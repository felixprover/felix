\section{Support and finite-support function spaces}

\begin{definition}\label{spike_support_on}
    $\supportOn{I}{f}{z}=\{i\in I\mid f(i)\neq z\}$.
\end{definition}

\begin{proposition}\label{spike_support_on_iff}
    $i\in\supportOn{I}{f}{z}$ iff $i\in I$ and $f(i)\neq z$.
\end{proposition}
\begin{proof}
    Follows by \auto{spike_support_on}.
\end{proof}

\begin{proposition}\label{spike_support_on_subseteq}
    $\supportOn{I}{f}{z}\subseteq I$.
\end{proposition}
\begin{proof}
    Follows by \auto{subseteq,spike_support_on_iff}.
\end{proof}

\begin{definition}\label{spike_dependent_support}
    $\dependentSupport{I}{f}{z}
        =\{i\in I\mid f(i)\neq z(i)\}$.
\end{definition}

\begin{proposition}\label{spike_dependent_support_iff}
    $i\in\dependentSupport{I}{f}{z}$ iff
        $i\in I$ and $f(i)\neq z(i)$.
\end{proposition}
\begin{proof}
    Follows by \auto{spike_dependent_support}.
\end{proof}

\begin{proposition}\label{spike_dependent_support_subseteq}
    $\dependentSupport{I}{f}{z}\subseteq I$.
\end{proposition}
\begin{proof}
    Follows by \auto{subseteq,spike_dependent_support_iff}.
\end{proof}

\begin{theorem}\label{spike_dependent_support_precomposition}
    Let $e\in\funs{J}{I}$.
    Let $f,z\in\dependentProduct{I}{A}$.
    Then
        $\dependentSupport{J}{f\circ e}{z\circ e}
            =\preimg{e}{\dependentSupport{I}{f}{z}}$.
\end{theorem}
\begin{proof}
    $e$ is a function and $\dom{e}=J$ and $\ran{e}\subseteq I$ by
        \auto{funs_is_function,funs,funs_ran}.
    $f,z\in\funs{I}{\unions{\ran{A}}}$ by
        \auto{dependent_product_iff}.
    Hence $f$ is a function and $z$ is a function and
        $\dom{f}=I=\dom{z}$ by
        \auto{funs_is_function,funs}.
    $f$ is composable with $e$ and $z$ is composable with $e$ by
        \auto.
    Show that for every $j$ we have
        $j\in\dependentSupport{J}{f\circ e}{z\circ e}$ iff
        $j\in\preimg{e}{\dependentSupport{I}{f}{z}}$.
    \begin{subproof}
        Fix $j$.
        \begin{byCase}
        \caseOf{$j\in J$.}
            Hence $j\in\dom{e}$ by \auto.
            $(f\circ e)(j)=f(e(j))$ and
                $(z\circ e)(j)=z(e(j))$ by
                \auto{circ_apply}.
            $e(j)\in I$ by \auto{funs_type_apply}.
            Follows by
                \auto{spike_dependent_support_iff,preimg_iff,
                    function_apply_iff}.
        \caseOf{$j\notin J$.}
            $j\notin\dom{e}$ by \auto.
            Follows by
                \auto{spike_dependent_support_iff,preimg_subseteq_dom,
                    subseteq}.
        \end{byCase}
    \end{subproof}
    Follows by \auto{setext}.
\end{proof}

\begin{definition}\label{spike_finitely_supported_functions}
    $\finitelySupportedFunctions{I}{A}{z}
        =\{f\in\funs{I}{A}\mid
            \text{$\supportOn{I}{f}{z}$ is finite}\}$.
\end{definition}

\begin{proposition}\label{spike_finitely_supported_functions_iff}
    $f\in\finitelySupportedFunctions{I}{A}{z}$ iff
        $f\in\funs{I}{A}$ and $\supportOn{I}{f}{z}$ is finite.
\end{proposition}
\begin{proof}
    Follows by \auto{spike_finitely_supported_functions}.
\end{proof}

\begin{definition}\label{spike_dependent_finitely_supported_functions}
    $\dependentFinitelySupportedFunctions{I}{A}{z}
        =\{f\in\dependentProduct{I}{A}\mid
            \text{$\dependentSupport{I}{f}{z}$ is finite}\}$.
\end{definition}

\begin{proposition}\label{spike_dependent_finitely_supported_functions_iff}
    $f\in\dependentFinitelySupportedFunctions{I}{A}{z}$ iff
        $f\in\dependentProduct{I}{A}$ and
        $\dependentSupport{I}{f}{z}$ is finite.
\end{proposition}
\begin{proof}
    Follows by
        \auto{spike_dependent_finitely_supported_functions}.
\end{proof}

\begin{theorem}\label{spike_constant_family_dependent_product}
    $\dependentProduct{I}{I\times\{A\}}=\funs{I}{A}$.
\end{theorem}
\begin{proof}
    Let $C=I\times\{A\}$.
    $C\in\funs{I}{\{A\}}$ by \auto{constant_graph_type}.
    Hence $C$ is a function and $\dom{C}=I$ by
        \auto{funs_is_function,funs}.
    Show that every element of $\dependentProduct{I}{C}$ is an element
        of $\funs{I}{A}$.
    \begin{subproof}
        Fix $f\in\dependentProduct{I}{C}$.
        Hence $f$ is a function and $\dom{f}=I$ by
            \auto{dependent_product_iff,funs_is_function,funs}.
        For every $i\in I$ we have $f(i)\in C(i)$ by
            \auto{dependent_product_coordinate}.
        Hence for every $i\in I$ we have $f(i)\in A$ by
            \auto{constant_graph_application}.
        Thus $f$ is a function to $A$ by \auto.
        Follows by \auto{funs_intro}.
    \end{subproof}
    Show that every element of $\funs{I}{A}$ is an element of
        $\dependentProduct{I}{C}$.
    \begin{subproof}
        Fix $f\in\funs{I}{A}$.
        Hence $f$ is a function and $\dom{f}=I$ by
            \auto{funs_is_function,funs}.
        Show that for every $i\in I$ we have $f(i)\in C(i)$.
        \begin{subproof}
            Fix $i\in I$.
            $f(i)\in A$ by \auto{funs_type_apply}.
            $C(i)=A$ by \auto{constant_graph_application}.
            Follows by \auto.
        \end{subproof}
        Show that $f\in\funs{I}{\unions{\ran{C}}}$.
        \begin{subproof}
            Show that for every $i\in\dom{f}$ we have
                $f(i)\in\unions{\ran{C}}$.
            \begin{subproof}
                Fix $i\in\dom{f}$.
                Hence $i\in I$ by \auto.
                $f(i)\in C(i)$ by \auto.
                Follows by \auto{indexed_family_member_union}.
            \end{subproof}
            Hence $f$ is a function to $\unions{\ran{C}}$ by \auto.
            Follows by \auto{funs_intro}.
        \end{subproof}
        Follows by \auto{dependent_product_iff}.
    \end{subproof}
    Follows by \auto{setext}.
\end{proof}

\begin{theorem}\label{spike_constant_family_support}
    Let $z\in A$ and $f\in\funs{I}{A}$.
    Then
        $\dependentSupport{I}{f}{I\times\{z\}}
            =\supportOn{I}{f}{z}$.
\end{theorem}
\begin{proof}
    For every $i\in I$ we have $(I\times\{z\})(i)=z$ by
        \auto{constant_graph_application}.
    Follows by
        \auto{setext,spike_dependent_support_iff,
            spike_support_on_iff}.
\end{proof}

\begin{theorem}\label{spike_constant_family_finite_support_space}
    Let $z\in A$.
    Then
        $\dependentFinitelySupportedFunctions
            {I}{I\times\{A\}}{I\times\{z\}}
        =\finitelySupportedFunctions{I}{A}{z}$.
\end{theorem}
\begin{proof}
    $\dependentProduct{I}{I\times\{A\}}=\funs{I}{A}$ by
        \auto{spike_constant_family_dependent_product}.
    Show that for every $f\in\funs{I}{A}$ we have
        $\dependentSupport{I}{f}{I\times\{z\}}
            =\supportOn{I}{f}{z}$.
    \begin{subproof}
        Fix $f\in\funs{I}{A}$.
        Follows by \auto{spike_constant_family_support}.
    \end{subproof}
    Follows by
        \auto{setext,
            spike_dependent_finitely_supported_functions_iff,
            spike_finitely_supported_functions_iff}.
\end{proof}

\begin{theorem}\label{spike_finite_preimage_under_finite_fibers}
    Suppose $F$ is finite and $F\subseteq I$.
    Suppose for every $i\in I$ we have $\preimg{e}{\{i\}}$ is finite.
    Then $\preimg{e}{F}$ is finite.
\end{theorem}
\begin{proof}
    Let $C=\{D\in\pow{F}\mid
        \text{$\preimg{e}{D}$ is finite}\}$.
    $\preimg{e}{\emptyset}=\emptyset$ by
        \auto{setext,preimg_iff,emptyset}.
    Hence $\preimg{e}{\emptyset}$ is finite by
        \auto{emptyset_is_finite}.
    $\emptyset\in\pow{F}$ by \auto{powerset_bottom}.
    Hence $\emptyset\in C$ by \auto.
    Show that for all $D,i$ such that $D\in C$ and $i\in F$ we have
        $\cons{i}{D}\in C$.
    \begin{subproof}
        Fix $D,i$.
        Assume $D\in C$ and $i\in F$.
        $D\subseteq F$ by \auto{powerset_elim}.
        Hence $\cons{i}{D}\subseteq F$ by
            \auto{cons_subseteq_intro}.
        Hence $\cons{i}{D}\in\pow{F}$ by
            \auto{powerset_intro}.
        $i\in I$ by \auto{subseteq}.
        Hence $\preimg{e}{\{i\}}$ is finite by \auto.
        $\preimg{e}{D}$ is finite.
        $\cons{i}{D}=\{i\}\union D$ by
            \auto{setext,cons_iff,union_iff,singleton_iff}.
        Hence
            $\preimg{e}{\cons{i}{D}}
                =\preimg{e}{\{i\}}\union\preimg{e}{D}$ by
            \auto{preimg_union}.
        Hence $\preimg{e}{\cons{i}{D}}$ is finite by
            \auto{union_of_finite_is_finite}.
        Follows.
    \end{subproof}
    $F\in C$ by \auto{finite_induction}.
    Follows.
\end{proof}

\begin{proposition}\label{spike_dependent_precomposition_type}
    Suppose $A$ is a function on $I$.
    Let $e\in\funs{J}{I}$ and $f\in\dependentProduct{I}{A}$.
    Then $f\circ e\in\dependentProduct{J}{A\circ e}$.
\end{proposition}
\begin{proof}
    $A$ is a function and $\dom{A}=I$ by \auto.
    $e$ is a function and $\dom{e}=J$ and $\ran{e}\subseteq I$ by
        \auto{funs_is_function,funs,funs_ran}.
    $A$ is composable with $e$ by \auto.
    Hence $A\circ e$ is a function by \auto{function_circ}.
    Show that $\dom{A\circ e}=J$.
    \begin{subproof}
        Show that for every $j$ we have
            $j\in\dom{A\circ e}$ iff $j\in J$.
        \begin{subproof}
            Fix $j$.
            Follows by
                \auto{dom_iff,circ_iff,funs_type_apply,
                    function_apply_elim}.
        \end{subproof}
        Follows by \auto{setext}.
    \end{subproof}
    $f\in\funs{I}{\unions{\ran{A}}}$ by
        \auto{dependent_product_iff}.
    Hence $f$ is a function and $\dom{f}=I$ by
        \auto{funs_is_function,funs}.
    $f$ is composable with $e$ by \auto.
    Hence $f\circ e$ is a function and $\dom{f\circ e}=J$ by
        \auto{funs_circ,funs_is_function,funs}.
    Show that for every $j\in J$ we have
        $(f\circ e)(j)\in(A\circ e)(j)$.
    \begin{subproof}
        Fix $j\in J$.
        Hence $j\in\dom{e}$ by \auto.
        $e(j)\in I$ by \auto{funs_type_apply}.
        $f(e(j))\in A(e(j))$ by
            \auto{dependent_product_coordinate}.
        $(f\circ e)(j)=f(e(j))$ by
            \auto{circ_apply,funs_is_function}.
        $(A\circ e)(j)=A(e(j))$ by
            \auto{circ_apply,funs_is_function}.
        Follows by \auto.
    \end{subproof}
    Show that $f\circ e\in\funs{J}{\unions{\ran{A\circ e}}}$.
    \begin{subproof}
        Show that for every $j\in\dom{f\circ e}$ we have
            $(f\circ e)(j)\in\unions{\ran{A\circ e}}$.
        \begin{subproof}
            Fix $j\in\dom{f\circ e}$.
            Hence $j\in J$ by \auto.
            $(f\circ e)(j)\in(A\circ e)(j)$ by \auto.
            Follows by \auto{indexed_family_member_union}.
        \end{subproof}
        Hence $f\circ e$ is a function to
            $\unions{\ran{A\circ e}}$ by \auto.
        Follows by \auto{funs_intro}.
    \end{subproof}
    Follows by \auto{dependent_product_iff}.
\end{proof}

\begin{theorem}\label{spike_finite_support_precomposition}
    Suppose $A$ is a function on $I$.
    Let $e\in\funs{J}{I}$.
    Let $z\in\dependentProduct{I}{A}$.
    Suppose $f\in\dependentFinitelySupportedFunctions{I}{A}{z}$.
    Suppose for every $i\in I$ we have
        $\preimg{e}{\{i\}}$ is finite.
    Then
        $f\circ e\in
            \dependentFinitelySupportedFunctions
                {J}{A\circ e}{z\circ e}$.
\end{theorem}
\begin{proof}
    $f\in\dependentProduct{I}{A}$ and
        $\dependentSupport{I}{f}{z}$ is finite by
        \auto{spike_dependent_finitely_supported_functions_iff}.
    $f\circ e\in\dependentProduct{J}{A\circ e}$ and
        $z\circ e\in\dependentProduct{J}{A\circ e}$ by
        \auto{spike_dependent_precomposition_type}.
    $\dependentSupport{I}{f}{z}\subseteq I$ by
        \auto{spike_dependent_support_subseteq}.
    Hence $\preimg{e}{\dependentSupport{I}{f}{z}}$ is finite by
        \auto{spike_finite_preimage_under_finite_fibers}.
    $\dependentSupport{J}{f\circ e}{z\circ e}
        =\preimg{e}{\dependentSupport{I}{f}{z}}$ by
        \auto{spike_dependent_support_precomposition}.
    Hence $\dependentSupport{J}{f\circ e}{z\circ e}$ is finite by
        \auto.
    Follows by
        \auto{spike_dependent_finitely_supported_functions_iff}.
\end{proof}

\subsection{Point masses}

\begin{definition}\label{spike_dependent_point_mass}
    $\dependentPointMass{I}{A}{z}{i}{a}
        =\cons{(i,a)}{\restrl{z}{I\setminus\{i\}}}$.
\end{definition}

\begin{proposition}\label{spike_dependent_point_mass_type}
    Suppose $A$ is a function on $I$.
    Let $z\in\dependentProduct{I}{A}$.
    Let $i\in I$ and $a\in A(i)$.
    Then
        $\dependentPointMass{I}{A}{z}{i}{a}
            \in\dependentProduct{I}{A}$.
\end{proposition}
\begin{proof}
    Let $p=\dependentPointMass{I}{A}{z}{i}{a}$.
    Let $D=I\setminus\{i\}$.
    $A$ is a function and $\dom{A}=I$ by \auto.
    $z\in\funs{I}{\unions{\ran{A}}}$ by
        \auto{dependent_product_iff}.
    Hence $z$ is a function and $\dom{z}=I$ by
        \auto{funs_is_function,funs}.
    $D\subseteq I$ by \auto{setminus,subseteq}.
    Hence $\restrl{z}{D}\in
        \funs{D}{\unions{\ran{A}}}$ by \auto{funs_restrl}.
    $i\notin D$ by \auto{setminus,singleton_iff}.
    $a\in\unions{\ran{A}}$ by
        \auto{indexed_family_member_union}.
    Hence
        $p\in\funs{\cons{i}{D}}{\unions{\ran{A}}}$ by
        \auto{spike_dependent_point_mass,
            extend_function_at_new_point}.
    $\cons{i}{D}=I$ by \auto{cons_remove}.
    Hence $p\in\funs{I}{\unions{\ran{A}}}$ by \auto.
    Show that for every $j\in I$ we have $p(j)\in A(j)$.
    \begin{subproof}
        Fix $j\in I$.
        \begin{byCase}
        \caseOf{$j=i$.}
            $(i,a)\in p$ by
                \auto{spike_dependent_point_mass,cons_iff}.
            $p$ is a function by \auto{funs_is_function}.
            Hence $p(i)=a$ by \auto{function_apply_intro}.
            Follows by \auto.
        \caseOf{$j\neq i$.}
            Hence $j\in D$ by \auto{setminus,singleton_iff}.
            $(j,z(j))\in z$ by \auto{function_apply_elim}.
            Hence $(j,z(j))\in\restrl{z}{D}$ by
                \auto{restrl_iff}.
            Hence $(j,z(j))\in p$ by
                \auto{spike_dependent_point_mass,cons_iff}.
            $p$ is a function by \auto{funs_is_function}.
            Hence $p(j)=z(j)$ by \auto{function_apply_intro}.
            $z(j)\in A(j)$ by
                \auto{dependent_product_coordinate}.
            Follows by \auto.
        \end{byCase}
    \end{subproof}
    Follows by \auto{dependent_product_iff}.
\end{proof}

\begin{proposition}\label{spike_dependent_point_mass_selected}
    Suppose $A$ is a function on $I$.
    Let $z\in\dependentProduct{I}{A}$.
    Let $i\in I$ and $a\in A(i)$.
    Then $\apply{\dependentPointMass{I}{A}{z}{i}{a}}{i}=a$.
\end{proposition}
\begin{proof}
    Let $p=\dependentPointMass{I}{A}{z}{i}{a}$.
    $p$ is a function by
        \auto{spike_dependent_point_mass_type,
            dependent_product_iff,funs_is_function}.
    $(i,a)\in p$ by
        \auto{spike_dependent_point_mass,cons_iff}.
    Follows by \auto{function_apply_intro}.
\end{proof}

\begin{proposition}\label{spike_dependent_point_mass_other}
    Suppose $A$ is a function on $I$.
    Let $z\in\dependentProduct{I}{A}$.
    Let $i,j\in I$ and $a\in A(i)$.
    Suppose $j\neq i$.
    Then $\apply{\dependentPointMass{I}{A}{z}{i}{a}}{j}=z(j)$.
\end{proposition}
\begin{proof}
    Let $p=\dependentPointMass{I}{A}{z}{i}{a}$.
    Let $D=I\setminus\{i\}$.
    $z\in\funs{I}{\unions{\ran{A}}}$ by
        \auto{dependent_product_iff}.
    Hence $z$ is a function and $\dom{z}=I$ by
        \auto{funs_is_function,funs}.
    $p$ is a function by
        \auto{spike_dependent_point_mass_type,
            dependent_product_iff,funs_is_function}.
    $j\in D$ by \auto{setminus,singleton_iff}.
    $(j,z(j))\in z$ by \auto{function_apply_elim}.
    Hence $(j,z(j))\in\restrl{z}{D}$ by \auto{restrl_iff}.
    Hence $(j,z(j))\in p$ by
        \auto{spike_dependent_point_mass,cons_iff}.
    Follows by \auto{function_apply_intro}.
\end{proof}

\begin{proposition}\label{spike_dependent_point_mass_support}
    Suppose $A$ is a function on $I$.
    Let $z\in\dependentProduct{I}{A}$.
    Let $i\in I$ and $a\in A(i)$.
    Then
        $(a=z(i)\implies
            \dependentSupport{I}{
                \dependentPointMass{I}{A}{z}{i}{a}}{z}=\emptyset)$ and
        $(a\neq z(i)\implies
            \dependentSupport{I}{
                \dependentPointMass{I}{A}{z}{i}{a}}{z}=\{i\})$.
\end{proposition}
\begin{proof}
    Let $p=\dependentPointMass{I}{A}{z}{i}{a}$.
    \begin{byCase}
    \caseOf{$a=z(i)$.}
        Show that $\dependentSupport{I}{p}{z}=\emptyset$.
        \begin{subproof}
            Show that $\dependentSupport{I}{p}{z}\subseteq\emptyset$.
            \begin{subproof}
                It suffices to show that every element of
                    $\dependentSupport{I}{p}{z}$ is an element of
                    $\emptyset$ by \auto{subseteq}.
                Fix $j\in\dependentSupport{I}{p}{z}$.
                Hence $j\in I$ and $p(j)\neq z(j)$ by
                    \auto{spike_dependent_support_iff}.
                $j=i$ or $j\neq i$ by \auto.
                $p(i)=a$ by
                    \auto{spike_dependent_point_mass_selected}.
                If $j\neq i$, then $p(j)=z(j)$ by
                    \auto{spike_dependent_point_mass_other}.
                Contradiction by \auto.
            \end{subproof}
            Follows by \auto{subseteq_emptyset_iff}.
        \end{subproof}
        Follows by \auto.
    \caseOf{$a\neq z(i)$.}
        Show that $\dependentSupport{I}{p}{z}=\{i\}$.
        \begin{subproof}
            $p(i)=a$ by
                \auto{spike_dependent_point_mass_selected}.
            Hence $i\in\dependentSupport{I}{p}{z}$ by
                \auto{spike_dependent_support_iff}.
            Show that for every $j\in\dependentSupport{I}{p}{z}$
                we have $j\in\{i\}$.
            \begin{subproof}
                Fix $j\in\dependentSupport{I}{p}{z}$.
                Hence $j\in I$ and $p(j)\neq z(j)$ by
                    \auto{spike_dependent_support_iff}.
                Suppose not.
                Hence $j\neq i$ by \auto{singleton_iff}.
                Then $p(j)=z(j)$ by
                    \auto{spike_dependent_point_mass_other}.
                Contradiction.
            \end{subproof}
            Follows by \auto{setext,singleton_iff}.
        \end{subproof}
        Follows by \auto.
    \end{byCase}
\end{proof}

\begin{proposition}\label{spike_dependent_point_mass_finitely_supported}
    Suppose $A$ is a function on $I$.
    Let $z\in\dependentProduct{I}{A}$.
    Let $i\in I$ and $a\in A(i)$.
    Then
        $\dependentPointMass{I}{A}{z}{i}{a}
            \in\dependentFinitelySupportedFunctions{I}{A}{z}$.
\end{proposition}
\begin{proof}
    $\dependentPointMass{I}{A}{z}{i}{a}
        \in\dependentProduct{I}{A}$ by
        \auto{spike_dependent_point_mass_type}.
    $\dependentSupport{I}{
        \dependentPointMass{I}{A}{z}{i}{a}}{z}$ is finite by
        \auto{spike_dependent_point_mass_support,
            emptyset_is_finite,singleton_is_finite}.
    Follows by
        \auto{spike_dependent_finitely_supported_functions_iff}.
\end{proof}

\begin{definition}\label{spike_point_mass}

            {I}{I\times\{A\}}{I\times\{z\}}{i}{a}$.

\begin{proposition}\label{spike_point_mass_finitely_supported}

            \in\dependentFinitelySupportedFunctions{I}{C}{q}$ by
        \auto{spike_dependent_point_mass_finitely_supported}.
    Follows by
        \auto{spike_point_mass,
            spike_constant_family_finite_support_space}.
\end{proof}

    $q\in\dependentProduct{I}{C}$ by

    $C(i)=A$ and $q(j)=z$ by

\begin{definition}\label{spike_dependent_point_mass_map}
    $\dependentPointMassMap{I}{A}{z}{i}
        =\{(a,\dependentPointMass{I}{A}{z}{i}{a})\mid a\in A(i)\}$.
\end{definition}

\begin{proposition}\label{spike_dependent_point_mass_map_type}
    Suppose $A$ is a function on $I$.
    Let $z\in\dependentProduct{I}{A}$ and $i\in I$.
    Then
        $\dependentPointMassMap{I}{A}{z}{i}
            \in\funs{A(i)}{
                \dependentFinitelySupportedFunctions{I}{A}{z}}$.
\end{proposition}
\begin{proof}
    Let $m=\dependentPointMassMap{I}{A}{z}{i}$.
    $m$ is a relation by
        \auto{spike_dependent_point_mass_map,relation}.
    $m$ is right-unique by
        \auto{spike_dependent_point_mass_map,rightunique,pair_eq_iff}.
    Hence $m$ is a function.
    $\dom{m}=A(i)$ by
        \auto{spike_dependent_point_mass_map,setext,dom_iff,pair_eq_iff}.
    Show that for every $a\in\dom{m}$ we have
        $m(a)\in\dependentFinitelySupportedFunctions{I}{A}{z}$.
    \begin{subproof}
        Fix $a\in\dom{m}$.
        Hence $a\in A(i)$ by \auto.
        $(a,\dependentPointMass{I}{A}{z}{i}{a})\in m$ by
            \auto{spike_dependent_point_mass_map,pair_eq_iff}.
        Hence
            $m(a)=\dependentPointMass{I}{A}{z}{i}{a}$ by
            \auto{function_apply_intro}.
        Follows by
            \auto{spike_dependent_point_mass_finitely_supported}.
    \end{subproof}
    Hence $m$ is a function to
        $\dependentFinitelySupportedFunctions{I}{A}{z}$ by \auto.
    Follows by \auto{funs_intro}.
\end{proof}

\begin{proposition}\label{spike_dependent_point_mass_map_apply}
