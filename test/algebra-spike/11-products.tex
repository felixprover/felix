\section{Dependent products and direct sums}

\subsection{Reindexing dependent functions}

Precomposition is the contravariant reindexing operation.  Its definition is
independent of finiteness; finiteness enters only when the graph is restricted
to finitely supported functions.

\begin{definition}\label{spike_precomposition_op}
    $\spikePrecompositionOp:\Set\to\Set\to\Set
        =\fun{e,x:\Set}{x\circ e}$.
\end{definition}

\begin{definition}\label{spike_dependent_reindex}
    $\dependentReindex{I}{J}{A}{e}
        =\Apply{\graphOnOp}{
            \dependentProduct{I}{A},
            \dependentProduct{J}{A\circ e},
            \Apply{\spikePrecompositionOp}{e}}$.
\end{definition}

\begin{proposition}\label{spike_dependent_reindex_type}
    Suppose $A$ is a function on $I$.
    Let $e\in\funs{J}{I}$.
    Then
        $\dependentReindex{I}{J}{A}{e}\in
            \funs{\dependentProduct{I}{A}}{
                \dependentProduct{J}{A\circ e}}$.
\end{proposition}
\begin{proof}
    For every $x\in\dependentProduct{I}{A}$ we have
        $x\circ e\in\dependentProduct{J}{A\circ e}$ by
        \auto{spike_dependent_precomposition_type}.
    Follows by
        \auto{spike_dependent_reindex,spike_precomposition_op,
            logical_graph_on_function}.
\end{proof}

\subsection{Pointwise group operations}

\begin{definition}\label{spike_group_carrier_op}
    $\spikeGroupCarrierOp:\Set\to\Set\to\Set
        =\fun{G,i:\Set}{\carrier[G(i)]}$.
\end{definition}

\begin{definition}\label{spike_group_carrier_family}
    $\spikeGroupCarrierFamily{I}{G}
        =\Apply{\graphOnDomainOp}{
            I,\Apply{\spikeGroupCarrierOp}{G}}$.
\end{definition}

\begin{proposition}\label{spike_group_carrier_family_type}
    Suppose $G$ is a function on $I$.
    Then $\spikeGroupCarrierFamily{I}{G}$ is a function on $I$.
\end{proposition}
\begin{proof}
    $\spikeGroupCarrierFamily{I}{G}\in
        \funs{I}{\ran{\spikeGroupCarrierFamily{I}{G}}}$ by
        \auto{spike_group_carrier_family,
            logical_graph_on_domain_function}.
    Follows by \auto{funs_is_function,funs}.
\end{proof}

\begin{proposition}\label{spike_group_carrier_family_apply}
    Suppose $G$ is a function on $I$ and $i\in I$.
    Then
        $\apply{\spikeGroupCarrierFamily{I}{G}}{i}
            =\carrier[G(i)]$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{spike_group_carrier_family,spike_group_carrier_op,
            logical_graph_on_domain_apply}.
\end{proof}

\begin{definition}\label{spike_group_neutral_op}
    $\spikeGroupNeutralOp:\Set\to\Set\to\Set
        =\fun{G,i:\Set}{\neutral[G(i)]}$.
\end{definition}

\begin{definition}\label{spike_product_group_neutral}
    $\spikeProductGroupNeutral{I}{G}
        =\Apply{\graphOnDomainOp}{
            I,\Apply{\spikeGroupNeutralOp}{G}}$.
\end{definition}

\begin{proposition}\label{spike_product_group_neutral_coordinate}
    Let $i\in I$.
    Then
        $\apply{\spikeProductGroupNeutral{I}{G}}{i}
            =\neutral[G(i)]$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{spike_product_group_neutral,spike_group_neutral_op,
            logical_graph_on_domain_apply}.
\end{proof}

\begin{proposition}\label{spike_product_group_neutral_type}
    Suppose $G$ is a function on $I$.
    Suppose for every $j\in I$ we have $G(j)$ is a group.
    Then
        $\spikeProductGroupNeutral{I}{G}\in
            \dependentProduct{I}{\spikeGroupCarrierFamily{I}{G}}$.
\end{proposition}
\begin{proof}
    Let $C=\spikeGroupCarrierFamily{I}{G}$.
    Let $z=\spikeProductGroupNeutral{I}{G}$.
    $C$ is a function on $I$ by
        \auto{spike_group_carrier_family_type}.
    $z\in\funs{I}{\ran{z}}$ by
        \auto{spike_product_group_neutral,
            logical_graph_on_domain_function}.
    Hence $z$ is a function and $\dom{z}=I$ by
        \auto{funs_is_function,funs}.
    Show that for every $i\in I$ we have $z(i)\in C(i)$.
    \begin{subproof}
        Fix $i\in I$.
        $z(i)=\neutral[G(i)]$ by
            \auto{spike_product_group_neutral_coordinate}.
        $C(i)=\carrier[G(i)]$ by
            \auto{spike_group_carrier_family_apply}.
        $G(i)$ is a group by \auto.
        Hence $G(i)$ is a monoid by \auto{group_is_monoid}.
        Hence $G(i)$ is a unital magma by
            \auto{monoid_is_unital_magma}.
        Follows by \auto{unitalmagma_type}.
    \end{subproof}
    Show that $z\in\funs{I}{\unions{\ran{C}}}$.
    \begin{subproof}
        Show that for every $i\in\dom{z}$ we have
            $z(i)\in\unions{\ran{C}}$.
        \begin{subproof}
            Fix $i\in\dom{z}$.
            Hence $i\in I$ by \auto.
            $z(i)\in C(i)$ by \auto.
            Follows by \auto{indexed_family_member_union}.
        \end{subproof}
        Hence $z$ is a function to $\unions{\ran{C}}$ by
            \auto.
        Follows by \auto{funs_intro}.
    \end{subproof}
    Follows by \auto{dependent_product_iff}.
\end{proof}

\begin{definition}\label{spike_group_product_coordinate_op}
    $\spikeGroupProductCoordinateOp:
        \Set\to\Set\to\Set\to\Set\to\Set
        =\fun{G,x,y,i:\Set}{\mul[G(i)](x(i),y(i))}$.
\end{definition}

\begin{definition}\label{spike_pointwise_group_product_op}
    $\spikePointwiseGroupProductOp:
        \Set\to\Set\to\Set\to\Set\to\Set
        =\fun{I,G,x,y:\Set}{
            \Apply{\graphOnDomainOp}{
                I,\Apply{\spikeGroupProductCoordinateOp}{G,x,y}}}$.
\end{definition}

\begin{proposition}\label{spike_pointwise_group_product_coordinate}
    Let $i\in I$.
    Then
        $\apply{\Apply{\spikePointwiseGroupProductOp}{I,G,x,y}}{i}
            =\mul[G(i)](x(i),y(i))$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{spike_pointwise_group_product_op,
            spike_group_product_coordinate_op,
            logical_graph_on_domain_apply}.
\end{proof}

\begin{proposition}\label{spike_pointwise_group_product_type}
    Suppose $G$ is a function on $I$.
    Suppose for every $j\in I$ we have $G(j)$ is a group.
    Let
        $x,y\in\dependentProduct
            {I}{\spikeGroupCarrierFamily{I}{G}}$.

            \dependentProduct{I}{\spikeGroupCarrierFamily{I}{G}}$.

\begin{proof}

        \auto{funs_is_function,funs}.
    Show that for every $i\in I$ we have $u(i)\in C(i)$.

        Fix $i\in I$.
        $x(i),y(i)\in C(i)$ by
            \auto{dependent_product_coordinate}.
        $C(i)=\carrier[G(i)]$ by
            \auto{spike_group_carrier_family_apply}.
        $G(i)$ is a group by \auto.
        Hence $G(i)$ is a monoid by \auto{group_is_monoid}.
        Hence $G(i)$ is a magma by \auto{monoid_is_magma}.
        $u(i)=\mul[G(i)](x(i),y(i))$ by
            \auto{spike_pointwise_group_product_coordinate}.
        Follows by \auto{magma_welldef}.
    \end{subproof}
    Show that $u\in\funs{I}{\unions{\ran{C}}}$.
    \begin{subproof}
        Show that for every $i\in\dom{u}$ we have
            $u(i)\in\unions{\ran{C}}$.
        \begin{subproof}
            Fix $i\in\dom{u}$.
            Hence $i\in I$ by \auto.
            $u(i)\in C(i)$ by \auto.
            Follows by \auto{indexed_family_member_union}.
        \end{subproof}
        Hence $u$ is a function to $\unions{\ran{C}}$ by
            \auto.
        Follows by \auto{funs_intro}.
    \end{subproof}
    Follows by \auto{dependent_product_iff}.
\end{proof}

\begin{proposition}\label{spike_dependent_product_extensionality}
    Suppose $A$ is a function on $I$.

\end{proposition}

\end{proof}

\subsection{The indexed product group}

\begin{definition}\label{spike_product_group}
    $\ProductGroup{I}{G}=
        \structure[group]{
            \carrier=
                \dependentProduct
                    {I}{\spikeGroupCarrierFamily{I}{G}},
            \mul=
                \Apply{\binaryGraphOnOp}{
                    \dependentProduct
                        {I}{\spikeGroupCarrierFamily{I}{G}},
                    \dependentProduct
                        {I}{\spikeGroupCarrierFamily{I}{G}},
                    \Apply{\spikePointwiseGroupProductOp}{I,G}},
            \neutral=\spikeProductGroupNeutral{I}{G},
            \groupInv=
                \Apply{\graphOnOp}{
                    \dependentProduct
                        {I}{\spikeGroupCarrierFamily{I}{G}},
                    \dependentProduct
                        {I}{\spikeGroupCarrierFamily{I}{G}},
                    \Apply{\spikePointwiseGroupInverseOp}{I,G}}
        }$.
\end{definition}

\begin{proposition}\label{spike_product_group_fields}
    Let $C=\spikeGroupCarrierFamily{I}{G}$.
    Let $D=\dependentProduct{I}{C}$.
    $\carrier[\ProductGroup{I}{G}]=D$ and
    $\mul[\ProductGroup{I}{G}]
        =\Apply{\binaryGraphOnOp}{
            D,D,\Apply{\spikePointwiseGroupProductOp}{I,G}}$ and
    $\neutral[\ProductGroup{I}{G}]
        =\spikeProductGroupNeutral{I}{G}$ and
    $\groupInv[\ProductGroup{I}{G}]
        =\Apply{\graphOnOp}{
            D,D,\Apply{\spikePointwiseGroupInverseOp}{I,G}}$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{spike_product_group,
            groupaggregateprojection_carrier,groupaggregateprojection_mul,
            groupaggregateprojection_neutral,groupaggregateprojection_groupInv}.
\end{proof}

\begin{proposition}\label{spike_product_group_mul_coordinate}
    Suppose $G$ is a function on $I$.
    Suppose for every $j\in I$ we have $G(j)$ is a group.
    Let $P=\ProductGroup{I}{G}$.
    Let $x,y\in\carrier[P]$ and $i\in I$.
    Then
        $\apply{\mul[P](x,y)}{i}
            =\mul[G(i)](x(i),y(i))$.
\end{proposition}
\begin{proof}
    Let $C=\spikeGroupCarrierFamily{I}{G}$.
    Let $D=\dependentProduct{I}{C}$.
    $\carrier[P]=D$ and
        $\mul[P]=\Apply{\binaryGraphOnOp}{
            D,D,\Apply{\spikePointwiseGroupProductOp}{I,G}}$ by
        \auto{spike_product_group_fields}.
    Hence $x,y\in D$ by \auto.
    $\mul[P](x,y)
        =\Apply{\spikePointwiseGroupProductOp}{I,G,x,y}$ by
        \auto{logical_binary_graph_on_apply}.
    Follows by
        \auto{spike_pointwise_group_product_coordinate}.
\end{proof}

\begin{proposition}\label{spike_product_group_inverse_coordinate}
    Suppose $G$ is a function on $I$.
    Suppose for every $j\in I$ we have $G(j)$ is a group.
    Let $P=\ProductGroup{I}{G}$.
    Let $x\in\carrier[P]$ and $i\in I$.
    Then
        $\apply{\groupInv[P](x)}{i}
            =\groupInv[G(i)](x(i))$.
\end{proposition}
\begin{proof}
    Let $C=\spikeGroupCarrierFamily{I}{G}$.
    Let $D=\dependentProduct{I}{C}$.
    $\carrier[P]=D$ and
        $\groupInv[P]=\Apply{\graphOnOp}{
            D,D,\Apply{\spikePointwiseGroupInverseOp}{I,G}}$ by
        \auto{spike_product_group_fields}.
    Hence $x\in D$ by \auto.
    $\groupInv[P](x)
        =\Apply{\spikePointwiseGroupInverseOp}{I,G,x}$ by
        \auto{logical_graph_on_apply}.
    Follows by
        \auto{spike_pointwise_group_inverse_coordinate}.
\end{proof}

\begin{proposition}\label{spike_product_group_neutral_coordinate_exact}
    Let $P=\ProductGroup{I}{G}$.
    Let $i\in I$.
    Then $\neutral[P](i)=\neutral[G(i)]$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{spike_product_group_fields,
            spike_product_group_neutral_coordinate}.
\end{proof}

\begin{theorem}\label{spike_product_group_is_exact}
    Suppose $G$ is a function on $I$.
    Suppose for every $i\in I$ we have $G(i)$ is a group.
    Then $\ProductGroup{I}{G}$ is an exact group.
\end{theorem}
\begin{proof}
    Let $C=\spikeGroupCarrierFamily{I}{G}$.
    Let $D=\dependentProduct{I}{C}$.
    Let $P=\ProductGroup{I}{G}$.
    $C$ is a function on $I$ by
        \auto{spike_group_carrier_family_type}.
    $\carrier[P]=D$ by \auto{spike_product_group_fields}.
    Show that $\mul[P]\in\funs{D\times D}{D}$.
    \begin{subproof}
        For all $x,y\in D$ we have
            $\Apply{\spikePointwiseGroupProductOp}{I,G,x,y}\in D$ by
            \auto{spike_pointwise_group_product_type}.
        Follows by
            \auto{spike_product_group_fields,
                logical_binary_graph_on_function}.
    \end{subproof}
    $\neutral[P]\in D$ by
        \auto{spike_product_group_fields,
            spike_product_group_neutral_type}.
    Show that $\groupInv[P]\in\funs{D}{D}$.
    \begin{subproof}
        For every $x\in D$ we have
            $\Apply{\spikePointwiseGroupInverseOp}{I,G,x}\in D$ by
            \auto{spike_pointwise_group_inverse_type}.
        Follows by
            \auto{spike_product_group_fields,
                logical_graph_on_function}.
    \end{subproof}
    Hence for all $x,y\in D$ we have $\mul[P](x,y)\in D$ by
        \auto{funs_type_apply,times_tuple_intro}.
    Hence for every $x\in D$ we have $\groupInv[P](x)\in D$ by
        \auto{funs_type_apply}.

    Show that for all $x,y,z\in D$ we have
        $\mul[P](x,\mul[P](y,z))=\mul[P](\mul[P](x,y),z)$.
    \begin{subproof}
        Fix $x,y,z\in D$.
        $\mul[P](y,z),\mul[P](x,y)\in D$ by \auto.
        Show that for every $i\in I$ we have
            $\apply{\mul[P](x,\mul[P](y,z))}{i}
                =\apply{\mul[P](\mul[P](x,y),z)}{i}$.
        \begin{subproof}
            Fix $i\in I$.
            $x(i),y(i),z(i)\in C(i)$ by
                \auto{dependent_product_coordinate}.
            $C(i)=\carrier[G(i)]$ by
                \auto{spike_group_carrier_family_apply}.
            Follows by
                \auto{spike_product_group_mul_coordinate,
                    group_is_monoid,monoid_assoc}.
        \end{subproof}
        Follows by \auto{spike_dependent_product_extensionality}.
    \end{subproof}

    Show that for every $x\in D$ we have
        $\mul[P](x,\neutral[P])=x$.
    \begin{subproof}
        Fix $x\in D$.
        Show that for every $i\in I$ we have
            $\apply{\mul[P](x,\neutral[P])}{i}=x(i)$.
        \begin{subproof}
            Fix $i\in I$.
            $x(i)\in C(i)$ by \auto{dependent_product_coordinate}.
            $C(i)=\carrier[G(i)]$ by
                \auto{spike_group_carrier_family_apply}.
            Follows by
                \auto{spike_product_group_mul_coordinate,
                    spike_product_group_neutral_coordinate_exact,
                    group_is_monoid,monoid_is_unital_magma,
                    unitalmagma_right}.
        \end{subproof}
        Follows by \auto{spike_dependent_product_extensionality}.
    \end{subproof}

    Show that for every $x\in D$ we have
        $\mul[P](\neutral[P],x)=x$.
    \begin{subproof}
        Fix $x\in D$.
        Show that for every $i\in I$ we have
            $\apply{\mul[P](\neutral[P],x)}{i}=x(i)$.
        \begin{subproof}
            Fix $i\in I$.
            $x(i)\in C(i)$ by \auto{dependent_product_coordinate}.
            $C(i)=\carrier[G(i)]$ by
                \auto{spike_group_carrier_family_apply}.
            Follows by
                \auto{spike_product_group_mul_coordinate,
                    spike_product_group_neutral_coordinate_exact,
                    group_is_monoid,monoid_is_unital_magma,
                    unitalmagma_left}.
        \end{subproof}
        Follows by \auto{spike_dependent_product_extensionality}.
    \end{subproof}

    Show that for every $x\in D$ we have
        $\mul[P](x,\groupInv[P](x))=\neutral[P]$.
    \begin{subproof}
        Fix $x\in D$.
        Show that for every $i\in I$ we have
            $\apply{\mul[P](x,\groupInv[P](x))}{i}
                =\neutral[P](i)$.
        \begin{subproof}
            Fix $i\in I$.
            $x(i)\in C(i)$ by \auto{dependent_product_coordinate}.
            $C(i)=\carrier[G(i)]$ by
                \auto{spike_group_carrier_family_apply}.
            Follows by
                \auto{spike_product_group_mul_coordinate,
                    spike_product_group_inverse_coordinate,
                    spike_product_group_neutral_coordinate_exact,
                    group_inverse_right}.
        \end{subproof}
        Follows by \auto{spike_dependent_product_extensionality}.
    \end{subproof}

    Show that for every $x\in D$ we have
        $\mul[P](\groupInv[P](x),x)=\neutral[P]$.
    \begin{subproof}
        Fix $x\in D$.
        Show that for every $i\in I$ we have
            $\apply{\mul[P](\groupInv[P](x),x)}{i}
                =\neutral[P](i)$.
        \begin{subproof}
            Fix $i\in I$.
            $x(i)\in C(i)$ by \auto{dependent_product_coordinate}.
            $C(i)=\carrier[G(i)]$ by
                \auto{spike_group_carrier_family_apply}.
            Follows by
                \auto{spike_product_group_mul_coordinate,
                    spike_product_group_inverse_coordinate,
                    spike_product_group_neutral_coordinate_exact,
                    group_inverse_left}.
        \end{subproof}
        Follows by \auto{spike_dependent_product_extensionality}.
    \end{subproof}

    Hence $P$ is a magma by
        \auto{spike_product_group,groupaggregateexact,magma}.
    Hence $P$ is a unital magma by
        \auto{spike_product_group,groupaggregateexact,unitalmagma}.
    Hence $P$ is a monoid by
        \auto{spike_product_group,groupaggregateexact,monoid}.
    Hence $P$ is a group by
        \auto{spike_product_group,groupaggregateexact,group}.
    Follows by \auto{spike_product_group,groupaggregateexact}.
\end{proof}

\begin{proposition}\label{spike_product_group_is_commutative}
    Suppose $G$ is a function on $I$.
    Suppose for every $i\in I$ we have $G(i)$ is a commutative group.
    Then $\ProductGroup{I}{G}$ is a commutative group.
\end{proposition}
\begin{proof}
    Let $C=\spikeGroupCarrierFamily{I}{G}$.
    Let $D=\dependentProduct{I}{C}$.
    Let $P=\ProductGroup{I}{G}$.
    For every $i\in I$ we have $G(i)$ is a group by
        \auto{commutative_group_is_group}.
    $C$ is a function on $I$ by
        \auto{spike_group_carrier_family_type}.
    Hence $P$ is an exact group by
        \auto{spike_product_group_is_exact}.
    Hence $P$ is a group by \auto.
    Hence $P$ is a monoid by \auto{group_is_monoid}.
    Hence $P$ is a magma by \auto{monoid_is_magma}.
    $\carrier[P]=D$ by \auto{spike_product_group_fields}.
    Show that for all $x,y\in\carrier[P]$ we have
        $\mul[P](x,y)=\mul[P](y,x)$.
    \begin{subproof}
        Fix $x,y\in\carrier[P]$.
        Hence $x,y\in D$ by \auto{spike_product_group_fields}.
        $\mul[P](x,y),\mul[P](y,x)\in\carrier[P]$ by
            \auto{magma_welldef}.

\begin{proposition}\label{spike_product_group_mul_support}
    Suppose $G$ is a function on $I$.
    Suppose for every $i\in I$ we have $G(i)$ is a group.
    Let $P=\ProductGroup{I}{G}$.
    Let $z=\neutral[P]$.
    Let $x,y\in\carrier[P]$.
    Then
        $\dependentSupport{I}{\mul[P](x,y)}{z}
            \subseteq
        \dependentSupport{I}{x}{z}\union
            \dependentSupport{I}{y}{z}$.
\end{proposition}
\begin{proof}
    Let $C=\spikeGroupCarrierFamily{I}{G}$.
    Let $D=\dependentProduct{I}{C}$.
    $\carrier[P]=D$ by \auto{spike_product_group_fields}.
    Hence $x,y\in D$ by \auto.
    Show that every element of
        $\dependentSupport{I}{\mul[P](x,y)}{z}$ is an element of
        $\dependentSupport{I}{x}{z}\union
            \dependentSupport{I}{y}{z}$.
    \begin{subproof}
        Fix $i\in\dependentSupport{I}{\mul[P](x,y)}{z}$.
        Hence $i\in I$ and
            $\apply{\mul[P](x,y)}{i}\neq z(i)$ by
            \auto{spike_dependent_support_iff}.
        Show that $i\in\dependentSupport{I}{x}{z}$ or
            $i\in\dependentSupport{I}{y}{z}$.
        \begin{subproof}
            Suppose not.
            Hence $i\notin\dependentSupport{I}{x}{z}$ and
                $i\notin\dependentSupport{I}{y}{z}$ by \auto.
            Hence $x(i)=z(i)$ and $y(i)=z(i)$ by
                \auto{spike_dependent_support_iff}.
            $z(i)=\neutral[G(i)]$ by
                \auto{spike_product_group_neutral_coordinate_exact}.
            $G(i)$ is a group by \auto.
            Hence $G(i)$ is a monoid by \auto{group_is_monoid}.
            Hence $G(i)$ is a unital magma by
                \auto{monoid_is_unital_magma}.
            $\apply{\mul[P](x,y)}{i}
                =\mul[G(i)](x(i),y(i))$ by
                \auto{spike_product_group_mul_coordinate}.
            Hence $\apply{\mul[P](x,y)}{i}
                =\mul[G(i)](\neutral[G(i)],\neutral[G(i)])$ by
                \auto.
            $\neutral[G(i)]\in\carrier[G(i)]$ by
                \auto{unitalmagma_type}.
            $\mul[G(i)](\neutral[G(i)],\neutral[G(i)])
                =\neutral[G(i)]$ by \auto{unitalmagma_left}.
            Hence $\apply{\mul[P](x,y)}{i}=z(i)$ by \auto.
            Contradiction.
        \end{subproof}
        Follows by \auto{union_iff}.
    \end{subproof}
    Follows by \auto{subseteq}.
\end{proof}

\begin{theorem}\label{spike_direct_sum_group_subgroup}
    Suppose $G$ is a function on $I$.
    Suppose for every $i\in I$ we have $G(i)$ is a commutative group.
    Let $P=\ProductGroup{I}{G}$.
    Let $C=\spikeGroupCarrierFamily{I}{G}$.
    Let $z=\spikeProductGroupNeutral{I}{G}$.
    Let $D=\dependentFinitelySupportedFunctions{I}{C}{z}$.
    Then $\subgroup{D}{P}$.
\end{theorem}
\begin{proof}
    For every $i\in I$ we have $G(i)$ is a group by
        \auto{commutative_group_is_group}.
    $P$ is an exact group by \auto{spike_product_group_is_exact}.
    Hence $P$ is a group by \auto.
    $\carrier[P]=\dependentProduct{I}{C}$ by
        \auto{spike_product_group_fields}.
    Show that every element of $D$ is an element of $\carrier[P]$.
    \begin{subproof}
        Fix $x\in D$.
        $x\in\dependentProduct{I}{C}$ by
            \auto{spike_dependent_finitely_supported_functions_iff}.
        Follows by \auto.
    \end{subproof}
    Hence $D\subseteq\carrier[P]$ by \auto{subseteq}.
    $z=\neutral[P]$ by \auto{spike_product_group_fields}.
    $z\in\dependentProduct{I}{C}$ by
        \auto{spike_product_group_neutral_type}.
    $\dependentSupport{I}{z}{z}=\emptyset$ by
        \auto{setext,spike_dependent_support_iff,emptyset}.
    Hence $\dependentSupport{I}{z}{z}$ is finite by
        \auto{emptyset_is_finite}.
    Hence $\neutral[P]\in D$ by
        \auto{spike_dependent_finitely_supported_functions_iff}.

    Show that for all $x,y\in D$ we have $\mul[P](x,y)\in D$.
    \begin{subproof}
        Fix $x,y\in D$.
        Hence $x,y\in\carrier[P]$ by \auto{subseteq}.
        $P$ is a monoid by \auto{group_is_monoid}.
        Hence $P$ is a magma by \auto{monoid_is_magma}.
        Hence $\mul[P](x,y)\in\carrier[P]$ by
            \auto{magma_welldef}.
        $\dependentSupport{I}{x}{z}$ and
            $\dependentSupport{I}{y}{z}$ are finite by
            \auto{spike_dependent_finitely_supported_functions_iff}.
        Hence
            $\dependentSupport{I}{x}{z}\union
                \dependentSupport{I}{y}{z}$ is finite by
            \auto{union_of_finite_is_finite}.
        $\dependentSupport{I}{\mul[P](x,y)}{z}
            \subseteq
            \dependentSupport{I}{x}{z}\union
                \dependentSupport{I}{y}{z}$ by
            \auto{spike_product_group_mul_support}.
        Hence $\dependentSupport{I}{\mul[P](x,y)}{z}$ is finite by
            \auto{subseteq_finite_is_finite}.
        $\mul[P](x,y)\in\dependentProduct{I}{C}$ by \auto.
        Follows by
            \auto{spike_dependent_finitely_supported_functions_iff}.
    \end{subproof}

    Show that for every $x\in D$ we have $\groupInv[P](x)\in D$.
    \begin{subproof}
        Fix $x\in D$.
        Hence $x\in\carrier[P]$ by \auto{subseteq}.
        $\groupInv[P](x)\in\carrier[P]$ by
            \auto{group_inverse_type}.
        $\dependentSupport{I}{x}{z}$ is finite by
            \auto{spike_dependent_finitely_supported_functions_iff}.
        $\dependentSupport{I}{\groupInv[P](x)}{z}
            \subseteq\dependentSupport{I}{x}{z}$ by
            \auto{spike_product_group_inverse_support}.
        Hence $\dependentSupport{I}{\groupInv[P](x)}{z}$ is finite by
            \auto{subseteq_finite_is_finite}.
        $\groupInv[P](x)\in\dependentProduct{I}{C}$ by \auto.
        Follows by
            \auto{spike_dependent_finitely_supported_functions_iff}.
    \end{subproof}
    Follows by \auto{subgroup}.
\end{proof}

    Hence $r$ is a function and $\dom{r}=D$ by
        \auto{funs_is_function,funs}.
    Show that $r$ is injective.
    \begin{subproof}
        Show that for all $x,y\in\dom{r}$ such that $r(x)=r(y)$
            we have $x=y$.

            Fix $x,y\in\dom{r}$.
            Assume $r(x)=r(y)$.
            Hence $x,y\in D$ by \auto.

                \auto{spike_dependent_direct_sum_reindex_apply}.
            Hence $x\circ e=y\circ e$ by \auto.
            $(x\circ e)\circ\converse{e}=x$ and
                $(y\circ e)\circ\converse{e}=y$ by
                \auto{spike_reindexed_section_round_trip}.
            Follows by \auto.
        \end{subproof}
        Follows by \auto{injective_function}.
    \end{subproof}
    Show that $r\in\surj{D}{E}$.
    \begin{subproof}
        Show that for every $y\in E$ there exists $x\in D$
            such that $r(x)=y$.
        \begin{subproof}

            Let $x=y\circ\converse{e}$.
            $A$ is a function and $\dom{A}=I$ by \auto.
            Hence $A\in\funs{I}{\ran{A}}$ by
                \auto{function_exact_type_to_range}.

                \auto{funs_circ}.
            Hence $A\circ e$ is a function on $J$ by
                \auto{funs_is_function,funs}.
            $\converse{e}\in\bijections{I}{J}$ by
                \auto{bijection_converse_is_bijection}.
            Hence $\converse{e}\in\funs{I}{J}$ by
                \auto{bijections_to_funs}.
            For every $j\in J$ we have
                $\preimg{\converse{e}}{\{j\}}$ is finite by
                \auto{spike_bijection_has_finite_fibers}.
            $y\in\dependentFinitelySupportedFunctions
                {J}{A\circ e}{z\circ e}$ by \auto.
            $z\circ e\in\dependentProduct{J}{A\circ e}$ by
                \auto{spike_dependent_precomposition_type}.
            Hence
                $x\in\dependentFinitelySupportedFunctions
                    {I}{(A\circ e)\circ\converse{e}}{
                        (z\circ e)\circ\converse{e}}$ by
                \auto{spike_finite_support_precomposition}.
            $(A\circ e)\circ\converse{e}=A$ by
                \auto{spike_pulled_back_family_round_trip}.
            $z\in\funs{I}{\unions{\ran{A}}}$ by
                \auto{dependent_product_iff}.
            $(z\circ e)\circ\converse{e}=z$ by

\begin{theorem}\label{spike_product_module_intro}

    Suppose
        $\scalarMul[P]\in
            \funs{\carrier[R]\times\carrier[P]}{\carrier[P]}$.
    Suppose for all $a,x,y$ such that $a\in\carrier[R]$ and
        $x,y\in\carrier[P]$ we have
        $\scalarMul[P](a,\mul[P](x,y))
            =\mul[P](\scalarMul[P](a,x),\scalarMul[P](a,y))$.
    Suppose for all $a,b,x$ such that $a,b\in\carrier[R]$ and
        $x\in\carrier[P]$ we have
        $\scalarMul[P](\ringAdd[R](a,b),x)
            =\mul[P](\scalarMul[P](a,x),\scalarMul[P](b,x))$.
    Suppose for all $a,b,x$ such that $a,b\in\carrier[R]$ and
        $x\in\carrier[P]$ we have
        $\scalarMul[P](a,\scalarMul[P](b,x))
            =\scalarMul[P](\ringMul[R](a,b),x)$.
    Suppose for every $x\in\carrier[P]$ we have
        $\scalarMul[P](\ringOne[R],x)=x$.
    Then $P$ is an exact left module over $R$.
\end{theorem}
\begin{proof}
    It is not the case that it is not the case that
        $P$ is a left module over $R$ by
        \auto{spike_product_module,left_moduleaggregateexact,
            left_module}.
    Hence $P$ is a left module over $R$ by \auto.
    Follows by
        \auto{spike_product_module,left_moduleaggregateexact}.
\end{proof}

\begin{theorem}\label{spike_product_module_is_exact}
    Let $R$ be a ring.
    Suppose $M$ is a function on $I$.
    Suppose for every $i\in I$ we have $M(i)$ is a left module over $R$.
    Then $\ProductModule{R}{I}{M}$ is an exact left module over $R$.
\end{theorem}
\begin{proof}
    Let $A=\spikeModuleAdditiveGroupFamily{I}{M}$.
    Let $Q=\ProductGroup{I}{A}$.
    Let $C=\spikeGroupCarrierFamily{I}{A}$.
    Let $P=\ProductModule{R}{I}{M}$.
    $A$ is a function on $I$ by
        \auto{spike_module_additive_group_family_type}.
    Show that for every $i\in I$ we have $A(i)$ is a commutative group.
    \begin{subproof}
        Fix $i\in I$.
        $A(i)=\ModuleAdditiveGroup{M(i)}$ by
            \auto{spike_module_additive_group_family_apply}.
        Follows by \auto{spike_module_additive_group_is_exact}.
    \end{subproof}
    Hence for every $i\in I$ we have $A(i)$ is a group by
        \auto{commutative_group_is_group}.
    Hence $Q$ is an exact group by
        \auto{spike_product_group_is_exact}.
    $Q$ is a commutative group by
        \auto{spike_product_group_is_commutative}.
    Hence $Q$ is a group by \auto{commutative_group_is_group}.
    Hence $Q$ is a monoid by \auto{group_is_monoid}.
    Hence $Q$ is a unital magma by \auto{monoid_is_unital_magma}.
    Hence $Q$ is a magma by \auto{monoid_is_magma}.
    $\carrier[P]=\carrier[Q]$ and $\mul[P]=\mul[Q]$ and
        $\neutral[P]=\neutral[Q]$ and
        $\groupInv[P]=\groupInv[Q]$ by
        \auto{spike_product_module_fields}.
    $\mul[P]\in
        \funs{\carrier[P]\times\carrier[P]}{\carrier[P]}$ by
        \auto{magma_mul_graph}.
    $\neutral[P]\in\carrier[P]$ by \auto{unitalmagma_type}.
    $\groupInv[P]\in\funs{\carrier[P]}{\carrier[P]}$ by
        \auto{group_inverse_graph}.
    For all $x,y,z\in\carrier[P]$ we have
        $\mul[P](x,\mul[P](y,z))=\mul[P](\mul[P](x,y),z)$ by
        \auto{monoid_assoc}.
    For every $x\in\carrier[P]$ we have
        $\mul[P](x,\neutral[P])=x$ and
        $\mul[P](\neutral[P],x)=x$ by
        \auto{unitalmagma_right,unitalmagma_left}.
    For every $x\in\carrier[P]$ we have
        $\mul[P](x,\groupInv[P](x))=\neutral[P]$ and
        $\mul[P](\groupInv[P](x),x)=\neutral[P]$ by
        \auto{group_inverse_right,group_inverse_left}.
    Hence $P$ is a magma by
        \auto{spike_product_module,left_moduleaggregateexact,magma}.
    Hence $P$ is a unital magma by
        \auto{spike_product_module,left_moduleaggregateexact,
            unitalmagma}.
    Hence $P$ is a monoid by
        \auto{spike_product_module,left_moduleaggregateexact,monoid}.
    Hence $P$ is a group by
        \auto{spike_product_module,left_moduleaggregateexact,group}.
    For all $x,y\in\carrier[P]$ we have
        $\mul[P](x,y)=\mul[P](y,x)$ by
        \auto{commutative_group_commutative}.
    Hence $P$ is a commutative group by
        \auto{group,commutative_group}.
    $\scalarMul[P]\in
        \funs{\carrier[R]\times\carrier[P]}{\carrier[P]}$ by
        \auto{spike_product_module_scalar_type}.
    $\carrier[P]=\dependentProduct{I}{C}$ by
        \auto{spike_product_group_fields}.
    $C$ is a function on $I$ by
        \auto{spike_group_carrier_family_type}.

    Show that for all $a,x,y$ such that $a\in\carrier[R]$ and
        $x,y\in\carrier[P]$ we have
        $\scalarMul[P](a,\mul[P](x,y))
            =\mul[P](\scalarMul[P](a,x),\scalarMul[P](a,y))$.
    \begin{subproof}
        Fix $a,x,y$.
        Assume $a\in\carrier[R]$ and $x,y\in\carrier[P]$.
        $\mul[P](x,y),\scalarMul[P](a,x),\scalarMul[P](a,y)
            \in\carrier[P]$ by
            \auto{magma_welldef,spike_product_module_scalar_type,
                funs_type_apply,times_tuple_intro}.
        Hence
            $\scalarMul[P](a,\mul[P](x,y)),
                \mul[P](\scalarMul[P](a,x),\scalarMul[P](a,y))
            \in\carrier[P]$ by
            \auto{magma_welldef,spike_product_module_scalar_type,
                funs_type_apply,times_tuple_intro}.
        Hence
            $\scalarMul[P](a,\mul[P](x,y)),
                \mul[P](\scalarMul[P](a,x),\scalarMul[P](a,y))
            \in\dependentProduct{I}{C}$ by \auto.
        Show that for every $i\in I$ we have
            $\apply{\scalarMul[P](a,\mul[P](x,y))}{i}
                =\apply{\mul[P]
                    (\scalarMul[P](a,x),\scalarMul[P](a,y))}{i}$.
        \begin{subproof}
            Fix $i\in I$.
            $x(i),y(i)\in C(i)$ by
                \auto{dependent_product_coordinate}.
            $C(i)=\carrier[M(i)]$ by
                \auto{spike_module_product_carrier_coordinate}.
            Follows by
                \auto{spike_product_module_scalar_coordinate,
                    spike_product_module_add_coordinate,
                    module_scalar_vector_distributive}.
        \end{subproof}
        Follows by \auto{spike_dependent_product_extensionality}.
    \end{subproof}

    Show that for all $a,b,x$ such that $a,b\in\carrier[R]$ and
        $x\in\carrier[P]$ we have
        $\scalarMul[P](\ringAdd[R](a,b),x)
            =\mul[P](\scalarMul[P](a,x),\scalarMul[P](b,x))$.
    \begin{subproof}
        Fix $a,b,x$.
        Assume $a,b\in\carrier[R]$ and $x\in\carrier[P]$.
        $\ringAdd[R](a,b)\in\carrier[R]$ by \auto{ring_add_type}.
        $\scalarMul[P](a,x),\scalarMul[P](b,x)\in\carrier[P]$ by
            \auto{spike_product_module_scalar_type,
                funs_type_apply,times_tuple_intro}.
        Hence
            $\scalarMul[P](\ringAdd[R](a,b),x),
                \mul[P](\scalarMul[P](a,x),\scalarMul[P](b,x))
            \in\carrier[P]$ by
            \auto{spike_product_module_scalar_type,
                funs_type_apply,times_tuple_intro,magma_welldef}.
        Hence
            $\scalarMul[P](\ringAdd[R](a,b),x),
                \mul[P](\scalarMul[P](a,x),\scalarMul[P](b,x))
            \in\dependentProduct{I}{C}$ by \auto.
        Show that for every $i\in I$ we have
            $\apply{\scalarMul[P](\ringAdd[R](a,b),x)}{i}
                =\apply{\mul[P]
                    (\scalarMul[P](a,x),\scalarMul[P](b,x))}{i}$.
        \begin{subproof}
            Fix $i\in I$.
            $x(i)\in C(i)$ by \auto{dependent_product_coordinate}.
            $C(i)=\carrier[M(i)]$ by
                \auto{spike_module_product_carrier_coordinate}.
            Follows by
                \auto{spike_product_module_scalar_coordinate,
                    spike_product_module_add_coordinate,
                    module_scalar_add_distributive}.
        \end{subproof}
        Follows by \auto{spike_dependent_product_extensionality}.
    \end{subproof}

    Show that for all $a,b,x$ such that $a,b\in\carrier[R]$ and
        $x\in\carrier[P]$ we have
        $\scalarMul[P](a,\scalarMul[P](b,x))
            =\scalarMul[P](\ringMul[R](a,b),x)$.
    \begin{subproof}
        Fix $a,b,x$.
        Assume $a,b\in\carrier[R]$ and $x\in\carrier[P]$.
        $\ringMul[R](a,b)\in\carrier[R]$ by \auto{ring_mul_type}.
        $\scalarMul[P](b,x)\in\carrier[P]$ by
            \auto{spike_product_module_scalar_type,

        Hence
            $\scalarMul[P](a,\scalarMul[P](b,x)),

        Hence
            $\scalarMul[P](a,\scalarMul[P](b,x)),

\begin{proposition}\label{spike_pointwise_module_scalar_support}
    Let $R$ be a ring.
    Suppose $M$ is a function on $I$.
    Suppose for every $i\in I$ we have $M(i)$ is a left module over $R$.
    Let $A=\spikeModuleAdditiveGroupFamily{I}{M}$.
    Let $C=\spikeGroupCarrierFamily{I}{A}$.
    Let $z=\spikeProductGroupNeutral{I}{A}$.
    Let $a\in\carrier[R]$ and $x\in\dependentProduct{I}{C}$.

        $\dependentSupport{I}{
            \Apply{\spikePointwiseModuleScalarOp}{I,M,a,x}}{z}
        \subseteq\dependentSupport{I}{x}{z}$.
\end{proposition}

    Let $u=\Apply{\spikePointwiseModuleScalarOp}{I,M,a,x}$.
    Show that every element of $\dependentSupport{I}{u}{z}$
        is an element of $\dependentSupport{I}{x}{z}$.
    \begin{subproof}
        Fix $i\in\dependentSupport{I}{u}{z}$.
        Hence $i\in I$ and $u(i)\neq z(i)$ by
            \auto{spike_dependent_support_iff}.
        Show that $i\in\dependentSupport{I}{x}{z}$.
        \begin{subproof}
            Suppose not.
            Hence $x(i)=z(i)$ by
                \auto{spike_dependent_support_iff}.
            $A$ is a function on $I$ by
                \auto{spike_module_additive_group_family_type}.
            $A(i)=\ModuleAdditiveGroup{M(i)}$ by
                \auto{spike_module_additive_group_family_apply}.
            $z(i)=\neutral[A(i)]$ by
                \auto{spike_product_group_neutral_coordinate}.
            Hence $z(i)=\neutral[M(i)]$ by
                \auto{spike_module_additive_group_fields}.
            $u(i)=\scalarMul[M(i)](a,x(i))$ by
                \auto{spike_pointwise_module_scalar_coordinate}.
            Hence $u(i)=\neutral[M(i)]$ by
                \auto{module_scalar_vector_zero}.
            Hence $u(i)=z(i)$ by \auto.
            Contradiction.
        \end{subproof}
        Follows.
    \end{subproof}
    Follows by \auto{subseteq}.
\end{proof}

\begin{theorem}\label{spike_direct_sum_module_intro}

            \funs{\carrier[R]\times\carrier[P]}{\carrier[P]}$.
    Suppose for all $a,x,y$ such that $a\in\carrier[R]$ and
        $x,y\in\carrier[P]$ we have
        $\scalarMul[P](a,\mul[P](x,y))
            =\mul[P](\scalarMul[P](a,x),\scalarMul[P](a,y))$.
    Suppose for all $a,b,x$ such that $a,b\in\carrier[R]$ and
        $x\in\carrier[P]$ we have
        $\scalarMul[P](\ringAdd[R](a,b),x)
            =\mul[P](\scalarMul[P](a,x),\scalarMul[P](b,x))$.
    Suppose for all $a,b,x$ such that $a,b\in\carrier[R]$ and
        $x\in\carrier[P]$ we have
        $\scalarMul[P](a,\scalarMul[P](b,x))
            =\scalarMul[P](\ringMul[R](a,b),x)$.
    Suppose for every $x\in\carrier[P]$ we have
        $\scalarMul[P](\ringOne[R],x)=x$.

\end{theorem}
\begin{proof}
    It is not the case that it is not the case that
        $P$ is a left module over $R$ by
        \auto{spike_direct_sum_module,left_moduleaggregateexact,
            left_module}.
    Hence $P$ is a left module over $R$ by \auto.
    Follows by
        \auto{spike_direct_sum_module,left_moduleaggregateexact}.
\end{proof}

\begin{theorem}\label{spike_direct_sum_module_is_exact}
    Let $R$ be a ring.
    Suppose $M$ is a function on $I$.
    Suppose for every $i\in I$ we have $M(i)$ is a left module over $R$.
    Then $\DirectSumModule{R}{I}{M}$
        is an exact left module over $R$.
\end{theorem}
\begin{proof}
    Let $A=\spikeModuleAdditiveGroupFamily{I}{M}$.
    Let $Q=\DirectSumGroup{I}{A}$.
    Let $C=\spikeGroupCarrierFamily{I}{A}$.
    Let $P=\DirectSumModule{R}{I}{M}$.
    $A$ is a function on $I$ by
        \auto{spike_module_additive_group_family_type}.
    Show that for every $i\in I$ we have $A(i)$ is a commutative group.
    \begin{subproof}
        Fix $i\in I$.
        $A(i)=\ModuleAdditiveGroup{M(i)}$ by
            \auto{spike_module_additive_group_family_apply}.
        Follows by \auto{spike_module_additive_group_is_exact}.
    \end{subproof}
    Hence $Q$ is an exact commutative group by
        \auto{spike_direct_sum_group_is_exact}.
    Hence $Q$ is a group by \auto{commutative_group_is_group}.
    Hence $Q$ is a monoid by \auto{group_is_monoid}.
    Hence $Q$ is a unital magma by \auto{monoid_is_unital_magma}.
    Hence $Q$ is a magma by \auto{monoid_is_magma}.
    $\carrier[P]=\carrier[Q]$ and $\mul[P]=\mul[Q]$ and
        $\neutral[P]=\neutral[Q]$ and
        $\groupInv[P]=\groupInv[Q]$ by
        \auto{spike_direct_sum_module_fields}.
    $\mul[P]\in
        \funs{\carrier[P]\times\carrier[P]}{\carrier[P]}$ by
        \auto{magma_mul_graph}.
    $\neutral[P]\in\carrier[P]$ by \auto{unitalmagma_type}.
    $\groupInv[P]\in\funs{\carrier[P]}{\carrier[P]}$ by
        \auto{group_inverse_graph}.
    For all $x,y,z\in\carrier[P]$ we have
        $\mul[P](x,\mul[P](y,z))=\mul[P](\mul[P](x,y),z)$ by
        \auto{monoid_assoc}.
    For every $x\in\carrier[P]$ we have
        $\mul[P](x,\neutral[P])=x$ and
        $\mul[P](\neutral[P],x)=x$ by
        \auto{unitalmagma_right,unitalmagma_left}.
    For every $x\in\carrier[P]$ we have
        $\mul[P](x,\groupInv[P](x))=\neutral[P]$ and
        $\mul[P](\groupInv[P](x),x)=\neutral[P]$ by
        \auto{group_inverse_right,group_inverse_left}.
    Hence $P$ is a magma by
        \auto{spike_direct_sum_module,left_moduleaggregateexact,magma}.
    Hence $P$ is a unital magma by
        \auto{spike_direct_sum_module,left_moduleaggregateexact,
            unitalmagma}.
    Hence $P$ is a monoid by
        \auto{spike_direct_sum_module,left_moduleaggregateexact,monoid}.
    Hence $P$ is a group by
        \auto{spike_direct_sum_module,left_moduleaggregateexact,group}.
    For all $x,y\in\carrier[P]$ we have
        $\mul[P](x,y)=\mul[P](y,x)$ by
        \auto{commutative_group_commutative}.
    Hence $P$ is a commutative group by
        \auto{group,commutative_group}.
    $\scalarMul[P]\in
        \funs{\carrier[R]\times\carrier[P]}{\carrier[P]}$ by
        \auto{spike_direct_sum_module_scalar_type}.
    $\carrier[P]=
        \dependentFinitelySupportedFunctions
            {I}{C}{\spikeProductGroupNeutral{I}{A}}$ by
        \auto{spike_direct_sum_module_fields,
            spike_direct_sum_group_carrier}.
    Hence $\carrier[P]\subseteq\dependentProduct{I}{C}$ by
        \auto{spike_dependent_finitely_supported_functions_iff,
            subseteq}.
    $C$ is a function on $I$ by
        \auto{spike_group_carrier_family_type}.

    Show that for all $a,x,y$ such that $a\in\carrier[R]$ and
        $x,y\in\carrier[P]$ we have
        $\scalarMul[P](a,\mul[P](x,y))
            =\mul[P](\scalarMul[P](a,x),\scalarMul[P](a,y))$.
    \begin{subproof}
        Fix $a,x,y$.
        Assume $a\in\carrier[R]$ and $x,y\in\carrier[P]$.
        $\mul[P](x,y),\scalarMul[P](a,x),\scalarMul[P](a,y)
            \in\carrier[P]$ by
            \auto{magma_welldef,spike_direct_sum_module_scalar_type,
                funs_type_apply,times_tuple_intro}.
        Hence
            $\scalarMul[P](a,\mul[P](x,y)),
                \mul[P](\scalarMul[P](a,x),\scalarMul[P](a,y))
            \in\carrier[P]$ by
            \auto{magma_welldef,spike_direct_sum_module_scalar_type,
                funs_type_apply,times_tuple_intro}.
        Hence
            $\scalarMul[P](a,\mul[P](x,y)),
                \mul[P](\scalarMul[P](a,x),\scalarMul[P](a,y))
            \in\dependentProduct{I}{C}$ by \auto{subseteq}.
        Show that for every $i\in I$ we have
            $\apply{\scalarMul[P](a,\mul[P](x,y))}{i}
                =\apply{\mul[P]
                    (\scalarMul[P](a,x),\scalarMul[P](a,y))}{i}$.
        \begin{subproof}
            Fix $i\in I$.
            $x(i),y(i)\in C(i)$ by
                \auto{dependent_product_coordinate,subseteq}.
            $C(i)=\carrier[M(i)]$ by
                \auto{spike_module_product_carrier_coordinate}.
            Follows by
                \auto{spike_direct_sum_module_scalar_coordinate,
                    spike_direct_sum_module_add_coordinate,
                    module_scalar_vector_distributive}.
        \end{subproof}
        Follows by \auto{spike_dependent_product_extensionality}.
    \end{subproof}

    Show that for all $a,b,x$ such that $a,b\in\carrier[R]$ and
        $x\in\carrier[P]$ we have
        $\scalarMul[P](\ringAdd[R](a,b),x)
            =\mul[P](\scalarMul[P](a,x),\scalarMul[P](b,x))$.
    \begin{subproof}
        Fix $a,b,x$.
        Assume $a,b\in\carrier[R]$ and $x\in\carrier[P]$.
        $\ringAdd[R](a,b)\in\carrier[R]$ by \auto{ring_add_type}.
        $\scalarMul[P](a,x),\scalarMul[P](b,x)\in\carrier[P]$ by
            \auto{spike_direct_sum_module_scalar_type,
                funs_type_apply,times_tuple_intro}.
        Hence
            $\scalarMul[P](\ringAdd[R](a,b),x),
                \mul[P](\scalarMul[P](a,x),\scalarMul[P](b,x))
            \in\carrier[P]$ by
            \auto{spike_direct_sum_module_scalar_type,
                funs_type_apply,times_tuple_intro,magma_welldef}.
        Hence
            $\scalarMul[P](\ringAdd[R](a,b),x),
                \mul[P](\scalarMul[P](a,x),\scalarMul[P](b,x))
            \in\dependentProduct{I}{C}$ by \auto{subseteq}.
        Show that for every $i\in I$ we have
            $\apply{\scalarMul[P](\ringAdd[R](a,b),x)}{i}
                =\apply{\mul[P]
                    (\scalarMul[P](a,x),\scalarMul[P](b,x))}{i}$.
        \begin{subproof}
            Fix $i\in I$.
            $x(i)\in C(i)$ by
                \auto{dependent_product_coordinate,subseteq}.
            $C(i)=\carrier[M(i)]$ by
                \auto{spike_module_product_carrier_coordinate}.
            Follows by
                \auto{spike_direct_sum_module_scalar_coordinate,
                    spike_direct_sum_module_add_coordinate,
                    module_scalar_add_distributive}.
        \end{subproof}
        Follows by \auto{spike_dependent_product_extensionality}.
    \end{subproof}

    Show that for all $a,b,x$ such that $a,b\in\carrier[R]$ and
        $x\in\carrier[P]$ we have
        $\scalarMul[P](a,\scalarMul[P](b,x))
            =\scalarMul[P](\ringMul[R](a,b),x)$.
    \begin{subproof}
        Fix $a,b,x$.
        Assume $a,b\in\carrier[R]$ and $x\in\carrier[P]$.
        $\ringMul[R](a,b)\in\carrier[R]$ by \auto{ring_mul_type}.
        $\scalarMul[P](b,x)\in\carrier[P]$ by
            \auto{spike_direct_sum_module_scalar_type,
                funs_type_apply,times_tuple_intro}.
        Hence
            $\scalarMul[P](a,\scalarMul[P](b,x)),
                \scalarMul[P](\ringMul[R](a,b),x)
            \in\carrier[P]$ by
            \auto{spike_direct_sum_module_scalar_type,
                funs_type_apply,times_tuple_intro}.
        Hence
            $\scalarMul[P](a,\scalarMul[P](b,x)),
                \scalarMul[P](\ringMul[R](a,b),x)
            \in\dependentProduct{I}{C}$ by \auto{subseteq}.
        Show that for every $i\in I$ we have
            $\apply{\scalarMul[P](a,\scalarMul[P](b,x))}{i}
                =\apply{\scalarMul[P](\ringMul[R](a,b),x)}{i}$.
        \begin{subproof}
            Fix $i\in I$.
            $x(i)\in C(i)$ by
                \auto{dependent_product_coordinate,subseteq}.
            $C(i)=\carrier[M(i)]$ by
                \auto{spike_module_product_carrier_coordinate}.
            Follows by
                \auto{spike_direct_sum_module_scalar_coordinate,
                    module_scalar_associative}.
        \end{subproof}
        Follows by \auto{spike_dependent_product_extensionality}.
    \end{subproof}

    Show that for every $x\in\carrier[P]$ we have
        $\scalarMul[P](\ringOne[R],x)=x$.
    \begin{subproof}
        Fix $x\in\carrier[P]$.
        $\ringOne[R]\in\carrier[R]$ by \auto{ring_one_type}.
        $\scalarMul[P](\ringOne[R],x)\in\carrier[P]$ by
            \auto{spike_direct_sum_module_scalar_type,

        Show that for every $i\in I$ we have
            $\apply{\scalarMul[P](\ringOne[R],x)}{i}=x(i)$.
        \begin{subproof}
            Fix $i\in I$.
            $x(i)\in C(i)$ by
                \auto{dependent_product_coordinate,subseteq}.
            $C(i)=\carrier[M(i)]$ by
                \auto{spike_module_product_carrier_coordinate}.
            Follows by
                \auto{spike_direct_sum_module_scalar_coordinate,
                    module_scalar_one}.
        \end{subproof}
        Follows by \auto{spike_dependent_product_extensionality,subseteq}.
    \end{subproof}
    Follows by \auto{spike_direct_sum_module_intro}.
\end{proof}

\begin{theorem}\label{spike_finite_direct_sum_module_eq_product_module}
    Let $R$ be a ring.
    Suppose $I$ is finite.
    Suppose $M$ is a function on $I$.
    Suppose for every $i\in I$ we have $M(i)$ is a left module over $R$.

\end{theorem}
\begin{proof}
    Let $A=\spikeModuleAdditiveGroupFamily{I}{M}$.
    Let $D=\DirectSumModule{R}{I}{M}$.
    Let $P=\ProductModule{R}{I}{M}$.
    $A$ is a function on $I$ by
        \auto{spike_module_additive_group_family_type}.

        $A(i)$ is a commutative group.
    \begin{subproof}

            \auto{spike_module_additive_group_family_apply}.
        Follows by \auto{spike_module_additive_group_is_exact}.
    \end{subproof}
    $\DirectSumGroup{I}{A}=\ProductGroup{I}{A}$ by
        \auto{spike_finite_direct_sum_group_eq_product_group}.
    $D$ is an exact left module over $R$ by
        \auto{spike_direct_sum_module_is_exact}.
    $P$ is an exact left module over $R$ by
        \auto{spike_product_module_is_exact}.
    $\carrier[D]=\carrier[P]$ and $\mul[D]=\mul[P]$ and
        $\neutral[D]=\neutral[P]$ and
        $\groupInv[D]=\groupInv[P]$ by
        \auto{spike_direct_sum_module_fields,
            spike_product_module_fields}.
    $\scalarMul[D]=\scalarMul[P]$ by
        \auto{spike_direct_sum_module_fields,
            spike_product_module_fields}.
    Follows by
        \auto{spike_direct_sum_module,spike_product_module,
            spike_direct_sum_module_additive_group,
            spike_product_module_additive_group,
            spike_direct_sum_module_scalar_graph,
            spike_product_module_scalar_graph}.
\end{proof}

\begin{theorem}\label{spike_direct_sum_group_inclusion_hom}
    Suppose $G$ is a function on $I$.
    Suppose for every $j\in I$ we have $G(j)$ is a commutative group.
    Let $i\in I$.
    Then
        $\groupHom
            {\directSumGroupInclusion{I}{G}{i}}
            {G(i)}{\DirectSumGroup{I}{G}}$.
\end{theorem}
\begin{proof}
    Let $C=\spikeGroupCarrierFamily{I}{G}$.
    Let $z=\spikeProductGroupNeutral{I}{G}$.
    Let $S=\DirectSumGroup{I}{G}$.
    Let $m=\directSumGroupInclusion{I}{G}{i}$.
    $G(i)$ is a group by
        \auto{commutative_group_is_group}.
    $S$ is a commutative group by
        \auto{spike_direct_sum_group_is_exact}.
    Hence $S$ is a group by \auto{commutative_group_is_group}.
    $m\in\funs{\carrier[G(i)]}{\carrier[S]}$ by
        \auto{spike_direct_sum_group_inclusion_type}.
    $C$ is a function on $I$ by
        \auto{spike_group_carrier_family_type}.
    For every $j\in I$ we have $G(j)$ is a group by
        \auto{commutative_group_is_group}.
    $z\in\dependentProduct{I}{C}$ by
        \auto{spike_product_group_neutral_type}.
    Show that for all $a,b\in\carrier[G(i)]$ we have
        $m(\mul[G(i)](a,b))=\mul[S](m(a),m(b))$.
    \begin{subproof}
        Fix $a,b\in\carrier[G(i)]$.
        $G(i)$ is a monoid by \auto{group_is_monoid}.
        Hence $G(i)$ is a magma by \auto{monoid_is_magma}.
        $\mul[G(i)](a,b)\in\carrier[G(i)]$ by
            \auto{magma_welldef}.
        $m(a),m(b)\in\carrier[S]$ by
            \auto{funs_type_apply}.
        Hence $m(\mul[G(i)](a,b)),\mul[S](m(a),m(b))
            \in\carrier[S]$ by
            \auto{funs_type_apply,group_is_monoid,
                monoid_is_magma,magma_welldef}.
        Hence $m(\mul[G(i)](a,b)),\mul[S](m(a),m(b))
            \in\dependentProduct{I}{C}$ by
            \auto{spike_direct_sum_group_carrier,
                spike_dependent_finitely_supported_functions_iff}.
        Show that for every $j\in I$ we have
            $\apply{m(\mul[G(i)](a,b))}{j}
                =\apply{\mul[S](m(a),m(b))}{j}$.
        \begin{subproof}
            Fix $j\in I$.
            \begin{byCase}
            \caseOf{$j=i$.}
                Follows by
                    \auto{spike_direct_sum_group_inclusion_apply,
                        spike_dependent_point_mass_selected,
                        spike_direct_sum_group_mul_coordinate,
                        spike_group_carrier_family_apply}.
            \caseOf{$j\neq i$.}
                $C(i)=\carrier[G(i)]$ by
                    \auto{spike_group_carrier_family_apply}.
                $\mul[G(i)](a,b)\in C(i)$ by \auto.
                $\apply{m(\mul[G(i)](a,b))}{j}=z(j)$ and
                    $\apply{m(a)}{j}=z(j)$ and
                    $\apply{m(b)}{j}=z(j)$ by
                    \auto{spike_direct_sum_group_inclusion_apply,
                        spike_dependent_point_mass_other}.
                $z(j)=\neutral[G(j)]$ by
                    \auto{spike_product_group_neutral_coordinate}.
                $G(j)$ is a monoid by \auto{group_is_monoid}.
                Hence $G(j)$ is a unital magma by
                    \auto{monoid_is_unital_magma}.
                $\apply{\mul[S](m(a),m(b))}{j}
                    =\mul[G(j)](\apply{m(a)}{j},\apply{m(b)}{j})$ by
                    \auto{spike_direct_sum_group_mul_coordinate}.
                Hence $\apply{\mul[S](m(a),m(b))}{j}
                    =\mul[G(j)](\neutral[G(j)],\neutral[G(j)])$ by
                    \auto.
                $\neutral[G(j)]\in\carrier[G(j)]$ by
                    \auto{unitalmagma_type}.
                $\mul[G(j)](\neutral[G(j)],\neutral[G(j)])
                    =\neutral[G(j)]$ by \auto{unitalmagma_left}.
                Follows by \auto.
            \end{byCase}
        \end{subproof}
        Follows by \auto{spike_dependent_product_extensionality}.
    \end{subproof}
    Follows by \auto{group_hom}.
\end{proof}

\subsection{Direct-sum assembly mediators}

For a finitely supported section, the assembly map folds the images of its
nonzero coordinates.  The fold is canonical, so the construction contains no
selected enumeration of the support.

\begin{definition}\label{spike_direct_sum_group_contribution_op}
    $\spikeDirectSumGroupContributionOp:
        \Set\to\Set\to\Set\to\Set
        =\fun{f,x,i:\Set}{\apply{\apply{f}{i}}{\apply{x}{i}}}$.
\end{definition}

\begin{definition}\label{spike_direct_sum_group_contribution}
    $\spikeDirectSumGroupContribution{I}{G}{H}{f}{x}
        =\Apply{\graphOnOp}{
            \dependentSupport{I}{x}{\spikeProductGroupNeutral{I}{G}},
            \carrier[H],
            \Apply{\spikeDirectSumGroupContributionOp}{f,x}}$.
\end{definition}

\begin{proposition}\label{spike_direct_sum_group_contribution_apply}
    Suppose $G$ is a function on $I$ and $f$ is a function on $I$.
    Let $i\in\dependentSupport{I}{x}{
        \spikeProductGroupNeutral{I}{G}}$.
    Then
        $\apply{\spikeDirectSumGroupContribution{I}{G}{H}{f}{x}}{i}
            =\apply{\apply{f}{i}}{x(i)}$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{spike_direct_sum_group_contribution,
            spike_direct_sum_group_contribution_op,
            logical_graph_on_apply}.
\end{proof}

\begin{proposition}\label{spike_direct_sum_group_contribution_type}
    Suppose $G$ is a function on $I$ and $H$ is a group.
    Suppose for every $i\in I$ we have $G(i)$ is a group.
    Suppose $f$ is a function on $I$ and
        for every $i\in I$ we have $\groupHom{f(i)}{G(i)}{H}$.
    Let $x\in\carrier[\DirectSumGroup{I}{G}]$.
    Then
        $\spikeDirectSumGroupContribution{I}{G}{H}{f}{x}\in
            \funs{\dependentSupport{I}{x}{
                \spikeProductGroupNeutral{I}{G}}}{\carrier[H]}$.
\end{proposition}
\begin{proof}
    Let $C=\spikeGroupCarrierFamily{I}{G}$.
    Let $z=\spikeProductGroupNeutral{I}{G}$.
    Let $D=\dependentSupport{I}{x}{z}$.
    $x\in\dependentFinitelySupportedFunctions{I}{C}{z}$ by
        \auto{spike_direct_sum_group_carrier}.
    Hence $x\in\dependentProduct{I}{C}$ by
        \auto{spike_dependent_finitely_supported_functions_iff}.
    Show that for every $i\in D$ we have
        $\apply{\apply{f}{i}}{x(i)}\in\carrier[H]$.
    \begin{subproof}
        Fix $i\in D$.
        Hence $i\in I$ by \auto{spike_dependent_support_iff}.
        $x(i)\in C(i)$ by \auto{dependent_product_coordinate}.
        $C(i)=\carrier[G(i)]$ by
            \auto{spike_group_carrier_family_apply}.
        $f(i)\in\funs{\carrier[G(i)]}{\carrier[H]}$ by
            \auto{group_hom_type}.
        Follows by \auto{funs_type_apply}.
    \end{subproof}
    Hence for every $i\in D$ we have
        $\Apply{\Apply{\spikeDirectSumGroupContributionOp}{f,x}}{i}
            \in\carrier[H]$ by
        \auto{spike_direct_sum_group_contribution_op}.
    $\Apply{\graphOnOp}{D,\carrier[H],
        \Apply{\spikeDirectSumGroupContributionOp}{f,x}}
        \in\funs{D}{\carrier[H]}$ by
        \auto{logical_graph_on_function}.

\begin{proposition}\label{spike_direct_sum_group_mediator_value_type}
    Suppose $G$ is a function on $I$.
    Suppose for every $i\in I$ we have $G(i)$ is a commutative group.
    Let $H$ be a commutative group.
    Suppose $f$ is a function on $I$ and
        for every $i\in I$ we have $\groupHom{f(i)}{G(i)}{H}$.
    Let $x\in\carrier[\DirectSumGroup{I}{G}]$.
    Then
        $\Apply{\spikeDirectSumGroupMediatorValueOp}{I,G,H,f,x}
            \in\carrier[H]$.
\end{proposition}
\begin{proof}
    Let $z=\spikeProductGroupNeutral{I}{G}$.
    Let $D=\dependentSupport{I}{x}{z}$.
    Let $c=\spikeDirectSumGroupContribution{I}{G}{H}{f}{x}$.
    $x\in\dependentFinitelySupportedFunctions
        {I}{\spikeGroupCarrierFamily{I}{G}}{z}$ by
        \auto{spike_direct_sum_group_carrier}.
    Hence $D$ is finite by
        \auto{spike_dependent_finitely_supported_functions_iff}.
    $H$ is a group by \auto{commutative_group_is_group}.
    Hence $H$ is a monoid by \auto{group_is_monoid}.
    For all $u,v\in\carrier[H]$ we have
        $\mul[H](u,v)=\mul[H](v,u)$ by
        \auto{commutative_group_commutative}.
    For every $i\in I$ we have $G(i)$ is a group by
        \auto{commutative_group_is_group}.
    $c\in\funs{D}{\carrier[H]}$ by
        \auto{spike_direct_sum_group_contribution_type}.
    $\finiteFold{H}{D}{c}\in\carrier[H]$ by
        \auto{spike_finite_fold_canonical}.
    Follows by \auto{spike_direct_sum_group_mediator_value_op}.
\end{proof}

\begin{proposition}\label{spike_linear_map_additive_group_hom}
    Suppose $f$ is a linear map from $M$ to $N$ over $R$.

        \auto{linear_map}.

        $\ModuleAdditiveGroup{N}$ is a commutative group by
        \auto{spike_module_additive_group_is_exact}.
    Hence $\ModuleAdditiveGroup{M}$ is a group and
        $\ModuleAdditiveGroup{N}$ is a group by
        \auto{commutative_group_is_group}.
    $f\in\funs{\carrier[\ModuleAdditiveGroup{M}]}{
        \carrier[\ModuleAdditiveGroup{N}]}$ by
        \auto{linear_map,spike_module_additive_group_fields}.
    For all $x,y\in\carrier[\ModuleAdditiveGroup{M}]$ we have
        $f(\mul[\ModuleAdditiveGroup{M}](x,y))
            =\mul[\ModuleAdditiveGroup{N}](f(x),f(y))$ by
        \auto{linear_map,spike_module_additive_group_fields}.
    Follows by \auto{group_hom}.
\end{proof}

\begin{proposition}\label{spike_direct_sum_module_mediator_type}
    Let $R$ be a ring.
    Suppose $M$ is a function on $I$.
    Suppose for every $i\in I$ we have $M(i)$ is a left module over $R$.
    Let $N$ be a left module over $R$.
    Suppose $f$ is a function on $I$ and for every $i\in I$ we have
        $f(i)$ is a linear map from $M(i)$ to $N$ over $R$.
    Then
        $\directSumModuleMediator{R}{I}{M}{N}{f}\in
            \funs{\carrier[\DirectSumModule{R}{I}{M}]}{
                \carrier[N]}$.
\end{proposition}
\begin{proof}
    Let $A=\spikeModuleAdditiveGroupFamily{I}{M}$.
    Let $B=\ModuleAdditiveGroup{N}$.
    Let $D=\DirectSumModule{R}{I}{M}$.
    Let $Q=\DirectSumGroup{I}{A}$.
    $A$ is a function on $I$ by
        \auto{spike_module_additive_group_family_type}.
    Show that for every $i\in I$ we have $A(i)$ is a commutative group.
    \begin{subproof}
        Fix $i\in I$.
        $A(i)=\ModuleAdditiveGroup{M(i)}$ by
            \auto{spike_module_additive_group_family_apply}.
        Follows by \auto{spike_module_additive_group_is_exact}.
    \end{subproof}
    $B$ is a commutative group by
        \auto{spike_module_additive_group_is_exact}.
    Show that for every $i\in I$ we have $\groupHom{f(i)}{A(i)}{B}$.
    \begin{subproof}
        Fix $i\in I$.
        $A(i)=\ModuleAdditiveGroup{M(i)}$ by
            \auto{spike_module_additive_group_family_apply}.
        Follows by \auto{spike_linear_map_additive_group_hom}.
    \end{subproof}
    $\directSumGroupMediator{I}{A}{B}{f}\in
        \funs{\carrier[Q]}{\carrier[B]}$ by
        \auto{spike_direct_sum_group_mediator_type}.
    $\carrier[D]=\carrier[Q]$ by
        \auto{spike_direct_sum_module_fields}.
    $\carrier[B]=\carrier[N]$ by
        \auto{spike_module_additive_group_fields}.
    Follows by \auto{spike_direct_sum_module_mediator}.
\end{proof}
