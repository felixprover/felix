\section{The canonical submodule predicate}

\begin{definition}\label{spike_prototype_submodule}
    $\prototypeSubmodule{R}{N}{M}$ iff
        $M$ is a left module over $R$ and
        $\subgroup{N}{\ModuleAdditiveGroup{M}}$ and
        for all $a,x$ such that
            $a\in\carrier[R]$ and $x\in N$ we have
            $\scalarMul[M](a,x)\in N$.
\end{definition}

\begin{proposition}\label{spike_submodule_implies_prototype}
    Suppose $N$ is a submodule of $M$ over $R$.
    Then $\prototypeSubmodule{R}{N}{M}$.
\end{proposition}
\begin{proof}
    $M$ is a left module over $R$ by \auto{submodule}.
    Hence $\ModuleAdditiveGroup{M}$ is an exact commutative group by
        \auto{spike_module_additive_group_is_exact}.
    Hence $\ModuleAdditiveGroup{M}$ is a group by
        \auto{commutative_group_is_group}.
    $N\subseteq\carrier[\ModuleAdditiveGroup{M}]$ and
        $\neutral[\ModuleAdditiveGroup{M}]\in N$ by
        \auto{submodule,spike_module_additive_group_fields}.
    For all $x,y\in N$ we have
        $\mul[\ModuleAdditiveGroup{M}](x,y)\in N$ by
        \auto{submodule,spike_module_additive_group_fields}.
    For every $x\in N$ we have
        $\groupInv[\ModuleAdditiveGroup{M}](x)\in N$ by
        \auto{submodule,spike_module_additive_group_fields}.
    Hence $\subgroup{N}{\ModuleAdditiveGroup{M}}$ by
        \auto{subgroup}.
    Follows by \auto{submodule,spike_prototype_submodule}.
\end{proof}

\begin{proposition}\label{spike_prototype_implies_submodule}
    Suppose $\prototypeSubmodule{R}{N}{M}$.
    Then $N$ is a submodule of $M$ over $R$.
\end{proposition}
\begin{proof}
    $M$ is a left module over $R$ and
        $\subgroup{N}{\ModuleAdditiveGroup{M}}$ by
        \auto{spike_prototype_submodule}.
    $N\subseteq\carrier[\ModuleAdditiveGroup{M}]$ and
        $\neutral[\ModuleAdditiveGroup{M}]\in N$ by
        \auto{subgroup_subseteq,subgroup_neutral}.
    Hence $N\subseteq\carrier[M]$ and $\neutral[M]\in N$ by
        \auto{spike_module_additive_group_fields}.
    For all $x,y\in N$ we have
        $\mul[M](x,y)\in N$ by
        \auto{subgroup_mul,spike_module_additive_group_fields}.
    For every $x\in N$ we have $\groupInv[M](x)\in N$ by
        \auto{subgroup_inv,spike_module_additive_group_fields}.
    Follows by \auto{spike_prototype_submodule,submodule}.
\end{proof}

\begin{theorem}\label{spike_submodule_equivalence}
    $N$ is a submodule of $M$ over $R$ iff
        $\prototypeSubmodule{R}{N}{M}$.
\end{theorem}
\begin{proof}
    Follows by
        \auto{spike_submodule_implies_prototype,
            spike_prototype_implies_submodule}.
\end{proof}
