\section{The canonical linear-map predicate}

\begin{definition}\label{spike_prototype_linear_map}
    $\prototypeLinearMap{R}{f}{M}{N}$ iff
        $M$ is a left module over $R$ and
        $N$ is a left module over $R$ and
        $\groupHom{f}{\ModuleAdditiveGroup{M}}{
            \ModuleAdditiveGroup{N}}$ and
        for all $a,x$ such that
            $a\in\carrier[R]$ and
            $x\in\carrier[M]$ we have
            $f(\scalarMul[M](a,x))=\scalarMul[N](a,f(x))$.
\end{definition}

\begin{proposition}\label{spike_linear_map_implies_prototype}
    Suppose $f$ is a linear map from $M$ to $N$ over $R$.
    Then $\prototypeLinearMap{R}{f}{M}{N}$.
\end{proposition}
\begin{proof}
    $M$ is a left module over $R$ and
        $N$ is a left module over $R$ by \auto{linear_map}.
    Hence $\ModuleAdditiveGroup{M}$ is an exact commutative group and
        $\ModuleAdditiveGroup{N}$ is an exact commutative group by
        \auto{spike_module_additive_group_is_exact}.
    Hence $\ModuleAdditiveGroup{M}$ is a group and
        $\ModuleAdditiveGroup{N}$ is a group by
        \auto{commutative_group_is_group}.
    $f\in\funs{
        \carrier[\ModuleAdditiveGroup{M}]}{
        \carrier[\ModuleAdditiveGroup{N}]}$ by
        \auto{linear_map,spike_module_additive_group_fields}.
    For all $x,y\in\carrier[\ModuleAdditiveGroup{M}]$ we have
        $f(\mul[\ModuleAdditiveGroup{M}](x,y))
            =\mul[\ModuleAdditiveGroup{N}](f(x),f(y))$ by
        \auto{linear_map,spike_module_additive_group_fields}.
    Hence
        $\groupHom{f}{\ModuleAdditiveGroup{M}}{
            \ModuleAdditiveGroup{N}}$ by
        \auto{group_hom}.
    Follows by \auto{linear_map,spike_prototype_linear_map}.
\end{proof}

\begin{proposition}\label{spike_prototype_implies_linear_map}
    Suppose $\prototypeLinearMap{R}{f}{M}{N}$.
    Then $f$ is a linear map from $M$ to $N$ over $R$.
\end{proposition}
\begin{proof}
    $M$ is a left module over $R$ and
        $N$ is a left module over $R$ and
        $\groupHom{f}{\ModuleAdditiveGroup{M}}{
            \ModuleAdditiveGroup{N}}$ by
        \auto{spike_prototype_linear_map}.
    $f\in\funs{\carrier[M]}{\carrier[N]}$ by
        \auto{group_hom_type,spike_module_additive_group_fields}.
    For all $x,y\in\carrier[M]$ we have
        $f(\mul[M](x,y))=\mul[N](f(x),f(y))$ by
        \auto{group_hom_mul,spike_module_additive_group_fields}.
    Follows by \auto{spike_prototype_linear_map,linear_map}.
\end{proof}

\begin{theorem}\label{spike_linear_map_equivalence}
    $f$ is a linear map from $M$ to $N$ over $R$ iff
        $\prototypeLinearMap{R}{f}{M}{N}$.
\end{theorem}
\begin{proof}
    Follows by
        \auto{spike_linear_map_implies_prototype,
            spike_prototype_implies_linear_map}.
\end{proof}
