\section{Commutator orientation}

This fixture freezes the convention
\( [x,y]=x^{-1}y^{-1}xy \) against the library's convention
\(a\mathbin{\cdot}x\mathbin{\cdot}a^{-1}\) for conjugation.

\begin{definition}\label{spike17_group_commutator}
    $\groupCommutator{G}{x}{y}=
        \mul[G](
            \mul[G](
                \mul[G](\groupInv[G](x),\groupInv[G](y)),x),y)$.
\end{definition}

\begin{proposition}\label{spike17_group_commutator_type}
    Let $G$ be a group.
    Let $x,y\in\carrier[G]$.
    Then $\groupCommutator{G}{x}{y}\in\carrier[G]$.
\end{proposition}
\begin{proof}
    $\groupInv[G](x),\groupInv[G](y)\in\carrier[G]$ by
        \auto{group_inverse_type}.
    Follows by
        \auto{spike17_group_commutator,group_is_monoid,
            monoid_is_magma,magma_welldef}.
\end{proof}

\begin{theorem}\label{spike17_group_commutator_inverse}
    Let $G$ be a group.
    Let $x,y\in\carrier[G]$.
    Then
        $\groupInv[G](\groupCommutator{G}{x}{y})
            =\groupCommutator{G}{y}{x}$.
\end{theorem}
\begin{proof}
    $\groupInv[G](x),\groupInv[G](y)\in\carrier[G]$ by
        \auto{group_inverse_type}.
    Follows by
        \auto{spike17_group_commutator,group_inverse_product,
            group_inverse_involutive,group_is_monoid,monoid_assoc,
            monoid_is_magma,magma_welldef}.
\end{proof}

\begin{lemma}\label{spike17_conjugate_product}
    Let $G$ be a group.
    Let $a,x,y\in\carrier[G]$.
    Then
        $\conjugate{G}{a}{\mul[G](x,y)}
            =\mul[G](\conjugate{G}{a}{x},\conjugate{G}{a}{y})$.
\end{lemma}
\begin{proof}
    Let $b=\groupInv[G](a)$.
    $b\in\carrier[G]$ by
        \auto{group_inverse_type}.
    $\mul[G](b,a)=\neutral[G]$ by
        \auto{group_inverse_left}.
    $\mul[G](\conjugate{G}{a}{x},\conjugate{G}{a}{y})
        =\mul[G](
            \mul[G](\mul[G](a,x),b),
            \mul[G](\mul[G](a,y),b))$
        by \auto{group_conjugate}.
    Hence
        $\mul[G](\conjugate{G}{a}{x},\conjugate{G}{a}{y})
        =\mul[G](
            \mul[G](\mul[G](\mul[G](a,x),b),\mul[G](a,y)),b)$
        by
        \auto{group_is_monoid,monoid_assoc,monoid_is_magma,
            magma_welldef}.
    Hence
        $\mul[G](\conjugate{G}{a}{x},\conjugate{G}{a}{y})
        =\mul[G](
            \mul[G](\mul[G](a,x),\mul[G](b,\mul[G](a,y))),b)$
        by
        \auto{group_is_monoid,monoid_assoc,monoid_is_magma,
            magma_welldef}.
    Hence
        $\mul[G](\conjugate{G}{a}{x},\conjugate{G}{a}{y})
        =\mul[G](
            \mul[G](\mul[G](a,x),\mul[G](\mul[G](b,a),y)),b)$
        by
        \auto{group_is_monoid,monoid_assoc,monoid_is_magma,
            magma_welldef}.
    Hence
        $\mul[G](\conjugate{G}{a}{x},\conjugate{G}{a}{y})
        =\mul[G](\mul[G](\mul[G](a,x),y),b)$
        by
        \auto{group_is_monoid,monoid_is_unital_magma,
            monoid_is_magma,unitalmagma_left,magma_welldef}.
    Hence
        $\mul[G](\conjugate{G}{a}{x},\conjugate{G}{a}{y})
        =\mul[G](\mul[G](a,\mul[G](x,y)),b)$
        by
        \auto{group_is_monoid,monoid_assoc,monoid_is_magma,
            magma_welldef}.
    Follows by \auto{group_conjugate}.
\end{proof}

\begin{lemma}\label{spike17_conjugate_inverse}
    Let $G$ be a group.
    Let $a,x\in\carrier[G]$.
    Then
        $\conjugate{G}{a}{\groupInv[G](x)}
            =\groupInv[G](\conjugate{G}{a}{x})$.
\end{lemma}
\begin{proof}
    $\groupInv[G](a),\groupInv[G](x)\in\carrier[G]$ by
        \auto{group_inverse_type}.
    Follows by
        \auto{group_conjugate,group_inverse_product,
            group_inverse_involutive,group_is_monoid,monoid_assoc,
            monoid_is_magma,magma_welldef}.
\end{proof}

\begin{theorem}\label{spike17_group_commutator_conjugate}
    Let $G$ be a group.
    Let $a,x,y\in\carrier[G]$.
    Then
        $\conjugate{G}{a}{\groupCommutator{G}{x}{y}}
            =\groupCommutator{G}
                {\conjugate{G}{a}{x}}{\conjugate{G}{a}{y}}$.
\end{theorem}
\begin{proof}
    $\groupInv[G](x),\groupInv[G](y)\in\carrier[G]$ by
        \auto{group_inverse_type}.
    $\conjugate{G}{a}{x},\conjugate{G}{a}{y}\in\carrier[G]$ by
        \auto{group_conjugate_type}.
    Follows by
        \auto{spike17_group_commutator,spike17_conjugate_product,
            spike17_conjugate_inverse,group_is_monoid,monoid_is_magma,
            magma_welldef}.
\end{proof}
