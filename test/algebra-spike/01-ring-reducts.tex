\section{Exact ring reduct prototypes}

\begin{definition}\label{spike_additive_group}
    $\AdditiveGroup{R}=
        \structure[group]{
            \carrier=\carrier[R],
            \mul=\ringAdd[R],
            \neutral=\ringZero[R],
            \groupInv=\ringNeg[R]
        }$.
\end{definition}

\begin{proposition}\label{spike_additive_group_fields}
    $\carrier[\AdditiveGroup{R}]=\carrier[R]$ and
    $\mul[\AdditiveGroup{R}]=\ringAdd[R]$ and
    $\neutral[\AdditiveGroup{R}]=\ringZero[R]$ and
    $\groupInv[\AdditiveGroup{R}]=\ringNeg[R]$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{spike_additive_group,
            groupaggregateprojection_carrier,groupaggregateprojection_mul,
            groupaggregateprojection_neutral,groupaggregateprojection_groupInv}.
\end{proof}

\begin{proposition}\label{spike_additive_group_is_exact}
    Let $R$ be a ring.
    Then $\AdditiveGroup{R}$ is an exact commutative group.
\end{proposition}
\begin{proof}
    Let $G=\AdditiveGroup{R}$.
    $\carrier[G]=\carrier[R]$ and $\mul[G]=\ringAdd[R]$ and
        $\neutral[G]=\ringZero[R]$ and
        $\groupInv[G]=\ringNeg[R]$ by
        \auto{spike_additive_group_fields}.
    $\mul[G]\in\funs{\carrier[G]\times\carrier[G]}{\carrier[G]}$ by
        \auto{ring_add_graph}.
    $\neutral[G]\in\carrier[G]$ by \auto{ring_zero_type}.
    $\groupInv[G]\in\funs{\carrier[G]}{\carrier[G]}$ by
        \auto{ring_neg_graph}.
    For all $x,y,z\in\carrier[G]$ we have
        $\mul[G](x,\mul[G](y,z))=\mul[G](\mul[G](x,y),z)$ by
        \auto{ring_add_associative}.
    For every $x\in\carrier[G]$ we have
        $\mul[G](x,\neutral[G])=x$ and
        $\mul[G](\neutral[G],x)=x$ by
        \auto{ring_add_zero_right,ring_add_zero_left}.
    For every $x\in\carrier[G]$ we have
        $\mul[G](x,\groupInv[G](x))=\neutral[G]$ and
        $\mul[G](\groupInv[G](x),x)=\neutral[G]$ by
        \auto{ring_add_inverse_right,ring_add_inverse_left}.
    Hence $G$ is a group by
        \auto{spike_additive_group,groupaggregateexact,
            magma,unitalmagma,monoid,group}.
    For all $x,y\in\carrier[G]$ we have
        $\mul[G](x,y)=\mul[G](y,x)$ by
        \auto{ring_add_commutative}.
    $G$ is a commutative group by
        \auto{group,commutative_group}.
    Follows by
        \auto{spike_additive_group,groupaggregateexact}.
\end{proof}

\begin{definition}\label{spike_multiplicative_monoid}
    $\MultiplicativeMonoid{R}=
        \structure[unital magma]{
            \carrier=\carrier[R],
            \mul=\ringMul[R],
            \neutral=\ringOne[R]
        }$.
\end{definition}

\begin{proposition}\label{spike_multiplicative_monoid_fields}
    $\carrier[\MultiplicativeMonoid{R}]=\carrier[R]$ and
    $\mul[\MultiplicativeMonoid{R}]=\ringMul[R]$ and
    $\neutral[\MultiplicativeMonoid{R}]=\ringOne[R]$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{spike_multiplicative_monoid,
            unitalmagmaaggregateprojection_carrier,
            unitalmagmaaggregateprojection_mul,
            unitalmagmaaggregateprojection_neutral}.
\end{proof}

\begin{proposition}\label{spike_multiplicative_monoid_is_exact}
    Let $R$ be a ring.
    Then $\MultiplicativeMonoid{R}$ is an exact monoid.
\end{proposition}
\begin{proof}
    Let $M=\MultiplicativeMonoid{R}$.
    $\carrier[M]=\carrier[R]$ and $\mul[M]=\ringMul[R]$ and
        $\neutral[M]=\ringOne[R]$ by
        \auto{spike_multiplicative_monoid_fields}.
    $\mul[M]\in\funs{\carrier[M]\times\carrier[M]}{\carrier[M]}$ by
        \auto{ring_mul_graph}.
    $\neutral[M]\in\carrier[M]$ by \auto{ring_one_type}.
    For all $x,y,z\in\carrier[M]$ we have
        $\mul[M](x,\mul[M](y,z))=\mul[M](\mul[M](x,y),z)$ by
        \auto{ring_mul_associative}.
    For every $x\in\carrier[M]$ we have
        $\mul[M](x,\neutral[M])=x$ and
        $\mul[M](\neutral[M],x)=x$ by
        \auto{ring_mul_one_right,ring_mul_one_left}.
    Hence $M$ is a monoid by
        \auto{spike_multiplicative_monoid,unitalmagmaaggregateexact,
            magma,unitalmagma,monoid}.
    Follows by
        \auto{spike_multiplicative_monoid,unitalmagmaaggregateexact}.
\end{proof}
