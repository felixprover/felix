\section{Free modules as direct sums}

The free module on an index set is not a second finite-support construction.
It is definitionally the direct sum of the constant family of left regular
modules.

\begin{definition}\label{spike_regular_module_family}
    $\spikeRegularModuleFamily{R}{I}
        =I\times\{\LeftRegularModule{R}\}$.
\end{definition}

\begin{proposition}\label{spike_regular_module_family_type}
    $\spikeRegularModuleFamily{R}{I}\in
        \funs{I}{\{\LeftRegularModule{R}\}}$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{spike_regular_module_family,constant_graph_type}.
\end{proof}

\begin{proposition}\label{spike_regular_module_family_apply}
    Let $i\in I$.
    Then
        $\apply{\spikeRegularModuleFamily{R}{I}}{i}
            =\LeftRegularModule{R}$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{spike_regular_module_family,
            constant_graph_application}.
\end{proof}

\begin{definition}\label{spike_free_module}
    $\FreeModule{R}{I}
        =\DirectSumModule{R}{I}{\spikeRegularModuleFamily{R}{I}}$.
\end{definition}

\begin{theorem}\label{spike_free_module_constructor_eq}
    $\FreeModule{R}{I}
        =\DirectSumModule{R}{I}{\spikeRegularModuleFamily{R}{I}}$.
\end{theorem}
\begin{proof}
    Follows by \auto{spike_free_module}.
\end{proof}

\begin{theorem}\label{spike_free_module_is_exact}
    Let $R$ be a ring.
    Then $\FreeModule{R}{I}$ is an exact left module over $R$.
\end{theorem}
\begin{proof}
    Let $F=\spikeRegularModuleFamily{R}{I}$.
    $F$ is a function on $I$ by
        \auto{spike_regular_module_family_type,
            funs_is_function,funs}.
    For every $i\in I$ we have $F(i)$ is a left module over $R$ by
        \auto{spike_regular_module_family_apply,
            spike_left_regular_module_is_exact}.
    Follows by
        \auto{spike_free_module,spike_direct_sum_module_is_exact}.
\end{proof}

\begin{proposition}\label{spike_free_module_carrier_family}
    Let $R$ be a ring.
    Let $F=\spikeRegularModuleFamily{R}{I}$.
    Let $A=\spikeModuleAdditiveGroupFamily{I}{F}$.
    Then
        $\spikeGroupCarrierFamily{I}{A}
            =I\times\{\carrier[R]\}$.
\end{proposition}
\begin{proof}
    Let $C=\spikeGroupCarrierFamily{I}{A}$.
    Let $K=I\times\{\carrier[R]\}$.
    $F$ is a function on $I$ by
        \auto{spike_regular_module_family_type,
            funs_is_function,funs}.
    Hence $A$ is a function on $I$ by
        \auto{spike_module_additive_group_family_type}.
    Hence $C$ is a function and $\dom{C}=I$ by
        \auto{spike_group_carrier_family_type}.
    $K\in\funs{I}{\{\carrier[R]\}}$ by
        \auto{constant_graph_type}.
    Hence $K$ is a function and $\dom{K}=I$ by
        \auto{funs_is_function,funs}.
    Show that for every $i\in I$ we have $C(i)=K(i)$.
    \begin{subproof}
        Fix $i\in I$.
        $C(i)=\carrier[A(i)]$ by
            \auto{spike_group_carrier_family_apply}.
        $A(i)=\ModuleAdditiveGroup{F(i)}$ by
            \auto{spike_module_additive_group_family_apply}.
        $F(i)=\LeftRegularModule{R}$ by
            \auto{spike_regular_module_family_apply}.
        $\carrier[\ModuleAdditiveGroup{\LeftRegularModule{R}}]
            =\carrier[\LeftRegularModule{R}]$ by
            \auto{spike_module_additive_group_fields}.
        $\carrier[\LeftRegularModule{R}]=\carrier[R]$ by
            \auto{spike_left_regular_module_fields,
                spike_additive_group_fields}.
        $K(i)=\carrier[R]$ by \auto{constant_graph_application}.
        Follows by \auto.
    \end{subproof}
    Follows by \auto{funext_on_domain}.
\end{proof}

\begin{proposition}\label{spike_free_module_zero_section}
    Let $R$ be a ring.
    Let $F=\spikeRegularModuleFamily{R}{I}$.
    Let $A=\spikeModuleAdditiveGroupFamily{I}{F}$.
    Then
        $\spikeProductGroupNeutral{I}{A}
            =I\times\{\ringZero[R]\}$.
\end{proposition}
\begin{proof}
    Let $z=\spikeProductGroupNeutral{I}{A}$.
    Let $e=I\times\{\ringZero[R]\}$.
    $F$ is a function on $I$ by
        \auto{spike_regular_module_family_type,
            funs_is_function,funs}.
    Hence $A$ is a function on $I$ by
        \auto{spike_module_additive_group_family_type}.
    $z\in\funs{I}{\ran{z}}$ by
        \auto{spike_product_group_neutral,
            logical_graph_on_domain_function}.
    Hence $z$ is a function and $\dom{z}=I$ by
        \auto{funs_is_function,funs}.
    $e\in\funs{I}{\{\ringZero[R]\}}$ by
        \auto{constant_graph_type}.
    Hence $e$ is a function and $\dom{e}=I$ by
        \auto{funs_is_function,funs}.
    Show that for every $i\in I$ we have $z(i)=e(i)$.
    \begin{subproof}
        Fix $i\in I$.
        $z(i)=\neutral[A(i)]$ by
            \auto{spike_product_group_neutral_coordinate}.
        $A(i)=\ModuleAdditiveGroup{F(i)}$ by
            \auto{spike_module_additive_group_family_apply}.
        $F(i)=\LeftRegularModule{R}$ by
            \auto{spike_regular_module_family_apply}.
        $\neutral[\ModuleAdditiveGroup{\LeftRegularModule{R}}]
            =\neutral[\LeftRegularModule{R}]$ by
            \auto{spike_module_additive_group_fields}.
        $\neutral[\LeftRegularModule{R}]=\ringZero[R]$ by
            \auto{spike_left_regular_module_fields,
                spike_additive_group_fields}.
        $e(i)=\ringZero[R]$ by \auto{constant_graph_application}.
        Follows by \auto.
    \end{subproof}
    Follows by \auto{funext_on_domain}.
\end{proof}

\begin{theorem}\label{spike_free_module_carrier}
    Let $R$ be a ring.
    Then
        $\carrier[\FreeModule{R}{I}]
            =\finitelySupportedFunctions
                {I}{\carrier[R]}{\ringZero[R]}$.
\end{theorem}
\begin{proof}
    Let $F=\spikeRegularModuleFamily{R}{I}$.
    Let $A=\spikeModuleAdditiveGroupFamily{I}{F}$.
    Let $C=\spikeGroupCarrierFamily{I}{A}$.
    Let $z=\spikeProductGroupNeutral{I}{A}$.
    Let $D=\DirectSumModule{R}{I}{F}$.
    Let $Q=\DirectSumGroup{I}{A}$.
    $C=I\times\{\carrier[R]\}$ by
        \auto{spike_free_module_carrier_family}.
    $z=I\times\{\ringZero[R]\}$ by
        \auto{spike_free_module_zero_section}.
    $\carrier[D]=\carrier[Q]$ by
        \auto{spike_direct_sum_module_fields}.
    $\carrier[Q]=\dependentFinitelySupportedFunctions{I}{C}{z}$ by
        \auto{spike_direct_sum_group_carrier}.
    $\ringZero[R]\in\carrier[R]$ by \auto{ring_zero_type}.
    $\dependentFinitelySupportedFunctions
        {I}{I\times\{\carrier[R]\}}{I\times\{\ringZero[R]\}}
        =\finitelySupportedFunctions
            {I}{\carrier[R]}{\ringZero[R]}$ by
        \auto{spike_constant_family_finite_support_space}.
    Follows by \auto{spike_free_module}.
\end{proof}

\begin{definition}\label{spike_free_basis_vector}
    $\freeBasisVector{R}{I}{i}
        =\apply{\directSumModuleInclusion
            {R}{I}{\spikeRegularModuleFamily{R}{I}}{i}}{
                \ringOne[R]}$.
\end{definition}

\begin{proposition}\label{spike_free_basis_vector_type}
    Let $R$ be a ring and $i\in I$.
    Then $\freeBasisVector{R}{I}{i}\in\carrier[\FreeModule{R}{I}]$.
\end{proposition}
\begin{proof}
    Let $F=\spikeRegularModuleFamily{R}{I}$.
    $F$ is a function on $I$ by
        \auto{spike_regular_module_family_type,
            funs_is_function,funs}.
    For every $j\in I$ we have $F(j)$ is a left module over $R$ by
        \auto{spike_regular_module_family_apply,
            spike_left_regular_module_is_exact}.
    $\directSumModuleInclusion{R}{I}{F}{i}\in
        \funs{\carrier[F(i)]}{\carrier[\DirectSumModule{R}{I}{F}]}$ by
        \auto{spike_direct_sum_module_inclusion_type}.
    $\carrier[F(i)]=\carrier[R]$ by
        \auto{spike_regular_module_family_apply,
            spike_left_regular_module_fields,
            spike_additive_group_fields}.
    $\ringOne[R]\in\carrier[F(i)]$ by \auto{ring_one_type}.
    $\apply{\directSumModuleInclusion{R}{I}{F}{i}}{\ringOne[R]}
        \in\carrier[\DirectSumModule{R}{I}{F}]$ by
        \auto{funs_type_apply}.
    Follows by \auto{spike_free_basis_vector,spike_free_module}.
\end{proof}

\begin{definition}\label{spike_free_generator_map_op}
    $\spikeFreeGeneratorMapOp:
        \Set\to\Set\to\Set\to\Set\to\Set
        =\fun{M,h,i,a:\Set}{\scalarMul[M](a,h(i))}$.
\end{definition}

\begin{definition}\label{spike_free_generator_map}
    $\spikeFreeGeneratorMap{R}{M}{h}{i}
        =\Apply{\graphOnOp}{
            \carrier[R],\carrier[M],
            \Apply{\spikeFreeGeneratorMapOp}{M,h,i}}$.
\end{definition}

\begin{proposition}\label{spike_free_generator_map_type}
    Let $R$ be a ring.
    Let $M$ be a left module over $R$.
    Let $h\in\funs{I}{\carrier[M]}$ and $i\in I$.
    Then
        $\spikeFreeGeneratorMap{R}{M}{h}{i}
            \in\funs{\carrier[R]}{\carrier[M]}$.
\end{proposition}
\begin{proof}
    $h(i)\in\carrier[M]$ by \auto{funs_type_apply}.
    For every $a\in\carrier[R]$ we have
        $\scalarMul[M](a,h(i))\in\carrier[M]$ by
        \auto{module_scalar_type}.
    Follows by
        \auto{spike_free_generator_map,
            spike_free_generator_map_op,
            logical_graph_on_function}.
\end{proof}

\begin{proposition}\label{spike_free_generator_map_apply}
    Let $R$ be a ring.
    Let $M$ be a left module over $R$.
    Let $h\in\funs{I}{\carrier[M]}$ and $i\in I$.
    Let $a\in\carrier[R]$.
    Then
        $\apply{\spikeFreeGeneratorMap{R}{M}{h}{i}}{a}
            =\scalarMul[M](a,h(i))$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{spike_free_generator_map,
            spike_free_generator_map_op,
            logical_graph_on_apply}.
\end{proof}

\begin{theorem}\label{spike_free_generator_map_linear}
    Let $R$ be a ring.
    Let $M$ be a left module over $R$.
    Let $h\in\funs{I}{\carrier[M]}$ and $i\in I$.
    Then $\spikeFreeGeneratorMap{R}{M}{h}{i}$ is a linear map from
        $\LeftRegularModule{R}$ to $M$ over $R$.
\end{theorem}
\begin{proof}
    Let $L=\LeftRegularModule{R}$.
    Let $f=\spikeFreeGeneratorMap{R}{M}{h}{i}$.
    $L$ is a left module over $R$ by
        \auto{spike_left_regular_module_is_exact}.
    $\carrier[L]=\carrier[R]$ and
        $\mul[L]=\ringAdd[R]$ and
        $\scalarMul[L]=\ringMul[R]$ by
        \auto{spike_left_regular_module_fields,
            spike_additive_group_fields}.
    $f\in\funs{\carrier[L]}{\carrier[M]}$ by
        \auto{spike_free_generator_map_type}.
    $h(i)\in\carrier[M]$ by \auto{funs_type_apply}.
    Show that for all $a,b\in\carrier[L]$ we have
        $f(\mul[L](a,b))=\mul[M](f(a),f(b))$.
    \begin{subproof}
        Fix $a,b\in\carrier[L]$.
        Hence $a,b\in\carrier[R]$ by \auto.
        $\ringAdd[R](a,b)\in\carrier[R]$ by \auto{ring_add_type}.
        $f(\mul[L](a,b))
            =\scalarMul[M](\ringAdd[R](a,b),h(i))$ by
            \auto{spike_free_generator_map_apply}.
        $f(a)=\scalarMul[M](a,h(i))$ and
            $f(b)=\scalarMul[M](b,h(i))$ by
            \auto{spike_free_generator_map_apply}.
        Follows by \auto{module_scalar_add_distributive}.
    \end{subproof}
    Show that for all $a,b$ such that $a\in\carrier[R]$ and
        $b\in\carrier[L]$ we have
        $f(\scalarMul[L](a,b))=\scalarMul[M](a,f(b))$.
    \begin{subproof}
        Fix $a,b$.
        Assume $a\in\carrier[R]$ and $b\in\carrier[L]$.
        Hence $b\in\carrier[R]$ by \auto.
        $\ringMul[R](a,b)\in\carrier[R]$ by \auto{ring_mul_type}.
        $f(\scalarMul[L](a,b))
            =\scalarMul[M](\ringMul[R](a,b),h(i))$ by
            \auto{spike_free_generator_map_apply}.
        $f(b)=\scalarMul[M](b,h(i))$ by
            \auto{spike_free_generator_map_apply}.
        Follows by \auto{module_scalar_associative}.
    \end{subproof}
    Follows by \auto{linear_map}.
\end{proof}

\begin{definition}\label{spike_free_generator_family_op}
    $\spikeFreeGeneratorFamilyOp:
        \Set\to\Set\to\Set\to\Set\to\Set
        =\fun{R,M,h,i:\Set}{\spikeFreeGeneratorMap{R}{M}{h}{i}}$.
\end{definition}

\begin{definition}\label{spike_free_generator_family}
    $\spikeFreeGeneratorFamily{R}{I}{M}{h}
        =\Apply{\graphOnDomainOp}{
            I,\Apply{\spikeFreeGeneratorFamilyOp}{R,M,h}}$.
\end{definition}

\begin{proposition}\label{spike_free_generator_family_type}
    $\spikeFreeGeneratorFamily{R}{I}{M}{h}$ is a function on $I$.
\end{proposition}
\begin{proof}
    $\spikeFreeGeneratorFamily{R}{I}{M}{h}\in
        \funs{I}{\ran{\spikeFreeGeneratorFamily{R}{I}{M}{h}}}$ by
        \auto{spike_free_generator_family,
            logical_graph_on_domain_function}.
    Follows by \auto{funs_is_function,funs}.
\end{proof}

\begin{proposition}\label{spike_free_generator_family_apply}
    Let $i\in I$.
    Then
        $\apply{\spikeFreeGeneratorFamily{R}{I}{M}{h}}{i}
            =\spikeFreeGeneratorMap{R}{M}{h}{i}$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{spike_free_generator_family,
            spike_free_generator_family_op,
            logical_graph_on_domain_apply}.
\end{proof}

\begin{definition}\label{spike_free_module_lift}
    $\freeModuleLift{R}{I}{M}{h}
        =\directSumModuleMediator
            {R}{I}{\spikeRegularModuleFamily{R}{I}}{M}{
                \spikeFreeGeneratorFamily{R}{I}{M}{h}}$.
\end{definition}

\begin{proposition}\label{spike_free_module_lift_type}
    Let $R$ be a ring.
    Let $M$ be a left module over $R$.
    Let $h\in\funs{I}{\carrier[M]}$.
    Then
        $\freeModuleLift{R}{I}{M}{h}\in
            \funs{\carrier[\FreeModule{R}{I}]}{\carrier[M]}$.
\end{proposition}
\begin{proof}
    Let $F=\spikeRegularModuleFamily{R}{I}$.
    Let $g=\spikeFreeGeneratorFamily{R}{I}{M}{h}$.
    $F$ is a function on $I$ by
        \auto{spike_regular_module_family_type,
            funs_is_function,funs}.
    For every $i\in I$ we have $F(i)$ is a left module over $R$ by
        \auto{spike_regular_module_family_apply,
            spike_left_regular_module_is_exact}.
    $g$ is a function on $I$ by
        \auto{spike_free_generator_family_type}.
    Show that for every $i\in I$ we have
        $g(i)$ is a linear map from $F(i)$ to $M$ over $R$.
    \begin{subproof}
        Fix $i\in I$.
        $g(i)=\spikeFreeGeneratorMap{R}{M}{h}{i}$ by
            \auto{spike_free_generator_family_apply}.
        $F(i)=\LeftRegularModule{R}$ by
            \auto{spike_regular_module_family_apply}.
        Follows by \auto{spike_free_generator_map_linear}.
    \end{subproof}
    $\directSumModuleMediator{R}{I}{F}{M}{g}\in
        \funs{\carrier[\DirectSumModule{R}{I}{F}]}{\carrier[M]}$ by
        \auto{spike_direct_sum_module_mediator_type}.
    Follows by \auto{spike_free_module_lift,spike_free_module}.
\end{proof}

\begin{theorem}\label{spike_free_module_lift_apply}
    Let $R$ be a ring.
    Let $M$ be a left module over $R$.
    Let $h\in\funs{I}{\carrier[M]}$.
    Let $x\in\carrier[\FreeModule{R}{I}]$.
    Let $F=\spikeRegularModuleFamily{R}{I}$.
    Let $A=\spikeModuleAdditiveGroupFamily{I}{F}$.
    Let $g=\spikeFreeGeneratorFamily{R}{I}{M}{h}$.
    Then
        $\apply{\freeModuleLift{R}{I}{M}{h}}{x}
            =\finiteFold{\ModuleAdditiveGroup{M}}{
                \dependentSupport{I}{x}{
                    \spikeProductGroupNeutral{I}{A}}}{
                \spikeDirectSumGroupContribution
                    {I}{A}{\ModuleAdditiveGroup{M}}{g}{x}}$.
\end{theorem}
\begin{proof}
    $F$ is a function on $I$ by
        \auto{spike_regular_module_family_type,
            funs_is_function,funs}.
    For every $i\in I$ we have $F(i)$ is a left module over $R$ by
        \auto{spike_regular_module_family_apply,
            spike_left_regular_module_is_exact}.
    $g$ is a function on $I$ by
        \auto{spike_free_generator_family_type}.
    Show that for every $i\in I$ we have
        $g(i)$ is a linear map from $F(i)$ to $M$ over $R$.
    \begin{subproof}
        Fix $i\in I$.
        $g(i)=\spikeFreeGeneratorMap{R}{M}{h}{i}$ by
            \auto{spike_free_generator_family_apply}.
        $F(i)=\LeftRegularModule{R}$ by
            \auto{spike_regular_module_family_apply}.
        Follows by \auto{spike_free_generator_map_linear}.
    \end{subproof}
    $\FreeModule{R}{I}=\DirectSumModule{R}{I}{F}$ by
        \auto{spike_free_module}.
    Hence $x\in\carrier[\DirectSumModule{R}{I}{F}]$ by \auto.
    Follows by
        \auto{spike_free_module_lift,
            spike_direct_sum_module_mediator_apply}.
\end{proof}
