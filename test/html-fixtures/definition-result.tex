\begin{definition}\label{html_special}
    $X$ is special iff $X=X$.
\end{definition}

\begin{definition}\label{html_notable}
    $X$ is notable iff $X=X$.
\end{definition}

\begin{definition}\label{html_clone}
    Let $X$ be a set.
    $\clone{X}=X$, which is a special notable set.
\end{definition}
