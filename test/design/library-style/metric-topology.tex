\begin{definition}\label{style_metric_topology}
    Let $M$ be a metric space. Then
    $\metricTopology{M}=
        \structure[topological space]{
            \carrier=\carrier[M],
            \opens=\metricOpens[M]
        }$.
\end{definition}

\begin{theorem}\label{style_metric_topology_is_topological}
    Let $M$ be a metric space. Then $\metricTopology{M}$ is a topological
    space.
\end{theorem}
