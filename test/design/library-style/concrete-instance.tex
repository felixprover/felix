\begin{instance}\label{style_two_point_indiscrete_space}
    The topological space
    $\TwoPointIndiscrete=
        \structure[topological space]{
            \carrier=\TwoPoint,
            \opens=\{\emptyset,\TwoPoint\}
        }$.
\end{instance}
\begin{proof}
    The empty set and the carrier are open. The displayed family is closed
    under arbitrary unions and finite intersections.
\end{proof}
