\begin{definition}\label{notion_style_compact}
    A topological space $X$ is compact iff for every $C$ such that
        $C\subseteq\opens[X]$ and $C$ covers $\carrier[X]$ there exists
        a finite set $D$ such that $D\subseteq C$ and
        $D$ covers $\carrier[X]$.
\end{definition}

% Accepted: the complete structure phrase supplies the core notion to its
% modifier.
Let $X$ be a compact topological space.

% Rejected: no topological-space evidence is active.
$Y$ is compact.

% Each coordinated modifier is checked against the same core and outer
% context.  One modifier never licenses its sibling.
Let $Z$ be a compact and finite topological space.
