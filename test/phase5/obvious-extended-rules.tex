\begin{proposition}\label{explicit_conjunctive_cases}
    Let $a,P,Q,R,S$ be sets.
    Suppose $a\in P$ or $a\in Q$.
    Suppose $a\in R$.
    Suppose for all $x$ we have $(x\in P\land x\in R)\implies x\in S$.
    Suppose for all $x$ we have $(x\in Q\land x\in R)\implies x\in S$.
    Then $a\in S$.
\end{proposition}
\begin{proof}
    \begin{byCase}
    \caseOf{$a\in P$.}
        Thus $a\in S$.
    \caseOf{$a\in Q$.}
        Thus $a\in S$.
    \end{byCase}
\end{proof}

\begin{proposition}\label{obvious_conjunctive_cases}
    Let $a,P,Q,R,S$ be sets.
    Suppose $a\in P$ or $a\in Q$.
    Suppose $a\in R$.
    Suppose for all $x$ we have $(x\in P\land x\in R)\implies x\in S$.
    Suppose for all $x$ we have $(x\in Q\land x\in R)\implies x\in S$.
    Then $a\in S$.
\end{proposition}
\begin{proof}
    Follows.
\end{proof}

\begin{proposition}\label{obvious_projected_cases}
    Let $a,P,Q,R,S$ be sets.
    Suppose $a\in P$ or $a\in Q$.
    Suppose $a\in R$.
    Suppose for all $x$ we have
        $((x\in P\land x\in R)\implies x\in S)\land ((x\in Q\land x\in R)\implies x\in S)$.
    Then $a\in S$.
\end{proposition}
\begin{proof}
    Follows.
\end{proof}
