\begin{datatype}\label{formation_data}
    Define $\formationdata{A}$ inductively as follows.
    \begin{enumerate}
        \item[Falsity] $\formationzero\in\formationdata{A}$.
        \item[Atom] $\formationatom{x}\in\formationdata{A}$ for $x\in A$.
        \item[Implication] $\formationjoin{x}{y}\in\formationdata{A}$ for
            $x\in\formationdata{A}$ and $y\in\formationdata{A}$.
    \end{enumerate}
\end{datatype}
