\section{Structural recursion on formulas}

\begin{recursive-definition}[Occurring atoms]\label{pl_atoms}
    Define $\plAtoms{A}{p}$ recursively for $p\in\plform{A}$ as follows.
    \begin{enumerate}
        \item[Falsity] $\plAtoms{A}{\propbot}=\emptyset$.
        \item[Atom] $\plAtoms{A}{\propvar{a}}=\{a\}$.
        \item[Implication] $\plAtoms{A}{p\propto q}=\plAtoms{A}{p}\union\plAtoms{A}{q}$.
    \end{enumerate}
\end{recursive-definition}

\begin{recursive-definition}[Reconstruction]\label{pl_rebuild}
    Define $\plRebuild{A}{p}$ recursively for $p\in\plform{A}$ as follows.
    \begin{enumerate}
        \item[Falsity] $\plRebuild{A}{\propbot}=\propbot$.
        \item[Atom] $\plRebuild{A}{\propvar{a}}=\propvar{a}$.
        \item[Implication] $\plRebuild{A}{p\propto q}=\plRebuild{A}{p}\propto\plRebuild{A}{q}$.
    \end{enumerate}
\end{recursive-definition}

\begin{recursive-definition}[Constant recursion]\label{pl_constant}
    Define $\plConstant{A}{X}{p}$ recursively for $p\in\plform{A}$ as follows.
    \begin{enumerate}
        \item[Falsity] $\plConstant{A}{X}{\propbot}=X$.
        \item[Atom] $\plConstant{A}{X}{\propvar{a}}=X$.
        \item[Implication] $\plConstant{A}{X}{p\propto q}=X$.
    \end{enumerate}
\end{recursive-definition}

\begin{recursive-definition}[Satisfaction]\label{pl_satisfaction}
    Define $\plSat{A}{V}{p}$ recursively for $p\in\plform{A}$ as follows.
    \begin{enumerate}
        \item[Falsity] $\plSat{A}{V}{\propbot}$ iff $\bot$.
        \item[Atom] $\plSat{A}{V}{\propvar{a}}$ iff $a\in V$.
        \item[Implication] $\plSat{A}{V}{p\propto q}$ iff
            if $\plSat{A}{V}{p}$ then $\plSat{A}{V}{q}$.
    \end{enumerate}
\end{recursive-definition}
