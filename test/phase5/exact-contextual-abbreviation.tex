\begin{signature}\label{phase5_context_apply}
    $\apply{f}{x}$ is a set.
\end{signature}

\begin{struct}\label{phase5_context_magma}
    A phase five context magma $A$ is a onesorted structure equipped with
    \begin{enumerate}
        \item $\phasefivecombine$;
        \item $\phasefivezero$;
        \item $\phasefiveneg$;
    \end{enumerate}
    such that
    \begin{enumerate}
        \item\label{phase5_context_refl} $A = A$.
    \end{enumerate}
\end{struct}

\begin{struct}\label{phase5_context_unital_magma}
    A phase five context unital magma $A$ is a phase five context magma.
\end{struct}

\begin{abbreviation}\label{phase5_context_dot}
    %! infixl 4
    $a\phasefivedot b = \phasefivecombine(a,b)$.
\end{abbreviation}

\begin{abbreviation}\label{phase5_context_indexed_dot}
    %! infixl 4
    $a\phasefivedot_{A} b = \phasefivecombine[A](a,b)$.
\end{abbreviation}

\begin{abbreviation}\label{phase5_context_indexed_plus}
    %! infixl 1
    $a+_{A}b = \phasefivecombine[A](a,b)$.
\end{abbreviation}

\begin{abbreviation}\label{phase5_context_indexed_zero}
    $0_{A} = \phasefivezero[A]$.
\end{abbreviation}

\begin{abbreviation}\label{phase5_context_indexed_negation}
    $-_{A}a = \phasefiveneg[A](a)$.
\end{abbreviation}

\begin{abbreviation}\label{phase5_context_commutes}
    $a$ commutes with $b$ in phase five iff
    $a\phasefivedot b = b\phasefivedot a$.
\end{abbreviation}

\begin{proposition}\label{phase5_context_dot_explicit}
    Let $A$ be a phase five context magma.
    Let $a,b$ be sets.
    Then $a\phasefivedot b = \phasefivecombine[A](a,b)$.
\end{proposition}

\begin{proposition}\label{phase5_context_indexed_dot_explicit}

    Let $a,b$ be sets.
    Then $a\phasefivedot_{A} b = \phasefivecombine[A](a,b)$.
\end{proposition}

\begin{proposition}\label{phase5_context_indexed_plus_explicit}
    Let $A$ be a phase five context magma.
    Let $a,b$ be sets.
    Then $a+_{A}b = \phasefivecombine[A](a,b)$.
\end{proposition}

\begin{proposition}\label{phase5_context_indexed_zero_explicit}
    Let $A$ be a phase five context magma.
    Then $0_{A} = \phasefivezero[A]$.
\end{proposition}

\begin{proposition}\label{phase5_context_indexed_negation_explicit}
    Let $A$ be a phase five context magma.
    Let $a$ be a set.
    Then $-_{A}a = \phasefiveneg[A](a)$.
\end{proposition}

\begin{proposition}\label{phase5_context_inherited}
    Let $A$ be a phase five context unital magma.
    Let $a,b$ be sets.
    Then $a\phasefivedot b = \phasefivecombine[A](a,b)$.
\end{proposition}

\begin{proposition}\label{phase5_context_nested}
    Let $A$ be a phase five context magma.
    Let $a,b$ be sets.
    Then $a$ commutes with $b$ in phase five iff
        $\phasefivecombine[A](a,b) = \phasefivecombine[A](b,a)$.
\end{proposition}

\begin{proposition}\label{phase5_context_explicit_unique}
    Let $A,a,b$ be sets.
    Then $\phasefivecombine[A](a,b) = \phasefivecombine[A](a,b)$.
\end{proposition}
