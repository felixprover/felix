\begin{proposition}\label{explicit_universal_disjunction}
    Let $a,P,Q$ be sets.
    Suppose for all $x$ we have $x\in P\lor x\in Q$.
    Suppose $a\notin P$.
    Then $a\in Q$.
\end{proposition}
\begin{proof}
    $a\in P$ or $a\in Q$.
    Follows.
\end{proof}

\begin{proposition}\label{obvious_universal_disjunction}
    Let $a,P,Q$ be sets.
    Suppose for all $x$ we have $x\in P\lor x\in Q$.
    Suppose $a\notin P$.
    Then $a\in Q$.
\end{proposition}
\begin{proof}
    Follows.
\end{proof}

\begin{proposition}\label{obvious_partial_universal_conjunction}
    Let $a,P,Q,R$ be sets.
    Suppose for all $x$ we have $x\in R\land(x\in P\lor x\in Q)$.
    Suppose $a\notin P$.
    Then $a\in Q$.
\end{proposition}
\begin{proof}
    Follows.
\end{proof}

\begin{proposition}\label{obvious_nested_universal_disjunction}
    Let $a,b,P,Q$ be sets.
    Suppose for all $x,y$ we have $x\in P\lor y\in Q$.
    Suppose $a\notin P$.
    Then $b\in Q$.
\end{proposition}
\begin{proof}
    Follows.
\end{proof}
