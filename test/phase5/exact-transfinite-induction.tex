\begin{definition}\label{phase5_ordinal_guard}
    $x$ is a phase five ordinal iff $x=x$.
\end{definition}

\begin{axiom}\label{phase5_ordinal_guard_hereditary}
    Let $\alpha$ be a phase five ordinal.
    Suppose $\beta\in\alpha$.
    Then $\beta$ is a phase five ordinal.
\end{axiom}

\begin{proposition}\label{phase5_transfinite_induction}
    For every phase five ordinal $\alpha$ we have $\alpha=\alpha$ or $\alpha\neq\alpha$.
\end{proposition}
\begin{proof}[Proof by transfinite induction]
    Follows by \auto.
\end{proof}
