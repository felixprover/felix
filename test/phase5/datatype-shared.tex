\begin{datatype}\label{shared_first}
    Define $\sharedfirst$ inductively as follows.
    \begin{enumerate}
        \item $\sharedzero \in \sharedfirst$.
        \item $\sharedstep{x} \in \sharedfirst$ for $x \in \sharedfirst$.
    \end{enumerate}
\end{datatype}

\begin{datatype}\label{shared_second}
    Define $\sharedsecond$ inductively as follows.
    \begin{enumerate}
        \item $\anotherzero \in \sharedsecond$.
        \item $\anotherstep{x} \in \sharedsecond$ for $x \in \sharedsecond$.
    \end{enumerate}
\end{datatype}
