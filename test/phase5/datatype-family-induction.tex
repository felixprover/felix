\begin{proposition}\label{family_induction}
    Then for every $p\in\familydata{A}{B}$ we have $p=p$.
\end{proposition}
\begin{proof}[Proof by structural induction on $p$]
    \begin{byCase}
        \caseOf{Familyzero.}
            Fix $w\in B$.
            Follows.
        \caseOf{Familyatom.}
            Fix $a\in A$.
            Follows.
        \caseOf{Familyjoin.}
            Fix $q,r\in\familydata{A}{B}$.
            Fix $z\in B$.
            Suppose $q=q$ and $r=r$.
            Follows.
        \caseOf{Familyrich.}
            Fix $z\in\{(x,y)\mid x\in A,y\in B\}$.
            Follows.
    \end{byCase}
\end{proof}
