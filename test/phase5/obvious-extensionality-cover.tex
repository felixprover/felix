\begin{proposition}\label{cover_extensionality}
    Let $A,B,U,V$ be sets.
    Suppose for all $x$ we have if $x\in U$, then $x\in A$ or $x\in B$.
    Suppose for all $x$ we have if $x\in A$, then $x\in V$.
    Suppose for all $x$ we have if $x\in B$, then $x\in V$.
    Suppose for all $x$ we have if $x\in V$, then $x\in U$.
    Then $U=V$.
\end{proposition}
\begin{proof}
    Follows by \obv{setext}.
\end{proof}
