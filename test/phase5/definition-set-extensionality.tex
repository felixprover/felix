\begin{definition}\label{phase5_contained}
    $A$ is phase five contained in $B$ iff
    for every $x\in A$ we have $x\in B$.
\end{definition}

\begin{abbreviation}\label{phase5_nested}
    $A$ is phase five nested in $B$ iff
    $A$ is phase five contained in $B$.
\end{abbreviation}

\begin{definition}\label{phase5_coextensional}
    $A$ is phase five coextensional with $B$ iff
    for every $x$ we have $x\in A$ iff $x\in B$.
\end{definition}

\begin{proposition}\label{definition_set_extensionality}
    Let $A,B$ be sets.
    Suppose $A$ is phase five contained in $B$.
    Suppose $B$ is phase five contained in $A$.
    Then $A=B$.
\end{proposition}
\begin{proof}
    Follows by set extensionality.
\end{proof}

\begin{proposition}\label{mixed_definition_set_extensionality}
    Let $A,B$ be sets.
    Suppose for every $x\in A$ we have $x\in B$.
    Suppose $B$ is phase five contained in $A$.
    Then $A=B$.
\end{proposition}
\begin{proof}
    Follows by set extensionality.
\end{proof}

\begin{proposition}\label{definition_iff_set_extensionality}
    Let $A,B$ be sets.
    Suppose $A$ is phase five coextensional with $B$.
    Then $A=B$.
\end{proposition}
\begin{proof}
    Follows by set extensionality.
\end{proof}

\begin{proposition}\label{chained_definition_set_extensionality}
    Let $A,B$ be sets.
    Suppose $A$ is phase five nested in $B$.
    Suppose $B$ is phase five nested in $A$.
    Then $A=B$.
\end{proposition}
\begin{proof}
    Follows by set extensionality.
\end{proof}
