\begin{proposition}\label{exact_record_extensionality}
    Let $A,B$ be sets.
    Assume $A$ is an exact matrix left.
    Assume $B$ is an exact matrix left.
    Assume $\carrier[A]=\carrier[B]$.
    Assume $\matrixPoint[A]=\matrixPoint[B]$.
    Then $A=B$.
\end{proposition}
\begin{proof}
    Follows by structure extensionality.
\end{proof}

\begin{proposition}\label{generic_record_extensionality}
    Let $A,B$ be sets.
    Assume $A$ is an exact matrix left.
    Assume $B$ is an exact matrix left.
    Assume $\carrier[A]=\carrier[B]$.
    Assume $\matrixPoint[A]=\matrixPoint[B]$.
    Then $A=B$.
\end{proposition}
\begin{proof}
    Follows by extensionality.
\end{proof}

\begin{proposition}\label{generic_record_falls_back_to_set}
    Let $A,B$ be sets.
    Assume $A$ is an exact matrix left.
    Assume $B$ is an exact matrix left.
    Suppose for all $x\in A$ we have $x\in B$.
    Suppose for all $x\in B$ we have $x\in A$.
    Then $A=B$.
\end{proposition}
\begin{proof}
    Follows by extensionality.
\end{proof}
