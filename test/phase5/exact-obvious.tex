\begin{proposition}\label{obvious_local_definition}
    For all $A$ we have $A=A$.
\end{proposition}
\begin{proof}
    Fix $A$.
    Let $B=A$.
    Then $B=A$ by definition.
    Follows.
\end{proof}

\begin{axiom}\label{obvious_rule}
    Let $A$ be a set.
    If $A = \emptyset$, then $A \in \emptyset$.
\end{axiom}

\begin{axiom}\label{obvious_irrelevant}
    $\emptyset = \emptyset$.
\end{axiom}

\begin{proposition}\label{obvious_application}
    $\emptyset \in \emptyset$.
\end{proposition}
\begin{proof}
    Follows by \obv{obvious_irrelevant,obvious_rule}.
\end{proof}

\begin{axiom}\label{obvious_transport_conjunction}
    $\emptyset=\emptyset\land\emptyset\in\emptyset$.
\end{axiom}

\begin{axiom}\label{obvious_transport_equality}
    $\emptyset=(\emptyset,\emptyset)$.
\end{axiom}

\begin{proposition}\label{obvious_projected_transport}
    $(\emptyset,\emptyset)\in\emptyset$.
\end{proposition}
\begin{proof}
    Follows by
        \obv{obvious_transport_conjunction,obvious_transport_equality}.
\end{proof}

\begin{proposition}\label{obvious_branch_rule_fixed_point}
    Let $A,B$ be sets.
    Suppose $A=\emptyset$ or $B=\emptyset$.
    Suppose $A=\emptyset\implies A=B$.
    Suppose $B=\emptyset\implies A=B$.
    Then $A=B$.
\end{proposition}
\begin{proof}
    Follows.
\end{proof}
