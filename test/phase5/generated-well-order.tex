\begin{proposition}\label{test_empty_graph_not_generated_one_point}
    $\emptyset$ is not a well ordering generated by $\emptyset$ on
        $\{\emptyset\}$.
\end{proposition}
\begin{proof}
    Assume $\emptyset$ is a well ordering generated by $\emptyset$ on
        $\{\emptyset\}$.
    Then $(\downward{\emptyset}{\emptyset},\emptyset)\in\emptyset$
        by \auto{generated_well_ordering_entry,singleton_iff}.
    Contradiction by \auto{emptyset}.
\end{proof}

\begin{proposition}\label{test_one_entry_graph_generated_one_point}
    Let $f=\{\emptyset\}\times\{\emptyset\}$.
    Then $\emptyset$ is a well ordering generated by $f$ on
        $\{\emptyset\}$.
\end{proposition}
\begin{proof}
    $f\in\funs{\{\emptyset\}}{\{\emptyset\}}$
        by \auto{constant_graph_type}.
    Hence $f$ is a function by \auto{funs_is_function}.

    $\emptyset$ is a binary relation on $\{\emptyset\}$
        by \auto{emptyset_subseteq}.
    $\emptyset$ is a strict order
        by \auto{relation,transitive,asymmetric,emptyset}.
    $\emptyset$ is connex on $\{\emptyset\}$
        by \auto{connex,singleton_iff}.
    Hence $\emptyset$ is a strict linear order on $\{\emptyset\}$
        by \auto{strict_linear_order}.
    For every $X$ such that $X$ is inhabited and
        $X\subseteq\{\emptyset\}$ we have $X=\{\emptyset\}$
        by \auto{setext,subseteq,singleton_iff}.
    Thus for all $X$ such that $X$ is inhabited and
        $X\subseteq\{\emptyset\}$ there exists $a\in X$ such that
        $a$ is a least element of $X$ with respect to $\emptyset$
        by \auto{singleton_least_element,singleton_iff}.
    Hence $\emptyset$ is a strict well order on $\{\emptyset\}$
        by \auto{strict_well_order}.
    $\downward{\emptyset}{\emptyset}=\emptyset$
        by \auto{setext,downward_closure_iff,emptyset}.
    $(\downward{\emptyset}{\emptyset},\emptyset)\in f$
        by \auto{times_tuple_intro,singleton_iff}.
    Follows by \auto{generated_well_ordering,singleton_iff}.
\end{proof}

\begin{proposition}\label{test_empty_graph_generated_empty_order}
    $\emptyset$ is a well ordering generated by $\emptyset$ on $\emptyset$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{empty_generated_well_ordering,emptyset_is_function}.
\end{proof}

\begin{proposition}\label{test_nonfunction_not_generated_empty_order}
    Let $f=\cons{(\emptyset,\{\emptyset\})}
        {\{(\emptyset,\emptyset)\}}$.
    Then $\emptyset$ is not a well ordering generated by $f$ on $\emptyset$.
\end{proposition}
\begin{proof}
    Assume $\emptyset$ is a well ordering generated by $f$ on $\emptyset$.
    Then $f$ is a function by \obv{generated_well_ordering}.
    $(\emptyset,\emptyset),(\emptyset,\{\emptyset\})\in f$
        by \auto{cons_iff,singleton_iff}.
    Thus $\emptyset=\{\emptyset\}$ by \auto{function_rightunique}.
    $\emptyset\neq\{\emptyset\}$ by \auto{emptyset,singleton_iff}.
    Contradiction.
\end{proof}

\begin{proposition}\label{test_generated_well_order_compatibility}
    Suppose $R$ is a well ordering generated by $f$ on $B$.
    Suppose $S$ is a well ordering generated by $f$ on $C$.
    Then $B$ is a common initial segment of $R$ on $B$ and $S$ on $C$
    or $C$ is a common initial segment of $R$ on $B$ and $S$ on $C$.
\end{proposition}
\begin{proof}
    Follows by \auto{generated_well_orderings_compatible}.
\end{proof}
