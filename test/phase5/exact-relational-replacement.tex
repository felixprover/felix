    For every set $A$ we have $A = A$.
\end{axiom}

\begin{definition}\label{phase5_relational_replacement_definition}
    $\phasefiverelational{A} = \{ y \mid \exists x\in A. y = x \}$.
\end{definition}

    For every set $A$ we have $A = A$.
\end{proposition}

\end{proof}
