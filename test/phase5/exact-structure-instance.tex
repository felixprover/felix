\begin{struct}\label{instance_syntax_structure}
    An instance syntax structure $X$ is a onesorted structure equipped with
    \begin{enumerate}
        \item\label{instance_syntax_point}
            a point $\instancePoint[X]\in\carrier[X]$;
    \end{enumerate}
    .
\end{struct}

\begin{instance}\label{instance_equation_surface}
    The instance syntax structure
    $\InstanceEquation=\structure{\{\emptyset\},\emptyset}$.
\end{instance}
\begin{proof}
    Follows by
    \auto{instance_syntax_structure,instance_syntax_structureaggregateexact,instance_syntax_structureaggregateprojection_carrier,instance_syntax_structureaggregateprojection_instancePoint}.
\end{proof}

\begin{struct}\label{instance_parameter_base}
    An instance parameter base $R$ is a onesorted structure such that
    \begin{enumerate}
        \item\label{instance_parameter_base_refl} $R=R$.
    \end{enumerate}
\end{struct}

\begin{instance}\label{instance_parameter_object}
    The instance parameter base
    $\InstanceParameter=\structure{\{\emptyset\}}$.
\end{instance}
\begin{proof}
    Follows by
    \auto{instance_parameter_base,instance_parameter_baseaggregateexact,instance_parameter_baseaggregateprojection_carrier}.
\end{proof}

\begin{struct}\label{instance_parameterized_child}
    Let $R$ be an instance parameter base.
    An instance parameterized child $X$ over $R$ is an instance syntax
    structure such that
    \begin{enumerate}
        \item\label{instance_parameterized_child_refl} $R=R$.
    \end{enumerate}
\end{struct}

\begin{instance}\label{instance_parameterized_object}
    The instance parameterized child $\InstanceParameterized$ over
    $\InstanceParameter$ is
    $\structure{\{\emptyset\},\emptyset}$.
\end{instance}
\begin{proof}
    We have $\InstanceParameterized$ is an instance syntax structure by
    \auto{instance_syntax_structure,instance_syntax_structureaggregateexact,instance_syntax_structureaggregateprojection_carrier,instance_syntax_structureaggregateprojection_instancePoint}.
    Follows by
    \obv{instance_parameterized_child,instance_parameter_object}.
\end{proof}

\begin{proposition}\label{instance_parameterized_parent_available}
    $\InstanceParameterized$ is an instance syntax structure.
\end{proposition}
\begin{proof}
    Follows by
    \auto{instance_parameterized_object,instance_parameterized_child}.
\end{proof}

\begin{proposition}\label{instance_parameter_is_not_stored}
    $\structure[instance parameterized child over \InstanceParameter]
        {\{\emptyset\},\emptyset}
    =\structure[instance syntax structure]
        {\{\emptyset\},\emptyset}$.
\end{proposition}
