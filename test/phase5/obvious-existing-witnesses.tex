\begin{proposition}\label{obvious_set_conditional_true}
    $\forall p:\Prop.\ \forall a,b:\Set.
        p\implies (\exists x:\Set.
            (p\land x=a)\lor(\lnot p\land x=b))$.
\end{proposition}
\begin{proof}
    Fix $p,a,b$.
    Assume $p$.
    Follows.
\end{proof}

\begin{proposition}\label{obvious_set_conditional_false}
    $\forall p:\Prop.\ \forall a,b:\Set.
        \lnot p\implies (\exists x:\Set.
            (p\land x=a)\lor(\lnot p\land x=b))$.
\end{proposition}
\begin{proof}
    Fix $p,a,b$.
    Assume $\lnot p$.
    Follows.
\end{proof}
