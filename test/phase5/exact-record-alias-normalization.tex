\begin{proposition}\label{exact_record_alias_normalization}
    For all $A,B$ we have $A=A$.
\end{proposition}
\begin{proof}
    Fix $A$ and $B$.
    Let $P=A$.
    Let $R=P$.
    Let $Q=B$.
    Let $S=Q$.
    Let $U=\emptyset$.
    Show if $\pow{R}$ is an exact matrix left, then if
        $\pow{S}$ is an exact matrix left, then if
        $\carrier[\pow{R}]=\carrier[\pow{S}]$, then if
        $\matrixPoint[\pow{R}]=\matrixPoint[\pow{S}]$, then
        $\pow{A}=\pow{B}$.
    \begin{subproof}
        Assume $\pow{R}$ is an exact matrix left.
        Assume $\pow{S}$ is an exact matrix left.
        Assume $\carrier[\pow{R}]=\carrier[\pow{S}]$.
        Assume $\matrixPoint[\pow{R}]=\matrixPoint[\pow{S}]$.
        Follows by structure extensionality.
    \end{subproof}
    Follows.
\end{proof}
