\begin{proposition}\label{incomplete_record_extensionality}
    Let $A,B$ be sets.
    Assume $A$ is an exact matrix left.
    Assume $B$ is an exact matrix left.
    Then $A=B$.
\end{proposition}
\begin{proof}
    Follows by structure extensionality.
\end{proof}
