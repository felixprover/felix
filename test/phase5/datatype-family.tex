\begin{datatype}\label{family_data}
    Define $\familydata{A}{B}$ inductively as follows.
    \begin{enumerate}
        \item $\familyzero\in\familydata{A}{B}$ for $w\in B$.
        \item $\familyatom{x}\in\familydata{A}{B}$ for $x\in A$.
        \item $\familyjoin{y}{x}\in\familydata{A}{B}$ for
            $x\in\familydata{A}{B}$, $y\in\familydata{A}{B}$, and $z\in B$.
        \item $\familyrich{z}\in\familydata{A}{B}$ for
            $z\in\{(x,y)\mid x\in A,y\in B\}$.
    \end{enumerate}
\end{datatype}
