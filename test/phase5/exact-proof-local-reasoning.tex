\begin{proposition}\label{suffices_exact}
    Let $A$ be a set.
    $A=A$.
\end{proposition}
\begin{proof}
    It suffices to show that $A=A\land A=A$.
    Follows.
\end{proof}

\begin{proposition}\label{calculation_equality_exact}
    Let $A$ be a set.
    Suppose $A=\emptyset\land \emptyset=\{A\}$.
    Then $A=\{A\}$.
\end{proposition}
\begin{proof}
    \begin{align*}
        A
            &= \emptyset
                \explanation{by \auto{suffices_exact}}
            \\
            &= \{A\}
    \end{align*}
    Follows.
\end{proof}

\begin{proposition}\label{calculation_biconditional_exact}
    Let $A$ be a set.
    $(A=A)\iff ((A=A)\lor(A\neq A))$.
\end{proposition}
\begin{proof}
    \begin{align*}
        A=A
            &\iff \top
                \explanation{by \auto}
                \\
            &\iff (A=A)\lor(A\neq A)
                \explanation{by \auto}
    \end{align*}
    Follows by assumption.
\end{proof}

\begin{proposition}\label{calculation_quantified_exact}
    Let $A$ be a set.
    For all $x,y\in A$ such that $x=y$ we have $x=y$.
\end{proposition}
\begin{proof}
    For all $x,y\in A$ such that $x=y$, we have
    \begin{align*}
        x
            &= y
            \\
            &= y
    \end{align*}
    Follows.
\end{proof}

\begin{proposition}\label{since_structural_exact}
    Let $A,B$ be sets.
    Suppose $A=A$.
    Then $A=A$.
\end{proposition}
\begin{proof}
    Since $A=A$ by assumption, we have $B=B$.
    Follows by assumption.
\end{proof}

\begin{proposition}\label{suffices_omitted_exact}
