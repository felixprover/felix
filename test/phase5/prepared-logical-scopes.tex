\begin{proposition}\label{prepared_logical_closed}
    Let $A$ be a set.
    Then for all $x$ we have if $x=A$, then $x=A$.
\end{proposition}
\begin{proof}
    Follows by \auto.
\end{proof}

\begin{proposition}\label{prepared_logical_residual}
    Let $A$ be a set.
    Then for all $x$ we have if $x=A$, then $x=A$ and $A=x$.
\end{proposition}
\begin{proof}
    Follows by \auto.
\end{proof}

\begin{proposition}\label{prepared_logical_nested}
    Let $A$ be a set.
    Then for all $x,y$ we have if $x=A$ and $y=x$, then $x=A$ and $A=y$.
\end{proposition}
\begin{proof}
    Follows by \auto.
\end{proof}

\begin{proposition}\label{prepared_logical_supplied_clause}
    Let $A,B$ be sets.
    Assume that for all $x$ we have if $x\in A$, then $x\in B$.
    Then for all $x$ we have if $x\in A$, then $x\in B$ and $x\in A$.
\end{proposition}
\begin{proof}
    Follows by \auto.
\end{proof}
