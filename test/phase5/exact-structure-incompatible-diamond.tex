\begin{struct}\label{incompatible_left}
    An incompatible left $X$ is a onesorted structure equipped with
    \begin{enumerate}
        \item\label{incompatible_left_point}
            a point $\incompatiblePoint[X]\in\carrier[X]$;
    \end{enumerate}
    .
\end{struct}

\begin{struct}\label{incompatible_right}
    An incompatible right $X$ is a onesorted structure equipped with
    \begin{enumerate}
        \item\label{incompatible_right_point}
            a point $\incompatiblePoint[X]\in\carrier[X]$;
    \end{enumerate}
    .
\end{struct}

\begin{struct}\label{incompatible_diamond}
    An incompatible diamond $X$ is an incompatible left and an incompatible
    right.
\end{struct}
