\begin{struct}\label{override_base}
    An override base $X$ is a onesorted structure equipped with
    \begin{enumerate}
        \item\label{override_base_point}
            a point $\overridePoint[X]\in\carrier[X]$;
    \end{enumerate}
    .
\end{struct}

\begin{struct}\label{override_child}
    An override child $X$ is an override base equipped with
    \begin{enumerate}
        \item\label{override_child_point}
            a point $\overridePoint[X]\in\carrier[X]$;
    \end{enumerate}
    .
\end{struct}
