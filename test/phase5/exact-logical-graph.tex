\import{set/logical-graph.tex}

\begin{definition}\label{phase5_logical_identity}
    $\phaseFiveLogicalIdentityOp:\Set\to\Set
        =\fun{x:\Set}{x}$.
\end{definition}

\begin{definition}\label{phase5_named_identity_graph}
    $\phaseFiveIdentityGraph{A}
        =\Apply{\Apply{\Apply{\graphOnOp}{A}}{A}}
            {\phaseFiveLogicalIdentityOp}$.
\end{definition}

\begin{proposition}\label{phase5_named_identity_graph_function}
    For every $A$ we have $\phaseFiveIdentityGraph{A}\in\funs{A}{A}$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{phase5_named_identity_graph,logical_graph_on_function,phase5_logical_identity}.
\end{proof}

\begin{proposition}\label{phase5_named_identity_graph_application}
    Suppose $x\in A$.
    Then $\apply{\phaseFiveIdentityGraph{A}}{x}=x$.
\end{proposition}
\begin{proof}
    $(x,x)\in\phaseFiveIdentityGraph{A}$ by
        \auto{phase5_named_identity_graph,logical_graph_on_membership,phase5_logical_identity}.
    Follows by
        \auto{phase5_named_identity_graph_function,funs_is_function,function_apply_intro}.
\end{proof}

\begin{proposition}\label{phase5_named_identity_graph_downstream}
    Suppose $x\in A$.
    Then $\apply{\phaseFiveIdentityGraph{A}}
        {\apply{\phaseFiveIdentityGraph{A}}{x}}=x$.
\end{proposition}
\begin{proof}
    Follows by \auto{phase5_named_identity_graph_application}.
\end{proof}

\begin{definition}\label{phase5_logical_equality}
    $\phaseFiveLogicalEqualityOp:\Set\to\Set\to\Prop
        =\fun{x:\Set}{\fun{y:\Set}{x=y}}$.
\end{definition}

\begin{definition}\label{phase5_named_diagonal_relation}
    $\phaseFiveDiagonalRelation{A}
        =\Apply{\Apply{\relationOnOp}{A}}{\phaseFiveLogicalEqualityOp}$.
\end{definition}

\begin{proposition}\label{phase5_named_diagonal_relation_membership}
    $(x,y)\in\phaseFiveDiagonalRelation{A}$ iff
        $x\in A$ and $y\in A$ and $x=y$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{phase5_named_diagonal_relation,logical_relation_on_membership,phase5_logical_equality}.
\end{proof}

\begin{proposition}\label{phase5_named_diagonal_relation_subseteq}
    $\phaseFiveDiagonalRelation{A}\subseteq A\times A$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{phase5_named_diagonal_relation,logical_relation_on_subseteq}.
\end{proof}

\begin{proposition}\label{phase5_named_diagonal_relation_downstream}
    Suppose $(x,y)\in\phaseFiveDiagonalRelation{A}$.
    Then $x=y$.
\end{proposition}
\begin{proof}
    Follows by \auto{phase5_named_diagonal_relation_membership}.
\end{proof}
