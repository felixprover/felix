\begin{recursive-definition}\label{guarded_constant}
    Let $X\in B$.
    Define $\guardedConstant{B}{A}{X}{p}$ recursively for $p\in\plform{A}$ as follows.
    \begin{enumerate}
        \item[Falsity] $\guardedConstant{B}{A}{X}{\propbot}=X$.
        \item[Atom] $\guardedConstant{B}{A}{X}{\propvar{a}}=X$.
        \item[Implication] $\guardedConstant{B}{A}{X}{p\propto q}=\guardedConstant{B}{A}{X}{q}$.
    \end{enumerate}
\end{recursive-definition}

\begin{recursive-definition}\label{guarded_predicate}
    Let $X\in B$.
    Define $\guardedPredicate{B}{A}{X}{p}$ recursively for $p\in\plform{A}$ as follows.
    \begin{enumerate}
        \item[Falsity] $\guardedPredicate{B}{A}{X}{\propbot}$ iff $\top$.
        \item[Atom] $\guardedPredicate{B}{A}{X}{\propvar{a}}$ iff $\top$.
        \item[Implication] $\guardedPredicate{B}{A}{X}{p\propto q}$ iff not $\guardedPredicate{B}{A}{X}{q}$.
    \end{enumerate}
\end{recursive-definition}

\begin{recursive-definition}\label{previous_recursion}
    Define $\previousRecursion{A}{p}$ recursively for $p\in\plform{A}$ as follows.
    \begin{enumerate}
        \item[Falsity] $\previousRecursion{A}{\propbot}=\plAtoms{A}{\propbot}$.
        \item[Atom] $\previousRecursion{A}{\propvar{a}}=\plAtoms{A}{\propvar{a}}$.
        \item[Implication] $\previousRecursion{A}{p\propto q}=\plAtoms{A}{p}$.
    \end{enumerate}
\end{recursive-definition}

\begin{proposition}\label{guarded_constant_use}
    Let $X\in B$.
    Then $\guardedConstant{B}{A}{X}{\propbot\propto\propbot}=X$.
\end{proposition}
\begin{proof}
    Follows by definition.
\end{proof}

\begin{proposition}\label{guarded_predicate_use}
    Let $X\in B$.
    Then not $\guardedPredicate{B}{A}{X}{\propbot\propto\propbot}$.
\end{proposition}
\begin{proof}
    Follows by definition.
\end{proof}

\begin{proposition}\label{previous_recursion_use}
    $\previousRecursion{A}{\propbot\propto\propbot}=\emptyset$.
\end{proposition}
\begin{proof}
    Follows by definition.
\end{proof}

\begin{recursive-definition}\label{binder_recursion}
    Define $\binderRecursion{A}{p}$ recursively for $p\in\plform{A}$ as follows.
    \begin{enumerate}
        \item[Falsity] $\binderRecursion{A}{\propbot}$ iff $\bot$.
        \item[Atom] $\binderRecursion{A}{\propvar{a}}$ iff $\top$.
        \item[Implication] $\binderRecursion{A}{p\propto q}$ iff
            for every set $x$ we have if $x=x$ then not $\binderRecursion{A}{p}$.
    \end{enumerate}
\end{recursive-definition}

\begin{proposition}\label{binder_recursion_use}
    $\binderRecursion{A}{\propbot\propto\propbot}$.
\end{proposition}
\begin{proof}
    Follows by definition.
\end{proof}
