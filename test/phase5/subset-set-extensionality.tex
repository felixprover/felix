\import{set.tex}

\begin{proposition}\label{subset_set_extensionality}
    Let $A,B$ be sets.
    Suppose $A\subseteq B$.
    Suppose $B\subseteq A$.
    Then $A=B$.
\end{proposition}
\begin{proof}
    Follows by set extensionality.
\end{proof}

\begin{proposition}\label{mixed_subset_set_extensionality}
    Let $A,B$ be sets.
    Suppose for every $x\in A$ we have $x\in B$.
    Suppose $B\subseteq A$.
    Then $A=B$.
\end{proposition}
\begin{proof}
    Follows by set extensionality.
\end{proof}

\begin{proposition}\label{chained_subset_set_extensionality}
    Let $A,B$ be sets.
    Suppose $A$ is a subset of $B$.
    Suppose $A\supseteq B$.
    Then $A=B$.
\end{proposition}
\begin{proof}
    Follows by set extensionality.
\end{proof}
