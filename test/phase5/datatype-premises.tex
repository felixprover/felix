\begin{datatype}\label{premise_data}
    Define $\premisedata$ inductively as follows.
    \begin{enumerate}
        \item $\premiseleaf \in \premisedata$.
        \item $\premiseatom{x} \in \premisedata$ for
            $x \in \{\premiseleaf\}$ and $w \in \emptyset$.
        \item $\premisejoin{y}{x} \in \premisedata$ for
            $x \in \premisedata$, $y \in \premisedata$, and
            $z \in \{\premiseleaf\}$.
    \end{enumerate}
\end{datatype}
