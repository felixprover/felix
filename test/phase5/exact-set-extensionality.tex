\begin{proposition}\label{exact_set_extensionality}
    Let $A,B$ be sets.
    Suppose for all $x\in A$ we have $x\in B$.
    Suppose for all $x\in B$ we have $x\in A$.
    Then $A=B$.
\end{proposition}
\begin{proof}
    Follows by set extensionality.
\end{proof}

\begin{proposition}\label{generic_set_extensionality}
    Let $A,B$ be sets.
    Suppose for all $x\in A$ we have $x\in B$.
    Suppose for all $x\in B$ we have $x\in A$.
    Then $A=B$.
\end{proposition}
\begin{proof}
    Follows by extensionality.
\end{proof}

\begin{proposition}\label{pointwise_iff_set_extensionality}
    Let $A,B$ be sets.
    Suppose for all $x$ we have $x\in A$ iff $x\in B$.
    Then $A=B$.
\end{proposition}
\begin{proof}
    Follows by set extensionality.
\end{proof}

\begin{proposition}\label{reversed_pointwise_iff_set_extensionality}
    Let $A,B$ be sets.
    Suppose for all $x$ we have $x\in B$ iff $x\in A$.
    Then $A=B$.
\end{proposition}
\begin{proof}
    Follows by set extensionality.
\end{proof}

\begin{proposition}\label{conjoined_pointwise_iff_set_extensionality}
    Let $A,B$ be sets.
    Suppose (for all $x$ we have $x\in A$ iff $x\in B$) and $A=A$.
    Then $A=B$.
\end{proposition}
\begin{proof}
    Follows by set extensionality.
\end{proof}
