\begin{proposition}\label{explicit_demorgan_premise}
    Let $a,P,Q,R$ be sets.
    Suppose $\lnot(a\notin P\land a\notin Q)$.
    Suppose for all $x$ we have $(x\in P\lor x\in Q)\implies x\in R$.
    Then $a\in R$.
\end{proposition}
\begin{proof}
    $a\in P$ or $a\in Q$.
    Follows.
\end{proof}

\begin{proposition}\label{obvious_excluded_middle_premise}
    Let $a,P,R$ be sets.
    Suppose for all $x$ we have $(x\in P\lor x\notin P)\implies x\in R$.
    Then $a\in R$.
\end{proposition}
\begin{proof}
    Follows.
\end{proof}

\begin{proposition}\label{obvious_two_boolean_premises}
    Let $a,P,Q,R$ be sets.
    Suppose for all $x$ we have $(x\in P\lor x\notin P)\implies
        ((x\in Q\lor x\notin Q)\implies x\in R)$.
    Then $a\in R$.
\end{proposition}
\begin{proof}
    Follows.
\end{proof}
