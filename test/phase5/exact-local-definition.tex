\begin{proposition}\label{phase5_local_definition}
    For all $A$ we have $A = A$.
\end{proposition}
\begin{proof}
    Fix $A$.
    Let $B = \{x \in A \mid x = x\}$.
    Show for all $x$ we have if $x \in B$, then $x \in A$.
    \begin{subproof}
        Fix $x$.
        Assume $x \in B$.
        Follows.
    \end{subproof}
    Follows by \auto.
\end{proof}
