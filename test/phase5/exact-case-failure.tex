\begin{proposition}\label{case_failure_exact}
    Let $A$ be a set.
    $A=A$.
\end{proposition}
\begin{proof}
    \begin{byCase}
    \caseOf{$A=\emptyset$.}
        Follows by \auto.
    \caseOf{$A=\{A\}$.}
        Follows by \auto.
    \caseOf{$A=\{\emptyset\}$.}
        Follows by \auto.
    \end{byCase}
\end{proof}
