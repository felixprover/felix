\begin{definition}\label{subseteq}
    $A\subseteq B$ iff for every $a\in A$ we have $a\in B$.
\end{definition}

\begin{proposition}[Carrier monotonicity]\label{pl_formula_monotone}
    Suppose $A\subseteq B$.
    Then for every $p\in\plform{A}$ we have $p\in\plform{B}$.
\end{proposition}
\begin{proof}[Proof by structural induction on $p$]
    \begin{byCase}
        \caseOf{Falsity.}
            Follows by \obv{pl_formula_falsity_intro}.
        \caseOf{Atom.}
            Fix $a\in A$.
            Then $a\in B$ by \obv{subseteq}.
            Follows by \obv{pl_formula_atom_intro}.
        \caseOf{Implication.}
            Fix $q,r\in\plform{A}$.
            Suppose $q\in\plform{B}$ and $r\in\plform{B}$.
            Follows by \obv{pl_formula_implication_intro}.
    \end{byCase}
\end{proof}

\begin{proposition}[Active membership]\label{pl_formula_monotone_active}

\end{proposition}

    \end{byCase}
\end{proof}

\begin{proposition}\label{pl_formula_implicational_motive}
    Then for every $p\in\plform{A}$ we have
    if $p\in C$ then $p=p$.
\end{proposition}
\begin{proof}[Proof by structural induction on $p$]
    \begin{byCase}
        \caseOf{Implication.}
            Fix $q,r\in\plform{A}$.
            Suppose if $q\in C$ then $q=q$.
            Suppose if $r\in C$ then $r=r$.
            Follows.
        \caseOf{Falsity.}
            \begin{byCase}
                \caseOf{$C=C$.} Follows.
                \caseOf{$C\neq C$.} Follows.
            \end{byCase}
        \caseOf{Atom.}
            Fix $a\in A$.
            Follows.
    \end{byCase}
\end{proof}
