\begin{definition}\label{prepared_evidence_distraction}
    $\preparedEvidenceDistraction{X}$ iff $X=X$.
\end{definition}

\begin{definition}\label{prepared_evidence_leaf}
    $\preparedEvidenceLeaf{A}{D}$ iff $A=A$ and $D=D$.
\end{definition}

\begin{definition}\label{prepared_evidence_package}
    $\preparedEvidencePackage{A}{D}$ iff $\preparedEvidenceLeaf{A}{D}$ and
    for all $x$ we have if $x=A$, then $A=x$.
\end{definition}

\begin{proposition}\label{prepared_evidence_closed}
    Suppose $\preparedEvidenceDistraction{X}$.
    Suppose $\preparedEvidencePackage{A}{D}$.
    Then $A=A$ and $D=D$ and for all $x$ we have if $x=A$, then $A=x$.
\end{proposition}
\begin{proof}
    Follows by \auto.
\end{proof}

\begin{proposition}\label{prepared_evidence_partial}
    Suppose $\preparedEvidenceDistraction{X}$.
    Suppose $\preparedEvidencePackage{A}{D}$.
    Then $A=A$ and $D=D$ and
    for all $x$ we have if $x=A$, then $A=x$ and
    for all $y$ we have $y=y$.
\end{proposition}
\begin{proof}
    Follows by \auto.
\end{proof}

\begin{proposition}\label{prepared_evidence_transport}
    Let $A,B,C$ be sets.
    Assume $A=B$ and $A\in C$.
    Then $B\in C$.
\end{proposition}
\begin{proof}
    Follows by \auto.
\end{proof}

\begin{proposition}\label{prepared_evidence_transport_binder}
    Let $A,B$ be sets.
    Assume $B=A$.
    Assume for every set $x$ we have $x\in A$.
    Then for every set $x$ we have $x\in B$.
\end{proposition}
\begin{proof}
    Follows by \auto.
\end{proof}
