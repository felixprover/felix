\import{recursion/well-founded-relation.tex}

\begin{datatype}\label{witness_tree}
    Define $\witnessTree{A}$ inductively as follows.
    \begin{enumerate}
        \item[Leaf] $\witnessLeaf\in\witnessTree{A}$.
        \item[Node] $\witnessNode{a}{x}\in\witnessTree{A}$ for
            $x\in\witnessTree{A}$, $w\in A$, and $a\in A$.
    \end{enumerate}
\end{datatype}

\begin{classification}\label{witness_node_formation}
    Let $a\in A$ and $x\in\witnessTree{A}$.
    Then $\witnessNode{a}{x}\in\witnessTree{A}$.
\end{classification}
\begin{proof}
    Follows by \obv{witness_tree_node_intro}.
\end{proof}

\begin{recursive-definition}\label{witness_copy}
    Define $\witnessCopy{A}{x}$ recursively for $x\in\witnessTree{A}$ as follows.
    \begin{enumerate}
        \item[Leaf] $\witnessCopy{A}{\witnessLeaf}=\witnessLeaf$.
        \item[Node] $\witnessCopy{A}{\witnessNode{a}{x}}=\witnessCopy{A}{x}$.
    \end{enumerate}
\end{recursive-definition}

\begin{proposition}\label{witness_copy_use}
    Let $a\in A$.
    Then $\witnessCopy{A}{\witnessNode{a}{\witnessLeaf}}=\witnessLeaf$.
\end{proposition}
\begin{proof}
    Follows by definition.
\end{proof}
