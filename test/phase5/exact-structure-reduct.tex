\begin{struct}\label{reduct_base}
    A reduct base $X$ is a onesorted structure such that
    \begin{enumerate}
        \item\label{reduct_base_reflexive} $X=X$.
    \end{enumerate}
\end{struct}

\begin{struct}\label{reduct_extension}
    A reduct extension $X$ is a reduct base equipped with
    \begin{enumerate}
        \item\label{reduct_extension_point}
            a point $\reductPoint[X]\in\carrier[X]$;
    \end{enumerate}
    .
\end{struct}

\begin{instance}\label{reduct_extension_object}
    The reduct extension $\ReductExtension$ is
    $\structure{\{\emptyset\},\emptyset}$.
\end{instance}
\begin{proof}
    Follows by
    \auto{reduct_extension,reduct_base,reduct_extensionaggregateexact,reduct_extensionaggregateprojection_carrier,reduct_extensionaggregateprojection_reductPoint}.
\end{proof}

\begin{proposition}\label{reduct_is_parent}
    $\reduct[reduct base]{\ReductExtension}$ is a reduct base.
\end{proposition}
\begin{proof}
    Follows by \auto{reduct_base,reduct_baseshape,reduct_basereductexact,reduct_extensioninherit_reduct_base,reduct_extension_object}.
\end{proof}

\begin{proposition}\label{reduct_is_exact_parent}
    $\reduct[reduct base]{\ReductExtension}$ is an exact reduct base.
\end{proposition}
\begin{proof}
    Follows by \auto{reduct_is_parent,reduct_baseshape,reduct_basereductexact,reduct_extensioninherit_reduct_base,reduct_extension_object}.
\end{proof}

\begin{proposition}\label{reduct_preserves_carrier}
    $\carrier[\reduct[reduct base]{\ReductExtension}]
        =\carrier[\ReductExtension]$.
\end{proposition}
\begin{proof}
    Follows by \auto{reduct_basereductprojection_carrier}.
\end{proof}

\begin{proposition}\label{reduct_local_projection}
    Let $A$ be a reduct extension.
    Assume $\reduct[reduct base]{A}=\reduct[reduct base]{A}$ and $A=A$.
    Then for every set $x$ we have $\reduct[reduct base]{A}$ is a reduct base.
\end{proposition}
\begin{proof}
    Fix $x$.
    Follows by \auto{reduct_base,reduct_baseshape,reduct_basereductexact,reduct_extensioninherit_reduct_base}.
\end{proof}

\begin{proposition}\label{reduct_local_missing}
    Let $A$ be a reduct extension.
    Then for every set $x$ we have $\reduct[reduct base]{A}$ is a reduct base.
\end{proposition}
\begin{proof}
    Fix $x$.
    Follows by \auto{reduct_base,reduct_baseshape,reduct_basereductexact,reduct_extensioninherit_reduct_base}.
\end{proof}

\begin{proposition}\label{reduct_predicate_substitution}
    Let $A,B$ be reduct bases.
    Let $C$ be a set.
    Assume $C=A$ or $C=B$.
    Then $C$ is a reduct base.
\end{proposition}
\begin{proof}
    Follows by \auto{reduct_basereductexact}.
\end{proof}

\begin{proposition}\label{reduct_wide_local_projection}
    Let $A$ be a reduct extension.
    Assume $\reduct[reduct base]{A}=\reduct[reduct base]{A}$.
    Then $\reduct[reduct base]{A}$ is a wide reduct base.
\end{proposition}
\begin{proof}
    Follows by \auto{reduct_wide,reduct_baseshape,reduct_basereductexact,reduct_extensioninherit_reduct_base}.
\end{proof}

\begin{proposition}\label{reduct_predicate_substitution_plain}
    Let $A,B$ be reduct bases.
    Let $C$ be a set.
    Assume $C=A$ or $C=B$.
    Then $C$ is a reduct base.
\end{proposition}
\begin{proof}
    Follows by \auto.
\end{proof}

\begin{proposition}\label{reduct_local_alias_chain}
    Let $A$ be a reduct base.
    Then $A$ is a reduct base.
\end{proposition}
\begin{proof}
    Let $B=A$.
    Let $C=B$.
    $C$ is a reduct base by \auto.
    Follows by \auto.
\end{proof}
