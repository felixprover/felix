\begin{proposition}\label{exact_local_disjunction_cases}
    Let $A$ be a set.
    Suppose $A=\emptyset$ or $A=\{A\}$.
    Then $A=A$.
\end{proposition}
\begin{proof}
    \begin{byCase}
    \caseOf{$A=\emptyset$.}
        Follows by \auto.
    \caseOf{$A=\{A\}$.}
        Follows by \auto.
    \end{byCase}
\end{proof}
