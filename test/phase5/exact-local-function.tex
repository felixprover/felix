\begin{proposition}\label{phase5_local_function}
    For all $A$ we have $A = A$.
\end{proposition}
\begin{proof}
    Fix $A$.
    Let $f(x) = x$ for $x\in A$.
    Show for all $y$ we have if $y\in A$, then $(y,y)\in f$.
    \begin{subproof}
        Fix $y$.
        Assume $y\in A$.
        Follows by \auto.
    \end{subproof}
    Follows.
\end{proof}
