\begin{definition}\label{replacement_test_definition}
    $\replacementTest{A} = \{ y \mid x \in A, y \in x \mid x = y \}$.
\end{definition}

\begin{definition}\label{separation_test_definition}
    $\separationTest{A} = \{x\in A \mid x\in x\}$.
\end{definition}

\begin{proposition}\label{replacement_separation_theorem}
    For all $A$ we have $\separationTest{A} = \replacementTest{A}$.
\end{proposition}
\begin{proof}
    Fix $A$.
    It suffices to show that for all $x$ we have $x\in \separationTest{A}$ iff $x\in \replacementTest{A}$ by \auto{setext}.
    Fix $x$.
    Follows by \auto{replacement_test_definition,separation_test_definition}.
\end{proof}
