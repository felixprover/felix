\begin{struct}\label{matrix_base}
    A matrix base $X$ is a onesorted structure.
\end{struct}

\begin{struct}\label{matrix_left}
    A matrix left $X$ is a matrix base equipped with
    \begin{enumerate}
        \item\label{matrix_left_point}
            a point $\matrixPoint[X]\in\carrier[X]$;
    \end{enumerate}
    .
\end{struct}

\begin{struct}\label{matrix_right}
    A matrix right $X$ is a matrix base such that
    \begin{enumerate}
        \item\label{matrix_right_reflexive} $X=X$.
    \end{enumerate}
\end{struct}

\begin{struct}\label{matrix_diamond}
    A matrix diamond $X$ is a matrix left and a matrix right.
\end{struct}

\begin{proposition}\label{matrix_cited_parent}
    Let $A$ be a matrix diamond.
    Then $A$ is a matrix left.
\end{proposition}
\begin{proof}
    Follows by \obv{matrix_diamond}.
\end{proof}

\begin{proposition}\label{matrix_cited_parent_witness}
    Let $A$ be a matrix diamond.
    Then there exists a set $B$ such that $B$ is a matrix left and $B=A$.
\end{proposition}
\begin{proof}
    Follows by \auto{matrix_diamond}.
\end{proof}

\begin{proposition}\label{matrix_annotation_is_not_a_tag}
    $\structure[matrix base]{A}=\structure[matrix right]{A}$.
\end{proposition}

\begin{definition}\label{matrix_family}
    $\matrixFamily{A}=\structure[matrix diamond]{\{A\},A}$.
\end{definition}

\begin{proposition}\label{matrix_family_is_diamond}
    $\matrixFamily{A}$ is a matrix diamond.
\end{proposition}
\begin{proof}
    Follows by \auto{matrix_family,matrix_diamond,matrix_left,matrix_right,matrix_base,matrix_leftaggregateexact,matrix_leftaggregateprojection_carrier,matrix_leftaggregateprojection_matrixPoint}.
\end{proof}

\begin{proposition}\label{exact_matrix_left_extensionality}
    Let $A$ be a set.
    Let $B$ be a set.
    Assume $A$ is an exact matrix left.
    Assume $B$ is an exact matrix left.
    Assume $\carrier[A]=\carrier[B]$.
    Assume $\matrixPoint[A]=\matrixPoint[B]$.
    Then $A=B$.
\end{proposition}
\begin{proof}
    Follows by \auto{matrix_leftprojectionextensionality}.
\end{proof}
