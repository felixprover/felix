\begin{definition}\label{phase5_apply}
    $\apply{f}{x} = \unions{\img{f}{\{x\}}}$.
\end{definition}

    Then $f(x) = y$.
\end{proposition}

    Follows by \auto.
\end{proof}

\begin{proof}
