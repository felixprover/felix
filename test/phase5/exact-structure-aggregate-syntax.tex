\begin{struct}\label{aggregate_syntax_structure}
    An aggregate syntax structure $X$ is a onesorted structure equipped with
    \begin{enumerate}
        \item\label{aggregate_syntax_point}
            a point $\aggregatePoint[X]\in\carrier[X]$;
    \end{enumerate}
    .
\end{struct}

\begin{proposition}\label{aggregate_syntax_contextual}
    Suppose $x\in A$.
    Then $\structure{A,x}$ is an aggregate syntax structure.
\end{proposition}
\begin{proof}
    Follows by
    \auto{aggregate_syntax_structure,aggregate_syntax_structureaggregateexact,aggregate_syntax_structureaggregateprojection_carrier,aggregate_syntax_structureaggregateprojection_aggregatePoint}.
\end{proof}

\begin{proposition}\label{aggregate_syntax_contextual_exact}
    Suppose $x\in A$.
    Then $\structure{A,x}$ is an exact aggregate syntax structure.
\end{proposition}
\begin{proof}
    Follows by
    \auto{aggregate_syntax_structure,aggregate_syntax_structureaggregateexact,aggregate_syntax_structureaggregateprojection_carrier,aggregate_syntax_structureaggregateprojection_aggregatePoint}.
\end{proof}

\begin{proposition}\label{aggregate_syntax_named}
    $\structure[aggregate syntax structure]{A,x}
        =\structure[aggregate syntax structure]{
            \aggregatePoint=x,
            \carrier=A
        }$.
\end{proposition}
\begin{proof}
    Follows.
\end{proof}

\begin{proposition}\label{aggregate_syntax_carrier}
    $\carrier[\structure[aggregate syntax structure]{A,x}]=A$.
\end{proposition}
\begin{proof}
    Follows by \auto{aggregate_syntax_structureaggregateprojection_carrier}.
\end{proof}

\begin{proposition}\label{aggregate_syntax_point_value}
    $\aggregatePoint[\structure[aggregate syntax structure]{A,x}]=x$.
\end{proposition}
\begin{proof}
    Follows by \auto{aggregate_syntax_structureaggregateprojection_aggregatePoint}.
\end{proof}
