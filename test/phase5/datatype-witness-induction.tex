\begin{proposition}\label{witness_induction}
    Then for every $p\in\witnessdata{A}$ we have $p=p$.
\end{proposition}
\begin{proof}[Proof by structural induction on $p$]
    \begin{byCase}
        \caseOf{Witnesszero.} Follows.
        \caseOf{Witnessnode.}
            Fix $a\in A$.
            Fix $w\in\witnessdata{A}$.
            Suppose $w=w$.
            Follows.
    \end{byCase}
\end{proof}
