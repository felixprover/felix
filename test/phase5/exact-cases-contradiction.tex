\begin{proposition}\label{cases_exact}
    Let $A$ be a set.
    $A=A$.
\end{proposition}
\begin{proof}
    \begin{byCase}
    \caseOf{$A=\emptyset$.}
        Follows.
    \caseOf{$A=\{A\}$.}
        Follows.
    \caseOf{$A=A$.}
        Follows.
    \end{byCase}
\end{proof}

\begin{proposition}\label{by_contradiction_exact}
    Let $A$ be a set.
    $A=A$.
\end{proposition}
\begin{proof}
    Suppose not.
    We have $A=A$.
    We have $\bot$.
    We have $A=\emptyset$.
    Follows.
\end{proof}

\begin{proposition}\label{arbitrary_contradiction_exact}
    Let $A$ be a set.
    Suppose $A\neq A$.
    Then $A=A$.
\end{proposition}
\begin{proof}
    We have $A=A$.
    Contradiction.
\end{proof}

\begin{proposition}\label{case_contradiction_intermediate_exact}
    Let $A$ be a set.
    $A=A$.
\end{proposition}
\begin{proof}
    \begin{byCase}
    \caseOf{$A=A$.}
        Follows.
    \caseOf{$A\neq A$.}
        We have $A=A$.
        We have $\bot$.
        We have $A=\emptyset$.
        Contradiction.
    \end{byCase}
\end{proof}

\begin{proposition}\label{omitted_case_exact}
    Let $A$ be a set.
    $A=A$.
\end{proposition}
\begin{proof}
    \begin{byCase}
    \caseOf{$A=A$.}
        Omitted.
    \caseOf{$A\neq A$.}
        Follows.
    \end{byCase}
\end{proof}
