\begin{proposition}\label{pl_atoms_subset}
    For every $p\in\plform{A}$ we have $\plAtoms{A}{p}\subseteq A$.
\end{proposition}
\begin{proof}[Proof by structural induction on $p$]
    \begin{byCase}
        \caseOf{Falsity.}
            $\plAtoms{A}{\propbot}=\emptyset$ by definition.
            Follows by \obv{emptyset_subseteq}.
        \caseOf{Atom.}
            Fix $a\in A$.
            $\plAtoms{A}{\propvar{a}}=\{a\}$ by definition.
            Follows by \obv{singleton_subset_intro}.
        \caseOf{Implication.}
            Fix $q,r\in\plform{A}$.
            Suppose $\plAtoms{A}{q}\subseteq A$ and $\plAtoms{A}{r}\subseteq A$.
            $\plAtoms{A}{q\propto r}=\plAtoms{A}{q}\union\plAtoms{A}{r}$ by definition.
            Follows by \obv{union_subsets_is_subset}.
    \end{byCase}
\end{proof}

\begin{proposition}[Reconstruction is the identity]\label{pl_rebuild_identity}
    For every $p\in\plform{A}$ we have $\plRebuild{A}{p}=p$.
\end{proposition}
\begin{proof}[Proof by structural induction on $p$]
    \begin{byCase}
        \caseOf{Falsity.}
            Follows by definition.
        \caseOf{Atom.}
            Fix $a\in A$.
            Follows by definition.
        \caseOf{Implication.}
            Fix $q,r\in\plform{A}$.
            Suppose $\plRebuild{A}{q}=q$ and $\plRebuild{A}{r}=r$.
            $\plRebuild{A}{q\propto r}=\plRebuild{A}{q}\propto\plRebuild{A}{r}$ by definition.
            Follows.
    \end{byCase}
\end{proof}
