\begin{proposition}\label{exact_record_theorem_equality_no_alias}
    Let $A,B,P,Q$ be sets.
    Assume $P=A$.
    Assume $Q=B$.
    Assume $\pow{P}$ is an exact matrix left.
    Assume $\pow{Q}$ is an exact matrix left.
    Assume $\carrier[\pow{P}]=\carrier[\pow{Q}]$.
    Assume $\matrixPoint[\pow{P}]=\matrixPoint[\pow{Q}]$.
    Then $\pow{A}=\pow{B}$.
\end{proposition}
\begin{proof}
    Follows by structure extensionality.
\end{proof}
