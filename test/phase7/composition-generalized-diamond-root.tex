\begin{proposition}\label{calculation_generalized_diamond_guarded}
    Let $X$ be a quasiordered set.
    For all $a,b,c,d\in X$ such that
        $a\calcorder b$ and $b\mathrel{\leq[X]}c$ and $c=d$
        we have $a\calcorder d$.
\end{proposition}
\begin{proof}
    For all $a,b,c,d\in X$ such that
        $a\calcorder b$ and $b\mathrel{\leq[X]}c$ and $c=d$, we have
    \begin{align*}
        a
            &\calcorder b
                \explanation{by assumption}
                \\
            &\mathrel{\leq[X]} c
                \explanation{by assumption}
                \\
            &= d
                \explanation{by assumption}
    \end{align*}
    Follows.
\end{proof}
