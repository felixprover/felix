\begin{proposition}\label{calculation_transported_conditions}
    Suppose $A=B$.
    Suppose $a\in B$ and $b\in B$ and $c\in B$.
    Suppose $a\calcguard{A}b\calcguard{A}c$.
    Then $a\calcguard{A}c$.
\end{proposition}
\begin{proof}
    \begin{align*}
        a &\calcguard{A}b \explanation{by assumption} \\
          &\calcguard{A}c \explanation{by assumption}
    \end{align*}
\end{proof}
