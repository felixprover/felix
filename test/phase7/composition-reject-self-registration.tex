\begin{abbreviation}\label{calculation_self_relation}
    $a\calcself{n}b$ iff $(a,n)=(b,n)$.
\end{abbreviation}

\begin{composition}\label{calculation_self_registration}
    Suppose $x\calcself{n}y\calcself{n}z$.
    Then $x\calcself{n}z$.
\end{composition}
\begin{proof}
    \begin{align*}
        x
            &\calcself{n} y
                \explanation{by assumption}
                \\
            &\calcself{n} z
                \explanation{by assumption}
    \end{align*}
\end{proof}
