\begin{proposition}\label{phase7_composition_diamond_root}
    Suppose $A\subseteq B$.
    Suppose $B=C$.
    Suppose $C\subseteq D$.
    Then $A\subseteq D$.
\end{proposition}
\begin{proof}
    \begin{align*}
        A
            &\subseteq B
                \explanation{by \auto}
                \\
            &= C
                \explanation{by \auto}
                \\
            &\subseteq D
                \explanation{by \auto}
    \end{align*}
\end{proof}
