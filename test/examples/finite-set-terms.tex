\begin{signature}\label{example_cons}
    $\cons{y}{X}$ is a set.
\end{signature}

\begin{axiom}\label{cons}
    $x\in \cons{y}{X}$ iff $x = y$ or $x\in X$.
\end{axiom}

\begin{definition}\label{unit}
    $\unit = \{\emptyset\}$.
\end{definition}

\begin{proposition}\label{emptyset_in_unit}
    $\emptyset\in\unit$.
\end{proposition}
