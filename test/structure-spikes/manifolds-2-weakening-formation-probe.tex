\section{Pseudogroup weakening formation}

\import{topology/coordinate-homeomorphism.tex}

\begin{theorem}\label{atlas_weakening_formation_control}
    Let $G,H$ be coordinate pseudogroups and $X$ be a topological space.
    Let $A$ be a coordinate atlas on $X$ under $G$.
    Suppose $G$ is a subpseudogroup of $H$.
    Then $A$ is a coordinate atlas on $X$ under $H$.
\end{theorem}
\begin{proof}
    Follows by notion.
\end{proof}

\begin{theorem}\label{relative_atlas_weakening_formation_control}
    Let $G,H$ be coordinate pseudogroups and $X$ be a topological space.
    Let $S$ be a relative coordinate structure on $X$ under $G$.
    Suppose $G$ is a subpseudogroup of $H$.
    Then $\atlas[S]$ is a coordinate atlas on $X$ under $H$.
\end{theorem}
\begin{proof}
    Follows by notion.
\end{proof}

\begin{theorem}\label{relative_atlas_weakening_guard_consumer}
    Let $G,H$ be coordinate pseudogroups and $X$ be a topological space.
    Let $S$ be a relative coordinate structure on $X$ under $G$.
    Suppose $G$ is a subpseudogroup of $H$.
    Then $\MaxAtlas{H}{X}{\atlas[S]}=\MaxAtlas{H}{X}{\atlas[S]}$.
\end{theorem}
\begin{proof}
    Follows.
\end{proof}
