\begin{proposition}\label{imported_clear_result}
    Let $X$ be a set.
    Then $\ImportedClear{X}$ is a coordinate record on $X$.
\end{proposition}
\begin{proof}
    Follows by \obv{imported_clear}.
\end{proof}

\begin{proposition}\label{imported_clear_shape}
    Let $X$ be a set.
    Then $\ImportedClear{X}$ is an exact coordinate record on $X$.
\end{proposition}
\begin{proof}
    Follows by \obv{imported_clear,imported_clearshape}.
\end{proof}

\begin{proposition}\label{imported_clear_atlas}
    Let $X$ be a set.
    Then $\atlas[\ImportedClear{X}]=\emptyset$.
\end{proposition}
\begin{proof}
    Follows by \obv{imported_clearprojection_atlas}.
\end{proof}

\begin{proposition}\label{imported_clear_reconstruction}
    Let $X$ be a set.
    Then $(coordinate record on X based on \ImportedClear{X})=
        \structure[coordinate record on X]{\atlas=\emptyset}$.
\end{proposition}
\begin{proof}
    Follows by \obv{imported_clearprojection_atlas}.
\end{proof}
