Reconstruction selects fields. Its target annotation does not establish laws.

\begin{proposition}\label{expression_original}
    Let $X,Y$ be sets.
    Then $(coordinate record on X based on \ImportedCoordinate{X})=
        \structure[coordinate record on X]{\atlas=X}$.
\end{proposition}
\begin{proof}
    Follows by \auto{imported_coordinate_base,coordinate_recordaggregateprojection_atlas}.
\end{proof}

\begin{proposition}\label{expression_target}
    Let $X,Y$ be sets.
    Then $(record based on \ImportedCoordinate{X})=\structure[record]{}$.
\end{proposition}
