\begin{construction}\label{unproved_parameter_copy}
    Let $X,Y$ be sets.
    Let $R$ be a coordinate record on $X$.
    The coordinate record $\ParameterCopy{X}{Y}{R}=
        coordinate record on Y based on R$ on $Y$.
\end{construction}
\begin{proof}
    $\atlas[R]\subseteq X$ by \obv{coordinate_atlas_bound}.
    Follows.
\end{proof}
