\begin{construction}\label{unproved_field_copy}
    Let $X,B$ be sets.
    Let $R$ be a coordinate record on $X$.
    The coordinate record $\FieldCopy{X}{B}{R}=
        coordinate record on X based on R with \atlas=B$ on $X$.
\end{construction}
\begin{proof}
    $\atlas[R]\subseteq X$ by \obv{coordinate_atlas_bound}.
    Follows.
\end{proof}
