\begin{theorem}\label{coordinate_map_parameter_control}
    Let $C$ be a local-map calculus.
    Let $G,H$ be coordinate pseudogroups.
    Let $M,N$ be equipped coordinate spaces.
    Then if $f$ is coordinate-admissible from $M$ under $G$ to $N$ under $H$ in $C$,
        then $f$ is coordinate-admissible from $M$ under $G$ to $N$ under $H$ in $C$.
\end{theorem}

\end{proof}
