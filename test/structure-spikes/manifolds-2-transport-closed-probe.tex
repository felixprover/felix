\section{Closed transport examples}

\import{topology/positive-coordinate-examples.tex}

\begin{theorem}\label{closed_reflection_transport_consumer}
    Then $\atlas[\TransportedCoordinates{\PositiveRealPseudogroup}{\RealTopology}{\RealTopology}
            {\GeneratedCoordinates{\PositiveRealPseudogroup}{\RealTopology}{\RealOneChart}}{\surrealRealNegationGraph}]
        =\atlas[\GeneratedCoordinates{\PositiveRealPseudogroup}{\RealTopology}{\RealReflectionChart}]$.
\end{theorem}
\begin{proof}
    Follows by \obv{real_reflection_transports_positive_coordinates}.
\end{proof}

\begin{theorem}\label{closed_reflection_admissible_consumer}
    Then $\surrealRealNegationGraph$ is coordinate-admissible
        from $\Equip{\RealTopology}{\GeneratedCoordinates{\PositiveRealPseudogroup}{\RealTopology}{\RealOneChart}}$
        under $\PositiveRealPseudogroup$
        to $\Equip{\RealTopology}{\GeneratedCoordinates{\PositiveRealPseudogroup}{\RealTopology}{\RealReflectionChart}}$
        under $\PositiveRealPseudogroup$ in $\ContinuousCalculus{\{\RealTopology\}}$.
\end{theorem}
\begin{proof}
    $\RealTopology$ is a topological space by \obv{real_topology_is_topological_space}.
    $\PositiveRealPseudogroup$ is a coordinate pseudogroup by \obv{positive_real_pseudogroup}.
    $\model[\PositiveRealPseudogroup]=\RealTopology$ by \obv{positive_real_pseudogroup_model}.
    $\{\RealTopology\}$ is a topological model family by \obv{singleton_topological_model_family}.
    $\model[\PositiveRealPseudogroup]\in\{\RealTopology\}$ by \obv{singleton_iff}.
    $\ContinuousCalculus{\{\RealTopology\}}$ is a local-map calculus by \obv{continuous_calculus}.
    $\PositiveRealPseudogroup$ is compatible with the calculus $\ContinuousCalculus{\{\RealTopology\}}$ by
        \obv{pseudogroup_continuous_calculus_compatible}.
    Follows by \obv{real_reflection_transported_admissible}.
\end{proof}

The identity of the empty open subspace is the empty graph. Transport must
retain its single empty chart, not mistake the maximal atlas for an empty family.

\begin{theorem}\label{closed_empty_transport_consumer}
    Let $G=\FullPseudogroup{\RealTopology}$ and $E=\subspaceTopology{\RealTopology}{\emptyset}$.
    Let $S=\RestrictedCoordinates{G}{\RealTopology}{\GeneratedCoordinates{G}{\RealTopology}{\RealOneChart}}{\emptyset}$.
    Then $\TransportedCoordinates{G}{E}{E}{S}{\identity{\carrier[E]}}=S$ and
        $\atlas[\TransportedCoordinates{G}{E}{E}{S}{\identity{\carrier[E]}}]=\{\emptyset\}$ and
        $\identity{\carrier[E]}=\emptyset$.
\end{theorem}
\begin{proof}
    $\RealTopology$ is a topological space by \obv{real_topology_is_topological_space}.
    $G$ is a coordinate pseudogroup by \auto{full_pseudogroup}.
    $E$ is a topological space by \auto{subspace_topology_is_topological_space}.
    $\emptyset\in\opens[\RealTopology]$ by \auto{emptyset_open,is_open}.
    $\RealOneChart$ is a coordinate atlas on $\RealTopology$ under $G$ by \obv{real_identity_full_atlas}.
    $\GeneratedCoordinates{G}{\RealTopology}{\RealOneChart}$ is a relative coordinate structure
        on $\RealTopology$ under $G$ by \obv{generated_coordinates}.
    $S$ is an exact relative coordinate structure on $E$ under $G$ by \auto{restricted_coordinates_is_exact}.
    $\TransportedCoordinates{G}{E}{E}{S}{\identity{\carrier[E]}}=S$ by \obv{transported_coordinates_identity}.
    $\atlas[S]=\{\emptyset\}$ by \auto{restricted_coordinates_empty_atlas}.
    $\carrier[E]=\emptyset$ by \auto{subspace_topology_carrier,inter_emptyset,inter_comm}.
    $\identity{\carrier[E]}=\emptyset$ by \auto{id,setext}.
\end{proof}
