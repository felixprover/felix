\begin{definition}\label{empty_coordinate_over}
    $\EmptyCoordinateOver{X}=\structure[coordinate record on X]{\atlas=\emptyset}$.
\end{definition}

\begin{definition}\label{other_empty_coordinate_over}
    $\OtherEmptyCoordinateOver{X}=\structure[coordinate record on X]{\atlas=\emptyset}$.
\end{definition}

\begin{struct}\label{marked_coordinate}
    Let $X$ be a set.
    A marked coordinate $R$ on $X$ is a coordinate record on $X$ such that
    \begin{enumerate}
        \item\label{marked_coordinate_reflexive} $R=R$.
    \end{enumerate}
\end{struct}

\begin{classification}\label{empty_coordinate_any_parameter}
    Let $X,Y$ be sets.
    Then $\EmptyCoordinateOver{X}$ is a coordinate record on $Y$.
\end{classification}
\begin{proof}
    Follows by \auto{empty_coordinate_over,coordinate_record,
        coordinate_recordaggregateprojection_atlas,emptyset_subseteq}.
\end{proof}
