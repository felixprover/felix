\section{A toy coordinate vocabulary}

This experiment tests record constructions, not manifold theory. An atlas is
only a family of open sets. The local-map vocabulary below remembers allowed
domains, not maps; its intersection law is the only pseudogroup operation
modelled here.

\begin{struct}\label{local_map_calculus}
    A local-map calculus $C$ is a record equipped with
    \begin{enumerate}
        \item a set $\domains[C]$;
    \end{enumerate}.
\end{struct}

\begin{struct}\label{coordinate_pseudogroup}
    A coordinate pseudogroup $H$ is a local-map calculus such that
    \begin{enumerate}
        \item\label{domains_inter}
            for all $U,V\in\domains[H]$ we have $U\inter V\in\domains[H]$.
    \end{enumerate}
\end{struct}

\begin{struct}\label{coordinate_datum}
    A coordinate datum $R$ is a record equipped with
    \begin{enumerate}
        \item a set $\atlas[R]$;
    \end{enumerate}.
\end{struct}

\begin{definition}\label{admissible_atlas}
    Let $X$ be a topological space.
    $A$ is an admissible atlas on $X$ iff $A\subseteq\opens[X]$.
\end{definition}

\begin{struct}\label{relative_coordinate_structure}
    Let $X$ be a topological space.
    A relative coordinate structure $R$ on $X$ is a coordinate datum such that
    \begin{enumerate}
        \item\label{relative_atlas_admissible}
            $\atlas[R]$ is an admissible atlas on $X$.
    \end{enumerate}
\end{struct}

\begin{struct}\label{equipped_coordinate_space}
    An equipped coordinate space $M$ is a topological space and a coordinate datum
    such that
    \begin{enumerate}
        \item\label{equipped_atlas_open} $\atlas[M]\subseteq\opens[M]$.
    \end{enumerate}
\end{struct}

\section{Completing, restricting, and enlarging atlases}

Completion adds the whole space. Restriction retains the old charts that are
open in the subspace. An admissible weakening allows more domains, still open
in the same topology. These operations are total set expressions; their useful
properties retain the hypotheses below.

\begin{definition}\label{maximal_atlas}
    $\MaximalAtlas{X}{A}=A\union\{\carrier[X]\}$.
\end{definition}

\begin{proposition}\label{maximal_atlas_admissible}
    Let $X$ be a topological space.
    Suppose $A$ is an admissible atlas on $X$.
    Then $\MaximalAtlas{X}{A}$ is an admissible atlas on $X$.
\end{proposition}
\begin{proof}
    $A\subseteq\opens[X]$ by \obv{admissible_atlas}.
    $\carrier[X]\in\opens[X]$ by \obv{carrier_open}.
    Follows by \auto{admissible_atlas,maximal_atlas,union,singleton_iff,subseteq}.
\end{proof}

\begin{proposition}\label{maximal_atlas_equal}
    Suppose $A\union\{\carrier[X]\}=B\union\{\carrier[X]\}$.
    Then $\MaximalAtlas{X}{A}=\MaximalAtlas{X}{B}$.
\end{proposition}
\begin{proof}
    Follows by \obv{maximal_atlas}.
\end{proof}

\begin{definition}\label{restricted_atlas}
    $\RestrictedAtlas{M}{U}=\atlas[M]\inter\subspaceOpens{M}{U}$.
\end{definition}

\begin{proposition}\label{restricted_atlas_open}
    $\RestrictedAtlas{M}{U}\subseteq\subspaceOpens{M}{U}$.
\end{proposition}
\begin{proof}
    Follows by \auto{restricted_atlas,inter_lower_right}.
\end{proof}

\begin{definition}\label{admissible_weakening}
    Let $M$ be an equipped coordinate space.
    Let $H$ be a coordinate pseudogroup.
    $H$ is an admissible weakening for $M$ iff
        $\atlas[M]\subseteq\domains[H]$ and $\domains[H]\subseteq\opens[M]$.
\end{definition}

\begin{definition}\label{allowed_atlas}
    Let $M$ be an equipped coordinate space.
    Let $H$ be a coordinate pseudogroup.
    $M$ has an atlas allowed by $H$ iff $\atlas[M]\subseteq\domains[H]$.
\end{definition}

\begin{proposition}\label{open_subspace_carrier}
    Let $M$ be a topological space.
    Suppose $U$ is open in $M$.
    Then $U\inter\carrier[M]=U$.
\end{proposition}
\begin{proof}
    $U\in\opens[M]$ by \obv{is_open}.
    $U\subseteq\carrier[M]$ by \obv{opens_type}.
    Follows by \obv{inter_absorb_supseteq_right}.
\end{proof}

\begin{proposition}\label{open_subspace_topology}
    Let $M$ be a topological space.
    Suppose $U$ is open in $M$.
    Then $\subspaceOpens{M}{U}$ is a topology on $U$.
\end{proposition}
\begin{proof}
    $U\inter\carrier[M]=U$ by \obv{open_subspace_carrier}.
    Follows by \obv{subspace_opens_form_topology}.
\end{proof}
