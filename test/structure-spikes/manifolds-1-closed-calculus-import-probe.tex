\begin{theorem}\label{closed_calculus_model_import_probe}
    Then $\RealTopology$ is a model of $\ContinuousCalculus{\{\RealTopology\}}$.
\end{theorem}
\begin{proof}
    $\RealTopology$ is a topological space by \obv{real_topology_is_topological_space}.
    $\{\RealTopology\}$ is a topological model family by \obv{singleton_topological_model_family}.
    $\RealTopology\in\{\RealTopology\}$ by \auto{singleton_iff}.
    Follows by \auto{continuous_calculus_model}.
\end{proof}
