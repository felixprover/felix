\begin{definition}\label{closed_value_guard}
    Let $X$ be a set.
    Let $R$ be a coordinate record on $X$.
    $A$ is governed by $R$ over $X$ iff $A=A$.
\end{definition}

\begin{construction}\label{closed_value_copy}
    Let $R$ be a coordinate record on $\{0\}$.
    The coordinate record $\ClosedValueCopy{R}=\structure{\atlas=\emptyset}$ on $\{0\}$.
\end{construction}
\begin{proof}
    Follows by \auto{coordinate_recordaggregateprojection_atlas,emptyset_subseteq}.
\end{proof}

\begin{proposition}\label{closed_value_same_file}
    $\emptyset$ is governed by $\ClosedValueZero$ over $0$.
\end{proposition}
\begin{proof}
    $\ClosedValueZero$ is a coordinate record on $0$ by notion.
    Follows by \obv{closed_value_guard}.
\end{proof}

\begin{proposition}\label{closed_value_unused_outer_binder}
    For every set $X$, $\emptyset$ is governed by $\ClosedValueCopy{\ClosedValueSingleton}$ over $\{0\}$.
\end{proposition}
\begin{proof}
    Fix $X$.
    Let $R=\ClosedValueSingleton$.
    $R$ is a coordinate record on $\{0\}$ by notion.
    $\ClosedValueCopy{R}$ is a coordinate record on $\{0\}$ by notion.
    Follows by \obv{closed_value_guard}.
\end{proof}
