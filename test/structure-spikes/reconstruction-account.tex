\begin{struct}\label{account_record}
    Let $X$ be a set.
    An account record $R$ on $X$ is a record equipped with
    \begin{enumerate}
        \item a set $\payload[R]$;
    \end{enumerate}.
\end{struct}

\begin{struct}\label{account_marked}
    Let $X$ be a set.
    A marked account $R$ on $X$ is an account record on $X$ such that
    \begin{enumerate}
        \item\label{account_reflexive} $R=R$.
    \end{enumerate}
\end{struct}

\begin{definition}\label{account_provider}
    $\Account{X}=\structure[account record on X]{\payload=X}$.
\end{definition}

\begin{classification}\label{account_undetermined}
    Let $X,Y$ be sets.
    Then $\Account{X}$ is an account record on $Y$.
\end{classification}
\begin{proof}
    Follows by \auto{account_provider,account_record}.
\end{proof}

\begin{classification}\label{account_failed}
    Let $X$ be a set.
    Suppose $X\in X$.
    Then $\Account{X}$ is a marked account on $X$.
\end{classification}
\begin{proof}
    $\Account{X}$ is an account record on $X$ by \obv{account_undetermined}.
    Follows by \obv{account_marked}.
\end{proof}

\begin{classification}\label{account_usable}
    Let $X$ be a set.
    Then $\Account{X}$ is an account record on $X$.
\end{classification}
\begin{proof}
    Follows by \obv{account_undetermined}.
\end{proof}

\begin{proposition}\label{discovery_result}
    Let $X$ be a set.
    Then $(account record on X based on \Account{X})=
        \structure[account record on X]{\payload=X}$.
\end{proposition}
\begin{proof}
    Follows by \auto{account_provider,account_recordaggregateprojection_payload}.
\end{proof}

\begin{definition}\label{account_unrelated_provider}
    $\UnrelatedAccount{X}=\structure[account record on X]{\payload=X}$.
\end{definition}

\begin{classification}\label{account_unrelated_result}
    Let $X$ be a set.
    Then $\UnrelatedAccount{X}$ is an account record on $X$.
\end{classification}
\begin{proof}
    Follows by \auto{account_unrelated_provider,account_record}.
\end{proof}
