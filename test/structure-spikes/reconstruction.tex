\section{Reconstruction preserves only the requested fields}

\begin{proposition}\label{reconstruction_copies_atlas}
    Let $R$ be a coordinate record on $X$.
    Then $(coordinate record on X based on R)=
        \structure[coordinate record on X]{\atlas=\atlas[R]}$.
\end{proposition}

\begin{proposition}\label{reconstruction_asserted_base}
    Let $R,X$ be sets.
    Suppose $R$ is a coordinate record on $X$.
    Then $(coordinate record on X based on R)=
        \structure[coordinate record on X]{\atlas=\atlas[R]}$.
\end{proposition}

\begin{proposition}\label{reconstruction_parent_base}
    Let $R$ be a record.
    Then $(coordinate record on X based on R with \atlas=A)=
        \structure[coordinate record on X]{\atlas=A}$.
\end{proposition}

\begin{proposition}\label{reconstruction_drops_extra_fields}
    Let $R$ be a mixed coordinate on $X$.
    Then $(coordinate record on X based on R)=
        \structure[coordinate record on X]{\atlas=\atlas[R]}$.
\end{proposition}

\begin{proposition}\label{reconstruction_diamond}
    Let $R$ be a coordinate diamond on $X$.
    Then $(coordinate diamond on X based on R with \atlas=A)=
        \structure[coordinate diamond on X]{\atlas=A}$.
\end{proposition}

\begin{proposition}\label{reconstruction_empty}
    Let $R$ be a record.
    Then $(record based on R)=\structure[record]{}$.
\end{proposition}

\begin{proposition}\label{reconstruction_nested}
    Let $R$ be a coordinate record on $X$.
    Then $(coordinate record on X based on
        (coordinate record on X based on R with \atlas=A) with \atlas=B)=
        \structure[coordinate record on X]{\atlas=B}$.
\end{proposition}

\begin{proposition}\label{reconstruction_aggregate_base}
    Then $(coordinate record on X based on \structure[coordinate record on X]{\atlas=A})=
        \structure[coordinate record on X]{\atlas=A}$.
\end{proposition}
\begin{proof}
    Follows by \auto{coordinate_recordaggregateprojection_atlas}.
\end{proof}

\begin{proposition}\label{reconstruction_reduct_base}
    Let $R$ be a mixed coordinate on $X$.
    Then $(coordinate record on X based on \reduct[coordinate record on X]{R})=
        \structure[coordinate record on X]{\atlas=\atlas[R]}$.
\end{proposition}
\begin{proof}
    Follows by \auto{coordinate_recordreductprojection_atlas}.
\end{proof}

\begin{proposition}\label{reconstruction_projection}
    Let $R$ be a coordinate record on $X$.
    Then $\atlas[(coordinate record on X based on R with \atlas=A)]=A$.
\end{proposition}
\begin{proof}
    Follows by \obv{coordinate_recordaggregateprojection_atlas}.
\end{proof}

\begin{instance}\label{reconstructed_empty_coordinate}
    The coordinate record $\ReconstructedEmptyCoordinate$ on $\emptyset$ is
        $(coordinate record on \emptyset based on \structure[record]{}
            with \atlas=\emptyset)$.
\end{instance}
\begin{proof}
    Follows by \auto{coordinate_record,emptyset_subseteq,
        coordinate_recordaggregateprojection_atlas}.
\end{proof}

\begin{proposition}\label{reconstructed_empty_coordinate_atlas}
    $\atlas[\ReconstructedEmptyCoordinate]=\emptyset$.
\end{proposition}
\begin{proof}
    Follows by \obv{reconstructed_empty_coordinateprojection_atlas}.
\end{proof}
