\begin{proposition}\label{forgetting_atlas_is_not_equality}
    Let $M,N$ be equipped coordinate spaces.
    Suppose $M$ is an exact equipped coordinate space.
    Suppose $N$ is an exact equipped coordinate space.
    Suppose $\carrier[M]=\carrier[N]$ and $\opens[M]=\opens[N]$.
    Then $M=N$.
\end{proposition}
\begin{proof}
    Follows by structure extensionality.
\end{proof}
