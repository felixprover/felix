\end{proof}

\begin{theorem}\label{positive_atlas_import_probe}
    Then $\RealOneChart$ is a coordinate atlas on $\RealTopology$ under $\PositiveRealPseudogroup$.
\end{theorem}
\begin{proof}
    $\RealTopology$ is a topological space by \obv{real_topology_is_topological_space}.
    $\PositiveRealPseudogroup$ is a coordinate pseudogroup by \obv{positive_real_pseudogroup}.
    $\model[\PositiveRealPseudogroup]=\RealTopology$ by
        \obv{positive_real_pseudogroup_model}.
    $\carrier[\RealTopology]=\reals$ by \obv{real_topology_carrier}.
    $\reals\in\opens[\RealTopology]$ by \obv{carrier_open}.
    $\identity{\reals}$ is an open partial homeomorphism from $\RealTopology$ to
        $\RealTopology$ by \obv{identity_open_partial_homeomorphism}.
    $\dom{\identity{\reals}}=\reals$ by \obv{id_dom}.
    Follows by \auto{real_one_chart,single_chart_atlas}.
\end{proof}
