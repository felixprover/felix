\section{Coordinate constructions}

These small models exercise the record interface, not manifold theory.
Relative structures store only an atlas; equipped spaces also store their
underlying topology. Every construction retains its stated hypotheses in
its public structure, exactness, and projection theorems.

\begin{construction}\label{underlying_topological_space}
    Let $M$ be an equipped coordinate space.
    The topological space $\UnderlyingTopologicalSpace{M}=
        topological space based on M$.
\end{construction}
\begin{proof}
    $\opens[M]$ is a topology on $\carrier[M]$
        by \obv{topological_space_opens_form_topology}.
    Follows by \auto{topology_family,
        topological_spaceaggregateprojection_carrier,
        topological_spaceaggregateprojection_opens}.
\end{proof}

\begin{construction}\label{open_restriction}
    Let $M$ be an equipped coordinate space.
    Suppose $U$ is open in $M$.
    The equipped coordinate space $\OpenRestriction{M}{U}=
        equipped coordinate space based on M with
        \carrier=U and \opens=\subspaceOpens{M}{U}
        and \atlas=\RestrictedAtlas{M}{U}$.
\end{construction}
\begin{proof}
    $\subspaceOpens{M}{U}$ is a topology on $U$
        by \obv{open_subspace_topology}.
    $\RestrictedAtlas{M}{U}\subseteq\subspaceOpens{M}{U}$
        by \obv{restricted_atlas_open}.
    Follows by \auto{topology_family,
        equipped_coordinate_spaceaggregateprojection_carrier,
        equipped_coordinate_spaceaggregateprojection_opens,
        equipped_coordinate_spaceaggregateprojection_atlas}.
\end{proof}

\begin{construction}\label{generated_coordinate_structure}
    Let $X$ be a topological space.
    Suppose $A$ is an admissible atlas on $X$.
    The relative coordinate structure
        $\GeneratedCoordinateStructure{X}{A}=
            \structure{\atlas=\MaximalAtlas{X}{A}}$ on $X$.
\end{construction}
\begin{proof}
    $\MaximalAtlas{X}{A}$ is an admissible atlas on $X$
        by \obv{maximal_atlas_admissible}.
    Follows by \auto{relative_coordinate_structureaggregateprojection_atlas}.
\end{proof}

\begin{construction}\label{equip}
    Let $X$ be a topological space.
    Suppose $A$ is an admissible atlas on $X$.
    The equipped coordinate space $\Equip{X}{A}=
        equipped coordinate space based on X with
        \atlas=\atlas[\GeneratedCoordinateStructure{X}{A}]$.
\end{construction}
\begin{proof}
    $\opens[X]$ is a topology on $\carrier[X]$
        by \obv{topological_space_opens_form_topology}.
    $\atlas[\GeneratedCoordinateStructure{X}{A}]=\MaximalAtlas{X}{A}$
        by \obv{generated_coordinate_structureprojection_atlas}.
    $\MaximalAtlas{X}{A}\subseteq\opens[X]$
        by \auto{maximal_atlas_admissible,admissible_atlas}.
    Follows by \auto{topology_family,
        equipped_coordinate_spaceaggregateprojection_carrier,
        equipped_coordinate_spaceaggregateprojection_opens,
        equipped_coordinate_spaceaggregateprojection_atlas}.
\end{proof}

\begin{construction}\label{weaken_coordinate_structure}
    Let $M$ be an equipped coordinate space.
    Let $H$ be a coordinate pseudogroup.
    Suppose $H$ is an admissible weakening for $M$.
    The equipped coordinate space $\WeakenCoordinateStructure{M}{H}=
        equipped coordinate space based on M with \atlas=\domains[H]$.
\end{construction}
\begin{proof}
    $\opens[M]$ is a topology on $\carrier[M]$
        by \obv{topological_space_opens_form_topology}.
    $\domains[H]\subseteq\opens[M]$ by \obv{admissible_weakening}.
    Follows by \auto{topology_family,
        equipped_coordinate_spaceaggregateprojection_carrier,
        equipped_coordinate_spaceaggregateprojection_opens,
        equipped_coordinate_spaceaggregateprojection_atlas}.
\end{proof}

\begin{theorem}\label{weaken_coordinate_structure_allowed}
    Let $M$ be an equipped coordinate space.
    Let $H$ be a coordinate pseudogroup.
    Suppose $H$ is an admissible weakening for $M$.
    Then $\WeakenCoordinateStructure{M}{H}$ has an atlas allowed by $H$.
\end{theorem}
\begin{proof}
    $\WeakenCoordinateStructure{M}{H}$ is an equipped coordinate space
        by \obv{weaken_coordinate_structure}.
    $\atlas[\WeakenCoordinateStructure{M}{H}]=\domains[H]$
        by \obv{weaken_coordinate_structureprojection_atlas}.
    Follows by \obv{allowed_atlas,subseteq_refl}.
\end{proof}

\section{Equality through the public interface}

\begin{theorem}\label{generated_coordinate_structure_equal}
    Let $X$ be a topological space.
    Suppose $A$ is an admissible atlas on $X$.
    Suppose $B$ is an admissible atlas on $X$.
    Suppose $A\union\{\carrier[X]\}=B\union\{\carrier[X]\}$.
    Then $\GeneratedCoordinateStructure{X}{A}=\GeneratedCoordinateStructure{X}{B}$.
\end{theorem}
\begin{proof}
    $\GeneratedCoordinateStructure{X}{A}$ is an exact relative coordinate structure on $X$
        by \obv{generated_coordinate_structure,generated_coordinate_structureshape}.
    $\GeneratedCoordinateStructure{X}{B}$ is an exact relative coordinate structure on $X$
        by \obv{generated_coordinate_structure,generated_coordinate_structureshape}.
    $\MaximalAtlas{X}{A}=\MaximalAtlas{X}{B}$ by \obv{maximal_atlas_equal}.
    $\atlas[\GeneratedCoordinateStructure{X}{A}]=\atlas[\GeneratedCoordinateStructure{X}{B}]$
        by \obv{generated_coordinate_structureprojection_atlas}.
    Follows by structure extensionality.
\end{proof}

\begin{theorem}\label{equip_equal}
    Let $X$ be a topological space.
    Suppose $A$ is an admissible atlas on $X$.
    Suppose $B$ is an admissible atlas on $X$.
    Suppose $A\union\{\carrier[X]\}=B\union\{\carrier[X]\}$.
    Then $\Equip{X}{A}=\Equip{X}{B}$.
\end{theorem}
\begin{proof}
    Let $P=\Equip{X}{A}$ and $Q=\Equip{X}{B}$.
    $P$ is an exact equipped coordinate space by \auto{equip,equipshape}.
    $Q$ is an exact equipped coordinate space by \auto{equip,equipshape}.
    $\carrier[P]=\carrier[Q]$ by \auto{equipprojection_carrier}.
    $\opens[P]=\opens[Q]$ by \auto{equipprojection_opens}.
    $\GeneratedCoordinateStructure{X}{A}=\GeneratedCoordinateStructure{X}{B}$
        by \obv{generated_coordinate_structure_equal}.
    $\atlas[P]=\atlas[Q]$ by \auto{equipprojection_atlas}.
    Follows by structure extensionality.
\end{proof}


\section{Composing the interface}

\begin{proposition}\label{coordinate_construction_chain}
    Let $X$ be a topological space.
    Suppose $A$ is an admissible atlas on $X$.
    Suppose $U$ is open in $\Equip{X}{A}$.
    Then $topological space based on
        \UnderlyingTopologicalSpace{\OpenRestriction{\Equip{X}{A}}{U}}
        =\structure[topological space]{\carrier=U,
            \opens=\subspaceOpens{\Equip{X}{A}}{U}}$.
\end{proposition}
\begin{proof}
    $\Equip{X}{A}$ is an equipped coordinate space by \obv{equip}.
    $\OpenRestriction{\Equip{X}{A}}{U}$ is an equipped coordinate space
        by \obv{open_restriction}.
    Follows by \auto{underlying_topological_spaceprojection_carrier,
        underlying_topological_spaceprojection_opens,
        open_restrictionprojection_carrier,open_restrictionprojection_opens}.
\end{proof}

\section{Forgetting fields and congruence}

\begin{proposition}\label{topological_reduct_exact}
    Let $X$ be a topological space.
    Then $\reduct[topological space]{X}$ is an exact topological space.
\end{proposition}
\begin{proof}
    $\opens[X]$ is a topology on $\carrier[X]$
        by \obv{topological_space_opens_form_topology}.
    Follows by \auto{topological_space,topology_family,topological_spacereductexact,
        topological_spacereductprojection_carrier,topological_spacereductprojection_opens}.
\end{proof}

\begin{proposition}\label{underlying_topological_space_reduct}
    Let $M$ be an equipped coordinate space.
    Then $\UnderlyingTopologicalSpace{M}=\reduct[topological space]{M}$.
\end{proposition}
\begin{proof}
    $M$ is a topological space by \obv{equipped_coordinate_space}.
    $\UnderlyingTopologicalSpace{M}$ is an exact topological space
        by \obv{underlying_topological_space,underlying_topological_spaceshape}.
    $\reduct[topological space]{M}$ is an exact topological space
        by \obv{topological_reduct_exact}.
    $\carrier[\UnderlyingTopologicalSpace{M}]=\carrier[\reduct[topological space]{M}]$
        by \auto{underlying_topological_spaceprojection_carrier,topological_spacereductprojection_carrier}.
    $\opens[\UnderlyingTopologicalSpace{M}]=\opens[\reduct[topological space]{M}]$
        by \auto{underlying_topological_spaceprojection_opens,topological_spacereductprojection_opens}.
    Follows by structure extensionality.
\end{proof}

\begin{proposition}\label{topological_reduct_idempotent}
    Let $X$ be a topological space.
    Then $\reduct[topological space]{\reduct[topological space]{X}}=\reduct[topological space]{X}$.
\end{proposition}
\begin{proof}
    $\reduct[topological space]{X}$ is an exact topological space
        by \obv{topological_reduct_exact}.
    $\reduct[topological space]{\reduct[topological space]{X}}$ is an exact topological space
        by \obv{topological_reduct_exact}.
    $\carrier[\reduct[topological space]{\reduct[topological space]{X}}]=
        \carrier[\reduct[topological space]{X}]$
        by \obv{topological_spacereductprojection_carrier}.
    $\opens[\reduct[topological space]{\reduct[topological space]{X}}]=
        \opens[\reduct[topological space]{X}]$
        by \obv{topological_spacereductprojection_opens}.
    Follows by structure extensionality.
\end{proof}

\begin{proposition}\label{underlying_carrier_reduct_composition}
    Let $M$ be an equipped coordinate space.
    Then $\reduct[onesorted structure]{\reduct[topological space]{M}}=
        \reduct[onesorted structure]{M}$.
\end{proposition}
\begin{proof}
    $M$ is a topological space by \obv{equipped_coordinate_space}.
    $\reduct[topological space]{M}$ is a topological space by \obv{topological_reduct_exact}.
    $\reduct[onesorted structure]{\reduct[topological space]{M}}$ is an exact onesorted structure
        by \auto{onesorted_structure,onesorted_structurereductexact,topological_space}.
    $\reduct[onesorted structure]{M}$ is an exact onesorted structure
        by \auto{onesorted_structure,onesorted_structurereductexact,topological_space}.
    $\carrier[\reduct[topological space]{M}]=\carrier[M]$
        by \obv{topological_spacereductprojection_carrier}.
    $\carrier[\reduct[onesorted structure]{\reduct[topological space]{M}}]=
        \carrier[\reduct[onesorted structure]{M}]$
        by \auto{onesorted_structurereductprojection_carrier}.
    Follows by structure extensionality.
\end{proof}

\begin{proposition}\label{equip_space_congruence}
    Let $X,Y$ be topological spaces.
    Suppose $A$ is an admissible atlas on $X$.
    Suppose $A$ is an admissible atlas on $Y$.
    Suppose $X=Y$.
    Then $\Equip{X}{A}=\Equip{Y}{A}$.
\end{proposition}

\begin{proposition}\label{equip_atlas_congruence}
    Let $X$ be a topological space.
    Suppose $A$ is an admissible atlas on $X$.
    Suppose $B$ is an admissible atlas on $X$.
    Suppose $A=B$.
    Then $\Equip{X}{A}=\Equip{X}{B}$.
\end{proposition}
