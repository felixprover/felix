\section{Constructions from classified exact records}

\begin{proposition}\label{classified_bundle_reconstruction}
    Let $X$ be a set.
    Then $coordinate record on X based on \ClassifiedCoordinate{X}
        =\structure[coordinate record on X]{\atlas[\ClassifiedCoordinate{X}]}$.
\end{proposition}
\begin{proof}
    Follows.
\end{proof}

\begin{construction}\label{copy_classified_coordinate}
    Let $X$ be a set.
    Let $R$ be a set.
    Suppose $R$ is an exact coordinate record on $X$.
    The coordinate record $\CopyClassifiedCoordinate{X}{R}=
        coordinate record on X based on R$ on $X$.
\end{construction}
\begin{proof}
    $\atlas[R]\subseteq X$ by \obv{coordinate_atlas_bound}.
    Follows by \auto{coordinate_recordaggregateprojection_atlas}.
\end{proof}

\begin{proposition}\label{copy_classified_bundle_identity}
    Let $X$ be a set.
    Then $\CopyClassifiedCoordinate{X}{\ClassifiedCoordinate{X}}=\ClassifiedCoordinate{X}$.
\end{proposition}
\begin{proof}
    $\ClassifiedCoordinate{X}$ is an exact coordinate record on $X$
        by \obv{coordinate_classification}.
    $\CopyClassifiedCoordinate{X}{\ClassifiedCoordinate{X}}$ is an exact coordinate record on $X$
        by \obv{copy_classified_coordinate,copy_classified_coordinateshape}.
    $\atlas[\CopyClassifiedCoordinate{X}{\ClassifiedCoordinate{X}}]=\atlas[\ClassifiedCoordinate{X}]$
        by \obv{copy_classified_coordinateprojection_atlas}.
    Follows by structure extensionality.
\end{proof}

\begin{proposition}\label{classified_copy_remains_total}
    Let $X$ be a set.
    Let $R$ be a coordinate record on $X$.
    Then $\CopyClassifiedCoordinate{X}{R}=\CopyClassifiedCoordinate{X}{R}$.
\end{proposition}
