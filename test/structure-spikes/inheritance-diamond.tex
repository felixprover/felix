\begin{struct}\label{coordinate_left}
    Let $X$ be a set.
    A coordinate left $R$ on $X$ is a coordinate record on $X$.
\end{struct}

\begin{struct}\label{coordinate_right}
    Let $X$ be a set.
    A coordinate right $R$ on $X$ is a coordinate record on $X$.
\end{struct}

\begin{struct}\label{coordinate_diamond}
    Let $X$ be a set.
    A coordinate diamond $R$ on $X$ is a coordinate left on $X$ and a coordinate right on $X$.
\end{struct}

\begin{struct}\label{mixed_coordinate}
    Let $X$ be a set.
    A mixed coordinate $R$ on $X$ is a coordinate record on $X$ and a onesorted structure.
\end{struct}

\begin{struct}\label{reverse_mixed_coordinate}
    Let $X$ be a set.
    A reverse mixed coordinate $R$ on $X$ is a onesorted structure and a coordinate record on $X$.
\end{struct}

\begin{proposition}\label{mixed_field_order}
    Let $X,A$ be sets.
    Then $\structure[mixed coordinate on X]{X,A}=\structure[reverse mixed coordinate on X]{A,X}$.
\end{proposition}

\begin{proposition}\label{mixed_carrier_projection}
    Let $X,A$ be sets.
    Then $\carrier[\structure[mixed coordinate on X]{X,A}]=A$.
\end{proposition}
\begin{proof}
    Follows by \obv{mixed_coordinateaggregateprojection_carrier}.
\end{proof}

\begin{proposition}\label{mixed_carrier_membership}
    Let $R$ be a mixed coordinate on $X$.
    Let $a\in R$.
    Then $a\in\carrier[R]$.
\end{proposition}
\begin{proof}
    Follows by assumption.
\end{proof}

\begin{proposition}\label{coordinate_diamond_inheritance}
    Let $R$ be a coordinate diamond on $X$.
    Then $R$ is a coordinate record on $X$.
\end{proposition}
\begin{proof}
    Follows by notion.
\end{proof}
