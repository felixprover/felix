\begin{definition}\label{classified_coordinate_definition}
    Let $X$ be a set.
    $\ClassifiedCoordinate{X}=\FullCoordinate{X}$.
\end{definition}

\begin{classification}\label{coordinate_classification}
    Let $X$ be a set.
    Then $\ClassifiedCoordinate{X}$ is an exact coordinate record on $X$.
\end{classification}
\begin{proof}
    $\FullCoordinate{X}$ is an exact coordinate record on $X$
        by \obv{full_coordinate_family,full_coordinate_familyshape}.
    Follows by \auto{classified_coordinate_definition}.
\end{proof}
