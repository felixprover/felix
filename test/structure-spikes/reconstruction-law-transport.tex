\section{The laws of a reconstructed record}

\begin{definition}\label{bounded_parameter_copy}
    Let $X,Y$ be sets.
    Let $R$ be a coordinate record on $X$.
    $\BoundedParameterCopy{X}{Y}{R}=
        coordinate record on Y based on R$.
\end{definition}

\begin{proposition}\label{bounded_parameter_copy_result}
    Let $X,Y$ be sets.
    Let $R$ be a coordinate record on $X$.
    Suppose $\atlas[R]\subseteq Y$.
    Then $\BoundedParameterCopy{X}{Y}{R}$ is a coordinate record on $Y$.
\end{proposition}
\begin{proof}
    Follows by \auto{bounded_parameter_copy,coordinate_record,
        coordinate_recordaggregateexact,coordinate_recordaggregateprojection_atlas}.
\end{proof}

\begin{construction}\label{bounded_field_copy}
    Let $X,B$ be sets.
    Let $R$ be a coordinate record on $X$.
    Suppose $B\subseteq X$.
    The coordinate record $\BoundedFieldCopy{X}{B}{R}=
        coordinate record on X based on R with \atlas=B$ on $X$.
\end{construction}
\begin{proof}
    Follows by \auto{coordinate_recordaggregateprojection_atlas}.
\end{proof}
