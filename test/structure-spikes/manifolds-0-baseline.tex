\section{The existing-syntax baseline}

These five definitions freeze the provider arguments and the complete ordered
headers before introducing construction syntax. Exactness and projections are
separate facts here; the new declarations should provide those facts directly.

\begin{definition}\label{underlying_topological_space}
    $\UnderlyingTopologicalSpace{M}=\structure[topological space]{
        \carrier=\carrier[M],\opens=\opens[M]}$.
\end{definition}

\begin{classification}\label{underlying_topological_space_result}
    Let $M$ be an equipped coordinate space.
    Then $\UnderlyingTopologicalSpace{M}$ is a topological space.
\end{classification}
\begin{proof}
    $\opens[M]$ is a topology on $\carrier[M]$
        by \obv{topological_space_opens_form_topology}.
    Follows by \auto{underlying_topological_space,topological_space,
        topology_family,topological_spaceaggregateexact,
        topological_spaceaggregateprojection_carrier,
        topological_spaceaggregateprojection_opens}.
\end{proof}

\begin{definition}\label{open_restriction}
    $\OpenRestriction{M}{U}=\structure[equipped coordinate space]{
        \carrier=U,\opens=\subspaceOpens{M}{U},
        \atlas=\RestrictedAtlas{M}{U}}$.
\end{definition}

\begin{classification}\label{open_restriction_result}
    Let $M$ be an equipped coordinate space.
    Suppose $U$ is open in $M$.
    Then $\OpenRestriction{M}{U}$ is an equipped coordinate space.
\end{classification}
\begin{proof}
    $\subspaceOpens{M}{U}$ is a topology on $U$
        by \obv{open_subspace_topology}.
    $\RestrictedAtlas{M}{U}\subseteq\subspaceOpens{M}{U}$
        by \obv{restricted_atlas_open}.
    Follows by \auto{open_restriction,equipped_coordinate_space,
        coordinate_datum,topological_space,topology_family,
        equipped_coordinate_spaceaggregateexact,
        equipped_coordinate_spaceaggregateprojection_carrier,
        equipped_coordinate_spaceaggregateprojection_opens,
        equipped_coordinate_spaceaggregateprojection_atlas}.
\end{proof}

\begin{definition}\label{generated_coordinate_structure}
    $\GeneratedCoordinateStructure{X}{A}=\structure[relative coordinate structure on X]{
        \atlas=\MaximalAtlas{X}{A}}$.
\end{definition}

\begin{classification}\label{generated_coordinate_structure_result}
    Let $X$ be a topological space.
    Suppose $A$ is an admissible atlas on $X$.
    Then $\GeneratedCoordinateStructure{X}{A}$ is a relative coordinate structure on $X$.
\end{classification}
\begin{proof}
    $\MaximalAtlas{X}{A}$ is an admissible atlas on $X$
        by \obv{maximal_atlas_admissible}.
    Follows by \auto{generated_coordinate_structure,relative_coordinate_structure,
        coordinate_datum,relative_coordinate_structureaggregateexact,
        relative_coordinate_structureaggregateprojection_atlas}.
\end{proof}

\begin{proposition}\label{generated_coordinate_structureprojection_atlas}
    Let $X$ be a topological space.
    Suppose $A$ is an admissible atlas on $X$.
    Then $\atlas[\GeneratedCoordinateStructure{X}{A}]=\MaximalAtlas{X}{A}$.
\end{proposition}
\begin{proof}
    Follows by \auto{generated_coordinate_structure,
        relative_coordinate_structureaggregateprojection_atlas}.
\end{proof}

\begin{theorem}\label{generated_coordinate_structureshape}
    Let $X$ be a topological space.
    Suppose $A$ is an admissible atlas on $X$.
    Then $\GeneratedCoordinateStructure{X}{A}$ is an exact relative coordinate structure on $X$.
\end{theorem}
\begin{proof}
    $\GeneratedCoordinateStructure{X}{A}$ is a relative coordinate structure on $X$
        by \obv{generated_coordinate_structure_result}.
    Follows by \auto{generated_coordinate_structure,
        relative_coordinate_structureaggregateexact}.
\end{proof}

\begin{definition}\label{equip}
    $\Equip{X}{A}=\structure[equipped coordinate space]{
        \carrier=\carrier[X],\opens=\opens[X],
        \atlas=\atlas[\GeneratedCoordinateStructure{X}{A}]}$.
\end{definition}

\begin{classification}\label{equip_result}
    Let $X$ be a topological space.
    Suppose $A$ is an admissible atlas on $X$.
    Then $\Equip{X}{A}$ is an equipped coordinate space.
\end{classification}
\begin{proof}
    $\opens[X]$ is a topology on $\carrier[X]$
        by \obv{topological_space_opens_form_topology}.
    $\atlas[\GeneratedCoordinateStructure{X}{A}]=\MaximalAtlas{X}{A}$
        by \obv{generated_coordinate_structureprojection_atlas}.
    $\MaximalAtlas{X}{A}\subseteq\opens[X]$
        by \auto{maximal_atlas_admissible,admissible_atlas}.
    Follows by \auto{equip,equipped_coordinate_space,coordinate_datum,
        topological_space,topology_family,equipped_coordinate_spaceaggregateexact,
        equipped_coordinate_spaceaggregateprojection_carrier,
        equipped_coordinate_spaceaggregateprojection_opens,
        equipped_coordinate_spaceaggregateprojection_atlas}.
\end{proof}

\begin{proposition}\label{equipprojection_carrier}

\begin{proposition}\label{equipprojection_opens}

\begin{proposition}\label{equipprojection_atlas}

\begin{theorem}\label{equipshape}
    Let $X$ be a topological space.

\end{theorem}
\begin{proof}

\begin{definition}\label{weaken_coordinate_structure}

\begin{classification}\label{weaken_coordinate_structure_result}

\begin{theorem}\label{weaken_coordinate_structure_allowed}

    Follows by \obv{allowed_atlas,subseteq_refl}.
\end{proof}

\section{Equality through the public interface}

\begin{theorem}\label{generated_coordinate_structure_equal}
    Let $X$ be a topological space.
    Suppose $A$ is an admissible atlas on $X$.
    Suppose $B$ is an admissible atlas on $X$.
    Suppose $A\union\{\carrier[X]\}=B\union\{\carrier[X]\}$.
    Then $\GeneratedCoordinateStructure{X}{A}=\GeneratedCoordinateStructure{X}{B}$.
\end{theorem}
\begin{proof}
    $\GeneratedCoordinateStructure{X}{A}$ is an exact relative coordinate structure on $X$
        by \obv{generated_coordinate_structureshape}.
    $\GeneratedCoordinateStructure{X}{B}$ is an exact relative coordinate structure on $X$
        by \obv{generated_coordinate_structureshape}.
    $\MaximalAtlas{X}{A}=\MaximalAtlas{X}{B}$ by \obv{maximal_atlas_equal}.
    $\atlas[\GeneratedCoordinateStructure{X}{A}]=\atlas[\GeneratedCoordinateStructure{X}{B}]$
        by \obv{generated_coordinate_structureprojection_atlas}.
    Follows by structure extensionality.
\end{proof}

\begin{theorem}\label{equip_equal}
    Let $X$ be a topological space.
    Suppose $A$ is an admissible atlas on $X$.
    Suppose $B$ is an admissible atlas on $X$.
    Suppose $A\union\{\carrier[X]\}=B\union\{\carrier[X]\}$.
    Then $\Equip{X}{A}=\Equip{X}{B}$.
\end{theorem}
\begin{proof}
    Let $P=\Equip{X}{A}$ and $Q=\Equip{X}{B}$.
    $P$ is an exact equipped coordinate space by \auto{equipshape}.
    $Q$ is an exact equipped coordinate space by \auto{equipshape}.
    $\carrier[P]=\carrier[Q]$ by \auto{equipprojection_carrier}.
    $\opens[P]=\opens[Q]$ by \auto{equipprojection_opens}.
    $\GeneratedCoordinateStructure{X}{A}=\GeneratedCoordinateStructure{X}{B}$
        by \obv{generated_coordinate_structure_equal}.
    $\atlas[P]=\atlas[Q]$ by \auto{equipprojection_atlas}.
    Follows by structure extensionality.
\end{proof}
