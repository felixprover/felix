The membership premise below is mathematically equivalent to openness. The
ordinary classification may use the finite-rule fallback, but its fresh
projection is outside the provider-result registration fragment. Reconstruction
discovery must not mistake that fallback for a provider-result registration.

\begin{definition}\label{direct_open_restriction}
    $\DirectOpenRestriction{M}{U}=\OpenRestriction{M}{U}$.
\end{definition}

\begin{classification}\label{direct_open_restriction_result}
    Let $M$ be an equipped coordinate space.
    Let $U\in\opens[M]$.
    Then $\DirectOpenRestriction{M}{U}$ is an equipped coordinate space.
\end{classification}
\begin{proof}
    $U$ is open in $M$ by \obv{is_open}.
    Follows by \auto{direct_open_restriction,open_restriction_result}.
\end{proof}

\begin{proposition}\label{direct_membership_discovery_rejected}
    Let $M$ be an equipped coordinate space.
    Let $U\in\opens[M]$.
    Then $(equipped coordinate space based on \DirectOpenRestriction{M}{U})=
        \structure[equipped coordinate space]{
            \carrier=\carrier[\DirectOpenRestriction{M}{U}],
            \opens=\opens[\DirectOpenRestriction{M}{U}],
            \atlas=\atlas[\DirectOpenRestriction{M}{U}]}$.
\end{proposition}
