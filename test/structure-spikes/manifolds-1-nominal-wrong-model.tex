\import{topology/full-pseudogroup.tex}

\begin{definition}\label{nominal_model_guard}
    Let $E$ be a topological space.
    Let $G$ be a pseudogroup on $E$.
    $A$ is supervised by $G$ on $E$ iff $A=A$.
\end{definition}

\begin{proposition}\label{nominal_wrong_model}
    Let $E,F$ be topological spaces.
    Suppose $E\neq F$.
    Then $\emptyset$ is supervised by $\FullPseudogroup{E}$ on $F$.
\end{proposition}
\begin{proof}
    Follows by notion.
\end{proof}
