\begin{struct}\label{foreign_coordinate_datum}
    A foreign coordinate datum $R$ is a record equipped with
    \begin{enumerate}
        \item a set $\atlas[R]$;
    \end{enumerate}.
\end{struct}

\begin{proposition}\label{reconstruction_disambiguates_projection}
    Let $X$ be a topological space.
    Suppose $A$ is an admissible atlas on $X$.
    Then $\atlas[(coordinate datum based on \Equip{X}{A})]=
        \atlas[\reduct[coordinate datum]{\Equip{X}{A}}]$.
\end{proposition}
\begin{proof}
    Follows by \auto{coordinate_datumaggregateprojection_atlas,
        coordinate_datumreductprojection_atlas}.
\end{proof}

\begin{proposition}\label{alias_disambiguates_projection}
    Let $X$ be a topological space.
    Suppose $A$ is an admissible atlas on $X$.
    Then $\atlas[(coordinate datum based on \Equip{X}{A})]=
        \atlas[\reduct[coordinate datum]{\Equip{X}{A}}]$.
\end{proposition}
\begin{proof}
    Let $N=\Equip{X}{A}$.
    $N$ is an equipped coordinate space by \auto{equip}.
    $\atlas[\reduct[coordinate datum]{N}]=\MaximalAtlas{X}{A}$
        by \auto{equipprojection_atlas,coordinate_datumreductprojection_atlas,
            generated_coordinate_structureprojection_atlas}.
    $\atlas[(coordinate datum based on N)]=\atlas[\reduct[coordinate datum]{N}]$
        by \auto{coordinate_datumaggregateprojection_atlas,
            coordinate_datumreductprojection_atlas}.
\end{proof}
