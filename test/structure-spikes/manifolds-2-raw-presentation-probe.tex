\section{Imported raw-presentation interfaces}

\import{topology/coordinate-map-examples.tex}

\begin{theorem}\label{raw_atlas_topology_consumer}
    Let $G$ be a coordinate pseudogroup and $X$ be a topological space.
    Let $A$ be a coordinate atlas on $X$ under $G$.
    Then $\TopologicalReduct{\SpaceFromCharts{\carrier[X]}{\model[G]}{A}}=\TopologicalReduct{X}$.
\end{theorem}
\begin{proof}
    $\model[G]$ is a topological space by \obv{coordinate_pseudogroup}.
    $A$ is a topological coordinate presentation on $\carrier[X]$ modelled on $\model[G]$ by
        \obv{coordinate_atlas_raw_presentation}.
    Every element of $A$ is an open partial homeomorphism from $X$ to $\model[G]$ by
        \obv{coordinate_atlas}.
    For every $x\in\carrier[X]$ there exists $c\in A$ such that $x\in\dom{c}$ by
        \obv{coordinate_atlas}.
    $\CoordinateTopology{\carrier[X]}{\model[G]}{A}=\TopologicalReduct{X}$ by
        \obv{coordinate_topology_recovers_reduct}.
    Follows by \obv{space_from_charts_topological_reduct}.
\end{proof}

\begin{theorem}\label{raw_space_presentation_consumer}
    Let $E$ be a topological space.
    Let $A$ be a topological coordinate presentation on $X$ modelled on $E$.
    Then $A$ presents $\SpaceFromCharts{X}{E}{A}$ on $\SpaceFromCharts{X}{E}{A}$
        under $\FullPseudogroup{E}$ and
        $\CoordinateStructure{\SpaceFromCharts{X}{E}{A}}
            =\GeneratedCoordinates{\FullPseudogroup{E}}{\CoordinateTopology{X}{E}{A}}{A}$.
\end{theorem}
\begin{proof}
    Follows by \obv{space_from_charts_presentation,space_from_charts_coordinate_structure}.
\end{proof}

\begin{theorem}\label{raw_space_chart_consumer}
    Let $E$ be a topological space.
    Let $A$ be a topological coordinate presentation on $X$ modelled on $E$.
    Then $c\in\atlas[\SpaceFromCharts{X}{E}{A}]$ iff
        $c$ is an open partial homeomorphism from $\SpaceFromCharts{X}{E}{A}$ to $E$.
\end{theorem}
\begin{proof}
    Follows by \obv{space_from_charts_chart_iff}.
\end{proof}

\begin{theorem}\label{raw_space_recovery_consumer}
    Let $E$ be a topological space.
    Suppose $M$ is an exact equipped coordinate space.
    Suppose $A$ presents $M$ on $M$ under $\FullPseudogroup{E}$.
    Then $\SpaceFromCharts{\carrier[M]}{E}{A}=M$.
\end{theorem}
\begin{proof}
    Follows by \obv{space_from_presented_coordinates}.
\end{proof}

\begin{theorem}\label{raw_space_equality_consumer}
    Let $E$ be a topological space.
    Suppose $A$ is a topological coordinate presentation on $X$ modelled on $E$ and
        $B$ is a topological coordinate presentation on $X$ modelled on $E$.
    Then $\SpaceFromCharts{X}{E}{A}=\SpaceFromCharts{X}{E}{B}$ iff
        $\CoordinateTopology{X}{E}{A}=\CoordinateTopology{X}{E}{B}$.
\end{theorem}
\begin{proof}
    Follows by \obv{space_from_charts_equal_iff}.
\end{proof}
