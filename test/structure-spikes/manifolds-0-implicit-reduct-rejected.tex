\begin{proposition}\label{open_inheritance_is_not_an_exact_reduct}
    Let $M$ be an equipped coordinate space.
    Then $M=\UnderlyingTopologicalSpace{M}$.
\end{proposition}
\begin{proof}
    $\UnderlyingTopologicalSpace{M}$ is an exact topological space
        by \auto{underlying_topological_space,underlying_topological_spaceshape}.
    Follows by structure extensionality.
\end{proof}
