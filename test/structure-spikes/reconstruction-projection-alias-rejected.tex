\begin{proposition}\label{ordinary_alias_projection_stays_ambiguous}
    Let $X$ be a topological space.
    Suppose $A$ is an admissible atlas on $X$.
    Then $X=X$.
\end{proposition}
\begin{proof}
    Let $N=\Equip{X}{A}$.
    $N$ is an equipped coordinate space by \obv{equip}.
    $\atlas[N]=\atlas[N]$.
    Follows.
\end{proof}
