\section{A ten-field diamond}

\begin{struct}\label{wide_base}
    A wide base $R$ is a record equipped with
    \begin{enumerate}
        \item a set $\north[R]$;
        \item a set $\south[R]$;
        \item a set $\east[R]$;
        \item a set $\west[R]$;
        \item a set $\center[R]$;
        \item a set $\upper[R]$;
        \item a set $\lower[R]$;
        \item a set $\inner[R]$;
        \item a set $\outer[R]$;
        \item a set $\middle[R]$;
    \end{enumerate}.
\end{struct}

\begin{struct}\label{wide_left}
    A wide left $R$ is a wide base.
\end{struct}

\begin{struct}\label{wide_right}
    A wide right $R$ is a wide base.
\end{struct}

\begin{struct}\label{wide_record}
    A wide record $R$ is a wide left and a wide right.
\end{struct}
