\import{set.tex}

\begin{struct}\label{instance_focus_probe_token}
    A focus probe token $T$ is a record.
\end{struct}

\begin{instance}\label{instance_focus_probe_value}
    The focus probe token $\InstanceFocusProbeValue=\structure{}$.
\end{instance}
\begin{proof}
    Follows.
\end{proof}

\begin{definition}\label{instance_focus_probe_predicate}
    Let $T$ be a focus probe token.
    $T$ is a marked token iff $T=T$.
\end{definition}

\begin{theorem}\label{instance_focus_probe_subject}
    Then $\InstanceFocusProbeValue$ is a marked token.
\end{theorem}
\begin{proof}
    $\InstanceFocusProbeValue$ is a focus probe token by \obv{instance_focus_probe_value}.
    Follows by \obv{instance_focus_probe_predicate}.
\end{proof}

\begin{definition}\label{instance_focus_probe_parameter_predicate}
    Let $T$ be a focus probe token.
    $A$ is a supported set under $T$ iff $A=A$.
\end{definition}

\begin{theorem}\label{instance_focus_probe_explicit_guard}
    Suppose $\InstanceFocusProbeValue$ is a focus probe token.
    Then $\emptyset$ is a supported set under $\InstanceFocusProbeValue$.
\end{theorem}
\begin{proof}
    Follows by \obv{instance_focus_probe_parameter_predicate}.
\end{proof}
