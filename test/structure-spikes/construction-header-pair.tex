\begin{construction}\label{copy_pair_coordinate}
    Let $X,Y$ be sets.
    Let $R$ be a coordinate record on $(X,Y)$.
    The coordinate record $\CopyPairCoordinate{X}{Y}{R}=
        coordinate record on (X,Y) based on R$ on $(X,Y)$.
\end{construction}
\begin{proof}
    $\atlas[R]\subseteq (X,Y)$ by \obv{coordinate_atlas_bound}.
    Follows by \auto{coordinate_recordaggregateprojection_atlas}.
\end{proof}

\begin{construction}\label{copy_nested_pair_coordinate}
    Let $X,Y$ be sets.
    Let $R$ be a coordinate record on $((X,Y),X)$.
    The coordinate record $\CopyNestedPairCoordinate{X}{Y}{R}=
        coordinate record on ((X,Y),X) based on R$ on $((X,Y),X)$.
\end{construction}
\begin{proof}
    $\atlas[R]\subseteq ((X,Y),X)$ by \obv{coordinate_atlas_bound}.
    Follows by \auto{coordinate_recordaggregateprojection_atlas}.
\end{proof}
