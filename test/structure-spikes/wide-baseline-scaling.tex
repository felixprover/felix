\begin{definition}\label{wide_copy}
    $\WideCopy{R}=\structure[wide record]{
        \north=\north[R],
        \south=\south[R],
        \east=\east[R],
        \west=\west[R],
        \center=\center[R],
        \upper=\upper[R],
        \lower=\lower[R],
        \inner=\inner[R],
        \outer=\outer[R],
        \middle=\middle[R]}$.
\end{definition}

\begin{classification}\label{wide_copy_result}
    Let $R$ be a wide record.
    Then $\WideCopy{R}$ is a wide record.
\end{classification}
\begin{proof}
    Follows by \auto{wide_copy,wide_baseaggregateexact}.
\end{proof}

\begin{proposition}\label{wide_copyshape}
    Let $R$ be a wide record.
    Then $\WideCopy{R}$ is an exact wide record.
\end{proposition}
\begin{proof}
    Follows by \auto{wide_copy,wide_copy_result,wide_baseaggregateexact}.
\end{proof}

\begin{proposition}\label{wide_copy_identity}
    Let $R$ be a wide record.
    Suppose $R$ is an exact wide record.
    Then $\WideCopy{R}=R$.
\end{proposition}
\begin{proof}
    $\WideCopy{R}$ is an exact wide record by \obv{wide_copyshape}.
    $\north[\WideCopy{R}]=\north[R]$ by \obv{wide_copyprojection_north}.
    $\south[\WideCopy{R}]=\south[R]$ by \obv{wide_copyprojection_south}.
    $\east[\WideCopy{R}]=\east[R]$ by \obv{wide_copyprojection_east}.
    $\west[\WideCopy{R}]=\west[R]$ by \obv{wide_copyprojection_west}.
    $\center[\WideCopy{R}]=\center[R]$ by \obv{wide_copyprojection_center}.
    $\upper[\WideCopy{R}]=\upper[R]$ by \obv{wide_copyprojection_upper}.
    $\lower[\WideCopy{R}]=\lower[R]$ by \obv{wide_copyprojection_lower}.
    $\inner[\WideCopy{R}]=\inner[R]$ by \obv{wide_copyprojection_inner}.
    $\outer[\WideCopy{R}]=\outer[R]$ by \obv{wide_copyprojection_outer}.
    $\middle[\WideCopy{R}]=\middle[R]$ by \obv{wide_copyprojection_middle}.
    Follows by structure extensionality.
\end{proof}
