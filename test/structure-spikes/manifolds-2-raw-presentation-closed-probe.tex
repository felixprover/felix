\begin{theorem}\label{raw_real_presentations_consumer}
    Then $\RealTwoCharts$ presents $\SpaceFromCharts{\reals}{\RealTopology}{\RealOneChart}$
        on $\SpaceFromCharts{\reals}{\RealTopology}{\RealOneChart}$ under $\FullPseudogroup{\RealTopology}$ and
        $\SpaceFromCharts{\reals}{\RealTopology}{\RealTwoCharts}
            =\SpaceFromCharts{\reals}{\RealTopology}{\RealOneChart}$.
\end{theorem}
\begin{proof}
    $\RealTopology$ is a topological space by \obv{real_topology_is_topological_space}.
    $\RealTwoCharts$ is a topological coordinate presentation on $\reals$ modelled on $\RealTopology$ by
        \obv{real_two_charts_presentation}.
    $\RealTwoCharts$ presents $\SpaceFromCharts{\reals}{\RealTopology}{\RealTwoCharts}$
        on $\SpaceFromCharts{\reals}{\RealTopology}{\RealTwoCharts}$ under $\FullPseudogroup{\RealTopology}$ by
        \obv{space_from_charts_presentation}.
    $\SpaceFromCharts{\reals}{\RealTopology}{\RealTwoCharts}
        =\SpaceFromCharts{\reals}{\RealTopology}{\RealOneChart}$ by \obv{real_chart_spaces_equal}.
\end{proof}

\begin{theorem}\label{raw_zero_chart_consumer}
    Then $\identity{\{\emptyset\}}\in\ChartsAt{
        \SpaceFromCharts{\{\emptyset\}}{\ZeroCoordinateModel}{\ZeroCoordinateCharts}}{\emptyset}$.
\end{theorem}
\begin{proof}
    $\ZeroCoordinateModel$ is a topological space by \obv{zero_coordinate_model_is_topological_space}.
    $\ZeroCoordinateCharts$ is a topological coordinate presentation on $\{\emptyset\}$
        modelled on $\ZeroCoordinateModel$ by \obv{zero_coordinate_charts_presentation}.
    Let $M=\SpaceFromCharts{\{\emptyset\}}{\ZeroCoordinateModel}{\ZeroCoordinateCharts}$.
    $M$ is a coordinate datum and $M$ is a topological space by
        \auto{space_from_charts,equipped_coordinate_space}.
    $\FullPseudogroup{\ZeroCoordinateModel}$ is a coordinate pseudogroup by \obv{full_pseudogroup}.
    $\ZeroCoordinateCharts$ presents $M$ on $M$ under $\FullPseudogroup{\ZeroCoordinateModel}$ by
        \auto{space_from_charts_presentation}.
    $\ZeroCoordinateCharts\subseteq\atlas[M]$ by \obv{presentation_chart_inclusion}.
    $\identity{\{\emptyset\}}\in\atlas[M]$ by \auto{zero_coordinate_charts,singleton_iff,subseteq}.
    $\emptyset\in\dom{\identity{\{\emptyset\}}}$ by \auto{id_dom,singleton_iff}.
    Follows by \auto{charts_at_iff}.
\end{proof}

\begin{theorem}\label{raw_empty_space_consumer}
    Then $\opens[\SpaceFromCharts{\emptyset}{\RealTopology}{\emptyset}]=\{\emptyset\}$ and
        $\atlas[\SpaceFromCharts{\emptyset}{\RealTopology}{\emptyset}]=\{\emptyset\}$ and
        $\ChartsAt{\SpaceFromCharts{\emptyset}{\RealTopology}{\emptyset}}{\emptyset}=\emptyset$.
\end{theorem}
\begin{proof}
    $\RealTopology$ is a topological space by \obv{real_topology_is_topological_space}.
    $\emptyset$ is a topological coordinate presentation on $\emptyset$ modelled on $\RealTopology$ by
        \obv{empty_coordinate_presentation}.
    Let $M=\SpaceFromCharts{\emptyset}{\RealTopology}{\emptyset}$.
    $M$ is a coordinate datum by \auto{space_from_charts,equipped_coordinate_space}.
    $\CoordinateTopology{\emptyset}{\RealTopology}{\emptyset}=\discreteTopology{\emptyset}$ by
        \obv{empty_coordinate_topology}.
    $\opens[M]=\pow{\emptyset}$ by
        \auto{space_from_chartsprojection_opens,discrete_topology_opens,discrete_opens}.
    $\opens[M]=\{\emptyset\}$ by \obv{powerset_emptyset}.
    $\atlas[M]=\{\emptyset\}$ by \auto{empty_chart_space}.
    $\dom{\emptyset}=\emptyset$ by \obv{dom_emptyset}.
    $\ChartsAt{M}{\emptyset}=\emptyset$ by \auto{charts_at_iff,singleton_iff,emptyset,setext}.
\end{proof}
