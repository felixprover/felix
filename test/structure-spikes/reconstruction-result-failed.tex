\begin{classification}\label{empty_coordinate_marked}
    Let $X$ be a set.
    Suppose $X\in X$.
    Then $\EmptyCoordinateOver{X}$ is a marked coordinate on $X$.
\end{classification}
\begin{proof}
    $\EmptyCoordinateOver{X}$ is a coordinate record on $X$
        by \obv{empty_coordinate_any_parameter}.
    Follows by \obv{marked_coordinate}.
\end{proof}
