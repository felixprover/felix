\section{Checked non-variable reconstruction bases}

\begin{proposition}\label{reconstruction_provider_base}
    Let $X$ be a topological space.
    Suppose $A$ is an admissible atlas on $X$.
    Then $(topological space based on \Equip{X}{A})=
        \structure[topological space]{
            \carrier=\carrier[\Equip{X}{A}],\opens=\opens[\Equip{X}{A}]}$.
\end{proposition}

\begin{proposition}\label{reconstruction_local_alias}
    Let $X$ be a topological space.
    Suppose $A$ is an admissible atlas on $X$.
    Then $(topological space based on \Equip{X}{A})=
        \structure[topological space]{\carrier=\carrier[X],\opens=\opens[X]}$.
\end{proposition}
\begin{proof}
    Let $N=\Equip{X}{A}$.
    $N$ is an equipped coordinate space by \auto{equip_result}.
    $(topological space based on N)=
        \structure[topological space]{\carrier=\carrier[N],\opens=\opens[N]}$.
    $\carrier[N]=\carrier[X]$ by \auto{equipprojection_carrier}.
    $\opens[N]=\opens[X]$ by \auto{equipprojection_opens}.
\end{proof}

\begin{proposition}\label{reconstruction_relative_provider}
    Let $X$ be a topological space.
    Suppose $A$ is an admissible atlas on $X$.
    Then $(relative coordinate structure on X based on \GeneratedCoordinateStructure{X}{A})=
        \structure[relative coordinate structure on X]{
            \atlas=\atlas[\GeneratedCoordinateStructure{X}{A}]}$.
\end{proposition}
