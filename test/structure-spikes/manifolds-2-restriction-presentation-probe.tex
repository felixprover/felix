\section{Restricted-coordinate presentation formation probe}

\import{topology/coordinate-restriction.tex}

This imported consumer forms the full presentation statement without a
redundant assumption about the structure supplied by its presentation.

\begin{theorem}\label{restricted_coordinates_presentation_consumer}
    Let $G$ be a coordinate pseudogroup and $X$ be a topological space.
    Let $S$ be a coordinate datum.
    Suppose $A$ presents $S$ on $X$ under $G$.
    Suppose $U$ is open in $X$.
    Then $\RestrictedAtlas{A}{U}$ presents $\RestrictedCoordinates{G}{X}{S}{U}$
        on $\subspaceTopology{X}{U}$ under $G$.
\end{theorem}
\begin{proof}
    Follows by \obv{restricted_coordinates_presentation}.
\end{proof}

\begin{theorem}\label{restricted_coordinates_two_presentations_consumer}
    Let $G$ be a coordinate pseudogroup and $X$ be a topological space.
    Let $S$ be a coordinate datum.
    Suppose $A$ presents $S$ on $X$ under $G$ and $B$ presents $S$ on $X$ under $G$.
    Suppose $U$ is open in $X$.
    Then $\GeneratedCoordinates{G}{\subspaceTopology{X}{U}}{\RestrictedAtlas{A}{U}}
        =\GeneratedCoordinates{G}{\subspaceTopology{X}{U}}{\RestrictedAtlas{B}{U}}$.
\end{theorem}
\begin{proof}
    $\RestrictedCoordinates{G}{X}{S}{U}
        =\GeneratedCoordinates{G}{\subspaceTopology{X}{U}}{\RestrictedAtlas{A}{U}}$ by
        \obv{restricted_coordinates_from_presentation}.
    $\RestrictedCoordinates{G}{X}{S}{U}
        =\GeneratedCoordinates{G}{\subspaceTopology{X}{U}}{\RestrictedAtlas{B}{U}}$ by
        \obv{restricted_coordinates_from_presentation}.
\end{proof}

\begin{theorem}\label{nested_restricted_coordinates_consumer}
    Let $G$ be a coordinate pseudogroup and $X$ be a topological space.
    Let $S$ be a relative coordinate structure on $X$ under $G$.
    Suppose $U$ is open in $X$ and $V$ is open in $\subspaceTopology{X}{U}$.
    Then $\RestrictedCoordinates{G}{\subspaceTopology{X}{U}}{\RestrictedCoordinates{G}{X}{S}{U}}{V}
        =\RestrictedCoordinates{G}{X}{S}{V}$.
\end{theorem}
\begin{proof}
    Follows by \obv{restricted_coordinates_nested}.
\end{proof}

\begin{theorem}\label{nested_restricted_charts_consumer}
    Let $G$ be a coordinate pseudogroup and $X$ be a topological space.
    Let $S$ be a relative coordinate structure on $X$ under $G$.
    Suppose $U$ is open in $X$ and $V$ is open in $\subspaceTopology{X}{U}$.
    Then $\ChartsAt{\RestrictedCoordinates{G}{\subspaceTopology{X}{U}}{\RestrictedCoordinates{G}{X}{S}{U}}{V}}{p}
        =\ChartsAt{\RestrictedCoordinates{G}{X}{S}{V}}{p}$.
\end{theorem}
\begin{proof}
    $\RestrictedCoordinates{G}{\subspaceTopology{X}{U}}{\RestrictedCoordinates{G}{X}{S}{U}}{V}

\begin{theorem}\label{full_restricted_coordinates_consumer}
    Let $G$ be a coordinate pseudogroup and $X$ be a topological space.
    Suppose $S$ is an exact relative coordinate structure on $X$ under $G$.
    Then $\RestrictedCoordinates{G}{X}{S}{\carrier[X]}=S$.
\end{theorem}
\begin{proof}
    Follows by \obv{restricted_coordinates_full}.
\end{proof}

\end{theorem}
\begin{proof}

\end{proof}

\begin{theorem}\label{equipped_restricted_presentation_consumer}
    Let $G$ be a coordinate pseudogroup and $X$ be a topological space.
    Let $S$ be a coordinate datum.
    Suppose $A$ presents $S$ on $X$ under $G$ and $U$ is open in $X$.
    Then $\RestrictedAtlas{A}{U}$ presents
        $\Equip{\subspaceTopology{X}{U}}{\RestrictedCoordinates{G}{X}{S}{U}}$ on
        $\Equip{\subspaceTopology{X}{U}}{\RestrictedCoordinates{G}{X}{S}{U}}$ under $G$.
\end{theorem}
\begin{proof}
    Follows by \obv{equipped_restricted_coordinates_presentation}.
\end{proof}

\begin{theorem}\label{restricted_inclusion_consumer}
    Let $C$ be a local-map calculus and $G$ be a coordinate pseudogroup.
    Let $X$ be a topological space and $S$ be a coordinate datum.
    Suppose $A$ presents $S$ on $X$ under $G$.
    Suppose $G$ is compatible with the calculus $C$ and $U$ is open in $X$.
    Then $\subspaceInclusion{X}{U}$ is coordinate-admissible from
        $\Equip{\subspaceTopology{X}{U}}{\RestrictedCoordinates{G}{X}{S}{U}}$ under $G$
        to $\Equip{X}{S}$ under $G$ in $C$.
\end{theorem}
\begin{proof}
    $S$ is a relative coordinate structure on $X$ under $G$ by \obv{presented_coordinates}.
    Follows by \obv{restricted_coordinates_inclusion}.
\end{proof}

\begin{theorem}\label{empty_restricted_inclusion_consumer}
    Let $C$ be a local-map calculus and $G$ be a coordinate pseudogroup.
    Let $X$ be a topological space.
    Let $S$ be a relative coordinate structure on $X$ under $G$.
    Suppose $G$ is compatible with the calculus $C$.
    Then $\emptyset$ is coordinate-admissible from
        $\Equip{\subspaceTopology{X}{\emptyset}}{\RestrictedCoordinates{G}{X}{S}{\emptyset}}$ under $G$
        to $\Equip{X}{S}$ under $G$ in $C$.
\end{theorem}
\begin{proof}
    $\emptyset$ is open in $X$ by \obv{emptyset_open}.
    $\subspaceInclusion{X}{\emptyset}=\emptyset$ by
        \auto{subspace_inclusion,inter_emptyset,inter_comm,id,setext}.
    Follows by \auto{restricted_coordinates_inclusion}.
\end{proof}

\begin{theorem}\label{restricted_equipped_extraction_consumer}
    Let $G$ be a coordinate pseudogroup and $X$ be a topological space.
    Let $S$ be a relative coordinate structure on $X$ under $G$.
    Suppose $U$ is open in $X$.
    Then $\CoordinateStructure{\Equip{\subspaceTopology{X}{U}}{\RestrictedCoordinates{G}{X}{S}{U}}}
        =\RestrictedCoordinates{G}{X}{S}{U}$.
\end{theorem}
\begin{proof}
    $\subspaceTopology{X}{U}$ is a topological space by \obv{subspace_topology_is_topological_space}.
    $\RestrictedCoordinates{G}{X}{S}{U}$ is an exact coordinate datum by
        \auto{restricted_coordinates,restricted_coordinatesshape,relative_coordinate_structure}.
    Follows by \obv{coordinate_structure_of_equip}.
\end{proof}

\begin{theorem}\label{full_restricted_charts_consumer}
    Let $G$ be a coordinate pseudogroup and $X$ be a topological space.
    Suppose $S$ is an exact relative coordinate structure on $X$ under $G$.
    Then $\ChartsAt{\RestrictedCoordinates{G}{X}{S}{\carrier[X]}}{p}=\ChartsAt{S}{p}$.
\end{theorem}
\begin{proof}
    $\RestrictedCoordinates{G}{X}{S}{\carrier[X]}=S$ by \obv{restricted_coordinates_full}.
\end{proof}
