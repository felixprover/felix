\section{Imported weakening laws}

\import{topology/coordinate-weakening.tex}

\begin{theorem}\label{weakened_presentations_consumer}
    Let $G,H$ be coordinate pseudogroups and $X$ be a topological space.
    Let $S$ be a coordinate datum.
    Suppose $A$ presents $S$ on $X$ under $G$ and $B$ presents $S$ on $X$ under $G$.
    Suppose $G$ is a subpseudogroup of $H$.
    Then $\GeneratedCoordinates{H}{X}{A}=\GeneratedCoordinates{H}{X}{B}$.
\end{theorem}
\begin{proof}
    $\WeakenedCoordinates{G}{H}{X}{S}=\GeneratedCoordinates{H}{X}{A}$ and
        $\WeakenedCoordinates{G}{H}{X}{S}=\GeneratedCoordinates{H}{X}{B}$ by
        \obv{weakened_coordinates_from_presentation}.
\end{proof}

\begin{theorem}\label{weakened_charts_transitive_consumer}
    Let $G,H,K$ be coordinate pseudogroups and $X$ be a topological space.
    Let $S$ be a relative coordinate structure on $X$ under $G$.
    Suppose $G$ is a subpseudogroup of $H$ and $H$ is a subpseudogroup of $K$.
    Then $\ChartsAt{\WeakenedCoordinates{H}{K}{X}{\WeakenedCoordinates{G}{H}{X}{S}}}{p}
        =\ChartsAt{\WeakenedCoordinates{G}{K}{X}{S}}{p}$.
\end{theorem}
\begin{proof}
    $\WeakenedCoordinates{H}{K}{X}{\WeakenedCoordinates{G}{H}{X}{S}}
        =\WeakenedCoordinates{G}{K}{X}{S}$ by \obv{weakened_coordinates_transitive}.
\end{proof}

\begin{theorem}\label{weakened_restriction_consumer}
    Let $G,H$ be coordinate pseudogroups and $X$ be a topological space.
    Let $S$ be a relative coordinate structure on $X$ under $G$.
    Suppose $G$ is a subpseudogroup of $H$ and $U$ is open in $X$.
    Then $\ChartsAt{\WeakenedCoordinates{G}{H}{\subspaceTopology{X}{U}}{\RestrictedCoordinates{G}{X}{S}{U}}}{p}
        =\ChartsAt{\RestrictedCoordinates{H}{X}{\WeakenedCoordinates{G}{H}{X}{S}}{U}}{p}$.
\end{theorem}
\begin{proof}
    $\WeakenedCoordinates{G}{H}{\subspaceTopology{X}{U}}{\RestrictedCoordinates{G}{X}{S}{U}}
        =\RestrictedCoordinates{H}{X}{\WeakenedCoordinates{G}{H}{X}{S}}{U}$ by
        \obv{weakened_coordinates_restriction}.
\end{proof}

\begin{theorem}\label{weakened_transport_consumer}
    Let $G,H$ be coordinate pseudogroups and $X,Y$ be topological spaces.
    Let $S$ be a relative coordinate structure on $X$ under $G$.
    Suppose $G$ is a subpseudogroup of $H$ and $h$ is a homeomorphism from $X$ to $Y$.
    Then $\ChartsAt{\WeakenedCoordinates{G}{H}{Y}{\TransportedCoordinates{G}{X}{Y}{S}{h}}}{p}
        =\ChartsAt{\TransportedCoordinates{H}{X}{Y}{\WeakenedCoordinates{G}{H}{X}{S}}{h}}{p}$.
\end{theorem}
\begin{proof}
    $\WeakenedCoordinates{G}{H}{Y}{\TransportedCoordinates{G}{X}{Y}{S}{h}}
        =\TransportedCoordinates{H}{X}{Y}{\WeakenedCoordinates{G}{H}{X}{S}}{h}$ by
        \obv{weakened_coordinates_transport}.
\end{proof}

\begin{theorem}\label{weakened_extraction_consumer}
    Let $G,H$ be coordinate pseudogroups and $X$ be a topological space.
    Let $S$ be a relative coordinate structure on $X$ under $G$.
    Suppose $G$ is a subpseudogroup of $H$.
    Then $\ChartsAt{\WeakenedCoordinates{G}{H}{\TopologicalReduct{\Equip{X}{S}}}{\CoordinateStructure{\Equip{X}{S}}}}{p}
        =\ChartsAt{\WeakenedCoordinates{G}{H}{X}{S}}{p}$.
\end{theorem}
\begin{proof}
    $\WeakenedCoordinates{G}{H}{\TopologicalReduct{\Equip{X}{S}}}{\CoordinateStructure{\Equip{X}{S}}}
        =\WeakenedCoordinates{G}{H}{X}{S}$ by \obv{weakened_coordinates_of_equip}.
\end{proof}
