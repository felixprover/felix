\import{set.tex}

\begin{struct}\label{coordinate_record}
    Let $X$ be a set.
    A coordinate record $R$ on $X$ is a record equipped with
    \begin{enumerate}
        \item a set $\atlas[R]$;
    \end{enumerate}
    such that
    \begin{enumerate}
        \item\label{coordinate_atlas_bound} $\atlas[R]\subseteq X$.
    \end{enumerate}
\end{struct}

\begin{struct}\label{reflexive_record}
    A reflexive record $R$ is a record such that
    \begin{enumerate}
        \item\label{reflexive_record_law} $R=R$.
    \end{enumerate}
\end{struct}

\begin{proposition}\label{record_plural}
    Let $R,S$ be records.
    Then $R=R$ and $S=S$.
\end{proposition}

\begin{proposition}\label{empty_record_term}
    $\structure[record]{}=\emptyset$.
\end{proposition}

\begin{instance}\label{empty_record}
    The record $\EmptyRecord=\structure{}$.
\end{instance}
\begin{proof}
    Follows by \auto{record,recordaggregateexact}.
\end{proof}

\begin{proposition}\label{coordinate_aggregate}
    Let $X$ be a set.
    Then $\structure[coordinate record on X]{X}$ is an exact coordinate record on $X$.
\end{proposition}
\begin{proof}
    Follows by \auto{coordinate_record,coordinate_recordaggregateexact,coordinate_recordaggregateprojection_atlas,subseteq}.
\end{proof}

\begin{instance}\label{empty_coordinate_record}
    The coordinate record $\EmptyCoordinate$ on $\emptyset$ is $\structure{\emptyset}$.
\end{instance}
\begin{proof}
    Follows by \auto{coordinate_record,coordinate_recordaggregateexact,coordinate_recordaggregateprojection_atlas,subseteq}.
\end{proof}

\begin{proposition}\label{coordinate_projection}
    $\atlas[\EmptyCoordinate]=\emptyset$.
\end{proposition}
\begin{proof}
    Follows by \obv{empty_coordinate_recordprojection_atlas}.
\end{proof}

\begin{proposition}\label{empty_record_extensionality}
    Let $A,B$ be sets.
    Assume $A$ is an exact record.
    Assume $B$ is an exact record.
    Then $A=B$.
\end{proposition}
\begin{proof}
    Follows by structure extensionality.
\end{proof}

\begin{proposition}\label{empty_record_generic_extensionality}
    Let $A,B$ be sets.
    Assume $A$ is an exact record.
    Assume $B$ is an exact record.
    Then $A=B$.
\end{proposition}
\begin{proof}
    Follows by extensionality.
\end{proof}

\begin{proposition}\label{coordinate_extensionality}
    Let $X,A,B$ be sets.
    Assume $A$ is an exact coordinate record on $X$.
    Assume $B$ is an exact coordinate record on $X$.
    Assume $\atlas[A]=\atlas[B]$.
    Then $A=B$.
\end{proposition}
\begin{proof}
    Follows by structure extensionality.
\end{proof}

\begin{proposition}\label{coordinate_root_reduct}
    Let $R$ be a coordinate record on $X$.
    Then $\reduct[record]{R}$ is an exact record.
\end{proposition}
\begin{proof}
    Follows by \auto{record,recordreductexact}.
\end{proof}

\begin{proposition}\label{coordinate_reduct_projection}
    Let $R$ be a coordinate record on $X$.
    Then $\atlas[\reduct[coordinate record on X]{R}]=\atlas[R]$.
\end{proposition}
\begin{proof}
    Follows by \obv{coordinate_recordreductprojection_atlas}.
\end{proof}
