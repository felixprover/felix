\begin{struct}\label{paired_coordinate}
    Let $X,Y$ be sets.
    A paired coordinate $R$ on $X$ and $Y$ is a coordinate record on $(X,Y)$ such that
    \begin{enumerate}
        \item\label{paired_copy_reflexive}
            $\CopyPairCoordinate{X}{Y}{R}=\CopyPairCoordinate{X}{Y}{R}$.
    \end{enumerate}
\end{struct}

\begin{proposition}\label{copy_pair_import_shape}
    Let $X,Y$ be sets.
    Let $R$ be a coordinate record on $(X,Y)$.
    Then $\CopyPairCoordinate{X}{Y}{\CopyPairCoordinate{X}{Y}{R}}$
        is an exact coordinate record on $(X,Y)$.
\end{proposition}
\begin{proof}
    Follows by \obv{copy_pair_coordinate,copy_pair_coordinateshape}.
\end{proof}

\begin{proposition}\label{copy_pair_baseline_import_shape}
    Let $X,Y$ be sets.
    Let $R$ be a coordinate record on $(X,Y)$.
    Then $\CopyPairCoordinate{X}{Y}{\CopyPairCoordinateBaseline{X}{Y}{R}}$
        is an exact coordinate record on $(X,Y)$.
\end{proposition}
\begin{proof}
    Follows by \obv{copy_pair_coordinate,copy_pair_coordinateshape}.
\end{proof}

\begin{proposition}\label{copy_pair_import_reconstruction}
    Let $X,Y$ be sets.
    Let $R$ be a coordinate record on $(X,Y)$.
    Then $coordinate record on (X,Y) based on \CopyPairCoordinate{X}{Y}{R}
        =coordinate record on (X,Y) based on \CopyPairCoordinateBaseline{X}{Y}{R}$.
\end{proposition}
\begin{proof}
    $\atlas[\CopyPairCoordinate{X}{Y}{R}]=\atlas[R]$
        by \obv{copy_pair_coordinateprojection_atlas}.
    $\atlas[\CopyPairCoordinateBaseline{X}{Y}{R}]=\atlas[R]$
        by \obv{copy_pair_coordinate_definition,coordinate_recordaggregateprojection_atlas}.
    Follows.
\end{proof}

\begin{proposition}\label{copy_nested_pair_import}
    Let $X,Y$ be sets.
    Let $R$ be a coordinate record on $((X,Y),X)$.
    Then $\atlas[\CopyPairCoordinate{(X,Y)}{X}{R}]=\atlas[R]$.
\end{proposition}
\begin{proof}
    Follows by \obv{copy_pair_coordinateprojection_atlas}.
\end{proof}
