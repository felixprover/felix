\import{set.tex}

\begin{struct}\label{instance_guard_probe_token}
    A probe token $T$ is a record.
\end{struct}

\begin{instance}\label{instance_guard_probe_value}
    The probe token $\InstanceGuardProbeValue=\structure{}$.
\end{instance}
\begin{proof}
    Follows.
\end{proof}

\begin{definition}\label{instance_guard_probe_predicate}
    Let $T$ be a probe token.
    $A$ is a marked set over $T$ iff $A=A$.
\end{definition}

\begin{theorem}\label{instance_guard_probe_parameter}
    Then $\emptyset$ is a marked set over $\InstanceGuardProbeValue$.
\end{theorem}
\begin{proof}
    $\InstanceGuardProbeValue$ is a probe token by \obv{instance_guard_probe_value}.
    Follows by \obv{instance_guard_probe_predicate}.
\end{proof}
