\section{Small construction families}

\begin{construction}\label{empty_record_family}
    Let $X$ be a set.
    The record $\EmptyRecordFamily{X}=\structure{}$.
\end{construction}
\begin{proof}
    Follows.
\end{proof}

\begin{construction}\label{reflexive_record_family}
    Let $X$ be a set.
    The reflexive record $\ReflexiveRecordFamily{X}=\structure{}$.
\end{construction}
\begin{proof}
    Follows.
\end{proof}

\begin{construction}\label{full_coordinate_family}
    Let $X$ be a set.
    The coordinate record $\FullCoordinate{X}=\structure{\atlas=X}$ on $X$.
\end{construction}
\begin{proof}
    Follows by \auto{coordinate_recordaggregateprojection_atlas,subseteq_refl}.
\end{proof}

\begin{construction}\label{full_coordinate_explicit_family}
    Let $X$ be a set.
    The coordinate record $\FullCoordinateExplicit{X}=
        \structure[coordinate record on X]{\atlas=X}$ on $X$.
\end{construction}
\begin{proof}
    Follows by \auto{coordinate_recordaggregateprojection_atlas,subseteq_refl}.
\end{proof}

\begin{proposition}\label{coordinate_family_annotation_agrees}
    Let $X$ be a set.
    Then $\FullCoordinateExplicit{X}=\FullCoordinate{X}$.
\end{proposition}

\begin{construction}\label{copy_coordinate_family}
    Let $X$ be a set.
    Let $R$ be a coordinate record on $X$.
    The coordinate record $\CopyCoordinate{X}{R}=
        coordinate record on X based on R$ on $X$.
\end{construction}
\begin{proof}
    $\atlas[R]\subseteq X$ by \obv{coordinate_atlas_bound}.
    Follows by \auto{coordinate_recordaggregateprojection_atlas}.
\end{proof}

\begin{proposition}\label{copy_coordinate_projection}
    Let $X$ be a set.
    Let $R$ be a coordinate record on $X$.
    Then $\atlas[\CopyCoordinate{X}{R}]=\atlas[R]$.
\end{proposition}
\begin{proof}
    Follows by \obv{copy_coordinate_familyprojection_atlas}.
\end{proof}

\begin{proposition}\label{full_coordinate_reconstruction}
    Let $X$ be a set.
    Then $coordinate record on X based on \FullCoordinate{X}
        =\structure[coordinate record on X]{X}$.
\end{proposition}
\begin{proof}
    Follows by \obv{full_coordinate_familyprojection_atlas}.
\end{proof}
