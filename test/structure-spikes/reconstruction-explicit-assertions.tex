\begin{proposition}\label{discovery_explicit_parameter}
    Let $X,Y$ be sets.
    Then $X=X$.
\end{proposition}
\begin{proof}
    $\EmptyCoordinateOver{X}$ is a coordinate record on $Y$
        by \obv{empty_coordinate_any_parameter}.
    $(coordinate record on Y based on \EmptyCoordinateOver{X})=
        \structure[coordinate record on Y]{\atlas=\emptyset}$
        by \auto{empty_coordinate_over,coordinate_recordaggregateprojection_atlas}.
    Follows.
\end{proof}

\begin{proposition}\label{discovery_explicit_alias}
    Let $R,X$ be sets.
    Suppose $R$ is a coordinate record on $X$.
    Then $R=R$.
\end{proposition}
\begin{proof}
    Let $N=R$.
    $N$ is a coordinate record on $X$.
    $(coordinate record on X based on N)=
        \structure[coordinate record on X]{\atlas=\atlas[N]}$.
    Follows.
\end{proof}
