\begin{proposition}\label{nominal_full_pair}
    Let $E$ be a topological space.
    Then $\PseudogroupIntersection{E}{\{\FullPseudogroup{E},\FullPseudogroup{E}\}}$
        is a pseudogroup on $E$.
\end{proposition}
\begin{proof}
    Follows by notion.
\end{proof}

\begin{proposition}\label{nominal_full_diffeomorphism}
    Let $C$ be a local-map calculus.
    Let $E$ be a model of $C$.
    Then $\PseudogroupIntersection{E}{\{\FullPseudogroup{E},\DiffeomorphismPseudogroup{C}{E}\}}$
        is a pseudogroup on $E$.
\end{proposition}
\begin{proof}
    Follows by notion.
\end{proof}

\begin{proposition}\label{nominal_empty_singleton_nested}

\begin{proposition}\label{nominal_family_member}
    Let $E$ be a topological space.
    Let $P$ be a pseudogroup family on $E$.
    Let $G\in P$.
    Then $G$ is a pseudogroup on $E$.
\end{proposition}
\begin{proof}
    Follows by notion.
\end{proof}
