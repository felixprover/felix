\begin{proposition}\label{empty_record_alias_extensionality}
    For all $A,B$ we have $A=A$.
\end{proposition}
\begin{proof}
    Fix $A$ and $B$.
    Let $P=A$.
    Let $Q=B$.
    Let $U=\emptyset$.
    Show if $P$ is an exact record, then if $Q$ is an exact record, then $A=B$.
    \begin{subproof}
        Assume $P$ is an exact record.
        Assume $Q$ is an exact record.
        Follows by structure extensionality.
    \end{subproof}
    Follows.
\end{proof}

\begin{proposition}\label{coordinate_alias_extensionality}
    For all $X,A,B$ we have $A=A$.
\end{proposition}
\begin{proof}
    Fix $X$, $A$, and $B$.
    Let $P=A$.
    Let $R=P$.
    Let $Q=B$.
    Let $S=Q$.
    Let $U=\emptyset$.
    Show if $\pow{R}$ is an exact coordinate record on $X$, then if
        $\pow{S}$ is an exact coordinate record on $X$, then if
        $\atlas[\pow{R}]=\atlas[\pow{S}]$, then $\pow{A}=\pow{B}$.
    \begin{subproof}
        Assume $\pow{R}$ is an exact coordinate record on $X$.
        Assume $\pow{S}$ is an exact coordinate record on $X$.
        Assume $\atlas[\pow{R}]=\atlas[\pow{S}]$.
        Follows by structure extensionality.
    \end{subproof}
    Follows.
\end{proof}
