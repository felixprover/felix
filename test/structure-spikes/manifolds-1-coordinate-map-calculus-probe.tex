\begin{theorem}\label{coordinate_map_closed_calculus_probe}
    Let $G,H$ be coordinate pseudogroups.
    Let $M,N$ be equipped coordinate spaces.
    Then if $f$ is coordinate-admissible from $M$ under $G$ to $N$ under $H$
        in $\ContinuousCalculus{\{\RealTopology\}}$,
        then $f$ is coordinate-admissible from $M$ under $G$ to $N$ under $H$
        in $\ContinuousCalculus{\{\RealTopology\}}$.
\end{theorem}
\begin{proof}
    Follows.
\end{proof}
