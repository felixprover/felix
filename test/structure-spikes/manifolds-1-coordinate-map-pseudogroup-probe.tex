\begin{theorem}\label{coordinate_map_closed_pseudogroup_probe}
    Let $C$ be a local-map calculus.
    Let $M,N$ be equipped coordinate spaces.
    Then if $f$ is coordinate-admissible from $M$ under $\FullPseudogroup{\RealTopology}$
        to $N$ under $\FullPseudogroup{\RealTopology}$ in $C$,
        then $f$ is coordinate-admissible from $M$ under $\FullPseudogroup{\RealTopology}$
        to $N$ under $\FullPseudogroup{\RealTopology}$ in $C$.
\end{theorem}
\begin{proof}
    Follows.
\end{proof}
