\import{set.tex}

\begin{struct}\label{unregistered_token}
    An unregistered token $R$ is a record.
\end{struct}

\begin{definition}\label{unregistered_value}
    $\UnregisteredValue:\Set=\structure[record]{}$.
\end{definition}

\begin{proposition}\label{unregistered_value_theorem}
    $\UnregisteredValue$ is an unregistered token.
\end{proposition}
\begin{proof}
    Follows by \auto{unregistered_value,unregistered_token,recordaggregateexact}.
\end{proof}

\begin{definition}\label{unregistered_guard}
    Let $R$ be an unregistered token.
    $A$ is supported over $R$ iff $A=A$.
\end{definition}

\begin{proposition}\label{unregistered_guard_rejected}
    $\emptyset$ is supported over $\UnregisteredValue$.
\end{proposition}
\begin{proof}
    Follows.
\end{proof}
