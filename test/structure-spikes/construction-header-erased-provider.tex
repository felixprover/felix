\begin{construction}\label{erased_provider_header}
    Let $X,Y$ be sets.
    Let $R$ be a coordinate record on $\OtherLeft{X}{Y}$.
    The coordinate record $\ErasedProviderHeader{X}{Y}{R}=
        coordinate record on \KeepLeft{X}{Y} based on R$
        on $\KeepLeft{X}{Y}$.
\end{construction}
\begin{proof}
    $\atlas[R]\subseteq X$ by \obv{coordinate_atlas_bound}.
    Follows by \auto{coordinate_recordaggregateprojection_atlas}.
\end{proof}
