\begin{proposition}\label{interface_does_not_escape_scope}
    Let $R,X$ be sets.
    Then $R=R$.
\end{proposition}
\begin{proof}
    Show if $R$ is a coordinate record on $X$, then $R=R$.
    \begin{subproof}
        Assume $R$ is a coordinate record on $X$.
        $(coordinate record on X based on R)=(coordinate record on X based on R)$.
        Follows.
    \end{subproof}
    $(coordinate record on X based on R)=(coordinate record on X based on R)$.
    Follows.
\end{proof}
