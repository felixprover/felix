\section{Introducing nominal openness from the topology}

\import{topology/subspace.tex}

\begin{theorem}\label{membership_open_formation_probe}
    Let $X$ be a topological space.
    Let $U\in\opens[X]$.
    Then $U$ is open in $X$.
\end{theorem}
\begin{proof}
    Follows by notion.
\end{proof}

\begin{theorem}\label{empty_open_formation_probe}
    Let $X$ be a topological space.
    Then $\emptyset$ is open in $X$.
\end{theorem}
\begin{proof}
    Follows by notion.
\end{proof}

\begin{theorem}\label{nested_open_formation_probe}
    Let $X$ be a topological space.
    Suppose $U$ is open in $X$ and $V$ is open in $\subspaceTopology{X}{U}$.
    Then $V$ is open in $X$.
\end{theorem}
\begin{proof}
    Follows by notion.
\end{proof}

\begin{theorem}\label{carrier_open_formation_probe}
    Let $X$ be a topological space.
    Then $\carrier[X]$ is open in $X$.
\end{theorem}
\begin{proof}
    Follows by notion.
\end{proof}
