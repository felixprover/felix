\begin{definition}\label{reduced_diagonal}
    % Keep dynamic relation notation: it supplies whole truth, not a membership row.
    $R$ is a diagonalising relation iff for all $x,y$ such that $x\mathrel{R}y$
    we have $x\mathrel{R}x$.
\end{definition}

\begin{definition}\label{reduced_incident}
    $x$ is an incident point iff there exist $R,y$ such that $(x,y)\in R$.
\end{definition}

\begin{classification}\label{reduced_incident_intro}
    Suppose $(x,y)\in R$.
    Then $x$ is an incident point.
\end{classification}
\begin{proof}
    Follows by \auto{reduced_incident}.
\end{proof}

\begin{definition}\label{reduced_loop}
    $x$ is a looped point iff there exists $R$ such that $(x,x)\in R$.
\end{definition}

\begin{classification}\label{reduced_loop_intro}
    Suppose $(x,x)\in R$.
    Then $x$ is a looped point.
\end{classification}
\begin{proof}
    Follows by \auto{reduced_loop}.
\end{proof}
