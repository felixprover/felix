\section{Restricted transport guards}

\import{topology/embedding.tex}

\begin{theorem}\label{transport_open_subset_guard}
    Let $X$ be a topological space.
    Suppose $U$ is open in $X$.
    Then $U\subseteq\carrier[X]$.
\end{theorem}
\begin{proof}
    Follows by notion.
\end{proof}

\begin{theorem}\label{transport_restricted_homeomorphism_guard}
    Suppose $h$ is a homeomorphism from $X$ to $Y$ and $U$ is open in $X$.
    Then $\restrl{h}{U}$ is a homeomorphism from $\subspaceTopology{X}{U}$
        to $\subspaceTopology{Y}{\img{h}{U}}$.
\end{theorem}
\begin{proof}
    Follows by notion.
\end{proof}

\begin{theorem}\label{transport_image_open_guard}
    Suppose $h$ is a homeomorphism from $X$ to $Y$ and $U$ is open in $X$.
    Then $\img{h}{U}$ is open in $Y$.
\end{theorem}
\begin{proof}
    Follows by notion.
\end{proof}
