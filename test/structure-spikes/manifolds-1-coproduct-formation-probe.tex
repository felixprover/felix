\section{Coproduct formation probe}

\import{topology/manifold-open.tex}
\import{topology/manifold-transport.tex}
\import{topology/indexed-coproduct-open.tex}

Each summand is homeomorphic to an open subspace of the indexed coproduct.
Local Euclidean structure therefore passes to arbitrary disjoint unions.
This local statement does not assert second countability for an uncountable
family of manifolds.

\begin{theorem}\label{indexed_coproduct_of_locally_euclidean_spaces}
    Let $n$ be a natural number.
    Suppose $X$ is a function on $I$.
    Suppose for every $i\in I$ we have $X(i)$ is a topological space.
    Suppose for every $i\in I$ we have $X(i)$ is locally Euclidean of dimension $n$.
    Then $\IndexedCoproductTopology{I}{X}$ is locally Euclidean of dimension $n$.
\end{theorem}
\begin{proof}
    Let $P=\IndexedCoproductTopology{I}{X}$ and $A=\indexedTopologicalCarrierFamily{I}{X}$.
    For every $i\in I$ we have $X(i)$ is a topological space.
    $P$ is a topological space by \obv{indexed_coproduct_topology_is_topological_space}.
    $A$ is a function on $I$ by \obv{indexed_topological_carrier_family_type}.
    $\carrier[P]=\indexedSum{I}{A}$ by \obv{indexed_coproduct_topology_carrier}.
    Show that for every $p\in\carrier[P]$ there exists $U\in\opens[P]$ such that
        $p\in U$ and $\subspaceTopology{P}{U}$ is locally Euclidean of dimension $n$.
    \begin{subproof}
        Fix $p\in\carrier[P]$.
        Take $i,x$ such that $p=(i,x)$ and $i\in I$ and $x\in A(i)$ by
            \auto{indexed_sum,times_elem_is_tuple,indexed_sum_iff}.
        $A(i)=\carrier[X(i)]$ by \obv{indexed_topological_carrier_family_apply}.
        $X(i)$ is a topological space and $X(i)$ is locally Euclidean of dimension $n$.
        Let $f=\indexedSumInjection{I}{A}{i}$.
        $f$ is an open embedding from $X(i)$ to $P$ by
            \auto{indexed_coproduct_injection_is_open_embedding}.
        $\ran{f}\in\opens[P]$ and $f$ is a homeomorphism from $X(i)$ to
            $\subspaceTopology{P}{\ran{f}}$ by \obv{open_embedding,topological_embedding}.
        $f(x)=(i,x)$ by \auto{indexed_sum_injection_apply}.
        $f\in\funs{\carrier[X(i)]}{\carrier[P]}$ by \obv{open_embedding,topological_embedding}.
        $p\in\ran{f}$ by \auto{funs,funs_is_function,function_apply_elim,ran_intro}.
        $\subspaceTopology{P}{\ran{f}}$ is a topological space by
            \obv{subspace_topology_is_topological_space}.
        $\subspaceTopology{P}{\ran{f}}$ is locally Euclidean of dimension $n$ by
            \obv{homeomorphism_preserves_locally_euclidean}.
        Follows by \auto.
    \end{subproof}
    Follows by \auto{locally_euclidean_from_open_neighbourhoods}.
\end{proof}
