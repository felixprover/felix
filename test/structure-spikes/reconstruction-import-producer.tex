\begin{proposition}\label{imported_anchor}
    Let $X$ be a set.
    Then $X=X$.
\end{proposition}

\begin{construction}\label{imported_clear}
    Let $X$ be a set.
    The coordinate record $\ImportedClear{X}=
        coordinate record on X based on \ImportedCoordinate{X}
            with \atlas=\emptyset$ on $X$.
\end{construction}
\begin{proof}
    $\atlas[\ImportedClear{X}]\subseteq X$
        by \auto{coordinate_recordaggregateprojection_atlas,emptyset_subseteq,subseteq_refl}.
    Follows.
\end{proof}

\begin{instance}\label{imported_closed_clear}
    The coordinate record $\ImportedClosedClear$ on $\{\emptyset\}$ is
        $(coordinate record on \{\emptyset\} based on
            \ImportedCoordinate{\{\emptyset\}} with \atlas=\emptyset)$.
\end{instance}
\begin{proof}
    $\atlas[\ImportedClosedClear]\subseteq\{\emptyset\}$
        by \auto{coordinate_recordaggregateprojection_atlas,emptyset_subseteq,subseteq_refl}.
    Follows.
\end{proof}
