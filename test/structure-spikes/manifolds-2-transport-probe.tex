\section{Transported-coordinate consumers}

\import{topology/coordinate-homeomorphism.tex}

These imported statements consume the checked constructor results, including
composed, inverse, and restricted homeomorphisms. Their assumptions describe
the input presentations and maps, not the structures to be reconstructed.

\begin{theorem}\label{transported_presentations_consumer}
    Let $G$ be a coordinate pseudogroup and $X,Y$ be topological spaces.
    Let $S$ be a coordinate datum.
    Suppose $A$ presents $S$ on $X$ under $G$ and $B$ presents $S$ on $X$ under $G$.
    Suppose $h$ is a homeomorphism from $X$ to $Y$.
    Then $\GeneratedCoordinates{G}{Y}{\TransportedAtlas{A}{h}}
        =\GeneratedCoordinates{G}{Y}{\TransportedAtlas{B}{h}}$.
\end{theorem}
\begin{proof}
    $\TransportedCoordinates{G}{X}{Y}{S}{h}=\GeneratedCoordinates{G}{Y}{\TransportedAtlas{A}{h}}$ by
        \obv{transported_coordinates_from_presentation}.
    $\TransportedCoordinates{G}{X}{Y}{S}{h}=\GeneratedCoordinates{G}{Y}{\TransportedAtlas{B}{h}}$ by
        \obv{transported_coordinates_from_presentation}.
\end{proof}

\begin{theorem}\label{transported_charts_composition_consumer}
    Let $G$ be a coordinate pseudogroup and $X,Y,Z$ be topological spaces.
    Let $S$ be a relative coordinate structure on $X$ under $G$.
    Suppose $h$ is a homeomorphism from $X$ to $Y$ and $k$ is a homeomorphism from $Y$ to $Z$.
    Then $\ChartsAt{\TransportedCoordinates{G}{Y}{Z}{\TransportedCoordinates{G}{X}{Y}{S}{h}}{k}}{p}
        =\ChartsAt{\TransportedCoordinates{G}{X}{Z}{S}{k\circ h}}{p}$.
\end{theorem}
\begin{proof}
    $\TransportedCoordinates{G}{Y}{Z}{\TransportedCoordinates{G}{X}{Y}{S}{h}}{k}
        =\TransportedCoordinates{G}{X}{Z}{S}{k\circ h}$ by \obv{transported_coordinates_composition}.
\end{proof}

\begin{theorem}\label{transported_charts_roundtrip_consumer}
    Let $G$ be a coordinate pseudogroup and $X,Y$ be topological spaces.
    Suppose $S$ is an exact relative coordinate structure on $X$ under $G$.
    Suppose $h$ is a homeomorphism from $X$ to $Y$.
    Then $\ChartsAt{\TransportedCoordinates{G}{Y}{X}{\TransportedCoordinates{G}{X}{Y}{S}{h}}{\converse{h}}}{p}
        =\ChartsAt{S}{p}$.
\end{theorem}
\begin{proof}
    $\TransportedCoordinates{G}{Y}{X}{\TransportedCoordinates{G}{X}{Y}{S}{h}}{\converse{h}}=S$ by
        \obv{transported_coordinates_roundtrip}.
\end{proof}

\begin{theorem}\label{transported_restriction_charts_consumer}
    Let $G$ be a coordinate pseudogroup and $X,Y$ be topological spaces.
    Let $S$ be a relative coordinate structure on $X$ under $G$.
    Suppose $h$ is a homeomorphism from $X$ to $Y$ and $U$ is open in $X$.
    Then $\ChartsAt{\TransportedCoordinates{G}{\subspaceTopology{X}{U}}{\subspaceTopology{Y}{\img{h}{U}}}
            {\RestrictedCoordinates{G}{X}{S}{U}}{\restrl{h}{U}}}{p}
        =\ChartsAt{\RestrictedCoordinates{G}{Y}{\TransportedCoordinates{G}{X}{Y}{S}{h}}{\img{h}{U}}}{p}$.
\end{theorem}
\begin{proof}
    $\TransportedCoordinates{G}{\subspaceTopology{X}{U}}{\subspaceTopology{Y}{\img{h}{U}}}
            {\RestrictedCoordinates{G}{X}{S}{U}}{\restrl{h}{U}}
        =\RestrictedCoordinates{G}{Y}{\TransportedCoordinates{G}{X}{Y}{S}{h}}{\img{h}{U}}$ by
        \obv{transported_coordinates_restriction}.
\end{proof}

\begin{theorem}\label{bijection_transport_presentation_consumer}
    Let $G$ be a coordinate pseudogroup and $X$ be a topological space.
    Let $S$ be a coordinate datum.
    Suppose $A$ presents $S$ on $X$ under $G$.
    Let $h\in\bijections{\carrier[X]}{B}$.
    Then $\TransportedAtlas{A}{h}$ presents
        $\TransportedCoordinates{G}{X}{\SingleInitialTopology{B}{X}{\converse{h}}}{S}{h}$
        on $\SingleInitialTopology{B}{X}{\converse{h}}$ under $G$.
\end{theorem}
\begin{proof}
    Follows by \obv{bijection_transported_coordinates_presentation}.
\end{proof}
