\import{topology/topological-space.tex}

\begin{construction}\label{unproved_opens_copy}
    Let $X$ be a topological space.
    Let $B$ be a set.
    The topological space $\OpensCopy{X}{B}=
        topological space based on X with \opens=B$.
\end{construction}
\begin{proof}
    $\opens[\OpensCopy{X}{B}]\subseteq\pow{\carrier[\OpensCopy{X}{B}]}$.
    Follows.
\end{proof}
