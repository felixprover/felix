\import{set.tex}

\begin{struct}\label{closed_indexed_token}
    Let $X$ be a set.
    A closed indexed token $R$ over $X$ is a record equipped with
    \begin{enumerate}
        \item a set $\atlas[R]$;
    \end{enumerate}
    such that
    \begin{enumerate}
        \item\label{closed_indexed_token_bound} $\atlas[R]\subseteq X$.
    \end{enumerate}
\end{struct}

\begin{definition}\label{closed_unregistered_value}
    $\ClosedUnregisteredValue:\Set=\structure[closed indexed token over \{0\}]{\atlas=\emptyset}$.
\end{definition}

\begin{proposition}\label{closed_unregistered_theorem}
    $\ClosedUnregisteredValue$ is a closed indexed token over $\{0\}$.
\end{proposition}
\begin{proof}
    Follows by \auto{closed_unregistered_value,closed_indexed_token,
        closed_indexed_tokenaggregateexact,closed_indexed_tokenaggregateprojection_atlas,emptyset_subseteq}.
\end{proof}

\begin{definition}\label{closed_unregistered_guard}
    Let $R$ be a closed indexed token over $\{0\}$.
    $A$ is supported by the indexed token $R$ iff $A=A$.
\end{definition}

\begin{proposition}\label{closed_unregistered_guard_rejected}
    $\emptyset$ is supported by the indexed token $\ClosedUnregisteredValue$.
\end{proposition}
\begin{proof}
    Follows.
\end{proof}
