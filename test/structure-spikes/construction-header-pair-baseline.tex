\begin{definition}\label{copy_pair_coordinate_definition}
    Let $X,Y,R$ be sets.
    $\CopyPairCoordinateBaseline{X}{Y}{R}=
        \structure[coordinate record on (X,Y)]{\atlas=\atlas[R]}$.
\end{definition}

\begin{classification}\label{copy_pair_coordinate_classification}
    Let $X,Y$ be sets.
    Let $R$ be a coordinate record on $(X,Y)$.
    Then $\CopyPairCoordinateBaseline{X}{Y}{R}$ is a coordinate record on $(X,Y)$.
\end{classification}
\begin{proof}
    $\atlas[R]\subseteq (X,Y)$ by \obv{coordinate_atlas_bound}.
    Follows by \auto{copy_pair_coordinate_definition,coordinate_record,
        coordinate_recordaggregateexact,coordinate_recordaggregateprojection_atlas}.
\end{proof}
