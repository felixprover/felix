\begin{proposition}\label{measured_extensionality}
    Let $R,S$ be wide records.
    Suppose $R$ is an exact wide record and $S$ is an exact wide record.
    Suppose $\north[R]=\north[S]$ and $\south[R]=\south[S]$.
    Suppose $\east[R]=\east[S]$ and $\west[R]=\west[S]$.
    Suppose $\center[R]=\center[S]$ and $\upper[R]=\upper[S]$.
    Suppose $\lower[R]=\lower[S]$ and $\inner[R]=\inner[S]$.
    Suppose $\outer[R]=\outer[S]$ and $\middle[R]=\middle[S]$.
    Then $R=S$.
\end{proposition}
\begin{proof}
    Follows by structure extensionality.
\end{proof}
