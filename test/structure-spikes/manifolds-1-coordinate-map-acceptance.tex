\subsection{Restricting presentations}

\begin{theorem}\label{restricted_presentations_same_maximal_atlas}
    Let $G$ be a coordinate pseudogroup and $X$ be a topological space.
    Suppose $A$ is a coordinate atlas on $X$ under $G$ and $B$ is a coordinate atlas on $X$ under $G$.
    Suppose for every $a\in A$ we have for every $b\in B$ we have
        $a$ is compatible with $b$ on $X$ under $G$.
    Let $U\in\opens[X]$.
    Then $\MaxAtlas{G}{\subspaceTopology{X}{U}}{\RestrictedAtlas{A}{U}}
        =\MaxAtlas{G}{\subspaceTopology{X}{U}}{\RestrictedAtlas{B}{U}}$.
\end{theorem}
\begin{proof}
    $\MaxAtlas{G}{X}{A}=\MaxAtlas{G}{X}{B}$ by \obv{max_atlas_equality_iff}.
    Follows by \auto{restricted_atlas_maximalization}.
\end{proof}

\subsection{Changing the working presentations}

The local hypotheses use one pair of presentations; the conclusion uses
another pair for the same endpoint structures. Both directions of the public
presentation criterion are needed, with no maximal-atlas unfolding.

\begin{theorem}\label{coordinate_map_presentation_replacement}
    Let $C$ be a local-map calculus.
    Let $G,H$ be coordinate pseudogroups.
    Let $M,N$ be equipped coordinate spaces.
    Suppose $G$ is compatible with the calculus $C$ and $H$ is compatible with the calculus $C$.
    Suppose $M$ has coordinates under $G$ and $N$ has coordinates under $H$.
    Suppose $A$ presents $M$ on $M$ under $G$ and $B$ presents $N$ on $N$ under $H$.
    Suppose $A'$ presents $M$ on $M$ under $G$ and $B'$ presents $N$ on $N$ under $H$.
    Suppose $\dom{f}\in\opens[M]$ and $f\in\funs{\dom{f}}{\carrier[N]}$.
    Suppose for every $q\in\dom{f}$ there exists $c\in A$ such that there exists $d\in B$
        such that there exists $U\in\opens[M]$ such that
            $q\in U$ and $U\subseteq\dom{f}$ and $U\subseteq\dom{c}$ and
            $\img{f}{U}\subseteq\dom{d}$ and
            $\CoordExpr{d}{f}{c}{U}$ is admissible from $\model[G]$ to $\model[H]$ in $C$.
    Let $p\in\dom{f}$.
    Then there exists $c\in A'$ such that there exists $d\in B'$
        such that there exists $U\in\opens[M]$ such that
            $p\in U$ and $U\subseteq\dom{f}$ and $U\subseteq\dom{c}$ and
            $\img{f}{U}\subseteq\dom{d}$ and
            $\CoordExpr{d}{f}{c}{U}$ is admissible from $\model[G]$ to $\model[H]$ in $C$.
\end{theorem}
\begin{proof}
    $f$ is coordinate-admissible from $M$ under $G$ to $N$ under $H$ in $C$ by
        \obv{coordinate_admissible_presentations}.
    Follows by \auto{coordinate_admissible_presentations}.
\end{proof}

\subsection{Weakening the calculus without changing endpoints}

\begin{theorem}\label{coordinate_map_continuous_weakening}
    Let $C$ be a local-map calculus.
    Let $G,H$ be coordinate pseudogroups.
    Let $M,N$ be equipped coordinate spaces.
    Suppose $f$ is coordinate-admissible from $M$ under $G$ to $N$ under $H$ in $C$.
    Then $f$ is coordinate-admissible from $M$ under $G$ to $N$ under $H$
        in $\ContinuousCalculus{\models[C]}$.
\end{theorem}
\begin{proof}
    $\models[C]$ is a topological model family by \obv{calculus_models_are_topological_family}.
    $\models[C]\subseteq\models[C]$ by \obv{subseteq_refl}.
    $C$ is a subcalculus of $\ContinuousCalculus{\models[C]}$ by
        \obv{continuous_calculus_maximal}.
    Follows by \obv{coordinate_admissible_subcalculus}.
\end{proof}

\begin{theorem}\label{relative_empty_coordinate_map}
    Let $C$ be a local-map calculus.
    Let $G,H$ be coordinate pseudogroups.
    Let $X,Y$ be topological spaces.
    Let $S$ be a relative coordinate structure on $X$ under $G$.
    Let $T$ be a relative coordinate structure on $Y$ under $H$.
    Suppose $G$ is compatible with the calculus $C$ and $H$ is compatible with the calculus $C$.
    Then $\emptyset$ is coordinate-admissible
        from $\Equip{X}{S}$ under $G$ to $\Equip{Y}{T}$ under $H$ in $C$.
\end{theorem}
\begin{proof}
    $\emptyset$ is a continuous partial map from $X$ to $Y$ by \obv{empty_continuous_partial_map}.
    $\dom{\emptyset}\in\opens[X]$ and $\emptyset\in\funs{\dom{\emptyset}}{\carrier[Y]}$ by
        \obv{continuous_partial_map,continuous_partial_map_type}.
    $\dom{\emptyset}=\emptyset$ by \obv{dom_emptyset}.
    Follows by \auto{coordinate_admissible_equip_iff}.
\end{proof}

\begin{theorem}\label{relative_identity_coordinate_neighborhood}
    Let $C$ be a local-map calculus.
    Let $G$ be a coordinate pseudogroup.
    Let $X$ be a topological space.
    Let $S$ be a relative coordinate structure on $X$ under $G$.
    Suppose $G$ is compatible with the calculus $C$.
    Let $W\in\opens[X]$ and $p\in W$.
    Then there exists $c\in\ChartsAt{S}{p}$ such that there exists $d\in\atlas[S]$
        such that there exists $U\in\opens[X]$ such that
            $p\in U$ and $U\subseteq W$ and $U\subseteq\dom{c}$ and
            $\img{\identity{W}}{U}\subseteq\dom{d}$ and
            $\CoordExpr{d}{\identity{W}}{c}{U}$ is admissible from $\model[G]$ to $\model[G]$ in $C$.
\end{theorem}
\begin{proof}
    $\Equip{X}{S}$ has coordinates under $G$ by \obv{equip_preserves_coordinates}.
    $\opens[\Equip{X}{S}]=\opens[X]$ by \obv{equipprojection_opens}.
    $\identity{W}$ is coordinate-admissible
        from $\Equip{X}{S}$ under $G$ to $\Equip{X}{S}$ under $G$ in $C$ by
        \obv{coordinate_admissible_identity}.
    $\dom{\identity{W}}=W$ by \obv{id_dom}.
    Follows by \auto{coordinate_admissible_equip_iff}.
\end{proof}

\subsection{Composition on a proper domain}

The second identity is defined on a proper open subset of the first one's
image. Composition must retain its natural domain rather than require total
coverage of the intermediate image.

\begin{theorem}\label{coordinate_identity_proper_composition}
    Let $C$ be a local-map calculus.
    Let $G$ be a coordinate pseudogroup.
    Let $M$ be an equipped coordinate space.
    Suppose $G$ is compatible with the calculus $C$ and $M$ has coordinates under $G$.
    Suppose $U,V\in\opens[M]$ and $V\subseteq U$ and $V\neq U$.
    Then $\identity{V}\circ\identity{U}$ is coordinate-admissible
        from $M$ under $G$ to $M$ under $G$ in $C$
        and $\dom{\identity{V}\circ\identity{U}}=V$
        and not $\ran{\identity{U}}\subseteq\dom{\identity{V}}$.
\end{theorem}
\begin{proof}
    $\identity{U}$ is coordinate-admissible from $M$ under $G$ to $M$ under $G$ in $C$ and
        $\identity{V}$ is coordinate-admissible from $M$ under $G$ to $M$ under $G$ in $C$ by
        \obv{coordinate_admissible_identity}.
    $\identity{V}\circ\identity{U}$ is coordinate-admissible
        from $M$ under $G$ to $M$ under $G$ in $C$ by \obv{coordinate_admissible_composition}.
    $U\inter V=V$ by \obv{inter_absorb_supseteq_left}.
    $\identity{V}\circ\identity{U}=\identity{V}$ by \auto{identity_partial_composition}.
    $\dom{\identity{V}}=V$ and $\ran{\identity{U}}=U$ by \obv{id_dom,id_ran}.
    Not $U\subseteq V$ by \auto{subseteq_antisymmetric}.
\end{proof}
