\section{Restricted-coordinate formation controls}

\import{topology/coordinate-restriction.tex}

\begin{theorem}\label{restricted_coordinates_result_control}
    Let $G$ be a coordinate pseudogroup and $X$ be a topological space.
    Let $S$ be a relative coordinate structure on $X$ under $G$.
    Suppose $U$ is open in $X$.
    Then $\RestrictedCoordinates{G}{X}{S}{U}$ is a relative coordinate structure
        on $\subspaceTopology{X}{U}$ under $G$.
\end{theorem}
\begin{proof}
    Follows by \obv{restricted_coordinates}.
\end{proof}

\begin{theorem}\label{restricted_coordinates_charts_at_control}
    Let $G$ be a coordinate pseudogroup and $X$ be a topological space.
    Let $S$ be a relative coordinate structure on $X$ under $G$.
    Suppose $U$ is open in $X$.
    Then $\ChartsAt{\RestrictedCoordinates{G}{X}{S}{U}}{p}
        =\ChartsAt{\RestrictedCoordinates{G}{X}{S}{U}}{p}$.
\end{theorem}
\begin{proof}
    Follows.
\end{proof}

\begin{theorem}\label{presented_restricted_coordinates_charts_at_probe}
    Let $G$ be a coordinate pseudogroup and $X$ be a topological space.
    Let $S$ be a coordinate datum.
    Suppose $A$ presents $S$ on $X$ under $G$.
    Suppose $U$ is open in $X$.
    Then $\ChartsAt{\RestrictedCoordinates{G}{X}{S}{U}}{p}
        =\ChartsAt{\RestrictedCoordinates{G}{X}{S}{U}}{p}$.
\end{theorem}
\begin{proof}
    Follows.
\end{proof}
