\section{Closed weakening examples}

\import{topology/positive-coordinate-examples.tex}

\begin{theorem}\label{closed_positive_weakening_consumer}
    Then $\atlas[\WeakenedCoordinates{\PositiveRealPseudogroup}{\FullPseudogroup{\RealTopology}}{\RealTopology}
            {\GeneratedCoordinates{\PositiveRealPseudogroup}{\RealTopology}{\RealOneChart}}]
        =\atlas[\WeakenedCoordinates{\PositiveRealPseudogroup}{\FullPseudogroup{\RealTopology}}{\RealTopology}
            {\GeneratedCoordinates{\PositiveRealPseudogroup}{\RealTopology}{\RealReflectionChart}}]$.
\end{theorem}
\begin{proof}
    Follows by \obv{real_positive_weakening_agrees}.
\end{proof}

\begin{theorem}\label{closed_weakened_reflection_consumer}
    Then $\surrealRealNegationGraph$ is coordinate-admissible
        from $\Equip{\RealTopology}{\WeakenedCoordinates{\PositiveRealPseudogroup}{\FullPseudogroup{\RealTopology}}{\RealTopology}
            {\GeneratedCoordinates{\PositiveRealPseudogroup}{\RealTopology}{\RealOneChart}}}$
        under $\FullPseudogroup{\RealTopology}$
        to $\Equip{\RealTopology}{\WeakenedCoordinates{\PositiveRealPseudogroup}{\FullPseudogroup{\RealTopology}}{\RealTopology}
            {\GeneratedCoordinates{\PositiveRealPseudogroup}{\RealTopology}{\RealReflectionChart}}}$
        under $\FullPseudogroup{\RealTopology}$ in $\ContinuousCalculus{\{\RealTopology\}}$.
\end{theorem}
\begin{proof}
    Let $X=\RealTopology$ and $G=\PositiveRealPseudogroup$.
    Let $H=\FullPseudogroup{X}$ and $C=\ContinuousCalculus{\{X\}}$.
    $X$ is a topological space by \obv{real_topology_is_topological_space}.
    $G$ is a coordinate pseudogroup and $H$ is a coordinate pseudogroup by
        \auto{positive_real_pseudogroup,full_pseudogroup}.
    $G$ is a subpseudogroup of $H$ by \auto{positive_real_pseudogroup_in_full}.
    $\model[G]=X$ and $\model[H]=X$ by \auto{positive_real_pseudogroup_model,full_pseudogroup_model}.
    $\{X\}$ is a topological model family by \obv{singleton_topological_model_family}.
    $\model[G]\in\{X\}$ and $\model[H]\in\{X\}$ by \obv{singleton_iff}.
    $C$ is a local-map calculus by \auto{continuous_calculus}.
    $G$ is compatible with the calculus $C$ and $H$ is compatible with the calculus $C$ by
        \auto{pseudogroup_continuous_calculus_compatible}.
    $\RealOneChart$ is a coordinate atlas on $X$ under $G$ and
        $\RealReflectionChart$ is a coordinate atlas on $X$ under $G$ by
        \obv{real_identity_positive_atlas,real_reflection_positive_atlas}.
    Let $S=\GeneratedCoordinates{G}{X}{\RealOneChart}$ and
        $T=\GeneratedCoordinates{G}{X}{\RealReflectionChart}$.
    $S$ is a relative coordinate structure on $X$ under $G$ and
        $T$ is a relative coordinate structure on $X$ under $G$ by \auto{generated_coordinates}.
    $\surrealRealNegationGraph$ is coordinate-admissible from $\Equip{X}{S}$ under $G$
        to $\Equip{X}{T}$ under $G$ in $C$ by \auto{real_reflection_transported_admissible}.
    Follows by \auto{weakened_coordinates_admissible}.
\end{proof}

\begin{theorem}\label{closed_empty_weakening_consumer}
    Let $G=\PositiveRealPseudogroup$ and $H=\FullPseudogroup{\RealTopology}$.
    Let $X=\RealTopology$ and $E=\subspaceTopology{X}{\emptyset}$.
    Let $S=\GeneratedCoordinates{G}{X}{\RealOneChart}$.
    Then $\WeakenedCoordinates{G}{H}{E}{\RestrictedCoordinates{G}{X}{S}{\emptyset}}
        =\RestrictedCoordinates{H}{X}{\GeneratedCoordinates{H}{X}{\RealOneChart}}{\emptyset}$ and
        $\atlas[\WeakenedCoordinates{G}{H}{E}{\RestrictedCoordinates{G}{X}{S}{\emptyset}}]=\{\emptyset\}$.
\end{theorem}
\begin{proof}
    $X$ is a topological space by \obv{real_topology_is_topological_space}.
    $G$ is a coordinate pseudogroup and $H$ is a coordinate pseudogroup by
        \auto{positive_real_pseudogroup,full_pseudogroup}.
    $G$ is a subpseudogroup of $H$ by \auto{positive_real_pseudogroup_in_full}.
    $\emptyset$ is open in $X$ by \obv{emptyset_open}.
    $\RealOneChart$ is a coordinate atlas on $X$ under $G$ and
        $\RealOneChart$ is a coordinate atlas on $X$ under $H$ by
        \obv{real_identity_positive_atlas,real_identity_full_atlas}.
    $S$ is a relative coordinate structure on $X$ under $G$ by \auto{generated_coordinates}.
    $\WeakenedCoordinates{G}{H}{X}{S}=\GeneratedCoordinates{H}{X}{\RealOneChart}$ by
        \obv{real_positive_coordinates_weaken_to_full}.
    $\WeakenedCoordinates{G}{H}{E}{\RestrictedCoordinates{G}{X}{S}{\emptyset}}
        =\RestrictedCoordinates{H}{X}{\WeakenedCoordinates{G}{H}{X}{S}}{\emptyset}$ by
        \auto{weakened_coordinates_restriction}.
    $\GeneratedCoordinates{H}{X}{\RealOneChart}$ is a relative coordinate structure on $X$ under $H$ by
        \obv{generated_coordinates}.
    $\atlas[\RestrictedCoordinates{H}{X}{\GeneratedCoordinates{H}{X}{\RealOneChart}}{\emptyset}]=\{\emptyset\}$ by
        \obv{restricted_coordinates_empty_atlas}.
\end{proof}
