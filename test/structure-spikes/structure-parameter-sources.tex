\begin{abbreviation}\label{parameter_keep_left}
    $\KeepLeft{X}{Y}=X$.
\end{abbreviation}

\begin{abbreviation}\label{parameter_other_left}
    $\OtherLeft{X}{Y}=X$.
\end{abbreviation}

\begin{definition}\label{parameter_source_copy}
    The parameter source copy of $X$ is $X$.
\end{definition}

\begin{definition}\label{parameter_record_provider}
    Let $X,Y,R$ be sets.
    $\ParameterRecord{X}{Y}{R}=\structure[coordinate record on \KeepLeft{X}{Y}]{\atlas=\atlas[R]}$.
\end{definition}

\begin{classification}\label{parameter_provider_result}
    Let $X,Y$ be sets.
    Let $R$ be a coordinate record on $\KeepLeft{X}{Y}$.
    Then $\ParameterRecord{X}{Y}{R}$ is a coordinate record on $\KeepLeft{X}{Y}$.
\end{classification}
\begin{proof}
    $\atlas[R]\subseteq X$ by \obv{coordinate_atlas_bound}.
    Follows by \auto{parameter_record_provider,coordinate_record,
        coordinate_recordaggregateexact,coordinate_recordaggregateprojection_atlas}.
\end{proof}

\begin{classification}\label{parameter_finite_rule}
    Let $X,Y$ be sets.
    Let $R$ be a coordinate record on $\KeepLeft{X}{Y}$.
    Let $S$ be a coordinate record on $\OtherLeft{X}{Y}$.
    Then $R$ is a coordinate record on $\OtherLeft{X}{Y}$.
\end{classification}
\begin{proof}
    Follows by assumption.
\end{proof}

\begin{proposition}\label{parameter_nested_source}
    Let $X,Y$ be sets.
    Let $R$ be a coordinate record on $(X,(\KeepLeft{X}{Y},Y))$.
    Then $R$ is a coordinate record on $(X,(\KeepLeft{X}{Y},Y))$.
\end{proposition}
\begin{proof}
    Follows by assumption.
\end{proof}

\begin{proposition}\label{parameter_field_source}
    Let $X$ be a set.
    Let $R$ be a coordinate record on $X$.
    Let $S$ be a coordinate record on $\atlas[R]$.
    Then $S$ is a coordinate record on $\atlas[R]$.
\end{proposition}
\begin{proof}
    Follows by assumption.
\end{proof}

\begin{proposition}\label{parameter_function_source}

    Then $R$ is a coordinate record on the parameter source copy of $X$.
\end{proposition}
\begin{proof}
    Follows by assumption.
\end{proof}
