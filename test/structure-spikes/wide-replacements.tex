\section{Simultaneous replacements}

\begin{proposition}\label{wide_swap_entry_order}
    Let $R$ be a wide record.
    Then $(wide record based on R with \north=\south[R] and \south=\north[R])=
        (wide record based on R with \south=\north[R] and \north=\south[R])$.
\end{proposition}

\begin{proposition}\label{wide_swap_north}
    Let $R$ be a wide record.
    Then $\north[(wide record based on R with \north=\south[R] and \south=\north[R])]=\south[R]$.
\end{proposition}
\begin{proof}
    Follows by \obv{wide_baseaggregateprojection_north}.
\end{proof}

\begin{proposition}\label{wide_swap_south}
    Let $R$ be a wide record.
    Then $\south[(wide record based on R with \north=\south[R] and \south=\north[R])]=\north[R]$.
\end{proposition}
\begin{proof}
    Follows by \obv{wide_baseaggregateprojection_south}.
\end{proof}

\begin{proposition}\label{wide_swap_untouched}

\begin{proof}

\begin{proposition}\label{wide_unchanged_replacements}
    Let $R$ be a wide record.
    Suppose $R$ is an exact wide record.
    Suppose $U=\north[R]$ and $V=\south[R]$.
    Then $(wide record based on R with \north=U and \south=V)=R$.
\end{proposition}
\begin{proof}
    Let $P=wide record based on R with \north=U and \south=V$.
    $wide record based on R with \north=U and \south=V$ is an exact wide record
        by \obv{wide_baseaggregateexact}.
    $P$ is an exact wide record by \auto{wide_baseaggregateexact}.
    $\north[P]=\north[R]$ by \auto{wide_baseaggregateprojection_north}.
    $\south[P]=\south[R]$ by \auto{wide_baseaggregateprojection_south}.
    $\east[P]=\east[R]$ by \auto{wide_baseaggregateprojection_east}.
    $\west[P]=\west[R]$ by \auto{wide_baseaggregateprojection_west}.
    $\center[P]=\center[R]$ by \auto{wide_baseaggregateprojection_center}.
    $\upper[P]=\upper[R]$ by \auto{wide_baseaggregateprojection_upper}.
    $\lower[P]=\lower[R]$ by \auto{wide_baseaggregateprojection_lower}.
    $\inner[P]=\inner[R]$ by \auto{wide_baseaggregateprojection_inner}.
    $\outer[P]=\outer[R]$ by \auto{wide_baseaggregateprojection_outer}.
    $\middle[P]=\middle[R]$ by \auto{wide_baseaggregateprojection_middle}.
    Follows by structure extensionality.
\end{proof}
