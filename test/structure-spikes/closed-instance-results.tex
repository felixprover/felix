\begin{definition}\label{closed_instance_guard}
    Let $R$ be a reflexive record.
    $A$ is supported by $R$ iff $A=A$.
\end{definition}

\begin{definition}\label{closed_parameter_guard}
    Let $R$ be a coordinate record on $\ClosedRefined$.
    $A$ is supported on the coordinates of $R$ iff $A=A$.
\end{definition}

\begin{proposition}\label{closed_parameter_guard_consumed}
    $\emptyset$ is supported on the coordinates of $\ClosedParameter$.
\end{proposition}
\begin{proof}
    $\ClosedParameter$ is a coordinate record on $\ClosedRefined$ by notion.
    Follows by \obv{closed_parameter_guard}.
\end{proof}

\begin{proposition}\label{closed_finite_notion}

\begin{construction}\label{closed_instance_copy}

\begin{proposition}\label{closed_instance_nested_guard}
    $\emptyset$ is supported by $\ClosedCopy{\ClosedCopy{\ClosedRefined}}$.
\end{proposition}
\begin{proof}
    $\ClosedCopy{\ClosedCopy{\ClosedRefined}}$ is a reflexive record by notion.
    Follows by \obv{closed_instance_guard}.
\end{proof}

\begin{proposition}\label{closed_instance_notion}

\end{proposition}
\begin{proof}

\begin{proposition}\label{closed_instance_local_alias}
    For every set $X$, $\emptyset$ is supported by $\ClosedRefined$.
\end{proposition}
\begin{proof}
    Fix $X$.
    Let $R=\ClosedRefined$.
    $R$ is a reflexive record by notion.
    $\ClosedCopy{R}$ is a reflexive record by notion.
    $\emptyset$ is supported by $\ClosedCopy{R}$ by \obv{closed_instance_guard}.
    $\ClosedRefined$ is a reflexive record by notion.
    Follows by \obv{closed_instance_guard}.
\end{proof}
