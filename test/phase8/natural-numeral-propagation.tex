\begin{definition}\label{natural_numeral_binding}
    $\naturalNumeralBinding=1$.
\end{definition}

\begin{struct}\label{natural_numeral_structure}
    A natural numeral structure $X$ is a onesorted structure equipped with
    \begin{enumerate}
        \item\label{natural_numeral_structure_point}
            a distinguished point
            $\naturalNumeralPoint[X]\in 1$;
    \end{enumerate}
    .
\end{struct}

\begin{datatype}\label{natural_numeral_datatype}
    Define $\naturalNumeralData$ inductively as follows.
    \begin{enumerate}
        \item $\naturalNumeralConstructor{n}\in\naturalNumeralData$ for
            $n\in 1$.
    \end{enumerate}
\end{datatype}
