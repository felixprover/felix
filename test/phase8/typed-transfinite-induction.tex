\begin{definition}\label{typed_transfinite_guard}
    $x$ is a typed transfinite ordinal iff $x=x$.
\end{definition}

\begin{proposition}\label{typed_transfinite_guard_hereditary}
    Let $\alpha$ be a typed transfinite ordinal.
    Let $\beta\in\alpha$.
    Then $\beta$ is a typed transfinite ordinal.
\end{proposition}
\begin{proof}
    Follows by \auto{typed_transfinite_guard}.
\end{proof}

\begin{theorem}\label{typed_transfinite_induction}
    For every typed transfinite ordinal $\alpha$ we have
    $\forall P:\Set\to\Prop.
        (\forall x:\Set.\ (\forall w:\Set.\ w\in x\implies\Apply{P}{w})
             \implies\Apply{P}{x})
        \implies\Apply{P}{\alpha}$.
\end{theorem}
\begin{proof}[Proof by transfinite induction]
    Fix $P$.
    Assume $\forall x:\Set.\ (\forall w:\Set.\ w\in x\implies\Apply{P}{w})
         \implies\Apply{P}{x}$.
    Follows by \auto.
\end{proof}
