\begin{signature}\label{typed_case_predicate}
    $\mathcalH : \Set\to(\Set\to\Prop)\to(\Set\to\Prop)\to\Prop$.
\end{signature}

\begin{proposition}\label{typed_case_analysis}
    $\forall \alpha:\Set.\ \forall p,q:\Set\to\Prop.
        \Apply{\Apply{\Apply{\mathcalH}{\alpha}}{p}}{q}\lor
        \lnot\Apply{\Apply{\Apply{\mathcalH}{\alpha}}{p}}{q}$.
\end{proposition}
\begin{proof}
    Fix $\alpha,p,q$.
    \begin{byCase}
    \caseOf{$\Apply{\Apply{\Apply{\mathcalH}{\alpha}}{p}}{q}$.}
        Follows by \auto.
    \caseOf{$\lnot\Apply{\Apply{\Apply{\mathcalH}{\alpha}}{p}}{q}$.}
        Follows by \auto.
    \end{byCase}
\end{proof}
