\begin{signature}\label{funs}
    $\funs{A}{B}$ is a set.
\end{signature}

\begin{axiom}\label{set_map_arrow}
    Let $f,A,B$ be sets.
    Then $f:A\to B$.
\end{axiom}

\begin{axiom}\label{set_map_expanded}
    Let $f,A,B$ be sets.
    Then $f\in\funs{A}{B}$.
\end{axiom}

\begin{proposition}\label{set_map_backend}
    Let $f,A,B$ be sets.
    Then $f:A\to B$.
\end{proposition}
\begin{proof}
    Follows by \auto{set_map_expanded}.
\end{proof}
