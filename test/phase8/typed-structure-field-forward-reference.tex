\begin{struct}\label{typed_field_forward_reference}
    A typed field forward reference $X$ is a onesorted structure equipped with
    \begin{enumerate}
        \item a distinguished point $\fieldFirst[X]\in\fieldLater[X]$;
        \item a later carrier element $\fieldLater[X]\in\carrier[X]$;
    \end{enumerate}
    .
\end{struct}
