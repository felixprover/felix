\begin{signature}\label{parameterized_funs}
    $\funs{A}{B}$ is a set.
\end{signature}

\begin{struct}\label{scalar_base}
    A scalar base $R$ is a onesorted structure such that
    \begin{enumerate}
        \item\label{scalar_base_refl} $R=R$.
    \end{enumerate}
\end{struct}

\begin{struct}\label{typed_field_schema}
    A typed field schema $X$ is a onesorted structure equipped with
    \begin{enumerate}
        \item\label{typed_field_point}
            a distinguished point
            $\fieldPoint[X]\in\carrier[X]$;
        \item\label{typed_field_map}
            a transition map
            $\fieldMap[X]:\carrier[X]\to\carrier[X]$;
    \end{enumerate}
    .
\end{struct}

\begin{struct}\label{left_action}
    Let $R$ be a scalar base.
    A left action $M$ over $R$ is a onesorted structure such that
    \begin{enumerate}
        \item\label{left_action_refl} $M=M$.
    \end{enumerate}
\end{struct}

\begin{struct}\label{unital_left_action}
    Let $R$ be a scalar base.
    A unital left action $M$ over $R$ is a left action over $R$ such that
    \begin{enumerate}
        \item\label{unital_left_action_refl} $R=R$.
    \end{enumerate}
\end{struct}

\begin{struct}\label{biaction}
    Let $R$ be a scalar base.
    Let $S$ be a scalar base.
    A biaction $M$ over $R$ and $S$ is a left action over $R$ such that
    \begin{enumerate}
        \item\label{biaction_refl} $S=S$.
    \end{enumerate}
\end{struct}

\begin{proposition}\label{left_action_parameter}
    Let $M$ be a left action over $R$.
    Then $R$ is a scalar base.
\end{proposition}
\begin{proof}
    Follows by \auto{left_action}.
\end{proof}

\begin{proposition}\label{unital_left_action_parent}
    Let $M$ be a unital left action over $R$.
    Then $M$ is a left action over $R$.
\end{proposition}
\begin{proof}
    Follows by \auto{unital_left_action}.
\end{proof}

\begin{proposition}\label{biaction_second_parameter}
    Let $M$ be a biaction over $R$ and $S$.
    Then $S$ is a scalar base.
\end{proposition}
\begin{proof}
    Follows by \auto{biaction}.
\end{proof}

\begin{proposition}\label{grouped_left_actions}
    Let $M,N$ be left actions over $R$.
    Then $M=M$ and $N=N$.
\end{proposition}
\begin{proof}
    Follows.
\end{proof}

\begin{struct}\label{named_biaction}
    Let $R$ be a scalar base.
    Let $S$ be a scalar base.
    A named biaction $M$ over $R$ and $S$ is
    a left action over $R$ \label{named_biaction_left}
    and a left action over $S$ \label{named_biaction_right}.
\end{struct}

\begin{proposition}\label{named_biaction_parent}
    Let $M$ be a named biaction over $R$ and $S$.
    Then $M$ is a left action over $S$.
\end{proposition}
\begin{proof}
    Follows by \obv{named_biaction_right}.
\end{proof}

\begin{definition}\label{presentation_relation}
    $x\presentationRelation{A}y$ iff $x\in A$ and $y\in A$.
\end{definition}

\begin{definition}\label{presentation_predicate}
    $\presentationPredicate{x}{y}$ iff $x=y$.
\end{definition}

\begin{struct}\label{secondary_field_schema}
    A secondary field schema $X$ is a onesorted structure equipped with
    \begin{enumerate}
        \item\label{secondary_field_point}
            a point $\fieldPoint[X]\in\carrier[X]$;
    \end{enumerate}
    .
\end{struct}

\begin{definition}\label{presentation_adjective}
    $x$ is \presentationClass\ iff $x=x$.
\end{definition}

\begin{struct}\label{symbolic_schema}
    A \presentationSchema\ $X$ is a onesorted structure.
\end{struct}
