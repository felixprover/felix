\begin{proposition}\label{eta_function_equality}
    $\forall f,g:\Set\to\Set.
        (\forall x:\Set. \Apply{f}{x}=\Apply{g}{x}) \implies f=g$.
\end{proposition}
\begin{proof}
    Fix $f,g$.
    Assume $\forall x:\Set. \Apply{f}{x}=\Apply{g}{x}$.
    Follows.
\end{proof}

\begin{proposition}\label{eta_predicate_equality}
    $\forall P,Q:\Set\to\Prop.
        (\forall x:\Set. \Apply{P}{x}\iff\Apply{Q}{x}) \implies P=Q$.
\end{proposition}
\begin{proof}
    Fix $P,Q$.
    Assume $\forall x:\Set. \Apply{P}{x}\iff\Apply{Q}{x}$.
    Follows.
\end{proof}

\begin{proposition}\label{eta_argument_transport}
    $\forall f,g:\Set\to\Set. \forall R:(\Set\to\Set)\to\Prop.
        (\forall x:\Set. \Apply{f}{x}=\Apply{g}{x})
        \implies \Apply{R}{f} \implies \Apply{R}{g}$.
\end{proposition}
\begin{proof}
    Fix $f,g,R$.
    Assume $\forall x:\Set. \Apply{f}{x}=\Apply{g}{x}$.
    Assume $\Apply{R}{f}$.
    Follows.
\end{proof}
