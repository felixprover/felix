\import{library/algebra/magma.tex}

\begin{definition}\label{notion_endpoint_field_domain_value}
    Let $x\in\dom{f}$.
    $\notionEndpointFieldDomainValue{f}{x}=x$.
\end{definition}

\begin{definition}\label{notion_endpoint_field_result_value}
    Let $y\in B$.
    $\notionEndpointFieldResultValue{y}{B}=y$.
\end{definition}

\begin{proposition}\label{notion_endpoint_field_domain}
    Let $A$ be a magma.
    Let $a,b\in\carrier[A]$.
    Then $\notionEndpointFieldDomainValue{\mul[A]}{(a,b)}=(a,b)$.
\end{proposition}
\begin{proof}
    Follows.
\end{proof}

\begin{proposition}\label{notion_endpoint_field_result}
    Let $A$ be a magma.
    Let $a,b\in\carrier[A]$.
    Then $\notionEndpointFieldResultValue
        {\mul[A](a,b)}{\carrier[A]}=\mul[A](a,b)$.
\end{proposition}
\begin{proof}
    Follows.
\end{proof}
