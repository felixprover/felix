\begin{definition}\label{horn_scoped_primary}
    $X$ is a scoped primary iff $X=\emptyset$.
\end{definition}

\begin{definition}\label{horn_scoped_secondary}
    $X$ is a scoped secondary iff $X=\{\emptyset\}$.
\end{definition}

\begin{abbreviation}\label{horn_scoped_joint}
    $I$ is scoped-joint iff for every $i\in I$ we have $i$ is a scoped primary and $i$ is a scoped secondary.
\end{abbreviation}

\begin{definition}\label{horn_scoped_combined}
    Let $I$ be a set that is scoped-joint.
    $\hornScopedCombined{I}=I$.
\end{definition}

\begin{proposition}\label{horn_scoped_conjunction_goal}
    Suppose for every $i\in I$ we have $i$ is a scoped primary.
    Suppose for every $j\in I$ we have $j$ is a scoped secondary.
    Then for every $k\in I$ we have $k$ is a scoped primary and $k$ is a scoped secondary.
\end{proposition}
\begin{proof}
    Follows by notion.
\end{proof}

\begin{proposition}\label{horn_scoped_contract}
    Suppose for every $i\in I$ we have $i$ is a scoped primary.
    Suppose for every $j\in I$ we have $j$ is a scoped secondary.
    Then $\hornScopedCombined{I}=I$.
\end{proposition}
\begin{proof}
    Follows by \obv{horn_scoped_combined}.
\end{proof}

\begin{definition}\label{horn_scoped_complete}
    $I$ is a scoped complete iff $I=I$.
\end{definition}

\begin{classification}\label{horn_scoped_whole_premise}
    Suppose for every $i\in I$ we have $i$ is a scoped primary and $i$ is a scoped secondary.
    Then $I$ is a scoped complete.
\end{classification}
\begin{proof}
    Follows by \obv{horn_scoped_complete}.
\end{proof}

\begin{proposition}\label{horn_scoped_whole_premise_consumer}
    Suppose for every $i\in I$ we have $i$ is a scoped primary.
    Suppose for every $j\in I$ we have $j$ is a scoped secondary.
    Then $I$ is a scoped complete.
\end{proposition}
\begin{proof}
    Follows by notion.
\end{proof}

\begin{proposition}\label{horn_scoped_implication}
    Let $A$ be a set.
    Then if $A$ is a scoped primary then $A$ is a scoped primary.
\end{proposition}
\begin{proof}
    Follows by notion.
\end{proof}


\begin{proposition}\label{horn_scoped_equality_assumption}
    Let $A$ be a scoped primary.
    Then for every $B$ if $A=B$ then $B$ is a scoped primary.
\end{proposition}
\begin{proof}
    Follows by notion.
\end{proof}

\begin{proposition}\label{horn_scoped_truth}
    Let $A$ be a set.
    Then $A$ is a set.
\end{proposition}
\begin{proof}
    Follows by notion.
\end{proof}
