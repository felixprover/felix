\begin{definition}\label{horn_positive_target}
    $X$ is a positive target iff $X=X$.
\end{definition}

\begin{classification}\label{horn_positive_disjunctive_parameter}
    Suppose $x\in A$ or $x$ is a connective primary.
    Then $A$ is a positive target.
\end{classification}
\begin{proof}
    Follows by \obv{horn_positive_target}.
\end{proof}

\begin{proposition}\label{horn_positive_parameter_consumer}
    Suppose $a\in A$.
    Then $A$ is a positive target.
\end{proposition}
\begin{proof}
    Follows by notion.
\end{proof}
