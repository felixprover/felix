\begin{definition}\label{notion_order_base_b}
    $X$ is a notion order beta base iff $X=X$.
\end{definition}

\begin{definition}\label{notion_order_base_a}
    $X$ is a notion order alpha base iff $X=X$.
\end{definition}

\begin{definition}\label{notion_ordered_object}
    Let $X$ be a notion order beta base.
    Let $X$ be a notion order alpha base.
    Let $X$ be a notion order beta base.
    Let $Y$ be a notion order beta base.
    Let $Y$ be a notion order alpha base.
    Let $Y$ be a notion order beta base.
    $X$ is a notion ordered object over $Y$ iff $X=X$.
\end{definition}
