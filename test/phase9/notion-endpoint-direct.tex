\import{library/function.tex}

\begin{definition}\label{notion_endpoint_domain_value}
    Let $x\in\dom{f}$.
    $\notionEndpointDomainValue{f}{x}=x$.
\end{definition}

\begin{definition}\label{notion_endpoint_result_value}
    Let $y\in B$.
    $\notionEndpointResultValue{y}{B}=y$.
\end{definition}

\begin{definition}\label{notion_endpoint_nested_value}
    Let $g$ be a function.
    Let $y\in\dom{g}$.
    $\notionEndpointNestedValue{g}{y}=y$.
\end{definition}

\begin{abbreviation}\label{notion_endpoint_tagged_function}
    $f$ is a notion tagged function from $A$ to $B$ over $T$ iff
        $f$ is a function from $A$ to $B$.
\end{abbreviation}

\begin{proposition}\label{notion_endpoint_domain_direct}
    Let $f$ be a function from $A$ to $B$.
    Let $x\in A$.
    Then $\notionEndpointDomainValue{f}{x}=x$.
\end{proposition}
\begin{proof}
    Follows.
\end{proof}

\begin{proposition}\label{notion_endpoint_nested_result}
    Let $f$ be a function from $A$ to $B$.
    Let $g$ be a function from $B$ to $C$.
    Let $x\in A$.
    Then $\notionEndpointNestedValue{g}{f(x)}=f(x)$.
\end{proposition}
\begin{proof}
    Follows.
\end{proof}

\begin{proposition}\label{notion_endpoint_bounded_fix}
    Let $f$ be a function from $A$ to $B$.
    Then for every $x\in A$ we have
        $\notionEndpointResultValue{f(x)}{B}=f(x)$.
\end{proposition}
\begin{proof}
    Fix $x\in A$.
    Follows.
\end{proof}

\begin{proposition}\label{notion_endpoint_bounded_take}
    Let $f$ be a function from $A$ to $B$.
    Suppose there exists $x\in A$ such that $x=x$.
    Then $A=A$.
\end{proposition}
\begin{proof}
    Take $x\in A$ such that $x=x$ by assumption.
    We have $\notionEndpointResultValue{f(x)}{B}=f(x)$.
    Follows.
\end{proof}

\begin{proposition}\label{notion_endpoint_alias}
    Let $f$ be a function from $A$ to $B$.
    Let $x\in A$.
    Then $x=x$.
\end{proposition}
\begin{proof}
    Let $g=f$.
    We have $\notionEndpointDomainValue{g}{x}=x$.
    We have $\notionEndpointResultValue{g(x)}{B}=g(x)$.
    Follows.
\end{proof}

\begin{proposition}\label{notion_endpoint_result_direct}
    Let $f$ be a function from $A$ to $B$.
    Let $x\in A$.
    Then $\notionEndpointResultValue{f(x)}{B}=f(x)$.
\end{proposition}
\begin{proof}
    Follows.
\end{proof}

\begin{proposition}\label{notion_endpoint_spelling_equivalence}
    Let $f\in\funs{A}{B}$.
    Then $f$ is a function from $A$ to $B$ and
        $f\in\funs{A}{B}$.
\end{proposition}
\begin{proof}
    $f$ is a function from $A$ to $B$ by notion.
    $f\in\funs{A}{B}$ by notion.
    Follows.
\end{proof}

\begin{proposition}\label{notion_endpoint_symbolic_direct}
    Let $f\in\funs{A}{B}$.
    Let $x\in A$.
    Then $\notionEndpointDomainValue{f}{x}=x$ and
        $\notionEndpointResultValue{f(x)}{B}=f(x)$.
\end{proposition}
\begin{proof}
    Follows.
\end{proof}

\begin{proposition}\label{notion_endpoint_symbolic_nested}

    Let $g\in\funs{B}{C}$.
    Let $x\in A$.
    Then $\notionEndpointNestedValue{g}{f(x)}=f(x)$.
\end{proposition}
\begin{proof}
    Follows.
\end{proof}

\begin{proposition}\label{notion_endpoint_symbolic_product_argument}
    Let $H\in\funs{A\times\pow{A\times B}}{B}$.
    Let $x\in A$ and $p\in\pow{A\times B}$.
    Then $\notionEndpointResultValue{H((x,p))}{B}=H((x,p))$.
\end{proposition}
\begin{proof}
    Follows.
\end{proof}

\begin{proposition}\label{notion_endpoint_terminal_alias_chain}
    Let $f$ be a notion tagged function from $A$ to $B$ over $T$.
    Let $x\in A$.
    Then $f\in\funs{A}{B}$ and
        $f$ is a function from $A$ to $B$ and
        $\notionEndpointResultValue{f(x)}{B}=f(x)$.
\end{proposition}
\begin{proof}
    $f\in\funs{A}{B}$ by notion.
    $f$ is a function from $A$ to $B$ by notion.
    Follows.
\end{proof}

\begin{proposition}\label{notion_endpoint_aliased_product_result}
    Let $H\in\funs{A\times\pow{A\times B}}{B}$.
    Let $p=\restrl{f}{D}$.
    Let $b=H((x,p))$.
    Suppose $(x,p)\in A\times\pow{A\times B}$.
    Then $b\in B$.
\end{proposition}
\begin{proof}
    $b\in B$ by notion.
    Follows.
\end{proof}
