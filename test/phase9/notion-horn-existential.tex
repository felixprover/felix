\begin{proposition}\label{horn_existential_later_disjunct}

    Follows by notion.

\begin{proposition}\label{horn_existential_exact_disjunction}

\end{proposition}
\begin{proof}
    Follows by notion.
\end{proof}

\begin{proposition}\label{horn_existential_derived_equality}
    Suppose $a\in A$.
    Suppose for every $x\in A$ we have $x=b$.
    Then there exists $x\in A$ such that $x=b$.
\end{proposition}
\begin{proof}
    Follows by notion.
\end{proof}

\begin{definition}\label{horn_existential_ready}
    $X$ is witness-ready iff $X=X$.
\end{definition}

\begin{classification}\label{horn_existential_scoped_rule}
    Suppose if $X\in\emptyset$ then $X\in\emptyset$.
    Then $X$ is witness-ready.
\end{classification}
\begin{proof}
    Follows by \auto{horn_existential_ready}.
\end{proof}

\begin{proposition}\label{horn_existential_scoped_producer}

\end{proposition}
\begin{proof}
    Follows by notion.
\end{proof}
