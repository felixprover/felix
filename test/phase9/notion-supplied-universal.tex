\begin{definition}\label{supplied_universal_result}
    $I$ is a supplied-universal result iff $I=I$.
\end{definition}

\begin{definition}\label{supplied_universal_admissible}
    Let $I$ be a supplied-universal result.
    $I$ is supplied-universal-admissible iff $I=I$.
\end{definition}

\begin{classification}\label{supplied_universal_classification}
    Suppose for every $i\in I$ we have $i$ is a quantified base.
    Then $I$ is a supplied-universal result.
\end{classification}
\begin{proof}
    Follows by \obv{supplied_universal_result}.
\end{proof}

\begin{theorem}\label{supplied_universal_identity}
    Suppose for every $j\in I$ we have $j$ is a quantified base.
    Then $I$ is a supplied-universal result.
\end{theorem}
\begin{proof}
    Follows by notion.
\end{proof}

\begin{theorem}\label{supplied_universal_projection}
    Suppose for every $j\in I$ we have $A$ is a quantified base and $j$ is a quantified base.
    Then $I$ is a supplied-universal result.
\end{theorem}
\begin{proof}
    Follows by notion.
\end{proof}

\begin{theorem}\label{supplied_universal_formation}
    Suppose for every $j\in I$ we have $A$ is a quantified base and $j$ is a quantified base.
    Suppose $I$ is supplied-universal-admissible.
    Then $I=I$.
\end{theorem}
\begin{proof}
    Follows.
\end{proof}
