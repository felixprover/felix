\begin{definition}\label{whole_template_primary}
    $X$ is a template primary iff $X=X$.
\end{definition}

\begin{definition}\label{whole_template_target}
    $X$ is template-ready iff $X=X$.
\end{definition}

\begin{classification}\label{whole_template_all_primary}
    Let $X$ be a set.
    Then $X$ is a template primary.
\end{classification}
\begin{proof}
    Follows by \obv{whole_template_primary}.
\end{proof}

\begin{classification}\label{whole_template_reflexive}
    Let $X$ be a set.
    Then $X=X$.
\end{classification}
\begin{proof}
    Follows.
\end{proof}

\begin{classification}\label{whole_template_consumer_rule}
    Suppose $X$ is a template primary.
    Suppose $X=(A,A)$.
    Then $A$ is template-ready.
\end{classification}
\begin{proof}
    Follows by \obv{whole_template_target}.
\end{proof}

\begin{proposition}\label{whole_template_source_consumer}
    Let $A$ be a set.
    Then $A$ is template-ready.
\end{proposition}
\begin{proof}
    Follows by notion.
\end{proof}
