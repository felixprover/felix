\import{library/function.tex}

\begin{definition}\label{notion_typed_image_consumer}
    Let $B$ be a subset of $Y$.
    $\notionTypedImageConsumer{B}{Y}=B$.
\end{definition}

\begin{abbreviation}\label{notion_typed_image_tagged_function}
    $f$ is a typed-image tagged function from $X$ to $Y$ over $T$ iff
        $f\in\funs{X}{Y}$.
\end{abbreviation}

\begin{definition}\label{notion_typed_image_provider_base}
    $Y$ is a typed-image provider base iff $Y=Y$.
\end{definition}

\begin{definition}\label{notion_typed_image_provider}
    $\notionTypedImageProvider{Y}=\emptyset$.
\end{definition}

\begin{classification}\label{notion_typed_image_provider_result}
    Let $Y$ be a typed-image provider base.
    Then $\notionTypedImageProvider{Y}\in\funs{\emptyset}{Y}$.
\end{classification}
\begin{proof}
    $\emptyset\in\funs{\emptyset}{Y}$ by
        \auto{emptyset_is_function_on_emptyset,
            codom_of_emptyset_can_be_anything,funs_intro}.
    Follows by \auto{notion_typed_image_provider}.
\end{proof}

\begin{proposition}\label{notion_typed_image_direct}
    Let $f\in\funs{X}{Y}$.
    Let $A$ be a subset of $X$.
    Then $\notionTypedImageConsumer{\img{f}{A}}{Y}=\img{f}{A}$.
\end{proposition}
\begin{proof}
    Follows by \auto{notion_typed_image_consumer}.
\end{proof}

\begin{proposition}\label{notion_typed_image_row_safe_unary}
    Let $f$ be a typed-image tagged function from $X$ to $Y$ over $T$.
    Let $A$ be a subset of $X$.
    Then $\notionTypedImageConsumer{\img{f}{A}}{Y}=\img{f}{A}$.
\end{proposition}
\begin{proof}
    Follows by \auto{notion_typed_image_consumer}.
\end{proof}

\begin{proposition}\label{notion_typed_image_provider_derived}
    Let $Y$ be a typed-image provider base.
    Let $A$ be a subset of $\emptyset$.
    Then $\notionTypedImageConsumer
        {\img{\notionTypedImageProvider{Y}}{A}}{Y}
        =\img{\notionTypedImageProvider{Y}}{A}$.
\end{proposition}
\begin{proof}
    Follows by \auto{notion_typed_image_consumer}.
\end{proof}

\begin{proposition}\label{notion_typed_image_alias_transport}
    Let $f\in\funs{X}{Y}$.
    Let $A$ be a subset of $X$.
    Then $f=f$.
\end{proposition}
\begin{proof}
    Let $g=f$.
    We have $\notionTypedImageConsumer{\img{g}{A}}{Y}=\img{g}{A}$
        by \auto{notion_typed_image_consumer}.
    Follows.
\end{proof}

\begin{proposition}\label{notion_typed_image_correlated_endpoints}
    Let $f\in\funs{U}{V}$.
    Suppose $f\in\funs{X}{Y}$.
    Let $A$ be a subset of $X$.
    Then $\notionTypedImageConsumer{\img{f}{A}}{Y}=\img{f}{A}$.
\end{proposition}
\begin{proof}
    Follows by \auto{notion_typed_image_consumer}.
\end{proof}
