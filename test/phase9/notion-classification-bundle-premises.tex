\begin{classification}\label{notion_bundle_anchored_results}
    Suppose $x$ is a finite-evidence row element of $X$.
    Let $X$ be a notion base.
    Then $x$ is finite-evidence selected and $x$ is finite-evidence directly-selected.
\end{classification}
\begin{proof}
    Follows by \auto{notion_finite_evidence_selected,notion_finite_evidence_directly_selected}.
\end{proof}

\begin{classification}\label{notion_bundle_unary_results}
    Suppose $X$ is a notion base.
    Then $X$ is notion guarded and $X$ is notion admissible.
\end{classification}
\begin{proof}
    Follows by \auto{notion_guarded,notion_guarded_alias}.
\end{proof}
