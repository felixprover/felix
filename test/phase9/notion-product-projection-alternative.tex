\import{library/set.tex}

\begin{definition}\label{notion_alternative_product}
    $\notionAlternativeProduct{A}{B}=A\times B$.
\end{definition}

\begin{classification}\label{notion_alternative_product_fst_type}
    Suppose $p\in\notionAlternativeProduct{A}{B}$.
    Then $\fst{p}\in A$.
\end{classification}
\begin{proof}
    Follows by \auto{notion_alternative_product,fst_type}.
\end{proof}

\begin{definition}\label{notion_alternative_product_consumer}
    Let $x\in A$.
    $\notionAlternativeProductConsumer{x}{A}=x$.
\end{definition}

\begin{proposition}\label{notion_alternative_product_client}
    Suppose $p\in\notionAlternativeProduct{A}{B}$.
    Then $\notionAlternativeProductConsumer{\fst{p}}{A}=\fst{p}$.
\end{proposition}
\begin{proof}
    Follows.
\end{proof}
