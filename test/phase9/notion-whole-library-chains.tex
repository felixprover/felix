\import{library/topology/basis.tex}
\import{library/set/filter-basis.tex}
\import{library/set/choice.tex}
\import{library/topology/indexed-coproduct.tex}

\begin{definition}\label{whole_topology_consumer}
    Let $T$ be a topology on $S$.
    $T$ is whole topology accepted on $S$ iff $T=T$.
\end{definition}

\begin{definition}\label{whole_membership_consumer}
    Let $x\in A$.
    $x$ is whole membership accepted in $A$ iff $x=x$.
\end{definition}

\begin{definition}\label{whole_space_consumer}
    Let $Y$ be a topological space.
    $Y$ is whole space accepted iff $Y=Y$.
\end{definition}

\begin{proposition}\label{whole_basis_generated_opens}
    Suppose $B$ is a topological basis for $S$.
    Then $\generatedOpens{S}{B}$ is whole topology accepted on $S$.
\end{proposition}
\begin{proof}
    $\generatedOpens{S}{B}$ is a topology on $S$ by notion.
    Follows by \obv{whole_topology_consumer}.
\end{proof}

\begin{proposition}\label{whole_filter_basis_choice}
    Suppose $B$ is a filter basis on $S$.
    Then $\choice{B}$ is whole membership accepted in $B$.
\end{proposition}
\begin{proof}
    $B$ is inhabited by notion.
    $\choice{B}\in B$ by notion.
    Follows by \obv{whole_membership_consumer}.
\end{proof}

\begin{proposition}\label{whole_function_on_coproduct}
    Suppose $X$ is a function on $I$.
    Suppose for every $i\in I$ we have $X(i)$ is a topological space.
    Then $\IndexedCoproductTopology{I}{X}$ is whole space accepted.
\end{proposition}
\begin{proof}
    $\IndexedCoproductTopology{I}{X}$ is a topological space by notion.
    Follows by \obv{whole_space_consumer}.
\end{proof}
