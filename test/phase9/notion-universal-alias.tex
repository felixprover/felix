\begin{theorem}\label{universal_alias_source}
    Suppose $k\in I$.
    Then $k=k$.
\end{theorem}
\begin{proof}
    Let $J=I$.
    Show that if for every $j\in J$ we have $j$ is a quantified base,
        then $k$ is a quantified base.
    \begin{subproof}
        Assume for every $j\in J$ we have $j$ is a quantified base.
        $k$ is a quantified base by notion.
        Follows.
    \end{subproof}
    Show that if for every $j\in J$ we have $j$ is a quantified base,
        then $I$ is a supplied-universal result.
    \begin{subproof}
        Assume for every $j\in J$ we have $j$ is a quantified base.
        $I$ is a supplied-universal result by notion.
        Follows.
    \end{subproof}
    Follows.
\end{proof}

\begin{theorem}\label{universal_alias_dependent_source}
    Suppose $k\in I$ and $l\in k$.
    Then $k=k$.
\end{theorem}
\begin{proof}
    Let $J=I$.
    Show that if for every $i\in J$ we have for every $j\in i$
        we have $(i,j)$ is a quantified base, then $(k,l)$ is a quantified base.
    \begin{subproof}
        Assume for every $i\in J$ we have for every $j\in i$
            we have $(i,j)$ is a quantified base.
        $(k,l)$ is a quantified base by notion.
        Follows.
    \end{subproof}
    Follows.
\end{proof}
