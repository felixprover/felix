\begin{proposition}\label{readability_interval_map_assertion_equivalence}
    Let $X$ be a topological space.
    (there exists $f\in\funs{\carrier[X]}{\RealUnitInterval}$ such that
        $f$ is continuous from $X$ to $\RealUnitIntervalTopology$ and
        (for every $x\in A$ we have $f(x)=c$) and
        (for every $x\in B$ we have $f(x)=d$))
    iff
    (there exists $f\in\funs{\carrier[X]}{\RealUnitInterval}$ such that
        $f$ is continuous from $X$ to $\RealUnitIntervalTopology$ and
        $f$ is constantly $c$ on $A$ and $f$ is constantly $d$ on $B$).
\end{proposition}
\begin{proof}
    Follows.
\end{proof}

\begin{proposition}\label{readability_urysohn_statement_equivalence}
    (if $X$ is a topological space, then if $X$ is normal, then
        for all $A,B\in\closeds{X}$ such that $A$ is disjoint from $B$
        there exists $f\in\funs{\carrier[X]}{\RealUnitInterval}$ such that
            $f$ is continuous from $X$ to $\RealUnitIntervalTopology$ and
            (for every $x\in A$ we have $f(x)=\surrealZero$) and
            (for every $x\in B$ we have $f(x)=\surrealOne$))
    iff
    (if $X$ is a normal space, then
        for all closed subsets $A,B$ of $X$ such that $A$ is disjoint from $B$
        there exists a continuous function $f$ from $X$ to
            $\RealUnitIntervalTopology$ such that
            $f$ is constantly $\surrealZero$ on $A$ and
            constantly $\surrealOne$ on $B$).
\end{proposition}
\begin{proof}
    Follows by \auto{normal_space_is_topological,closed_subset_of,is_closed_in,
        continuous_map,real_unit_interval_topology_carrier}.
\end{proof}
