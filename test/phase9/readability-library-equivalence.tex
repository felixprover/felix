\begin{proposition}\label{readability_silver_assumptions}
    Let $\kappa$ be a singular cardinal.
    ($\naturals\in\cofinality{\kappa}$ and
        for every $\mu\in\kappa$ such that $\mu$ is an infinite cardinal
        we have $\cardPower{\cardTwo}{\mu}=\cardSucc{\mu}$)
    iff
    ($\kappa$ is of uncountable cofinality and
        for every infinite cardinal $\mu\in\kappa$ we have
        $\cardPower{\cardTwo}{\mu}=\cardSucc{\mu}$).
\end{proposition}
\begin{proof}
    Follows by \auto.
\end{proof}

\begin{proposition}\label{readability_cardinal_bound}
    (for every $\mu\in\kappa$ such that $\mu$ is a cardinal we have
        $\cardPower{\mu}{\nu}\in\kappa$)
    iff
    (for every cardinal $\mu\in\kappa$ we have
        $\cardPower{\mu}{\nu}\in\kappa$).
\end{proposition}
\begin{proof}
    Follows by \auto.
\end{proof}

\begin{proposition}\label{readability_limit_ordinal_bound}
    (for every $\gamma\in\alpha$ such that $\gamma$ is a limit ordinal
        and $\ordinalUnbounded{C\inter\gamma}{\gamma}$ we have $\gamma\in C$)
    iff
    (for every limit ordinal $\gamma\in\alpha$ such that
        $\ordinalUnbounded{C\inter\gamma}{\gamma}$ we have $\gamma\in C$).
\end{proposition}
\begin{proof}
    Follows by \auto.
\end{proof}
