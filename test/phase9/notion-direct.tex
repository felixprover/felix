\begin{definition}\label{notion_base}
    $X$ is a notion base iff $X=X$.
\end{definition}

\begin{definition}\label{notion_guarded}
    Let $X$ be a notion base.
    $X$ is notion guarded iff $X=X$.
\end{definition}

\begin{definition}\label{notion_guarded_alias}
    Let $X$ be a notion base.
    $X$ is notion admissible iff $X$ is notion guarded.
\end{definition}
