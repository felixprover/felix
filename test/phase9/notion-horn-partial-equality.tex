\begin{definition}\label{horn_partial_seed}
    $X$ is a partial seed iff $X=X$.
\end{definition}

\begin{definition}\label{horn_partial_middle}
    $X$ is a partial middle iff $X=\emptyset$.
\end{definition}

\begin{definition}\label{horn_partial_target}
    $A$ is a partial target iff there exists $x$ such that $(x,x)$ is a partial middle.
\end{definition}

\begin{classification}\label{horn_partial_consumer}
    Suppose $(x,x)$ is a partial middle.
    Then $A$ is a partial target.
\end{classification}
\begin{proof}
    Follows by \auto{horn_partial_target}.
\end{proof}

\begin{proposition}\label{horn_partial_equality}
    Let $a$ be a partial seed.
    Suppose for every $z$ if $z$ is a partial seed then $B$ is a partial middle.
    Suppose $B=(a,a)$.
    Then $A$ is a partial target.
\end{proposition}
\begin{proof}
    Follows by notion.
\end{proof}

\begin{definition}\label{shared_parameter_diagonal}
    $X$ is a shared diagonal over $A$ iff $X=(A,A)$.
\end{definition}

\begin{classification}\label{shared_parameter_rule}
    Let $A$ be a partial seed.
    Then $(A,A)$ is a shared diagonal over $A$.
\end{classification}
\begin{proof}
    Follows by \obv{shared_parameter_diagonal}.
\end{proof}

\begin{proposition}\label{shared_parameter_transport}
    Let $A$ be a partial seed.
    Suppose $A=B$.
    Then $(A,A)$ is a shared diagonal over $B$.
\end{proposition}
\begin{proof}
    Follows by notion.
\end{proof}

\begin{definition}\label{matching_premise_copy}
    $\matchingCopy{X}{Y}=X$.
\end{definition}

\begin{definition}\label{matching_premise_target}
    $X$ is matching-ready iff $X$ is a partial seed and
        there exist sets $Y,z$ such that $z\in Y$.
\end{definition}

\begin{classification}\label{matching_premise_rule}
    Suppose $\matchingCopy{X}{Y}$ is a partial seed.
    Suppose there exists $z\in Y$.
    Then $X$ is matching-ready.
\end{classification}
\begin{proof}
    Follows by \auto{matching_premise_target,matching_premise_copy}.
\end{proof}

\begin{proposition}\label{matching_premise_consumer}
    Suppose $\matchingCopy{A}{B}$ is a partial seed.
    Suppose $\matchingCopy{A}{B}=\matchingCopy{A}{C}$.
    Suppose there exists $z\in C$.
    Then $A$ is matching-ready.
\end{proposition}
\begin{proof}
    Follows by notion.
\end{proof}
