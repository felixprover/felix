\begin{definition}\label{notion_term_value}
    Let $X$ be a notion base.
    $\notionTermValue{X}=X$.
\end{definition}

\begin{definition}\label{notion_term_wrapper}
    Let $X$ be a notion base.
    The notion term wrapper of $X$ is $\notionTermValue{X}$.
\end{definition}

\begin{definition}\label{notion_membership_value}
    Let $X\in A$.
    $\notionMembershipValue{X}{A}=X$.
\end{definition}

\begin{definition}\label{notion_membership_wrapper}
    Let $X\in A$.
    $\notionMembershipWrapper{X}{A}=\notionMembershipValue{X}{A}$.
\end{definition}

\begin{definition}\label{notion_binary_term}
    Let $X$ be a notion base.
    Let $Y$ be a notion base.
    $\notionBinaryTerm{X}{Y}=X$.
\end{definition}


\begin{abbreviation}\label{notion_discard_argument}
    $\notionDiscard{X}{Y}=X$.
\end{abbreviation}
