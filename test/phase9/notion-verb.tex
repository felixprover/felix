\begin{definition}\label{notion_verb_base}
    $X$ is a notion verb base iff $X=X$.
\end{definition}

\begin{definition}\label{notion_verb_seal}
    $X$ has a notion verb seal over $Y$ iff $X=X$.
\end{definition}

\begin{definition}\label{notion_verb_accepted}
    $X$ is notion verb accepted over $Y$ iff $X=X$.
\end{definition}

\begin{definition}\label{notion_verb_guarded_seal}
    Let $X$ be a notion verb base.
    $X$ carries a notion verb guarded seal over $Y$ iff $X=X$.
\end{definition}
