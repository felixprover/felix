\import{library/set/finite.tex}
\import{library/set/choice.tex}

\begin{definition}\label{notion_intrinsic_finite_consumer}
    Let $X$ be a finite set.
    $\notionIntrinsicFiniteConsumer{X}=X$.
\end{definition}

\begin{definition}\label{notion_intrinsic_membership_consumer}
    Let $x\in A$.
    $\notionIntrinsicMembershipConsumer{x}{A}=x$.
\end{definition}

\begin{proposition}\label{notion_intrinsic_powerset_result}
    Suppose $X$ is finite.
    Then $\notionIntrinsicFiniteConsumer{\pow{X}}=\pow{X}$.
\end{proposition}
\begin{proof}
    Follows.
\end{proof}

\begin{proposition}\label{notion_intrinsic_projection_result}
    Suppose $p\in A\times B$.
    Then $\notionIntrinsicMembershipConsumer{\fst{p}}{A}=\fst{p}$ and
    $\notionIntrinsicMembershipConsumer{\fst{p}}{A}=\fst{p}$ and
    $\notionIntrinsicMembershipConsumer{\snd{p}}{B}=\snd{p}$ and
    $\notionIntrinsicMembershipConsumer{\snd{p}}{B}=\snd{p}$.
\end{proposition}
\begin{proof}
    Follows.
\end{proof}

\begin{proposition}\label{notion_intrinsic_projection_guard}
    If $p\in A\times B$, then
    $\notionIntrinsicMembershipConsumer{\fst{p}}{A}=\fst{p}$.
\end{proposition}
\begin{proof}
    Follows.
\end{proof}

\begin{proposition}\label{notion_intrinsic_projection_multiple_rows}
    Suppose $p\in A\times B$ and $p\in A\times C$.
    Then $\notionIntrinsicMembershipConsumer{\fst{p}}{A}=\fst{p}$.
\end{proposition}
\begin{proof}
    Follows.
\end{proof}

\begin{proposition}\label{notion_intrinsic_first_projection_core_coordinates}
    Suppose $p\in A\times B$.
    Then $p=p$.
\end{proposition}
\begin{proof}
    Let $q=p$.
    We have $\notionIntrinsicMembershipConsumer{\fst{q}}{A}=\fst{q}$
        by \auto{notion_intrinsic_membership_consumer}.
    Follows.
\end{proof}

\begin{proposition}\label{notion_intrinsic_second_projection_core_coordinates}
    Suppose $p\in A\times B$.
    Then $p=p$.
\end{proposition}
\begin{proof}
    Let $q=p$.
    We have $\notionIntrinsicMembershipConsumer{\snd{q}}{B}=\snd{q}$
        by \auto{notion_intrinsic_membership_consumer}.
    Follows.
\end{proof}

\begin{proposition}\label{notion_intrinsic_choice_result}
    Let $A$ be an inhabited set.
    Then $\notionIntrinsicMembershipConsumer{\choice{A}}{A}=\choice{A}$.
\end{proposition}
\begin{proof}
    Follows.
\end{proof}

\begin{proposition}\label{notion_intrinsic_empty_result}
    $\notionIntrinsicFiniteConsumer{\emptyset}=\emptyset$.
\end{proposition}
\begin{proof}
    Follows.
\end{proof}
