\begin{struct}\label{notion_bundle_carrier}
    A notion bundle carrier $X$ is a onesorted structure.
\end{struct}

\begin{struct}\label{notion_bundle_reflexive_carrier}
    A notion bundle reflexive carrier $X$ is a notion bundle carrier such that
    \begin{enumerate}
        \item\label{notion_bundle_reflexive_law} $X=X$.
    \end{enumerate}
\end{struct}

\begin{definition}\label{notion_bundle_value}
    $\notionBundleValue=\structure[notion bundle carrier]{\emptyset}$.
\end{definition}

\begin{classification}\label{notion_bundle_exact}
    $\notionBundleValue$ is an exact notion bundle carrier.
\end{classification}
\begin{proof}
    Follows by \auto{notion_bundle_value,notion_bundle_carrier,notion_bundle_carrieraggregateexact}.
\end{proof}

\begin{proposition}\label{notion_bundle_open_replay}
    $\notionBundleValue$ is a notion bundle carrier.
\end{proposition}
\begin{proof}
    Follows by notion.
\end{proof}

\begin{proposition}\label{notion_bundle_exact_replay}
    $\notionBundleValue$ is an exact notion bundle carrier.
\end{proposition}
\begin{proof}
    Follows by \obv{notion_bundle_exact}.
\end{proof}

\begin{classification}\label{notion_bundle_nested}
    $\notionBundleValue$ is an exact notion bundle carrier and
    $\notionBundleValue$ is a notion bundle carrier.
\end{classification}
\begin{proof}
    Follows by \obv{notion_bundle_exact}.
\end{proof}

\begin{classification}\label{notion_bundle_mixed}
    $\notionBundleValue$ is a notion bundle carrier and
    $\emptyset$ is a notion base.
\end{classification}
\begin{proof}
    Follows by \auto{notion_bundle_exact,notion_base}.
\end{proof}
