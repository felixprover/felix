\begin{struct}\label{supplied_empty_record}
    A supplied empty record $R$ is a record.
\end{struct}

\begin{construction}\label{supplied_guarded_construction}
    Let $I$ be a set.
    Suppose for every $i\in I$ we have $i$ is a quantified base.
    Suppose $I=I$.
    The supplied empty record $\suppliedGuardedConstruction{I}=\structure[supplied empty record]{}$.
\end{construction}
\begin{proof}
    Follows.
\end{proof}

\begin{proposition}\label{supplied_construction_consumer}
    Suppose for every $j\in I$ we have $A$ is a quantified base and $j$ is a quantified base.
    Suppose $I=I$.
    Then $\suppliedGuardedConstruction{I}$ is a supplied empty record.
\end{proposition}
\begin{proof}
    Follows by notion.
\end{proof}

\begin{abbreviation}\label{contract_guarded}
    $I$ is contract guarded iff
        (for every $i\in I$ we have $i$ is a quantified base) and $I=I$.
\end{abbreviation}

\begin{definition}\label{contract_guarded_value}
    Let $I$ be a set that is contract guarded.
    $\contractGuardedValue{I}=I$.
\end{definition}

\begin{proposition}\label{guarded_contract_consumer}
    Suppose for every $j\in I$ we have $A$ is a quantified base and $j$ is a quantified base.
    Suppose $I=I$.
    Then $\contractGuardedValue{I}=I$.
\end{proposition}
\begin{proof}
    Follows by \obv{contract_guarded_value}.
\end{proof}

\begin{proposition}\label{supplied_root_goal}
    Suppose for every $j$ we have if $j\in J$ and $j\in I$ then $A$ is a quantified base and $j$ is a quantified base.
    Then for every $k\in I$ such that $k\in J$ we have $k$ is a quantified base.
\end{proposition}
\begin{proof}
    Follows by notion.
\end{proof}

\begin{abbreviation}\label{contract_populated}
    $A$ is contract populated by $B$ iff there exists $x$ such that $x\in A$ and $x\in B$.
\end{abbreviation}

\begin{definition}\label{contract_populated_value}
    Let $A$ be a set that is contract populated by $B$.
    $\contractPopulatedValue{A}{B}=A$.
\end{definition}

\begin{proposition}\label{contract_populated_consumer}
    Suppose there exists $y$ such that $y\in X$ and $y\in Y$.
    Then $\contractPopulatedValue{X}{Y}=X$.
\end{proposition}
\begin{proof}
    Follows by \obv{contract_populated_value}.
\end{proof}
