\begin{definition}\label{horn_connective_primary}
    $X$ is a connective primary iff $X=\emptyset$.
\end{definition}

\begin{definition}\label{horn_connective_secondary}
    $X$ is a connective secondary iff $X=\{\emptyset\}$.
\end{definition}

\begin{proposition}\label{horn_connective_left}
    Let $X$ be a connective primary.
    Then $X$ is a connective primary or $X$ is a connective secondary.
\end{proposition}
\begin{proof}
    Follows by notion.
\end{proof}

\begin{proposition}\label{horn_connective_equality_chain}
    Suppose $A=B$ and $C=B$.
    Then $A=C$.
\end{proposition}
\begin{proof}
    Follows by notion.
\end{proof}

\begin{proposition}\label{horn_connective_equality_reflexive}
    Let $A$ be a set.
    Then $A=A$.
\end{proposition}
\begin{proof}
    Follows by notion.
\end{proof}

\begin{proposition}\label{horn_connective_right}
    Let $X$ be a connective secondary.
    Then $X$ is a connective primary or $X$ is a connective secondary.
\end{proposition}
\begin{proof}
    Follows by notion.
\end{proof}

\begin{proposition}\label{horn_connective_iff_introduction}
    Suppose if $X$ is a connective primary then $X$ is a connective secondary.
    Suppose if $X$ is a connective secondary then $X$ is a connective primary.
    Then $X$ is a connective primary iff $X$ is a connective secondary.
\end{proposition}
\begin{proof}
    Follows by notion.
\end{proof}

\begin{proposition}\label{horn_connective_iff_forward}
    Suppose $X$ is a connective primary iff $X$ is a connective secondary.
    Let $X$ be a connective primary.
    Then $X$ is a connective secondary.
\end{proposition}
\begin{proof}
    Follows by notion.
\end{proof}

\begin{proposition}\label{horn_connective_iff_backward}
    Suppose $X$ is a connective primary iff $X$ is a connective secondary.
    Let $X$ be a connective secondary.
    Then $X$ is a connective primary.
\end{proposition}
\begin{proof}
    Follows by notion.
\end{proof}

\begin{proposition}\label{horn_connective_iff_scoped}
    Let $X$ be a set.
    Then $X$ is a connective primary iff $X$ is a connective primary.
\end{proposition}
\begin{proof}
    Follows by notion.
\end{proof}
