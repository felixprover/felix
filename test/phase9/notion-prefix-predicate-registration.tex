\import{order/boolean-algebra/subalgebra.tex}

\begin{classification}\label{notion_boolean_subalgebra_structure_result}
    Suppose $\booleanSubalgebra{A}{B}$.
    Then $\booleanSubalgebraStructure{B}{A}$ is a Boolean algebra.
\end{classification}
\begin{proof}
    Follows by
        \obv{boolean_subalgebra_structure_is_boolean_algebra}.
\end{proof}

\begin{definition}\label{notion_boolean_subalgebra_witness}
    $A$ is a notion Boolean subalgebra witness iff $A=A$.
\end{definition}

\begin{classification}\label{notion_boolean_subalgebra_witness_view}
    Suppose $\booleanSubalgebra{A}{B}$.
    Then $A$ is a notion Boolean subalgebra witness.
\end{classification}
\begin{proof}
    Follows.
\end{proof}

\begin{abbreviation}\label{notion_boolean_subalgebra_alias}
    $A$ is a notion Boolean subalgebra of $B$ iff
        $\booleanSubalgebra{A}{B}$.
\end{abbreviation}
