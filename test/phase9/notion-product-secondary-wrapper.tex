% The transparent wrapper expands to the original endpoint.
\begin{abbreviation}\label{notion_product_secondary_wrapper}
    $\notionProductSecondaryWrapper{A}=A$.
\end{abbreviation}

\begin{definition}\label{notion_product_secondary_wrapper_consumer}
    Let $x\in A$.
    $\notionProductSecondaryWrapperConsumer{x}{A}=x$.
\end{definition}

\begin{proposition}\label{notion_product_secondary_wrapper_preserves_membership}
    Suppose $p\in\notionProductSecondaryWrapper{A}\times B$.
    Then $\notionProductSecondaryWrapperConsumer{\fst{p}}{A}=\fst{p}$.
\end{proposition}

    Follows.
\end{proof}
