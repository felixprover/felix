\begin{definition}\label{horn_initial_probe}
    Let $X$ be a topological space.
    Let $g\in\funs{S}{\carrier[X]}$.
    $\hornInitialProbe{X}{S}{g}=g$.
\end{definition}

\begin{proposition}\label{horn_initial_pointwise}
    Suppose $f$ is an initial mapping family from $S$ to $Y$ indexed by $I$.
    Let $i\in I$.
    Then $Y(i)$ is a topological space and
        $f(i)\in\funs{S}{\carrier[Y(i)]}$.
\end{proposition}
\begin{proof}
    Follows by notion.
\end{proof}

\begin{proposition}\label{horn_initial_source_call}
    Suppose $f$ is an initial mapping family from $S$ to $Y$ indexed by $I$.
    Let $i\in I$.
    Then $\hornInitialProbe{Y(i)}{S}{f(i)}=f(i)$.
\end{proposition}
\begin{proof}
    Follows by \obv{horn_initial_probe}.
\end{proof}

\begin{proposition}\label{horn_proper_basis_member}
    Suppose $B$ is a proper filter basis on $S$.
    Let $A\in B$.
    Then $A\subseteq S$ and $A$ is inhabited.
\end{proposition}
\begin{proof}
    Follows by notion.
\end{proof}

\begin{proposition}\label{horn_proper_basis_choice}
    Suppose $B$ is a proper filter basis on $S$.
    Let $A\in B$.
    Then $\choice{A}\in A$.
\end{proposition}
\begin{proof}
    Follows by notion.
\end{proof}

\begin{proposition}\label{horn_local_inhabited_member}
    Suppose every element of $B$ is inhabited.
    Let $A\in B$.
    Then $A$ is inhabited.
\end{proposition}
\begin{proof}
    Follows by notion.
\end{proof}

\begin{proposition}\label{horn_local_whole_implication}
    Let $A,B,C$ be sets.
    Suppose if $A\in B$ then $C$ is inhabited.
    Let $D$ be a set.
    Suppose $A\in B$.
    Then $C$ is inhabited.
\end{proposition}
\begin{proof}
    Follows by notion.
\end{proof}
