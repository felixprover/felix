\begin{definition}\label{whole_family}
    $X$ is a whole family for $A$ iff
    for every $x\in X$ we have $x\in A$.
\end{definition}

\begin{definition}\label{whole_marked}
    $x$ is whole marked iff $x=x$.
\end{definition}

\begin{definition}\label{whole_equal}
    $x$ is whole equal to $y$ iff $x=y$.
\end{definition}

\begin{abbreviation}\label{whole_populated}
    $A$ is whole populated iff there exists $x$ such that $x\in A$.
\end{abbreviation}

\begin{proposition}\label{whole_universal_becomes_supplied}
    Suppose $X$ is a whole family for $A$.
    Let $x\in X$.
    Then $x\in A$.
\end{proposition}
\begin{proof}
    For every $y\in X$ we have $y\in A$ by notion.
    Follows by notion.
\end{proof}

\begin{proposition}\label{whole_equality_becomes_supplied}
    Suppose $x$ is whole marked and $x$ is whole equal to $y$.
    Then $y$ is whole marked.
\end{proposition}
\begin{proof}
    $x=y$ by notion.
    Follows by notion.
\end{proof}

\begin{proposition}\label{whole_existential_exact}
    Suppose $A$ is whole populated.
    Then there exists $x$ such that $x\in A$.
\end{proposition}
\begin{proof}
    Follows by notion.
\end{proof}
