\begin{proposition}\label{readability_constant_expansion}
    $f$ is constantly $c$ on $A$ iff for every $x\in A$ we have $f(x)=c$.
\end{proposition}

    Follows.
\end{proof}

\begin{proposition}\label{readability_constant_empty}
    $f$ is constantly $c$ on $\emptyset$.
\end{proposition}
\begin{proof}
    Follows by \auto{emptyset}.
\end{proof}

\begin{proposition}\label{readability_cofinality_expansion}
    $\kappa$ is of uncountable cofinality iff $\naturals\in\cofinality{\kappa}$.
\end{proposition}
\begin{proof}
    Follows.
\end{proof}

\begin{proposition}\label{readability_closed_expansion}
    Let $X$ be a topological space.
    $A$ is a closed subset of $X$ iff $A\in\closeds{X}$.
\end{proposition}
\begin{proof}
    Follows by \auto{closed_subset_of,is_closed_in}.
\end{proof}

\begin{proposition}\label{readability_continuous_expansion}
    $f$ is a continuous function from $X$ to $Y$ iff
        $f$ is continuous from $X$ to $Y$.
\end{proposition}
\begin{proof}
    Follows.
\end{proof}

\begin{proposition}\label{readability_closed_head}
    Let $X$ be a normal space.
    Let $A,B$ be closed subsets of $X$ such that $A$ is disjoint from $B$.
    Then $A$ is closed in $X$ and $B$ is closed in $X$.
\end{proposition}
\begin{proof}
    Follows.
\end{proof}

\begin{proposition}\label{readability_continuous_noun}
    Let $X,Y$ be topological spaces.
    Let $f$ be a continuous function from $X$ to $Y$ such that
        $f$ is constantly $c$ on $A$ and constantly $d$ on $B$.
    Then $f$ is continuous from $X$ to $Y$.
\end{proposition}
\begin{proof}
    Follows.
\end{proof}
