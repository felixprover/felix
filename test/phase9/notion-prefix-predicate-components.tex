\begin{proposition}\label{notion_prefix_predicate_components}
    Suppose $B$ is a Boolean algebra.
    Suppose $A\subseteq\carrier[B]$.
    Suppose $\latticeBottom[B],\latticeTop[B]\in A$.
    Suppose for every $x\in A$ we have
        $\booleanComplement[B](x)\in A$.
    Suppose for all $x,y\in A$ we have $\meet[B](x,y)\in A$.
    Suppose for all $x,y\in A$ we have $\join[B](x,y)\in A$.
    Then $\booleanSubalgebraStructure{B}{A}$ is a Boolean algebra.
\end{proposition}
\begin{proof}
    Follows by notion.
\end{proof}
