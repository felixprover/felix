\begin{theorem}\label{quantified_member_proof}
    Suppose for every $i\in I$ we have $i$ is a quantified base.
    Suppose $k\in I$.
    Then $k$ is a quantified base.
\end{theorem}
\begin{proof}
    Follows by notion.
\end{proof}

\begin{theorem}\label{quantified_dependent_members}
    Suppose for every $i\in I$ we have for every $j\in i$ we have $(i,j)$ is a quantified base.
    Suppose $k\in I$ and $l\in k$.
    Then $(k,l)$ is a quantified base.
\end{theorem}
\begin{proof}
    Follows by notion.
\end{proof}

\begin{theorem}\label{quantified_result_projection}
    Suppose for every $i\in I$ we have $A$ is a quantified base and $i$ is a quantified base.

    Then $k$ is a quantified base.
\end{theorem}
\begin{proof}
    Follows by notion.
\end{proof}

\begin{theorem}\label{quantified_repeated_guards}
    Suppose for every $i\in I$ such that $i\in J$ we have $i$ is a quantified base.
    Suppose $k\in I$ and $k\in J$.
    Then $k$ is a quantified base.
\end{theorem}
\begin{proof}
    Follows by notion.
\end{proof}

\begin{theorem}\label{quantified_noun_term}
