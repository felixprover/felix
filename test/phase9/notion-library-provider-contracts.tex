\import{algebra/group/normal-closure.tex}
\import{algebra/group/cyclic.tex}
\import{topology/topological-space.tex}

\begin{definition}\label{notion_library_guarded_closure}
    Let $X$ be a topological space.
    $\notionLibraryGuardedClosure{A}{X}=\closure{A}{X}$.
\end{definition}

\begin{definition}\label{notion_library_guarded_generated_subgroup}
    Let $G$ be a group.
    Let $A$ be a subset of $\carrier[G]$.
    $\notionLibraryGuardedGeneratedSubgroup{A}{G}
        =\generatedSubgroup{A}{G}$.
\end{definition}

\begin{definition}\label{notion_library_guarded_normal_closure}
    Let $G$ be a group.
    Let $A$ be a subset of $\carrier[G]$.
    $\notionLibraryGuardedNormalClosure{A}{G}=\normalClosure{A}{G}$.
\end{definition}

\begin{definition}\label{notion_library_guarded_cyclic_subgroup}
    Let $G$ be a group.
    Let $a$ be an element of $\carrier[G]$.
    $\notionLibraryGuardedCyclicSubgroup{G}{a}=\cyclicSubgroup{G}{a}$.
\end{definition}
