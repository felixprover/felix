\begin{abbreviation}\label{horn_indexed_requirements}
    $I$ is horn-compatible with $f$ from $S$ to $Y$ iff
        for every $i\in I$ we have $Y(i)$ is a topological space and
            $f(i)\in\funs{S}{\carrier[Y(i)]}$.
\end{abbreviation}

\begin{definition}\label{horn_indexed_contract}
    Let $I$ be a set that is horn-compatible with $f$ from $S$ to $Y$.
    $\hornIndexedContract{I}{f}{S}{Y}=I$.
\end{definition}

\begin{proposition}\label{horn_indexed_contract_consumer}
    Suppose for every $i\in I$ we have $Y(i)$ is a topological space.
    Suppose for every $j\in I$ we have $f(j)\in\funs{S}{\carrier[Y(j)]}$.
    Then $\hornIndexedContract{I}{f}{S}{Y}=I$.
\end{proposition}
\begin{proof}
    Follows by \obv{horn_indexed_contract}.
\end{proof}

\begin{proposition}\label{horn_codomain_weakening}
    Let $f$ be a function to $B$.
    Suppose $B\subseteq C$.
    Then $f$ is a function to $C$.
\end{proposition}
\begin{proof}
    Follows by notion.
\end{proof}

\begin{classification}\label{horn_composition_type}
    Let $f\in\funs{A}{B}$.
    Let $g\in\funs{B}{C}$.
    Then $g\circ f\in\funs{A}{C}$.
\end{classification}
\begin{proof}
    Follows by \obv{funs_circ}.
\end{proof}

\begin{proposition}\label{horn_pointwise_composition}
    Suppose for every $i\in I$ we have $f(i)\in\funs{A}{B}$.
    Suppose for every $j\in I$ we have $g(j)\in\funs{B}{C}$.
    Then for every $k\in I$ we have $g(k)\circ f(k)\in\funs{A}{C}$.
\end{proposition}
\begin{proof}
    Follows by notion.
\end{proof}
