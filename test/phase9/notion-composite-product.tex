\begin{definition}\label{notion_product}
    $\notionProduct{A}{B}=\{(a,b)\mid a\in A, b\in B\}$.
\end{definition}

\begin{classification}\label{notion_product_intro}
    Suppose $x\in X$ and $y\in Y$.
    Then $(x,y)\in\notionProduct{X}{Y}$.
\end{classification}
\begin{proof}
    Follows by \auto{notion_product}.
\end{proof}

\begin{definition}\label{notion_product_element}
    $z$ is a notion product element of $P$ iff $z\in P$.
\end{definition}

\begin{definition}\label{notion_product_output}
    $z$ is a notion product output of $P$ iff $z=z$.
\end{definition}

\begin{classification}\label{notion_product_element_intro}
    Suppose $x\in X$ and $y\in Y$.
    Then $(x,y)$ is a notion product element of
        $\notionProduct{X}{Y}$.
\end{classification}
\begin{proof}
    Follows by \auto{notion_product, notion_product_element}.
\end{proof}

\begin{classification}\label{notion_product_element_output}
    Let $z$ be a notion product element of $P$.
    Then $z$ is a notion product output of $P$.
\end{classification}
\begin{proof}
    Follows by \auto{notion_product_output}.
\end{proof}

\begin{definition}\label{notion_product_output_witness}
    Let $P$ be a set.
    Let $z$ be a notion product output of $P$.
    $\notionProductOutputWitness{z}{P}=z$.
\end{definition}

\begin{definition}\label{notion_product_witness}
    Let $X,Y$ be sets.
    Let $z\in\notionProduct{X}{Y}$.
    $\notionProductWitness{z}{X}{Y}=z$.
\end{definition}

\begin{proposition}\label{notion_composite_product_client}
    Suppose $x\in X$ and $y\in Y$.
    Then $\notionProductWitness{(x,y)}{X}{Y}=(x,y)$.
\end{proposition}
\begin{proof}
    Follows.
\end{proof}

\begin{proposition}\label{notion_composite_product_unary_client}
    Suppose $x\in X$ and $y\in Y$.
    Then $\notionProductOutputWitness{(x,y)}{\notionProduct{X}{Y}}
        =(x,y)$.
\end{proposition}
\begin{proof}
    Follows.
\end{proof}
