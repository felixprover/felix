\begin{definition}\label{notion_equality_paired}
    $X$ is a notion equality paired object over $A$ iff $X=X$.
\end{definition}

\begin{definition}\label{notion_equality_conditional_result}
    $X$ is a notion conditional paired object over $A$ and $B$ iff $X=X$.
\end{definition}

\begin{classification}\label{notion_equality_conditional_classification}
    Suppose if $A=B$ then $X$ is a notion equality paired object over $B$.
    Then $X$ is a notion conditional paired object over $A$ and $B$.
\end{classification}
\begin{proof}
    Follows by \auto{notion_equality_conditional_result}.
\end{proof}
