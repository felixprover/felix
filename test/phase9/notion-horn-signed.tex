\begin{classification}\label{horn_not_in_subset}
    Suppose $A\subseteq B$ and $c\notin B$.
    Then $c\notin A$.
\end{classification}
\begin{proof}
    Follows by \obv{not_in_subseteq}.
\end{proof}

\begin{definition}\label{horn_exclusion_contract}
    Let $c$ be a set such that $c\notin A$.
    $\hornExcluded{c}{A}=c$.
\end{definition}

\begin{proposition}\label{horn_exclusion_consumer}
    Suppose $A\subseteq B$ and $c\notin B$.
    Then $\hornExcluded{c}{A}=c$.
\end{proposition}
\begin{proof}
    Follows by \obv{horn_exclusion_contract}.
\end{proof}

\begin{proposition}\label{horn_proper_filter_negative}
    Let $F$ be a proper filter on $S$.
    Then $\emptyset\notin F$.
\end{proposition}
\begin{proof}
    Follows by notion.
\end{proof}

\begin{proposition}\label{horn_indexed_negative}
    Suppose for every $i\in I$ we have $f(i)\notin B$.
    Suppose $A\subseteq B$.
    Then for every $j\in I$ we have $f(j)\notin A$.
\end{proposition}
\begin{proof}
    Follows by notion.
\end{proof}

\begin{definition}\label{horn_negative_certified}
    $A$ is negatively certified iff $A=A$.
\end{definition}

\begin{classification}\label{horn_negative_body_witness}
    Let $A$ be a set.
    Suppose $c\notin B$.
    Then $A$ is negatively certified.
\end{classification}
\begin{proof}
    Follows by \obv{horn_negative_certified}.
\end{proof}

\begin{proposition}\label{horn_negative_body_consumer}
    Let $A$ be a set.
    Suppose $d\notin C$.
    Then $A$ is negatively certified.
\end{proposition}
\begin{proof}
    Follows by notion.
\end{proof}

\begin{definition}\label{horn_self_containing}
    $A$ is self-containing iff $A\in A$.
\end{definition}

\begin{classification}\label{horn_not_self_containing}
    Suppose $A\notin A$.
    Then $A$ is not self-containing.
\end{classification}
\begin{proof}
    Follows by \obv{horn_self_containing}.
\end{proof}

\begin{proposition}\label{horn_negative_named_consumer}
    Suppose $A\notin A$.
    Then $A$ is not self-containing.
\end{proposition}
\begin{proof}
    Follows by notion.
\end{proof}
