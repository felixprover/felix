\import{test/phase9/notion-direct.tex}

\begin{definition}\label{notion_alias_paired}
    $X$ is a notion alias paired over $Y$ iff $X=X$.
\end{definition}

\begin{definition}\label{notion_alias_pair_value}
    Let $X$ be a notion alias paired over $Y$.
    $\notionAliasPairValue{X}{Y}=X$.
\end{definition}

\begin{proposition}\label{notion_alias_repeated_negative}
    Let $X$ be a notion alias paired over $X$.
    Let $W$ be a set.
    Then $X=X$.
\end{proposition}
\begin{proof}
    Let $Z=X$.
    We have $\notionAliasPairValue{Z}{W}=Z$
        by \auto{notion_alias_pair_value}.
    Follows.
\end{proof}
