\begin{classification}\label{notion_pair_subset_intro}
    Suppose $a\in C$ and $b\in C$.
    Then $\{a,b\}$ is a subset of $C$.
\end{classification}
\begin{proof}
    Follows by \auto{cons_iff,subseteq}.
\end{proof}

\begin{definition}\label{notion_guarded_generated_subgroup}
    Let $G$ be a group.
    Let $A$ be a subset of $\carrier[G]$.
    $\notionGuardedGeneratedSubgroup{A}{G}=
        \generatedSubgroup{A}{G}$.
\end{definition}
