\begin{proposition}\label{notion_neighbourhood_direct_guard}
    Let $X$ be a topological space and $x\in\carrier[X]$.
    Suppose $B$ is a local basis at $x$ in $X$.
    Suppose $C$ is a basis of $\neighbourhoods{x}{X}$ on $\carrier[X]$.
    Then $C$ is a basis of $\neighbourhoods{x}{X}$ on $\carrier[X]$.
\end{proposition}
\begin{proof}
    Follows.
\end{proof}

\begin{proposition}\label{notion_neighbourhood_application_guard}
    Let $X,Y$ be topological spaces.
    Let $f\in\funs{\carrier[X]}{\carrier[Y]}$.
    Suppose $f$ is a homeomorphism from $X$ to $Y$.
    Let $x\in\carrier[X]$.
    Suppose $B$ is a basis of $\neighbourhoods{x}{X}$ on $\carrier[X]$.
    Suppose $C$ is a basis of $\neighbourhoods{f(x)}{Y}$ on $\carrier[Y]$.
    Then $C$ is a basis of $\neighbourhoods{f(x)}{Y}$ on $\carrier[Y]$.
\end{proposition}
\begin{proof}
    Follows.
\end{proof}
