\import{algebra/monoid/unit.tex}

\begin{classification}\label{horn_unit_product}
    Suppose $u$ is a unit of $M$ and $w$ is a unit of $M$.
    Then $\mul[M](u,w)$ is a unit of $M$.
\end{classification}
\begin{proof}
    Follows by \obv{monoid_unit_product}.
\end{proof}

\begin{classification}\label{horn_unit_image}
    Suppose $f$ is a monoid homomorphism from $M$ to $N$.
    Suppose $u$ is a unit of $M$.
    Then $f(u)$ is a unit of $N$.
\end{classification}
\begin{proof}
    Follows by \obv{monoid_hom_preserves_unit}.
\end{proof}

\begin{definition}\label{horn_unit_probe}
    Let $u$ be a unit of $M$.
    The unit probe of $u$ in $M$ is $u$, which is a unit of $M$.
\end{definition}

\begin{proposition}\label{horn_unit_contract}
    Suppose $u$ is a unit of $M$ and $w$ is a unit of $M$.
    Suppose $f$ is a monoid homomorphism from $M$ to $N$.
    Then the unit probe of $f(\mul[M](u,w))$ in $N$ is a unit of $N$.
\end{proposition}
\begin{proof}
    Follows by notion.
\end{proof}
