\begin{proposition}\label{notion_structure_shadow_restored}
    Let $a,b$ be sets.
    Let $A$ be a phase five fieldless carrier.
    Suppose $A$ is a phase five context magma such that
        $\phasefivecombine(a,b)=\phasefivecombine[A](a,b)$.
    Suppose $\phasefivecombine(a,b)=\phasefivecombine(a,b)$.
    Then $A=A$.
\end{proposition}
