\begin{definition}\label{notion_guarded_cyclic_subgroup}
    Let $G$ be a group.
    Let $a\in\carrier[G]$.
    $\notionGuardedCyclicSubgroup{G}{a}=
        \notionGuardedGeneratedSubgroup{\{a\}}{G}$.
\end{definition}
