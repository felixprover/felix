\import{function.tex}

\begin{definition}\label{horn_copy_function}
    Let $f$ be a function on $I$.
    $\hornCopyFunction{f}{I}=f$, which is a function on $I$.
\end{definition}

\begin{proposition}\label{horn_copy_function_consumer}
    Let $f$ be a function on $I$.
    Then $\hornCopyFunction{f}{I}$ is a function on $I$.
\end{proposition}
\begin{proof}
    Follows by notion.
\end{proof}

\begin{definition}\label{horn_copy_codomain}
    Let $B$ be a subset of $C$.
    Let $f$ be a function to $B$.
    $\hornCopyCodomain{f}{B}{C}=f$, which is a function to $C$.
\end{definition}

\begin{proposition}\label{horn_copy_codomain_consumer}
    Let $f$ be a function to $B$.
    Suppose $B\subseteq C$.
    Then $\hornCopyCodomain{f}{B}{C}$ is a function to $C$.
\end{proposition}
\begin{proof}
    Follows by notion.
\end{proof}
