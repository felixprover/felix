\begin{proposition}\label{horn_choice_inhabited}

    Then $A$ is inhabited.
\end{proposition}
\begin{proof}
    Follows by notion.
\end{proof}

\begin{proposition}\label{horn_choice_member}
    Suppose $a\in A$.
    Then $\choice{A}\in A$.
\end{proposition}
\begin{proof}
    Follows by notion.
\end{proof}
