\begin{definition}\label{notion_alias_domain_value}
    Let $x\in\notionGraphDomain{R}$.
    $\notionAliasDomainValue{x}{R}=x$.
\end{definition}

\begin{proposition}\label{notion_alias_domain}
    Suppose $(x,y)\in f$.
    Then $x=x$.
\end{proposition}
\begin{proof}
    Let $g=f$.
    We have $\notionAliasDomainValue{x}{g}=x$
        by \auto{notion_alias_domain_value}.
    Follows.
\end{proof}
