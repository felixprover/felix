\begin{definition}\label{whole_support_primary}
    $X$ is a support primary iff $X=X$.
\end{definition}

\begin{definition}\label{whole_support_target}
    $X$ is support-ready iff $X=\emptyset$.
\end{definition}

\begin{classification}\label{whole_support_all_primary}
    Let $X$ be a set.
    Then $X$ is a support primary.
\end{classification}
\begin{proof}
    Follows by \obv{whole_support_primary}.
\end{proof}

\begin{classification}\label{whole_support_consumer_rule}
    Suppose $(X,X)$ is a support primary.
    Suppose $X=\{\emptyset\}$.
    Suppose there exists $y\in\{\emptyset\}$.
    Suppose $A=\emptyset$.
    Then $A$ is support-ready.
\end{classification}
\begin{proof}
    Follows by \obv{whole_support_target}.
\end{proof}

\begin{proposition}\label{whole_support_source_consumer}
    Suppose $A=\emptyset$.
    Suppose there exists $y\in\{\emptyset\}$.
    Then $A$ is support-ready.
\end{proposition}
\begin{proof}
    Follows by notion.
\end{proof}

\begin{proposition}\label{whole_support_unused_binder}
    Let $A$ be a set.
    Suppose for every $X$ we have $A$ is support-ready.
    Then $A$ is support-ready.
\end{proposition}
\begin{proof}
    Follows by notion.
\end{proof}


\begin{definition}\label{whole_support_opaque}
    $\wholeOpaque : \Set\to\Prop = \fun{x:\Set}{x=x}$.
\end{definition}

\begin{classification}\label{whole_support_mixed_rule}
    Suppose $(X,X)$ is a support primary.
    Suppose $X=\{\emptyset\}$.
    Suppose $\Apply{\wholeOpaque}{\{\emptyset\}}\lor A=\emptyset$.
    Suppose $A=\emptyset$.
    Then $A$ is support-ready.
\end{classification}
\begin{proof}
    Follows by \obv{whole_support_target}.
\end{proof}

\begin{proposition}\label{whole_support_mixed_consumer}
    Suppose $A=\emptyset$.
    Suppose $\Apply{\wholeOpaque}{\{\emptyset\}}\lor A=\emptyset$.
    Then $A$ is support-ready.
\end{proposition}
\begin{proof}
    Follows by notion.
\end{proof}
