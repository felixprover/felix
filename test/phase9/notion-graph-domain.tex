\begin{definition}\label{notion_graph_domain}
    $\notionGraphDomain{R}
        =\{\fst{w}\mid w\in R\mid \exists x,y. w=(x,y)\}$.
\end{definition}

\begin{classification}\label{notion_graph_domain_intro}
    Suppose $(a,b)\in R$.
    Then $a\in\notionGraphDomain{R}$.
\end{classification}
\begin{proof}
    Follows by \auto{notion_graph_domain}.
\end{proof}

\begin{definition}\label{notion_graph_domain_witness}
    Let $a\in\notionGraphDomain{R}$.
    $\notionGraphDomainWitness{a}{R}=a$.
\end{definition}

\begin{proposition}\label{notion_graph_domain_direct}
    Suppose $(a,b)\in R$.
    Then $\notionGraphDomainWitness{a}{R}=a$.
\end{proposition}
\begin{proof}
    Follows.
\end{proof}

\begin{proposition}\label{notion_graph_domain_nested}
    Suppose $((a,b),c)\in R$.
    Then $\notionGraphDomainWitness{a}{\notionGraphDomain{R}}=a$.
\end{proposition}
\begin{proof}
    Follows.
\end{proof}
