\begin{definition}\label{notion_remaining_package}
    $\notionRemainingPackage{X}{D}$ iff $X$ is a notion base and $D=D$.
\end{definition}

\begin{definition}\label{notion_tail_package}
    $\notionTailPackage{X}{D}$ iff $\notionRemainingPackage{X}{D}$ and $D=D$.
\end{definition}

\begin{definition}\label{notion_package_noun}
    $X$ is a notion package over $D$ iff $\notionTailPackage{X}{D}$.
\end{definition}

\begin{definition}\label{notion_package_adjective}
    $X$ is notion packaged over $D$ iff $X$ is a notion package over $D$.
\end{definition}

\begin{definition}\label{notion_package_verb}
    $X$ carries a notion package over $D$ iff $X$ is notion packaged over $D$.
\end{definition}

\begin{definition}\label{notion_package_relation}
    $X\notionPackageRelation{D}Y$ iff
        $X$ carries a notion package over $D$ and $Y=Y$.
\end{definition}

\begin{proposition}\label{notion_tail_package_formation}
    Suppose $\notionTailPackage{X}{\emptyset}$.
    Then $X$ is notion guarded.
\end{proposition}
\begin{proof}
    Follows.
\end{proof}

\begin{proposition}\label{notion_package_chain_replay}
    Suppose $X\notionPackageRelation{\emptyset}Y$.
    Then $X$ is a notion base.
\end{proposition}
\begin{proof}
    Follows by notion.
\end{proof}
