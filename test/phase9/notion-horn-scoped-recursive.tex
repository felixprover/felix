\begin{abbreviation}\label{horn_scoped_equality_ready}
    $A$ is equality-ready iff for every $B$ if $A=B$ then $B$ is a scoped primary.
\end{abbreviation}

\begin{definition}\label{horn_scoped_equality_result}
    Let $A$ be a scoped primary.
    $\hornScopedEqualityResult{A}=A$, which is an equality-ready set.
\end{definition}

\begin{classification}\label{horn_scoped_recursive}
    Suppose for every $x$ we have $(A,x)$ is a scoped complete.
    Then $A$ is a scoped complete.
\end{classification}
\begin{proof}
    Follows by \obv{horn_scoped_complete}.
\end{proof}

\begin{classification}\label{horn_scoped_easy}
    Let $A$ be a scoped primary.
    Then $A$ is a scoped complete.
\end{classification}
\begin{proof}
    Follows by \obv{horn_scoped_complete}.
\end{proof}

\begin{proposition}\label{horn_scoped_fair}
    Let $A$ be a scoped primary.
    Then $A$ is a scoped complete.
\end{proposition}
\begin{proof}
    Follows by notion.
\end{proof}
