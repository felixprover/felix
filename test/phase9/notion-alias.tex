\begin{proposition}\label{notion_alias_direct}
    Let $X$ be a notion base.
    Then $X=X$.
\end{proposition}
\begin{proof}
    Let $Y=X$.
    We have $\notionTermValue{Y}=Y$ by \auto{notion_term_value}.
    Follows.
\end{proof}
