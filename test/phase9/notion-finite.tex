\import{structure/finite.tex}

\begin{struct}\label{notion_finite_group}
    A group $G$ is a onesorted structure.
\end{struct}

\begin{proposition}\label{notion_finite_set_reading}
    Let $X$ be a finite set.
    Then $X$ is a finite set.
\end{proposition}
\begin{proof}
    Follows by assumption.
\end{proof}

\begin{proposition}\label{notion_finite_group_reading}
    Let $G$ be a finite group.
    Then $\carrier[G]$ is a finite set.
\end{proposition}
\begin{proof}
    Follows by \auto{finite_structure}.
\end{proof}
