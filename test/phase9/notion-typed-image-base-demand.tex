\import{library/function.tex}

\begin{definition}\label{notion_typed_image_base_demand_consumer}
    Let $B$ be a subset of $Y$.
    $\notionTypedImageBaseDemandConsumer{B}{Y}=B$.
\end{definition}

\begin{definition}\label{notion_typed_image_deferred_function}
    $f$ is a typed-image deferred function from $X$ to $Y$ over $Z$ iff
        $f\in\funs{X}{Y}$.
\end{definition}

\begin{definition}\label{notion_typed_image_deferred_guard}
    $Z$ is a typed-image deferred guard iff $Z=Z$.
\end{definition}

\begin{classification}\label{notion_typed_image_deferred_result}
    Suppose $f$ is a typed-image deferred function from $X$ to $Y$ over $Z$.
    Let $Z$ be a typed-image deferred guard.
    Then $f\in\funs{X}{Y}$.
\end{classification}
\begin{proof}
    Follows by \auto{notion_typed_image_deferred_function}.
\end{proof}

\begin{proposition}\label{notion_typed_image_base_demand}
    Suppose $f$ is a typed-image deferred function from $X$ to $Y$ over $Z$.
    Let $Z$ be a typed-image deferred guard.
    Let $A$ be a subset of $X$.
    Then $\notionTypedImageBaseDemandConsumer{\img{f}{A}}{Y}=\img{f}{A}$.
\end{proposition}
\begin{proof}
    Follows by \auto{notion_typed_image_base_demand_consumer}.
\end{proof}
