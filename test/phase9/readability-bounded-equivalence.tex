% Independent controls for moving a noun guard into a bounded binder.
% A, F, and B are arbitrary, including empty sets.
\begin{proposition}\label{readability_bounded_universal_equivalence}
    (for every $x\in F$ such that $x$ is an element of $A$ we have $x\in B$)
    iff
    (for every element $x\in F$ of $A$ we have $x\in B$).
\end{proposition}
\begin{proof}
    Follows by \auto.
\end{proof}

\begin{proposition}\label{readability_bounded_existential_equivalence}
    (there exists $x\in F$ such that $x$ is an element of $A$ and $x\in B$)
    iff
    (there exists an element $x\in F$ of $A$ such that $x\in B$).
\end{proposition}
\begin{proof}
    Follows by \auto.
\end{proof}

\begin{proposition}\label{readability_bounded_negative_equivalence}
    (there exists no $x\in F$ such that $x$ is an element of $A$ and $x\in B$)
    iff
    (there exists no element $x\in F$ of $A$ such that $x\in B$).
\end{proposition}
\begin{proof}
    Follows by \auto.
\end{proof}

\begin{proposition}\label{readability_bounded_group_equivalence}
    (there exists $x,y\in F$ such that
        $x$ is an element of $A$ and $y$ is an element of $A$ and $x\neq y$)
    iff
    (there exist elements $x,y\in F$ of $A$ such that $x\neq y$).
\end{proposition}
\begin{proof}
    Follows by \auto.
\end{proof}
