\section{Reciprocals and division}

\subsection{Finite reciprocal approximations}

For a positive surreal $x$, Conway's reciprocal construction repeatedly
improves lower and upper candidates.  If $u$ is an earlier positive option of
$x$ whose reciprocal has already been constructed, the update based at $u$
is the following ring expression.

\begin{definition}\label{surreal_reciprocal_positive_left}
    $\surrealReciprocalPositiveLeft{x}=
        \{u\in\surrealLeft{x}\mid
            \surrealLt{\surrealZero}{u}\}$.
\end{definition}

\begin{definition}\label{surreal_reciprocal_update}
    $\surrealReciprocalUpdate{y}{x}{u}{v}=
        \surrealMul{
            \surrealAdd{\surrealOne}{
                \surrealMul{\surrealAdd{u}{\surrealNeg{x}}}{y}}}{v}$.
\end{definition}

\begin{definition}\label{surreal_reciprocal_update_set}
    $\surrealReciprocalUpdateSetOp:
        \Set\to\Set\to\Set\to(\Set\to\Set)\to\Set
        =\fun{Y,x,X:\Set}{
                    \fun{g:\Set\to\Set}{
                        \{\surrealReciprocalUpdate
                            {\fst{p}}{x}{\snd{p}}{\Apply{g}{\snd{p}}}
                            \mid p\in Y\times X\}}}$.
\end{definition}

\begin{proposition}\label{surreal_reciprocal_update_set_iff}
    $\forall a,x,b:\Set.\ \forall g:\Set\to\Set.\ \forall z:\Set.
        z\in\Apply{\surrealReciprocalUpdateSetOp}{a,x,b,g}
        \iff
        (\exists w,u:\Set.\ w\in a\land u\in b\land
            z=\surrealReciprocalUpdate{w}{x}{u}{\Apply{g}{u}})$.
\end{proposition}
\begin{proof}
    Fix $a,x,b,g,z$.
    Show that if
        $z\in\Apply{\surrealReciprocalUpdateSetOp}{a,x,b,g}$, then
        there exist $w,u$ such that $w\in a$ and $u\in b$ and
            $z=\surrealReciprocalUpdate{w}{x}{u}{\Apply{g}{u}}$.
    \begin{subproof}
        Assume
            $z\in\Apply{\surrealReciprocalUpdateSetOp}{a,x,b,g}$.
        Take $w,u$ such that $w\in a$ and $u\in b$ and
            $z=\surrealReciprocalUpdate{w}{x}{u}{\Apply{g}{u}}$ by
            \auto{surreal_reciprocal_update_set,times_elem_is_tuple,fst_eq,snd_eq}.
        Follows by \auto.
    \end{subproof}
    Show that if there exist $w,u$ such that
        $w\in a$ and $u\in b$ and
        $z=\surrealReciprocalUpdate{w}{x}{u}{\Apply{g}{u}}$, then
        $z\in\Apply{\surrealReciprocalUpdateSetOp}{a,x,b,g}$.
    \begin{subproof}
        Assume that there exist $w,u$ such that
            $w\in a$ and $u\in b$ and
            $z=\surrealReciprocalUpdate{w}{x}{u}{\Apply{g}{u}}$.
        Take $w,u$ such that $w\in a$ and $u\in b$ and
            $z=\surrealReciprocalUpdate{w}{x}{u}{\Apply{g}{u}}$.
        $w\in a$.
        $u\in b$.
        $(w,u)\in a\times b$ by \obv{times_tuple_intro}.
        $\fst{(w,u)}=w$ and $\snd{(w,u)}=u$ by
            \obv{fst_eq,snd_eq}.
        Follows by \auto{surreal_reciprocal_update_set}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{proposition}\label{surreal_reciprocal_update_set_extensionality}
    $\forall a,x,b:\Set.\ \forall g,h:\Set\to\Set.
        (\forall u:\Set.\ u\in b\implies\Apply{g}{u}=\Apply{h}{u})
        \implies
        \Apply{\surrealReciprocalUpdateSetOp}{a,x,b,g}
        =\Apply{\surrealReciprocalUpdateSetOp}{a,x,b,h}$.
\end{proposition}
\begin{proof}
    Fix $a,x,b,g,h$.
    Assume $\forall u:\Set.\ u\in b\implies
        \Apply{g}{u}=\Apply{h}{u}$.
    Show that for every $p\in a\times b$ we have
        $\surrealReciprocalUpdate
            {\fst{p}}{x}{\snd{p}}{\Apply{g}{\snd{p}}}
        =\surrealReciprocalUpdate
            {\fst{p}}{x}{\snd{p}}{\Apply{h}{\snd{p}}}$.
    \begin{subproof}
        Fix $p\in a\times b$.
        $\snd{p}\in b$ by \auto{times_elem_is_tuple,snd_eq}.
        Hence $\Apply{g}{\snd{p}}=\Apply{h}{\snd{p}}$.
    \end{subproof}
    Follows by
        \auto{surreal_reciprocal_update_set,logical_replacement_extensionality}.
\end{proof}

\begin{definition}\label{surreal_reciprocal_approximation_step}
    $\surrealReciprocalApproxStepOp:
        \Set\to(\Set\to\Set)\to\Set\to\Set\to\Set
        =\fun{x:\Set}{
            \fun{g:\Set\to\Set}{
                \fun{n,p:\Set}{
                        (\fst{p}
                            \union
                            \Apply{\surrealReciprocalUpdateSetOp}{\fst{p},x,\surrealRight{x},g}
                            \union
                            \Apply{\surrealReciprocalUpdateSetOp}{\snd{p},x,\surrealReciprocalPositiveLeft{x},g},
                         \snd{p}
                            \union
                            \Apply{\surrealReciprocalUpdateSetOp}{\fst{p},x,\surrealReciprocalPositiveLeft{x},g}
                            \union
                            \Apply{\surrealReciprocalUpdateSetOp}{\snd{p},x,\surrealRight{x},g})}}}$.
\end{definition}

\begin{definition}\label{surreal_reciprocal_approximation}
    $\surrealReciprocalApproxOp:
        \Set\to(\Set\to\Set)\to\Set\to\Set
        =\fun{x:\Set}{
            \fun{g:\Set\to\Set}{
                \fun{n:\Set}{
                    \Apply{\naturalRecOp}
                        {
                                (\{\surrealZero\},\emptyset),
                            \Apply{\surrealReciprocalApproxStepOp}{x,g},n}}}}$.
\end{definition}

\begin{definition}\label{surreal_reciprocal_lower_approximation}
    $\surrealReciprocalLowerApproxOp:
        \Set\to(\Set\to\Set)\to\Set\to\Set
        =\fun{x:\Set}{
            \fun{g:\Set\to\Set}{
                \fun{n:\Set}{
                    \fst{\Apply{\surrealReciprocalApproxOp}{x,g,n}}}}}$.
\end{definition}

\begin{definition}\label{surreal_reciprocal_upper_approximation}
    $\surrealReciprocalUpperApproxOp:
        \Set\to(\Set\to\Set)\to\Set\to\Set
        =\fun{x:\Set}{
            \fun{g:\Set\to\Set}{
                \fun{n:\Set}{
                    \snd{\Apply{\surrealReciprocalApproxOp}{x,g,n}}}}}$.
\end{definition}

\begin{proposition}\label{surreal_reciprocal_approximation_zero}
    $\forall x:\Set.\ \forall g:\Set\to\Set.
        \Apply{\surrealReciprocalApproxOp}
            {x,g,\emptyset}
        =(\{\surrealZero\},\emptyset)$.
\end{proposition}
\begin{proof}
    Follows by
        \obv{surreal_reciprocal_approximation,natural_recursion_zero}.
\end{proof}

\begin{proposition}\label{surreal_reciprocal_approximation_successor}
    $\forall x:\Set.\ \forall g:\Set\to\Set.\ \forall n:\Set.
        n\in\naturals\implies
        \Apply{\surrealReciprocalApproxOp}{x,g,\suc{n}}
        =\Apply{\surrealReciprocalApproxStepOp}{x,g,n,
                \Apply{\surrealReciprocalApproxOp}{x,g,n}}$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{surreal_reciprocal_approximation,natural_recursion_successor}.
\end{proof}

\begin{definition}\label{surreal_reciprocal_lower_options}
    $\surrealReciprocalLowerOptionsOp:
        \Set\to(\Set\to\Set)\to\Set
        =\fun{x:\Set}{
            \fun{g:\Set\to\Set}{
                \unions{\{\Apply{\surrealReciprocalLowerApproxOp}{x,g,n}
                    \mid n\in\naturals\}}}}$.
\end{definition}

\begin{definition}\label{surreal_reciprocal_upper_options}
    $\surrealReciprocalUpperOptionsOp:
        \Set\to(\Set\to\Set)\to\Set
        =\fun{x:\Set}{
            \fun{g:\Set\to\Set}{
                \unions{\{\Apply{\surrealReciprocalUpperApproxOp}{x,g,n}
                    \mid n\in\naturals\}}}}$.
\end{definition}

\subsection{The reciprocal update invariant}

\begin{lemma}\label{surreal_subtraction_negative}
    Suppose $\surreal{a}$ and $\surreal{b}$ and $\surrealLt{a}{b}$.
    Then $\surrealLt{\surrealAdd{a}{\surrealNeg{b}}}{\surrealZero}$.
\end{lemma}
\begin{proof}
    $\surrealLt{
        \surrealAdd{a}{\surrealNeg{b}}}{
        \surrealAdd{b}{\surrealNeg{b}}}$
        by \obv{surreal_add_lt_left,surreal_negation_is_surreal}.
    Follows by \obv{surreal_add_inverse_right}.
\end{proof}

\begin{lemma}\label{surreal_subtraction_positive}
    Suppose $\surreal{a}$ and $\surreal{b}$ and $\surrealLt{a}{b}$.
    Then $\surrealLt{\surrealZero}{
        \surrealAdd{b}{\surrealNeg{a}}}$.
\end{lemma}
\begin{proof}
    $\surrealLt{
        \surrealAdd{a}{\surrealNeg{a}}}{
        \surrealAdd{b}{\surrealNeg{a}}}$
        by \obv{surreal_add_lt_left,surreal_negation_is_surreal}.
    Follows by \obv{surreal_add_inverse_right}.
\end{proof}

\begin{lemma}\label{surreal_subtraction_negative_elim}
    Suppose $\surreal{a}$ and $\surreal{b}$.
    Suppose $\surrealLt{\surrealAdd{a}{\surrealNeg{b}}}{\surrealZero}$.
    Then $\surrealLt{a}{b}$.
\end{lemma}
\begin{proof}
    Follows by
        \auto{surreal_subtract_lt_add,surreal_add_zero_left,empty_surreal}.
\end{proof}

\begin{lemma}\label{surreal_subtraction_positive_elim}
    Suppose $\surreal{a}$ and $\surreal{b}$.
    Suppose $\surrealLt{\surrealZero}{
        \surrealAdd{a}{\surrealNeg{b}}}$.
    Then $\surrealLt{b}{a}$.
\end{lemma}
\begin{proof}
    Follows by
        \auto{surreal_lt_subtract_add,surreal_add_zero_left,empty_surreal}.
\end{proof}

\begin{lemma}\label{surreal_reciprocal_of_positive_is_positive}
    Suppose $\surreal{x}$ and $\surreal{v}$.
    Suppose $\surrealLt{\surrealZero}{x}$ and
        $\surrealMul{x}{v}=\surrealOne$.
    Then $\surrealLt{\surrealZero}{v}$.
\end{lemma}
\begin{proof}
    $\surrealLt{\surrealZero}{v}$ or $v=\surrealZero$ or
        $\surrealLt{v}{\surrealZero}$
        by \obv{surreal_lt_trichotomy,empty_surreal}.
    \begin{byCase}
        \caseOf{$\surrealLt{\surrealZero}{v}$.}
            Follows.
        \caseOf{not $\surrealLt{\surrealZero}{v}$.}
            \begin{byCase}
                \caseOf{$v=\surrealZero$.}
                    $\surrealMul{x}{v}=\surrealZero$
                        by \auto{surreal_mul_zero_right}.
                    $\surrealZero\neq\surrealOne$
                        by \obv{surreal_zero_lt_one,surreal_lt_irreflexive}.
                    Contradiction.
                \caseOf{$v\neq\surrealZero$.}
                    Hence $\surrealLt{v}{\surrealZero}$.
                    $\surrealLt{\surrealMul{x}{v}}{\surrealZero}$
                        by \obv{surreal_mul_pos_neg}.
                    Hence $\surrealLt{\surrealOne}{\surrealZero}$.
                    Contradiction by
                        \obv{surreal_zero_lt_one,surreal_lt_asymmetric,empty_surreal,surreal_one_is_surreal}.
            \end{byCase}
    \end{byCase}
\end{proof}

\begin{lemma}\label{surreal_reciprocal_update_identity}
    Suppose $\surreal{x}$ and $\surreal{u}$ and $\surreal{v}$ and
        $\surreal{y}$.
    Suppose $\surrealMul{u}{v}=\surrealOne$.
    Let $y'=\surrealReciprocalUpdate{y}{x}{u}{v}$.
    Then
        $\surrealAdd{\surrealOne}{\surrealNeg{\surrealMul{x}{y'}}}
        =\surrealMul{
            \surrealMul{
                \surrealAdd{\surrealOne}{
                    \surrealNeg{\surrealMul{x}{y}}}}
                {\surrealAdd{u}{\surrealNeg{x}}}}{v}$.
\end{lemma}
\begin{proof}
    Let $d=\surrealAdd{u}{\surrealNeg{x}}$.
    $\surreal{d}$ and $\surreal{y'}$
        by \auto{surreal_reciprocal_update,surreal_add_is_surreal,surreal_mul_is_surreal,surreal_negation_is_surreal}.
    $y'=\surrealMul{
        \surrealAdd{\surrealOne}{\surrealMul{d}{y}}}{v}$
        by \auto{surreal_reciprocal_update}.
    $\surrealMul{x}{\surrealAdd{\surrealOne}{\surrealMul{d}{y}}}
        =\surrealAdd{x}{\surrealMul{\surrealMul{x}{d}}{y}}$
        by \auto{surreal_mul_distributive_left,surreal_mul_one_right,surreal_mul_associative,surreal_mul_is_surreal}.
    Hence $\surrealMul{x}{y'}=
        \surrealAdd{\surrealMul{x}{v}}{
            \surrealMul{
                \surrealMul{\surrealMul{x}{d}}{y}}{v}}$
        by \auto{surreal_mul_distributive_right,surreal_mul_associative,surreal_mul_is_surreal,surreal_add_is_surreal}.
    Hence
        $\surrealAdd{\surrealOne}{\surrealNeg{\surrealMul{x}{y'}}}
        =\surrealAdd{
            \surrealAdd{\surrealOne}{
                \surrealNeg{\surrealMul{x}{v}}}}{
            \surrealNeg{
                \surrealMul{
                    \surrealMul{\surrealMul{x}{d}}{y}}{v}}}$
        by \auto{surreal_negation_additive,surreal_add_associative,surreal_mul_is_surreal,surreal_add_is_surreal,surreal_negation_is_surreal}.
    $\surrealMul{d}{v}=
        \surrealAdd{\surrealOne}{
            \surrealNeg{\surrealMul{x}{v}}}$
        by \auto{surreal_mul_distributive_right,surreal_mul_neg_left,surreal_mul_neg_right,surreal_mul_is_surreal,surreal_negation_is_surreal}.
    $\surrealMul{
        \surrealMul{\surrealMul{x}{d}}{y}}{v}
        =\surrealMul{
            \surrealMul{\surrealMul{x}{y}}{d}}{v}$
        by \auto{surreal_mul_associative,surreal_mul_commutative,surreal_mul_is_surreal}.
    $\surrealMul{
            \surrealMul{
                \surrealAdd{\surrealOne}{
                    \surrealNeg{\surrealMul{x}{y}}}}{d}}{v}
        =\surrealAdd{
            \surrealMul{d}{v}}{
            \surrealNeg{
                \surrealMul{
                    \surrealMul{\surrealMul{x}{y}}{d}}{v}}}$
        by \auto{surreal_mul_distributive_right,surreal_mul_neg_left,surreal_mul_neg_right,surreal_mul_one_left,surreal_mul_associative,surreal_mul_is_surreal,surreal_add_is_surreal,surreal_negation_is_surreal}.
\end{proof}

\begin{lemma}\label{surreal_reciprocal_update_left_lower}
    Suppose $\surreal{x}$ and $\surreal{u}$ and $\surreal{v}$ and
        $\surreal{y}$.
    Suppose $\surrealLt{\surrealZero}{x}$ and
        $\surrealLt{\surrealZero}{u}$ and
        $\surrealLt{u}{x}$ and
        $\surrealMul{u}{v}=\surrealOne$ and
        $\surrealLt{\surrealMul{x}{y}}{\surrealOne}$.
    Then $\surreal{\surrealReciprocalUpdate{y}{x}{u}{v}}$ and
        $\surrealLt{\surrealOne}{
            \surrealMul{x}{\surrealReciprocalUpdate{y}{x}{u}{v}}}$.
\end{lemma}
\begin{proof}
    Let $y'=\surrealReciprocalUpdate{y}{x}{u}{v}$.
    $\surrealLt{\surrealZero}{v}$
        by \obv{surreal_reciprocal_of_positive_is_positive}.
    $\surreal{y'}$ by
        \auto{surreal_reciprocal_update,surreal_add_is_surreal,surreal_mul_is_surreal,surreal_negation_is_surreal}.
    $\surrealLt{\surrealZero}{
        \surrealAdd{\surrealOne}{
            \surrealNeg{\surrealMul{x}{y}}}}$
        by \obv{surreal_subtraction_positive,surreal_mul_is_surreal,surreal_one_is_surreal}.
    $\surrealLt{\surrealAdd{u}{\surrealNeg{x}}}{\surrealZero}$
        by \obv{surreal_subtraction_negative}.
    Hence $\surrealLt{
        \surrealMul{
            \surrealMul{
                \surrealAdd{\surrealOne}{
                    \surrealNeg{\surrealMul{x}{y}}}}
                {\surrealAdd{u}{\surrealNeg{x}}}}{v}}
        {\surrealZero}$
        by \auto{surreal_mul_pos_neg,surreal_mul_neg_pos,surreal_add_is_surreal,surreal_mul_is_surreal,surreal_negation_is_surreal}.
    Hence $\surrealLt{
        \surrealAdd{\surrealOne}{
            \surrealNeg{\surrealMul{x}{y'}}}}{\surrealZero}$
        by \obv{surreal_reciprocal_update_identity}.
    Follows by
        \obv{surreal_subtraction_negative_elim,surreal_mul_is_surreal,surreal_one_is_surreal}.
\end{proof}

\begin{theorem}\label{surreal_reciprocal_approximation_invariant}
    $\forall x:\Set.\ \forall g:\Set\to\Set.\ \forall n:\Set.
        (\surreal{x}\land
         \surrealLt{\surrealZero}{x}\land
         n\in\naturals\land
         (\forall u:\Set.
            (\surreal{u}\land\birthday{u}\in\birthday{x}\land
                \surrealLt{\surrealZero}{u})
            \implies
            (\surreal{\Apply{g}{u}}\land
                \surrealMul{u}{\Apply{g}{u}}=\surrealOne)))
        \implies
        ((\forall y:\Set.
            y\in\Apply{\surrealReciprocalLowerApproxOp}{x,g,n}
            \implies
            (\surreal{y}\land
                \surrealLt{\surrealMul{x}{y}}{\surrealOne}))
         \land
         (\forall y:\Set.
            y\in\Apply{\surrealReciprocalUpperApproxOp}{x,g,n}
            \implies
            (\surreal{y}\land
                \surrealLt{\surrealOne}{\surrealMul{x}{y}})))$.
\end{theorem}
\begin{proof}
    Fix $x,g,n$.
    Assume $\surreal{x}$ and $\surrealLt{\surrealZero}{x}$ and
        $n\in\naturals$.
    Assume $\forall u:\Set.

\begin{theorem}\label{surreal_reciprocal_options_cut_pair}

    Let $A=\Apply{\surrealReciprocalLowerOptionsOp}{x,g}$.
    Let $B=\Apply{\surrealReciprocalUpperOptionsOp}{x,g}$.

        $\surrealLt{\surrealMul{x}{a}}{\surrealOne}$.

        Take $n\in\naturals$ such that
            $a\in\Apply{\surrealReciprocalLowerApproxOp}{x,g,n}$
            by \auto{surreal_reciprocal_lower_options_iff}.
        Follows by
            \auto{surreal_reciprocal_approximation_invariant}.

    Show that for every $b\in B$ we have $\surreal{b}$ and
        $\surrealLt{\surrealOne}{\surrealMul{x}{b}}$.
    \begin{subproof}
        Fix $b\in B$.
        Take $n\in\naturals$ such that
            $b\in\Apply{\surrealReciprocalUpperApproxOp}{x,g,n}$
            by \auto{surreal_reciprocal_upper_options_iff}.
        Follows by
            \auto{surreal_reciprocal_approximation_invariant}.

        (\forall b:\Set.\ b\in B\implies\surrealLt{a}{b})$.

        Fix $a$.
        Assume $a\in A$.
        Fix $b$.
        Assume $b\in B$.

            $\surrealLt{\surrealMul{x}{a}}{\surrealOne}$.
        $\surreal{b}$ and
            $\surrealLt{\surrealOne}{\surrealMul{x}{b}}$.

            \caseOf{$\surrealLt{a}{b}$.}

            \caseOf{not $\surrealLt{a}{b}$.}

                            {\surrealMul{x}{a}}$
                            by \obv{surreal_pos_mul_lt_left}.
                        Hence $\surrealLt{\surrealOne}{\surrealOne}$
                            by \auto{surreal_lt_transitive,surreal_mul_is_surreal,surreal_one_is_surreal}.
                        Contradiction by
                            \obv{surreal_lt_irreflexive,surreal_one_is_surreal}.
                \end{byCase}
        \end{byCase}
    \end{subproof}
    Follows by \auto{surreal_cut_pair}.
\end{proof}

\begin{lemma}\label{surreal_positive_small_birthday_is_one}
    Suppose $\surreal{z}$ and
        $\surrealLt{\surrealZero}{z}$ and
        $\birthday{z}\subseteq\birthday{\surrealOne}$.
    Then $z=\surrealOne$.
\end{lemma}
\begin{proof}
    $\birthday{z}$ and $\birthday{\surrealOne}$ are ordinals
        by \obv{surreal_birthday_ordinal,surreal_one_is_surreal}.
    $\birthday{\surrealOne}=\suc{\emptyset}$
        by \obv{surreal_one_is_surreal}.
    $\birthday{z}\in\suc{\emptyset}$ or
        $\birthday{z}=\suc{\emptyset}$
        by \auto{ordinal_subseteq_implies_elem_or_eq}.
    \begin{byCase}
        \caseOf{$\birthday{z}=\suc{\emptyset}$.}
            $\surrealAt{\suc{\emptyset}}{z}$
                by \auto{surreal_at_birthday}.
            $\surrealAt{\suc{\emptyset}}{\surrealOne}$
                by \auto{surreal_at_birthday,surreal_one_is_surreal}.
            Show that $\surrealEq{\suc{\emptyset}}{z}{\surrealOne}$.
            \begin{subproof}
                Show that for every $\delta\in\suc{\emptyset}$ we have
                    $\apply{z}{\delta}=\apply{\surrealOne}{\delta}$.
                \begin{subproof}
                    Fix $\delta\in\suc{\emptyset}$.
                    Hence $\delta=\emptyset$ by \auto{suc_iff,emptyset}.
                    $\birthday{\surrealZero}=\emptyset$
                        by \obv{empty_surreal}.
                    $\surrealLtLeftShort{\surrealZero}{z}$
                        by \auto{surreal_lt,surreal_lt_common,surreal_lt_right_short,inter,emptyset}.
                    Hence $\apply{z}{\emptyset}=\suc{\emptyset}$
                        by \auto{surreal_lt_left_short}.
                    $\apply{\surrealOne}{\emptyset}=\suc{\emptyset}$
                        by \obv{ordinal_surreal_value,suc_intro_self}.
                \end{subproof}
                Follows by \obv{surreal_eq_initial}.
            \end{subproof}
            Follows by \obv{surreal_eq_birthday_implies_eq}.
        \caseOf{$\birthday{z}\neq\suc{\emptyset}$.}
            Hence $\birthday{z}\in\suc{\emptyset}$.
            Hence $\birthday{z}=\emptyset$
                by \auto{suc_iff,emptyset}.
            Hence $z=\surrealZero$
                by \obv{surreal_empty_birthday}.
            Contradiction by
                \obv{surreal_lt_irreflexive,empty_surreal}.
    \end{byCase}
\end{proof}

            \surrealMul{\surrealAdd{u}{\surrealNeg{x}}}{w}}$

        =\surrealAdd{\surrealMul{u}{y}}{

        by \auto{surreal_negation_additive,surreal_one_is_surreal,surreal_mul_is_surreal,surreal_add_is_surreal,surreal_negation_is_surreal}.

        \surrealMul{\surrealAdd{u}{\surrealNeg{x}}}{w}}

        by \auto{surreal_mul_neg_left,surreal_negation_additive,surreal_negation_involutive,surreal_add_is_surreal,surreal_negation_is_surreal}.

    Hence $\surrealMul{u}{\surrealAdd{y}{\surrealNeg{z}}}

            \surrealAdd{\surrealNeg{\surrealOne}}{
                \surrealMul{\surrealAdd{\surrealNeg{u}}{x}}{w}}}$
        by \auto.
    $\surreal{\surrealMul{u}{y}}$ and
        $\surreal{
            \surrealMul{\surrealAdd{\surrealNeg{u}}{x}}{w}}$
        by \obv{surreal_mul_is_surreal,surreal_add_is_surreal,surreal_negation_is_surreal}.

    $\surrealAdd{\surrealOne}{
        \surrealAdd{\surrealMul{u}{y}}{
            \surrealAdd{\surrealNeg{\surrealOne}}{
                \surrealMul{\surrealAdd{\surrealNeg{u}}{x}}{w}}}}
        =\surrealAdd{\surrealMul{u}{y}}{
            \surrealAdd{\surrealOne}{
                \surrealAdd{\surrealNeg{\surrealOne}}{
                    \surrealMul{\surrealAdd{\surrealNeg{u}}{x}}{w}}}}$
        by \auto{surreal_add_associative,surreal_add_commutative,surreal_one_is_surreal,surreal_add_is_surreal,surreal_negation_is_surreal}.

\subsection{Reciprocal and division operations}

\begin{definition}\label{surreal_reciprocal_typed}
    $\surrealReciprocalOp:\Set\to\Set
        =\fun{x:\Set}{
            \Apply{\logicalIfSetOp}{
                \surrealLt{\surrealZero}{x},
                \surrealPositiveReciprocal{x},
                \Apply{\logicalIfSetOp}{
                    \surrealLt{x}{\surrealZero},
                    \surrealNeg{
                        \surrealPositiveReciprocal{\surrealNeg{x}}},
                    \surrealZero}}}$.
\end{definition}

\begin{definition}\label{surreal_reciprocal}
    $\surrealReciprocal{x}=\Apply{\surrealReciprocalOp}{x}$.
\end{definition}

\begin{proposition}\label{surreal_reciprocal_positive_equation}
    Suppose $\surrealLt{\surrealZero}{x}$.
    Then $\surrealReciprocal{x}=\surrealPositiveReciprocal{x}$.
\end{proposition}
\begin{proof}
    Follows by
        \obv{surreal_reciprocal,surreal_reciprocal_typed,logical_if_set_true}.
\end{proof}

\begin{proposition}\label{surreal_reciprocal_negative_equation}
    Suppose $\surreal{x}$ and $\surrealLt{x}{\surrealZero}$.
    Then $\surrealReciprocal{x}
        =\surrealNeg{\surrealPositiveReciprocal{\surrealNeg{x}}}$.
\end{proposition}
\begin{proof}
    Show that not $\surrealLt{\surrealZero}{x}$.
    \begin{subproof}
        Suppose $\surrealLt{\surrealZero}{x}$.
        Hence $\surrealLt{\surrealZero}{\surrealZero}$
            by \obv{surreal_lt_transitive,empty_surreal}.
        Contradiction by \obv{surreal_lt_irreflexive,empty_surreal}.
    \end{subproof}
    Follows by
        \auto{surreal_reciprocal,surreal_reciprocal_typed,logical_if_set_false,logical_if_set_true}.
\end{proof}

\begin{proposition}\label{surreal_reciprocal_zero}
    $\surrealReciprocal{\surrealZero}=\surrealZero$.
\end{proposition}
\begin{proof}
    $\lnot\surrealLt{\surrealZero}{\surrealZero}$
        by \obv{surreal_lt_irreflexive,empty_surreal}.
    Follows by
        \auto{surreal_reciprocal,surreal_reciprocal_typed,logical_if_set_false}.
\end{proof}

\begin{theorem}\label{surreal_reciprocal_is_surreal}
    Suppose $\surreal{x}$.
    Then $\surreal{\surrealReciprocal{x}}$.
\end{theorem}
\begin{proof}
    $\surrealLt{x}{\surrealZero}$ or $x=\surrealZero$ or
        $\surrealLt{\surrealZero}{x}$
        by \obv{surreal_lt_trichotomy,empty_surreal}.
    \begin{byCase}
        \caseOf{$\surrealLt{\surrealZero}{x}$.}
            $\surreal{\surrealPositiveReciprocal{x}}$
                by \auto{surreal_positive_reciprocal_inverse}.
            Follows by \obv{surreal_reciprocal_positive_equation}.
        \caseOf{not $\surrealLt{\surrealZero}{x}$.}
            \begin{byCase}
                \caseOf{$\surrealLt{x}{\surrealZero}$.}
                    $\surreal{\surrealNeg{x}}$
                        by \obv{surreal_negation_is_surreal}.
                    $\surrealLt{\surrealZero}{\surrealNeg{x}}$
                        by \auto{surreal_negation_lt_reverses,surreal_negation_empty,empty_surreal}.
                    Hence $\surreal{
                        \surrealPositiveReciprocal{\surrealNeg{x}}}$
                        by \auto{surreal_positive_reciprocal_inverse}.
                    Hence $\surreal{
                        \surrealNeg{
                            \surrealPositiveReciprocal{\surrealNeg{x}}}}$
                        by \obv{surreal_negation_is_surreal}.
                    Follows by
                        \obv{surreal_reciprocal_negative_equation}.
                \caseOf{not $\surrealLt{x}{\surrealZero}$.}
                    Hence $x=\surrealZero$.
                    Follows by \obv{surreal_reciprocal_zero,empty_surreal}.
            \end{byCase}
    \end{byCase}
\end{proof}

\begin{theorem}\label{surreal_reciprocal_inverse}
    Suppose $\surreal{x}$ and $x\neq\surrealZero$.
    Then $\surrealMul{x}{\surrealReciprocal{x}}=\surrealOne$.
\end{theorem}
\begin{proof}
    $\surrealLt{x}{\surrealZero}$ or $x=\surrealZero$ or
        $\surrealLt{\surrealZero}{x}$
        by \obv{surreal_lt_trichotomy,empty_surreal}.
    \begin{byCase}
        \caseOf{$\surrealLt{\surrealZero}{x}$.}
            $\surrealMul{x}{\surrealPositiveReciprocal{x}}
                =\surrealOne$
                by \obv{surreal_positive_reciprocal_inverse}.
            Follows by \auto{surreal_reciprocal_positive_equation}.
        \caseOf{not $\surrealLt{\surrealZero}{x}$.}
            Hence $\surrealLt{x}{\surrealZero}$.
            Let $q=\surrealNeg{x}$.
            $\surreal{q}$ by \obv{surreal_negation_is_surreal}.
            $\surrealLt{\surrealZero}{q}$
                by \auto{surreal_negation_lt_reverses,surreal_negation_empty,empty_surreal}.
            Let $r=\surrealPositiveReciprocal{q}$.
            $\surreal{r}$ and $\surrealMul{q}{r}=\surrealOne$
                by \auto{surreal_positive_reciprocal_inverse}.
            $\surrealReciprocal{x}=\surrealNeg{r}$
                by \auto{surreal_reciprocal_negative_equation}.
            $\surrealMul{x}{\surrealNeg{r}}=\surrealMul{q}{r}$
                by \auto{surreal_mul_both_negated,surreal_negation_involutive,surreal_negation_is_surreal}.
            Follows by \auto.
    \end{byCase}
\end{proof}

\begin{corollary}\label{surreal_reciprocal_inverse_right}
    Suppose $\surreal{x}$ and $x\neq\surrealZero$.
    Then $\surrealMul{\surrealReciprocal{x}}{x}=\surrealOne$.
\end{corollary}
\begin{proof}
    Follows by
        \auto{surreal_reciprocal_inverse,surreal_mul_commutative,surreal_reciprocal_is_surreal}.
\end{proof}

\begin{theorem}\label{surreal_reciprocal_positive}
    Suppose $\surreal{x}$ and
        $\surrealLt{\surrealZero}{x}$.
    Then
        $\surrealLt{\surrealZero}{\surrealReciprocal{x}}$.
\end{theorem}
\begin{proof}
    Let $r=\surrealReciprocal{x}$.
    $\surreal{r}$ by \obv{surreal_reciprocal_is_surreal}.
    Show that $x\neq\surrealZero$.
    \begin{subproof}
        Suppose $x=\surrealZero$.
        Hence $\surrealLt{\surrealZero}{\surrealZero}$.
        Contradiction by
            \obv{surreal_lt_irreflexive,empty_surreal}.
    \end{subproof}
    Hence $\surrealMul{x}{r}=\surrealOne$
        by \auto{surreal_reciprocal_inverse}.
    $\surrealLt{r}{\surrealZero}$ or $r=\surrealZero$ or
        $\surrealLt{\surrealZero}{r}$
        by \auto{surreal_lt_trichotomy,empty_surreal}.
    \begin{byCase}
        \caseOf{$\surrealLt{r}{\surrealZero}$.}
            $\surrealLt{\surrealMul{x}{r}}{\surrealZero}$
                by \obv{surreal_mul_pos_neg}.
            Hence $\surrealLt{\surrealOne}{\surrealZero}$.
            Contradiction by
                \obv{surreal_zero_lt_one,surreal_lt_asymmetric,empty_surreal,surreal_one_is_surreal}.
        \caseOf{not $\surrealLt{r}{\surrealZero}$.}
            \begin{byCase}
                \caseOf{$r=\surrealZero$.}
                    $\surrealMul{x}{r}=\surrealZero$
                        by \auto{surreal_mul_zero_right}.
                    Hence $\surrealOne=\surrealZero$.
                    Contradiction by
                        \obv{surreal_zero_lt_one,surreal_lt_irreflexive,empty_surreal}.
                \caseOf{$r\neq\surrealZero$.}
                    Follows.
            \end{byCase}
    \end{byCase}
\end{proof}

\begin{corollary}\label{surreal_reciprocal_nonzero}
    Suppose $\surreal{x}$ and $x\neq\surrealZero$.
    Then $\surrealReciprocal{x}\neq\surrealZero$.
\end{corollary}
\begin{proof}
    Suppose $\surrealReciprocal{x}=\surrealZero$.
    Hence $\surrealMul{x}{\surrealReciprocal{x}}=\surrealZero$
        by \auto{surreal_mul_zero_right}.
    Hence $\surrealOne=\surrealZero$
        by \auto{surreal_reciprocal_inverse}.
    Contradiction by
        \obv{surreal_zero_lt_one,surreal_lt_irreflexive,empty_surreal}.
\end{proof}

\begin{theorem}\label{surreal_reciprocal_unique}
    Suppose $\surreal{x}$ and $\surreal{y}$.
    Suppose $\surrealMul{x}{y}=\surrealOne$.
    Then $x\neq\surrealZero$ and
        $y=\surrealReciprocal{x}$.
\end{theorem}
\begin{proof}
    Show that $x\neq\surrealZero$.
    \begin{subproof}
        Suppose $x=\surrealZero$.
        Hence $\surrealMul{x}{y}=\surrealZero$
            by \auto{surreal_mul_zero_left}.
        Hence $\surrealOne=\surrealZero$.
        Contradiction by
            \obv{surreal_zero_lt_one,surreal_lt_irreflexive,empty_surreal}.
    \end{subproof}
    $\surreal{\surrealReciprocal{x}}$
        by \obv{surreal_reciprocal_is_surreal}.
    $\surrealMul{x}{\surrealReciprocal{x}}=\surrealOne$
        by \obv{surreal_reciprocal_inverse}.
    Hence $\surrealMul{x}{y}
        =\surrealMul{x}{\surrealReciprocal{x}}$.
    Hence $y=\surrealReciprocal{x}$
        by \obv{surreal_mul_nonzero_cancel_left}.
\end{proof}

\begin{corollary}\label{surreal_reciprocal_involutive}
    Suppose $\surreal{x}$ and $x\neq\surrealZero$.
    Then $\surrealReciprocal{\surrealReciprocal{x}}=x$.
\end{corollary}
\begin{proof}
    $\surreal{\surrealReciprocal{x}}$
        by \obv{surreal_reciprocal_is_surreal}.
    $\surrealMul{\surrealReciprocal{x}}{x}=\surrealOne$
        by \obv{surreal_reciprocal_inverse_right}.
    Hence $x=\surrealReciprocal{\surrealReciprocal{x}}$
        by \obv{surreal_reciprocal_unique}.
\end{proof}

\begin{corollary}\label{surreal_reciprocal_one}
    $\surrealReciprocal{\surrealOne}=\surrealOne$.
\end{corollary}
\begin{proof}
    $\surrealMul{\surrealOne}{\surrealOne}=\surrealOne$
        by \obv{surreal_mul_one_left,surreal_one_is_surreal}.
    Follows by
        \auto{surreal_reciprocal_unique,surreal_one_is_surreal}.
\end{proof}

\begin{lemma}\label{surreal_mul_nonzero}
    Suppose $\surreal{x}$ and $\surreal{y}$.
    Suppose $x\neq\surrealZero$ and $y\neq\surrealZero$.
    Then $\surrealMul{x}{y}\neq\surrealZero$.
\end{lemma}
\begin{proof}
    Suppose $\surrealMul{x}{y}=\surrealZero$.
    $\surrealMul{x}{\surrealZero}=\surrealZero$
        by \obv{surreal_mul_zero_right}.
    Hence $\surrealMul{x}{y}=\surrealMul{x}{\surrealZero}$.
    Hence $y=\surrealZero$
        by \obv{surreal_mul_nonzero_cancel_left,empty_surreal}.
    Contradiction.
\end{proof}

\begin{theorem}\label{surreal_reciprocal_multiplicative}
    Suppose $\surreal{x}$ and $\surreal{y}$.
    Suppose $x\neq\surrealZero$ and $y\neq\surrealZero$.
    Then
        $\surrealReciprocal{\surrealMul{x}{y}}
        =\surrealMul{\surrealReciprocal{x}}{
            \surrealReciprocal{y}}$.
\end{theorem}
\begin{proof}
    $\surreal{\surrealReciprocal{x}}$ and
        $\surreal{\surrealReciprocal{y}}$
        by \obv{surreal_reciprocal_is_surreal}.
    $\surrealMul{x}{\surrealReciprocal{x}}=\surrealOne$ and
        $\surrealMul{y}{\surrealReciprocal{y}}=\surrealOne$
        by \obv{surreal_reciprocal_inverse}.
    $\surrealMul{\surrealMul{x}{y}}{
        \surrealMul{\surrealReciprocal{x}}{
            \surrealReciprocal{y}}}=\surrealOne$
        by \auto{surreal_mul_associative,surreal_mul_commutative,surreal_mul_one_left,surreal_mul_one_right,surreal_one_is_surreal,surreal_mul_is_surreal}.
    Hence $\surrealMul{\surrealReciprocal{x}}{
        \surrealReciprocal{y}}
        =\surrealReciprocal{\surrealMul{x}{y}}$
        by \auto{surreal_reciprocal_unique,surreal_mul_is_surreal}.
\end{proof}
