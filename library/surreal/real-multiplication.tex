\section{Multiplication of positive surreal reals}

\begin{definition}\label{surreal_real_positive_lower}
    $\realPositiveLower{x}=
        \{q\in\realLower{x}\mid\surrealLt{\surrealZero}{q}\}$.
\end{definition}

\begin{proposition}\label{surreal_real_positive_lower_iff}
    $q\in\realPositiveLower{x}$ iff
        $q\in\surrealStage{\naturals}$ and
        $\surrealLt{\surrealZero}{q}$ and $\surrealLt{q}{x}$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{surreal_real_positive_lower,surreal_real_lower_iff}.
\end{proof}

\begin{lemma}\label{surreal_product_perturbation_identity}
    Suppose $\surreal{a}$ and $\surreal{b}$ and $\surreal{e}$.
    Then
        $\surrealMul{\surrealAdd{a}{e}}{\surrealAdd{b}{e}}
        =\surrealAdd{\surrealMul{a}{b}}
            {\surrealAdd
                {\surrealAdd{\surrealMul{e}{a}}{\surrealMul{e}{b}}}
                {\surrealMul{e}{e}}}$.
\end{lemma}
\begin{proof}
    Follows by
        \auto{surreal_mul_foil,surreal_mul_commutative,surreal_add_associative,surreal_add_commutative,surreal_mul_is_surreal,surreal_add_is_surreal}.
\end{proof}

\begin{lemma}\label{surreal_positive_product_perturbation_bound}
    Suppose $\surreal{a}$ and $\surreal{b}$ and
        $\surreal{e}$ and $\surreal{h}$.
    Suppose
        $\surrealLt{\surrealMul{e}{a}}{h}$ and
        $\surrealLt{\surrealMul{e}{b}}{h}$ and
        $\surrealLt{\surrealMul{e}{e}}{h}$.
    Then
        $\surrealLt
            {\surrealMul{\surrealAdd{a}{e}}{\surrealAdd{b}{e}}}
            {\surrealAdd{\surrealMul{a}{b}}
                {\surrealAdd{\surrealAdd{h}{h}}{h}}}$.
\end{lemma}
\begin{proof}
    $\surrealLt
        {\surrealAdd{\surrealMul{e}{a}}{\surrealMul{e}{b}}}
        {\surrealAdd{h}{h}}$
        by \obv{surreal_add_lt_both,surreal_mul_is_surreal}.
    Hence
        $\surrealLt
            {\surrealAdd
                {\surrealAdd{\surrealMul{e}{a}}{\surrealMul{e}{b}}}
                {\surrealMul{e}{e}}}
            {\surrealAdd{\surrealAdd{h}{h}}{h}}$
        by \obv{surreal_add_lt_both,surreal_add_is_surreal,surreal_mul_is_surreal}.
    Hence
        $\surrealLt
            {\surrealAdd{\surrealMul{a}{b}}
                {\surrealAdd
                    {\surrealAdd{\surrealMul{e}{a}}{\surrealMul{e}{b}}}
                    {\surrealMul{e}{e}}}}
            {\surrealAdd{\surrealMul{a}{b}}
                {\surrealAdd{\surrealAdd{h}{h}}{h}}}$
        by \obv{surreal_add_lt_right,surreal_mul_is_surreal,surreal_add_is_surreal}.
    Follows by \obv{surreal_product_perturbation_identity}.
\end{proof}

\begin{lemma}\label{surreal_epsilon_triple_bound}
    Let $n$ be a natural number.
    Let $h=\surrealEpsilon{\suc{\suc{n}}}$.
    Then
        $\surrealLt{\surrealAdd{\surrealAdd{h}{h}}{h}}
            {\surrealEpsilon{n}}$.
\end{lemma}
\begin{proof}
    $\suc{n}$ and $\suc{\suc{n}}$ are natural numbers
        by \obv{omega_successor}.
    $\surrealAdd{h}{h}=\surrealEpsilon{\suc{n}}$
        by
        \auto{surreal_epsilon_halving,natural_number_is_ordinal}.
    Hence
        $\surrealAdd{\surrealAdd{h}{h}}
            {\surrealAdd{h}{h}}=\surrealEpsilon{n}$
        by
        \auto{surreal_epsilon_halving,natural_number_is_ordinal}.
    $\surrealLt{\surrealZero}{h}$
        by \obv{surreal_epsilon_positive}.
    Hence $\surrealLt{h}{\surrealAdd{h}{h}}$
        by
        \auto{surreal_add_lt_right,surreal_add_zero_right,empty_surreal,surreal_epsilon_is_surreal}.
    Hence
        $\surrealLt{\surrealAdd{\surrealAdd{h}{h}}{h}}
            {\surrealAdd{\surrealAdd{h}{h}}{\surrealAdd{h}{h}}}$
        by
        \auto{surreal_add_lt_right,surreal_add_is_surreal,surreal_epsilon_is_surreal}.
\end{proof}

\begin{proposition}\label{surreal_real_positive_lower_inhabited}
    Suppose $x\in\reals$ and $\surrealLt{\surrealZero}{x}$.
    Then $\realPositiveLower{x}$ is inhabited.
\end{proposition}
\begin{proof}
    $\surrealZero\in\surrealStage{\naturals}$
        by
        \auto{surreal_stage_characterization,empty_surreal,omega_empty,omega_is_ordinal}.
    Hence $\surrealZero\in\realLower{x}$
        by \obv{surreal_real_lower_iff}.
    Take $q\in\realLower{x}$ such that
        $\surrealLt{\surrealZero}{q}$
        by \obv{surreal_real_lower_no_maximum}.
    Hence $q\in\realPositiveLower{x}$
        by
        \obv{surreal_real_positive_lower_iff,surreal_real_lower_iff}.
\end{proof}

\begin{proposition}\label{surreal_real_positive_lower_no_maximum}
    Suppose $x\in\reals$.
    Then for every $q\in\realPositiveLower{x}$ there exists
        $q'\in\realPositiveLower{x}$ such that $\surrealLt{q}{q'}$.
\end{proposition}
\begin{proof}
    Fix $q\in\realPositiveLower{x}$.
    $q\in\surrealStage{\naturals}$ and
        $q\in\realLower{x}$ and $\surrealLt{\surrealZero}{q}$
        by
        \auto{surreal_real_positive_lower_iff,surreal_real_lower_iff}.
    Take $q'\in\realLower{x}$ such that $\surrealLt{q}{q'}$
        by \obv{surreal_real_lower_no_maximum}.
    $q'\in\surrealStage{\naturals}$ and $\surrealLt{q'}{x}$
        by \obv{surreal_real_lower_iff}.
    $\surreal{q}$ and $\surreal{q'}$
        by
        \auto{surreal_stage_characterization,omega_is_ordinal}.
    $\surrealLt{\surrealZero}{q'}$
        by
        \obv{surreal_lt_transitive,empty_surreal,surreal_stage_characterization,omega_is_ordinal}.
    Hence $q'\in\realPositiveLower{x}$
        by \obv{surreal_real_positive_lower_iff}.
\end{proof}

\begin{definition}\label{surreal_real_mul_lower}
    $\realMulLower{x}{y}=
        \{\surrealMul{\fst{p}}{\snd{p}}
            \mid p\in\realPositiveLower{x}\times
                \realPositiveLower{y}\}$.
\end{definition}

\begin{definition}\label{surreal_real_mul_upper}
    $\realMulUpper{x}{y}=
        \{\surrealMul{\fst{p}}{\snd{p}}
            \mid p\in\realUpper{x}\times\realUpper{y}\}$.
\end{definition}

\begin{proposition}\label{surreal_real_mul_lower_iff}
    $z\in\realMulLower{x}{y}$ iff there exist $a,b$ such that
        $a\in\realPositiveLower{x}$ and
        $b\in\realPositiveLower{y}$ and
        $z=\surrealMul{a}{b}$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{surreal_real_mul_lower,times_elem_is_tuple,times_tuple_intro,fst_eq,snd_eq}.
\end{proof}

\begin{proposition}\label{surreal_real_mul_upper_iff}
    $z\in\realMulUpper{x}{y}$ iff there exist $a,b$ such that
        $a\in\realUpper{x}$ and $b\in\realUpper{y}$ and
        $z=\surrealMul{a}{b}$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{surreal_real_mul_upper,times_elem_is_tuple,times_tuple_intro,fst_eq,snd_eq}.
\end{proof}

\begin{proposition}\label{surreal_real_mul_bounds_finite_stage}
    Suppose $x,y\in\reals$.
    Then $\realMulLower{x}{y}\union\realMulUpper{x}{y}
        \subseteq\surrealStage{\naturals}$.
\end{proposition}
\begin{proof}
    Show that for every
        $z\in\realMulLower{x}{y}\union\realMulUpper{x}{y}$ we have
        $z\in\surrealStage{\naturals}$.
    \begin{subproof}
        Fix $z\in\realMulLower{x}{y}\union\realMulUpper{x}{y}$.
        \begin{byCase}
            \caseOf{$z\in\realMulLower{x}{y}$.}
                Take $a,b$ such that
                    $a\in\realPositiveLower{x}$ and
                    $b\in\realPositiveLower{y}$ and
                    $z=\surrealMul{a}{b}$
                    by \auto{surreal_real_mul_lower_iff}.
                $a,b\in\surrealStage{\naturals}$
                    by \auto{surreal_real_positive_lower_iff}.
                Follows by \auto{surreal_multiplication_finite_stage}.
            \caseOf{$z\notin\realMulLower{x}{y}$.}
                Hence $z\in\realMulUpper{x}{y}$
                    by \auto{union_iff}.
                Take $a,b$ such that
                    $a\in\realUpper{x}$ and
                    $b\in\realUpper{y}$ and
                    $z=\surrealMul{a}{b}$
                    by \auto{surreal_real_mul_upper_iff}.
                $a,b\in\surrealStage{\naturals}$
                    by \obv{surreal_real_upper_iff}.
                Follows by \auto{surreal_multiplication_finite_stage}.
        \end{byCase}
    \end{subproof}
    Follows by \obv{subseteq}.
\end{proof}

\begin{proposition}\label{surreal_real_mul_cut_pair}
    Suppose $x,y\in\reals$.
    Suppose $\surrealLt{\surrealZero}{x}$ and
        $\surrealLt{\surrealZero}{y}$.
    Then $\surrealCutPair{\realMulLower{x}{y}}
        {\realMulUpper{x}{y}}$.
\end{proposition}
\begin{proof}
    For every $z\in\realMulLower{x}{y}\union
        \realMulUpper{x}{y}$ we have
        $z\in\surrealStage{\naturals}$
        by
        \auto{surreal_real_mul_bounds_finite_stage,subseteq}.
    Hence for every $z\in\realMulLower{x}{y}\union
        \realMulUpper{x}{y}$ we have $\surreal{z}$
        by
        \auto{surreal_stage_characterization,omega_is_ordinal}.
    Show that for all $l\in\realMulLower{x}{y}$ we have for all
        $r\in\realMulUpper{x}{y}$ we have $\surrealLt{l}{r}$.
    \begin{subproof}
        Fix $l\in\realMulLower{x}{y}$.
        Fix $r\in\realMulUpper{x}{y}$.
        Take $a,b$ such that
            $a\in\realPositiveLower{x}$ and
            $b\in\realPositiveLower{y}$ and
            $l=\surrealMul{a}{b}$
            by \auto{surreal_real_mul_lower_iff}.
        Take $c,d$ such that
            $c\in\realUpper{x}$ and $d\in\realUpper{y}$ and
            $r=\surrealMul{c}{d}$
            by \obv{surreal_real_mul_upper_iff}.
        $a,b,c,d\in\surrealStage{\naturals}$ and
            $\surrealLt{\surrealZero}{a}$ and
            $\surrealLt{\surrealZero}{b}$ and
            $\surrealLt{a}{x}$ and $\surrealLt{x}{c}$ and
            $\surrealLt{b}{y}$ and $\surrealLt{y}{d}$
            by
            \auto{surreal_real_positive_lower_iff,surreal_real_upper_iff}.
        Hence $\surreal{a}$ and $\surreal{b}$ and
            $\surreal{c}$ and $\surreal{d}$
            by
            \auto{surreal_stage_characterization,omega_is_ordinal}.
        $\surrealLt{a}{c}$ and $\surrealLt{b}{d}$ and
            $\surrealLt{\surrealZero}{c}$
            by
            \obv{surreal_lt_transitive,surreal_real_is_surreal,empty_surreal}.
        $\surrealLt{\surrealMul{a}{b}}{\surrealMul{c}{b}}$
            by \obv{surreal_pos_mul_lt_right}.
        $\surrealLt{\surrealMul{c}{b}}{\surrealMul{c}{d}}$
            by \obv{surreal_pos_mul_lt_left}.
        Follows by
            \auto{surreal_lt_transitive,surreal_mul_is_surreal}.
    \end{subproof}
    For every $l\in\realMulLower{x}{y}$ we have $\surreal{l}$
        by \auto{union_iff}.
    For every $r\in\realMulUpper{x}{y}$ we have $\surreal{r}$
        by \auto{union_iff}.
    Follows by \obv{surreal_cut_pair}.
\end{proof}

\subsection{Quantitative product approximation}

\begin{lemma}\label{surreal_epsilon_at_most_one}
    Let $n$ be an ordinal.
    Suppose $n$ is a natural number.
    Then $\surrealLe{\surrealEpsilon{n}}{\surrealOne}$.
\end{lemma}
\begin{proof}[Proof by transfinite induction]
    $n=\emptyset$ or there exists a natural number $k$ such that
        $n=\suc{k}$ by \obv{natural_zero_or_successor}.
    \begin{byCase}
        \caseOf{$n=\emptyset$.}
            $\surrealEpsilon{n}=\surrealOne$
                by \obv{surreal_epsilon_zero}.
            Follows by
                \obv{surreal_le,surreal_one_is_surreal}.
        \caseOf{$n\neq\emptyset$.}
            Take $k$ such that $k$ is a natural number and
                $n=\suc{k}$.
            $k\in n$ by \obv{suc_intro_self}.
            $\surrealLe{\surrealEpsilon{k}}{\surrealOne}$.
            $\surrealLt{\surrealEpsilon{n}}{\surrealEpsilon{k}}$
                by \obv{surreal_epsilon_decreasing}.
            Follows by
                \auto{surreal_le,surreal_lt_le_transitive,surreal_epsilon_is_surreal,surreal_one_is_surreal}.
    \end{byCase}
\end{proof}

\begin{lemma}\label{surreal_epsilon_times_natural_at_most_one}
    Let $n$ be an ordinal.
    Suppose $n$ is a natural number.
    Then
        $\surrealLe
            {\surrealMul{\surrealEpsilon{n}}{\ordinalSurreal{n}}}
            {\surrealOne}$.
\end{lemma}
\begin{proof}[Proof by transfinite induction]
    $n=\emptyset$ or there exists a natural number $k$ such that
        $n=\suc{k}$ by \obv{natural_zero_or_successor}.
    \begin{byCase}
        \caseOf{$n=\emptyset$.}
            $\ordinalSurreal{n}=\surrealZero$
                by \obv{empty_ordinal_surreal}.
            $\surrealMul{\surrealEpsilon{n}}{\ordinalSurreal{n}}
                =\surrealZero$
                by
                \auto{surreal_mul_zero_right,surreal_epsilon_is_surreal}.
            Follows by
                \obv{surreal_le,surreal_zero_lt_one,empty_surreal,surreal_one_is_surreal}.
        \caseOf{$n\neq\emptyset$.}
            Take $k$ such that $k$ is a natural number and
                $n=\suc{k}$.
            $k\in n$ by \obv{suc_intro_self}.
            Let $z=\surrealMul{\surrealEpsilon{n}}
                {\ordinalSurreal{n}}$.
            $\surrealLe
                {\surrealMul{\surrealEpsilon{k}}
                    {\ordinalSurreal{k}}}{\surrealOne}$.
            $\surrealLe{\surrealEpsilon{k}}{\surrealOne}$
                by
                \obv{surreal_epsilon_at_most_one,natural_number_is_ordinal}.
            Hence
                $\surrealLe
                    {\surrealAdd
                        {\surrealMul{\surrealEpsilon{k}}
                            {\ordinalSurreal{k}}}
                        {\surrealEpsilon{k}}}
                    {\surrealAdd{\surrealOne}{\surrealOne}}$
                by
                \auto{surreal_add_le_both,surreal_mul_is_surreal,surreal_epsilon_is_surreal,ordinal_surreal_is_surreal,natural_number_is_ordinal,surreal_one_is_surreal}.
            $\surrealAdd{\surrealEpsilon{n}}{\surrealEpsilon{n}}
                =\surrealEpsilon{k}$
                by
                \auto{surreal_epsilon_halving,natural_number_is_ordinal}.
            $\ordinalSurreal{n}
                =\surrealAdd{\ordinalSurreal{k}}{\surrealOne}$
                by
                \auto{ordinal_surreal_successor_addition,natural_number_is_ordinal}.
            Hence
                $\surrealAdd{z}{z}
                    =\surrealAdd
                        {\surrealMul{\surrealEpsilon{k}}
                            {\ordinalSurreal{k}}}
                        {\surrealEpsilon{k}}$
                by
                \auto{surreal_mul_distributive_left,surreal_mul_distributive_right,surreal_mul_one_right,surreal_add_associative,surreal_add_commutative,surreal_mul_is_surreal,surreal_epsilon_is_surreal,ordinal_surreal_is_surreal,natural_number_is_ordinal,surreal_one_is_surreal}.
            Hence
                $\surrealLe{\surrealAdd{z}{z}}
                    {\surrealAdd{\surrealOne}{\surrealOne}}$.
            $\surreal{z}$ by
                \auto{surreal_mul_is_surreal,surreal_epsilon_is_surreal,ordinal_surreal_is_surreal,natural_number_is_ordinal}.
            Show that $\surrealLe{z}{\surrealOne}$.
            \begin{subproof}
                Suppose not.
                $\surrealLe{\surrealOne}{z}$
                    by \auto{surreal_le_total,surreal_one_is_surreal}.
                Hence $\surrealLt{\surrealOne}{z}$
                    by \auto{surreal_le}.
                Hence
                    $\surrealLt
                        {\surrealAdd{\surrealOne}{\surrealOne}}
                        {\surrealAdd{z}{z}}$
                    by
                    \obv{surreal_add_lt_both,surreal_one_is_surreal}.
                Contradiction by
                    \auto{surreal_le,surreal_lt_asymmetric,surreal_add_is_surreal,surreal_one_is_surreal}.
            \end{subproof}
            Follows by \auto.
    \end{byCase}
\end{proof}

\begin{lemma}\label{positive_real_scaled_epsilon_bound}
    Suppose $x\in\reals$ and $\surrealLt{\surrealZero}{x}$.
    Then there exists a natural number $K$ such that
        $\surrealLt{\surrealMul{\surrealEpsilon{K}}{x}}
            {\surrealOne}$.
\end{lemma}
\begin{proof}
    Take a natural number $N$ such that
        $\surrealLt{x}{\ordinalSurreal{N}}$
        by
        \auto{successor_omega_surreal_natural_bound_above,surreal_real_elim,surreal_real_bounds}.
    Let $K=\suc{N}$.
    $K$ is a natural number by \auto{omega_successor}.
    $\surrealLt{\ordinalSurreal{N}}{\ordinalSurreal{K}}$
        by
        \auto{ordinal_surreal_lt_iff,natural_number_is_ordinal,suc_intro_self}.
    Hence $\surrealLt{x}{\ordinalSurreal{K}}$
        by
        \obv{surreal_lt_transitive,surreal_real_is_surreal,ordinal_surreal_is_surreal,natural_number_is_ordinal}.
    $\surrealLt{\surrealZero}{\surrealEpsilon{K}}$
        by \obv{surreal_epsilon_positive}.
    Hence
        $\surrealLt{\surrealMul{\surrealEpsilon{K}}{x}}
            {\surrealMul{\surrealEpsilon{K}}{\ordinalSurreal{K}}}$
        by
        \auto{surreal_pos_mul_lt_left,surreal_real_is_surreal,surreal_epsilon_is_surreal,ordinal_surreal_is_surreal,natural_number_is_ordinal}.
    $\surrealLe
        {\surrealMul{\surrealEpsilon{K}}{\ordinalSurreal{K}}}
        {\surrealOne}$
        by
        \obv{surreal_epsilon_times_natural_at_most_one,natural_number_is_ordinal}.
    $\surreal{\surrealMul{\surrealEpsilon{K}}{x}}$ and
        $\surreal{\surrealMul{\surrealEpsilon{K}}
            {\ordinalSurreal{K}}}$
        by
        \auto{surreal_mul_is_surreal,surreal_epsilon_is_surreal,surreal_real_is_surreal,ordinal_surreal_is_surreal,natural_number_is_ordinal}.
    Hence
        $\surrealLt{\surrealMul{\surrealEpsilon{K}}{x}}
            {\surrealOne}$
        by
        \auto{surreal_lt_le_transitive,surreal_mul_is_surreal,surreal_one_is_surreal}.
    Follows by \auto.

\begin{lemma}\label{positive_reals_common_scaled_epsilon_bound}

        by \auto{surreal_real_mul_cut_pair}.

        by \auto{surreal_mul_cut_pair}.

        $\surrealLt{l}{\surrealCut{A}{B}}$.

        Hence $\surrealLt{\surrealMul{x}{y}}{r}$
            by
            \auto{surreal_lt_transitive,surreal_mul_is_surreal,surreal_real_is_surreal}.
        Follows by \auto{surreal_mul_equation}.
    \end{subproof}
    For every $l\in A$ we have $\surrealLt{l}{c}$
        by \auto{positive_reals_mul_left_option_below_cut}.
    For every $r\in B$ we have $\surrealLt{c}{r}$
        by \auto{positive_reals_mul_right_option_above_cut}.
    Hence $c=\surrealCut{A}{B}$
        by \auto{surreal_cut_extensionality}.
    Follows by \auto{surreal_mul_equation}.
\end{proof}

\begin{theorem}\label{surreal_positive_real_multiplication_closed}
    Suppose $x,y\in\reals$.
    Suppose $\surrealLt{\surrealZero}{x}$ and
        $\surrealLt{\surrealZero}{y}$.
    Then $\surrealMul{x}{y}\in\reals$.
\end{theorem}
\begin{proof}
    $\surrealCut{\realMulLower{x}{y}}{\realMulUpper{x}{y}}
        \in\reals$ by \obv{surreal_positive_real_mul_cut_is_real}.
    Follows by \auto{surreal_positive_real_mul_cut_equation}.
\end{proof}

\begin{theorem}\label{surreal_real_multiplication_closed}
    Suppose $x,y\in\reals$.
    Then $\surrealMul{x}{y}\in\reals$.
\end{theorem}
\begin{proof}
    $\surreal{x}$ and $\surreal{y}$
        by \obv{surreal_real_is_surreal}.
    $\surrealLt{\surrealZero}{x}$ or $x=\surrealZero$ or
        $\surrealLt{x}{\surrealZero}$
        by \obv{surreal_lt_trichotomy,empty_surreal}.
    $\surrealLt{\surrealZero}{y}$ or $y=\surrealZero$ or
        $\surrealLt{y}{\surrealZero}$
        by \auto{surreal_lt_trichotomy,empty_surreal}.
    \begin{byCase}
        \caseOf{$x=\surrealZero$.}
            $\surrealMul{x}{y}=\surrealZero$
                by \auto{surreal_mul_zero_left}.
            Follows by \obv{surreal_zero_is_real}.
        \caseOf{$x\neq\surrealZero$.}
            \begin{byCase}
                \caseOf{$y=\surrealZero$.}
                    $\surrealMul{x}{y}=\surrealZero$
                        by \auto{surreal_mul_zero_right}.
                    Follows by \obv{surreal_zero_is_real}.
                \caseOf{$y\neq\surrealZero$.}
                    \begin{byCase}
                        \caseOf{$\surrealLt{\surrealZero}{x}$.}
                            \begin{byCase}
                                \caseOf{$\surrealLt{\surrealZero}{y}$.}
                                    Follows by
                                        \obv{surreal_positive_real_multiplication_closed}.
                                \caseOf{not $\surrealLt{\surrealZero}{y}$.}
                                    Hence
                                        $\surrealLt{y}{\surrealZero}$.
                                    Hence
                                        $\surrealLt{\surrealZero}
                                            {\surrealNeg{y}}$
                                        by
                                        \auto{surreal_negation_lt_reverses,surreal_negation_empty,empty_surreal}.
                                    $\surrealNeg{y}\in\reals$
                                        by
                                        \obv{surreal_real_negation_closed}.
                                    Hence
                                        $\surrealMul{x}{\surrealNeg{y}}
                                            \in\reals$
                                        by
                                        \obv{surreal_positive_real_multiplication_closed}.
                                    Hence
                                        $\surrealNeg{\surrealMul{x}
                                            {\surrealNeg{y}}}\in\reals$
                                        by
                                        \obv{surreal_real_negation_closed}.
                                    $\surrealNeg{\surrealMul{x}
                                        {\surrealNeg{y}}}
                                        =\surrealMul{x}{y}$
                                        by
                                        \auto{surreal_mul_neg_right,surreal_negation_involutive,surreal_real_is_surreal,surreal_negation_is_surreal}.
                            \end{byCase}
                        \caseOf{not $\surrealLt{\surrealZero}{x}$.}
                            Hence $\surrealLt{x}{\surrealZero}$.
                            Hence
                                $\surrealLt{\surrealZero}
                                    {\surrealNeg{x}}$
                                by
                                \auto{surreal_negation_lt_reverses,surreal_negation_empty,empty_surreal}.
                            $\surrealNeg{x}\in\reals$
                                by \obv{surreal_real_negation_closed}.
                            \begin{byCase}
                                \caseOf{$\surrealLt{\surrealZero}{y}$.}
                                    $\surrealMul{\surrealNeg{x}}{y}
                                        \in\reals$
                                        by
                                        \obv{surreal_positive_real_multiplication_closed}.
                                    Hence
                                        $\surrealNeg{\surrealMul
                                            {\surrealNeg{x}}{y}}
                                            \in\reals$
                                        by
                                        \obv{surreal_real_negation_closed}.
                                    $\surrealNeg{\surrealMul
                                        {\surrealNeg{x}}{y}}
                                        =\surrealMul{x}{y}$
                                        by
                                        \auto{surreal_mul_neg_left,surreal_negation_involutive,surreal_real_is_surreal,surreal_negation_is_surreal}.
                                \caseOf{not $\surrealLt{\surrealZero}{y}$.}
                                    Hence $\surrealLt{y}{\surrealZero}$.
                                    Hence
                                        $\surrealLt{\surrealZero}
                                            {\surrealNeg{y}}$
                                        by
                                        \auto{surreal_negation_lt_reverses,surreal_negation_empty,empty_surreal}.
                                    $\surrealNeg{y}\in\reals$
                                        by
                                        \obv{surreal_real_negation_closed}.
                                    Hence
                                        $\surrealMul{\surrealNeg{x}}
                                            {\surrealNeg{y}}\in\reals$
                                        by
                                        \obv{surreal_positive_real_multiplication_closed}.
                                    $\surrealMul{\surrealNeg{x}}
                                        {\surrealNeg{y}}
                                        =\surrealMul{x}{y}$
                                        by
                                        \obv{surreal_mul_both_negated,surreal_real_is_surreal}.
                            \end{byCase}
                    \end{byCase}
            \end{byCase}
    \end{byCase}
\end{proof}
