\section{Arithmetic of the surreal integers}

\import{surreal/embedding.tex}

The integer carrier from the embedding layer consists of surreal values
representable as differences of embedded naturals.  The following
abbreviation names such a presentation without imposing a canonical sign
normal form.  It denotes the surreal value itself, not a pair or quotient
representative; distinct presentations may therefore be equal.

\subsection{Presentations and normal forms}

\begin{abbreviation}\label{surreal_integer_of}
    $\surrealIntegerOf{n}{m}=
        \surrealAdd{\ordinalSurreal{n}}
            {\surrealNeg{\ordinalSurreal{m}}}$.
\end{abbreviation}

\begin{proposition}\label{surreal_integer_of_iff}
    $x\in\surrealIntegers$ iff there exist natural numbers $n,m$ such that
        $x=\surrealIntegerOf{n}{m}$.
\end{proposition}
\begin{proof}
    Follows by \obv{surreal_integer_iff}.
\end{proof}

\begin{lemma}\label{surreal_difference_equality_iff}
    Suppose $\surreal{a}$ and $\surreal{b}$ and
        $\surreal{c}$ and $\surreal{d}$.
    Then
        $\surrealAdd{a}{\surrealNeg{b}}
            =\surrealAdd{c}{\surrealNeg{d}}$ iff
        $\surrealAdd{a}{d}=\surrealAdd{c}{b}$.
\end{lemma}
\begin{proof}
    $\surreal{\surrealNeg{b}}$ and $\surreal{\surrealNeg{d}}$
        by \obv{surreal_negation_is_surreal}.
    Follows by
        \auto{surreal_add_associative,surreal_add_commutative,surreal_add_inverse_left,surreal_add_inverse_right,surreal_add_zero_left,surreal_add_zero_right,surreal_add_cancel_left,surreal_add_cancel_right,surreal_add_is_surreal}.
\end{proof}

\begin{lemma}\label{surreal_difference_order_iff}
    Suppose $\surreal{a}$ and $\surreal{b}$ and
        $\surreal{c}$ and $\surreal{d}$.
    Then
        $\surrealLt
            {\surrealAdd{a}{\surrealNeg{b}}}
            {\surrealAdd{c}{\surrealNeg{d}}}$ iff
        $\surrealLt{\surrealAdd{a}{d}}{\surrealAdd{c}{b}}$.
\end{lemma}
\begin{proof}
    $\surreal{\surrealNeg{b}}$ and $\surreal{\surrealNeg{d}}$
        by \obv{surreal_negation_is_surreal}.
    Follows by
        \auto{surreal_add_associative,surreal_add_commutative,surreal_add_inverse_left,surreal_add_inverse_right,surreal_add_zero_left,surreal_add_zero_right,surreal_add_lt_cancel_left,surreal_add_lt_cancel_right,surreal_add_lt_left,surreal_add_lt_right,surreal_add_is_surreal}.
\end{proof}

\begin{theorem}\label{surreal_integer_of_equality_iff}
    Let $n,m,p,q$ be natural numbers.
    Then
        $\surrealIntegerOf{n}{m}=\surrealIntegerOf{p}{q}$ iff
        $n+q=p+m$.
\end{theorem}
\begin{proof}
    $n+q$ and $p+m$ are natural numbers
        by \obv{num_addition_on_naturals}.
    $n$, $m$, $p$, $q$, $n+q$, and $p+m$ are ordinals
        by \obv{natural_number_is_ordinal}.
    $\surreal{\ordinalSurreal{n}}$ and
        $\surreal{\ordinalSurreal{m}}$ and
        $\surreal{\ordinalSurreal{p}}$ and
        $\surreal{\ordinalSurreal{q}}$
        by \obv{ordinal_surreal_is_surreal}.
    $\surrealAdd{\ordinalSurreal{n}}{\ordinalSurreal{q}}
        =\ordinalSurreal{n+q}$ and
        $\surrealAdd{\ordinalSurreal{p}}{\ordinalSurreal{m}}
        =\ordinalSurreal{p+m}$
        by \obv{natural_surreal_addition_compatibility}.
    Follows by
        \auto{surreal_difference_equality_iff,ordinal_surreal_injective}.
\end{proof}

\begin{theorem}\label{surreal_integer_of_order_iff}
    Let $n,m,p,q$ be natural numbers.
    Then
        $\surrealLt
            {\surrealIntegerOf{n}{m}}
            {\surrealIntegerOf{p}{q}}$ iff
        $n+q\in p+m$.
\end{theorem}
\begin{proof}
    $n+q$ and $p+m$ are natural numbers
        by \obv{num_addition_on_naturals}.
    $n$, $m$, $p$, $q$, $n+q$, and $p+m$ are ordinals
        by \obv{natural_number_is_ordinal}.
    $\surreal{\ordinalSurreal{n}}$ and
        $\surreal{\ordinalSurreal{m}}$ and
        $\surreal{\ordinalSurreal{p}}$ and
        $\surreal{\ordinalSurreal{q}}$
        by \obv{ordinal_surreal_is_surreal}.
    $\surrealAdd{\ordinalSurreal{n}}{\ordinalSurreal{q}}
        =\ordinalSurreal{n+q}$ and
        $\surrealAdd{\ordinalSurreal{p}}{\ordinalSurreal{m}}
        =\ordinalSurreal{p+m}$
        by \obv{natural_surreal_addition_compatibility}.
    Follows by
        \auto{surreal_difference_order_iff,ordinal_surreal_lt_iff}.
\end{proof}

\begin{proposition}\label{surreal_integer_positive_presentation}
    Let $n$ be a natural number.
    Then $\surrealIntegerOf{n}{\emptyset}=\ordinalSurreal{n}$.
\end{proposition}
\begin{proof}
    $n$ is an ordinal by \obv{natural_number_is_ordinal}.
    $\ordinalSurreal{\emptyset}=\emptyset$
        by \obv{empty_ordinal_surreal}.
    Follows by
        \auto{surreal_negation_empty,surreal_add_zero_right,ordinal_surreal_is_surreal}.
\end{proof}

\begin{proposition}\label{surreal_integer_negative_presentation}
    Let $n$ be a natural number.
    Then $\surrealIntegerOf{\emptyset}{n}
        =\surrealNeg{\ordinalSurreal{n}}$.
\end{proposition}
\begin{proof}
    $n$ is an ordinal by \obv{natural_number_is_ordinal}.
    $\ordinalSurreal{\emptyset}=\emptyset$
        by \obv{empty_ordinal_surreal}.
    $\surreal{\surrealNeg{\ordinalSurreal{n}}}$
        by \obv{ordinal_surreal_is_surreal,surreal_negation_is_surreal}.
    Follows by \auto{surreal_add_zero_left}.
\end{proof}

\begin{theorem}\label{surreal_integer_canonical_normal_form}
    Let $x\in\surrealIntegers$.
    Then there exists a natural number $n$ such that
        $x=\ordinalSurreal{n}$ or
        ($n\neq\emptyset$ and
        $x=\surrealNeg{\ordinalSurreal{n}}$).
\end{theorem}
\begin{proof}
    Take $a,b\in\naturals$ such that
        $x=\surrealIntegerOf{a}{b}$
        by \auto{surreal_integer_of_iff}.
    $a$ and $b$ are ordinals by \obv{natural_number_is_ordinal}.
    We have $b\subseteq a$ or $a\subseteq b$
        by \obv{ordinal_subseteq_or_subseteq}.
    \begin{byCase}
        \caseOf{$b\subseteq a$.}
            Take a natural number $n$ such that $a=b+n$
                by \obv{natural_additive_difference}.
            $a+\emptyset=n+b$
                by \obv{num_addition_zero_right,natural_addition_commutative}.
            Hence $\surrealIntegerOf{a}{b}
                =\surrealIntegerOf{n}{\emptyset}$
                by \obv{surreal_integer_of_equality_iff,omega_empty}.
            Hence $x=\ordinalSurreal{n}$
                by \obv{surreal_integer_positive_presentation}.
        \caseOf{not $b\subseteq a$.}
            Hence $a\subseteq b$.
            Take a natural number $n$ such that $b=a+n$
                by \obv{natural_additive_difference}.
            \begin{byCase}
                \caseOf{$n=\emptyset$.}
                    $a=b$ by \auto{num_addition_zero_right}.
                    Hence $b\subseteq a$ by \obv{subseteq_refl}.
                    Contradiction.
                \caseOf{$n\neq\emptyset$.}
                    $a+n=\emptyset+b$
                        by \obv{num_addition_zero_left}.
                    Hence $\surrealIntegerOf{a}{b}
                        =\surrealIntegerOf{\emptyset}{n}$
                        by \obv{surreal_integer_of_equality_iff,omega_empty}.
                    Hence $x=\surrealNeg{\ordinalSurreal{n}}$
                        by \obv{surreal_integer_negative_presentation}.
            \end{byCase}
    \end{byCase}
\end{proof}

\begin{theorem}\label{surreal_integer_positive_normal_form_injective}
    Let $n,m$ be natural numbers.
    Then $\ordinalSurreal{n}=\ordinalSurreal{m}$ iff $n=m$.
\end{theorem}
\begin{proof}
    $n$ and $m$ are ordinals by \obv{natural_number_is_ordinal}.
    Follows by \auto{ordinal_surreal_injective}.
\end{proof}

\begin{theorem}\label{surreal_integer_negative_normal_form_injective}
    Let $n,m$ be natural numbers.
    Then
        $\surrealNeg{\ordinalSurreal{n}}
            =\surrealNeg{\ordinalSurreal{m}}$ iff $n=m$.
\end{theorem}
\begin{proof}
    Follows by
        \auto{surreal_integer_negative_presentation,surreal_integer_of_equality_iff,num_addition_zero_left,omega_empty}.
\end{proof}

\begin{theorem}\label{surreal_integer_normal_form_signs_disjoint}
    Let $n,m$ be natural numbers.
    Suppose $m\neq\emptyset$.
    Then $\ordinalSurreal{n}
        \neq\surrealNeg{\ordinalSurreal{m}}$.
\end{theorem}
\begin{proof}
    $\emptyset$ is a natural number by \obv{omega_empty}.
    Take a natural number $k$ such that $m=\suc{k}$
        by \obv{natural_zero_or_successor}.
    $n+m=\suc{n+k}$
        by \auto{num_addition_successor_right}.
    Hence $n+m\neq\emptyset$ by \obv{suc_neq_emptyset}.
    $\emptyset+\emptyset=\emptyset$
        by \obv{omega_empty,num_addition_zero_left}.
    Hence $\surrealIntegerOf{n}{\emptyset}
        \neq\surrealIntegerOf{\emptyset}{m}$
        by \auto{surreal_integer_of_equality_iff}.
    Follows by
        \obv{surreal_integer_positive_presentation,surreal_integer_negative_presentation}.
\end{proof}

\begin{theorem}\label{surreal_integer_canonical_normal_form_unique}
    Let $n,m$ be natural numbers.
    Suppose
        $(x=\ordinalSurreal{n}
          \lor (n\neq\emptyset\land
            x=\surrealNeg{\ordinalSurreal{n}}))$
        and
        $(x=\ordinalSurreal{m}
          \lor (m\neq\emptyset\land
            x=\surrealNeg{\ordinalSurreal{m}}))$.
    Then
        $n=m\land
        ((x=\ordinalSurreal{n}\land x=\ordinalSurreal{m})
         \lor
         (n\neq\emptyset\land m\neq\emptyset\land
          x=\surrealNeg{\ordinalSurreal{n}}\land
          x=\surrealNeg{\ordinalSurreal{m}}))$.
\end{theorem}
\begin{proof}
    Follows by
        \auto{surreal_integer_positive_normal_form_injective,surreal_integer_negative_normal_form_injective,surreal_integer_normal_form_signs_disjoint}.
\end{proof}

\subsection{Order of normal forms}

\begin{theorem}\label{surreal_integer_positive_normal_form_order}
    Let $n,m$ be natural numbers.
    Then $\surrealLt{\ordinalSurreal{n}}{\ordinalSurreal{m}}$
        iff $n\in m$.
\end{theorem}
\begin{proof}
    Follows by
        \obv{natural_number_is_ordinal,ordinal_surreal_lt_iff}.
\end{proof}

\begin{theorem}\label{surreal_integer_negative_normal_form_order}
    Let $n,m$ be natural numbers.
    Then
        $\surrealLt
            {\surrealNeg{\ordinalSurreal{n}}}
            {\surrealNeg{\ordinalSurreal{m}}}$ iff
        $m\in n$.
\end{theorem}
\begin{proof}
    Follows by
        \auto{natural_number_is_ordinal,ordinal_surreal_is_surreal,surreal_negation_lt_iff,ordinal_surreal_lt_iff}.
\end{proof}

\begin{theorem}\label{surreal_integer_mixed_normal_form_order}
    Let $n,m$ be natural numbers.
    Suppose $n\neq\emptyset$.
    Then
        $\surrealLt
            {\surrealNeg{\ordinalSurreal{n}}}
            {\ordinalSurreal{m}}$ and
        not $\surrealLt
            {\ordinalSurreal{m}}
            {\surrealNeg{\ordinalSurreal{n}}}$.
\end{theorem}
\begin{proof}
    Take a natural number $k$ such that $n=\suc{k}$
        by \obv{natural_zero_or_successor}.
    $m+n=\suc{m+k}$ by \auto{num_addition_successor_right}.
    Hence $m+n\neq\emptyset$ by \obv{suc_neq_emptyset}.
    Hence $\emptyset\in m+n$
        by \obv{num_addition_on_naturals,natural_nonzero_contains_zero}.
    $\emptyset+\emptyset=\emptyset$
        by \obv{omega_empty,num_addition_zero_left}.
    $\emptyset\notin\emptyset$ by \obv{emptyset}.
    $\emptyset$ is a natural number by \obv{omega_empty}.
    Follows by
        \auto{surreal_integer_negative_presentation,surreal_integer_positive_presentation,surreal_integer_of_order_iff}.
\end{proof}

\subsection{Arithmetic and ordered-ring laws}

\begin{proposition}\label{surreal_integer_zero}
    $\surrealZero\in\surrealIntegers$.
\end{proposition}
\begin{proof}
    $\emptyset$ is a natural number by \obv{omega_empty}.
    $\ordinalSurreal{\emptyset}=\emptyset$
        by \obv{empty_ordinal_surreal}.
    Follows by \auto{natural_surreal_is_integer}.
\end{proof}

\begin{proposition}\label{surreal_integer_one}
    $\surrealOne\in\surrealIntegers$.
\end{proposition}
\begin{proof}
    $\suc{\emptyset}$ is a natural number
        by \obv{omega_empty,omega_successor}.
    Follows by \obv{natural_surreal_is_integer}.
\end{proof}

\begin{proposition}\label{surreal_integer_zero_neq_one}
    $\surrealZero\neq\surrealOne$.
\end{proposition}
\begin{proof}
    Follows by
        \obv{surreal_zero_lt_one,surreal_lt_irreflexive,empty_surreal,surreal_one_is_surreal}.
\end{proof}

\begin{theorem}\label{surreal_integer_of_negation}
    Let $n,m$ be natural numbers.
    Then $\surrealNeg{\surrealIntegerOf{n}{m}}
        =\surrealIntegerOf{m}{n}$.
\end{theorem}
\begin{proof}
    $n$ and $m$ are ordinals by \obv{natural_number_is_ordinal}.
    Hence $\surreal{\ordinalSurreal{n}}$ and
        $\surreal{\ordinalSurreal{m}}$
        by \obv{ordinal_surreal_is_surreal}.
    Follows by
        \auto{surreal_negation_additive,surreal_negation_involutive,surreal_add_commutative,surreal_negation_is_surreal}.
\end{proof}

\begin{theorem}\label{surreal_integer_of_addition}
    Let $n,m,p,q$ be natural numbers.
    Then
        $\surrealAdd{\surrealIntegerOf{n}{m}}
            {\surrealIntegerOf{p}{q}}
        =\surrealIntegerOf{n+p}{m+q}$.
\end{theorem}
\begin{proof}
    $n$, $m$, $p$, and $q$ are ordinals
        by \obv{natural_number_is_ordinal}.
    Hence $\surreal{\ordinalSurreal{n}}$ and
        $\surreal{\ordinalSurreal{m}}$ and
        $\surreal{\ordinalSurreal{p}}$ and
        $\surreal{\ordinalSurreal{q}}$
        by \obv{ordinal_surreal_is_surreal}.
    $\surrealAdd{\surrealIntegerOf{n}{m}}
            {\surrealIntegerOf{p}{q}}
        =\surrealAdd
            {\surrealAdd{\ordinalSurreal{n}}{\ordinalSurreal{p}}}
            {\surrealNeg{
                \surrealAdd{\ordinalSurreal{m}}{\ordinalSurreal{q}}}}$
        by
        \auto{surreal_add_associative,surreal_add_commutative,surreal_negation_additive,surreal_negation_is_surreal,surreal_add_is_surreal}.
    Follows by
        \obv{natural_surreal_addition_compatibility}.
\end{proof}

\begin{lemma}\label{surreal_difference_product}
    Suppose $\surreal{a}$ and $\surreal{b}$ and
        $\surreal{c}$ and $\surreal{d}$.
    Then
    $\surrealMul{\surrealAdd{a}{\surrealNeg{b}}}
        {\surrealAdd{c}{\surrealNeg{d}}}
    =\surrealAdd
        {\surrealAdd{\surrealMul{a}{c}}{\surrealMul{b}{d}}}
        {\surrealNeg{
            \surrealAdd{\surrealMul{a}{d}}{\surrealMul{b}{c}}}}$.
\end{lemma}
\begin{proof}
    Let $A=\surrealMul{a}{c}$.
    Let $B=\surrealMul{b}{d}$.
    Let $C=\surrealMul{a}{d}$.
    Let $D=\surrealMul{b}{c}$.
    $\surreal{A}$ and $\surreal{B}$ and
        $\surreal{C}$ and $\surreal{D}$
        by \obv{surreal_mul_is_surreal}.
    $\surrealMul{\surrealAdd{a}{\surrealNeg{b}}}
            {\surrealAdd{c}{\surrealNeg{d}}}
        =\surrealAdd
            {\surrealAdd{A}{\surrealNeg{C}}}
            {\surrealAdd{\surrealNeg{D}}{B}}$
        by
        \auto{surreal_mul_foil,surreal_mul_neg_left,surreal_mul_neg_right,surreal_mul_both_negated,surreal_negation_is_surreal}.
    $\surrealNeg{\surrealAdd{C}{D}}
        =\surrealAdd{\surrealNeg{C}}{\surrealNeg{D}}$
        by \obv{surreal_negation_additive}.
    Follows by
        \auto{surreal_add_associative,surreal_add_commutative,surreal_negation_is_surreal,surreal_add_is_surreal}.
\end{proof}

\begin{theorem}\label{surreal_integer_of_multiplication}
    Let $n,m,p,q$ be natural numbers.
    Then
    $\surrealMul{\surrealIntegerOf{n}{m}}
        {\surrealIntegerOf{p}{q}}
    =\surrealIntegerOf
        {\naturalMul{n}{p}+\naturalMul{m}{q}}
        {\naturalMul{n}{q}+\naturalMul{m}{p}}$.
\end{theorem}
\begin{proof}
    $n$, $m$, $p$, and $q$ are ordinals
        by \obv{natural_number_is_ordinal}.
    Hence $\surreal{\ordinalSurreal{n}}$ and
        $\surreal{\ordinalSurreal{m}}$ and
        $\surreal{\ordinalSurreal{p}}$ and
        $\surreal{\ordinalSurreal{q}}$
        by \obv{ordinal_surreal_is_surreal}.
    Let $r=\naturalMul{n}{p}$.
    Let $s=\naturalMul{m}{q}$.
    Let $t=\naturalMul{n}{q}$.
    Let $u=\naturalMul{m}{p}$.
    $r$, $s$, $t$, and $u$ are natural numbers
        by \auto{natural_multiplication_closed}.
    $\surrealMul{\surrealIntegerOf{n}{m}}
            {\surrealIntegerOf{p}{q}}
        =\surrealAdd
            {\surrealAdd
                {\surrealMul{\ordinalSurreal{n}}{\ordinalSurreal{p}}}
                {\surrealMul{\ordinalSurreal{m}}{\ordinalSurreal{q}}}}
            {\surrealNeg{
                \surrealAdd
                    {\surrealMul{\ordinalSurreal{n}}{\ordinalSurreal{q}}}
                    {\surrealMul{\ordinalSurreal{m}}{\ordinalSurreal{p}}}}}$
        by \obv{surreal_difference_product}.
    Hence
    $\surrealMul{\surrealIntegerOf{n}{m}}
            {\surrealIntegerOf{p}{q}}
        =\surrealAdd
            {\surrealAdd{\ordinalSurreal{r}}{\ordinalSurreal{s}}}
            {\surrealNeg{
                \surrealAdd{\ordinalSurreal{t}}{\ordinalSurreal{u}}}}$
        by \auto{natural_surreal_multiplication_compatibility}.
    Hence
    $\surrealMul{\surrealIntegerOf{n}{m}}
            {\surrealIntegerOf{p}{q}}
        =\surrealAdd{\ordinalSurreal{r+s}}
            {\surrealNeg{\ordinalSurreal{t+u}}}$
        by \obv{natural_surreal_addition_compatibility}.
\end{proof}

\begin{theorem}\label{surreal_integer_subtraction_closed}
    Let $x,y\in\surrealIntegers$.
    Then $\surrealAdd{x}{\surrealNeg{y}}\in\surrealIntegers$.
\end{theorem}
\begin{proof}
    $\surrealNeg{y}\in\surrealIntegers$
        by \obv{surreal_integer_negation_closed}.
    Follows by \obv{surreal_integer_addition_closed}.
\end{proof}

\begin{theorem}\label{surreal_integer_additive_laws}
    Let $x,y,z\in\surrealIntegers$.
    Then
        $\surrealAdd{x}{\surrealZero}=x$ and
        $\surrealAdd{x}{\surrealNeg{x}}=\surrealZero$ and
        $\surrealAdd{x}{y}=\surrealAdd{y}{x}$ and
        $\surrealAdd{x}{\surrealAdd{y}{z}}
            =\surrealAdd{\surrealAdd{x}{y}}{z}$.
\end{theorem}
\begin{proof}
    $\surreal{x}$ and $\surreal{y}$ and $\surreal{z}$
        by \obv{surreal_integer_is_surreal}.
    Follows by
        \obv{surreal_add_zero_right,surreal_add_inverse_right,surreal_add_commutative,surreal_add_associative}.
\end{proof}

\begin{theorem}\label{surreal_integer_multiplicative_laws}
    Let $x,y,z\in\surrealIntegers$.
    Then
        $\surrealMul{x}{\surrealOne}=x$ and
        $\surrealMul{x}{y}=\surrealMul{y}{x}$ and
        $\surrealMul{x}{\surrealMul{y}{z}}
            =\surrealMul{\surrealMul{x}{y}}{z}$ and
        $\surrealMul{x}{\surrealAdd{y}{z}}
            =\surrealAdd{\surrealMul{x}{y}}{\surrealMul{x}{z}}$.
\end{theorem}
\begin{proof}
    $\surreal{x}$ and $\surreal{y}$ and $\surreal{z}$
        by \obv{surreal_integer_is_surreal}.
    Follows by
        \obv{surreal_mul_one_right,surreal_mul_commutative,surreal_mul_associative,surreal_mul_distributive_left}.
\end{proof}

\begin{theorem}\label{surreal_integer_no_zero_divisors}
    Let $x,y\in\surrealIntegers$.
    Suppose $\surrealMul{x}{y}=\surrealZero$.
    Then $x=\surrealZero$ or $y=\surrealZero$.
\end{theorem}
\begin{proof}
    $\surreal{x}$ and $\surreal{y}$
        by \obv{surreal_integer_is_surreal}.
    \begin{byCase}
        \caseOf{$x=\surrealZero$.}
            Follows.
        \caseOf{$x\neq\surrealZero$.}
            $\surrealMul{x}{\surrealZero}=\surrealZero$
                by \obv{surreal_mul_zero_right}.
            Hence $y=\surrealZero$
                by \obv{empty_surreal,surreal_mul_nonzero_cancel_left}.
    \end{byCase}
\end{proof}

\begin{theorem}\label{surreal_integer_order_trichotomy}
    Let $x,y\in\surrealIntegers$.
    Then $\surrealLt{x}{y}$ or $x=y$ or $\surrealLt{y}{x}$.
\end{theorem}
\begin{proof}
    Follows by
        \obv{surreal_integer_is_surreal,surreal_lt_trichotomy}.
\end{proof}

\begin{theorem}\label{surreal_integer_addition_order_compatible}
    Let $x,y,z\in\surrealIntegers$.
    Suppose $\surrealLt{x}{y}$.
    Then $\surrealLt{\surrealAdd{x}{z}}{\surrealAdd{y}{z}}$.
\end{theorem}
\begin{proof}
    Follows by
        \obv{surreal_integer_is_surreal,surreal_add_lt_left}.
\end{proof}

\begin{theorem}\label{surreal_integer_positive_multiplication_order_compatible}
    Let $x,y,z\in\surrealIntegers$.
    Suppose $\surrealLt{\surrealZero}{z}$ and $\surrealLt{x}{y}$.
    Then $\surrealLt{\surrealMul{x}{z}}{\surrealMul{y}{z}}$.
\end{theorem}
\begin{proof}
    Follows by
        \obv{surreal_integer_is_surreal,surreal_pos_mul_lt_right}.
\end{proof}

\subsection{Discrete order and induction}

\begin{definition}\label{surreal_integer_successor}
    $\surrealIntegerSuccessor{x}=\surrealAdd{x}{\surrealOne}$.
\end{definition}

\begin{definition}\label{surreal_integer_predecessor}
    $\surrealIntegerPredecessor{x}=
        \surrealAdd{x}{\surrealNeg{\surrealOne}}$.
\end{definition}

\begin{proposition}\label{surreal_integer_successor_closed}
    Let $x\in\surrealIntegers$.
    Then $\surrealIntegerSuccessor{x}\in\surrealIntegers$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{surreal_integer_successor,surreal_integer_one,surreal_integer_addition_closed}.
\end{proof}

\begin{proposition}\label{surreal_integer_predecessor_closed}
    Let $x\in\surrealIntegers$.
    Then $\surrealIntegerPredecessor{x}\in\surrealIntegers$.
\end{proposition}
\begin{proof}
    $\surrealNeg{\surrealOne}\in\surrealIntegers$
        by \obv{surreal_integer_one,surreal_integer_negation_closed}.
    Follows by
        \auto{surreal_integer_predecessor,surreal_integer_addition_closed}.
\end{proof}

\begin{theorem}\label{surreal_integer_successor_predecessor}
    Let $x\in\surrealIntegers$.
    Then
        $\surrealIntegerPredecessor{\surrealIntegerSuccessor{x}}=x$
        and
        $\surrealIntegerSuccessor{\surrealIntegerPredecessor{x}}=x$.
\end{theorem}
\begin{proof}
    $\surreal{x}$ and $\surreal{\surrealOne}$

\begin{theorem}\label{surreal_integer_lt_successor}
    Let $x\in\surrealIntegers$.
    Then $\surrealLt{x}{\surrealIntegerSuccessor{x}}$.
\end{theorem}
\begin{proof}
    $\surreal{x}$ and $\surreal{\surrealZero}$ and
        $\surreal{\surrealOne}$
        by \obv{surreal_integer_is_surreal,empty_surreal,surreal_one_is_surreal}.
    $\surrealLt{\surrealAdd{x}{\surrealZero}}
        {\surrealAdd{x}{\surrealOne}}$
        by \obv{surreal_zero_lt_one,surreal_add_lt_right}.
    Follows by
        \obv{surreal_integer_successor,surreal_add_zero_right}.
\end{proof}

\begin{proposition}\label{surreal_integer_successor_presentation}
    Let $n,m$ be natural numbers.
    Then
        $\surrealIntegerSuccessor{\surrealIntegerOf{n}{m}}
        =\surrealIntegerOf{\suc{n}}{m}$.
\end{proposition}
\begin{proof}
    $n$ and $m$ are ordinals by \obv{natural_number_is_ordinal}.
    $\surreal{\ordinalSurreal{n}}$ and
        $\surreal{\ordinalSurreal{m}}$ and
        $\surreal{\surrealOne}$
        by \obv{ordinal_surreal_is_surreal,surreal_one_is_surreal}.
    $\ordinalSurreal{\suc{n}}
        =\surrealAdd{\ordinalSurreal{n}}{\surrealOne}$
        by \obv{ordinal_surreal_successor_addition}.
    Follows by
        \auto{surreal_integer_successor,surreal_add_associative,surreal_add_commutative,surreal_negation_is_surreal,surreal_add_is_surreal}.
\end{proof}

\begin{theorem}\label{surreal_integer_no_between_successors}
    Let $x,y\in\surrealIntegers$.
    Suppose $\surrealLt{x}{y}$.
    Then not
        $\surrealLt{y}{\surrealIntegerSuccessor{x}}$.
\end{theorem}
\begin{proof}
    Suppose not.
    Take $n,m\in\naturals$ such that
        $x=\surrealIntegerOf{n}{m}$
        by \auto{surreal_integer_of_iff}.
    Take $p,q\in\naturals$ such that
        $y=\surrealIntegerOf{p}{q}$
        by \auto{surreal_integer_of_iff}.
    $\suc{n}$ is a natural number by \obv{omega_successor}.
    $n+q\in p+m$ by \auto{surreal_integer_of_order_iff}.
    $p+m\in\suc{n}+q$
        by \auto{surreal_integer_successor_presentation,surreal_integer_of_order_iff}.
    $\suc{n}+q=\suc{n+q}$
        by \obv{num_addition_successor_left}.
    Hence $p+m=n+q$ or $p+m\in n+q$ by \auto{suc_iff}.
    Contradiction by \obv{in_irrefl,in_asymmetric}.
\end{proof}

\begin{theorem}\label{surreal_integer_successor_is_least_greater}
    Let $x,y\in\surrealIntegers$.
    Suppose $\surrealLt{x}{y}$.
    Then $\surrealLe{\surrealIntegerSuccessor{x}}{y}$.
\end{theorem}
\begin{proof}
    $\surrealIntegerSuccessor{x}\in\surrealIntegers$
        by \obv{surreal_integer_successor_closed}.
    We have
        $\surrealLt{\surrealIntegerSuccessor{x}}{y}$ or
        $\surrealIntegerSuccessor{x}=y$ or
        $\surrealLt{y}{\surrealIntegerSuccessor{x}}$
        by \obv{surreal_integer_order_trichotomy}.
    Not $\surrealLt{y}{\surrealIntegerSuccessor{x}}$
        by \obv{surreal_integer_no_between_successors}.
    Follows by \obv{surreal_le}.
\end{proof}

\begin{theorem}\label{surreal_integer_predecessor_lt}
    Let $x\in\surrealIntegers$.
    Then $\surrealLt{\surrealIntegerPredecessor{x}}{x}$.
\end{theorem}
\begin{proof}
    $\surrealIntegerPredecessor{x}\in\surrealIntegers$
        by \obv{surreal_integer_predecessor_closed}.
    $\surrealLt
        {\surrealIntegerPredecessor{x}}
        {\surrealIntegerSuccessor{\surrealIntegerPredecessor{x}}}$
        by \obv{surreal_integer_lt_successor}.
    Follows by \auto{surreal_integer_successor_predecessor}.
\end{proof}

\begin{theorem}\label{surreal_integer_predecessor_is_greatest_smaller}
    Let $x,y\in\surrealIntegers$.
    Suppose $\surrealLt{y}{x}$.
    Then $\surrealLe{y}{\surrealIntegerPredecessor{x}}$.
\end{theorem}
\begin{proof}
    $\surrealIntegerPredecessor{x}\in\surrealIntegers$
        by \obv{surreal_integer_predecessor_closed}.
    Show that not
        $\surrealLt{\surrealIntegerPredecessor{x}}{y}$.
    \begin{subproof}
        Suppose not.
        Then not
            $\surrealLt
                {y}
                {\surrealIntegerSuccessor{\surrealIntegerPredecessor{x}}}$
            by \obv{surreal_integer_no_between_successors}.
        $\surrealIntegerSuccessor{\surrealIntegerPredecessor{x}}=x$
            by \auto{surreal_integer_successor_predecessor}.
        Contradiction.
    \end{subproof}
    We have
        $\surrealLt{y}{\surrealIntegerPredecessor{x}}$ or
        $y=\surrealIntegerPredecessor{x}$ or
        $\surrealLt{\surrealIntegerPredecessor{x}}{y}$
        by \obv{surreal_integer_order_trichotomy}.
    Follows by \obv{surreal_le}.
\end{proof}

\begin{proposition}\label{surreal_integer_natural_successor}
    Let $n$ be a natural number.
    Then
        $\surrealIntegerSuccessor{\ordinalSurreal{n}}
        =\ordinalSurreal{\suc{n}}$.
\end{proposition}
\begin{proof}
    $\emptyset$ and $\suc{n}$ are natural numbers
        by \obv{omega_empty,omega_successor}.
    $\ordinalSurreal{n}=\surrealIntegerOf{n}{\emptyset}$
        by \obv{surreal_integer_positive_presentation}.
    Follows by
        \auto{surreal_integer_successor_presentation,surreal_integer_positive_presentation}.
\end{proof}

\begin{proposition}\label{surreal_integer_negative_predecessor}
    Let $n$ be a natural number.
    Then
        $\surrealIntegerPredecessor{
            \surrealNeg{\ordinalSurreal{n}}}
        =\surrealNeg{\ordinalSurreal{\suc{n}}}$.
\end{proposition}
\begin{proof}
    $n$ and $\suc{n}$ are ordinals
        by \obv{natural_number_is_ordinal,omega_successor}.
    $\surreal{\ordinalSurreal{n}}$ and
        $\surreal{\surrealOne}$
        by \obv{ordinal_surreal_is_surreal,surreal_one_is_surreal}.
    $\ordinalSurreal{\suc{n}}
        =\surrealAdd{\ordinalSurreal{n}}{\surrealOne}$
        by \obv{ordinal_surreal_successor_addition}.
    $\surrealNeg{\ordinalSurreal{\suc{n}}}
        =\surrealAdd
            {\surrealNeg{\ordinalSurreal{n}}}
            {\surrealNeg{\surrealOne}}$
        by \auto{surreal_negation_additive}.
    Follows by \obv{surreal_integer_predecessor}.
\end{proof}

\begin{theorem}\label{surreal_integer_induction}
    $\forall P:\Set\to\Prop.
        (\Apply{P}{\surrealZero}
        \land
        (\forall z:\Set.
            (z\in\surrealIntegers\land\Apply{P}{z})
            \implies\Apply{P}{\surrealIntegerSuccessor{z}})
        \land
        (\forall z:\Set.
            (z\in\surrealIntegers\land\Apply{P}{z})
            \implies\Apply{P}{\surrealIntegerPredecessor{z}}))
        \implies
        (\forall z:\Set.
            z\in\surrealIntegers\implies\Apply{P}{z})$.
\end{theorem}
\begin{proof}
    Fix $P$.
    Assume $(\Apply{P}{\surrealZero}
        \land
        (\forall z:\Set.
            (z\in\surrealIntegers\land\Apply{P}{z})
            \implies\Apply{P}{\surrealIntegerSuccessor{z}})
        \land
        (\forall z:\Set.
            (z\in\surrealIntegers\land\Apply{P}{z})
            \implies\Apply{P}{\surrealIntegerPredecessor{z}}))$.

    Let $Q:\Set\to\Prop=
        \fun{k:\Set}{\Apply{P}{\ordinalSurreal{k}}}$.
    $\Apply{Q}{\emptyset}$
        by \obv{empty_ordinal_surreal}.
    Show that $\forall k:\Set. k\in\naturals\implies
        (\Apply{Q}{k}\implies\Apply{Q}{\suc{k}})$.
    \begin{subproof}
        Fix $k$.
        Assume $k\in\naturals$.
        Assume $\Apply{Q}{k}$.
        $\ordinalSurreal{k}\in\surrealIntegers$
            by \obv{natural_surreal_is_integer}.
        Hence $\Apply{P}{
            \surrealIntegerSuccessor{\ordinalSurreal{k}}}$.
        Follows by \obv{surreal_integer_natural_successor}.
    \end{subproof}
    Let $C=\sep{\naturals}{Q}$.
    $\emptyset\in C$
        by \auto{logical_separation_iff,omega_empty}.
    For every $k\in C$ we have $\suc{k}\in C$
        by \auto{logical_separation_iff,omega_successor}.
    Hence $C$ is an inductive set by \obv{inductive_set}.
    Hence every natural number is an element of $C$
        by \obv{omega_minimal}.

    Let $R:\Set\to\Prop=
        \fun{k:\Set}{
            \Apply{P}{\surrealNeg{\ordinalSurreal{k}}}}$.
    $\Apply{R}{\emptyset}$
        by \obv{empty_ordinal_surreal,surreal_negation_empty}.
    Show that $\forall k:\Set. k\in\naturals\implies
        (\Apply{R}{k}\implies\Apply{R}{\suc{k}})$.
    \begin{subproof}
        Fix $k$.
        Assume $k\in\naturals$.
        Assume $\Apply{R}{k}$.
        $\surrealNeg{\ordinalSurreal{k}}\in\surrealIntegers$
            by \obv{natural_surreal_is_integer,surreal_integer_negation_closed}.
        Hence $\Apply{P}{
            \surrealIntegerPredecessor{
                \surrealNeg{\ordinalSurreal{k}}}}$.
        Follows by \obv{surreal_integer_negative_predecessor}.
    \end{subproof}
    Let $D=\sep{\naturals}{R}$.
    $\emptyset\in D$
        by \auto{logical_separation_iff,omega_empty}.
    For every $k\in D$ we have $\suc{k}\in D$
        by \auto{logical_separation_iff,omega_successor}.
    Hence $D$ is an inductive set by \obv{inductive_set}.
    Hence every natural number is an element of $D$
        by \obv{omega_minimal}.

    Fix $x$.
    Assume $x\in\surrealIntegers$.
    Take a natural number $n$ such that
        $x=\ordinalSurreal{n}$ or
        ($n\neq\emptyset$ and
        $x=\surrealNeg{\ordinalSurreal{n}}$)
        by \obv{surreal_integer_canonical_normal_form}.
    \begin{byCase}
        \caseOf{$x=\ordinalSurreal{n}$.}
            $n\in C$.
            Hence $\Apply{Q}{n}$
                by \auto{logical_separation_iff}.
        \caseOf{$x\neq\ordinalSurreal{n}$.}
            Hence $x=\surrealNeg{\ordinalSurreal{n}}$.
            $n\in D$.
            Hence $\Apply{R}{n}$
                by \auto{logical_separation_iff}.
    \end{byCase}
\end{proof}
