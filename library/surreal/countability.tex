\section{Countability of finite-birthday surreals}

\import{surreal/finite.tex}
\import{cardinal/countability.tex}

\begin{theorem}\label{finite_birthday_surreals_are_countable}
    $\surrealStage{\naturals}$ is countable.
\end{theorem}
\begin{proof}
    Let $F=\{\surrealStage{n}\mid n\in\naturals\}$.
    $\naturals$ is countable by \obv{naturals_is_countable}.
    $\cardLe{F}{\naturals}$ by \auto{countable_replacement}.
    Hence $F$ is countable by \obv{cardinal_le,at_most_countable}.
    Show that every element of $F$ is countable.
    \begin{subproof}
        Fix $A\in F$.
        Take $n\in\naturals$ such that $A=\surrealStage{n}$.
        $A$ is finite by \auto{finite_surreal_stage}.
        Follows by \obv{finite_is_countable}.
    \end{subproof}
    Hence $\unions{F}$ is countable by \obv{countable_union_of_countable_sets}.
    Show that every element of $\surrealStage{\naturals}$ is an element of $\unions{F}$.
    \begin{subproof}
        Fix $x\in\surrealStage{\naturals}$.
        $\surreal{x}$ and $\birthday{x}\in\naturals$ by
            \obv{surreal_stage_characterization,omega_is_ordinal}.
        $\suc{\birthday{x}}\in\naturals$ by \obv{omega_successor}.
        Hence $\surrealStage{\suc{\birthday{x}}}\in F$.
        $x\in\surrealStage{\suc{\birthday{x}}}$ by \obv{surreal_in_successor_stage}.
        Follows by \auto{unions_iff}.
    \end{subproof}
    $\surrealStage{\naturals}\subseteq\unions{F}$ by \obv{subseteq}.
    Follows by \obv{subseteq_countable_is_countable}.
\end{proof}
