\import{surreal/additive.tex}

\section{Surreal multiplication}

For a recursively available binary operation $g$, the Conway multiplication
option associated with a pair $p=(u,v)$ is
\[
    g(u,y)+g(x,v)-g(u,v).
\]
Keeping this term and the four option quadrants explicit makes the locality
of the recursion visible to the type checker.

\subsection{Recursive construction}

\begin{definition}\label{surreal_mul_term_typed}
    $\surrealMulTermOp:
        (\Set\to\Set\to\Set)\to\Set\to\Set\to\Set\to\Set
        =\fun{g:\Set\to\Set\to\Set}{
            \fun{x,y,p:\Set}{
                \surrealAdd
                    {\surrealAdd
                        {\Apply{g}{\fst{p},y}}
                        {\Apply{g}{x,\snd{p}}}}
                    {\surrealNeg{\Apply{g}{\fst{p},\snd{p}}}}}}$.
\end{definition}

\begin{definition}\label{surreal_mul_step_left}
    $\surrealMulStepLeftOp:
        (\Set\to\Set\to\Set)\to\Set\to\Set\to\Set
        =\fun{g:\Set\to\Set\to\Set}{
            \fun{x,y:\Set}{
                \{\Apply{\surrealMulTermOp}{g,x,y,p}\mid
                    p\in\surrealLeft{x}\times\surrealLeft{y}\}
                \union
                \{\Apply{\surrealMulTermOp}{g,x,y,p}\mid
                    p\in\surrealRight{x}\times\surrealRight{y}\}}}$.
\end{definition}

\begin{definition}\label{surreal_mul_step_right}
    $\surrealMulStepRightOp:
        (\Set\to\Set\to\Set)\to\Set\to\Set\to\Set
        =\fun{g:\Set\to\Set\to\Set}{
            \fun{x,y:\Set}{
                \{\Apply{\surrealMulTermOp}{g,x,y,p}\mid
                    p\in\surrealLeft{x}\times\surrealRight{y}\}
                \union
                \{\Apply{\surrealMulTermOp}{g,x,y,p}\mid
                    p\in\surrealRight{x}\times\surrealLeft{y}\}}}$.
\end{definition}

\begin{definition}\label{surreal_mul_step}
    $\surrealMulStepOp:
        \Set\to\Set\to(\Set\to\Set\to\Set)\to\Set
        =\fun{x,y:\Set}{
            \fun{g:\Set\to\Set\to\Set}{
                \surrealCut
                    {\Apply{\surrealMulStepLeftOp}{g,x,y}}
                    {\Apply{\surrealMulStepRightOp}{g,x,y}}}}$.
\end{definition}

\begin{lemma}\label{surreal_mul_term_functional}
    Suppose $\surreal{x}$ and $\surreal{y}$.
    Suppose
        $p\in(\surrealLeft{x}\union\surrealRight{x})
            \times(\surrealLeft{y}\union\surrealRight{y})$.
    Then
        $\forall g,h:\Set\to\Set\to\Set.
            ((\forall u:\Set.
                (\surreal{u}\land\birthday{u}\in\birthday{x})
                \implies
                (\forall v:\Set.\ \surreal{v}\implies
                    \Apply{g}{u,v}=\Apply{h}{u,v}))
             \land
             (\forall v:\Set.
                (\surreal{v}\land\birthday{v}\in\birthday{y})
                \implies
                \Apply{g}{x,v}=\Apply{h}{x,v}))
            \implies
            \Apply{\surrealMulTermOp}{g,x,y,p}
                =\Apply{\surrealMulTermOp}{h,x,y,p}$.
\end{lemma}
\begin{proof}
    Fix $g,h$.
    Assume
        $(\forall u:\Set.
            (\surreal{u}\land\birthday{u}\in\birthday{x})
            \implies
            (\forall v:\Set.\ \surreal{v}\implies
                \Apply{g}{u,v}=\Apply{h}{u,v}))
         \land
         (\forall v:\Set.
            (\surreal{v}\land\birthday{v}\in\birthday{y})
            \implies
            \Apply{g}{x,v}=\Apply{h}{x,v})$.
    $\fst{p}\in\surrealLeft{x}\union\surrealRight{x}$ and
        $\snd{p}\in\surrealLeft{y}\union\surrealRight{y}$
        by \obv{fst_type,snd_type}.
    $\surreal{\fst{p}}$ and
        $\birthday{\fst{p}}\in\birthday{x}$
        by \auto{union,surreal_left_elim,surreal_right_elim}.
    $\surreal{\snd{p}}$ and
        $\birthday{\snd{p}}\in\birthday{y}$
        by \auto{union,surreal_left_elim,surreal_right_elim}.
    $\Apply{g}{\fst{p},y}
        =\Apply{h}{\fst{p},y}$ and
        $\Apply{g}{x,\snd{p}}
        =\Apply{h}{x,\snd{p}}$ and
        $\Apply{g}{\fst{p},\snd{p}}
        =\Apply{h}{\fst{p},\snd{p}}$.
    Follows by \auto{surreal_mul_term_typed}.
\end{proof}

\begin{lemma}\label{surreal_mul_quadrant_functional}
    Suppose $\surreal{x}$ and $\surreal{y}$.
    Suppose
        $A\subseteq\surrealLeft{x}\union\surrealRight{x}$ and
        $B\subseteq\surrealLeft{y}\union\surrealRight{y}$.
    Then
        $\forall g,h:\Set\to\Set\to\Set.
            ((\forall u:\Set.
                (\surreal{u}\land\birthday{u}\in\birthday{x})
                \implies
                (\forall v:\Set.\ \surreal{v}\implies
                    \Apply{g}{u,v}=\Apply{h}{u,v}))
             \land
             (\forall v:\Set.
                (\surreal{v}\land\birthday{v}\in\birthday{y})
                \implies
                \Apply{g}{x,v}=\Apply{h}{x,v}))
            \implies
            \{\Apply{\surrealMulTermOp}{g,x,y,p}
                \mid p\in A\times B\}
            =\{\Apply{\surrealMulTermOp}{h,x,y,p}
                \mid p\in A\times B\}$.
\end{lemma}
\begin{proof}
    Fix $g,h$.
    Assume
        $(\forall u:\Set.
            (\surreal{u}\land\birthday{u}\in\birthday{x})
            \implies
            (\forall v:\Set.\ \surreal{v}\implies
                \Apply{g}{u,v}=\Apply{h}{u,v}))
         \land
         (\forall v:\Set.
            (\surreal{v}\land\birthday{v}\in\birthday{y})
            \implies
            \Apply{g}{x,v}=\Apply{h}{x,v})$.
    Show that for every $p\in A\times B$ we have
        $\Apply{\surrealMulTermOp}{g,x,y,p}
            =\Apply{\surrealMulTermOp}{h,x,y,p}$.
    \begin{subproof}
        Fix $p\in A\times B$.
        $\fst{p}\in A$ and $\snd{p}\in B$
            by \obv{fst_type,snd_type}.
        $A\times B\subseteq
            (\surrealLeft{x}\union\surrealRight{x})
            \times(\surrealLeft{y}\union\surrealRight{y})$
            by \auto{subseteq,times}.
        Hence $p\in(\surrealLeft{x}\union\surrealRight{x})
            \times(\surrealLeft{y}\union\surrealRight{y})$
            by \obv{subseteq}.
        Follows by \obv{surreal_mul_term_functional}.
    \end{subproof}
    Follows by \auto{setext}.
\end{proof}

\begin{proposition}\label{surreal_mul_step_functional}
    $\Apply{\surrealBinaryRecFunctionalOp}{\surrealMulStepOp}$.
\end{proposition}
\begin{proof}
    Show that $\forall x,y:\Set.
        (\surreal{x}\land\surreal{y})\implies
        (\forall g,h:\Set\to\Set\to\Set.
            ((\forall u:\Set.
                (\surreal{u}\land\birthday{u}\in\birthday{x})
                \implies
                (\forall v:\Set.\ \surreal{v}\implies
                    \Apply{g}{u,v}
                    =\Apply{h}{u,v}))
             \land
             (\forall v:\Set.
                (\surreal{v}\land\birthday{v}\in\birthday{y})
                \implies
                \Apply{g}{x,v}
                =\Apply{h}{x,v}))
            \implies
            \Apply{\surrealMulStepOp}{x,y,g}
            =\Apply{\surrealMulStepOp}{x,y,h})$.
    \begin{subproof}
        Fix $x,y$.
        Assume $\surreal{x}\land\surreal{y}$.
        Fix $g,h$.
        Assume
            $(\forall u:\Set.
                (\surreal{u}\land\birthday{u}\in\birthday{x})
                \implies
                (\forall v:\Set.\ \surreal{v}\implies
                    \Apply{g}{u,v}
                    =\Apply{h}{u,v}))
             \land
             (\forall v:\Set.
                (\surreal{v}\land\birthday{v}\in\birthday{y})
                \implies
                \Apply{g}{x,v}
                =\Apply{h}{x,v})$.
        $\surrealLeft{x},\surrealRight{x}\subseteq
            \surrealLeft{x}\union\surrealRight{x}$ and
            $\surrealLeft{y},\surrealRight{y}\subseteq
            \surrealLeft{y}\union\surrealRight{y}$
            by \auto{subseteq,union}.
        $\{\Apply{\surrealMulTermOp}{g,x,y,p}\mid
                p\in\surrealLeft{x}\times\surrealLeft{y}\}
            =\{\Apply{\surrealMulTermOp}{h,x,y,p}\mid
                p\in\surrealLeft{x}\times\surrealLeft{y}\}$
            by \obv{surreal_mul_quadrant_functional}.
        $\{\Apply{\surrealMulTermOp}{g,x,y,p}\mid
                p\in\surrealRight{x}\times\surrealRight{y}\}
            =\{\Apply{\surrealMulTermOp}{h,x,y,p}\mid
                p\in\surrealRight{x}\times\surrealRight{y}\}$
            by \obv{surreal_mul_quadrant_functional}.
        $\{\Apply{\surrealMulTermOp}{g,x,y,p}\mid
                p\in\surrealLeft{x}\times\surrealRight{y}\}
            =\{\Apply{\surrealMulTermOp}{h,x,y,p}\mid
                p\in\surrealLeft{x}\times\surrealRight{y}\}$
            by \obv{surreal_mul_quadrant_functional}.
        $\{\Apply{\surrealMulTermOp}{g,x,y,p}\mid
                p\in\surrealRight{x}\times\surrealLeft{y}\}
            =\{\Apply{\surrealMulTermOp}{h,x,y,p}\mid
                p\in\surrealRight{x}\times\surrealLeft{y}\}$
            by \obv{surreal_mul_quadrant_functional}.
        Follows by
            \auto{surreal_mul_step,surreal_mul_step_left,surreal_mul_step_right}.
    \end{subproof}
    Follows by \obv{surreal_binary_recursion_functional}.
\end{proof}

\begin{definition}\label{surreal_mul_typed}
    $\surrealMulOp:\Set\to\Set\to\Set
        =\Apply{\surrealBinaryRecOp}{\surrealMulStepOp}$.
\end{definition}

\begin{definition}\label{surreal_mul}
    $\surrealMul{x}{y}=\Apply{\surrealMulOp}{x,y}$.
\end{definition}

\begin{definition}\label{surreal_mul_option}
    $\surrealMulOption{x}{y}{p}=
        \Apply{\surrealMulTermOp}{\surrealMulOp,x,y,p}$.
\end{definition}

\begin{definition}\label{surreal_mul_ll}
    $\surrealMulLL{x}{y}=
        \{\surrealMulOption{x}{y}{p}\mid
            p\in\surrealLeft{x}\times\surrealLeft{y}\}$.
\end{definition}

\begin{definition}\label{surreal_mul_rr}
    $\surrealMulRR{x}{y}=
        \{\surrealMulOption{x}{y}{p}\mid
            p\in\surrealRight{x}\times\surrealRight{y}\}$.
\end{definition}

\begin{definition}\label{surreal_mul_lr}
    $\surrealMulLR{x}{y}=
        \{\surrealMulOption{x}{y}{p}\mid
            p\in\surrealLeft{x}\times\surrealRight{y}\}$.
\end{definition}

\begin{definition}\label{surreal_mul_rl}
    $\surrealMulRL{x}{y}=
        \{\surrealMulOption{x}{y}{p}\mid
            p\in\surrealRight{x}\times\surrealLeft{y}\}$.
\end{definition}

\begin{definition}\label{surreal_mul_left}
    $\surrealMulLeft{x}{y}=
        \Apply{\surrealMulStepLeftOp}{\surrealMulOp,x,y}$.
\end{definition}

\begin{definition}\label{surreal_mul_right}
    $\surrealMulRight{x}{y}=
        \Apply{\surrealMulStepRightOp}{\surrealMulOp,x,y}$.
\end{definition}

\begin{theorem}\label{surreal_mul_equation}
    Suppose $\surreal{x}$ and $\surreal{y}$.
    Then $\surrealMul{x}{y}=
        \surrealCut{\surrealMulLeft{x}{y}}{\surrealMulRight{x}{y}}$.
\end{theorem}
\begin{proof}
    $\Apply{\surrealMulOp}{x,y}
        =\Apply{\surrealMulStepOp}
            {x,y,\surrealMulOp}$
        by
        \auto{surreal_mul_typed,surreal_binary_recursion_equation,surreal_mul_step_functional}.
    $\Apply{\surrealMulStepOp}{x,y,\surrealMulOp}
        =\surrealCut{\surrealMulLeft{x}{y}}{\surrealMulRight{x}{y}}$
        by \auto{surreal_mul_step,surreal_mul_left,surreal_mul_right}.
    Hence $\Apply{\surrealMulOp}{x,y}
        =\surrealCut{\surrealMulLeft{x}{y}}{\surrealMulRight{x}{y}}$.
    Follows by \obv{surreal_mul}.
\end{proof}

\begin{proposition}\label{surreal_mul_option_pair}
    $\surrealMulOption{x}{y}{(u,v)}=
        \surrealAdd
            {\surrealAdd{\surrealMul{u}{y}}{\surrealMul{x}{v}}}
            {\surrealNeg{\surrealMul{u}{v}}}$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{surreal_mul_option,surreal_mul_term_typed,surreal_mul,fst_eq,snd_eq}.
\end{proof}

\begin{proposition}\label{surreal_mul_left_iff}
    $a\in\surrealMulLeft{x}{y}$ iff
        (there exists $p\in\surrealLeft{x}\times\surrealLeft{y}$
            such that $a=\surrealMulOption{x}{y}{p}$) or
        (there exists $p\in\surrealRight{x}\times\surrealRight{y}$
            such that $a=\surrealMulOption{x}{y}{p}$).
\end{proposition}
\begin{proof}
    $a\in\surrealMulLeft{x}{y}$ iff
        (there exists $p\in\surrealLeft{x}\times\surrealLeft{y}$
            such that
            $a=\Apply{\surrealMulTermOp}{\surrealMulOp,x,y,p}$) or
        (there exists $p\in\surrealRight{x}\times\surrealRight{y}$
            such that
            $a=\Apply{\surrealMulTermOp}{\surrealMulOp,x,y,p}$)
        by \auto{surreal_mul_left,surreal_mul_step_left,union}.
    Follows by \auto{surreal_mul_option}.
\end{proof}

\begin{proposition}\label{surreal_mul_right_iff}
    $a\in\surrealMulRight{x}{y}$ iff
        (there exists $p\in\surrealLeft{x}\times\surrealRight{y}$
            such that $a=\surrealMulOption{x}{y}{p}$) or
        (there exists $p\in\surrealRight{x}\times\surrealLeft{y}$
            such that $a=\surrealMulOption{x}{y}{p}$).
\end{proposition}
\begin{proof}
    $a\in\surrealMulRight{x}{y}$ iff
        (there exists $p\in\surrealLeft{x}\times\surrealRight{y}$
            such that
            $a=\Apply{\surrealMulTermOp}{\surrealMulOp,x,y,p}$) or
        (there exists $p\in\surrealRight{x}\times\surrealLeft{y}$
            such that
            $a=\Apply{\surrealMulTermOp}{\surrealMulOp,x,y,p}$)
        by \auto{surreal_mul_right,surreal_mul_step_right,union}.
    Follows by \auto{surreal_mul_option}.
\end{proof}

\begin{proposition}\label{surreal_mul_left_pair_iff}
    $a\in\surrealMulLeft{x}{y}$ iff there exist $u,v$ such that
        $((u\in\surrealLeft{x}\land v\in\surrealLeft{y})
          \lor(u\in\surrealRight{x}\land v\in\surrealRight{y}))
         \land a=\surrealMulOption{x}{y}{(u,v)}$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{surreal_mul_left_iff,times_elem_is_tuple,times_tuple_intro}.
\end{proof}

\begin{proposition}\label{surreal_mul_right_pair_iff}
    $a\in\surrealMulRight{x}{y}$ iff there exist $u,v$ such that
        $((u\in\surrealLeft{x}\land v\in\surrealRight{y})
          \lor(u\in\surrealRight{x}\land v\in\surrealLeft{y}))
         \land a=\surrealMulOption{x}{y}{(u,v)}$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{surreal_mul_right_iff,times_elem_is_tuple,times_tuple_intro}.
\end{proof}

\begin{definition}\label{surreal_mul_property}
    $\surrealMulPropertyOp:\Set\to\Set\to\Prop
        =\fun{x,y:\Set}{
            \surreal{\surrealMul{x}{y}}
            \land
            \surrealCutPair{\surrealMulLeft{x}{y}}
                {\surrealMulRight{x}{y}}
            \land
            (\forall u,v:\Set.
                (u\in\surrealLeft{x}\land v\in\surrealLeft{y})
                \implies
                \surrealLt
                    {\surrealAdd{\surrealMul{u}{y}}{\surrealMul{x}{v}}}
                    {\surrealAdd{\surrealMul{x}{y}}{\surrealMul{u}{v}}})
            \land
            (\forall u,v:\Set.
                (u\in\surrealRight{x}\land v\in\surrealRight{y})
                \implies
                \surrealLt
                    {\surrealAdd{\surrealMul{u}{y}}{\surrealMul{x}{v}}}
                    {\surrealAdd{\surrealMul{x}{y}}{\surrealMul{u}{v}}})
            \land
            (\forall u,v:\Set.
                (u\in\surrealLeft{x}\land v\in\surrealRight{y})
                \implies
                \surrealLt
                    {\surrealAdd{\surrealMul{x}{y}}{\surrealMul{u}{v}}}
                    {\surrealAdd{\surrealMul{u}{y}}{\surrealMul{x}{v}}})
            \land
            (\forall u,v:\Set.
                (u\in\surrealRight{x}\land v\in\surrealLeft{y})
                \implies
                \surrealLt
                    {\surrealAdd{\surrealMul{x}{y}}{\surrealMul{u}{v}}}
                    {\surrealAdd{\surrealMul{u}{y}}{\surrealMul{x}{v}}})}$.
\end{definition}

\begin{lemma}\label{surreal_mul_property_left_left}
    Suppose $\Apply{\surrealMulPropertyOp}{x,y}$.
    Let $u\in\surrealLeft{x}$ and $v\in\surrealLeft{y}$.
    Then
        $\surrealLt
            {\surrealAdd{\surrealMul{u}{y}}{\surrealMul{x}{v}}}
            {\surrealAdd{\surrealMul{x}{y}}{\surrealMul{u}{v}}}$.
\end{lemma}
\begin{proof}
    Follows by \obv{surreal_mul_property}.
\end{proof}

\begin{lemma}\label{surreal_mul_property_right_left}
    Suppose $\Apply{\surrealMulPropertyOp}{x,y}$.
    Let $u\in\surrealRight{x}$ and $v\in\surrealLeft{y}$.
    Then
        $\surrealLt
            {\surrealAdd{\surrealMul{x}{y}}{\surrealMul{u}{v}}}
            {\surrealAdd{\surrealMul{u}{y}}{\surrealMul{x}{v}}}$.
\end{lemma}
\begin{proof}
    Follows by \obv{surreal_mul_property}.
\end{proof}

\begin{definition}\label{surreal_mul_available}
    $\surrealMulAvailableOp:\Set\to\Set\to\Prop
        =\fun{X,Y:\Set}{
        (\forall a:\Set.
            (\surreal{a}\land\birthday{a}\in\birthday{X})
            \implies
            (\forall b:\Set.\ \surreal{b}\implies
                \Apply{\surrealMulPropertyOp}{a,b}))
        \land
        (\forall b:\Set.
            (\surreal{b}\land\birthday{b}\in\birthday{Y})
            \implies
            \Apply{\surrealMulPropertyOp}{X,b})}$.
\end{definition}

\begin{lemma}\label{surreal_mul_available_at}
    Suppose $\Apply{\surrealMulAvailableOp}{X,Y}$.
    Suppose $\surreal{a}$ and $\surreal{b}$.
    Suppose $\birthday{a}\in\birthday{X}$ or
        ($a=X$ and $\birthday{b}\in\birthday{Y}$).
    Then $\Apply{\surrealMulPropertyOp}{a,b}$.
\end{lemma}
\begin{proof}
    Follows by \obv{surreal_mul_available}.
\end{proof}

\begin{lemma}\label{surreal_mul_option_is_surreal_from_available}
    Suppose $\surreal{X}$ and $\surreal{Y}$.
    Suppose $\Apply{\surrealMulAvailableOp}{X,Y}$.
    Suppose
        $p\in(\surrealLeft{X}\union\surrealRight{X})
            \times(\surrealLeft{Y}\union\surrealRight{Y})$.
    Then $\surreal{\surrealMulOption{X}{Y}{p}}$.
\end{lemma}
\begin{proof}
    Take $a,b$ such that
        $a\in\surrealLeft{X}\union\surrealRight{X}$ and
        $b\in\surrealLeft{Y}\union\surrealRight{Y}$ and $p=(a,b)$
        by \obv{times_elem_is_tuple}.
    $\surreal{a}$ and $\birthday{a}\in\birthday{X}$
        by \auto{union,surreal_left_elim,surreal_right_elim}.
    $\surreal{b}$ and $\birthday{b}\in\birthday{Y}$
        by \auto{union,surreal_left_elim,surreal_right_elim}.
    $\Apply{\surrealMulPropertyOp}{a,Y}
        \land\Apply{\surrealMulPropertyOp}{X,b}
        \land\Apply{\surrealMulPropertyOp}{a,b}$
        by \obv{surreal_mul_available_at}.
    $\surreal{\surrealMul{a}{Y}}$ and
        $\surreal{\surrealMul{X}{b}}$ and
        $\surreal{\surrealMul{a}{b}}$
        by \obv{surreal_mul_property}.
    Follows by
        \auto{surreal_mul_option_pair,surreal_add_is_surreal,surreal_negation_is_surreal}.
\end{proof}

\begin{lemma}\label{surreal_mul_cross_from_available_properties}
    Suppose $\surreal{X}$ and $\surreal{Y}$.
    Suppose $\Apply{\surrealMulAvailableOp}{X,Y}$.
    Suppose $\surreal{u}$ and $\surreal{x}$ and
        $\surreal{v}$ and $\surreal{y}$.
    Suppose $(u=X\lor\birthday{u}\in\birthday{X})$ and
        $(x=X\lor\birthday{x}\in\birthday{X})$.
    Suppose $(v=Y\lor\birthday{v}\in\birthday{Y})$ and
        $(y=Y\lor\birthday{y}\in\birthday{Y})$.
    Suppose
        $(\birthday{u}\in\birthday{X}\land
            \birthday{x}\in\birthday{X})$
        or
        $(\birthday{v}\in\birthday{Y}\land
            \birthday{y}\in\birthday{Y})$.
    Suppose $\surrealLt{u}{x}$ and $\surrealLt{v}{y}$.
    Then
        $\surrealLt
            {\surrealAdd{\surrealMul{u}{y}}{\surrealMul{x}{v}}}
            {\surrealAdd{\surrealMul{x}{y}}{\surrealMul{u}{v}}}$.
\end{lemma}
\begin{proof}
    $\Apply{\surrealMulPropertyOp}{u,v}
        \land\Apply{\surrealMulPropertyOp}{u,y}
        \land\Apply{\surrealMulPropertyOp}{x,v}
        \land\Apply{\surrealMulPropertyOp}{x,y}$
        by \auto{surreal_mul_available_at}.
    $\surreal{\surrealMul{u}{v}}$ and
        $\surreal{\surrealMul{u}{y}}$ and
        $\surreal{\surrealMul{x}{v}}$ and
        $\surreal{\surrealMul{x}{y}}$
        by \obv{surreal_mul_property}.
    $(\exists w:\Set.\ \surreal{w}\land
            \birthday{w}\in\birthday{u}\inter\birthday{x}\land
            \surrealLt{u}{w}\land\surrealLt{w}{x})
        \lor u\in\surrealLeft{x}
        \lor x\in\surrealRight{u}$
        by \obv{surreal_lt_elimination}.
    \begin{byCase}
        \caseOf{$\exists w:\Set.\ \surreal{w}\land
                \birthday{w}\in\birthday{u}\inter\birthday{x}\land
                \surrealLt{u}{w}\land\surrealLt{w}{x}$.}
            Take $w$ such that $\surreal{w}$ and
                $\birthday{w}\in\birthday{u}\inter\birthday{x}$ and
                $\surrealLt{u}{w}$ and $\surrealLt{w}{x}$.
            $w\in\surrealRight{u}$ and $w\in\surrealLeft{x}$
                by \obv{surreal_right_intro,surreal_left_intro,inter}.
            $\birthday{w}\in\birthday{X}$
                by
                \auto{inter,ordinal_transitivity,surreal_birthday_ordinal}.
            $\Apply{\surrealMulPropertyOp}{w,v}
                \land\Apply{\surrealMulPropertyOp}{w,y}$
                by \obv{surreal_mul_available_at}.
            $(\exists z:\Set.\ \surreal{z}\land
                    \birthday{z}\in\birthday{v}\inter\birthday{y}\land
                    \surrealLt{v}{z}\land\surrealLt{z}{y})
                \lor v\in\surrealLeft{y}

\begin{lemma}\label{surreal_mul_left_inequality_from_available}

        {\surrealMulRight{X}{Y}}$.

        or $(u\in\surrealRight{X}\land v\in\surrealRight{Y})$.
    Then
        $\surrealLt
            {\surrealAdd{\surrealMul{u}{Y}}{\surrealMul{X}{v}}}
            {\surrealAdd{\surrealMul{X}{Y}}{\surrealMul{u}{v}}}$.
\end{lemma}
\begin{proof}
    $\surreal{u}$ and $\birthday{u}\in\birthday{X}$ and
        $\surreal{v}$ and $\birthday{v}\in\birthday{Y}$
        by \auto{surreal_left_elim,surreal_right_elim}.

        \land\Apply{\surrealMulPropertyOp}{X,v}
        \land\Apply{\surrealMulPropertyOp}{u,v}$
        by \obv{surreal_mul_available_at}.

        \in\surrealMulLeft{X}{Y}$

            {\surrealCut{\surrealMulLeft{X}{Y}}{\surrealMulRight{X}{Y}}}$
        by \obv{surreal_cut_pair_left}.

        $\surrealLt{\surrealMulOption{X}{Y}{(u,v)}}
            {\surrealMul{X}{Y}}$
        by \obv{surreal_mul_equation}.

            {\surrealAdd
                {\surrealAdd{\surrealMul{u}{Y}}{\surrealMul{X}{v}}}
                {\surrealNeg{\surrealMul{u}{v}}}}
            {\surrealMul{X}{Y}}$
        by \auto{surreal_mul_option_pair}.
    $\surreal{\surrealMul{X}{Y}}$
        by \obv{surreal_mul_equation,surreal_cut_pair_is_surreal}.

        \obv{surreal_subtract_lt_add,surreal_add_is_surreal}.

\begin{theorem}\label{surreal_mul_properties}

                (\surreal{b}\land\birthday{b}\in\birthday{Y})
                \implies
                \Apply{\surrealMulPropertyOp}{X,b}))
         \implies\Apply{\surrealMulPropertyOp}{X,Y})$.

            $(\forall a:\Set.

                \implies

                    \Apply{\surrealMulPropertyOp}{a,b}))

                (\surreal{b}\land\birthday{b}\in\birthday{Y})

                \Apply{\surrealMulPropertyOp}{X,b})$.

            by \obv{surreal_mul_available}.
        Show that for every $l\in\surrealMulLeft{X}{Y}$ we have

                  \lor(u\in\surrealRight{X}\land v\in\surrealRight{Y}))

\begin{theorem}\label{surreal_mul_one_right}
    Suppose $\surreal{x}$.
    Then $\surrealMul{x}{\ordinalSurreal{\suc{\emptyset}}}=x$.
\end{theorem}
\begin{proof}
    Let $P:\Set\to\Prop=\fun{u:\Set}{
        \surrealMul{u}{\ordinalSurreal{\suc{\emptyset}}}=u}$.
    Show that $\forall u:\Set.\ \surreal{u}\implies
        ((\forall w:\Set.
            (\surreal{w}\land\birthday{w}\in\birthday{u})
            \implies\Apply{P}{w})
         \implies\Apply{P}{u})$.
    \begin{subproof}
        Fix $u$.
        Assume $\surreal{u}$.
        Assume $\forall w:\Set.
            (\surreal{w}\land\birthday{w}\in\birthday{u})
            \implies\Apply{P}{w}$.
        Show that for every
            $a\in\surrealMulLeft{u}{\ordinalSurreal{\suc{\emptyset}}}$
            we have $a\in\surrealLeft{u}$.
        \begin{subproof}
            Fix $a\in
                \surrealMulLeft{u}{\ordinalSurreal{\suc{\emptyset}}}$.
            Take $w,v$ such that
                $((w\in\surrealLeft{u}\land
                        v\in\surrealLeft{\ordinalSurreal{\suc{\emptyset}}})
                  \lor(w\in\surrealRight{u}\land
                        v\in\surrealRight{\ordinalSurreal{\suc{\emptyset}}}))
                 \land
                 a=\surrealMulOption{u}
                    {\ordinalSurreal{\suc{\emptyset}}}{(w,v)}$
                by \auto{surreal_mul_left_pair_iff}.
            $w\in\surrealLeft{u}$ and $v=\emptyset$
                by
                \auto{surreal_one_left,surreal_one_right,singleton_iff,emptyset}.
            $\surreal{w}$ and $\birthday{w}\in\birthday{u}$
                by \auto{surreal_left_elim}.
            Hence $\Apply{P}{w}$.
            Hence
                $\surrealMul{w}{\ordinalSurreal{\suc{\emptyset}}}=w$.
            $\surrealMulOption{u}{\ordinalSurreal{\suc{\emptyset}}}
                    {(w,\emptyset)}
                =\surrealMul{w}{\ordinalSurreal{\suc{\emptyset}}}$
                by \obv{surreal_mul_option_one_zero}.
        \end{subproof}
        Show that for every $w\in\surrealLeft{u}$ we have
            $w\in\surrealMulLeft{u}
                {\ordinalSurreal{\suc{\emptyset}}}$.
        \begin{subproof}
            Fix $w\in\surrealLeft{u}$.
            $\surreal{w}$ and $\birthday{w}\in\birthday{u}$
                by \auto{surreal_left_elim}.
            Hence $\Apply{P}{w}$.
            Hence
                $\surrealMul{w}{\ordinalSurreal{\suc{\emptyset}}}=w$.
            $\surrealMulOption{u}{\ordinalSurreal{\suc{\emptyset}}}
                    {(w,\emptyset)}
                =w$
                by \auto{surreal_mul_option_one_zero}.
            $\emptyset\in
                \surrealLeft{\ordinalSurreal{\suc{\emptyset}}}$
                by \auto{surreal_one_left,singleton_iff}.
            $\surrealMulOption{u}{\ordinalSurreal{\suc{\emptyset}}}
                    {(w,\emptyset)}
                \in\surrealMulLeft{u}
                    {\ordinalSurreal{\suc{\emptyset}}}$
                by \auto{surreal_mul_left_pair_iff}.
        \end{subproof}
        Hence
            $\surrealMulLeft{u}{\ordinalSurreal{\suc{\emptyset}}}
                =\surrealLeft{u}$
            by set extensionality.
        Show that for every
            $a\in\surrealMulRight{u}{\ordinalSurreal{\suc{\emptyset}}}$
            we have $a\in\surrealRight{u}$.
        \begin{subproof}
            Fix $a\in
                \surrealMulRight{u}{\ordinalSurreal{\suc{\emptyset}}}$.
            Take $w,v$ such that
                $((w\in\surrealLeft{u}\land
                        v\in\surrealRight{\ordinalSurreal{\suc{\emptyset}}})
                  \lor(w\in\surrealRight{u}\land
                        v\in\surrealLeft{\ordinalSurreal{\suc{\emptyset}}}))
                 \land
                 a=\surrealMulOption{u}
                    {\ordinalSurreal{\suc{\emptyset}}}{(w,v)}$
                by \auto{surreal_mul_right_pair_iff}.
            $w\in\surrealRight{u}$ and $v=\emptyset$
                by
                \auto{surreal_one_left,surreal_one_right,singleton_iff,emptyset}.
            $\surreal{w}$ and $\birthday{w}\in\birthday{u}$
                by \auto{surreal_right_elim}.
            Hence $\Apply{P}{w}$.
            Hence
                $\surrealMul{w}{\ordinalSurreal{\suc{\emptyset}}}=w$.
            $\surrealMulOption{u}{\ordinalSurreal{\suc{\emptyset}}}
                    {(w,\emptyset)}
                =\surrealMul{w}{\ordinalSurreal{\suc{\emptyset}}}$
                by \obv{surreal_mul_option_one_zero}.
        \end{subproof}
        Hence
            $\surrealMulRight{u}{\ordinalSurreal{\suc{\emptyset}}}
                \subseteq\surrealRight{u}$
            by \obv{subseteq}.
        Show that for every $w\in\surrealRight{u}$ we have
            $w\in\surrealMulRight{u}
                {\ordinalSurreal{\suc{\emptyset}}}$.
        \begin{subproof}
            Fix $w\in\surrealRight{u}$.
            $\surreal{w}$ and $\birthday{w}\in\birthday{u}$
                by \auto{surreal_right_elim}.
            Hence $\Apply{P}{w}$.
            Hence
                $\surrealMul{w}{\ordinalSurreal{\suc{\emptyset}}}=w$.
            $\surrealMulOption{u}{\ordinalSurreal{\suc{\emptyset}}}
                    {(w,\emptyset)}
                =w$
                by \auto{surreal_mul_option_one_zero}.
            $\emptyset\in
                \surrealLeft{\ordinalSurreal{\suc{\emptyset}}}$
                by \auto{surreal_one_left,singleton_iff}.
            $\surrealMulOption{u}{\ordinalSurreal{\suc{\emptyset}}}
                    {(w,\emptyset)}
                \in\surrealMulRight{u}
                    {\ordinalSurreal{\suc{\emptyset}}}$
                by \auto{surreal_mul_right_pair_iff}.
        \end{subproof}
        Hence
            $\surrealRight{u}\subseteq
                \surrealMulRight{u}{\ordinalSurreal{\suc{\emptyset}}}$
            by \obv{subseteq}.
        Hence
            $\surrealMulRight{u}{\ordinalSurreal{\suc{\emptyset}}}
                =\surrealRight{u}$
            by \obv{subseteq_antisymmetric}.
        $\surrealMul{u}{\ordinalSurreal{\suc{\emptyset}}}
            =\surrealCut
                {\surrealMulLeft{u}{\ordinalSurreal{\suc{\emptyset}}}}
                {\surrealMulRight{u}{\ordinalSurreal{\suc{\emptyset}}}}$
            by \obv{surreal_mul_equation,surreal_one_is_surreal}.
        Hence
            $\surrealMul{u}{\ordinalSurreal{\suc{\emptyset}}}
                =\surrealCut{\surrealLeft{u}}{\surrealRight{u}}$
            by \auto.
        Hence
            $\surrealMul{u}{\ordinalSurreal{\suc{\emptyset}}}=u$
            by \auto{surreal_eta}.
    \end{subproof}
    Follows by \auto{surreal_induction}.
\end{proof}

\begin{theorem}\label{surreal_mul_commutative}
    Suppose $\surreal{x}$ and $\surreal{y}$.
    Then $\surrealMul{x}{y}=\surrealMul{y}{x}$.
\end{theorem}
\begin{proof}
    Let $P:\Set\to\Set\to\Prop=\fun{u,v:\Set}{
        \surrealMul{u}{v}=\surrealMul{v}{u}}$.
    Show that $\forall u,v:\Set.\ (\surreal{u}\land\surreal{v})\implies
        (((\forall a:\Set.
                (\surreal{a}\land\birthday{a}\in\birthday{u})
                \implies
                (\forall b:\Set.\ \surreal{b}\implies
                    \Apply{P}{a,b}))
          \land
          (\forall b:\Set.
                (\surreal{b}\land\birthday{b}\in\birthday{v})
                \implies\Apply{P}{u,b}))
         \implies\Apply{P}{u,v})$.
    \begin{subproof}
        Fix $u,v$.
        Assume $\surreal{u}\land\surreal{v}$.
        Assume
            $(\forall a:\Set.
                (\surreal{a}\land\birthday{a}\in\birthday{u})
                \implies
                (\forall b:\Set.\ \surreal{b}\implies
                    \Apply{P}{a,b}))
             \land
             (\forall b:\Set.
                (\surreal{b}\land\birthday{b}\in\birthday{v})
                \implies\Apply{P}{u,b})$.
        Show that for all $a,b$ such that
            $a\in\surrealLeft{u}\union\surrealRight{u}$ and
            $b\in\surrealLeft{v}\union\surrealRight{v}$ we have
            $\surrealMulOption{u}{v}{(a,b)}
                =\surrealMulOption{v}{u}{(b,a)}$.
        \begin{subproof}
            Fix $a,b$.
            Assume $a\in\surrealLeft{u}\union\surrealRight{u}$ and
                $b\in\surrealLeft{v}\union\surrealRight{v}$.
            $\surreal{a}$ and $\birthday{a}\in\birthday{u}$ and
                $\surreal{b}$ and $\birthday{b}\in\birthday{v}$
                by
                \auto{union,surreal_left_elim,surreal_right_elim}.
            $\Apply{P}{a,v}
                \land\Apply{P}{u,b}
                \land\Apply{P}{a,b}$.
            Hence
                $\surrealMul{a}{v}=\surrealMul{v}{a}$ and
                $\surrealMul{u}{b}=\surrealMul{b}{u}$ and
                $\surrealMul{a}{b}=\surrealMul{b}{a}$.
            Follows by
                \obv{surreal_mul_option_swap_from_equalities}.
        \end{subproof}
        Show that for every $z\in\surrealMulLeft{u}{v}$ we have
            $z\in\surrealMulLeft{v}{u}$.
        \begin{subproof}
            Fix $z\in\surrealMulLeft{u}{v}$.
            Take $a,b$ such that
                $((a\in\surrealLeft{u}\land b\in\surrealLeft{v})
                  \lor(a\in\surrealRight{u}\land b\in\surrealRight{v}))
                 \land z=\surrealMulOption{u}{v}{(a,b)}$
                by \auto{surreal_mul_left_pair_iff}.
            $a\in\surrealLeft{u}\union\surrealRight{u}$ and
                $b\in\surrealLeft{v}\union\surrealRight{v}$
                by \auto{union}.
            Hence
                $\surrealMulOption{u}{v}{(a,b)}
                    =\surrealMulOption{v}{u}{(b,a)}$.
            $\surrealMulOption{v}{u}{(b,a)}
                \in\surrealMulLeft{v}{u}$
                by \auto{surreal_mul_left_pair_iff}.
        \end{subproof}
        Hence $\surrealMulLeft{u}{v}\subseteq\surrealMulLeft{v}{u}$
            by \obv{subseteq}.
        Show that for every $z\in\surrealMulLeft{v}{u}$ we have
            $z\in\surrealMulLeft{u}{v}$.
        \begin{subproof}
            Fix $z\in\surrealMulLeft{v}{u}$.
            Take $b,a$ such that
                $((b\in\surrealLeft{v}\land a\in\surrealLeft{u})
                  \lor(b\in\surrealRight{v}\land a\in\surrealRight{u}))
                 \land z=\surrealMulOption{v}{u}{(b,a)}$
                by \auto{surreal_mul_left_pair_iff}.
            $a\in\surrealLeft{u}\union\surrealRight{u}$ and
                $b\in\surrealLeft{v}\union\surrealRight{v}$
                by \auto{union}.
            Hence
                $\surrealMulOption{u}{v}{(a,b)}
                    =\surrealMulOption{v}{u}{(b,a)}$.
            $\surrealMulOption{u}{v}{(a,b)}
                \in\surrealMulLeft{u}{v}$
                by \auto{surreal_mul_left_pair_iff}.
        \end{subproof}
        Hence $\surrealMulLeft{v}{u}\subseteq\surrealMulLeft{u}{v}$
            by \obv{subseteq}.
        Hence $\surrealMulLeft{u}{v}=\surrealMulLeft{v}{u}$
            by \obv{subseteq_antisymmetric}.
        Show that for every $z\in\surrealMulRight{u}{v}$ we have
            $z\in\surrealMulRight{v}{u}$.
        \begin{subproof}
            Fix $z\in\surrealMulRight{u}{v}$.
            Take $a,b$ such that
                $((a\in\surrealLeft{u}\land b\in\surrealRight{v})
                  \lor(a\in\surrealRight{u}\land b\in\surrealLeft{v}))
                 \land z=\surrealMulOption{u}{v}{(a,b)}$
                by \auto{surreal_mul_right_pair_iff}.
            $a\in\surrealLeft{u}\union\surrealRight{u}$ and
                $b\in\surrealLeft{v}\union\surrealRight{v}$
                by \auto{union}.
            Hence
                $\surrealMulOption{u}{v}{(a,b)}
                    =\surrealMulOption{v}{u}{(b,a)}$.
            $\surrealMulOption{v}{u}{(b,a)}
                \in\surrealMulRight{v}{u}$
                by \auto{surreal_mul_right_pair_iff}.
        \end{subproof}
        Hence $\surrealMulRight{u}{v}\subseteq\surrealMulRight{v}{u}$
            by \obv{subseteq}.
        Show that for every $z\in\surrealMulRight{v}{u}$ we have
            $z\in\surrealMulRight{u}{v}$.
        \begin{subproof}
            Fix $z\in\surrealMulRight{v}{u}$.
            Take $b,a$ such that
                $((b\in\surrealLeft{v}\land a\in\surrealRight{u})
                  \lor(b\in\surrealRight{v}\land a\in\surrealLeft{u}))
                 \land z=\surrealMulOption{v}{u}{(b,a)}$
                by \auto{surreal_mul_right_pair_iff}.
            $a\in\surrealLeft{u}\union\surrealRight{u}$ and
                $b\in\surrealLeft{v}\union\surrealRight{v}$
                by \auto{union}.
            Hence
                $\surrealMulOption{u}{v}{(a,b)}
                    =\surrealMulOption{v}{u}{(b,a)}$.
            $\surrealMulOption{u}{v}{(a,b)}
                \in\surrealMulRight{u}{v}$
                by \auto{surreal_mul_right_pair_iff}.
        \end{subproof}
        Hence $\surrealMulRight{v}{u}\subseteq\surrealMulRight{u}{v}$
            by \obv{subseteq}.
        Hence $\surrealMulRight{u}{v}=\surrealMulRight{v}{u}$
            by \obv{subseteq_antisymmetric}.
        $\surrealMul{u}{v}
            =\surrealCut{\surrealMulLeft{u}{v}}{\surrealMulRight{u}{v}}$
            by \obv{surreal_mul_equation}.
        $\surrealMul{v}{u}
            =\surrealCut{\surrealMulLeft{v}{u}}{\surrealMulRight{v}{u}}$
            by \obv{surreal_mul_equation}.
        Hence $\surrealMul{u}{v}=\surrealMul{v}{u}$ by \auto.
    \end{subproof}
    Follows by \auto{surreal_binary_induction}.
\end{proof}

\begin{theorem}\label{surreal_mul_nonzero_cancel_left}
    Suppose $\surreal{x}$ and $x\neq\emptyset$ and
        $\surreal{y}$ and $\surreal{z}$.
    Suppose $\surrealMul{x}{y}=\surrealMul{x}{z}$.
    Then $y=z$.
\end{theorem}
\begin{proof}
    $\surrealLt{x}{\emptyset}$ or $x=\emptyset$ or
        $\surrealLt{\emptyset}{x}$
        by \obv{surreal_lt_trichotomy,empty_surreal}.
    $\surrealLt{y}{z}$ or $y=z$ or $\surrealLt{z}{y}$
        by \obv{surreal_lt_trichotomy}.
    \begin{byCase}
        \caseOf{$\surrealLt{y}{z}$.}
            \begin{byCase}
                \caseOf{$\surrealLt{\emptyset}{x}$.}
                    $\surrealLt{\surrealMul{x}{y}}{\surrealMul{x}{z}}$
                        by \obv{surreal_pos_mul_lt_left}.
                    Contradiction by
                        \obv{surreal_mul_is_surreal,surreal_lt_irreflexive}.
                \caseOf{$\lnot\surrealLt{\emptyset}{x}$.}
                    $\surrealLt{x}{\emptyset}$.
                    $\surrealLt{\surrealMul{x}{z}}{\surrealMul{x}{y}}$
                        by \obv{surreal_neg_mul_lt_left}.
                    Contradiction by
                        \obv{surreal_mul_is_surreal,surreal_lt_irreflexive}.
            \end{byCase}
        \caseOf{$\lnot\surrealLt{y}{z}$.}
            \begin{byCase}
                \caseOf{$y=z$.}
                    Follows.
                \caseOf{$y\neq z$.}
                    $\surrealLt{z}{y}$.
                    \begin{byCase}
                        \caseOf{$\surrealLt{\emptyset}{x}$.}
                            $\surrealLt{\surrealMul{x}{z}}
                                {\surrealMul{x}{y}}$
                                by \obv{surreal_pos_mul_lt_left}.
                            Contradiction by
                                \obv{surreal_mul_is_surreal,surreal_lt_irreflexive}.
                        \caseOf{$\lnot\surrealLt{\emptyset}{x}$.}
                            $\surrealLt{x}{\emptyset}$.
                            $\surrealLt{\surrealMul{x}{y}}
                                {\surrealMul{x}{z}}$
                                by \obv{surreal_neg_mul_lt_left}.
                            Contradiction by
                                \obv{surreal_mul_is_surreal,surreal_lt_irreflexive}.
                    \end{byCase}
            \end{byCase}
    \end{byCase}
\end{proof}

\begin{corollary}\label{surreal_mul_nonzero_cancel_right}

\begin{theorem}\label{surreal_square_strict_monotone_nonnegative}
    Suppose $\surreal{x}$ and $\surreal{y}$.
    Suppose $\surrealLe{\emptyset}{x}$ and $\surrealLt{x}{y}$.
    Then $\surrealLt{\surrealMul{x}{x}}{\surrealMul{y}{y}}$.
\end{theorem}
\begin{proof}
    $\surrealLt{\emptyset}{y}$
        by \obv{surreal_le_lt_transitive,empty_surreal}.
    \begin{byCase}
        \caseOf{$x=\emptyset$.}
            $\surrealMul{x}{x}=\emptyset$
                by \auto{surreal_mul_zero_left}.
            Follows by \obv{surreal_mul_pos_pos}.
        \caseOf{$x\neq\emptyset$.}
            $\surrealLt{\emptyset}{x}$
                by \auto{surreal_le,surreal_lt_trichotomy,empty_surreal}.
            $\surrealLt{\surrealMul{x}{x}}{\surrealMul{x}{y}}$
                by \obv{surreal_pos_mul_lt_left}.
            $\surrealLt{\surrealMul{x}{y}}{\surrealMul{y}{y}}$
                by \obv{surreal_pos_mul_lt_right}.
            Follows by
                \obv{surreal_lt_transitive,surreal_mul_is_surreal}.
    \end{byCase}
\end{proof}

\begin{theorem}\label{surreal_nonnegative_square_unique}
    Suppose $\surreal{x}$ and $\surreal{y}$.
    Suppose $\surrealLe{\emptyset}{x}$ and
        $\surrealLe{\emptyset}{y}$.
    Suppose $\surrealMul{x}{x}=\surrealMul{y}{y}$.
    Then $x=y$.
\end{theorem}
\begin{proof}
    $\surrealLt{x}{y}$ or $x=y$ or $\surrealLt{y}{x}$
        by \obv{surreal_lt_trichotomy}.
    \begin{byCase}
        \caseOf{$\surrealLt{x}{y}$.}
            $\surrealLt{\surrealMul{x}{x}}{\surrealMul{y}{y}}$
                by \obv{surreal_square_strict_monotone_nonnegative}.
            Contradiction by
                \obv{surreal_mul_is_surreal,surreal_lt_irreflexive}.
        \caseOf{$\lnot\surrealLt{x}{y}$.}
            \begin{byCase}
                \caseOf{$x=y$.}
                    Follows.
                \caseOf{$x\neq y$.}
                    $\surrealLt{y}{x}$.
                    $\surrealLt{\surrealMul{y}{y}}{\surrealMul{x}{x}}$
                        by
                        \obv{surreal_square_strict_monotone_nonnegative}.
                    Contradiction by
                        \obv{surreal_mul_is_surreal,surreal_lt_irreflexive}.
            \end{byCase}
    \end{byCase}
\end{proof}
