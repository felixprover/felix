\import{surreal/real.tex}

\section{Dyadic approximation of real surreals}

\begin{theorem}\label{finite_surreal_dyadic_interval}
    Suppose $x\in\surrealStage{\naturals}$.
    Let $n$ be a natural number.
    Then there exists $q\in\surrealStage{\naturals}$ such that
        $\surrealLt{q}{x}$ and
        $\surrealLt{x}{\surrealAdd{q}{\surrealEpsilon{n}}}$.
\end{theorem}
\begin{proof}
    Let $e=\surrealEpsilon{\suc{n}}$.
    Let $q=\surrealAdd{x}{\surrealNeg{e}}$.
    $\suc{n}$ is a natural number by \obv{omega_successor}.
    Hence $e\in\surrealStage{\naturals}$
        by \auto{surreal_epsilon_in_finite_stage}.
    Hence $\surrealNeg{e}\in\surrealStage{\naturals}$
        by \obv{surreal_negation_finite_stage}.
    Hence $q\in\surrealStage{\naturals}$
        by \auto{surreal_addition_finite_stage}.
    $\surreal{x}$ and $\surreal{e}$
        by
        \auto{surreal_stage_characterization,omega_is_ordinal,surreal_epsilon_is_surreal}.
    $\surrealLt{\surrealZero}{e}$
        by \obv{surreal_epsilon_positive}.
    Hence $\surrealLt{\surrealNeg{e}}{\surrealZero}$
        by
        \obv{surreal_negation_lt_reverses,surreal_negation_empty,empty_surreal}.
    Hence $\surrealLt{q}{x}$
        by
        \auto{surreal_add_lt_right,surreal_add_zero_right,empty_surreal,surreal_negation_is_surreal}.
    $\surrealAdd{e}{e}=\surrealEpsilon{n}$
        by
        \auto{surreal_epsilon_halving,natural_number_is_ordinal}.
    $\surrealAdd{q}{\surrealEpsilon{n}}
        =\surrealAdd{x}{e}$
        by
        \auto{surreal_add_associative,surreal_add_commutative,surreal_add_inverse_left,surreal_add_zero_right,surreal_negation_is_surreal,surreal_add_is_surreal}.
    $\surrealLt{x}{\surrealAdd{x}{e}}$
        by
        \auto{surreal_add_lt_right,surreal_add_zero_right,empty_surreal}.
\end{proof}

\begin{lemma}\label{negative_natural_successor_plus_one}
    Let $n$ be a natural number.
    Then
        $\surrealAdd{\surrealNeg{\ordinalSurreal{\suc{n}}}}
            {\surrealOne}=\surrealNeg{\ordinalSurreal{n}}$.
\end{lemma}
\begin{proof}
    $n$ and $\suc{n}$ are ordinals
        by \obv{natural_number_is_ordinal,suc_ordinal}.
    $\surrealAdd{\ordinalSurreal{n}}{\surrealOne}
        =\ordinalSurreal{\suc{n}}$
        by \obv{ordinal_surreal_successor_addition}.
    Follows by
        \auto{surreal_negation_additive,surreal_add_associative,surreal_add_commutative,surreal_add_inverse_left,surreal_add_zero_right,ordinal_surreal_is_surreal,surreal_one_is_surreal,surreal_negation_is_surreal}.
\end{proof}

\begin{lemma}\label{nonfinite_surreal_unit_interval_in_symmetric_bound}
    Let $N$ be an ordinal.
    Suppose $N$ is a natural number.
    Suppose $x\in\surrealStage{\suc{\naturals}}$ and
        $x\notin\surrealStage{\naturals}$.
    Suppose
        $\surrealLt{\surrealNeg{\ordinalSurreal{N}}}{x}$ and
        $\surrealLt{x}{\ordinalSurreal{N}}$.
    Then there exists $q\in\surrealStage{\naturals}$ such that
        $\surrealLt{q}{x}$ and
        $\surrealLt{x}{\surrealAdd{q}{\surrealOne}}$.
\end{lemma}
\begin{proof}[Proof by transfinite induction]
    $\surreal{x}$
        by
        \auto{surreal_stage_characterization,suc_ordinal,omega_is_ordinal}.
    $N=\emptyset$ or there exists a natural number $k$ such that
        $N=\suc{k}$ by \obv{natural_zero_or_successor}.
    \begin{byCase}
        \caseOf{$N=\emptyset$.}
            $\ordinalSurreal{N}=\surrealZero$ and
                $\surrealNeg{\ordinalSurreal{N}}=\surrealZero$
                by
                \obv{empty_ordinal_surreal,surreal_negation_empty}.
            Contradiction by
                \auto{surreal_lt_asymmetric,empty_surreal}.
        \caseOf{$N\neq\emptyset$.}
            Take $k\in\naturals$ such that $N=\suc{k}$.
            $N=\suc{k}$.
            $k\in N$ by \obv{suc_intro_self}.
            $\surreal{\ordinalSurreal{k}}$ and
                $\surreal{\surrealNeg{\ordinalSurreal{k}}}$
                by
                \obv{ordinal_surreal_is_surreal,natural_number_is_ordinal,surreal_negation_is_surreal}.
            $\surrealLt{x}{\ordinalSurreal{k}}$ or
                $x=\ordinalSurreal{k}$ or
                $\surrealLt{\ordinalSurreal{k}}{x}$
                by \obv{surreal_lt_trichotomy}.
            \begin{byCase}
                \caseOf{$x=\ordinalSurreal{k}$.}
                    Contradiction by
                        \auto{surreal_stage_characterization,ordinal_surreal_is_surreal,ordinal_surreal_birthday,natural_number_is_ordinal,omega_is_ordinal}.
                \caseOf{$x\neq\ordinalSurreal{k}$.}
                    \begin{byCase}
                        \caseOf{$\surrealLt{\ordinalSurreal{k}}{x}$.}
                            $\ordinalSurreal{k}\in
                                \surrealStage{\naturals}$
                                by
                                \auto{surreal_stage_characterization,ordinal_surreal_is_surreal,ordinal_surreal_birthday,natural_number_is_ordinal,omega_is_ordinal}.
                            $\surrealAdd{\ordinalSurreal{k}}
                                {\surrealOne}
                                =\ordinalSurreal{N}$
                                by
                                \auto{ordinal_surreal_successor_addition,natural_number_is_ordinal}.
                        \caseOf{not $\surrealLt{\ordinalSurreal{k}}{x}$.}
                            Hence $\surrealLt{x}{\ordinalSurreal{k}}$.
                            $\surrealLt{x}
                                {\surrealNeg{\ordinalSurreal{k}}}$ or
                                $x=\surrealNeg{\ordinalSurreal{k}}$ or
                                $\surrealLt
                                    {\surrealNeg{\ordinalSurreal{k}}}{x}$
                                by \obv{surreal_lt_trichotomy}.
                            \begin{byCase}
                                \caseOf{$x=\surrealNeg{\ordinalSurreal{k}}$.}
                                    Contradiction by
                                        \auto{surreal_stage_characterization,surreal_negation_is_surreal,surreal_negation_birthday,ordinal_surreal_is_surreal,ordinal_surreal_birthday,natural_number_is_ordinal,omega_is_ordinal}.
                                \caseOf{$x\neq
                                    \surrealNeg{\ordinalSurreal{k}}$.}
                                    \begin{byCase}
                                        \caseOf{$\surrealLt{x}
                                            {\surrealNeg{\ordinalSurreal{k}}}$.}
                                            $\surrealNeg{\ordinalSurreal{N}}
                                                \in\surrealStage{\naturals}$
                                                by
                                                \auto{surreal_stage_characterization,surreal_negation_is_surreal,surreal_negation_birthday,ordinal_surreal_is_surreal,ordinal_surreal_birthday,natural_number_is_ordinal,omega_successor,omega_is_ordinal}.
                                            $\surrealAdd
                                                {\surrealNeg{\ordinalSurreal{N}}}
                                                {\surrealOne}
                                                =\surrealNeg{\ordinalSurreal{k}}$
                                                by
                                                \auto{negative_natural_successor_plus_one}.
                                        \caseOf{not $\surrealLt{x}
                                            {\surrealNeg{\ordinalSurreal{k}}}$.}
                                            Hence
                                                $\surrealLt
                                                    {\surrealNeg{\ordinalSurreal{k}}}{x}$.
                                            Take $q\in
                                                \surrealStage{\naturals}$ such that
                                                $\surrealLt{q}{x}$ and
                                                $\surrealLt{x}
                                                    {\surrealAdd{q}{\surrealOne}}$.
                                    \end{byCase}
                            \end{byCase}
                    \end{byCase}
            \end{byCase}
    \end{byCase}
\end{proof}

\begin{theorem}\label{successor_omega_surreal_natural_bound_above}
    Suppose $x\in\surrealStage{\suc{\naturals}}$ and
        $\surrealLt{x}{\ordinalSurreal{\naturals}}$.
    Then there exists a natural number $N$ such that
        $\surrealLt{x}{\ordinalSurreal{N}}$.
\end{theorem}
\begin{proof}
    $\surreal{x}$ and $\birthday{x}\in\suc{\naturals}$
        by
        \auto{surreal_stage_characterization,suc_ordinal,omega_is_ordinal}.
    $\birthday{x}$ is an ordinal
        by \obv{surreal_birthday_ordinal}.
    $\birthday{x}\subseteq\naturals$
        by
        \auto{suc_iff,ordinal_elem_implies_subseteq,subseteq_refl,omega_is_ordinal}.
    $(\exists \delta:\Set.\ \surrealLtCommon{\delta}{x}
        {\ordinalSurreal{\naturals}})
        \lor
        \surrealLtLeftShort{x}{\ordinalSurreal{\naturals}}
        \lor
        \surrealLtRightShort{x}{\ordinalSurreal{\naturals}}$
        by \obv{surreal_lt}.
    \begin{byCase}
        \caseOf{$\exists \delta:\Set.\ \surrealLtCommon{\delta}{x}
                {\ordinalSurreal{\naturals}}$.}
            Take $\delta$ such that
                $\surrealLtCommon{\delta}{x}
                    {\ordinalSurreal{\naturals}}$.
            $\delta\in\birthday{x}\inter\naturals$ and
                $\surrealEq{\delta}{x}
                    {\ordinalSurreal{\naturals}}$ and
                $x(\delta)=\emptyset$
                by
                \obv{surreal_lt_common,ordinal_surreal_birthday,omega_is_ordinal}.
            Hence $\delta\in\naturals$ by \obv{inter_elim_right}.
            Let $N=\suc{\delta}$.
            $N\in\naturals$ by \auto{omega_successor}.
            $\delta\in N$ by \obv{suc_intro_self}.
            $\surrealEq{\delta}
                {\ordinalSurreal{\naturals}}{\ordinalSurreal{N}}$
                by
                \obv{ordinal_surreal_agree,ordinal_elem_implies_subseteq,natural_number_is_ordinal,omega_is_ordinal}.
            Hence $\surrealEq{\delta}{x}{\ordinalSurreal{N}}$
                by \obv{surreal_eq_initial_transitive}.
            $\apply{\ordinalSurreal{N}}{\delta}=\suc{\emptyset}$
                by \obv{ordinal_surreal_value}.
            $\delta\in\birthday{x}$
                by \obv{inter_elim_left}.
            $\birthday{\ordinalSurreal{N}}=N$
                by
                \obv{ordinal_surreal_birthday,natural_number_is_ordinal}.
            Hence $\delta\in\birthday{\ordinalSurreal{N}}$.
            Hence $\delta\in\birthday{x}\inter
                \birthday{\ordinalSurreal{N}}$
                by \obv{inter_intro}.
            Hence $\surrealLtCommon{\delta}{x}
                {\ordinalSurreal{N}}$
                by
                \obv{surreal_lt_common}.
            Hence $\surrealLt{x}{\ordinalSurreal{N}}$
                by \obv{surreal_lt}.
        \caseOf{not $\exists \delta:\Set.\ \surrealLtCommon{\delta}{x}
                {\ordinalSurreal{\naturals}}$.}
            $\surrealLtLeftShort{x}{\ordinalSurreal{\naturals}}$
                or
                $\surrealLtRightShort{x}
                    {\ordinalSurreal{\naturals}}$.
            \begin{byCase}
                \caseOf{$\surrealLtLeftShort{x}
                    {\ordinalSurreal{\naturals}}$.}
                    $\birthday{x}\in\naturals$
                        by
                        \obv{surreal_lt_left_short,ordinal_surreal_birthday,omega_is_ordinal}.
                    Let $N=\suc{\birthday{x}}$.
                    $N\in\naturals$ by \auto{omega_successor}.
                    $\birthday{x}\in N$ by \obv{suc_intro_self}.
                    $\surrealLt{x}{\ordinalSurreal{N}}$
                        by
                        \obv{ordinal_surreal_strictly_dominates_birthday,natural_number_is_ordinal}.
                \caseOf{not $\surrealLtLeftShort{x}
                    {\ordinalSurreal{\naturals}}$.}
                    Hence $\surrealLtRightShort{x}
                        {\ordinalSurreal{\naturals}}$.
                    Hence $\naturals\in\birthday{x}$
                        by
                        \obv{surreal_lt_right_short,ordinal_surreal_birthday,omega_is_ordinal}.
                    Hence $\naturals\in\naturals$
                        by \obv{subseteq}.
                    Contradiction by \obv{in_irrefl}.
            \end{byCase}
    \end{byCase}
\end{proof}

\begin{theorem}\label{successor_omega_surreal_natural_bound_below}
    Suppose $x\in\surrealStage{\suc{\naturals}}$ and
        $\surrealLt{\surrealNeg{\ordinalSurreal{\naturals}}}{x}$.
    Then there exists a natural number $N$ such that
        $\surrealLt{\surrealNeg{\ordinalSurreal{N}}}{x}$.
\end{theorem}
\begin{proof}
    $\surreal{x}$
        by
        \auto{surreal_stage_characterization,suc_ordinal,omega_is_ordinal}.
    Hence $\surrealNeg{x}\in\surrealStage{\suc{\naturals}}$
        by \obv{surreal_negation_stage,suc_ordinal,omega_is_ordinal}.
    $\surrealLt{\surrealNeg{x}}{\ordinalSurreal{\naturals}}$
        by
        \auto{surreal_negation_lt_reverses,surreal_negation_involutive,surreal_negation_is_surreal,ordinal_surreal_is_surreal,omega_is_ordinal}.
    Take a natural number $N$ such that
        $\surrealLt{\surrealNeg{x}}{\ordinalSurreal{N}}$
        by \obv{successor_omega_surreal_natural_bound_above}.
    $\surreal{\ordinalSurreal{N}}$
        by
        \obv{ordinal_surreal_is_surreal,natural_number_is_ordinal}.
    Hence $\surrealLt{\surrealNeg{\ordinalSurreal{N}}}{x}$
        by
        \auto{surreal_negation_lt_reverses,surreal_negation_involutive,surreal_negation_is_surreal}.
\end{proof}
