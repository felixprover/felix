\section{The real subfield of the surreals}

The real carrier consists of the finite-birthday surreals together with the
canonical limits of finite-birthday surreals.  The last clause excludes a
nonzero infinitesimal displacement from a finite-birthday surreal.  It is
kept primitively as two-sided inequalities.  Once absolute value has been
introduced, the public real interface restates the same condition as one
ordinary absolute-value bound.

\begin{definition}\label{surreal_reals}
    $\reals=
        \{x\in\surrealStage{\suc{\naturals}}\mid
            \text{$x\neq\ordinalSurreal{\naturals}$ and
            $x\neq\surrealNeg{\ordinalSurreal{\naturals}}$ and
            for every $q\in\surrealStage{\naturals}$ such that
                for every natural number $n$ we have
                    $\surrealLt{q}
                        {\surrealAdd{x}{\surrealEpsilon{n}}}$ and
                    $\surrealLt{x}
                        {\surrealAdd{q}{\surrealEpsilon{n}}}$
                we have $q=x$}\}$.
\end{definition}

\begin{theorem}\label{surreal_real_intro}
    Suppose $x\in\surrealStage{\suc{\naturals}}$.
    Suppose $x\neq\ordinalSurreal{\naturals}$ and
        $x\neq\surrealNeg{\ordinalSurreal{\naturals}}$.
    Suppose for every $q\in\surrealStage{\naturals}$ such that
        for every natural number $n$ we have
            $\surrealLt{q}{\surrealAdd{x}{\surrealEpsilon{n}}}$ and
            $\surrealLt{x}{\surrealAdd{q}{\surrealEpsilon{n}}}$
        we have $q=x$.
    Then $x\in\reals$.
\end{theorem}
\begin{proof}
    Follows by \obv{surreal_reals}.
\end{proof}

\begin{theorem}\label{surreal_real_elim}
    Suppose $x\in\reals$.
    Then $x\in\surrealStage{\suc{\naturals}}$ and
        $x\neq\ordinalSurreal{\naturals}$ and
        $x\neq\surrealNeg{\ordinalSurreal{\naturals}}$ and
        for every $q\in\surrealStage{\naturals}$ such that
        for every natural number $n$ we have
            $\surrealLt{q}{\surrealAdd{x}{\surrealEpsilon{n}}}$ and
            $\surrealLt{x}{\surrealAdd{q}{\surrealEpsilon{n}}}$
        we have $q=x$.
\end{theorem}
\begin{proof}
    Follows by \obv{surreal_reals}.
\end{proof}

\begin{corollary}\label{surreal_real_is_surreal}
    For every $x\in\reals$ we have $\surreal{x}$.
\end{corollary}
\begin{proof}
    Follows by
        \auto{surreal_real_elim,surreal_stage_characterization,suc_ordinal,omega_is_ordinal}.
\end{proof}

\begin{lemma}\label{surreal_successor_omega_stage_bound_above}
    Suppose $\surreal{x}$ and
        $\birthday{x}\in\suc{\naturals}$.
    Suppose $x\neq\ordinalSurreal{\naturals}$.
    Then $\surrealLt{x}{\ordinalSurreal{\naturals}}$.
\end{lemma}
\begin{proof}
    $\birthday{x}\in\naturals$ or $\birthday{x}=\naturals$
        by \auto{suc_iff}.
    \begin{byCase}
        \caseOf{$\birthday{x}\in\naturals$.}
            Follows by
                \obv{ordinal_surreal_strictly_dominates_birthday,omega_is_ordinal}.
        \caseOf{$\birthday{x}\notin\naturals$.}
            Hence $\birthday{x}=\naturals$.
            $\surrealLe{x}{\ordinalSurreal{\naturals}}$
                by
                \obv{ordinal_surreal_maximum_at_birthday,omega_is_ordinal}.
            Follows by \obv{surreal_le}.
    \end{byCase}
\end{proof}

\begin{lemma}\label{surreal_successor_omega_stage_bound_below}
    Suppose $\surreal{x}$ and
        $\birthday{x}\in\suc{\naturals}$.
    Suppose $x\neq\surrealNeg{\ordinalSurreal{\naturals}}$.
    Then
        $\surrealLt{\surrealNeg{\ordinalSurreal{\naturals}}}{x}$.
\end{lemma}
\begin{proof}
    $\surreal{\surrealNeg{x}}$ and
        $\birthday{\surrealNeg{x}}=\birthday{x}$
        by
        \obv{surreal_negation_is_surreal,surreal_negation_birthday}.
    $\surrealNeg{x}\neq\ordinalSurreal{\naturals}$
        by \auto{surreal_negation_involutive}.
    Hence
        $\surrealLt{\surrealNeg{x}}{\ordinalSurreal{\naturals}}$
        by \obv{surreal_successor_omega_stage_bound_above}.
    Follows by
        \auto{surreal_negation_lt_reverses,surreal_negation_involutive,ordinal_surreal_is_surreal,omega_is_ordinal}.
\end{proof}

\begin{corollary}\label{surreal_real_bounds}
    Suppose $x\in\reals$.
    Then
        $\surrealLt{\surrealNeg{\ordinalSurreal{\naturals}}}{x}$ and
        $\surrealLt{x}{\ordinalSurreal{\naturals}}$.
\end{corollary}
\begin{proof}
    $\surreal{x}$ and $\birthday{x}\in\suc{\naturals}$ and
        $x\neq\ordinalSurreal{\naturals}$ and
        $x\neq\surrealNeg{\ordinalSurreal{\naturals}}$
        by
        \auto{surreal_real_elim,surreal_stage_characterization,suc_ordinal,omega_is_ordinal}.
    Follows by
        \obv{surreal_successor_omega_stage_bound_above,surreal_successor_omega_stage_bound_below}.
\end{proof}

\subsection{Finite-birthday and dyadic reals}

\begin{theorem}\label{finite_surreal_stage_subset_reals}
    $\surrealStage{\naturals}\subseteq\reals$.
\end{theorem}
\begin{proof}
    Show that for every $x\in\surrealStage{\naturals}$ we have
        $x\in\reals$.
    \begin{subproof}
        Fix $x\in\surrealStage{\naturals}$.
        $\surreal{x}$ and $\birthday{x}\in\naturals$
            by
            \auto{surreal_stage_characterization,omega_is_ordinal}.
        $\naturals\subseteq\suc{\naturals}$
            by \auto{subseteq,suc_iff}.
        Hence $x\in\surrealStage{\suc{\naturals}}$
            by
            \auto{surreal_stage_monotone,omega_is_ordinal,suc_ordinal,subseteq}.
        $\birthday{\ordinalSurreal{\naturals}}=\naturals$ and
            $\birthday{\surrealNeg{\ordinalSurreal{\naturals}}}
                =\naturals$
            by
            \obv{ordinal_surreal_birthday,ordinal_surreal_is_surreal,surreal_negation_birthday,omega_is_ordinal}.
        Hence $x\neq\ordinalSurreal{\naturals}$ and
            $x\neq\surrealNeg{\ordinalSurreal{\naturals}}$
            by \obv{in_irrefl}.
        Show that $\forall q:\Set.\ q\in\surrealStage{\naturals}
            \implies
            ((\forall n:\Set.\ n\in\naturals\implies
                (\surrealLt{q}{\surrealAdd{x}{\surrealEpsilon{n}}}
                 \land
                 \surrealLt{x}{\surrealAdd{q}{\surrealEpsilon{n}}}))
             \implies q=x)$.
        \begin{subproof}
            Fix $q$.
            Assume $q\in\surrealStage{\naturals}$.
            Assume for every natural number $n$ we have
                $\surrealLt{q}{\surrealAdd{x}{\surrealEpsilon{n}}}$
                and
                $\surrealLt{x}{\surrealAdd{q}{\surrealEpsilon{n}}}$.
            $\surreal{q}$ by
                \auto{surreal_stage_characterization,omega_is_ordinal}.
            $\surrealLt{q}{x}$ or $q=x$ or $\surrealLt{x}{q}$
                by \obv{surreal_lt_trichotomy}.
            \begin{byCase}
                \caseOf{$q=x$.}
                    Follows.
                \caseOf{$q\neq x$.}
                    \begin{byCase}
                        \caseOf{$\surrealLt{q}{x}$.}
                            Let $d=\surrealAdd{x}{\surrealNeg{q}}$.
                            $\surrealNeg{q}\in
                                \surrealStage{\naturals}$
                                by \obv{surreal_negation_finite_stage}.
                            Hence $d\in\surrealStage{\naturals}$
                                by \auto{surreal_addition_finite_stage}.
                            Hence $\birthday{d}\in\naturals$ and
                                $\surreal{d}$
                                by
                                \auto{surreal_stage_characterization,omega_is_ordinal}.
                            $\surrealLt{\surrealZero}{d}$
                                by
                                \auto{surreal_add_lt_left,surreal_add_inverse_right,surreal_add_zero_right,surreal_negation_is_surreal}.
                            $d\in\surrealStage{\suc{\birthday{d}}}$
                                by
                                \auto{surreal_stage_characterization,suc_ordinal,surreal_birthday_ordinal,suc_intro_self}.
                            Hence
                                $\surrealLt{\surrealEpsilon{\birthday{d}}}{d}$
                                by
                                \obv{positive_finite_surreal_dominates_epsilon}.
                            $\surrealLt{x}
                                {\surrealAdd{q}
                                    {\surrealEpsilon{\birthday{d}}}}$
                                by \auto.
                            Hence
                                $\surrealLt{d}
                                    {\surrealAdd
                                        {\surrealAdd{q}
                                            {\surrealEpsilon{\birthday{d}}}}
                                        {\surrealNeg{q}}}$
                                by
                                \auto{surreal_add_lt_left,surreal_negation_is_surreal,surreal_add_is_surreal,surreal_epsilon_is_surreal}.
                            Hence
                                $\surrealLt{d}
                                    {\surrealEpsilon{\birthday{d}}}$
                                by
                                \auto{surreal_add_associative,surreal_add_commutative,surreal_add_inverse_right,surreal_add_zero_right,surreal_epsilon_is_surreal,surreal_negation_is_surreal,surreal_add_is_surreal}.
                            Contradiction by
                                \auto{surreal_lt_asymmetric,surreal_epsilon_is_surreal}.
                        \caseOf{not $\surrealLt{q}{x}$.}
                            Hence $\surrealLt{x}{q}$.
                            Let $d=\surrealAdd{q}{\surrealNeg{x}}$.
                            $\surrealNeg{x}\in
                                \surrealStage{\naturals}$
                                by \obv{surreal_negation_finite_stage}.
                            Hence $d\in\surrealStage{\naturals}$
                                by \auto{surreal_addition_finite_stage}.
                            Hence $\birthday{d}\in\naturals$ and
                                $\surreal{d}$
                                by
                                \auto{surreal_stage_characterization,omega_is_ordinal}.
                            $\surrealLt{\surrealZero}{d}$
                                by
                                \auto{surreal_add_lt_left,surreal_add_inverse_right,surreal_add_zero_right,surreal_negation_is_surreal}.
                            $d\in\surrealStage{\suc{\birthday{d}}}$
                                by
                                \auto{surreal_stage_characterization,suc_ordinal,surreal_birthday_ordinal,suc_intro_self}.
                            Hence
                                $\surrealLt{\surrealEpsilon{\birthday{d}}}{d}$
                                by
                                \obv{positive_finite_surreal_dominates_epsilon}.
                            $\surrealLt{q}
                                {\surrealAdd{x}
                                    {\surrealEpsilon{\birthday{d}}}}$
                                by \auto.
                            Hence
                                $\surrealLt{d}
                                    {\surrealAdd
                                        {\surrealAdd{x}
                                            {\surrealEpsilon{\birthday{d}}}}
                                        {\surrealNeg{x}}}$
                                by
                                \auto{surreal_add_lt_left,surreal_negation_is_surreal,surreal_add_is_surreal,surreal_epsilon_is_surreal}.
                            Hence
                                $\surrealLt{d}
                                    {\surrealEpsilon{\birthday{d}}}$
                                by
                                \auto{surreal_add_associative,surreal_add_commutative,surreal_add_inverse_right,surreal_add_zero_right,surreal_epsilon_is_surreal,surreal_negation_is_surreal,surreal_add_is_surreal}.
                            Contradiction by
                                \auto{surreal_lt_asymmetric,surreal_epsilon_is_surreal}.
                    \end{byCase}
            \end{byCase}
        \end{subproof}
        Follows by \obv{surreal_real_intro}.
    \end{subproof}
    Follows by \obv{subseteq}.
\end{proof}

\begin{lemma}\label{surreal_integer_finite_stage}
    $\surrealIntegers\subseteq\surrealStage{\naturals}$.
\end{lemma}
\begin{proof}
    Show that for every $z\in\surrealIntegers$ we have
        $z\in\surrealStage{\naturals}$.
    \begin{subproof}
        Fix $z\in\surrealIntegers$.
        Take $n,m\in\naturals$ such that
            $z=\surrealAdd{\ordinalSurreal{n}}
                {\surrealNeg{\ordinalSurreal{m}}}$
            by \auto{surreal_integer_iff}.
        $\ordinalSurreal{n},\ordinalSurreal{m}
            \in\surrealStage{\naturals}$
            by
            \auto{surreal_stage_characterization,ordinal_surreal_is_surreal,ordinal_surreal_birthday,omega_is_ordinal,natural_number_is_ordinal}.
        Hence $\surrealNeg{\ordinalSurreal{m}}
            \in\surrealStage{\naturals}$
            by \obv{surreal_negation_finite_stage}.
        Follows by \auto{surreal_addition_finite_stage}.
    \end{subproof}
    Follows by \obv{subseteq}.
\end{proof}

\begin{theorem}\label{surreal_dyadics_finite_stage}
    $\surrealDyadics\subseteq\surrealStage{\naturals}$.
\end{theorem}
\begin{proof}
    Show that for every $x\in\surrealDyadics$ we have
        $x\in\surrealStage{\naturals}$.
    \begin{subproof}
        Fix $x\in\surrealDyadics$.
        Take $k,z$ such that $k\in\naturals$ and
            $z\in\surrealIntegers$ and
            $x=\surrealMul{\surrealEpsilon{k}}{z}$
            by \obv{surreal_dyadic_iff}.
        $\surrealEpsilon{k}\in\surrealStage{\naturals}$
            by \obv{surreal_epsilon_in_finite_stage}.
        $z\in\surrealStage{\naturals}$
            by \obv{surreal_integer_finite_stage,subseteq}.
        Follows by \auto{surreal_multiplication_finite_stage}.
    \end{subproof}
    Follows by \obv{subseteq}.
\end{proof}

\begin{corollary}\label{surreal_dyadics_subset_reals}
    $\surrealDyadics\subseteq\reals$.
\end{corollary}
\begin{proof}
    Follows by
        \obv{surreal_dyadics_finite_stage,finite_surreal_stage_subset_reals,subseteq_transitive}.
\end{proof}

\begin{corollary}\label{natural_ordinal_surreal_is_real}
    Let $n$ be a natural number.
    Then $\ordinalSurreal{n}\in\reals$.
\end{corollary}
\begin{proof}
    $\ordinalSurreal{n}\in\surrealStage{\naturals}$
        by
        \auto{surreal_stage_characterization,ordinal_surreal_is_surreal,ordinal_surreal_birthday,natural_number_is_ordinal,omega_is_ordinal}.
    Follows by \obv{finite_surreal_stage_subset_reals,subseteq}.
\end{proof}

\begin{corollary}\label{surreal_epsilon_is_real}
    Let $n$ be a natural number.
    Then $\surrealEpsilon{n}\in\reals$.
\end{corollary}
\begin{proof}
    $\surrealEpsilon{n}\in\surrealDyadics$
        by \obv{surreal_epsilon_is_dyadic}.
    Follows by \obv{surreal_dyadics_subset_reals,subseteq}.
\end{proof}

\begin{corollary}\label{surreal_zero_is_real}

\begin{proof}

\begin{corollary}\label{surreal_one_is_real}
    $\surrealOne\in\reals$.
\end{corollary}

\subsection{Negation}

\begin{theorem}\label{surreal_real_negation_closed}
    Suppose $x\in\reals$.
    Then $\surrealNeg{x}\in\reals$.
\end{theorem}
\begin{proof}
    $x\in\surrealStage{\suc{\naturals}}$ and
        $x\neq\ordinalSurreal{\naturals}$ and
        $x\neq\surrealNeg{\ordinalSurreal{\naturals}}$
        by \obv{surreal_real_elim}.
    Hence $\surrealNeg{x}\in\surrealStage{\suc{\naturals}}$
        by \obv{surreal_negation_stage,suc_ordinal,omega_is_ordinal}.
    $\surrealNeg{x}\neq\ordinalSurreal{\naturals}$ and
        $\surrealNeg{x}\neq
            \surrealNeg{\ordinalSurreal{\naturals}}$
        by
        \auto{surreal_negation_involutive,ordinal_surreal_is_surreal,omega_is_ordinal,surreal_real_is_surreal}.
    Show that $\forall q:\Set.\ q\in\surrealStage{\naturals}
        \implies
        ((\forall n:\Set.\ n\in\naturals\implies
            (\surrealLt{q}
                {\surrealAdd{\surrealNeg{x}}{\surrealEpsilon{n}}}
             \land
             \surrealLt{\surrealNeg{x}}
                {\surrealAdd{q}{\surrealEpsilon{n}}}))
         \implies q=\surrealNeg{x})$.
    \begin{subproof}
        Fix $q$.
        Assume $q\in\surrealStage{\naturals}$.
        Assume for every natural number $n$ we have
            $\surrealLt{q}
                {\surrealAdd{\surrealNeg{x}}{\surrealEpsilon{n}}}$
            and
            $\surrealLt{\surrealNeg{x}}
                {\surrealAdd{q}{\surrealEpsilon{n}}}$.
        Let $r=\surrealNeg{q}$.
        $r\in\surrealStage{\naturals}$
            by \auto{surreal_negation_finite_stage}.
        Show that for every natural number $n$ we have
            $\surrealLt{r}{\surrealAdd{x}{\surrealEpsilon{n}}}$
            and
            $\surrealLt{x}{\surrealAdd{r}{\surrealEpsilon{n}}}$.
        \begin{subproof}
            Fix $n\in\naturals$.
            $\surreal{q}$ and $\surreal{x}$ and
                $\surreal{\surrealEpsilon{n}}$
                by
                \auto{surreal_stage_characterization,omega_is_ordinal,surreal_real_is_surreal,surreal_epsilon_is_surreal}.
            $\surrealLt{\surrealNeg{x}}
                {\surrealAdd{q}{\surrealEpsilon{n}}}$
                by \auto.
            Hence
                $\surrealLt
                    {\surrealNeg{\surrealAdd{q}{\surrealEpsilon{n}}}}
                    {\surrealNeg{\surrealNeg{x}}}$
                by
                \obv{surreal_negation_lt_reverses,surreal_add_is_surreal,surreal_negation_is_surreal}.
            Hence
                $\surrealLt
                    {\surrealAdd{r}
                        {\surrealNeg{\surrealEpsilon{n}}}}{x}$
                by
                \auto{surreal_negation_additive,surreal_negation_involutive,surreal_negation_is_surreal,surreal_real_is_surreal,surreal_epsilon_is_surreal}.
            Hence
                $\surrealLt{r}{\surrealAdd{x}{\surrealEpsilon{n}}}$
                by
                \auto{surreal_subtract_lt_add,surreal_stage_characterization,omega_is_ordinal,surreal_real_is_surreal,surreal_epsilon_is_surreal}.
            $\surrealLt{q}
                {\surrealAdd{\surrealNeg{x}}{\surrealEpsilon{n}}}$.
            Hence
                $\surrealLt
                    {\surrealNeg{\surrealAdd
                        {\surrealNeg{x}}{\surrealEpsilon{n}}}}
                    {\surrealNeg{q}}$
                by
                \obv{surreal_negation_lt_reverses,surreal_add_is_surreal,surreal_negation_is_surreal}.
            Hence
                $\surrealLt
                    {\surrealAdd{x}
                        {\surrealNeg{\surrealEpsilon{n}}}}{r}$
                by
                \auto{surreal_negation_additive,surreal_negation_involutive,surreal_negation_is_surreal,surreal_real_is_surreal,surreal_epsilon_is_surreal}.
            Hence
                $\surrealLt{x}{\surrealAdd{r}{\surrealEpsilon{n}}}$
                by
                \auto{surreal_subtract_lt_add,surreal_stage_characterization,omega_is_ordinal,surreal_real_is_surreal,surreal_epsilon_is_surreal}.
        \end{subproof}
        Hence $r=x$ by \obv{surreal_real_elim}.
        Follows by
            \auto{surreal_negation_involutive,surreal_real_is_surreal,surreal_stage_characterization,omega_is_ordinal}.
    \end{subproof}
    Follows by \obv{surreal_real_intro}.
\end{proof}
