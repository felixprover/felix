\section{Finite-stage cuts of real surreals}

The completeness arguments use cuts whose options all have finite birthday.
The cut itself may first appear at birthday $\naturals$, but never later.

\begin{theorem}\label{finite_stage_surreal_cut_successor_stage}
    Suppose $\surrealCutPair{L}{R}$.
    Suppose $L\union R\subseteq\surrealStage{\naturals}$.
    Then $\surrealCut{L}{R}\in
        \surrealStage{\suc{\naturals}}$.
\end{theorem}
\begin{proof}
    Let $c=\surrealCut{L}{R}$.
    Let $\kappa=\surrealCutBound{L}{R}$.
    $\surreal{c}$ and $\kappa$ is an ordinal and
        $\birthday{c}\in\suc{\kappa}$
        by
        \auto{surreal_cut_pair_is_surreal,surreal_cut_bound_ordinal,surreal_cut_pair_birthday_bound}.
    Show that for every $\alpha\in\kappa$ we have
        $\alpha\in\naturals$.
    \begin{subproof}
        Fix $\alpha\in\kappa$.
        Take $\gamma\in\surrealCutLengths{L}{R}$ such that
            $\alpha\in\gamma$
            by \auto{surreal_cut_bound,unions_iff}.
        Take $z\in L\union R$ such that
            $\gamma=\suc{\birthday{z}}$
            by \obv{surreal_cut_lengths}.
        $z\in\surrealStage{\naturals}$ by \obv{subseteq}.
        Hence $\birthday{z}\in\naturals$
            by
            \auto{surreal_stage_characterization,omega_is_ordinal}.
        $\birthday{z}\subseteq\naturals$
            by
            \auto{ordinal_elem_implies_subseteq,omega_is_ordinal,surreal_stage_characterization,surreal_birthday_ordinal}.
        Hence $\suc{\birthday{z}}\subseteq\naturals$
            by \obv{suc_subseteq_intro}.
        Follows by \obv{subseteq}.
    \end{subproof}
    Hence $\kappa\subseteq\naturals$ by \obv{subseteq}.
    $\kappa\in\naturals$ or $\kappa=\naturals$
        by
        \obv{ordinal_subseteq_implies_elem_or_eq,omega_is_ordinal}.
    Show that $\birthday{c}\in\suc{\naturals}$.
    \begin{subproof}
        \begin{byCase}
            \caseOf{$\kappa=\naturals$.}
                Follows.
            \caseOf{$\kappa\neq\naturals$.}
                Hence $\kappa\in\naturals$.
                $\suc{\kappa}\subseteq\naturals$
                    by \obv{suc_subseteq_intro}.
                Hence $\birthday{c}\in\naturals$
                    by \obv{subseteq}.
                Follows by \auto{suc_iff}.
        \end{byCase}
    \end{subproof}
    Follows by
        \auto{surreal_stage_characterization,suc_ordinal,omega_is_ordinal}.
\end{proof}

\subsection{Canonical finite-stage bounds}

\begin{definition}\label{surreal_real_lower}
    $\realLower{x}=
        \{q\in\surrealStage{\naturals}\mid\surrealLt{q}{x}\}$.
\end{definition}

\begin{definition}\label{surreal_real_upper}
    $\realUpper{x}=
        \{q\in\surrealStage{\naturals}\mid\surrealLt{x}{q}\}$.
\end{definition}

\begin{proposition}\label{surreal_real_lower_iff}
    $q\in\realLower{x}$ iff
        $q\in\surrealStage{\naturals}$ and $\surrealLt{q}{x}$.
\end{proposition}
\begin{proof}
    Follows by \obv{surreal_real_lower}.
\end{proof}

\begin{proposition}\label{surreal_real_upper_iff}
    $q\in\realUpper{x}$ iff
        $q\in\surrealStage{\naturals}$ and $\surrealLt{x}{q}$.
\end{proposition}
\begin{proof}
    Follows by \obv{surreal_real_upper}.
\end{proof}

\begin{proposition}\label{surreal_real_bounds_finite_stage}
    Suppose $x\in\reals$.
    Then $\realLower{x}\union\realUpper{x}
        \subseteq\surrealStage{\naturals}$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{subseteq,union_iff,surreal_real_lower_iff,surreal_real_upper_iff}.
\end{proof}

\begin{proposition}\label{surreal_real_bounds_cut_pair}
    Suppose $x\in\reals$.
    Then $\surrealCutPair{\realLower{x}}{\realUpper{x}}$.
\end{proposition}
\begin{proof}
    $\surreal{x}$ by \obv{surreal_real_is_surreal}.
    Show that for every $l\in\realLower{x}$ we have $\surreal{l}$.
    \begin{subproof}
        Fix $l\in\realLower{x}$.
        $l\in\surrealStage{\naturals}$
            by \obv{surreal_real_lower_iff}.
        Follows by
            \auto{surreal_stage_characterization,omega_is_ordinal}.
    \end{subproof}
    Show that for every $r\in\realUpper{x}$ we have $\surreal{r}$.
    \begin{subproof}
        Fix $r\in\realUpper{x}$.
        $r\in\surrealStage{\naturals}$
            by \obv{surreal_real_upper_iff}.
        Follows by
            \auto{surreal_stage_characterization,omega_is_ordinal}.
    \end{subproof}
    For all $l\in\realLower{x}$ we have for all
        $r\in\realUpper{x}$ we have $\surrealLt{l}{r}$
        by
        \obv{surreal_real_lower_iff,surreal_real_upper_iff,surreal_lt_transitive}.
    Follows by \obv{surreal_cut_pair}.
\end{proof}

\begin{proposition}\label{surreal_real_lower_inhabited}
    Suppose $x\in\reals$.
    Then $\realLower{x}$ is inhabited.
\end{proposition}
\begin{proof}
    Take $q\in\surrealStage{\naturals}$ such that
        $\surrealLt{q}{x}$ and
        $\surrealLt{x}{\surrealAdd{q}{\surrealEpsilon{\emptyset}}}$
        by
        \obv{surreal_real_dyadic_interval,emptyset_is_ordinal,omega_empty}.
    $q\in\realLower{x}$ by \obv{surreal_real_lower_iff}.
\end{proof}

\begin{proposition}\label{surreal_real_upper_inhabited}
    Suppose $x\in\reals$.
    Then $\realUpper{x}$ is inhabited.
\end{proposition}
\begin{proof}
    Take $q\in\surrealStage{\naturals}$ such that
        $\surrealLt{q}{x}$ and
        $\surrealLt{x}{\surrealAdd{q}{\surrealEpsilon{\emptyset}}}$
        by
        \obv{surreal_real_dyadic_interval,emptyset_is_ordinal,omega_empty}.
    $\surrealEpsilon{\emptyset}\in\surrealStage{\naturals}$
        by
        \obv{surreal_epsilon_in_finite_stage,omega_empty}.
    Hence $\surrealAdd{q}{\surrealEpsilon{\emptyset}}
        \in\surrealStage{\naturals}$
        by \obv{surreal_addition_finite_stage}.
    Hence $\surrealAdd{q}{\surrealEpsilon{\emptyset}}
        \in\realUpper{x}$
        by \obv{surreal_real_upper_iff}.
\end{proof}

\begin{theorem}\label{surreal_real_lower_no_maximum}
    Suppose $x\in\reals$.
    Then for every $w\in\realLower{x}$ there exists
        $w'\in\realLower{x}$ such that $\surrealLt{w}{w'}$.
\end{theorem}
\begin{proof}
    Fix $w\in\realLower{x}$.
    $w\in\surrealStage{\naturals}$ and $\surrealLt{w}{x}$
        by \obv{surreal_real_lower_iff}.
    $\surreal{w}$ and $\surreal{x}$
        by
        \auto{surreal_stage_characterization,omega_is_ordinal,surreal_real_is_surreal}.
    Suppose not.
    Show that for every natural number $n$ we have
        $\surrealLt{w}{\surrealAdd{x}{\surrealEpsilon{n}}}$ and
        $\surrealLt{x}{\surrealAdd{w}{\surrealEpsilon{n}}}$.
    \begin{subproof}
        Fix $n\in\naturals$.
        $\suc{n}$ is a natural number
            by \obv{omega_successor}.
        Let $e=\surrealEpsilon{n}$.
        Let $e'=\surrealEpsilon{\suc{n}}$.
        Let $t=\surrealAdd{w}{e'}$.
        $\surreal{e}$ and $\surreal{e'}$ and
            $\surrealLt{\surrealZero}{e}$ and
            $\surrealLt{\surrealZero}{e'}$
            by
            \auto{surreal_epsilon_is_surreal,surreal_epsilon_positive}.
        $\surrealLt{x}{\surrealAdd{x}{e}}$
            by
            \auto{surreal_add_lt_right,surreal_add_zero_right,empty_surreal}.
        Hence $\surrealLt{w}{\surrealAdd{x}{e}}$
            by
            \auto{surreal_lt_transitive,surreal_add_is_surreal}.
        $e'\in\surrealStage{\naturals}$
            by \auto{surreal_epsilon_in_finite_stage}.
        Hence $t\in\surrealStage{\naturals}$
            by \auto{surreal_addition_finite_stage}.
        $\surreal{t}$
            by
            \auto{surreal_stage_characterization,omega_is_ordinal}.
        $\surrealLt{w}{t}$
            by
            \auto{surreal_add_lt_right,surreal_add_zero_right,empty_surreal}.
        Show that not $\surrealLt{t}{x}$.
        \begin{subproof}
            Suppose $\surrealLt{t}{x}$.
            Hence $t\in\realLower{x}$
                by \obv{surreal_real_lower_iff}.
            Contradiction by \auto.
        \end{subproof}
        Hence $\surrealLe{x}{t}$
            by \auto{surreal_lt_trichotomy,surreal_le}.
        $n\in\suc{n}$ by \obv{suc_intro_self}.
        Hence $\surrealLt{e'}{e}$
            by \auto{surreal_epsilon_decreasing}.
        Hence $\surrealLt{t}{\surrealAdd{w}{e}}$
            by \auto{surreal_add_lt_right}.
        Hence $\surrealLt{x}{\surrealAdd{w}{e}}$
            by
            \auto{surreal_le_lt_transitive,surreal_add_is_surreal}.
        Follows by \auto.
    \end{subproof}
    Hence $w=x$ by \obv{surreal_real_elim}.
    Contradiction by \obv{surreal_lt_irreflexive}.
\end{proof}

\begin{theorem}\label{surreal_real_upper_no_minimum}
    Suppose $x\in\reals$.
    Then for every $r\in\realUpper{x}$ there exists
        $r'\in\realUpper{x}$ such that $\surrealLt{r'}{r}$.
\end{theorem}
\begin{proof}
    Fix $r\in\realUpper{x}$.
    $r\in\surrealStage{\naturals}$ and $\surrealLt{x}{r}$
        by \obv{surreal_real_upper_iff}.
    Let $y=\surrealNeg{x}$.
    Let $w=\surrealNeg{r}$.
    $y\in\reals$ by \auto{surreal_real_negation_closed}.
    $w\in\surrealStage{\naturals}$
        by \auto{surreal_negation_finite_stage}.
    $\surrealLt{w}{y}$
        by
        \auto{surreal_negation_lt_reverses,surreal_real_is_surreal,surreal_stage_characterization,omega_is_ordinal}.
    Hence $w\in\realLower{y}$
        by \obv{surreal_real_lower_iff}.
    Take $w'\in\realLower{y}$ such that $\surrealLt{w}{w'}$
        by \obv{surreal_real_lower_no_maximum}.
    Let $r'=\surrealNeg{w'}$.
    $w'\in\surrealStage{\naturals}$ and $\surrealLt{w'}{y}$
        by \auto{surreal_real_lower_iff}.
    Hence $r'\in\surrealStage{\naturals}$
        by \auto{surreal_negation_finite_stage}.
    $\surrealLt{x}{r'}$
        by
        \auto{surreal_negation_lt_reverses,surreal_negation_involutive,surreal_real_is_surreal,surreal_stage_characterization,omega_is_ordinal}.
    Hence $r'\in\realUpper{x}$
        by \obv{surreal_real_upper_iff}.
    $\surrealLt{r'}{r}$
        by
        \auto{surreal_negation_lt_reverses,surreal_negation_involutive,surreal_stage_characterization,omega_is_ordinal}.
    Follows by \auto.
\end{proof}

\begin{lemma}\label{surreal_earlier_than_real_finite_stage}
    Suppose $x\in\reals$ and $\surreal{w}$.
    Suppose $\birthday{w}\in\birthday{x}$.
    Then $w\in\surrealStage{\naturals}$.
\end{lemma}
\begin{proof}
    $\birthday{x}\in\suc{\naturals}$
        by
        \auto{surreal_real_elim,surreal_stage_characterization,suc_ordinal,omega_is_ordinal}.
    $\birthday{x}\in\naturals$ or
        $\birthday{x}=\naturals$ by \auto{suc_iff}.
    Show that $\birthday{w}\in\naturals$.
    \begin{subproof}
        \begin{byCase}
            \caseOf{$\birthday{x}=\naturals$.}
                Follows.
            \caseOf{$\birthday{x}\neq\naturals$.}
                Hence $\birthday{x}\in\naturals$.
                Follows by
                    \obv{naturals_is_transitive_set,transitiveset_iff_subseteq,subseteq}.
        \end{byCase}
    \end{subproof}
    Follows by
        \auto{surreal_stage_characterization,omega_is_ordinal}.
\end{proof}

\begin{theorem}\label{surreal_real_canonical_cut}
    Suppose $x\in\reals$.
    Then $x=\surrealCut{\realLower{x}}{\realUpper{x}}$.
\end{theorem}
\begin{proof}
    Let $L=\realLower{x}$.
    Let $R=\realUpper{x}$.
    $\surreal{x}$ by \obv{surreal_real_is_surreal}.
    $\surrealCutPair{L}{R}$
        by \auto{surreal_real_bounds_cut_pair}.
    $\surrealCutPair{\surrealLeft{x}}{\surrealRight{x}}$
        by \obv{surreal_options_form_cut_pair}.
    For every $l\in L$ we have
        $\surrealLt{l}{\surrealCut{\surrealLeft{x}}{\surrealRight{x}}}$
        by \auto{surreal_real_lower_iff,surreal_eta}.
    For every $r\in R$ we have
        $\surrealLt{\surrealCut{\surrealLeft{x}}{\surrealRight{x}}}{r}$
        by \auto{surreal_real_upper_iff,surreal_eta}.
    Show that for every $l\in\surrealLeft{x}$ we have
        $\surrealLt{l}{\surrealCut{L}{R}}$.
    \begin{subproof}
        Fix $l\in\surrealLeft{x}$.
        $\surreal{l}$ and $\birthday{l}\in\birthday{x}$ and
            $\surrealLt{l}{x}$
            by \obv{surreal_left_elim}.
        Hence $l\in\surrealStage{\naturals}$
            by \obv{surreal_earlier_than_real_finite_stage}.
        Hence $l\in L$ by \obv{surreal_real_lower_iff}.
        Follows by \obv{surreal_cut_pair_left}.
    \end{subproof}
    Show that for every $r\in\surrealRight{x}$ we have
        $\surrealLt{\surrealCut{L}{R}}{r}$.
    \begin{subproof}
        Fix $r\in\surrealRight{x}$.
        $\surreal{r}$ and $\birthday{r}\in\birthday{x}$ and
            $\surrealLt{x}{r}$
            by \obv{surreal_right_elim}.
        Hence $r\in\surrealStage{\naturals}$
            by \obv{surreal_earlier_than_real_finite_stage}.
        Hence $r\in R$ by \obv{surreal_real_upper_iff}.
        Follows by \obv{surreal_cut_pair_right}.
    \end{subproof}
    Hence $\surrealCut{L}{R}
        =\surrealCut{\surrealLeft{x}}{\surrealRight{x}}$
        by \obv{surreal_cut_extensionality}.
    Follows by \obv{surreal_eta}.
\end{proof}

\subsection{Dense finite-stage cuts}

\begin{theorem}\label{dense_finite_stage_surreal_cut_is_real}
    Suppose $\surrealCutPair{L}{R}$.
    Suppose $L\union R\subseteq\surrealStage{\naturals}$.
    Suppose $L$ is inhabited and $R$ is inhabited.
    Suppose for every $w\in L$ there exists $w'\in L$ such that
        $\surrealLt{w}{w'}$.
    Suppose for every $z\in R$ there exists $z'\in R$ such that
        $\surrealLt{z'}{z}$.
    Then $\surrealCut{L}{R}\in\reals$.
\end{theorem}
\begin{proof}
    Let $c=\surrealCut{L}{R}$.
    $c\in\surrealStage{\suc{\naturals}}$
        by \auto{finite_stage_surreal_cut_successor_stage}.
    $\surreal{c}$ and $\birthday{c}\in\suc{\naturals}$
        by
        \auto{surreal_stage_characterization,suc_ordinal,omega_is_ordinal}.
    \begin{byCase}
        \caseOf{$c\in\surrealStage{\naturals}$.}
            Follows by \obv{finite_surreal_stage_subset_reals,subseteq}.
        \caseOf{$c\notin\surrealStage{\naturals}$.}
            Hence $\birthday{c}\notin\naturals$
                by
                \auto{surreal_stage_characterization,omega_is_ordinal}.
            $\birthday{c}\in\naturals$ or
                $\birthday{c}=\naturals$ by \auto{suc_iff}.
            Hence $\birthday{c}=\naturals$.
            Take $l$ such that $l\in L$.
            Take $r$ such that $r\in R$.
            $l\in L\union R$ and $r\in L\union R$
                by \auto{union_iff}.
            Hence $l\in\surrealStage{\naturals}$ and
                $r\in\surrealStage{\naturals}$ by \obv{subseteq}.
            Hence $l\in\reals$ and $r\in\reals$
                by \obv{finite_surreal_stage_subset_reals,subseteq}.
            $\surrealLt{l}{c}$ and $\surrealLt{c}{r}$
                by \auto{surreal_cut_pair_left,surreal_cut_pair_right}.
            $\surrealLt
                {\surrealNeg{\ordinalSurreal{\naturals}}}{l}$ and
                $\surrealLt{r}{\ordinalSurreal{\naturals}}$
                by \obv{surreal_real_bounds}.
            Show that $c\neq\ordinalSurreal{\naturals}$.
            \begin{subproof}
                Suppose $c=\ordinalSurreal{\naturals}$.
                Hence
                    $\surrealLt{\ordinalSurreal{\naturals}}{r}$.
                Contradiction by
                    \obv{surreal_lt_asymmetric,surreal_real_is_surreal,ordinal_surreal_is_surreal,omega_is_ordinal}.
            \end{subproof}
            Show that
                $c\neq\surrealNeg{\ordinalSurreal{\naturals}}$.
            \begin{subproof}
                Suppose
                    $c=\surrealNeg{\ordinalSurreal{\naturals}}$.
                Hence
                    $\surrealLt{l}
                        {\surrealNeg{\ordinalSurreal{\naturals}}}$.
                Contradiction by
                    \obv{surreal_lt_asymmetric,surreal_real_is_surreal,surreal_negation_is_surreal,ordinal_surreal_is_surreal,omega_is_ordinal}.
            \end{subproof}
            Show that $\forall q:\Set.\ q\in\surrealStage{\naturals}
                \implies
                ((\forall n:\Set.\ n\in\naturals\implies
                    (\surrealLt{q}
                        {\surrealAdd{c}{\surrealEpsilon{n}}}
                     \land
                     \surrealLt{c}
                        {\surrealAdd{q}{\surrealEpsilon{n}}}))
                 \implies q=c)$.
            \begin{subproof}
                Fix $q$.
                Assume $q\in\surrealStage{\naturals}$.
                Assume for every natural number $n$ we have
                    $\surrealLt{q}
                        {\surrealAdd{c}{\surrealEpsilon{n}}}$ and
                    $\surrealLt{c}
                        {\surrealAdd{q}{\surrealEpsilon{n}}}$.
                $\surreal{q}$ and $\birthday{q}\in\naturals$
                    by
                    \auto{surreal_stage_characterization,omega_is_ordinal}.
                Show that for every $w\in L$ we have $\surrealLt{w}{q}$.
                \begin{subproof}
                    Fix $w\in L$.
                    $w\in L\union R$ by \auto{union_iff}.
                    Hence $w\in\surrealStage{\naturals}$
                        by \obv{subseteq}.
                    Hence $\surreal{w}$
                        by
                        \obv{surreal_stage_characterization,omega_is_ordinal}.
                    $\surrealLt{w}{q}$ or $\surrealLe{q}{w}$
                        by \auto{surreal_le_total,surreal_le}.
                    \begin{byCase}
                        \caseOf{$\surrealLt{w}{q}$.}
                            Follows.
                        \caseOf{not $\surrealLt{w}{q}$.}
                            Hence $\surrealLe{q}{w}$.
                            Take $w'\in L$ such that
                                $\surrealLt{w}{w'}$.
                            $w'\in L\union R$ by \auto{union_iff}.
                            Hence $w'\in\surrealStage{\naturals}$
                                by \obv{subseteq}.
                            Hence $\surreal{w'}$ by
                                \obv{surreal_stage_characterization,omega_is_ordinal}.
                            $\surrealLt{q}{w'}$
                                by \obv{surreal_le_lt_transitive}.
                            Let $d=\surrealAdd{w'}{\surrealNeg{q}}$.
                            $\surrealNeg{q}\in
                                \surrealStage{\naturals}$
                                by \obv{surreal_negation_finite_stage}.
                            Hence $d\in\surrealStage{\naturals}$
                                by \auto{surreal_addition_finite_stage}.
                            Hence $\surreal{d}$ and
                                $\birthday{d}\in\naturals$
                                by
                                \auto{surreal_stage_characterization,omega_is_ordinal}.
                            $\surrealLt{\surrealZero}{d}$
                                by \auto{surreal_subtraction_positive}.
                            $d\in\surrealStage{\suc{\birthday{d}}}$
                                by \obv{surreal_in_successor_stage}.
                            Hence
                                $\surrealLt
                                    {\surrealEpsilon{\birthday{d}}}{d}$
                                by
                                \obv{positive_finite_surreal_dominates_epsilon}.
                            $\surreal{\surrealEpsilon{\birthday{d}}}$ and
                                $\surreal{\surrealAdd{q}
                                    {\surrealEpsilon{\birthday{d}}}}$
                                by
                                \auto{surreal_epsilon_is_surreal,surreal_add_is_surreal}.
                            $\surrealLt{w'}{c}$
                                by \auto{surreal_cut_pair_left}.
                            $\surrealLt{c}
                                {\surrealAdd{q}
                                    {\surrealEpsilon{\birthday{d}}}}$
                                by \auto.
                            Hence $\surrealLt{w'}
                                {\surrealAdd{q}
                                    {\surrealEpsilon{\birthday{d}}}}$
                                by
                                \obv{surreal_lt_transitive,surreal_add_is_surreal,surreal_epsilon_is_surreal}.
                            Hence $\surrealLt{d}
                                {\surrealAdd
                                    {\surrealAdd{q}
                                        {\surrealEpsilon{\birthday{d}}}}
                                    {\surrealNeg{q}}}$
                                by
                                \auto{surreal_add_lt_left,surreal_negation_is_surreal,surreal_add_is_surreal}.
                            Hence $\surrealLt{d}
                                {\surrealEpsilon{\birthday{d}}}$
                                by
                                \auto{surreal_add_associative,surreal_add_commutative,surreal_add_inverse_right,surreal_add_zero_right,surreal_negation_is_surreal,surreal_epsilon_is_surreal,surreal_add_is_surreal}.
                            Contradiction by
                                \obv{surreal_lt_asymmetric,surreal_epsilon_is_surreal}.
                    \end{byCase}
                \end{subproof}
                Show that for every $z\in R$ we have $\surrealLt{q}{z}$.
                \begin{subproof}
                    Fix $z\in R$.
                    $z\in L\union R$ by \auto{union_iff}.
                    Hence $z\in\surrealStage{\naturals}$
                        by \obv{subseteq}.
                    Hence $\surreal{z}$
                        by
                        \auto{surreal_stage_characterization,omega_is_ordinal}.
                    $\surrealLt{q}{z}$ or $\surrealLe{z}{q}$
                        by \auto{surreal_le_total,surreal_le}.
                    \begin{byCase}
                        \caseOf{$\surrealLt{q}{z}$.}
                            Follows.
                        \caseOf{not $\surrealLt{q}{z}$.}
                            Hence $\surrealLe{z}{q}$ by \auto.
                            Take $z'\in R$ such that
                                $\surrealLt{z'}{z}$.
                            $z'\in L\union R$ by \auto{union_iff}.
                            Hence $z'\in\surrealStage{\naturals}$
                                by \obv{subseteq}.
                            Hence $\surreal{z'}$ by
                                \auto{surreal_stage_characterization,omega_is_ordinal}.
                            $\surrealLt{z'}{q}$
                                by \auto{surreal_lt_le_transitive}.
                            Let $d=\surrealAdd{q}{\surrealNeg{z'}}$.
                            $\surrealNeg{z'}\in
                                \surrealStage{\naturals}$
                                by \obv{surreal_negation_finite_stage}.
                            Hence $d\in\surrealStage{\naturals}$
                                by \auto{surreal_addition_finite_stage}.
                            Hence $\surreal{d}$ and
                                $\birthday{d}\in\naturals$
                                by
                                \auto{surreal_stage_characterization,omega_is_ordinal}.
                            $\surrealLt{\surrealZero}{d}$
                                by \auto{surreal_subtraction_positive}.
                            $d\in\surrealStage{\suc{\birthday{d}}}$
                                by \obv{surreal_in_successor_stage}.
                            Hence
                                $\surrealLt
                                    {\surrealEpsilon{\birthday{d}}}{d}$
                                by
                                \obv{positive_finite_surreal_dominates_epsilon}.
                            $\surreal{\surrealEpsilon{\birthday{d}}}$ and
                                $\surreal{\surrealAdd{c}
                                    {\surrealEpsilon{\birthday{d}}}}$
                                by
                                \auto{surreal_epsilon_is_surreal,surreal_add_is_surreal,surreal_cut_pair_is_surreal}.
                            $\surrealLt{c}{z'}$
                                by \auto{surreal_cut_pair_right}.
                            Hence
                                $\surrealLt{\surrealNeg{z'}}
                                    {\surrealNeg{c}}$
                                by
                                \obv{surreal_negation_lt_reverses,surreal_cut_pair_is_surreal,surreal_negation_is_surreal}.
                            $\surreal{\surrealNeg{z'}}$ and
                                $\surreal{\surrealNeg{c}}$
                                by
                                \obv{surreal_negation_is_surreal,surreal_cut_pair_is_surreal}.
                            Hence $\surrealLt{d}
                                {\surrealAdd{q}{\surrealNeg{c}}}$
                                by \auto{surreal_add_lt_right}.
                            $\surrealLt{q}
                                {\surrealAdd{c}
                                    {\surrealEpsilon{\birthday{d}}}}$
                                by \auto.
                            Hence
                                $\surrealLt
                                    {\surrealAdd{q}{\surrealNeg{c}}}
                                    {\surrealAdd
                                        {\surrealAdd{c}
                                            {\surrealEpsilon{\birthday{d}}}}
                                        {\surrealNeg{c}}}$
                                by
                                \obv{surreal_add_lt_left,surreal_negation_is_surreal,surreal_add_is_surreal}.
                            Hence
                                $\surrealLt
                                    {\surrealAdd{q}{\surrealNeg{c}}}
                                    {\surrealEpsilon{\birthday{d}}}$
                                by
                                \auto{surreal_add_associative,surreal_add_commutative,surreal_add_inverse_right,surreal_add_zero_right,surreal_negation_is_surreal,surreal_epsilon_is_surreal,surreal_add_is_surreal}.
                            Hence $\surrealLt{d}
                                {\surrealEpsilon{\birthday{d}}}$
                                by
                                \auto{surreal_lt_transitive,surreal_add_is_surreal,surreal_negation_is_surreal,surreal_epsilon_is_surreal}.
                            Contradiction by
                                \obv{surreal_lt_asymmetric,surreal_epsilon_is_surreal}.
                    \end{byCase}
                \end{subproof}
                Hence $\birthday{c}\subseteq\birthday{q}$
                    by \auto{surreal_cut_pair_simplest}.
                Hence $\naturals\subseteq\birthday{q}$
                    by \auto.
                Hence $\birthday{q}\in\birthday{q}$
                    by \obv{subseteq}.
                Contradiction by \auto{in_irrefl}.
            \end{subproof}
            Follows by \auto{surreal_real_intro}.
    \end{byCase}
\end{proof}

\begin{corollary}\label{dense_finite_stage_real_cut_is_real}
    Suppose $L\union R\subseteq\reals$.
    Suppose $L\union R\subseteq\surrealStage{\naturals}$.
    Suppose $\surrealCutPair{L}{R}$.
    Suppose $L$ is inhabited and $R$ is inhabited.
    Suppose for every $w\in L$ there exists $w'\in L$ such that
        $\surrealLt{w}{w'}$.
    Suppose for every $z\in R$ there exists $z'\in R$ such that
        $\surrealLt{z'}{z}$.
    Then $\surrealCut{L}{R}\in\reals$.
\end{corollary}
\begin{proof}
    Follows by \obv{dense_finite_stage_surreal_cut_is_real}.
\end{proof}
