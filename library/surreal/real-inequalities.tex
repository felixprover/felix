\section{Elementary real inequalities}

\begin{theorem}\label{real_affine_unit_interval_range}
    Let $a,b,u\in\reals$.
    Suppose $\surrealLe{a}{b}$ and
        $\surrealLe{\surrealZero}{u}$ and
        $\surrealLe{u}{\surrealOne}$.
    Let $d=\surrealAdd{b}{\surrealNeg{a}}$.
    Then
        $\surrealLe{a}{\surrealAdd{a}{\surrealMul{d}{u}}}$ and
        $\surrealLe{\surrealAdd{a}{\surrealMul{d}{u}}}{b}$.
\end{theorem}
\begin{proof}
    $\surrealNeg{a}\in\reals$ by \obv{surreal_real_negation_closed}.
    Hence $d\in\reals$ by \auto{surreal_real_addition_closed}.
    $\surrealLe{\surrealAdd{a}{\surrealNeg{a}}}
        {\surrealAdd{b}{\surrealNeg{a}}}$
        by \obv{surreal_add_le_left,surreal_real_is_surreal}.
    $\surrealAdd{a}{\surrealNeg{a}}=\surrealZero$
        by
        \obv{surreal_add_inverse_right,surreal_real_is_surreal}.
    Hence $\surrealLe{\surrealZero}{d}$.
    Hence $\surrealLe{\surrealZero}{\surrealMul{d}{u}}$
        by
        \obv{surreal_mul_nonneg_nonneg,surreal_real_is_surreal,empty_surreal}.
    Hence
        $\surrealLe{\surrealAdd{a}{\surrealZero}}
            {\surrealAdd{a}{\surrealMul{d}{u}}}$
        by
        \obv{surreal_add_le_right,surreal_mul_is_surreal,surreal_real_is_surreal,empty_surreal}.
    Hence $\surrealLe{a}{\surrealAdd{a}{\surrealMul{d}{u}}}$
        by \obv{surreal_add_zero_right,surreal_real_is_surreal}.
    $\surrealLe{\surrealMul{d}{u}}{\surrealMul{d}{\surrealOne}}$
        by
        \obv{surreal_nonneg_mul_le_left,surreal_real_is_surreal,surreal_one_is_surreal}.
    $\surrealMul{d}{\surrealOne}=d$
        by \obv{surreal_mul_one_right,surreal_real_is_surreal}.
    Hence $\surrealLe{\surrealMul{d}{u}}{d}$.
    Hence
        $\surrealLe{\surrealAdd{a}{\surrealMul{d}{u}}}
            {\surrealAdd{a}{d}}$
        by
        \obv{surreal_add_le_right,surreal_mul_is_surreal,surreal_real_is_surreal}.
    $\surrealAdd{a}{d}
        =\surrealAdd{a}{\surrealAdd{b}{\surrealNeg{a}}}$.
    $\surrealAdd{a}{\surrealAdd{b}{\surrealNeg{a}}}=b$
        by
        \auto{surreal_add_associative,surreal_add_commutative,surreal_add_inverse_right,surreal_add_zero_left,surreal_negation_is_surreal,surreal_real_is_surreal}.
    Hence $\surrealAdd{a}{d}=b$.
\end{proof}

\begin{corollary}\label{real_affine_unit_interval_abs_bound}
    Let $r,u\in\reals$.
    Suppose $\surrealLe{\surrealZero}{r}$ and
        $\surrealLe{\surrealZero}{u}$ and
        $\surrealLe{u}{\surrealOne}$.
    Let $a=\surrealNeg{r}$.
    Let $d=\surrealAdd{r}{\surrealNeg{a}}$.
    Then $\surrealLe{\abs{\surrealAdd{a}{\surrealMul{d}{u}}}}{r}$.
\end{corollary}
\begin{proof}
    $a\in\reals$ by \auto{surreal_real_negation_closed}.
    $\surrealLe{\surrealNeg{r}}{\surrealNeg{\surrealZero}}$
        by
        \obv{surreal_negation_le_reverses,surreal_real_is_surreal,empty_surreal}.
    $\surrealNeg{\surrealZero}=\surrealZero$
        by \obv{surreal_negation_empty}.
    Hence $\surrealLe{a}{\surrealZero}$.
    $\surrealLe{\surrealZero}{r}$ by assumption.
    Hence $\surrealLe{a}{r}$
        by \obv{surreal_le_transitive,surreal_real_is_surreal,empty_surreal}.
    Hence
        $\surrealLe{a}{\surrealAdd{a}{\surrealMul{d}{u}}}$ and
        $\surrealLe{\surrealAdd{a}{\surrealMul{d}{u}}}{r}$
        by \obv{real_affine_unit_interval_range}.
    Let $z=\surrealAdd{a}{\surrealMul{d}{u}}$.
    $z\in\reals$
        by
        \auto{surreal_real_negation_closed,surreal_real_addition_closed,surreal_real_multiplication_closed}.
    $\surrealLe{\surrealNeg{z}}{\surrealNeg{a}}$
        by \obv{surreal_negation_le_reverses,surreal_real_is_surreal}.
    $\surrealNeg{a}=r$
        by
        \auto{surreal_negation_involutive,surreal_real_is_surreal}.
    Hence $\surrealLe{\surrealNeg{z}}{r}$.
    Follows by
        \auto{surreal_abs_le_iff,surreal_real_is_surreal}.
\end{proof}
