\section{The native field of surreal reals}

\import{surreal/real-field.tex}
\import{algebra/ordered.tex}
\import{set/logical-graph.tex}

The logical surreal operations are restricted to the real carrier to form
the set-coded operation graphs required by a native field.  The primitive
record order is non-strict; the abstract strict order obtained by deleting
its diagonal agrees with the ambient strict surreal order.

\begin{definition}\label{surreal_real_negation_op}
    $\surrealRealNegationOp:\Set\to\Set
        =\fun{x:\Set}{\surrealNeg{x}}$.
\end{definition}

\begin{definition}\label{surreal_real_nonstrict_order_op}
    $\surrealRealNonstrictOrderOp:\Set\to\Set\to\Prop
        =\fun{x,y:\Set}{\surrealLe{x}{y}}$.
\end{definition}

\begin{definition}\label{surreal_real_addition_graph}
    $\surrealRealAdditionGraph=
        \Apply{\binaryGraphOnOp}
            {\reals,\reals,\surrealAddOp}$.
\end{definition}

\begin{definition}\label{surreal_real_negation_graph}
    $\surrealRealNegationGraph=
        \Apply{\graphOnOp}
            {\reals,\reals,\surrealRealNegationOp}$.
\end{definition}

\begin{definition}\label{surreal_real_multiplication_graph}
    $\surrealRealMultiplicationGraph=
        \Apply{\binaryGraphOnOp}
            {\reals,\reals,\surrealMulOp}$.
\end{definition}

\begin{definition}\label{surreal_real_inverse_graph}
    $\surrealRealInverseGraph=
        \Apply{\graphOnOp}
            {\reals,\reals,\surrealReciprocalOp}$.
\end{definition}

\begin{definition}\label{surreal_real_nonstrict_order}
    $\surrealRealNonstrictOrder=
        \Apply{\relationOnOp}
            {\reals,\surrealRealNonstrictOrderOp}$.
\end{definition}

\begin{proposition}\label{surreal_real_nonstrict_order_iff}
    Let $x,y\in\reals$.
    Then
        $x\mathrel{\surrealRealNonstrictOrder}y$ iff
        $\surrealLe{x}{y}$.
\end{proposition}
\begin{proof}

\end{proof}

\begin{theorem}\label{surreal_real_nonstrict_order_is_order}
    $\surrealRealNonstrictOrder$ is an order on $\reals$.
\end{theorem}
\begin{proof}
    $\surrealRealNonstrictOrder$ is a binary relation on $\reals$
        by \obv{surreal_real_nonstrict_order_type}.
    $\surrealRealNonstrictOrder$ is reflexive on $\reals$
        by
        \auto{reflexive_on,surreal_real_nonstrict_order_iff,surreal_real_is_surreal,surreal_le_reflexive}.
    $\surrealRealNonstrictOrder$ is transitive
        by
        \auto{transitive,surreal_real_nonstrict_order_iff,surreal_real_nonstrict_order_type,subseteq,times_tuple_elim,surreal_real_is_surreal,surreal_le_transitive}.
    $\surrealRealNonstrictOrder$ is antisymmetric
        by
        \auto{antisymmetric,surreal_real_nonstrict_order_iff,surreal_real_nonstrict_order_type,subseteq,times_tuple_elim,surreal_real_is_surreal,surreal_le_antisymmetric}.
\end{proof}

\begin{proposition}\label{surreal_real_nonstrict_order_total}
    For all $x,y\in\reals$ we have
        $x\mathrel{\surrealRealNonstrictOrder}y$ or
        $y\mathrel{\surrealRealNonstrictOrder}x$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{surreal_real_nonstrict_order_iff,surreal_real_lt_trichotomy,surreal_le}.
\end{proof}

\begin{proposition}\label{surreal_real_strict_order_iff}
    Let $x,y\in\reals$.
    Then
        $x\mathrel{\tostrictorder{\surrealRealNonstrictOrder}}y$ iff
        $\surrealLt{x}{y}$.
\end{proposition}
\begin{proof}
    $\surreal{x}$ and $\surreal{y}$
        by \obv{surreal_real_is_surreal}.
    Follows by
        \auto{tostrictorder_iff,surreal_real_nonstrict_order_iff,surreal_le,surreal_lt_irreflexive}.
\end{proof}

\begin{proposition}\label{surreal_real_nonstrict_order_addition}
    Let $x,y,z\in\reals$.
    Suppose $x\mathrel{\surrealRealNonstrictOrder}y$.
    Then $\surrealAdd{x}{z}
            \mathrel{\surrealRealNonstrictOrder}
        \surrealAdd{y}{z}$.
\end{proposition}
\begin{proof}
    $(\surrealAdd{x}{z}\in\reals\land
        \surrealAdd{y}{z}\in\reals)$
        by \obv{surreal_real_addition_closed}.
    Follows by
        \auto{surreal_real_nonstrict_order_iff,surreal_real_is_surreal,surreal_add_le_left}.
\end{proof}

\begin{proposition}\label{surreal_real_nonstrict_order_multiplication}
    Let $x,y\in\reals$.
    Suppose $\surrealZero\mathrel{\surrealRealNonstrictOrder}x$ and
        $\surrealZero\mathrel{\surrealRealNonstrictOrder}y$.
    Then $\surrealZero\mathrel{\surrealRealNonstrictOrder}
        \surrealMul{x}{y}$.
\end{proposition}
\begin{proof}
    $\surrealZero\in\reals$ by \obv{surreal_zero_is_real}.
    $\surrealMul{x}{y}\in\reals$
        by \obv{surreal_real_multiplication_closed}.
    Follows by
        \obv{surreal_real_nonstrict_order_iff,surreal_real_is_surreal,surreal_mul_nonneg_nonneg}.
\end{proof}

\begin{instance}\label{surreal_real_ordered_field}
    The ordered field
    $\RealField=
        \structure[ordered field]{
            \carrier=\reals,
            \ringAdd=\surrealRealAdditionGraph,
            \ringZero=\surrealZero,
            \ringNeg=\surrealRealNegationGraph,
            \ringMul=\surrealRealMultiplicationGraph,
            \ringOne=\surrealOne,
            \ringInv=\surrealRealInverseGraph,
            \leq=\surrealRealNonstrictOrder
        }$.
\end{instance}
\begin{proof}
    $\carrier[\RealField]=\reals$
        by \obv{ordered_fieldaggregateprojection_carrier}.
    $\ringAdd[\RealField]=\surrealRealAdditionGraph$
        by \obv{ordered_fieldaggregateprojection_ringAdd}.
    $\ringZero[\RealField]=\surrealZero$
        by \obv{ordered_fieldaggregateprojection_ringZero}.
    $\ringNeg[\RealField]=\surrealRealNegationGraph$
        by \obv{ordered_fieldaggregateprojection_ringNeg}.
    $\ringMul[\RealField]=\surrealRealMultiplicationGraph$
        by \obv{ordered_fieldaggregateprojection_ringMul}.
    $\ringOne[\RealField]=\surrealOne$
        by \obv{ordered_fieldaggregateprojection_ringOne}.
    $\ringInv[\RealField]=\surrealRealInverseGraph$
        by \obv{ordered_fieldaggregateprojection_ringInv}.
    $\leq[\RealField]=\surrealRealNonstrictOrder$
        by \obv{ordered_fieldaggregateprojection_leq}.

    $\ringAdd[\RealField]
        \in\funs
            {\carrier[\RealField]\times\carrier[\RealField]}
            {\carrier[\RealField]}$
        by \obv{surreal_real_addition_graph_type}.
    $\ringZero[\RealField]\in\carrier[\RealField]$
        by \obv{surreal_zero_is_real}.
    $\ringNeg[\RealField]
        \in\funs{\carrier[\RealField]}{\carrier[\RealField]}$
        by \obv{surreal_real_negation_graph_type}.
    $\ringMul[\RealField]
        \in\funs
            {\carrier[\RealField]\times\carrier[\RealField]}
            {\carrier[\RealField]}$
        by \obv{surreal_real_multiplication_graph_type}.
    $\ringOne[\RealField]\in\carrier[\RealField]$
        by \obv{surreal_one_is_real}.
    $\ringInv[\RealField]
        \in\funs{\carrier[\RealField]}{\carrier[\RealField]}$
        by \obv{surreal_real_inverse_graph_type}.
    $\leq[\RealField]
        \subseteq
        \carrier[\RealField]\times\carrier[\RealField]$
        by \obv{surreal_real_nonstrict_order_type}.

    For all $x,y\in\carrier[\RealField]$ we have
        $\ringAdd[\RealField](x,y)=\surrealAdd{x}{y}$
        by \auto{surreal_real_addition_graph_application}.
    For all $x\in\carrier[\RealField]$ we have
        $\ringNeg[\RealField](x)=\surrealNeg{x}$
        by \auto{surreal_real_negation_graph_application}.
    For all $x,y\in\carrier[\RealField]$ we have
        $\ringMul[\RealField](x,y)=\surrealMul{x}{y}$
        by \auto{surreal_real_multiplication_graph_application}.
    For all $x\in\carrier[\RealField]$ we have
        $\ringInv[\RealField](x)=\surrealReciprocal{x}$
        by \auto{surreal_real_inverse_graph_application}.
    For all $x,y\in\carrier[\RealField]$ we have
        $x\mathrel{\leq[\RealField]}y$ iff $\surrealLe{x}{y}$
        by \obv{surreal_real_nonstrict_order_iff}.
    For all $x,y\in\carrier[\RealField]$ we have
        $x\mathrel{\ringStrictOrder[\RealField]}y$ iff
        $\surrealLt{x}{y}$
        by
        \auto{ordered_algebra_strict_relation,surreal_real_strict_order_iff}.

    For all $x,y\in\carrier[\RealField]$ we have
        $\ringAdd[\RealField](x,y)\in\carrier[\RealField]$
        by \auto{surreal_real_addition_closed}.
    For all $x\in\carrier[\RealField]$ we have
        $\ringNeg[\RealField](x)\in\carrier[\RealField]$
        by \auto{surreal_real_negation_closed}.
    For all $x,y\in\carrier[\RealField]$ we have
        $\ringMul[\RealField](x,y)\in\carrier[\RealField]$
        by \auto{surreal_real_multiplication_closed}.
    For all $x\in\carrier[\RealField]$ we have
        $\ringInv[\RealField](x)\in\carrier[\RealField]$
        by \auto{surreal_real_reciprocal_closed}.

    For all $x,y,z\in\carrier[\RealField]$ we have
        $\ringAdd[\RealField](x,\ringAdd[\RealField](y,z))
            =\ringAdd[\RealField](\ringAdd[\RealField](x,y),z)$
        by \auto{surreal_add_associative,surreal_real_is_surreal}.
    For all $x,y\in\carrier[\RealField]$ we have
        $\ringAdd[\RealField](x,y)=\ringAdd[\RealField](y,x)$
        by \auto{surreal_add_commutative,surreal_real_is_surreal}.
    For all $x\in\carrier[\RealField]$ we have
        $\ringAdd[\RealField](\ringZero[\RealField],x)=x$
        by \auto{surreal_add_zero_left,surreal_real_is_surreal,empty_surreal}.
    For all $x\in\carrier[\RealField]$ we have
        $\ringAdd[\RealField](\ringNeg[\RealField](x),x)
            =\ringZero[\RealField]$
        by
        \auto{surreal_add_inverse_left,surreal_real_is_surreal,empty_surreal}.
    For all $x,y,z\in\carrier[\RealField]$ we have
        $\ringMul[\RealField](x,\ringMul[\RealField](y,z))
            =\ringMul[\RealField](\ringMul[\RealField](x,y),z)$
        by \auto{surreal_mul_associative,surreal_real_is_surreal}.
    For all $x\in\carrier[\RealField]$ we have
        $\ringMul[\RealField](\ringOne[\RealField],x)=x$
        by \auto{surreal_mul_one_left,surreal_real_is_surreal}.
    For all $x\in\carrier[\RealField]$ we have
        $\ringMul[\RealField](x,\ringOne[\RealField])=x$
        by \auto{surreal_mul_one_right,surreal_real_is_surreal}.
    For all $x,y,z\in\carrier[\RealField]$ we have
        $\ringMul[\RealField](x,\ringAdd[\RealField](y,z))
            =\ringAdd[\RealField]
                (\ringMul[\RealField](x,y),
                 \ringMul[\RealField](x,z))$
        by
        \auto{surreal_mul_distributive_left,surreal_real_is_surreal}.
    Show that for all $x,y,z\in\carrier[\RealField]$ we have
        $\ringMul[\RealField](\ringAdd[\RealField](x,y),z)
            =\ringAdd[\RealField]
                (\ringMul[\RealField](x,z),
                 \ringMul[\RealField](y,z))$.
    \begin{subproof}
        Fix $x,y,z\in\carrier[\RealField]$.
        Then $x,y,z\in\reals$ by \auto.
        Hence $\surreal{x}$ and $\surreal{y}$ and $\surreal{z}$
            by \obv{surreal_real_is_surreal}.
        $\ringMul[\RealField](\ringAdd[\RealField](x,y),z)
            =\surrealMul{\surrealAdd{x}{y}}{z}$ by \auto.
        Hence $\ringMul[\RealField](\ringAdd[\RealField](x,y),z)
            =\surrealAdd{\surrealMul{x}{z}}{\surrealMul{y}{z}}$
            by \auto{surreal_mul_distributive_right}.
        Follows by \auto.
    \end{subproof}

    $\ringZero[\RealField]\neq\ringOne[\RealField]$
        by
        \auto{surreal_zero_lt_one,surreal_lt_irreflexive,empty_surreal}.
    $\ringInv[\RealField](\ringZero[\RealField])
        =\ringZero[\RealField]$
        by \auto{surreal_reciprocal_zero}.
    For all $x\in\carrier[\RealField]$ such that
        $x\neq\ringZero[\RealField]$ we have
        $\ringMul[\RealField](\ringInv[\RealField](x),x)
            =\ringOne[\RealField]$
        by
        \auto{surreal_reciprocal_inverse_right,surreal_real_is_surreal}.
    For all $x\in\carrier[\RealField]$ such that
        $x\neq\ringZero[\RealField]$ we have
        $\ringMul[\RealField](x,\ringInv[\RealField](x))
            =\ringOne[\RealField]$
        by
        \auto{surreal_reciprocal_inverse,surreal_real_is_surreal}.
    For all $x,y\in\carrier[\RealField]$ we have
        $\ringMul[\RealField](x,y)=\ringMul[\RealField](y,x)$
        by \auto{surreal_mul_commutative,surreal_real_is_surreal}.

    $\leq[\RealField]$ is an order on $\carrier[\RealField]$
        by \auto{surreal_real_nonstrict_order_is_order}.
    For all $x,y\in\carrier[\RealField]$ we have
        $x\mathrel{\leq[\RealField]}y$ or
        $y\mathrel{\leq[\RealField]}x$
        by \auto{surreal_real_nonstrict_order_total}.
    For all $x,y,z\in\carrier[\RealField]$ such that
        $x\mathrel{\leq[\RealField]}y$ we have
        $\ringAdd[\RealField](x,z)
            \mathrel{\leq[\RealField]}
        \ringAdd[\RealField](y,z)$
        by \auto{surreal_real_nonstrict_order_addition}.
    For all $x,y\in\carrier[\RealField]$ such that
        $\ringZero[\RealField]\mathrel{\leq[\RealField]}x$ and
        $\ringZero[\RealField]\mathrel{\leq[\RealField]}y$ we have
        $\ringZero[\RealField]\mathrel{\leq[\RealField]}
            \ringMul[\RealField](x,y)$
        by \auto{surreal_real_nonstrict_order_multiplication}.

\end{proof}

\begin{proposition}\label{surreal_real_field_natural}
    Let $n$ be a natural number.
    Then $\ringNatural{\RealField}{n}=\ordinalSurreal{n}$.
\end{proposition}
\begin{proof}
    $\RealField$ is an ordered field by
        \obv{surreal_real_ordered_field}.
    Hence $\RealField$ is a ring by
        \auto{ordered_field,field,division_ring}.
    Let
        $C=\{k\in\naturals\mid
            \ringNatural{\RealField}{k}=\ordinalSurreal{k}\}$.
    $\emptyset\in C$
        by
        \auto{omega_empty,ring_natural_zero,surreal_real_field_zero,empty_ordinal_surreal}.
    Show that for every $k\in C$ we have $\suc{k}\in C$.
    \begin{subproof}
        Fix $k\in C$.
        Then $k\in\naturals$ and
            $\ringNatural{\RealField}{k}=\ordinalSurreal{k}$.
        $\ordinalSurreal{k}\in\surrealStage{\naturals}$
            by
            \auto{surreal_stage_characterization,ordinal_surreal_is_surreal,ordinal_surreal_birthday,natural_number_is_ordinal,omega_is_ordinal}.
        Hence $\ordinalSurreal{k}\in\reals$
            by \obv{finite_surreal_stage_subset_reals,subseteq}.
        Hence $\ringNatural{\RealField}{k}\in\reals$.
        $\surrealOne\in\reals$ by \obv{surreal_one_is_real}.
        Hence $\ringOne[\RealField]\in\reals$
            by \obv{surreal_real_field_one}.
        $\ringNatural{\RealField}{\suc{k}}
            =\ringAdd[\RealField]
                (\ringNatural{\RealField}{k},\ringOne[\RealField])$
            by \obv{ring_natural_successor}.
        Hence
            $\ringNatural{\RealField}{\suc{k}}
                =\surrealAdd{\ordinalSurreal{k}}{\surrealOne}$
            by
            \auto{surreal_real_field_addition,surreal_real_field_one}.
        Hence
            $\ringNatural{\RealField}{\suc{k}}
                =\ordinalSurreal{\suc{k}}$
            by
            \auto{ordinal_surreal_successor_addition,natural_number_is_ordinal}.
        $\suc{k}\in\naturals$ by \obv{omega_successor}.
    \end{subproof}
    Hence $C$ is an inductive set by \obv{inductive_set}.
    Hence every natural number is an element of $C$
        by \obv{omega_minimal}.
    Thus $n\in C$.
\end{proof}

\begin{proposition}\label{surreal_real_field_archimedean_law}
    Let $x,y\in\carrier[\RealField]$.
    Suppose $\ringZero[\RealField]
        \mathrel{\ringStrictOrder[\RealField]}x$ and
        $\ringZero[\RealField]\mathrel{\leq[\RealField]}y$.
    Then there exists a natural number $n$ such that
        $y\mathrel{\leq[\RealField]}
            \ringMul[\RealField](\ringNatural{\RealField}{n},x)$.
\end{proposition}
\begin{proof}
    $x,y\in\reals$ by \obv{surreal_real_field_carrier}.
    $\surrealZero\in\reals$ by \obv{surreal_zero_is_real}.
    $\surrealLt{\surrealZero}{x}$
        by
        \auto{surreal_real_field_zero,surreal_real_field_strict_order}.
    $\surrealLe{\surrealZero}{y}$
        by \auto{surreal_real_field_zero,surreal_real_field_order}.
    Take $n\in\naturals$ such that
        $\surrealLe{y}{\surrealMul{\ordinalSurreal{n}}{x}}$
        by \obv{surreal_reals_archimedean}.
    $\ordinalSurreal{n}\in\surrealStage{\naturals}$
        by
        \auto{surreal_stage_characterization,ordinal_surreal_is_surreal,ordinal_surreal_birthday,natural_number_is_ordinal,omega_is_ordinal}.
    Hence $\ordinalSurreal{n}\in\reals$
        by \obv{finite_surreal_stage_subset_reals,subseteq}.
    $\ringNatural{\RealField}{n}=\ordinalSurreal{n}$
        by \obv{surreal_real_field_natural}.
    Hence
        $\ringMul[\RealField](\ringNatural{\RealField}{n},x)
            =\surrealMul{\ordinalSurreal{n}}{x}$
        by \auto{surreal_real_field_multiplication}.
    Hence
        $y\mathrel{\leq[\RealField]}
            \ringMul[\RealField](\ringNatural{\RealField}{n},x)$
        by \auto{surreal_real_field_order,surreal_real_multiplication_closed}.
    Follows by \auto.
\end{proof}

\begin{proposition}\label{surreal_real_field_completeness_law}
    Let $A$ be a subset of $\carrier[\RealField]$.
    Suppose $A$ is inhabited and there exists
        $b\in\carrier[\RealField]$ such that for every $a\in A$
        we have $a\mathrel{\leq[\RealField]}b$.
    Then there exists $s\in\carrier[\RealField]$ such that
        (for every $a\in A$ we have
            $a\mathrel{\leq[\RealField]}s$) and
        (for every $u\in\carrier[\RealField]$ such that
            for every $r\in A$ we have
            $r\mathrel{\leq[\RealField]}u$ we have
            $s\mathrel{\leq[\RealField]}u$).
\end{proposition}
\begin{proof}
    $A\subseteq\reals$ by \obv{surreal_real_field_carrier}.
    Take $b\in\carrier[\RealField]$ such that for every $a\in A$
        we have $a\mathrel{\leq[\RealField]}b$.
    $b\in\reals$ by \obv{surreal_real_field_carrier}.
    Show that for every $a\in A$ we have $\surrealLe{a}{b}$.
    \begin{subproof}
        Fix $a\in A$.
        $a\in\reals$ by \obv{subseteq}.
        Follows by \auto{surreal_real_field_order}.
    \end{subproof}
    Take $s\in\reals$ such that
        $(\forall a:\Set.\ a\in A\implies\surrealLe{a}{s})\land
        (\forall u:\Set.
            (u\in\reals\land
             (\forall a:\Set.\ a\in A\implies\surrealLe{a}{u}))
            \implies\surrealLe{s}{u})$
        by \auto{surreal_reals_least_upper_bound}.
    $s\in\carrier[\RealField]$ by \obv{surreal_real_field_carrier}.
    Show that for every $a\in A$ we have
        $a\mathrel{\leq[\RealField]}s$.
    \begin{subproof}
        Fix $a\in A$.
        $a\in\reals$ by \obv{subseteq}.
        Follows by \auto{surreal_real_field_order}.
    \end{subproof}
    Show that
        $\forall u:\Set.
            (u\in\carrier[\RealField]\land
             (\forall a:\Set.\ a\in A\implies
                a\mathrel{\leq[\RealField]}u))
            \implies s\mathrel{\leq[\RealField]}u$.
    \begin{subproof}
        Fix $u$.
        Assume $u\in\carrier[\RealField]$ and for every $a\in A$
            we have $a\mathrel{\leq[\RealField]}u$.
        $u\in\reals$ by \obv{surreal_real_field_carrier}.
        Show that for every $a\in A$ we have $\surrealLe{a}{u}$.
        \begin{subproof}
            Fix $a\in A$.
            $a\in\reals$ by \obv{subseteq}.
            Follows by \auto{surreal_real_field_order}.
        \end{subproof}
        $\surrealLe{s}{u}$.
        Follows by \obv{surreal_real_field_order}.
    \end{subproof}
    Follows by \auto.
\end{proof}
