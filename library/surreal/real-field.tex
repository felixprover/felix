\section{The complete Archimedean ordered field of surreal reals}

The carrier-parametric predicates below collect the checked field, order,
Archimedean, and completeness laws for canonical surreal operations. They
remain useful as guarded support interfaces. The companion native-structure
module packages their canonical realization as an ordered-field record and
proves its Archimedean and completeness refinements.

\begin{definition}\label{surreal_field_carrier}
    $\surrealFieldCarrier{R}$ iff
        $\surrealZero,\surrealOne\in R$ and
        $\surrealZero\neq\surrealOne$ and
        (for every $x\in R$ we have
            $\surreal{x}$ and
            $\surrealNeg{x},\surrealReciprocal{x}\in R$ and
            $\surrealAdd{x}{\surrealZero}=x$ and
            $\surrealAdd{x}{\surrealNeg{x}}=\surrealZero$ and
            $\surrealMul{x}{\surrealOne}=x$ and
            $x\neq\surrealZero\implies
                \surrealMul{x}{\surrealReciprocal{x}}=\surrealOne$) and
        (for every $x,y\in R$ we have
            $\surrealAdd{x}{y},\surrealMul{x}{y},\surrealDiv{x}{y}\in R$ and
            $\surrealAdd{x}{y}=\surrealAdd{y}{x}$ and
            $\surrealMul{x}{y}=\surrealMul{y}{x}$) and
        (for every $x,y,z\in R$ we have
            $\surrealAdd{x}{\surrealAdd{y}{z}}
                =\surrealAdd{\surrealAdd{x}{y}}{z}$ and
            $\surrealMul{x}{\surrealMul{y}{z}}
                =\surrealMul{\surrealMul{x}{y}}{z}$ and
            $\surrealMul{x}{\surrealAdd{y}{z}}
                =\surrealAdd{\surrealMul{x}{y}}{\surrealMul{x}{z}}$).
\end{definition}

\begin{theorem}\label{surreal_reals_field_carrier}
    $\surrealFieldCarrier{\reals}$.
\end{theorem}
\begin{proof}
    $\surrealZero\in\reals$ and $\surrealOne\in\reals$
        by \obv{surreal_zero_is_real,surreal_one_is_real}.
    $\surrealZero\neq\surrealOne$
        by \obv{surreal_zero_lt_one,surreal_lt_irreflexive,empty_surreal}.

    Show that for every $x\in\reals$ we have
        $\surreal{x}$ and
        $\surrealNeg{x},\surrealReciprocal{x}\in\reals$ and
        $\surrealAdd{x}{\surrealZero}=x$ and
        $\surrealAdd{x}{\surrealNeg{x}}=\surrealZero$ and
        $\surrealMul{x}{\surrealOne}=x$ and
        $x\neq\surrealZero\implies
            \surrealMul{x}{\surrealReciprocal{x}}=\surrealOne$.
    \begin{subproof}
        Fix $x\in\reals$.
        $\surreal{x}$ by \obv{surreal_real_is_surreal}.
        $\surrealNeg{x}\in\reals$
            by \obv{surreal_real_negation_closed}.
        $\surrealReciprocal{x}\in\reals$
            by \obv{surreal_real_reciprocal_closed}.
        $\surrealAdd{x}{\surrealZero}=x$
            by \obv{surreal_add_zero_right,empty_surreal}.
        $\surrealAdd{x}{\surrealNeg{x}}=\surrealZero$
            by \obv{surreal_add_inverse_right,empty_surreal}.
        $\surrealMul{x}{\surrealOne}=x$
            by \obv{surreal_mul_one_right}.
        If $x\neq\surrealZero$ then
            $\surrealMul{x}{\surrealReciprocal{x}}=\surrealOne$
            by \obv{surreal_reciprocal_inverse}.
    \end{subproof}

    Show that for every $x,y\in\reals$ we have
        $\surrealAdd{x}{y},\surrealMul{x}{y},\surrealDiv{x}{y}\in\reals$ and
        $\surrealAdd{x}{y}=\surrealAdd{y}{x}$ and
        $\surrealMul{x}{y}=\surrealMul{y}{x}$.
    \begin{subproof}
        Fix $x,y\in\reals$.
        $\surreal{x}$ and $\surreal{y}$
            by \obv{surreal_real_is_surreal}.
        $\surrealAdd{x}{y}\in\reals$
            by \obv{surreal_real_addition_closed}.
        $\surrealMul{x}{y}\in\reals$
            by \obv{surreal_real_multiplication_closed}.
        $\surrealDiv{x}{y}\in\reals$
            by \obv{surreal_real_division_closed}.
        $\surrealAdd{x}{y}=\surrealAdd{y}{x}$
            by \obv{surreal_add_commutative}.
        $\surrealMul{x}{y}=\surrealMul{y}{x}$
            by \obv{surreal_mul_commutative}.
    \end{subproof}

    Show that for every $x,y,z\in\reals$ we have
        $\surrealAdd{x}{\surrealAdd{y}{z}}
            =\surrealAdd{\surrealAdd{x}{y}}{z}$ and
        $\surrealMul{x}{\surrealMul{y}{z}}
            =\surrealMul{\surrealMul{x}{y}}{z}$ and
        $\surrealMul{x}{\surrealAdd{y}{z}}
            =\surrealAdd{\surrealMul{x}{y}}{\surrealMul{x}{z}}$.
    \begin{subproof}
        Fix $x,y,z\in\reals$.
        $\surreal{x}$ and $\surreal{y}$ and $\surreal{z}$
            by \obv{surreal_real_is_surreal}.
        Follows by
            \obv{surreal_add_associative,surreal_mul_associative,surreal_mul_distributive_left}.
    \end{subproof}
    Follows by \obv{surreal_field_carrier}.
\end{proof}

\begin{definition}\label{surreal_ordered_field_carrier}
    $\surrealOrderedFieldCarrier{R}$ iff
        $\surrealFieldCarrier{R}$ and
        (for every $x\in R$ we have $\lnot\surrealLt{x}{x}$) and
        (for every $x,y\in R$ we have
            $\surrealLt{x}{y}$ or $x=y$ or $\surrealLt{y}{x}$) and
        (for every $x,y,z\in R$ we have
            $\surrealLt{x}{y}\land\surrealLt{y}{z}
                \implies\surrealLt{x}{z}$ and
            $\surrealLt{x}{y}\implies
                \surrealLt{\surrealAdd{x}{z}}{\surrealAdd{y}{z}}$ and
            $\surrealLt{\surrealZero}{z}\land\surrealLt{x}{y}
                \implies
                \surrealLt{\surrealMul{x}{z}}{\surrealMul{y}{z}}$).
\end{definition}

\begin{theorem}\label{surreal_reals_ordered_field_carrier}
    $\surrealOrderedFieldCarrier{\reals}$.
\end{theorem}
\begin{proof}
    $\surrealFieldCarrier{\reals}$
        by \obv{surreal_reals_field_carrier}.
    Show that for every $x\in\reals$ we have
        $\lnot\surrealLt{x}{x}$.
    \begin{subproof}
        Follows by
            \obv{surreal_lt_irreflexive,surreal_real_is_surreal}.
    \end{subproof}
    Show that for every $x,y\in\reals$ we have
        $\surrealLt{x}{y}$ or $x=y$ or $\surrealLt{y}{x}$.
    \begin{subproof}
        Follows by \obv{surreal_real_lt_trichotomy}.
    \end{subproof}
    Show that for every $x,y,z\in\reals$ we have
        $\surrealLt{x}{y}\land\surrealLt{y}{z}
            \implies\surrealLt{x}{z}$ and
        $\surrealLt{x}{y}\implies
            \surrealLt{\surrealAdd{x}{z}}{\surrealAdd{y}{z}}$ and
        $\surrealLt{\surrealZero}{z}\land\surrealLt{x}{y}
            \implies
            \surrealLt{\surrealMul{x}{z}}{\surrealMul{y}{z}}$.
    \begin{subproof}
        Fix $x,y,z\in\reals$.
        $\surreal{x}$ and $\surreal{y}$ and $\surreal{z}$
            by \obv{surreal_real_is_surreal}.
        If $\surrealLt{x}{y}$ and $\surrealLt{y}{z}$ then
            $\surrealLt{x}{z}$
            by \obv{surreal_lt_transitive}.
        If $\surrealLt{x}{y}$ then
            $\surrealLt{\surrealAdd{x}{z}}{\surrealAdd{y}{z}}$
            by \obv{surreal_add_lt_left}.
        If $\surrealLt{\surrealZero}{z}$ and
            $\surrealLt{x}{y}$ then
            $\surrealLt{\surrealMul{x}{z}}{\surrealMul{y}{z}}$
            by \obv{surreal_pos_mul_lt_right}.
    \end{subproof}
    Follows by \obv{surreal_ordered_field_carrier}.
\end{proof}

\begin{definition}\label{surreal_archimedean_ordered_field_carrier}
    $\surrealArchimedeanOrderedFieldCarrier{R}$ iff
        $\surrealOrderedFieldCarrier{R}$ and
        (for every $x,y\in R$ such that
            $\surrealLt{\surrealZero}{x}$ and
            $\surrealLe{\surrealZero}{y}$ there exists a natural number
            $N$ such that
            $\surrealLe{y}{\surrealMul{\ordinalSurreal{N}}{x}}$).
\end{definition}

\begin{theorem}\label{surreal_reals_archimedean_ordered_field_carrier}
    $\surrealArchimedeanOrderedFieldCarrier{\reals}$.
\end{theorem}
\begin{proof}
    $\surrealOrderedFieldCarrier{\reals}$
        by \obv{surreal_reals_ordered_field_carrier}.
    Show that for every $x,y\in\reals$ such that
        $\surrealLt{\surrealZero}{x}$ and
        $\surrealLe{\surrealZero}{y}$ there exists a natural number
        $N$ such that
        $\surrealLe{y}{\surrealMul{\ordinalSurreal{N}}{x}}$.
    \begin{subproof}
        Follows by \obv{surreal_reals_archimedean}.
    \end{subproof}
    Follows by \obv{surreal_archimedean_ordered_field_carrier}.
\end{proof}

\begin{definition}\label{surreal_complete_archimedean_ordered_field_carrier}
    $\surrealCompleteArchimedeanOrderedFieldCarrier{R}$ iff
        $\surrealArchimedeanOrderedFieldCarrier{R}$ and
        (for every $A$ such that $A\subseteq R$ and $A$ is inhabited and
            there exists $b\in R$ such that for every $r\in A$ we have
            $\surrealLe{r}{b}$ there exists $s\in R$ such that
            (for every $a\in A$ we have $\surrealLe{a}{s}$) and
            (for every $u\in R$ such that for every $r\in A$ we have
                $\surrealLe{r}{u}$ we have $\surrealLe{s}{u}$)).
\end{definition}

\begin{theorem}\label{surreal_reals_complete_archimedean_ordered_field}
    $\surrealCompleteArchimedeanOrderedFieldCarrier{\reals}$.
\end{theorem}
\begin{proof}
    $\surrealArchimedeanOrderedFieldCarrier{\reals}$
        by \obv{surreal_reals_archimedean_ordered_field_carrier}.
    Show that for every $A$ such that $A\subseteq\reals$ and
        $A$ is inhabited and there exists $b\in\reals$ such that
        for every $r\in A$ we have
        $\surrealLe{r}{b}$ there exists $s\in\reals$ such that
        (for every $a\in A$ we have $\surrealLe{a}{s}$) and
        (for every $u\in\reals$ such that for every $r\in A$ we have
            $\surrealLe{r}{u}$ we have $\surrealLe{s}{u}$).
    \begin{subproof}
        Fix $A$.
        Assume $A\subseteq\reals$ and $A$ is inhabited and there exists
            $b\in\reals$ such that for every $r\in A$ we have
            $\surrealLe{r}{b}$.
        Take $a_0$ such that $a_0\in A$.
        Hence $A$ is inhabited.
        Take $b\in\reals$ such that for every $r\in A$ we have
            $\surrealLe{r}{b}$.
        Follows by \obv{surreal_reals_least_upper_bound}.
    \end{subproof}
    Follows by
        \obv{surreal_complete_archimedean_ordered_field_carrier}.
\end{proof}
