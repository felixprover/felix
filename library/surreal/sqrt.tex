\import{surreal/reciprocal.tex}

\section{Nonnegative square roots}

The construction first produces finite Newton-style lower and upper
approximations.  Their unions form a Conway cut, which defines the
nonnegative square-root operation by surreal recursion.

\subsection{Finite square-root approximations}

\begin{definition}\label{surreal_sqrt_nonnegative_left}
    $\surrealSqrtNonnegativeLeft{x}=
        \{u\in\surrealLeft{x}\mid
            \surrealLe{\surrealZero}{u}\}$.
\end{definition}

\begin{definition}\label{surreal_sqrt_update}
    $\surrealSqrtUpdate{x}{y}{z}=
        \surrealDiv{
            \surrealAdd{x}{\surrealMul{y}{z}}}{
            \surrealAdd{y}{z}}$.
\end{definition}

\begin{definition}\label{surreal_sqrt_update_set}
    $\surrealSqrtUpdateSetOp:\Set\to\Set\to\Set\to\Set
        =\fun{Y,Z,x:\Set}{
                    \{\surrealSqrtUpdate{x}{\fst{p}}{\snd{p}}
                        \mid p\in Y\times Z
                        \mid \surrealLt{\surrealZero}{
                            \surrealAdd{\fst{p}}{\snd{p}}}\}}$.
\end{definition}

\begin{proposition}\label{surreal_sqrt_update_set_iff}
    $\forall Y,Z,x,q:\Set.
        q\in\Apply{\surrealSqrtUpdateSetOp}{Y,Z,x}
        \iff
        (\exists y,z:\Set.
            y\in Y\land z\in Z\land
            \surrealLt{\surrealZero}{\surrealAdd{y}{z}}\land
            q=\surrealSqrtUpdate{x}{y}{z})$.
\end{proposition}
\begin{proof}
    Fix $Y,Z,x,q$.
    Show that if
        $q\in\Apply{\surrealSqrtUpdateSetOp}{Y,Z,x}$, then
        there exist $y,z$ such that
            $y\in Y$ and $z\in Z$ and
            $\surrealLt{\surrealZero}{\surrealAdd{y}{z}}$ and
            $q=\surrealSqrtUpdate{x}{y}{z}$.
    \begin{subproof}
        Assume
            $q\in\Apply{\surrealSqrtUpdateSetOp}{Y,Z,x}$.
        Take $p\in Y\times Z$ such that
            $\surrealLt{\surrealZero}{
                \surrealAdd{\fst{p}}{\snd{p}}}$ and
            $q=\surrealSqrtUpdate{x}{\fst{p}}{\snd{p}}$ by
            \auto{surreal_sqrt_update_set}.
        Take $y,z$ such that $y\in Y$ and $z\in Z$ and
            $p=(y,z)$ by \obv{times_elem_is_tuple}.
        $\fst{p}=y$ and $\snd{p}=z$ by \auto{fst_eq,snd_eq}.
        Follows by \auto.
    \end{subproof}
    Show that if there exist $y,z$ such that
        $y\in Y$ and $z\in Z$ and
        $\surrealLt{\surrealZero}{\surrealAdd{y}{z}}$ and
        $q=\surrealSqrtUpdate{x}{y}{z}$, then
        $q\in\Apply{\surrealSqrtUpdateSetOp}{Y,Z,x}$.
    \begin{subproof}
        Assume that there exist $y,z$ such that
            $y\in Y$ and $z\in Z$ and
            $\surrealLt{\surrealZero}{\surrealAdd{y}{z}}$ and
            $q=\surrealSqrtUpdate{x}{y}{z}$.
        Take $y,z$ such that
            $y\in Y$ and $z\in Z$ and
            $\surrealLt{\surrealZero}{\surrealAdd{y}{z}}$ and
            $q=\surrealSqrtUpdate{x}{y}{z}$.
        $(y,z)\in Y\times Z$ by \obv{times_tuple_intro}.
        $\fst{(y,z)}=y$ and $\snd{(y,z)}=z$ by
            \obv{fst_eq,snd_eq}.
        Follows by \auto{surreal_sqrt_update_set}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{proposition}\label{surreal_sqrt_update_is_surreal}
    Suppose $\surreal{x}$ and $\surreal{y}$ and $\surreal{z}$.
    Then $\surreal{\surrealSqrtUpdate{x}{y}{z}}$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{surreal_sqrt_update,surreal_add_is_surreal,surreal_mul_is_surreal,surreal_division_is_surreal}.
\end{proof}

\begin{proposition}\label{surreal_sqrt_update_nonnegative}
    Suppose $\surreal{x}$ and $\surreal{y}$ and $\surreal{z}$.
    Suppose $\surrealLe{\surrealZero}{x}$ and
        $\surrealLe{\surrealZero}{y}$ and
        $\surrealLe{\surrealZero}{z}$ and
        $\surrealLt{\surrealZero}{\surrealAdd{y}{z}}$.
    Then $\surrealLe{\surrealZero}{\surrealSqrtUpdate{x}{y}{z}}$.
\end{proposition}
\begin{proof}
    $\surreal{\surrealMul{y}{z}}$ and
        $\surrealLe{\surrealZero}{\surrealMul{y}{z}}$
        by \obv{surreal_mul_is_surreal,surreal_mul_nonneg_nonneg}.
    $\surreal{\surrealAdd{x}{\surrealMul{y}{z}}}$ and
        $\surrealLe{\surrealZero}{
            \surrealAdd{x}{\surrealMul{y}{z}}}$
        by \auto{surreal_add_is_surreal,surreal_add_le_both,surreal_add_zero_left,empty_surreal}.
    $\surreal{\surrealAdd{y}{z}}$
        by \obv{surreal_add_is_surreal}.
    Follows by
        \auto{surreal_sqrt_update,surreal_division_nonnegative}.
\end{proof}

\begin{lemma}\label{surreal_mul_positive_right_negative_elim}
    Suppose $\surreal{a}$ and $\surreal{d}$.
    Suppose $\surrealLt{\surrealZero}{d}$ and
        $\surrealLt{\surrealMul{a}{d}}{\surrealZero}$.
    Then $\surrealLt{a}{\surrealZero}$.
\end{lemma}
\begin{proof}
    $\surrealLt{a}{\surrealZero}$ or $a=\surrealZero$ or
        $\surrealLt{\surrealZero}{a}$
        by \obv{surreal_lt_trichotomy,empty_surreal}.
    \begin{byCase}
        \caseOf{$\surrealLt{a}{\surrealZero}$.}
            Follows.
        \caseOf{not $\surrealLt{a}{\surrealZero}$.}
            \begin{byCase}
                \caseOf{$a=\surrealZero$.}
                    $\surrealMul{a}{d}=\surrealZero$
                        by \auto{surreal_mul_zero_left}.
                    Hence $\surrealLt{\surrealZero}{\surrealZero}$.
                    Contradiction by
                        \obv{surreal_lt_irreflexive,empty_surreal}.
                \caseOf{$a\neq\surrealZero$.}
                    $\surrealLt{\surrealZero}{a}$.
                    $\surrealLt{\surrealZero}{\surrealMul{a}{d}}$
                        by \obv{surreal_mul_pos_pos}.
                    Contradiction by
                        \obv{surreal_lt_asymmetric,empty_surreal,surreal_mul_is_surreal}.
            \end{byCase}
    \end{byCase}
\end{proof}

\begin{lemma}\label{surreal_mul_positive_right_positive_elim}

            \end{byCase}
    \end{byCase}
\end{proof}

\begin{lemma}\label{surreal_sqrt_update_difference_identity}
    Suppose $\surreal{x}$ and $\surreal{y}$ and $\surreal{z}$.
    Suppose $\surrealLt{\surrealZero}{\surrealAdd{y}{z}}$.
    Let $m=\surrealSqrtUpdate{x}{y}{z}$.
    Let $d=\surrealAdd{y}{z}$.
    Then
        $\surrealMul{
            \surrealAdd{\surrealMul{m}{m}}{
                \surrealNeg{x}}}{\surrealMul{d}{d}}
        =\surrealMul{
            \surrealAdd{x}{\surrealNeg{\surrealMul{y}{y}}}}{
            \surrealAdd{x}{\surrealNeg{\surrealMul{z}{z}}}}$.
\end{lemma}
\begin{proof}
    $\surreal{d}$ by \obv{surreal_add_is_surreal}.
    $d\neq\surrealZero$
        by \obv{surreal_lt_irreflexive,empty_surreal}.
    $\surreal{\surrealAdd{x}{\surrealMul{y}{z}}}$
        by \obv{surreal_mul_is_surreal,surreal_add_is_surreal}.
    Hence $\surreal{m}$
        by \obv{surreal_sqrt_update_is_surreal}.
    $\surrealMul{m}{d}=\surrealAdd{x}{\surrealMul{y}{z}}$
        by
        \auto{surreal_sqrt_update,surreal_division_cancel_right}.
    Let $p=\surrealMul{y}{z}$.
    $\surreal{p}$ by \obv{surreal_mul_is_surreal}.
    Let $A=\surrealMul{x}{x}$.
    Let $B=\surrealMul{x}{p}$.
    Let $C=\surrealMul{p}{p}$.
    Let $E=\surrealMul{x}{\surrealMul{y}{y}}$.
    Let $F=\surrealMul{x}{\surrealMul{z}{z}}$.
    $\surreal{A}$ and $\surreal{B}$ and $\surreal{C}$ and
        $\surreal{E}$ and $\surreal{F}$
        by \obv{surreal_mul_is_surreal}.
    Let $S=\surrealAdd{\surrealAdd{A}{C}}{
        \surrealAdd{\surrealNeg{E}}{\surrealNeg{F}}}$.
    Let $T=\surrealAdd{B}{B}$.
    Let $U=\surrealAdd{A}{C}$.
    Let $V=\surrealAdd{E}{F}$.
    $\surreal{T}$ and $\surreal{U}$ and $\surreal{V}$
        by \obv{surreal_add_is_surreal}.
    $\surrealMul{\surrealMul{m}{m}}{\surrealMul{d}{d}}
        =\surrealMul{\surrealMul{m}{d}}{\surrealMul{m}{d}}$
        by \obv{surreal_mul_four_rearrange}.
    Hence
        $\surrealMul{\surrealMul{m}{m}}{\surrealMul{d}{d}}
        =\surrealMul{\surrealAdd{x}{p}}{\surrealAdd{x}{p}}$.
    $\surrealMul{\surrealAdd{x}{p}}{\surrealAdd{x}{p}}
        =\surrealAdd{
            \surrealAdd{\surrealMul{x}{x}}{\surrealMul{x}{p}}}{
            \surrealAdd{\surrealMul{x}{p}}{\surrealMul{p}{p}}}$
        by
        \auto{surreal_mul_foil,surreal_mul_commutative,surreal_mul_is_surreal}.
    $\surrealMul{d}{d}
        =\surrealAdd{
            \surrealAdd{\surrealMul{y}{y}}{p}}{
            \surrealAdd{p}{\surrealMul{z}{z}}}$
        by
        \auto{surreal_mul_foil,surreal_mul_commutative,surreal_mul_is_surreal}.
    Hence $\surrealMul{x}{\surrealMul{d}{d}}
        =\surrealMul{x}{
            \surrealAdd{
                \surrealAdd{\surrealMul{y}{y}}{p}}{
                \surrealAdd{p}{\surrealMul{z}{z}}}}$.
    Hence $\surrealMul{x}{\surrealMul{d}{d}}
        =\surrealAdd{
            \surrealAdd{\surrealMul{x}{\surrealMul{y}{y}}}{
                \surrealMul{x}{p}}}{
            \surrealAdd{\surrealMul{x}{p}}{
                \surrealMul{x}{\surrealMul{z}{z}}}}$
        by
        \auto{surreal_mul_distributive_left,surreal_add_is_surreal,surreal_mul_is_surreal}.
    Hence
        $\surrealMul{
            \surrealAdd{\surrealMul{m}{m}}{\surrealNeg{x}}}{
            \surrealMul{d}{d}}
        =\surrealAdd{
            \surrealMul{\surrealMul{m}{m}}{\surrealMul{d}{d}}}{
            \surrealNeg{\surrealMul{x}{\surrealMul{d}{d}}}}$
        by
        \obv{surreal_mul_subtract_left,surreal_mul_neg_left,surreal_mul_is_surreal}.
    Hence
        $\surrealMul{
            \surrealAdd{\surrealMul{m}{m}}{\surrealNeg{x}}}{
            \surrealMul{d}{d}}
        =\surrealAdd{
            \surrealAdd{\surrealAdd{A}{B}}{\surrealAdd{B}{C}}}{
            \surrealNeg{
                \surrealAdd{\surrealAdd{E}{B}}{
                    \surrealAdd{B}{F}}}}$.
    $\surrealAdd{\surrealAdd{A}{B}}{\surrealAdd{B}{C}}
        =\surrealAdd{U}{T}$
        by
        \auto{surreal_add_associative,surreal_add_commutative,surreal_add_is_surreal}.
    $\surrealAdd{\surrealAdd{E}{B}}{\surrealAdd{B}{F}}
        =\surrealAdd{V}{T}$
        by
        \auto{surreal_add_associative,surreal_add_commutative,surreal_add_is_surreal}.
    $S=\surrealAdd{U}{\surrealNeg{V}}$
        by
        \auto{surreal_negation_additive,surreal_add_associative,surreal_add_commutative,surreal_add_is_surreal,surreal_negation_is_surreal}.
    $\surrealAdd{\surrealAdd{U}{T}}{
        \surrealNeg{\surrealAdd{V}{T}}}
        =\surrealAdd{U}{\surrealNeg{V}}$
        by
        \auto{surreal_negation_additive,surreal_add_associative,surreal_add_commutative,surreal_add_inverse_left,surreal_add_inverse_right,surreal_add_zero_left,surreal_add_zero_right,surreal_add_is_surreal,surreal_negation_is_surreal}.
    Hence
        $\surrealAdd{
            \surrealAdd{\surrealAdd{A}{B}}{\surrealAdd{B}{C}}}{
            \surrealNeg{
                \surrealAdd{\surrealAdd{E}{B}}{
                    \surrealAdd{B}{F}}}}=S$.
    Hence
        $\surrealMul{
            \surrealAdd{\surrealMul{m}{m}}{\surrealNeg{x}}}{
            \surrealMul{d}{d}}=S$.
    $\surrealMul{x}{\surrealNeg{\surrealMul{z}{z}}}
        =\surrealNeg{F}$
        by \auto{surreal_mul_neg_right,surreal_mul_is_surreal}.
    $\surrealMul{\surrealNeg{\surrealMul{y}{y}}}{x}
        =\surrealNeg{E}$
        by
        \auto{surreal_mul_neg_left,surreal_mul_commutative,surreal_mul_is_surreal}.
    $\surrealMul{\surrealNeg{\surrealMul{y}{y}}}{
        \surrealNeg{\surrealMul{z}{z}}}=C$
        by
        \auto{surreal_mul_both_negated,surreal_mul_four_rearrange,surreal_mul_is_surreal}.
    Hence $\surrealMul{
        \surrealAdd{x}{\surrealNeg{\surrealMul{y}{y}}}}{
        \surrealAdd{x}{\surrealNeg{\surrealMul{z}{z}}}}
        =\surrealAdd{\surrealAdd{A}{\surrealNeg{F}}}{
            \surrealAdd{\surrealNeg{E}}{C}}$
        by
        \auto{surreal_mul_foil,surreal_mul_is_surreal,surreal_negation_is_surreal}.
    $\surrealAdd{\surrealAdd{A}{\surrealNeg{F}}}{
        \surrealAdd{\surrealNeg{E}}{C}}=S$
        by
        \auto{surreal_add_associative,surreal_add_commutative,surreal_add_is_surreal,surreal_negation_is_surreal}.
    Hence $\surrealMul{
        \surrealAdd{x}{\surrealNeg{\surrealMul{y}{y}}}}{
        \surrealAdd{x}{\surrealNeg{\surrealMul{z}{z}}}}=S$.
\end{proof}

\begin{theorem}\label{surreal_sqrt_update_lower_upper}
    Suppose $\surreal{x}$ and $\surreal{y}$ and $\surreal{z}$.
    Suppose $\surrealLe{\surrealZero}{x}$ and
        $\surrealLe{\surrealZero}{y}$ and
        $\surrealLe{\surrealZero}{z}$.
    Suppose
        $\surrealLt{\surrealMul{y}{y}}{x}$ and
        $\surrealLt{x}{\surrealMul{z}{z}}$ and
        $\surrealLt{\surrealZero}{\surrealAdd{y}{z}}$.
    Let $m=\surrealSqrtUpdate{x}{y}{z}$.
    Then $\surreal{m}$ and
        $\surrealLe{\surrealZero}{m}$ and
        $\surrealLt{\surrealMul{m}{m}}{x}$.
\end{theorem}
\begin{proof}
    Let $d=\surrealAdd{y}{z}$.
    $\surreal{d}$ by \obv{surreal_add_is_surreal}.
    $\surreal{m}$ and $\surrealLe{\surrealZero}{m}$
        by
        \obv{surreal_sqrt_update_is_surreal,surreal_sqrt_update_nonnegative}.
    Let $D=\surrealMul{d}{d}$.
    $\surreal{D}$ and $\surrealLt{\surrealZero}{D}$
        by \obv{surreal_mul_is_surreal,surreal_mul_pos_pos}.
    Let $a=\surrealAdd{\surrealMul{m}{m}}{\surrealNeg{x}}$.
    Let $b=\surrealAdd{x}{\surrealNeg{\surrealMul{y}{y}}}$.
    Let $c=\surrealAdd{x}{\surrealNeg{\surrealMul{z}{z}}}$.
    $\surreal{a}$ and $\surreal{b}$ and $\surreal{c}$
        by
        \obv{surreal_mul_is_surreal,surreal_negation_is_surreal,surreal_add_is_surreal}.
    $\surrealLt{\surrealZero}{b}$ and
        $\surrealLt{c}{\surrealZero}$
        by
        \auto{surreal_subtraction_positive,surreal_subtraction_negative,surreal_mul_is_surreal}.
    Hence $\surrealLt{\surrealMul{b}{c}}{\surrealZero}$
        by \obv{surreal_mul_pos_neg}.
    $\surrealMul{a}{D}=\surrealMul{b}{c}$
        by \auto{surreal_sqrt_update_difference_identity}.
    Hence $\surrealLt{\surrealMul{a}{D}}{\surrealZero}$.
    Hence $\surrealLt{a}{\surrealZero}$
        by \obv{surreal_mul_positive_right_negative_elim}.
    Hence $\surrealLt{\surrealMul{m}{m}}{x}$
        by
        \auto{surreal_subtraction_negative_elim,surreal_mul_is_surreal}.
\end{proof}

\begin{definition}\label{surreal_sqrt_approximation_step}
    $\surrealSqrtApproxStepOp:\Set\to\Set\to\Set\to\Set
        =\fun{x,n,p:\Set}{
                    (\fst{p}
                        \union
                        \Apply{\surrealSqrtUpdateSetOp}{\fst{p},\snd{p},x},
                     \snd{p}
                        \union
                        \Apply{\surrealSqrtUpdateSetOp}{\fst{p},\fst{p},x}
                        \union
                        \Apply{\surrealSqrtUpdateSetOp}{\snd{p},\snd{p},x})}$.
\end{definition}

\begin{definition}\label{surreal_sqrt_approximation}
    $\surrealSqrtApproxOp:
        \Set\to(\Set\to\Set)\to\Set\to\Set
        =\fun{x:\Set}{
            \fun{g:\Set\to\Set}{
                \fun{n:\Set}{
                    \Apply{\naturalRecOp}{
                                (\{\Apply{g}{w}
                                    \mid w\in\surrealSqrtNonnegativeLeft{x}\},
                                 \{\Apply{g}{z}
                                    \mid z\in\surrealRight{x}\}),
                            \Apply{\surrealSqrtApproxStepOp}{x},n}}}}$.
\end{definition}

\begin{definition}\label{surreal_sqrt_lower_approximation}
    $\surrealSqrtLowerApproxOp:
        \Set\to(\Set\to\Set)\to\Set\to\Set
        =\fun{x:\Set}{
            \fun{g:\Set\to\Set}{
                \fun{n:\Set}{
                    \fst{\Apply{\surrealSqrtApproxOp}{x,g,n}}}}}$.
\end{definition}

\begin{definition}\label{surreal_sqrt_upper_approximation}
    $\surrealSqrtUpperApproxOp:
        \Set\to(\Set\to\Set)\to\Set\to\Set
        =\fun{x:\Set}{
            \fun{g:\Set\to\Set}{
                \fun{n:\Set}{
                    \snd{\Apply{\surrealSqrtApproxOp}{x,g,n}}}}}$.

\begin{proposition}\label{surreal_sqrt_approximation_zero}

\begin{theorem}\label{surreal_sqrt_approximation_monotone_add}
    $\forall x,m,n:\Set.\ \forall g:\Set\to\Set.
        (m\in\naturals\land n\in\naturals)
        \implies
        (\Apply{\surrealSqrtLowerApproxOp}{x,g,m}
            \subseteq
         \Apply{\surrealSqrtLowerApproxOp}{x,g,m+n}
         \land
         \Apply{\surrealSqrtUpperApproxOp}{x,g,m}
            \subseteq
         \Apply{\surrealSqrtUpperApproxOp}{x,g,m+n})$.
\end{theorem}
\begin{proof}
    Fix $x,m,n,g$.
    Assume $m\in\naturals$ and $n\in\naturals$.
    Let $P:\Set\to\Prop=\fun{k:\Set}{
        (\Apply{\surrealSqrtLowerApproxOp}{x,g,m}
            \subseteq
         \Apply{\surrealSqrtLowerApproxOp}{x,g,m+k})
        \land
        (\Apply{\surrealSqrtUpperApproxOp}{x,g,m}
            \subseteq
         \Apply{\surrealSqrtUpperApproxOp}{x,g,m+k})}$.
    $\Apply{P}{\emptyset}$
        by \obv{num_addition_zero_right,subseteq_refl}.
    Show that $\forall k:\Set.\ k\in\naturals\implies
        (\Apply{P}{k}\implies\Apply{P}{\suc{k}})$.
    \begin{subproof}
        Fix $k$.
        Assume $k\in\naturals$.
        Assume $\Apply{P}{k}$.
        $m+k\in\naturals$
            by \obv{num_addition_on_naturals}.
        $m+\suc{k}=\suc{m+k}$
            by \obv{num_addition_successor_right}.
        $\Apply{\surrealSqrtLowerApproxOp}{x,g,m+k}
            \subseteq
         \Apply{\surrealSqrtLowerApproxOp}{x,g,\suc{m+k}}$
            by
            \auto{surreal_sqrt_lower_approximation_successor,subseteq,union_iff}.
        $\Apply{\surrealSqrtUpperApproxOp}{x,g,m+k}
            \subseteq
         \Apply{\surrealSqrtUpperApproxOp}{x,g,\suc{m+k}}$
            by
            \auto{surreal_sqrt_upper_approximation_successor,subseteq,union_iff}.
        Follows by \obv{subseteq_transitive}.
    \end{subproof}
    Let $C=\sep{\naturals}{P}$.
    $\emptyset\in C$ by
        \auto{logical_separation_iff,omega_empty}.
    For every $k\in C$ we have $\suc{k}\in C$
        by \auto{logical_separation_iff,omega_successor}.
    Hence $C$ is an inductive set by \obv{inductive_set}.
    Hence every natural number is an element of $C$
        by \obv{omega_minimal}.
    Hence $n\in C$.
    Hence $\Apply{P}{n}$ by \auto{logical_separation_iff}.
\end{proof}

\begin{theorem}\label{surreal_sqrt_options_invariant}
    $\forall x:\Set.\ \forall g:\Set\to\Set.

             \surrealMul{\Apply{g}{u}}{\Apply{g}{u}}=u)))

            \implies

             \surrealLe{\surrealZero}{y}\land

            \implies
            (\surreal{y}\land
             \surrealLe{\surrealZero}{y}\land

\begin{proof}

\begin{theorem}\label{surreal_sqrt_options_cut_pair}
    $\forall x:\Set.\ \forall g:\Set\to\Set.

         \surrealLe{\surrealZero}{x}\land
         (\forall u:\Set.
            (\surreal{u}\land\birthday{u}\in\birthday{x}\land
                \surrealLe{\surrealZero}{u})
            \implies
            (\surreal{\Apply{g}{u}}\land
             \surrealLe{\surrealZero}{\Apply{g}{u}}\land
             \surrealMul{\Apply{g}{u}}{\Apply{g}{u}}=u)))
        \implies
        \surrealCutPair{
            \Apply{\surrealSqrtLowerOptionsOp}{x,g}}{
            \Apply{\surrealSqrtUpperOptionsOp}{x,g}}$.
\end{theorem}
\begin{proof}
    Fix $x,g$.
    Assume $\surreal{x}$ and
        $\surrealLe{\surrealZero}{x}$.
    Assume $\forall u:\Set.
        (\surreal{u}\land\birthday{u}\in\birthday{x}\land
            \surrealLe{\surrealZero}{u})
        \implies
        (\surreal{\Apply{g}{u}}\land
         \surrealLe{\surrealZero}{\Apply{g}{u}}\land
         \surrealMul{\Apply{g}{u}}{\Apply{g}{u}}=u)$.
    Let $L=\Apply{\surrealSqrtLowerOptionsOp}{x,g}$.
    Let $R=\Apply{\surrealSqrtUpperOptionsOp}{x,g}$.
    $\forall y:\Set.\ y\in L\implies
        (\surreal{y}\land
         \surrealLe{\surrealZero}{y}\land
         \surrealLt{\surrealMul{y}{y}}{x})$
        by \auto{surreal_sqrt_options_invariant}.
    $\forall y:\Set.\ y\in R\implies
        (\surreal{y}\land
         \surrealLe{\surrealZero}{y}\land
         \surrealLt{x}{\surrealMul{y}{y}})$
        by \auto{surreal_sqrt_options_invariant}.
    Show that $\forall l:\Set.\ l\in L\implies
        (\forall r:\Set.\ r\in R\implies\surrealLt{l}{r})$.
    \begin{subproof}
        Fix $l$.
        Assume $l\in L$.
        Fix $r$.
        Assume $r\in R$.
        $\surreal{l}$ and $\surrealLe{\surrealZero}{l}$ and
            $\surrealLt{\surrealMul{l}{l}}{x}$.
        $\surreal{r}$ and $\surrealLe{\surrealZero}{r}$ and
            $\surrealLt{x}{\surrealMul{r}{r}}$.
        $\surrealLt{l}{r}$ or $l=r$ or $\surrealLt{r}{l}$
            by \obv{surreal_lt_trichotomy}.
        \begin{byCase}
            \caseOf{$\surrealLt{l}{r}$.}
                Follows.
            \caseOf{not $\surrealLt{l}{r}$.}
                \begin{byCase}
                    \caseOf{$l=r$.}
                        $\surrealLt{\surrealMul{l}{l}}{
                            \surrealMul{l}{l}}$
                            by
                            \auto{surreal_lt_transitive,surreal_mul_is_surreal}.
                        Contradiction by
                            \obv{surreal_lt_irreflexive,surreal_mul_is_surreal}.
                    \caseOf{$l\neq r$.}
                        Hence $\surrealLt{r}{l}$.
                        $\surrealLt{\surrealMul{r}{r}}{
                            \surrealMul{l}{l}}$
                            by
                            \obv{surreal_square_strict_monotone_nonnegative}.
                        Hence $\surrealLt{x}{
                            \surrealMul{l}{l}}$
                            by
                            \auto{surreal_lt_transitive,surreal_mul_is_surreal}.
                        Hence $\surrealLt{x}{x}$
                            by
                            \auto{surreal_lt_transitive,surreal_mul_is_surreal}.
                        Contradiction by
                            \obv{surreal_lt_irreflexive}.
                \end{byCase}
        \end{byCase}
    \end{subproof}
    Follows by \auto{surreal_cut_pair}.
\end{proof}

\begin{theorem}\label{surreal_sqrt_cut_not_above}
    $\forall x:\Set.\ \forall g:\Set\to\Set.
        (\surreal{x}\land
         \surrealLe{\surrealZero}{x}\land
         (\forall u:\Set.
            (\surreal{u}\land\birthday{u}\in\birthday{x}\land
                \surrealLe{\surrealZero}{u})
            \implies
            (\surreal{\Apply{g}{u}}\land
             \surrealLe{\surrealZero}{\Apply{g}{u}}\land
             \surrealMul{\Apply{g}{u}}{\Apply{g}{u}}=u)))
        \implies
        \lnot\surrealLt{x}{
            \surrealMul{
                \surrealCut{
                    \Apply{\surrealSqrtLowerOptionsOp}{x,g}}{
                    \Apply{\surrealSqrtUpperOptionsOp}{x,g}}}{
                \surrealCut{
                    \Apply{\surrealSqrtLowerOptionsOp}{x,g}}{
                    \Apply{\surrealSqrtUpperOptionsOp}{x,g}}}}$.
\end{theorem}
\begin{proof}
    Fix $x,g$.
    Assume $\surreal{x}$ and
        $\surrealLe{\surrealZero}{x}$.
    Assume $\forall u:\Set.
        (\surreal{u}\land\birthday{u}\in\birthday{x}\land
            \surrealLe{\surrealZero}{u})
        \implies
        (\surreal{\Apply{g}{u}}\land
         \surrealLe{\surrealZero}{\Apply{g}{u}}\land
         \surrealMul{\Apply{g}{u}}{\Apply{g}{u}}=u)$.
    Let $L=\Apply{\surrealSqrtLowerOptionsOp}{x,g}$.
    Let $R=\Apply{\surrealSqrtUpperOptionsOp}{x,g}$.
    Let $y=\surrealCut{L}{R}$.
    $\surrealCutPair{L}{R}$
        by \auto{surreal_sqrt_options_cut_pair}.
    $\surreal{y}$ by \obv{surreal_cut_pair_is_surreal}.
    $\surrealLe{\surrealZero}{y}$
        by \auto{surreal_sqrt_cut_nonnegative}.
    $\forall l:\Set.\ l\in L\implies
        (\surreal{l}\land
         \surrealLe{\surrealZero}{l}\land
         \surrealLt{\surrealMul{l}{l}}{x})$
        by \auto{surreal_sqrt_options_invariant}.
    $\forall r:\Set.\ r\in R\implies
        (\surreal{r}\land
         \surrealLe{\surrealZero}{r}\land
         \surrealLt{x}{\surrealMul{r}{r}})$
        by \auto{surreal_sqrt_options_invariant}.
    Show that not $\surrealLt{x}{\surrealMul{y}{y}}$.
    \begin{subproof}
    Suppose $\surrealLt{x}{\surrealMul{y}{y}}$.
    $\surreal{\surrealMul{y}{y}}$
        by \obv{surreal_mul_is_surreal}.
    $(\exists q:\Set.\ \surreal{q}\land
        \birthday{q}\in\birthday{x}
            \inter\birthday{\surrealMul{y}{y}}\land
        \surrealLt{x}{q}\land
        \surrealLt{q}{\surrealMul{y}{y}})
      \lor x\in\surrealLeft{\surrealMul{y}{y}}
      \lor \surrealMul{y}{y}\in\surrealRight{x}$
        by \obv{surreal_lt_elimination}.
    \begin{byCase}
        \caseOf{$\exists q:\Set.\ \surreal{q}\land
            \birthday{q}\in\birthday{x}
                \inter\birthday{\surrealMul{y}{y}}\land
            \surrealLt{x}{q}\land
            \surrealLt{q}{\surrealMul{y}{y}}$.}
            Take $q$ such that $\surreal{q}$ and
                $\birthday{q}\in\birthday{x}
                    \inter\birthday{\surrealMul{y}{y}}$ and
                $\surrealLt{x}{q}$ and
                $\surrealLt{q}{\surrealMul{y}{y}}$.
            $\birthday{q}\in\birthday{x}$ by \obv{inter_elim_left}.
            $\surrealLt{\surrealZero}{q}$
                by
                \obv{surreal_le_lt_transitive,empty_surreal}.
            Hence $\surrealLe{\surrealZero}{q}$
                by \obv{surreal_le}.
            Let $s=\Apply{g}{q}$.
            $\surreal{s}$ and $\surrealLe{\surrealZero}{s}$ and
                $\surrealMul{s}{s}=q$ by \auto.
            $q\in\surrealRight{x}$
                by \obv{surreal_right_intro}.
            Hence $s\in R$
                by \auto{surreal_sqrt_recursive_right_option}.
            Hence $\surrealLt{y}{s}$
                by \auto{surreal_cut_pair_right}.
            Hence $\surrealLt{\surrealMul{y}{y}}{
                \surrealMul{s}{s}}$
                by
                \obv{surreal_square_strict_monotone_nonnegative}.
            Hence $\surrealLt{q}{q}$
                by
                \auto{surreal_lt_transitive,surreal_mul_is_surreal}.
            Contradiction by \obv{surreal_lt_irreflexive}.
        \caseOf{not $\exists q:\Set.\ \surreal{q}\land
            \birthday{q}\in\birthday{x}
                \inter\birthday{\surrealMul{y}{y}}\land
            \surrealLt{x}{q}\land
            \surrealLt{q}{\surrealMul{y}{y}}$.}
            \begin{byCase}

                        (($u\in\surrealLeft{y}$ and
                            $v\in\surrealLeft{y}$) or
                         ($u\in\surrealRight{y}$ and
                            $v\in\surrealRight{y}$)) and
                        $\surrealLe{
                            \surrealAdd{x}{\surrealMul{u}{v}}}{
                            \surrealAdd{\surrealMul{u}{y}}{
                                \surrealMul{y}{v}}}$
                        by \auto{surreal_mul_left_interpolation}.
                    $\surreal{u}$ and $\surreal{v}$
                        by
                        \auto{surreal_left_elim,surreal_right_elim}.
                    \begin{byCase}
                        \caseOf{$u\in\surrealLeft{y}$ and
                            $v\in\surrealLeft{y}$.}
                            Take $l\in L$ such that
                                $\surrealLe{u}{l}$
                                by \auto{surreal_cut_left_interpolation}.
                            Take $r\in L$ such that
                                $\surrealLe{v}{r}$
                                by \auto{surreal_cut_left_interpolation}.
                            $\surreal{l}$ and
                                $\surrealLe{\surrealZero}{l}$ and

                                $\surrealLe{\surrealZero}{r}$ and
                                $\surrealLt{\surrealMul{r}{r}}{x}$
                                by \auto.
                            $\surrealLt{l}{y}$ and
                                $\surrealLt{r}{y}$
                                by \obv{surreal_cut_pair_left}.
                            Hence $\surrealLe{l}{y}$ and
                                $\surrealLe{r}{y}$
                                by \obv{surreal_le}.
                            $\surrealLe{

                                \surrealAdd{\surrealMul{l}{y}}{
                                    \surrealMul{y}{r}}}$

                            $\surrealLt{\surrealZero}{

                                \auto{surreal_sqrt_same_update_contradiction}.
                        \caseOf{$\lnot(u\in\surrealLeft{y}\land
                            v\in\surrealLeft{y})$.}
                            Hence $u\in\surrealRight{y}$ and
                                $v\in\surrealRight{y}$.
                            Take $l\in R$ such that
                                $\surrealLe{l}{u}$
                                by \auto{surreal_cut_right_interpolation}.
                            Take $r\in R$ such that
                                $\surrealLe{r}{v}$
                                by \auto{surreal_cut_right_interpolation}.
                            $\surreal{l}$ and
                                $\surrealLe{\surrealZero}{l}$ and
                                $\surrealLt{x}{\surrealMul{l}{l}}$
                                by \auto.
                            $\surreal{r}$ and
                                $\surrealLe{\surrealZero}{r}$ and
                                $\surrealLt{x}{\surrealMul{r}{r}}$
                                by \auto.
                            $\surrealLt{y}{l}$ and

                                by \auto{surreal_cut_pair_right}.
                            Hence $\surrealLe{y}{l}$ and

                \caseOf{$x\notin\surrealLeft{\surrealMul{y}{y}}$.}

                    $\surreal{s}$ and

                    Hence $\surrealLt{y}{y}$ by \auto.

\begin{theorem}\label{surreal_sqrt_approximation_extensionality}
    $\forall x,n:\Set.\ \forall g,h:\Set\to\Set.
        (\surreal{x}\land n\in\naturals\land
         (\forall u:\Set.
            (\surreal{u}\land\birthday{u}\in\birthday{x})
            \implies\Apply{g}{u}=\Apply{h}{u}))
        \implies
        \Apply{\surrealSqrtApproxOp}{x,g,n}
        =\Apply{\surrealSqrtApproxOp}{x,h,n}$.
\end{theorem}
\begin{proof}
    Fix $x,n,g,h$.
    Assume $\surreal{x}$ and $n\in\naturals$.
    Assume $\forall u:\Set.
        (\surreal{u}\land\birthday{u}\in\birthday{x})
        \implies\Apply{g}{u}=\Apply{h}{u}$.
    Let $P:\Set\to\Prop=\fun{k:\Set}{
        \Apply{\surrealSqrtApproxOp}{x,g,k}
        =\Apply{\surrealSqrtApproxOp}{x,h,k}}$.

    Show that $\Apply{P}{\emptyset}$.
    \begin{subproof}
        Show that for every $u\in\surrealSqrtNonnegativeLeft{x}$
            we have $\Apply{g}{u}=\Apply{h}{u}$.
        \begin{subproof}
            Fix $u\in\surrealSqrtNonnegativeLeft{x}$.
            Hence $u\in\surrealLeft{x}$
                by \obv{surreal_sqrt_nonnegative_left}.
            Hence $\surreal{u}$ and
                $\birthday{u}\in\birthday{x}$
                by \auto{surreal_left_elim}.
        \end{subproof}
        Hence
            $\{\Apply{g}{u}
                \mid u\in\surrealSqrtNonnegativeLeft{x}\}
            =\{\Apply{h}{u}
                \mid u\in\surrealSqrtNonnegativeLeft{x}\}$
            by \obv{logical_replacement_extensionality}.
        Show that for every $u\in\surrealRight{x}$
            we have $\Apply{g}{u}=\Apply{h}{u}$.
        \begin{subproof}
            Fix $u\in\surrealRight{x}$.
            Hence $\surreal{u}$ and
                $\birthday{u}\in\birthday{x}$
                by \auto{surreal_right_elim}.
        \end{subproof}
        Hence
            $\{\Apply{g}{u}\mid u\in\surrealRight{x}\}
            =\{\Apply{h}{u}\mid u\in\surrealRight{x}\}$
            by \obv{logical_replacement_extensionality}.
        Follows by \auto{surreal_sqrt_approximation_zero}.
    \end{subproof}

    Show that $\forall k:\Set.\ k\in\naturals\implies
        (\Apply{P}{k}\implies\Apply{P}{\suc{k}})$.
    \begin{subproof}
        Fix $k$.
        Assume $k\in\naturals$.
        Assume $\Apply{P}{k}$.
        Follows by
            \auto{surreal_sqrt_approximation_successor}.
    \end{subproof}

    Let $C=\sep{\naturals}{P}$.
    $\emptyset\in C$ by
        \auto{logical_separation_iff,omega_empty}.
    For every $k\in C$ we have $\suc{k}\in C$
        by \auto{logical_separation_iff,omega_successor}.
    Hence $C$ is an inductive set by \obv{inductive_set}.
    Hence every natural number is an element of $C$
        by \obv{omega_minimal}.
    Hence $n\in C$.
    Hence $\Apply{P}{n}$ by \auto{logical_separation_iff}.
\end{proof}

\subsection{The nonnegative square-root operation}

\begin{definition}\label{surreal_nonnegative_sqrt_step}
    $\surrealNonnegativeSqrtStepOp:

            \fun{g:\Set\to\Set}{
                \surrealCut
                    {\Apply{\surrealSqrtLowerOptionsOp}{x,g}}
                    {\Apply{\surrealSqrtUpperOptionsOp}{x,g}}}}$.
\end{definition}

\begin{proposition}\label{surreal_nonnegative_sqrt_step_functional}

        \surrealNonnegativeSqrtStepOp}$.
\end{proposition}
\begin{proof}
    Show that $\forall x:\Set.\ \surreal{x}\implies
        (\forall g,h:\Set\to\Set.
            (\forall u:\Set.
                (\surreal{u}\land\birthday{u}\in\birthday{x})
                \implies\Apply{g}{u}=\Apply{h}{u})
            \implies
            \Apply{\surrealNonnegativeSqrtStepOp}{x,g}
            =\Apply{\surrealNonnegativeSqrtStepOp}{x,h})$.
    \begin{subproof}
        Fix $x$.
        Assume $\surreal{x}$.
        Fix $g,h$.
        Assume $\forall u:\Set.
            (\surreal{u}\land\birthday{u}\in\birthday{x})
            \implies\Apply{g}{u}=\Apply{h}{u}$.
        Show that for every $n\in\naturals$ we have
            $\Apply{\surrealSqrtLowerApproxOp}{x,g,n}
            =\Apply{\surrealSqrtLowerApproxOp}{x,h,n}$.
        \begin{subproof}
            Fix $n\in\naturals$.
            $\Apply{\surrealSqrtApproxOp}{x,g,n}
            =\Apply{\surrealSqrtApproxOp}{x,h,n}$
                by \obv{surreal_sqrt_approximation_extensionality}.
            Follows by \auto{surreal_sqrt_lower_approximation}.
        \end{subproof}
        Hence
            $\{\Apply{\surrealSqrtLowerApproxOp}{x,g,n}
                \mid n\in\naturals\}
            =\{\Apply{\surrealSqrtLowerApproxOp}{x,h,n}
                \mid n\in\naturals\}$
            by \obv{logical_replacement_extensionality}.
        Hence
            $\Apply{\surrealSqrtLowerOptionsOp}{x,g}
            =\Apply{\surrealSqrtLowerOptionsOp}{x,h}$
            by \auto{surreal_sqrt_lower_options}.

        Show that for every $n\in\naturals$ we have
            $\Apply{\surrealSqrtUpperApproxOp}{x,g,n}
            =\Apply{\surrealSqrtUpperApproxOp}{x,h,n}$.
        \begin{subproof}
            Fix $n\in\naturals$.
            $\Apply{\surrealSqrtApproxOp}{x,g,n}
            =\Apply{\surrealSqrtApproxOp}{x,h,n}$
                by \obv{surreal_sqrt_approximation_extensionality}.
            Follows by \auto{surreal_sqrt_upper_approximation}.
        \end{subproof}
        Hence
            $\{\Apply{\surrealSqrtUpperApproxOp}{x,g,n}
                \mid n\in\naturals\}
            =\{\Apply{\surrealSqrtUpperApproxOp}{x,h,n}
                \mid n\in\naturals\}$
            by \obv{logical_replacement_extensionality}.
        Hence
            $\Apply{\surrealSqrtUpperOptionsOp}{x,g}
            =\Apply{\surrealSqrtUpperOptionsOp}{x,h}$
            by \auto{surreal_sqrt_upper_options}.
        Follows by \auto{surreal_nonnegative_sqrt_step}.
    \end{subproof}
    Follows by \obv{surreal_recursion_functional}.
\end{proof}

\begin{definition}\label{surreal_nonnegative_sqrt_typed}

\begin{proposition}\label{surreal_nonnegative_sqrt_equation}

\begin{theorem}\label{surreal_nonnegative_sqrt_properties}
    Suppose $\surreal{x}$ and
        $\surrealLe{\surrealZero}{x}$.
    Then $\surreal{\surrealNonnegativeSqrt{x}}$ and
        $\surrealLe{\surrealZero}{\surrealNonnegativeSqrt{x}}$ and
        $\surrealMul{\surrealNonnegativeSqrt{x}}{
            \surrealNonnegativeSqrt{x}}=x$.
\end{theorem}
\begin{proof}
    Let $P:\Set\to\Prop=\fun{a:\Set}{
        \surrealLe{\surrealZero}{a}\implies
        (\surreal{\surrealNonnegativeSqrt{a}}\land
         \surrealLe{\surrealZero}{\surrealNonnegativeSqrt{a}}\land
         \surrealMul{\surrealNonnegativeSqrt{a}}{
            \surrealNonnegativeSqrt{a}}=a)}$.
    Show that $\forall a:\Set.\ \surreal{a}\implies
        ((\forall u:\Set.
            (\surreal{u}\land\birthday{u}\in\birthday{a})
            \implies\Apply{P}{u})
         \implies\Apply{P}{a})$.
    \begin{subproof}
        Fix $a$.
        Assume $\surreal{a}$.
        Assume $\forall u:\Set.
            (\surreal{u}\land\birthday{u}\in\birthday{a})
            \implies\Apply{P}{u}$.
        Show that if $\surrealLe{\surrealZero}{a}$ then
            $\surreal{\surrealNonnegativeSqrt{a}}$ and
            $\surrealLe{\surrealZero}{\surrealNonnegativeSqrt{a}}$ and
            $\surrealMul{\surrealNonnegativeSqrt{a}}{
                \surrealNonnegativeSqrt{a}}=a$.
        \begin{subproof}
            Suppose $\surrealLe{\surrealZero}{a}$.
            Let $g:\Set\to\Set=\surrealNonnegativeSqrtOp$.
            Let $L=\Apply{\surrealSqrtLowerOptionsOp}{a,g}$.
            Let $R=\Apply{\surrealSqrtUpperOptionsOp}{a,g}$.
            Let $y=\surrealNonnegativeSqrt{a}$.
            Show that $\forall u:\Set.
                (\surreal{u}\land\birthday{u}\in\birthday{a}\land
                    \surrealLe{\surrealZero}{u})
                \implies
                (\surreal{\Apply{g}{u}}\land
                 \surrealLe{\surrealZero}{\Apply{g}{u}}\land
                 \surrealMul{\Apply{g}{u}}{\Apply{g}{u}}=u)$.
            \begin{subproof}
                Fix $u$.
                Assume $\surreal{u}$ and
                    $\birthday{u}\in\birthday{a}$ and
                    $\surrealLe{\surrealZero}{u}$.
                $\Apply{P}{u}$.
                Follows by
                    \auto{surreal_nonnegative_sqrt_typed,surreal_nonnegative_sqrt}.
            \end{subproof}
            $\surrealCutPair{L}{R}$
                by \auto{surreal_sqrt_options_cut_pair}.
            $y=\surrealCut{L}{R}$
                by \auto{surreal_nonnegative_sqrt_equation}.
            Hence $\surreal{y}$
                by \obv{surreal_cut_pair_is_surreal}.
            $\surrealLe{\surrealZero}{y}$
                by \auto{surreal_sqrt_cut_nonnegative}.
            not $\surrealLt{\surrealMul{y}{y}}{a}$
                by \auto{surreal_sqrt_cut_not_below}.
            not $\surrealLt{a}{\surrealMul{y}{y}}$
                by \auto{surreal_sqrt_cut_not_above}.
            $\surrealMul{y}{y}=a$
                by
                \auto{surreal_lt_trichotomy,surreal_mul_is_surreal}.
            \begin{align*}
                \surrealMul{\surrealNonnegativeSqrt{a}}{
                    \surrealNonnegativeSqrt{a}}
                    &= \surrealMul{y}{y} \\
                    &= a
            \end{align*}
        \end{subproof}
    \end{subproof}
    $\forall a:\Set.\ \surreal{a}\implies\Apply{P}{a}$
        by \auto{surreal_induction}.
    Hence $\Apply{P}{x}$.
\end{proof}

\begin{proposition}\label{surreal_nonnegative_sqrt_zero}
