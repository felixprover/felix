\section{Order and Archimedean bounds for surreal reals}

The strict order on the real carrier is the restriction of the surreal
order.  Consequently its order laws need no duplicate relation or coercion.

\begin{proposition}\label{surreal_real_lt_trichotomy}
    Suppose $x,y\in\reals$.
    Then $\surrealLt{x}{y}$ or $x=y$ or $\surrealLt{y}{x}$.
\end{proposition}
\begin{proof}
    Follows by
        \obv{surreal_lt_trichotomy,surreal_real_is_surreal}.
\end{proof}

\begin{proposition}\label{surreal_real_lt_transitive}
    Suppose $x,y,z\in\reals$.
    Suppose $\surrealLt{x}{y}$ and $\surrealLt{y}{z}$.
    Then $\surrealLt{x}{z}$.
\end{proposition}
\begin{proof}
    Follows by
        \obv{surreal_lt_transitive,surreal_real_is_surreal}.
\end{proof}

\begin{theorem}\label{surreal_reals_archimedean}
    Suppose $x,y\in\reals$.
    Suppose $\surrealLt{\surrealZero}{x}$ and
        $\surrealLe{\surrealZero}{y}$.
    Then there exists a natural number $N$ such that
        $\surrealLe{y}{
            \surrealMul{\ordinalSurreal{N}}{x}}$.
\end{theorem}
\begin{proof}
    Let $z=\surrealDiv{y}{x}$.
    $z\in\reals$ by \auto{surreal_real_division_closed}.
    $z\in\surrealStage{\suc{\naturals}}$ and
        $\surrealLt{z}{\ordinalSurreal{\naturals}}$
        by \obv{surreal_real_elim,surreal_real_bounds}.
    Take a natural number $N$ such that
        $\surrealLt{z}{\ordinalSurreal{N}}$
        by \obv{successor_omega_surreal_natural_bound_above}.
    $\surreal{\ordinalSurreal{N}}$
        by
        \obv{ordinal_surreal_is_surreal,natural_number_is_ordinal}.
    Hence
        $\surrealLt{\surrealMul{z}{x}}
            {\surrealMul{\ordinalSurreal{N}}{x}}$
        by
        \obv{surreal_pos_mul_lt_right,surreal_real_is_surreal}.
    $x\neq\surrealZero$
        by
        \obv{surreal_lt_asymmetric,surreal_real_is_surreal,empty_surreal}.
    $\surrealMul{z}{x}=y$
        by
        \auto{surreal_division_cancel_right,surreal_real_is_surreal}.
    Hence
        $\surrealLt{y}{\surrealMul{\ordinalSurreal{N}}{x}}$.
    Hence
        $\surrealLe{y}{\surrealMul{\ordinalSurreal{N}}{x}}$
        by \obv{surreal_le}.
\end{proof}
