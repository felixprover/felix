\import{surreal/absolute-value.tex}

\section{Absolute-value estimates}

The following inequalities keep their natural surreal scope.  Their later
metric applications need no separate real-only copies.

\begin{theorem}\label{surreal_abs_difference_of_sums_bound}
    Suppose $\surreal{x_1}$ and $\surreal{x_2}$ and
        $\surreal{y_1}$ and $\surreal{y_2}$.
    Then
        $\surrealLe{
            \abs{\surrealAdd{\surrealAdd{x_1}{y_1}}
                {\surrealNeg{\surrealAdd{x_2}{y_2}}}}}
            {\surrealAdd{
                \abs{\surrealAdd{x_1}{\surrealNeg{x_2}}}}
                {\abs{\surrealAdd{y_1}{\surrealNeg{y_2}}}}}$.
\end{theorem}
\begin{proof}
    $\surrealAdd{\surrealAdd{x_1}{y_1}}
        {\surrealNeg{\surrealAdd{x_2}{y_2}}}
        =\surrealAdd{
            \surrealAdd{x_1}{\surrealNeg{x_2}}}
            {\surrealAdd{y_1}{\surrealNeg{y_2}}}$
        by
        \auto{surreal_negation_additive,surreal_add_associative,surreal_add_commutative,surreal_negation_is_surreal,surreal_add_is_surreal}.
    Follows by
        \auto{surreal_abs_triangle,surreal_negation_is_surreal,surreal_add_is_surreal}.
\end{proof}

\begin{theorem}\label{surreal_abs_scaled_difference}
    Suppose $\surreal{c}$ and $\surreal{x}$ and $\surreal{y}$.
    Then
        $\abs{\surrealAdd{\surrealMul{c}{x}}
            {\surrealNeg{\surrealMul{c}{y}}}}
        =\surrealMul{\abs{c}}
            {\abs{\surrealAdd{x}{\surrealNeg{y}}}}$.
\end{theorem}
\begin{proof}
    $\surrealAdd{\surrealMul{c}{x}}
        {\surrealNeg{\surrealMul{c}{y}}}
        =\surrealMul{c}{\surrealAdd{x}{\surrealNeg{y}}}$
        by \obv{surreal_mul_subtract_right}.
    Follows by
        \auto{surreal_abs_multiplicative,surreal_negation_is_surreal,surreal_add_is_surreal}.
\end{proof}

\begin{theorem}\label{surreal_reverse_triangle_for_abs}
    Suppose $\surreal{x}$ and $\surreal{y}$.
    Then
        $\surrealLe{
            \abs{\surrealAdd{\abs{x}}{\surrealNeg{\abs{y}}}}}
            {\abs{\surrealAdd{x}{\surrealNeg{y}}}}$.
\end{theorem}
\begin{proof}
    Let $d=\abs{\surrealAdd{x}{\surrealNeg{y}}}$.
    $\surreal{d}$ and $\surreal{\abs{x}}$ and $\surreal{\abs{y}}$
        by
        \obv{surreal_abs_is_surreal,surreal_negation_is_surreal,surreal_add_is_surreal}.
    $x=\surrealAdd{\surrealAdd{x}{\surrealNeg{y}}}{y}$
        by
        \auto{surreal_add_associative,surreal_add_inverse_left,surreal_add_zero_right,surreal_negation_is_surreal}.
    Hence
        $\surrealLe{\abs{x}}{\surrealAdd{d}{\abs{y}}}$
        by
        \auto{surreal_abs_triangle,surreal_negation_is_surreal,surreal_add_is_surreal}.
    Hence
        $\surrealLe{\abs{x}}{\surrealAdd{\abs{y}}{d}}$
        by
        \obv{surreal_add_commutative,surreal_abs_is_surreal}.
    $\surrealNeg{\surrealAdd{x}{\surrealNeg{y}}}
        =\surrealAdd{y}{\surrealNeg{x}}$
        by \obv{surreal_difference_negation}.
    Hence
        $\abs{\surrealAdd{y}{\surrealNeg{x}}}=d$
        by
        \auto{surreal_abs_negation,surreal_negation_is_surreal,surreal_add_is_surreal}.
    $y=\surrealAdd{\surrealAdd{y}{\surrealNeg{x}}}{x}$
        by
        \auto{surreal_add_associative,surreal_add_inverse_left,surreal_add_zero_right,surreal_negation_is_surreal}.
    Hence
        $\surrealLe{\abs{y}}{
            \surrealAdd{\abs{\surrealAdd{y}{\surrealNeg{x}}}}
                {\abs{x}}}$
        by
        \auto{surreal_abs_triangle,surreal_negation_is_surreal,surreal_add_is_surreal}.
    Hence $\surrealLe{\abs{y}}{\surrealAdd{\abs{x}}{d}}$
        by
        \auto{surreal_add_commutative,surreal_abs_is_surreal}.
    Follows by \auto{surreal_abs_difference_le_iff}.
\end{proof}

\begin{theorem}\label{surreal_product_difference_bound}
    Suppose $\surreal{p}$ and $\surreal{q}$ and
        $\surreal{x}$ and $\surreal{y}$.
    Then
        $\surrealLe{
            \abs{\surrealAdd{\surrealMul{p}{q}}
                {\surrealNeg{\surrealMul{x}{y}}}}}
            {\surrealAdd{
                \surrealMul{\abs{p}}
                    {\abs{\surrealAdd{q}{\surrealNeg{y}}}}}
                {\surrealMul{\abs{y}}
                    {\abs{\surrealAdd{p}{\surrealNeg{x}}}}}}$.
\end{theorem}
\begin{proof}
    Let $u=\surrealMul{p}{\surrealAdd{q}{\surrealNeg{y}}}$ and
        $v=\surrealMul{y}{\surrealAdd{p}{\surrealNeg{x}}}$.
    $\surreal{u}$ and $\surreal{v}$ by
        \obv{surreal_mul_is_surreal,surreal_add_is_surreal,surreal_negation_is_surreal}.
    $u=\surrealAdd{\surrealMul{p}{q}}{\surrealNeg{\surrealMul{p}{y}}}$
        by \obv{surreal_mul_subtract_right}.
    $\surrealAdd{u}{\surrealMul{p}{y}}=\surrealMul{p}{q}$
        by \auto{surreal_add_negated_cancel_right,surreal_mul_is_surreal}.
    $v=\surrealAdd{\surrealMul{p}{y}}{\surrealNeg{\surrealMul{x}{y}}}$
        by \auto{surreal_mul_subtract_right,surreal_mul_commutative}.
    $\surrealAdd{u}{v}
        =\surrealAdd{\surrealAdd{u}{\surrealMul{p}{y}}}
            {\surrealNeg{\surrealMul{x}{y}}}$
        by \auto{surreal_add_associative,surreal_mul_is_surreal,surreal_negation_is_surreal}.
    Hence $\surrealAdd{\surrealMul{p}{q}}
        {\surrealNeg{\surrealMul{x}{y}}}=\surrealAdd{u}{v}$.
    $\surrealLe{\abs{\surrealAdd{u}{v}}}
        {\surrealAdd{\abs{u}}{\abs{v}}}$
        by \obv{surreal_abs_triangle}.
    $\abs{u}=\surrealMul{\abs{p}}{\abs{\surrealAdd{q}{\surrealNeg{y}}}}$
        and $\abs{v}=\surrealMul{\abs{y}}{\abs{\surrealAdd{p}{\surrealNeg{x}}}}$
        by
        \auto{surreal_abs_multiplicative,surreal_add_is_surreal,surreal_negation_is_surreal}.
\end{proof}

\begin{theorem}\label{surreal_product_difference_local_bound}
    Suppose $\surreal{p}$ and $\surreal{q}$ and
        $\surreal{x}$ and $\surreal{y}$ and $\surreal{d}$.
    Suppose $\surrealLt{\surrealZero}{d}$ and
        $\surrealLe{d}{\surrealOne}$.
    Suppose
        $\surrealLt{\abs{\surrealAdd{p}{\surrealNeg{x}}}}{d}$ and
        $\surrealLt{\abs{\surrealAdd{q}{\surrealNeg{y}}}}{d}$.
    Then
        $\surrealLt{
            \abs{\surrealAdd{\surrealMul{p}{q}}
                {\surrealNeg{\surrealMul{x}{y}}}}}
            {\surrealMul{d}{
                \surrealAdd{\surrealAdd{\abs{x}}{\abs{y}}}
                    {\surrealOne}}}$.
\end{theorem}
\begin{proof}
    Let $d_p=\abs{\surrealAdd{p}{\surrealNeg{x}}}$ and
        $d_q=\abs{\surrealAdd{q}{\surrealNeg{y}}}$.
    Let $c=\surrealAdd{\abs{x}}{\surrealOne}$.
    $\surreal{d_p}$ and $\surreal{d_q}$ and $\surreal{c}$
        by
        \obv{surreal_abs_is_surreal,surreal_add_is_surreal,surreal_negation_is_surreal,surreal_one_is_surreal}.
    $\surreal{\surrealAdd{p}{\surrealNeg{x}}}$ and
        $\surreal{\surrealAdd{q}{\surrealNeg{y}}}$
        by \obv{surreal_add_is_surreal,surreal_negation_is_surreal}.
    $\surrealLe{\surrealZero}{d_p}$
        by \obv{surreal_abs_nonnegative_value}.
    $\surrealLe{\surrealZero}{d_q}$
        by \obv{surreal_abs_nonnegative_value}.
    $\surrealLe{\surrealZero}{\abs{x}}$
        by \obv{surreal_abs_nonnegative_value}.
    $\surrealLe{\surrealZero}{\abs{y}}$
        by \obv{surreal_abs_nonnegative_value}.
    $\surrealLt{\surrealZero}{c}$
        by
        \auto{surreal_add_le_lt,surreal_add_zero_left,surreal_abs_is_surreal,surreal_one_is_surreal,surreal_zero_lt_one,empty_surreal}.
    $\surrealLe{
        \abs{\surrealAdd{\abs{p}}{\surrealNeg{\abs{x}}}}}{d_p}$
        by \obv{surreal_reverse_triangle_for_abs}.
    Hence $\surrealLe{\abs{p}}{\surrealAdd{\abs{x}}{d_p}}$
        by
        \auto{surreal_abs_difference_le_iff,surreal_abs_is_surreal}.
    $\surrealLt{\surrealAdd{\abs{x}}{d_p}}
        {\surrealAdd{\abs{x}}{d}}$
        by \auto{surreal_add_lt_right,surreal_abs_is_surreal}.
    Hence $\surrealLt{\abs{p}}{\surrealAdd{\abs{x}}{d}}$
        by
        \auto{surreal_le_lt_transitive,surreal_abs_is_surreal,surreal_add_is_surreal}.
    $\surrealLe{\surrealAdd{\abs{x}}{d}}{c}$
        by
        \obv{surreal_add_le_right,surreal_abs_is_surreal,surreal_one_is_surreal}.
    Hence $\surrealLe{\abs{p}}{c}$
        by
        \auto{surreal_lt_le_transitive,surreal_le,surreal_abs_is_surreal,surreal_add_is_surreal}.
    Hence $\surrealLe{\surrealMul{\abs{p}}{d_q}}
        {\surrealMul{c}{d_q}}$
        by
        \obv{surreal_nonneg_mul_le_right,surreal_abs_is_surreal}.
    $\surrealLt{\surrealMul{c}{d_q}}{\surrealMul{c}{d}}$
        by \auto{surreal_pos_mul_lt_left}.
    Hence $\surrealLt{\surrealMul{\abs{p}}{d_q}}
        {\surrealMul{c}{d}}$
        by
        \auto{surreal_le_lt_transitive,surreal_mul_is_surreal,surreal_abs_is_surreal}.
    $\surrealLe{d_p}{d}$ by \auto{surreal_le}.
    Hence $\surrealLe{\surrealMul{\abs{y}}{d_p}}
        {\surrealMul{\abs{y}}{d}}$
        by
        \obv{surreal_nonneg_mul_le_left,surreal_abs_is_surreal}.
    Hence
        $\surrealLt{
            \surrealAdd{\surrealMul{\abs{p}}{d_q}}
                {\surrealMul{\abs{y}}{d_p}}}
            {\surrealAdd{\surrealMul{c}{d}}
                {\surrealMul{\abs{y}}{d}}}$
        by
        \obv{surreal_add_lt_le,surreal_mul_is_surreal,surreal_abs_is_surreal}.
    $\surrealAdd{\surrealMul{c}{d}}{\surrealMul{\abs{y}}{d}}
        =\surrealMul{d}{
            \surrealAdd{\surrealAdd{\abs{x}}{\abs{y}}}
                {\surrealOne}}$
        by
        \auto{surreal_mul_distributive_right,surreal_mul_commutative,surreal_add_associative,surreal_add_commutative,surreal_abs_is_surreal,surreal_add_is_surreal,surreal_one_is_surreal}.
    $\surrealLe{
        \abs{\surrealAdd{\surrealMul{p}{q}}
            {\surrealNeg{\surrealMul{x}{y}}}}}
        {\surrealAdd{\surrealMul{\abs{p}}{d_q}}
            {\surrealMul{\abs{y}}{d_p}}}$
        by \auto{surreal_product_difference_bound}.
    Follows by
        \auto{surreal_le_lt_transitive,surreal_abs_is_surreal,surreal_mul_is_surreal,surreal_add_is_surreal,surreal_negation_is_surreal}.
\end{proof}

\begin{proposition}\label{surreal_reciprocal_difference_identity}
    Suppose $\surreal{x}$ and $\surreal{y}$.
    Suppose $x\neq\surrealZero$ and $y\neq\surrealZero$.
    Then
        $\surrealAdd{\surrealReciprocal{x}}
            {\surrealNeg{\surrealReciprocal{y}}}
        =\surrealMul{\surrealAdd{y}{\surrealNeg{x}}}
            {\surrealMul{\surrealReciprocal{x}}
                {\surrealReciprocal{y}}}$.
\end{proposition}
\begin{proof}
    Let $r_x=\surrealReciprocal{x}$ and
        $r_y=\surrealReciprocal{y}$.
    $\surreal{r_x}$ and $\surreal{r_y}$
        by \obv{surreal_reciprocal_is_surreal}.
    $\surrealMul{y}{\surrealMul{r_x}{r_y}}=r_x$
        by
        \auto{surreal_mul_associative,surreal_mul_commutative,surreal_reciprocal_inverse,surreal_mul_one_right,surreal_mul_is_surreal}.
    $\surrealMul{x}{\surrealMul{r_x}{r_y}}=r_y$
        by
        \auto{surreal_mul_associative,surreal_reciprocal_inverse,surreal_mul_one_left,surreal_mul_is_surreal}.
    $\surrealMul{\surrealAdd{y}{\surrealNeg{x}}}
        {\surrealMul{r_x}{r_y}}
        =\surrealAdd{
            \surrealMul{y}{\surrealMul{r_x}{r_y}}}
            {\surrealNeg{\surrealMul{x}{\surrealMul{r_x}{r_y}}}}$
        by
        \auto{surreal_mul_distributive_right,surreal_mul_neg_left,surreal_mul_neg_right,surreal_mul_is_surreal,surreal_negation_is_surreal}.
    Follows by \auto.
\end{proof}

\begin{theorem}\label{surreal_positive_reciprocal_reverses_le}
    Suppose $\surreal{a}$ and $\surreal{b}$.
    Suppose $\surrealLt{\surrealZero}{a}$ and $\surrealLe{a}{b}$.
    Then
        $\surrealLe{\surrealReciprocal{b}}{\surrealReciprocal{a}}$.
\end{theorem}
\begin{proof}
    $\surrealLt{\surrealZero}{b}$
        by \obv{surreal_lt_le_transitive,empty_surreal}.
    Hence $a\neq\surrealZero$ and $b\neq\surrealZero$
        by \obv{surreal_lt_irreflexive,empty_surreal}.
    Let $r_a=\surrealReciprocal{a}$ and
        $r_b=\surrealReciprocal{b}$.
    $\surreal{r_a}$ and $\surreal{r_b}$
        by \obv{surreal_reciprocal_is_surreal}.
    $\surrealLt{\surrealZero}{r_a}$ and
        $\surrealLt{\surrealZero}{r_b}$
        by \obv{surreal_reciprocal_positive}.
    Hence $\surrealLe{\surrealZero}{\surrealMul{r_a}{r_b}}$
        by
        \obv{surreal_mul_nonneg_nonneg,surreal_le,empty_surreal}.
    Hence
        $\surrealLe{
            \surrealMul{\surrealMul{r_a}{r_b}}{a}}
            {\surrealMul{\surrealMul{r_a}{r_b}}{b}}$
        by \obv{surreal_nonneg_mul_le_left,surreal_mul_is_surreal}.
    $\surrealMul{\surrealMul{r_a}{r_b}}{a}=r_b$ and
        $\surrealMul{\surrealMul{r_a}{r_b}}{b}=r_a$
        by
        \auto{surreal_mul_associative,surreal_mul_commutative,surreal_reciprocal_inverse_right,surreal_mul_one_left,surreal_mul_one_right,surreal_mul_is_surreal}.
    Follows by \auto.
\end{proof}

\begin{proposition}\label{surreal_abs_reciprocal}
    Suppose $\surreal{x}$ and $x\neq\surrealZero$.
    Then
        $\abs{\surrealReciprocal{x}}
            =\surrealReciprocal{\abs{x}}$.
\end{proposition}
\begin{proof}
    $\surreal{\surrealReciprocal{x}}$
        by \obv{surreal_reciprocal_is_surreal}.
    Hence $\surreal{\abs{\surrealReciprocal{x}}}$ and
        $\surreal{\abs{x}}$
        by \obv{surreal_abs_is_surreal}.
    $\surrealMul{x}{\surrealReciprocal{x}}=\surrealOne$
        by \obv{surreal_reciprocal_inverse}.
    Hence
        $\abs{\surrealMul{x}{\surrealReciprocal{x}}}
            =\abs{\surrealOne}$.
    $\abs{\surrealMul{x}{\surrealReciprocal{x}}}
        =\surrealMul{\abs{x}}{\abs{\surrealReciprocal{x}}}$
        by \obv{surreal_abs_multiplicative}.
    $\abs{\surrealOne}=\surrealOne$
        by
        \obv{surreal_abs_nonnegative,surreal_le,surreal_zero_lt_one,surreal_one_is_surreal}.
    Hence
        $\surrealMul{\abs{x}}{\abs{\surrealReciprocal{x}}}
            =\surrealOne$.
    Follows by \auto{surreal_reciprocal_unique}.
\end{proof}

\begin{lemma}\label{surreal_abs_lower_bound_from_reserve}
    Suppose $\surreal{x}$ and $\surreal{y}$ and $\surreal{r}$.
    Suppose $\surrealLt{\surrealZero}{r}$ and
        $\surrealLe{\surrealAdd{r}{r}}{\abs{x}}$ and
        $\surrealLt{\abs{\surrealAdd{y}{\surrealNeg{x}}}}{r}$.
    Then $\surrealLt{r}{\abs{y}}$.
\end{lemma}
\begin{proof}
    Let $d=\abs{\surrealAdd{y}{\surrealNeg{x}}}$.
    $\surreal{d}$ and $\surreal{\abs{x}}$ and $\surreal{\abs{y}}$
        by
        \obv{surreal_abs_is_surreal,surreal_add_is_surreal,surreal_negation_is_surreal}.
    $\surrealLe{
        \abs{\surrealAdd{\abs{x}}{\surrealNeg{\abs{y}}}}}{d}$
        by
        \auto{surreal_reverse_triangle_for_abs,surreal_abs_negation,surreal_difference_negation,surreal_negation_is_surreal,surreal_add_is_surreal}.
    Hence $\surrealLe{\abs{x}}{\surrealAdd{\abs{y}}{d}}$
        by \auto{surreal_abs_difference_le_iff}.
    $\surrealLt{\surrealAdd{\abs{y}}{d}}
        {\surrealAdd{\abs{y}}{r}}$
        by \auto{surreal_add_lt_right}.
    Hence $\surrealLt{\abs{x}}{\surrealAdd{\abs{y}}{r}}$
        by \auto{surreal_le_lt_transitive,surreal_add_is_surreal}.
    Hence
        $\surrealLt{\surrealAdd{r}{r}}
            {\surrealAdd{\abs{y}}{r}}$
        by \auto{surreal_le_lt_transitive,surreal_add_is_surreal}.
    Follows by \obv{surreal_add_lt_cancel_right}.
\end{proof}

\begin{theorem}\label{surreal_reciprocal_difference_local_bound}
    Suppose $\surreal{x}$ and $\surreal{y}$ and
        $\surreal{r}$ and $\surreal{d}$.
    Suppose $x\neq\surrealZero$ and $\surrealLt{\surrealZero}{r}$.
    Suppose $\surrealLe{\surrealAdd{r}{r}}{\abs{x}}$ and
        $\surrealLt{\surrealZero}{d}$ and $\surrealLt{d}{r}$.
    Suppose $\surrealLt{\abs{\surrealAdd{y}{\surrealNeg{x}}}}{d}$.
    Then $y\neq\surrealZero$ and
        $\surrealLt{
            \abs{\surrealAdd{\surrealReciprocal{y}}
                {\surrealNeg{\surrealReciprocal{x}}}}}
            {\surrealMul{d}{
                \surrealMul{\abs{\surrealReciprocal{x}}}
                    {\surrealReciprocal{r}}}}$.
\end{theorem}
\begin{proof}
    Let $d_{yx}=\abs{\surrealAdd{y}{\surrealNeg{x}}}$.
    $\surreal{\surrealAdd{y}{\surrealNeg{x}}}$ and $\surreal{d_{yx}}$
        by
        \obv{surreal_add_is_surreal,surreal_negation_is_surreal,surreal_abs_is_surreal}.
    $\surrealLt{d_{yx}}{r}$
        by \obv{surreal_lt_transitive,surreal_abs_is_surreal}.
    Hence $\surrealLt{r}{\abs{y}}$
        by \obv{surreal_abs_lower_bound_from_reserve}.
    $\surrealLt{\surrealZero}{\abs{y}}$
        by \auto{surreal_lt_transitive,surreal_abs_is_surreal,empty_surreal}.
    Hence $y\neq\surrealZero$
        by \obv{surreal_abs_positive_iff}.
    $\surreal{\abs{y}}$ by \obv{surreal_abs_is_surreal}.
    $\surrealLe{r}{\abs{y}}$ by \obv{surreal_le}.
    Hence
        $\surrealLe{\surrealReciprocal{\abs{y}}}
            {\surrealReciprocal{r}}$
        by \obv{surreal_positive_reciprocal_reverses_le}.
    $\abs{\surrealReciprocal{x}}=\surrealReciprocal{\abs{x}}$ and
        $\abs{\surrealReciprocal{y}}=\surrealReciprocal{\abs{y}}$
        by \obv{surreal_abs_reciprocal}.
    Hence
        $\surrealLe{\abs{\surrealReciprocal{y}}}
            {\surrealReciprocal{r}}$.
    $\surrealLt{\surrealZero}{\abs{\surrealReciprocal{x}}}$
        by
        \auto{surreal_reciprocal_nonzero,surreal_abs_positive_iff,surreal_reciprocal_is_surreal}.
    Hence
        $\surrealLe{
            \surrealMul{\abs{\surrealReciprocal{x}}}
                {\abs{\surrealReciprocal{y}}}}
            {\surrealMul{\abs{\surrealReciprocal{x}}}
                {\surrealReciprocal{r}}}$
        by
        \obv{surreal_nonneg_mul_le_left,surreal_le,surreal_abs_is_surreal,surreal_reciprocal_is_surreal,empty_surreal}.
    Let $c=\surrealMul{\abs{\surrealReciprocal{x}}}
        {\surrealReciprocal{r}}$.
    $\surrealLt{\surrealZero}{\abs{\surrealReciprocal{x}}}$
        by
        \obv{surreal_reciprocal_nonzero,surreal_abs_positive_iff,surreal_reciprocal_is_surreal}.
    $\surrealLt{\surrealZero}{\surrealReciprocal{r}}$
        by \obv{surreal_reciprocal_positive}.
    Hence $\surrealLt{\surrealZero}{c}$
        by
        \obv{surreal_mul_pos_pos,surreal_reciprocal_nonzero,surreal_abs_positive_iff,surreal_reciprocal_is_surreal,surreal_reciprocal_positive,surreal_abs_is_surreal}.
    $\surreal{c}$
        by
        \obv{surreal_mul_is_surreal,surreal_abs_is_surreal,surreal_reciprocal_is_surreal}.
    $\surrealLe{\surrealZero}{d_{yx}}$
        by
        \obv{surreal_abs_nonnegative_value,surreal_add_is_surreal,surreal_negation_is_surreal}.
    $\surrealLe{
            \surrealMul{\abs{\surrealReciprocal{x}}}
                {\abs{\surrealReciprocal{y}}}}{c}$ and
        $\surrealLe{\surrealZero}{d_{yx}}$.
    Hence
        $\surrealLe{
            \surrealMul{d_{yx}}{
                \surrealMul{\abs{\surrealReciprocal{x}}}
                    {\abs{\surrealReciprocal{y}}}}}
            {\surrealMul{d_{yx}}{c}}$
        by
        \obv{surreal_nonneg_mul_le_left,surreal_mul_is_surreal,surreal_abs_is_surreal,surreal_reciprocal_is_surreal}.
    $\surrealLt{d_{yx}}{d}$ and $\surrealLt{\surrealZero}{c}$.
    $\surrealLt{\surrealMul{d_{yx}}{c}}{\surrealMul{d}{c}}$
        by \obv{surreal_pos_mul_lt_right}.
    $\surrealLe{
            \surrealMul{d_{yx}}{
                \surrealMul{\abs{\surrealReciprocal{x}}}
                    {\abs{\surrealReciprocal{y}}}}}
            {\surrealMul{d_{yx}}{c}}$ and
        $\surrealLt{\surrealMul{d_{yx}}{c}}{\surrealMul{d}{c}}$.
    Hence
        $\surrealLt{
            \surrealMul{d_{yx}}{
                \surrealMul{\abs{\surrealReciprocal{x}}}
                    {\abs{\surrealReciprocal{y}}}}}
            {\surrealMul{d}{c}}$
        by
        \auto{surreal_le_lt_transitive,surreal_mul_is_surreal,surreal_abs_is_surreal,surreal_reciprocal_is_surreal}.
    $\surrealAdd{\surrealReciprocal{y}}
        {\surrealNeg{\surrealReciprocal{x}}}
        =\surrealMul{\surrealAdd{x}{\surrealNeg{y}}}
            {\surrealMul{\surrealReciprocal{y}}
                {\surrealReciprocal{x}}}$
        by \obv{surreal_reciprocal_difference_identity}.
    $\abs{\surrealAdd{x}{\surrealNeg{y}}}=d_{yx}$
        by
        \auto{surreal_abs_negation,surreal_difference_negation,surreal_add_is_surreal,surreal_negation_is_surreal}.
    $\abs{\surrealMul{\surrealReciprocal{y}}
        {\surrealReciprocal{x}}}
        =\surrealMul{\abs{\surrealReciprocal{y}}}
            {\abs{\surrealReciprocal{x}}}$
        by
        \obv{surreal_abs_multiplicative,surreal_reciprocal_is_surreal}.
    $\abs{\surrealMul{\surrealAdd{x}{\surrealNeg{y}}}
        {\surrealMul{\surrealReciprocal{y}}
            {\surrealReciprocal{x}}}}
        =\surrealMul{\abs{\surrealAdd{x}{\surrealNeg{y}}}}
            {\abs{\surrealMul{\surrealReciprocal{y}}
                {\surrealReciprocal{x}}}}$
        by
        \obv{surreal_abs_multiplicative,surreal_add_is_surreal,surreal_negation_is_surreal,surreal_mul_is_surreal,surreal_reciprocal_is_surreal}.
    $\abs{\surrealAdd{\surrealReciprocal{y}}
        {\surrealNeg{\surrealReciprocal{x}}}}
        =\abs{\surrealMul{\surrealAdd{x}{\surrealNeg{y}}}
            {\surrealMul{\surrealReciprocal{y}}
                {\surrealReciprocal{x}}}}$.
    Hence
        $\abs{\surrealAdd{\surrealReciprocal{y}}
            {\surrealNeg{\surrealReciprocal{x}}}}
        =\surrealMul{d_{yx}}{
            \surrealMul{\abs{\surrealReciprocal{y}}}
                {\abs{\surrealReciprocal{x}}}}$
        by \auto.
    Hence
        $\abs{\surrealAdd{\surrealReciprocal{y}}
            {\surrealNeg{\surrealReciprocal{x}}}}
        =\surrealMul{d_{yx}}{
            \surrealMul{\abs{\surrealReciprocal{x}}}
                {\abs{\surrealReciprocal{y}}}}$
        by
        \auto{surreal_mul_commutative,surreal_abs_is_surreal,surreal_reciprocal_is_surreal}.
    Follows by \auto.
\end{proof}
