\section{Elementary dyadic real estimates}

The half, quarter, and three-quarter constants are named directly from the
existing dyadic scale.  These transparent names keep later quantitative
arguments readable without introducing rational division or powers.

\begin{definition}\label{real_half}
    $\realHalf=\surrealEpsilon{\suc{\emptyset}}$.
\end{definition}

\begin{definition}\label{real_quarter}
    $\realQuarter=\surrealEpsilon{\suc{\suc{\emptyset}}}$.
\end{definition}

\begin{definition}\label{real_three_quarters}
    $\realThreeQuarters=\surrealAdd{\realHalf}{\realQuarter}$.
\end{definition}

\subsection{Typing and arithmetic}

\begin{proposition}\label{real_half_is_real}
    $\realHalf\in\reals$.
\end{proposition}
\begin{proof}
    $\suc{\emptyset}$ is a natural number by \obv{omega_successor,omega_empty}.
    Follows by \auto{real_half,surreal_epsilon_is_real}.
\end{proof}

\begin{proposition}\label{real_quarter_is_real}
    $\realQuarter\in\reals$.
\end{proposition}
\begin{proof}
    $\suc{\suc{\emptyset}}$ is a natural number
        by \obv{omega_successor,omega_empty}.
    Follows by \auto{real_quarter,surreal_epsilon_is_real}.
\end{proof}

\begin{proposition}\label{real_three_quarters_is_real}
    $\realThreeQuarters\in\reals$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{real_three_quarters,real_half_is_real,real_quarter_is_real,surreal_real_addition_closed}.
\end{proof}

\begin{proposition}\label{real_half_positive}
    $\surrealLt{\surrealZero}{\realHalf}$.
\end{proposition}
\begin{proof}
    $\suc{\emptyset}$ is a natural number by \obv{omega_successor,omega_empty}.
    Follows by \obv{real_half,surreal_epsilon_positive}.
\end{proof}

\begin{proposition}\label{real_quarter_positive}
    $\surrealLt{\surrealZero}{\realQuarter}$.
\end{proposition}
\begin{proof}
    $\suc{\suc{\emptyset}}$ is a natural number
        by \obv{omega_successor,omega_empty}.
    Follows by \obv{real_quarter,surreal_epsilon_positive}.
\end{proof}

\begin{proposition}\label{real_three_quarters_positive}
    $\surrealLt{\surrealZero}{\realThreeQuarters}$.
\end{proposition}
\begin{proof}
    $\surreal{\realHalf}$ and $\surreal{\realQuarter}$
        by
        \obv{real_half_is_real,real_quarter_is_real,surreal_real_is_surreal}.
    $\surrealLt{\surrealAdd{\surrealZero}{\surrealZero}}
        {\surrealAdd{\realHalf}{\realQuarter}}$
        by
        \obv{real_half_positive,real_quarter_positive,surreal_add_lt_both,empty_surreal}.
    $\surrealAdd{\surrealZero}{\surrealZero}=\surrealZero$
        by \obv{surreal_add_zero_left,empty_surreal}.
    Follows by \obv{real_three_quarters}.
\end{proof}

\begin{proposition}\label{real_half_plus_half_eq_one}
    $\surrealAdd{\realHalf}{\realHalf}=\surrealOne$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{real_half,surreal_epsilon_halving,surreal_epsilon_zero,omega_empty,natural_number_is_ordinal}.
\end{proof}

\begin{proposition}\label{real_quarter_plus_quarter_eq_half}
    $\surrealAdd{\realQuarter}{\realQuarter}=\realHalf$.
\end{proposition}
\begin{proof}
    $\suc{\emptyset}$ is a natural number by \obv{omega_successor,omega_empty}.
    Follows by
        \auto{real_half,real_quarter,surreal_epsilon_halving,natural_number_is_ordinal}.
\end{proof}

\begin{proposition}\label{real_quarter_plus_three_quarters_eq_one}
    $\surrealAdd{\realQuarter}{\realThreeQuarters}=\surrealOne$.
\end{proposition}
\begin{proof}
    $\surrealAdd{\realQuarter}
        {\surrealAdd{\realHalf}{\realQuarter}}
        =\surrealAdd{\surrealAdd{\realQuarter}{\realQuarter}}
            {\realHalf}$
        by
        \auto{surreal_add_associative,surreal_add_commutative,real_half_is_real,real_quarter_is_real,surreal_real_is_surreal}.
    Hence
        $\surrealAdd{\realQuarter}
            {\surrealAdd{\realHalf}{\realQuarter}}
            =\surrealAdd{\realHalf}{\realHalf}$
        by \obv{real_quarter_plus_quarter_eq_half}.
    Hence
        $\surrealAdd{\realQuarter}
            {\surrealAdd{\realHalf}{\realQuarter}}=\surrealOne$
        by \obv{real_half_plus_half_eq_one}.
    Follows by \obv{real_three_quarters}.
\end{proof}

\begin{proposition}\label{real_three_quarters_lt_one}
    $\surrealLt{\realThreeQuarters}{\surrealOne}$.
\end{proposition}
\begin{proof}
    $\surreal{\realQuarter}$ and $\surreal{\realHalf}$
        by
        \obv{real_quarter_is_real,real_half_is_real,surreal_real_is_surreal}.
    $\surrealLt{\surrealAdd{\surrealZero}{\realQuarter}}
        {\surrealAdd{\realQuarter}{\realQuarter}}$
        by \obv{real_quarter_positive,surreal_add_lt_left,empty_surreal}.
    $\surrealAdd{\surrealZero}{\realQuarter}=\realQuarter$
        by \obv{surreal_add_zero_left}.
    Hence $\surrealLt{\realQuarter}{\realHalf}$
        by \obv{real_quarter_plus_quarter_eq_half}.
    Hence
        $\surrealLt{\surrealAdd{\realHalf}{\realQuarter}}
            {\surrealAdd{\realHalf}{\realHalf}}$
        by \obv{surreal_add_lt_right}.
    Follows by
        \obv{real_three_quarters,real_half_plus_half_eq_one}.
\end{proof}

\begin{proposition}\label{real_quarter_scale_nonnegative}
    Suppose $r\in\reals$ and $\surrealLe{\surrealZero}{r}$.
    Then
        $\surrealLe{\surrealZero}{\surrealMul{\realQuarter}{r}}$.
\end{proposition}
\begin{proof}
    $\surrealLe{\surrealZero}{\realQuarter}$ by
        \obv{real_quarter_positive,surreal_le,empty_surreal}.
    Follows by
        \obv{surreal_mul_nonneg_nonneg,real_quarter_is_real,surreal_real_is_surreal,empty_surreal}.
\end{proof}

\begin{proposition}\label{real_three_quarters_scale_nonnegative}
    Suppose $r\in\reals$ and $\surrealLe{\surrealZero}{r}$.
    Then
        $\surrealLe{\surrealZero}{\surrealMul{\realThreeQuarters}{r}}$.
\end{proposition}
\begin{proof}
    $\surrealLe{\surrealZero}{\realThreeQuarters}$ by
        \obv{real_three_quarters_positive,surreal_le,empty_surreal}.
    Follows by
        \obv{surreal_mul_nonneg_nonneg,real_three_quarters_is_real,surreal_real_is_surreal,empty_surreal}.
\end{proof}

\begin{proposition}\label{real_quarter_spending_identity}
    Suppose $r\in\reals$ and $\surrealLe{\surrealZero}{r}$.
    Then
        $\surrealAdd{\surrealMul{\realQuarter}{r}}
            {\surrealMul{\realThreeQuarters}{r}}=r$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{real_quarter_plus_three_quarters_eq_one,surreal_mul_distributive_right,surreal_mul_one_left,real_quarter_is_real,real_three_quarters_is_real,surreal_real_is_surreal}.
\end{proof}

\subsection{Splitting positive tolerances}

\begin{proposition}\label{surreal_epsilon_le_one}
    Let $n$ be a natural number.
    Then $\surrealLe{\surrealEpsilon{n}}{\surrealOne}$.
\end{proposition}
\begin{proof}
    \begin{byCase}
    \caseOf{$n=\emptyset$.}
        $\surrealEpsilon{n}=\surrealOne$
            by \obv{surreal_epsilon_zero}.
        Follows by
            \obv{surreal_le_reflexive,surreal_one_is_surreal}.
    \caseOf{$n\neq\emptyset$.}
        $n$ is an ordinal by \obv{natural_number_is_ordinal}.
        Hence $\emptyset\in n$
            by \obv{ordinal_empty_or_emptyset_elem}.
        $\surrealLt{\surrealEpsilon{n}}{\surrealEpsilon{\emptyset}}$
            by \obv{surreal_epsilon_decreasing}.
        $\surrealEpsilon{\emptyset}=\surrealOne$
            by \obv{surreal_epsilon_zero}.
        Follows by \obv{surreal_le}.
    \end{byCase}
\end{proof}

\begin{theorem}\label{positive_real_has_twofold_dyadic_budget}
    Suppose $e\in\reals$ and $\surrealLt{\surrealZero}{e}$.
    Then there exists $d\in\reals$ such that
        $\surrealLt{\surrealZero}{d}$ and
        $\surrealLt{\surrealAdd{d}{d}}{e}$.
\end{theorem}
\begin{proof}
    Take a natural number $n$ such that
        $\surrealLt{\surrealMul{\surrealEpsilon{n}}{\surrealOne}}{e}$
        by
        \obv{positive_real_scaled_epsilon_below,surreal_one_is_real,surreal_zero_lt_one}.
    Let $d=\surrealEpsilon{\suc{n}}$.
    $\suc{n}$ is a natural number by \obv{omega_successor}.
    Hence $d\in\reals$ and $\surrealLt{\surrealZero}{d}$
        by \auto{surreal_epsilon_is_real,surreal_epsilon_positive}.
    $\surrealAdd{d}{d}=\surrealEpsilon{n}$
        by \auto{surreal_epsilon_halving,natural_number_is_ordinal}.
    $\surrealMul{\surrealEpsilon{n}}{\surrealOne}
        =\surrealEpsilon{n}$
        by \auto{surreal_mul_one_right,surreal_epsilon_is_surreal}.
    Follows by \auto.
\end{proof}

\begin{theorem}\label{positive_real_has_lower_scaled_budget}
    Suppose $r,c,e\in\reals$.
    Suppose $\surrealLt{\surrealZero}{r}$ and
        $\surrealLt{\surrealZero}{c}$ and
        $\surrealLt{\surrealZero}{e}$.
    Then there exists $d\in\reals$ such that
        $\surrealLt{\surrealZero}{d}$ and $\surrealLt{d}{r}$ and
        $\surrealLt{\surrealMul{d}{c}}{e}$.
\end{theorem}
\begin{proof}
    Take $d_1\in\reals$ such that
        $\surrealLt{\surrealZero}{d_1}$ and
        $\surrealLt{\surrealAdd{d_1}{d_1}}{r}$
        by \obv{positive_real_has_twofold_dyadic_budget}.
    $\surrealLt{\surrealAdd{d_1}{\surrealZero}}
        {\surrealAdd{d_1}{d_1}}$
        by
        \obv{surreal_add_lt_right,surreal_real_is_surreal,empty_surreal}.
    $\surrealAdd{d_1}{\surrealZero}=d_1$
        by \obv{surreal_add_zero_right,surreal_real_is_surreal}.
    Hence $\surrealLt{d_1}{r}$
        by
        \obv{surreal_lt_transitive,surreal_add_is_surreal,surreal_real_is_surreal}.
    Take a natural number $n$ such that
        $\surrealLt{\surrealMul{\surrealEpsilon{n}}{c}}{e}$
        by \obv{positive_real_scaled_epsilon_below}.
    Let $d_2=\surrealEpsilon{n}$.
    $d_2\in\reals$ and $\surrealLt{\surrealZero}{d_2}$
        by \auto{surreal_epsilon_is_real,surreal_epsilon_positive}.
    $\surrealLe{d_1}{d_2}$ or $\surrealLe{d_2}{d_1}$
        by \obv{surreal_le_total,surreal_real_is_surreal}.
    \begin{byCase}
    \caseOf{$\surrealLe{d_1}{d_2}$.}
        $\surrealLe{\surrealMul{d_1}{c}}{\surrealMul{d_2}{c}}$
            by
            \obv{surreal_nonneg_mul_le_right,surreal_le,surreal_real_is_surreal,empty_surreal}.
        Hence $\surrealLt{\surrealMul{d_1}{c}}{e}$
            by
            \auto{surreal_le_lt_transitive,surreal_mul_is_surreal,surreal_real_is_surreal}.
        Follows by \auto.
    \caseOf{not $\surrealLe{d_1}{d_2}$.}
        Hence $\surrealLe{d_2}{d_1}$.
        Hence $\surrealLt{d_2}{r}$
            by
            \obv{surreal_le_lt_transitive,surreal_real_is_surreal}.
        Follows by \auto.
    \end{byCase}
\end{proof}

\begin{theorem}\label{positive_real_has_threefold_dyadic_budget}
    Suppose $e\in\reals$ and $\surrealLt{\surrealZero}{e}$.
    Then there exists $d\in\reals$ such that
        $\surrealLt{\surrealZero}{d}$ and
        $\surrealLt{\surrealAdd{\surrealAdd{d}{d}}{d}}{e}$.
\end{theorem}
\begin{proof}
    Take a natural number $n$ such that
        $\surrealLt{\surrealMul{\surrealEpsilon{n}}{\surrealOne}}{e}$
        by
        \obv{positive_real_scaled_epsilon_below,surreal_one_is_real,surreal_zero_lt_one}.
    Let $d=\surrealEpsilon{\suc{\suc{n}}}$.
    $\suc{\suc{n}}$ is a natural number by \obv{omega_successor}.
    Hence $d\in\reals$ and $\surrealLt{\surrealZero}{d}$
        by \auto{surreal_epsilon_is_real,surreal_epsilon_positive}.
    $\surrealLt{\surrealAdd{\surrealAdd{d}{d}}{d}}
        {\surrealEpsilon{n}}$
        by \obv{surreal_epsilon_triple_bound}.
    $\surrealMul{\surrealEpsilon{n}}{\surrealOne}
        =\surrealEpsilon{n}$
        by \auto{surreal_mul_one_right,surreal_epsilon_is_surreal}.
    Hence $\surrealLt{\surrealEpsilon{n}}{e}$.
    $\surreal{\surrealAdd{\surrealAdd{d}{d}}{d}}$ and
        $\surreal{\surrealEpsilon{n}}$ and $\surreal{e}$
        by
        \auto{surreal_add_is_surreal,surreal_epsilon_is_surreal,surreal_real_is_surreal}.
    Follows by \auto{surreal_lt_transitive}.
\end{proof}

\subsection{Three-quarter contraction}

\begin{lemma}\label{real_three_quarters_cube_remainder}
    Let $q=\realQuarter$.
    Let $t=\realThreeQuarters$.
    Let $u=\surrealMul{q}{\surrealMul{q}{q}}$.
    Then
        $\surrealAdd{
            \surrealMul{t}{\surrealMul{t}{t}}}
            {\surrealAdd{
                \surrealAdd{\surrealAdd{\surrealAdd{u}{u}}{u}}{u}}
                {u}}
        =\realHalf$.
\end{lemma}
\begin{proof}
    Let $s=\surrealMul{q}{q}$.
    $\surreal{q}$ and $\surreal{t}$ and $\surreal{s}$ and $\surreal{u}$
        by
        \obv{real_quarter_is_real,real_three_quarters_is_real,surreal_real_is_surreal,surreal_mul_is_surreal}.
    $\surrealAdd{\surrealAdd{q}{q}}{\surrealAdd{q}{q}}
        =\surrealOne$
        by
        \obv{real_quarter_plus_quarter_eq_half,real_half_plus_half_eq_one}.
    Hence
        $\surrealMul{q}{
            \surrealAdd{\surrealAdd{q}{q}}{\surrealAdd{q}{q}}}
        =\surrealMul{q}{\surrealOne}$.
    $\surrealMul{q}{
        \surrealAdd{\surrealAdd{q}{q}}{\surrealAdd{q}{q}}}
        =\surrealAdd{
            \surrealMul{q}{\surrealAdd{q}{q}}}
            {\surrealMul{q}{\surrealAdd{q}{q}}}$
        by \obv{surreal_mul_distributive_left,surreal_add_is_surreal}.
    $\surrealMul{q}{\surrealAdd{q}{q}}=\surrealAdd{s}{s}$
        by \auto{surreal_mul_distributive_left,surreal_add_is_surreal}.
    $\surrealMul{q}{\surrealOne}=q$
        by \obv{surreal_mul_one_right}.
    Thus
        $\surrealAdd{\surrealAdd{s}{s}}{\surrealAdd{s}{s}}=q$
        by \auto.
    Hence
        $\surrealMul{q}{
            \surrealAdd{\surrealAdd{s}{s}}{\surrealAdd{s}{s}}}
        =\surrealMul{q}{q}$.
    $\surrealMul{q}{
        \surrealAdd{\surrealAdd{s}{s}}{\surrealAdd{s}{s}}}
        =\surrealAdd{
            \surrealMul{q}{\surrealAdd{s}{s}}}
            {\surrealMul{q}{\surrealAdd{s}{s}}}$
        by \obv{surreal_mul_distributive_left,surreal_add_is_surreal}.
    $\surrealMul{q}{\surrealAdd{s}{s}}=\surrealAdd{u}{u}$
        by \auto{surreal_mul_distributive_left,surreal_mul_associative}.
    $\surrealMul{q}{q}=s$.
    Thus
        $\surrealAdd{\surrealAdd{u}{u}}{\surrealAdd{u}{u}}=s$
        by \auto.
    $t=\surrealAdd{\realHalf}{q}$
        by \auto{real_three_quarters}.
    Hence $t=\surrealAdd{\surrealAdd{q}{q}}{q}$
        by \auto{real_quarter_plus_quarter_eq_half}.
    Hence $t=\surrealAdd{q}{\surrealAdd{q}{q}}$
        by \auto{surreal_add_associative}.
    Let $a=\surrealAdd{s}{\surrealAdd{s}{s}}$.
    Let $b=\surrealAdd{u}{\surrealAdd{u}{u}}$.
    Let $c=\surrealAdd{b}{\surrealAdd{b}{b}}$.
    Let $w=\surrealAdd{\surrealAdd{u}{u}}{\surrealAdd{u}{u}}$.
    Let $v=\surrealAdd{
        \surrealAdd{\surrealAdd{\surrealAdd{u}{u}}{u}}{u}}{u}$.
    Let $h=\surrealAdd{\surrealAdd{w}{w}}{\surrealAdd{w}{w}}$.
    $\surreal{a}$ and $\surreal{b}$ and $\surreal{c}$ and
    $\surreal{w}$ and $\surreal{v}$ and $\surreal{h}$
        by \auto{surreal_add_is_surreal}.
    $\surrealMul{t}{t}
        =\surrealMul{\surrealAdd{q}{\surrealAdd{q}{q}}}{t}$.
    $\surrealMul{\surrealAdd{q}{\surrealAdd{q}{q}}}{t}
            =\surrealAdd{\surrealMul{q}{t}}
                {\surrealMul{\surrealAdd{q}{q}}{t}}$
        by \obv{surreal_mul_distributive_right,surreal_add_is_surreal}.
    Hence
        $\surrealMul{t}{t}
            =\surrealAdd{\surrealMul{q}{t}}
                {\surrealMul{\surrealAdd{q}{q}}{t}}$.
    $\surrealMul{q}{t}
        =\surrealMul{q}{\surrealAdd{q}{\surrealAdd{q}{q}}}$.
    $\surrealMul{q}{\surrealAdd{q}{\surrealAdd{q}{q}}}
        =\surrealAdd{\surrealMul{q}{q}}
            {\surrealAdd{\surrealMul{q}{q}}{\surrealMul{q}{q}}}$
        by \auto{surreal_mul_distributive_left,surreal_add_is_surreal}.
    Hence
        $\surrealMul{q}{t}
            =\surrealAdd{s}{\surrealAdd{s}{s}}$
        by \auto.
    Hence $\surrealMul{q}{t}=a$.
    $\surrealMul{\surrealAdd{q}{q}}{t}
        =\surrealAdd{\surrealMul{q}{t}}{\surrealMul{q}{t}}$
        by \obv{surreal_mul_distributive_right,surreal_add_is_surreal}.
    Hence
        $\surrealMul{\surrealAdd{q}{q}}{t}=\surrealAdd{a}{a}$
        by \auto.
    Hence $\surrealMul{t}{t}=\surrealAdd{a}{\surrealAdd{a}{a}}$
        by \auto.
    Hence
        $\surrealMul{t}{\surrealMul{t}{t}}
            =\surrealMul{t}{\surrealAdd{a}{\surrealAdd{a}{a}}}$.
    $\surrealMul{q}{a}
        =\surrealMul{q}{\surrealAdd{s}{\surrealAdd{s}{s}}}$.
    $\surrealMul{q}{\surrealAdd{s}{\surrealAdd{s}{s}}}
        =\surrealAdd{\surrealMul{q}{s}}
            {\surrealAdd{\surrealMul{q}{s}}{\surrealMul{q}{s}}}$
        by \auto{surreal_mul_distributive_left,surreal_add_is_surreal}.
    $\surrealMul{q}{s}=u$
        by \auto{surreal_mul_associative}.
    Hence
        $\surrealMul{q}{a}=\surrealAdd{u}{\surrealAdd{u}{u}}$
        by \auto.
    Hence $\surrealMul{q}{a}=b$.
    $\surrealMul{q}{\surrealAdd{a}{\surrealAdd{a}{a}}}
        =\surrealAdd{\surrealMul{q}{a}}
            {\surrealAdd{\surrealMul{q}{a}}{\surrealMul{q}{a}}}$
        by \auto{surreal_mul_distributive_left,surreal_add_is_surreal}.
    Hence
        $\surrealMul{q}{\surrealAdd{a}{\surrealAdd{a}{a}}}=c$
        by \auto.
    $\surrealMul{t}{\surrealAdd{a}{\surrealAdd{a}{a}}}
        =\surrealMul{\surrealAdd{q}{\surrealAdd{q}{q}}}
            {\surrealAdd{a}{\surrealAdd{a}{a}}}$.
    $\surrealMul{\surrealAdd{q}{\surrealAdd{q}{q}}}
            {\surrealAdd{a}{\surrealAdd{a}{a}}}
        =\surrealAdd{
            \surrealMul{q}{\surrealAdd{a}{\surrealAdd{a}{a}}}}
            {\surrealMul{\surrealAdd{q}{q}}
                {\surrealAdd{a}{\surrealAdd{a}{a}}}}$
        by \obv{surreal_mul_distributive_right,surreal_add_is_surreal}.
    Hence
        $\surrealMul{t}{\surrealAdd{a}{\surrealAdd{a}{a}}}
        =\surrealAdd{
            \surrealMul{q}{\surrealAdd{a}{\surrealAdd{a}{a}}}}
            {\surrealMul{\surrealAdd{q}{q}}
                {\surrealAdd{a}{\surrealAdd{a}{a}}}}$.
    $\surrealMul{\surrealAdd{q}{q}}
        {\surrealAdd{a}{\surrealAdd{a}{a}}}
        =\surrealAdd{
            \surrealMul{q}{\surrealAdd{a}{\surrealAdd{a}{a}}}}
            {\surrealMul{q}{\surrealAdd{a}{\surrealAdd{a}{a}}}}$
        by \obv{surreal_mul_distributive_right,surreal_add_is_surreal}.
    Hence
        $\surrealMul{\surrealAdd{q}{q}}
            {\surrealAdd{a}{\surrealAdd{a}{a}}}
            =\surrealAdd{c}{c}$
        by \auto.
    Hence
        $\surrealMul{t}{\surrealAdd{a}{\surrealAdd{a}{a}}}
            =\surrealAdd{c}{\surrealAdd{c}{c}}$
        by \auto.
    Hence
        $\surrealMul{t}{\surrealMul{t}{t}}
            =\surrealAdd{c}{\surrealAdd{c}{c}}$.
    $\surrealAdd{
            \surrealAdd{u}{\surrealAdd{u}{u}}}
            {\surrealAdd{
                \surrealAdd{u}{\surrealAdd{u}{u}}}
                {\surrealAdd{u}{\surrealAdd{u}{u}}}}
        =\surrealAdd{
            \surrealAdd{
                \surrealAdd{\surrealAdd{u}{u}}{\surrealAdd{u}{u}}}
                {\surrealAdd{\surrealAdd{u}{u}}{\surrealAdd{u}{u}}}}
            {u}$
        by
        \auto{surreal_add_associative,surreal_add_commutative,surreal_add_is_surreal}.
    Hence $c=\surrealAdd{\surrealAdd{w}{w}}{u}$ by \auto.
    $w=s$.
    Hence $h=q$ by \auto.
    $\surrealAdd{
        \surrealAdd{
                \surrealAdd{\surrealAdd{w}{w}}{u}}
                {\surrealAdd{
                    \surrealAdd{\surrealAdd{w}{w}}{u}}
                    {\surrealAdd{\surrealAdd{w}{w}}{u}}}}
            {v}
        =\surrealAdd{h}{h}$
        by
        \auto{surreal_add_associative,surreal_add_commutative,surreal_add_is_surreal}.
    Hence
        $\surrealAdd{\surrealAdd{c}{\surrealAdd{c}{c}}}{v}
            =\surrealAdd{h}{h}$
        by \auto.
    Hence
        $\surrealAdd{\surrealMul{t}{\surrealMul{t}{t}}}{v}
            =\surrealAdd{q}{q}$
        by \auto.
    Hence
        $\surrealAdd{\surrealMul{t}{\surrealMul{t}{t}}}{v}
            =\realHalf$
        by \auto{real_quarter_plus_quarter_eq_half}.
    Follows by \auto.
\end{proof}

\begin{lemma}\label{real_three_quarters_cube_le_half}
    Then
        $\surrealLe{
            \surrealMul{\realThreeQuarters}
                {\surrealMul{\realThreeQuarters}{\realThreeQuarters}}}
            {\realHalf}$.
\end{lemma}
\begin{proof}
    Let $q=\realQuarter$.
    Let $t=\realThreeQuarters$.
    Let $u=\surrealMul{q}{\surrealMul{q}{q}}$.
    Let $u_2=\surrealAdd{u}{u}$.
    Let $u_3=\surrealAdd{u_2}{u}$.
    Let $u_4=\surrealAdd{u_3}{u}$.
    Let $v=\surrealAdd{u_4}{u}$.
    $\surreal{q}$ and $\surreal{\surrealMul{q}{q}}$ and $\surreal{u}$
        by
        \auto{real_quarter_is_real,surreal_real_is_surreal,surreal_mul_is_surreal}.
    $\surrealLe{\surrealZero}{q}$
        by \auto{real_quarter_positive,surreal_le,empty_surreal}.
    Hence $\surrealLe{\surrealZero}{\surrealMul{q}{q}}$
        by \obv{surreal_mul_nonneg_nonneg,empty_surreal}.
    Hence $\surrealLe{\surrealZero}{u}$
        by \auto{surreal_mul_nonneg_nonneg,empty_surreal}.
    $\surreal{u_2}$ and $\surreal{u_3}$ and $\surreal{u_4}$ and $\surreal{v}$

\begin{theorem}\label{real_three_quarters_triple_product_le_half}

\end{proof}
