\section{Infima of bounded real sets}

\import{surreal/real-completeness.tex}

The infimum of a real set is obtained from the supremum of its pointwise
negation.  This definition makes the order-dual construction explicit and
reuses the canonical real supremum rather than introducing a second cut
construction.

\begin{definition}\label{real_infimum}
    $\realInfimum{A}
        =\surrealNeg{\realSupremum{\surrealNegSet{A}}}$.
\end{definition}

\begin{proposition}\label{real_negation_set_subset_reals}
    Suppose $A\subseteq\reals$.
    Then $\surrealNegSet{A}\subseteq\reals$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{surreal_negation_set_iff,
            surreal_real_negation_closed,subseteq}.
\end{proof}

\begin{proposition}\label{real_negation_set_inhabited}
    Suppose $A$ is inhabited.
    Then $\surrealNegSet{A}$ is inhabited.
\end{proposition}
\begin{proof}
    Take $a$ such that $a\in A$.
    $\surrealNeg{a}\in\surrealNegSet{A}$ by
        \obv{surreal_negation_set_iff}.
\end{proof}

\begin{proposition}\label{real_negation_set_upper_bound}
    Suppose $A\subseteq\reals$ and $b\in\reals$.
    Suppose for every $a\in A$ we have $\surrealLe{b}{a}$.
    Then for every $y\in\surrealNegSet{A}$ we have
        $\surrealLe{y}{\surrealNeg{b}}$.
\end{proposition}
\begin{proof}
    Fix $y\in\surrealNegSet{A}$.
    Take $a\in A$ such that $y=\surrealNeg{a}$ by
        \obv{surreal_negation_set_iff}.
    $a\in\reals$ by \obv{subseteq}.
    Follows by
        \auto{surreal_negation_le_reverses,
            surreal_real_is_surreal}.
\end{proof}

\begin{theorem}\label{real_infimum_is_real}
    Suppose $A\subseteq\reals$ and $A$ is inhabited.
    Suppose $b\in\reals$ and for every $a\in A$ we have
        $\surrealLe{b}{a}$.
    Then $\realInfimum{A}\in\reals$.
\end{theorem}
\begin{proof}
    $\surrealNegSet{A}\subseteq\reals$ by
        \obv{real_negation_set_subset_reals}.
    $\surrealNegSet{A}$ is inhabited by
        \obv{real_negation_set_inhabited}.
    $\surrealNeg{b}\in\reals$ by
        \obv{surreal_real_negation_closed}.
    For every $y\in\surrealNegSet{A}$ we have
        $\surrealLe{y}{\surrealNeg{b}}$ by
        \obv{real_negation_set_upper_bound}.
    Hence $\realSupremum{\surrealNegSet{A}}\in\reals$ by
        \obv{real_supremum_is_real}.
    Hence $\surrealNeg{\realSupremum{\surrealNegSet{A}}}\in\reals$ by
        \obv{surreal_real_negation_closed}.
    Follows by \obv{real_infimum}.
\end{proof}

\begin{theorem}\label{real_infimum_lower_bound}
    Suppose $A\subseteq\reals$ and $A$ is inhabited.
    Suppose $b\in\reals$ and for every $z\in A$ we have
        $\surrealLe{b}{z}$.
    Let $a\in A$.
    Then $\surrealLe{\realInfimum{A}}{a}$.
\end{theorem}
\begin{proof}
    $\surrealNegSet{A}\subseteq\reals$ by
        \obv{real_negation_set_subset_reals}.
    $\surrealNegSet{A}$ is inhabited by
        \obv{real_negation_set_inhabited}.
    $\surrealNeg{b}\in\reals$ by
        \obv{surreal_real_negation_closed}.
    For every $y\in\surrealNegSet{A}$ we have
        $\surrealLe{y}{\surrealNeg{b}}$ by
        \obv{real_negation_set_upper_bound}.
    $\surrealNeg{a}\in\surrealNegSet{A}$ by
        \obv{surreal_negation_set_iff}.
    Hence $\surrealLe{\surrealNeg{a}}
        {\realSupremum{\surrealNegSet{A}}}$ by
        \obv{real_supremum_upper_bound}.
    $a\in\reals$ by \obv{subseteq}.
    $\realSupremum{\surrealNegSet{A}}\in\reals$ by
        \obv{real_supremum_is_real}.
    Hence $\surrealLe
        {\surrealNeg{\realSupremum{\surrealNegSet{A}}}}
        {\surrealNeg{\surrealNeg{a}}}$ by
        \obv{surreal_negation_le_reverses,surreal_real_is_surreal}.
    $\surrealNeg{\surrealNeg{a}}=a$ by
        \obv{surreal_negation_involutive,surreal_real_is_surreal}.
    Follows by \obv{real_infimum}.
\end{proof}

\begin{theorem}\label{real_infimum_greatest}
    Suppose $A\subseteq\reals$ and $A$ is inhabited.
    Suppose $b\in\reals$ and for every $z\in A$ we have
        $\surrealLe{b}{z}$.
    Suppose $u\in\reals$ and for every $v\in A$ we have
        $\surrealLe{u}{v}$.
    Then $\surrealLe{u}{\realInfimum{A}}$.
\end{theorem}
\begin{proof}
    $\surrealNegSet{A}\subseteq\reals$ by
        \obv{real_negation_set_subset_reals}.
    $\surrealNegSet{A}$ is inhabited by
        \obv{real_negation_set_inhabited}.
    $\surrealNeg{b},\surrealNeg{u}\in\reals$ by
        \obv{surreal_real_negation_closed}.
    For every $y\in\surrealNegSet{A}$ we have
        $\surrealLe{y}{\surrealNeg{b}}$ by
        \obv{real_negation_set_upper_bound}.
    Show that for every $y\in\surrealNegSet{A}$ we have
        $\surrealLe{y}{\surrealNeg{u}}$.
    \begin{subproof}
        Fix $y\in\surrealNegSet{A}$.
        Take $v\in A$ such that $y=\surrealNeg{v}$ by
            \obv{surreal_negation_set_iff}.
        $v\in\reals$ by \obv{subseteq}.
        Follows by
            \auto{surreal_negation_le_reverses,
                surreal_real_is_surreal}.
    \end{subproof}
    Hence $\surrealLe{\realSupremum{\surrealNegSet{A}}}
        {\surrealNeg{u}}$ by
        \obv{real_supremum_least}.
    $\realSupremum{\surrealNegSet{A}}\in\reals$ by
        \obv{real_supremum_is_real}.
    Hence $\surrealLe{\surrealNeg{\surrealNeg{u}}}
        {\surrealNeg{\realSupremum{\surrealNegSet{A}}}}$ by
        \obv{surreal_negation_le_reverses,surreal_real_is_surreal}.
    $\surrealNeg{\surrealNeg{u}}=u$ by
        \obv{surreal_negation_involutive,surreal_real_is_surreal}.
    Follows by \obv{real_infimum}.
\end{proof}

\begin{theorem}\label{surreal_reals_greatest_lower_bound}
    Suppose $A\subseteq\reals$ and $A$ is inhabited.
    Suppose there exists $b\in\reals$ such that for every $a\in A$
        we have $\surrealLe{b}{a}$.
    Then there exists $s\in\reals$ such that
        (for every $a\in A$ we have $\surrealLe{s}{a}$) and
        (for every $u\in\reals$ such that for every $r\in A$ we have
            $\surrealLe{u}{r}$ we have $\surrealLe{u}{s}$).
\end{theorem}
\begin{proof}
    Take $b\in\reals$ such that for every $a\in A$ we have
        $\surrealLe{b}{a}$.
    $\realInfimum{A}\in\reals$ by \obv{real_infimum_is_real}.
    For every $a\in A$ we have $\surrealLe{\realInfimum{A}}{a}$ by
        \obv{real_infimum_lower_bound}.
    For every $u\in\reals$ such that for every $r\in A$ we have
        $\surrealLe{u}{r}$ we have
        $\surrealLe{u}{\realInfimum{A}}$ by
        \obv{real_infimum_greatest}.
\end{proof}
