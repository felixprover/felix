\section{Quasigroups}

As for magmas, quasigroup carriers may be empty. The division graphs and laws
then have their ordinary empty-domain interpretation.

\begin{struct}\label{quasigroup}
    A quasigroup $A$ is a magma equipped with
    \begin{enumerate}
        \item\label{quasigroup_ldiv_graph}
            a left division
            $\ldiv[A]:\carrier[A]\times\carrier[A]\to\carrier[A]$;
        \item\label{quasigroup_rdiv_graph}
            a right division
            $\rdiv[A]:\carrier[A]\times\carrier[A]\to\carrier[A]$;
    \end{enumerate}
    such that
    \begin{enumerate}
        \item\label{quasigroup_mul_ldiv}
            for all $a,b\in\carrier[A]$ we have
                $b=\mul[A](a,\ldiv[A](a,b))$.
        \item\label{quasigroup_ldiv_mul}
            for all $a,b\in\carrier[A]$ we have
                $b=\ldiv[A](a,\mul[A](a,b))$.
        \item\label{quasigroup_rdiv_mul}
            for all $a,b\in\carrier[A]$ we have
                $b=\mul[A](\rdiv[A](b,a),a)$.
        \item\label{quasigroup_mul_rdiv}
            for all $a,b\in\carrier[A]$ we have
                $b=\rdiv[A](\mul[A](b,a),a)$.
    \end{enumerate}
\end{struct}

\begin{proposition}\label{quasigroup_ldiv_type}
    Let $A$ be a quasigroup.
    Let $a,b\in\carrier[A]$.
    Then $\ldiv[A](a,b)\in\carrier[A]$.
\end{proposition}
\begin{proof}
    $(a,b)\in\carrier[A]\times\carrier[A]$
        by \obv{times_tuple_intro}.
    Follows by \obv{quasigroup_ldiv_graph,funs_type_apply}.
\end{proof}

\begin{proposition}\label{quasigroup_rdiv_type}
    Let $A$ be a quasigroup.
    Let $a,b\in\carrier[A]$.
    Then $\rdiv[A](a,b)\in\carrier[A]$.
\end{proposition}
\begin{proof}
    $(a,b)\in\carrier[A]\times\carrier[A]$
        by \obv{times_tuple_intro}.
    Follows by \obv{quasigroup_rdiv_graph,funs_type_apply}.
\end{proof}

\begin{lemma}\label{quasigroup_cancel_left}
    Let $A$ be a quasigroup.
    Let $a,b,c\in\carrier[A]$.
    Suppose $\mul[A](a,b) = \mul[A](a,c)$.
    Then $b = c$.
\end{lemma}
\begin{proof}
    We have $b=\ldiv[A](a,\mul[A](a,b))$ by \obv{quasigroup_ldiv_mul}.
    We have $c=\ldiv[A](a,\mul[A](a,c))$ by \obv{quasigroup_ldiv_mul}.
\end{proof}

\begin{lemma}\label{quasigroup_cancel_right}
    Let $A$ be a quasigroup.
    Let $a,b,c\in\carrier[A]$.
    Suppose $\mul[A](a,c) = \mul[A](b,c)$.
    Then $a = b$.
\end{lemma}
\begin{proof}
    We have $a=\rdiv[A](\mul[A](a,c),c)$ by \obv{quasigroup_mul_rdiv}.
    We have $b=\rdiv[A](\mul[A](b,c),c)$ by \obv{quasigroup_mul_rdiv}.
\end{proof}
