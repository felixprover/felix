\section{Right modules}

A right module is not a second record type.  It is a left module over the
opposite ring, with the scalar displayed on the right.  In particular, the
stored scalar graph remains ring-first and all additive data are inherited
literally from the left-module structure.

\begin{definition}\label{right_module}
    $M$ is a right module over $R$ iff
        $R$ is a ring and
        $M$ is a left module over $\OppositeRing{R}$.
\end{definition}

\begin{proposition}\label{right_module_iff}
    $M$ is a right module over $R$ iff
        $R$ is a ring and
        $M$ is a left module over $\OppositeRing{R}$.
\end{proposition}
\begin{proof}
    Follows by \obv{right_module}.
\end{proof}

\begin{proposition}\label{right_module_ring}
    Suppose $M$ is a right module over $R$.
    Then $R$ is a ring.
\end{proposition}
\begin{proof}
    Follows by \obv{right_module}.
\end{proof}

\begin{proposition}\label{right_module_is_left_module_over_opposite}
    Suppose $M$ is a right module over $R$.
    Then $M$ is a left module over $\OppositeRing{R}$.
\end{proposition}
\begin{proof}
    Follows by \obv{right_module}.
\end{proof}

\begin{proposition}\label{left_module_over_opposite_is_right_module}
    Let $R$ be a ring.
    Suppose $M$ is a left module over $\OppositeRing{R}$.
    Then $M$ is a right module over $R$.
\end{proposition}
\begin{proof}
    Follows by \obv{right_module}.
\end{proof}

\begin{corollary}\label{right_module_is_commutative_group}
    Suppose $M$ is a right module over $R$.
    Then $M$ is a commutative group.
\end{corollary}
\begin{proof}
    Follows by
        \obv{right_module_is_left_module_over_opposite,
            module_is_commutative_group}.
\end{proof}

\begin{definition}\label{right_scalar_multiplication}
    $\rightScalarMul{M}{x}{a}=\scalarMul[M](a,x)$.
\end{definition}

\begin{proposition}\label{right_scalar_type}
    Suppose $M$ is a right module over $R$.
    Let $x\in\carrier[M]$ and $a\in\carrier[R]$.
    Then $\rightScalarMul{M}{x}{a}\in\carrier[M]$.
\end{proposition}
\begin{proof}
    Let $S=\OppositeRing{R}$.
    $M$ is a left module over $S$ by
        \obv{right_module_is_left_module_over_opposite}.
    Hence $S$ is a ring by \obv{module_scalar_ring}.
    $a\in\carrier[S]$ by \auto{opposite_ring_fields}.
    Follows by
        \auto{right_scalar_multiplication,module_scalar_type}.
\end{proof}

\begin{proposition}\label{right_scalar_vector_distributive}
    Suppose $M$ is a right module over $R$.
    Let $x,y\in\carrier[M]$ and $a\in\carrier[R]$.
    Then
        $\rightScalarMul{M}{\mul[M](x,y)}{a}
            =\mul[M](\rightScalarMul{M}{x}{a},
                \rightScalarMul{M}{y}{a})$.
\end{proposition}
\begin{proof}
    $M$ is a left module over $\OppositeRing{R}$ by
        \obv{right_module_is_left_module_over_opposite}.
    $a\in\carrier[\OppositeRing{R}]$ by
        \auto{opposite_ring_fields}.
    Follows by
        \auto{right_scalar_multiplication,
            module_scalar_vector_distributive}.
\end{proof}

\begin{proposition}\label{right_scalar_add_distributive}
    Suppose $M$ is a right module over $R$.
    Let $x\in\carrier[M]$ and $a,b\in\carrier[R]$.
    Then
        $\rightScalarMul{M}{x}{\ringAdd[R](a,b)}
            =\mul[M](\rightScalarMul{M}{x}{a},
                \rightScalarMul{M}{x}{b})$.
\end{proposition}
\begin{proof}
    Let $S=\OppositeRing{R}$.
    $M$ is a left module over $S$ by
        \obv{right_module_is_left_module_over_opposite}.
    $a,b\in\carrier[S]$ and
        $\ringAdd[S](a,b)=\ringAdd[R](a,b)$ by
        \auto{opposite_ring_fields}.
    Follows by
        \auto{right_scalar_multiplication,
            module_scalar_add_distributive}.
\end{proof}

\begin{proposition}\label{right_scalar_associative}
    Suppose $M$ is a right module over $R$.
    Let $x\in\carrier[M]$ and $a,b\in\carrier[R]$.
    Then
        $\rightScalarMul{M}{\rightScalarMul{M}{x}{a}}{b}
            =\rightScalarMul{M}{x}{\ringMul[R](a,b)}$.
\end{proposition}
\begin{proof}
    Let $S=\OppositeRing{R}$.
    $R$ is a ring by \obv{right_module_ring}.
    $M$ is a left module over $S$ by
        \obv{right_module_is_left_module_over_opposite}.
    $a,b\in\carrier[S]$ by \auto{opposite_ring_fields}.
    $\ringMul[S](b,a)=\ringMul[R](a,b)$ by
        \auto{opposite_ring_mul_apply}.
    Follows by
        \auto{right_scalar_multiplication,module_scalar_associative}.
\end{proof}

\begin{proposition}\label{right_scalar_one}
    Suppose $M$ is a right module over $R$.
    Let $x\in\carrier[M]$.
    Then $\rightScalarMul{M}{x}{\ringOne[R]}=x$.
\end{proposition}
\begin{proof}
    Let $S=\OppositeRing{R}$.
    $M$ is a left module over $S$ by
        \obv{right_module_is_left_module_over_opposite}.
    $\ringOne[S]=\ringOne[R]$ by
        \auto{opposite_ring_fields}.
    Follows by
        \auto{right_scalar_multiplication,module_scalar_one}.
\end{proof}
