\section{Submodules}

A submodule is a raw carrier inside an explicit ambient module.  Its additive
closure is expressed once through the canonical additive-group reduct; scalar
closure is the only additional carrier condition.

\begin{definition}\label{submodule}
    $N$ is a submodule of $M$ over $R$ iff
        $M$ is a left module over $R$ and
        $\subgroup{N}{\ModuleAdditiveGroup{M}}$ and
        for all $a,x$ such that
            $a\in\carrier[R]$ and $x\in N$ we have
            $\scalarMul[M](a,x)\in N$.
\end{definition}

\begin{theorem}\label{submodule_expanded_iff}
    $N$ is a submodule of $M$ over $R$ iff
        $M$ is a left module over $R$ and
        $N\subseteq\carrier[M]$ and
        $\neutral[M]\in N$ and
        (for all $x,y\in N$ we have $\mul[M](x,y)\in N$) and
        (for all $x\in N$ we have $\groupInv[M](x)\in N$) and
        (for all $a,x$ such that
            $a\in\carrier[R]$ and $x\in N$ we have
            $\scalarMul[M](a,x)\in N$).
\end{theorem}
\begin{proof}
    Let $G=\ModuleAdditiveGroup{M}$.
    Show that if $N$ is a submodule of $M$ over $R$, then
        $M$ is a left module over $R$ and
        $N\subseteq\carrier[M]$ and
        $\neutral[M]\in N$ and
        (for all $x,y\in N$ we have $\mul[M](x,y)\in N$) and
        (for all $x\in N$ we have $\groupInv[M](x)\in N$) and
        (for all $a,x$ such that $a\in\carrier[R]$ and $x\in N$
            we have $\scalarMul[M](a,x)\in N$).
    \begin{subproof}
        Assume $N$ is a submodule of $M$ over $R$.
        $M$ is a left module over $R$ and $\subgroup{N}{G}$ by
            \obv{submodule}.
        $N\subseteq\carrier[G]$ and $\neutral[G]\in N$ by
            \obv{subgroup_subseteq,subgroup_neutral}.
        For all $x,y\in N$ we have $\mul[G](x,y)\in N$ by
            \obv{subgroup_mul}.
        For every $x\in N$ we have $\groupInv[G](x)\in N$ by
            \obv{subgroup_inv}.
        Follows by
            \auto{submodule,module_additive_group_fields}.
    \end{subproof}
    Show that if
        $M$ is a left module over $R$ and
        $N\subseteq\carrier[M]$ and
        $\neutral[M]\in N$ and
        (for all $x,y\in N$ we have $\mul[M](x,y)\in N$) and
        (for all $x\in N$ we have $\groupInv[M](x)\in N$) and
        (for all $a,x$ such that $a\in\carrier[R]$ and $x\in N$
            we have $\scalarMul[M](a,x)\in N$),
        then $N$ is a submodule of $M$ over $R$.
    \begin{subproof}
        Assume
            $M$ is a left module over $R$ and
            $N\subseteq\carrier[M]$ and
            $\neutral[M]\in N$ and
            (for all $x,y\in N$ we have $\mul[M](x,y)\in N$) and
            (for all $x\in N$ we have $\groupInv[M](x)\in N$) and
            (for all $a,x$ such that $a\in\carrier[R]$ and $x\in N$
                we have $\scalarMul[M](a,x)\in N$).
        $G$ is a group by \obv{module_additive_group_is_group}.
        $N\subseteq\carrier[G]$ and $\neutral[G]\in N$ by
            \auto{module_additive_group_fields}.
        For all $x,y\in N$ we have $\mul[G](x,y)\in N$ by
            \auto{module_additive_group_fields}.
        For every $x\in N$ we have $\groupInv[G](x)\in N$ by
            \auto{module_additive_group_fields}.
        Hence $\subgroup{N}{G}$ by \obv{subgroup}.
        Follows by \obv{submodule}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{proposition}\label{submodule_module}

\end{proposition}
\begin{proof}
    Follows by \obv{submodule}.
\end{proof}

\subsection{Elementary submodule constructions}

\begin{proposition}\label{zero_submodule}
    Let $R$ be a ring.
    Let $M$ be a left module over $R$.
    Then $\trivialSubgroupCarrier{\ModuleAdditiveGroup{M}}$
        is a submodule of $M$ over $R$.
\end{proposition}
\begin{proof}
    Let $G=\ModuleAdditiveGroup{M}$ and
        $Z=\trivialSubgroupCarrier{G}$.
    $G$ is a group by \obv{module_additive_group_is_group}.
    Hence $\subgroup{Z}{G}$ by \auto{trivial_carrier_subgroup}.
    Show that for all $a,x$ such that $a\in\carrier[R]$ and $x\in Z$
        we have $\scalarMul[M](a,x)\in Z$.
    \begin{subproof}
        Fix $a,x$.
        Assume $a\in\carrier[R]$ and $x\in Z$.
        $x=\neutral[G]$ by
            \auto{trivial_subgroup_carrier,singleton_iff}.
        Hence $x=\neutral[M]$ by
            \auto{module_additive_group_fields}.
        $\scalarMul[M](a,x)=\neutral[M]$ by
            \auto{module_scalar_vector_zero}.
        Follows by
            \obv{trivial_subgroup_carrier,singleton_intro,
                module_additive_group_fields}.
    \end{subproof}
    Follows by \obv{submodule}.
\end{proof}

\begin{proposition}\label{module_carrier_submodule}
    Let $R$ be a ring.
    Let $M$ be a left module over $R$.
    Then $\carrier[M]$ is a submodule of $M$ over $R$.
\end{proposition}
\begin{proof}
    Let $G=\ModuleAdditiveGroup{M}$.
    $G$ is a group by \obv{module_additive_group_is_group}.
    $\carrier[G]=\carrier[M]$ by
        \auto{module_additive_group_fields}.
    Hence $\subgroup{\carrier[M]}{G}$ by
        \auto{full_carrier_subgroup}.
    For all $a,x$ such that $a\in\carrier[R]$ and
        $x\in\carrier[M]$ we have
        $\scalarMul[M](a,x)\in\carrier[M]$ by
        \obv{module_scalar_type}.
    Follows by \auto{submodule}.
\end{proof}

\begin{theorem}\label{submodule_intersection}
    Suppose $N$ is a submodule of $M$ over $R$ and
        $P$ is a submodule of $M$ over $R$.
    Then $N\inter P$ is a submodule of $M$ over $R$.
\end{theorem}
\begin{proof}
    Let $G=\ModuleAdditiveGroup{M}$.
    $M$ is a left module over $R$ by \obv{submodule_module}.
    $\subgroup{N}{G}$ and $\subgroup{P}{G}$ by
        \obv{submodule_additive_subgroup}.
    Hence $\subgroup{N\inter P}{G}$ by
        \obv{subgroup_intersection}.
    Show that for all $a,x$ such that $a\in\carrier[R]$ and
        $x\in N\inter P$ we have $\scalarMul[M](a,x)\in N\inter P$.
    \begin{subproof}
        Follows by
            \obv{inter_elim_left,inter_elim_right,
                submodule_scalar,inter_intro}.
    \end{subproof}
    Follows by \auto{submodule}.
\end{proof}

\begin{theorem}\label{submodule_indexed_intersection}
    Let $M$ be a left module over $R$.
    Let $F\in\funs{I}{\pow{\carrier[M]}}$.
    Suppose for every $i\in I$ we have
        $F(i)$ is a submodule of $M$ over $R$.
    Then $\indexedIntersection{I}{F}{\carrier[M]}$
        is a submodule of $M$ over $R$.
\end{theorem}
\begin{proof}
    Let $G=\ModuleAdditiveGroup{M}$ and
        $C=\indexedIntersection{I}{F}{\carrier[M]}$.
    $G$ is a group by \obv{module_additive_group_is_group}.
    $F\in\funs{I}{\pow{\carrier[G]}}$ by
        \auto{module_additive_group_fields}.
    For every $i\in I$ we have $\subgroup{F(i)}{G}$ by
        \auto{submodule_additive_subgroup}.
    Hence $\subgroup{C}{G}$ by
        \auto{subgroup_indexed_intersection,
            module_additive_group_fields}.
    Show that for all $a,x$ such that $a\in\carrier[R]$ and $x\in C$
        we have $\scalarMul[M](a,x)\in C$.
    \begin{subproof}
        Fix $a,x$.
        Assume $a\in\carrier[R]$ and $x\in C$.
        Hence $x\in\carrier[M]$ and for every $i\in I$ we have
            $x\in F(i)$ by \obv{indexed_intersection_iff}.
        $\scalarMul[M](a,x)\in\carrier[M]$ by
            \obv{module_scalar_type}.
        For every $i\in I$ we have $\scalarMul[M](a,x)\in F(i)$ by
            \auto{submodule_scalar}.
        Follows by \obv{indexed_intersection}.
    \end{subproof}
    Follows by \obv{submodule}.
\end{proof}

\begin{theorem}\label{directed_submodule_union}
    Let $M$ be a left module over $R$.
    Let $F\in\funs{I}{\pow{\carrier[M]}}$.
    Suppose $I$ is inhabited.
    Suppose for every $i\in I$ we have
        $F(i)$ is a submodule of $M$ over $R$.
    Suppose for all $i,j\in I$ there exists $k\in I$ such that
        $F(i)\union F(j)\subseteq F(k)$.
    Then $\unions{\ran{F}}$ is a submodule of $M$ over $R$.
\end{theorem}
\begin{proof}
    Let $G=\ModuleAdditiveGroup{M}$ and $U=\unions{\ran{F}}$.
    $G$ is a group by \obv{module_additive_group_is_group}.
    $F\in\funs{I}{\pow{\carrier[G]}}$ by
        \auto{module_additive_group_fields}.
    For every $i\in I$ we have $\subgroup{F(i)}{G}$ by
        \auto{submodule_additive_subgroup}.
    Hence $\subgroup{U}{G}$ by \auto{directed_subgroup_union}.
    Show that for all $a,x$ such that $a\in\carrier[R]$ and $x\in U$
        we have $\scalarMul[M](a,x)\in U$.
    \begin{subproof}
        Fix $a,x$.
        Assume $a\in\carrier[R]$ and $x\in U$.
        Take $A\in\ran{F}$ such that $x\in A$ by \obv{unions_iff}.
        Take $i\in\dom{F}$ such that $F(i)=A$ by
            \auto{fun_ran_iff,funs_is_function}.
        Hence $i\in I$ by \obv{funs}.
        $\scalarMul[M](a,x)\in F(i)$ by \auto{submodule_scalar}.
        $F(i)\in\ran{F}$ by
            \obv{fun_ran_iff,funs_is_function}.
        Follows by \auto{unions_iff}.
    \end{subproof}
    Follows by \obv{submodule}.
\end{proof}

\subsection{The module induced on a submodule}

\begin{definition}\label{induced_submodule}
    $\InducedSubmodule{R}{M}{N}=
        \structure[left module over R]{
            \carrier=
                \carrier[\subgroupStructure{\ModuleAdditiveGroup{M}}{N}],
            \mul=\mul[\subgroupStructure{\ModuleAdditiveGroup{M}}{N}],
            \neutral=
                \neutral[\subgroupStructure{\ModuleAdditiveGroup{M}}{N}],
            \groupInv=
                \groupInv[\subgroupStructure{\ModuleAdditiveGroup{M}}{N}],
            \scalarMul=
                \restrl{\scalarMul[M]}{\carrier[R]\times N}
        }$.
\end{definition}

\begin{proposition}\label{induced_submodule_fields}
    Let $K=\subgroupStructure{\ModuleAdditiveGroup{M}}{N}$.
    Then
        $\carrier[\InducedSubmodule{R}{M}{N}]=\carrier[K]$ and
        $\mul[\InducedSubmodule{R}{M}{N}]=\mul[K]$ and
        $\neutral[\InducedSubmodule{R}{M}{N}]=\neutral[K]$ and
        $\groupInv[\InducedSubmodule{R}{M}{N}]=\groupInv[K]$ and
        $\scalarMul[\InducedSubmodule{R}{M}{N}]
            =\restrl{\scalarMul[M]}{\carrier[R]\times N}$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{induced_submodule,
            left_moduleaggregateprojection_carrier,
            left_moduleaggregateprojection_mul,
            left_moduleaggregateprojection_neutral,
            left_moduleaggregateprojection_groupInv,
            left_moduleaggregateprojection_scalarMul}.
\end{proof}

\begin{proposition}\label{induced_submodule_carrier}
    Suppose $N$ is a submodule of $M$ over $R$.
    Then $\carrier[\InducedSubmodule{R}{M}{N}]=N$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{induced_submodule_fields,subgroup_structure_carrier}.
\end{proof}

\begin{proposition}\label{induced_submodule_scalar_graph}
    Suppose $N$ is a submodule of $M$ over $R$.
    Then
        $\restrl{\scalarMul[M]}{\carrier[R]\times N}
            \in\funs{\carrier[R]\times N}{N}$.
\end{proposition}
\begin{proof}
    Let $f=\scalarMul[M]$ and
        $s=\restrl{f}{\carrier[R]\times N}$.
    $M$ is a left module over $R$ by \obv{submodule_module}.
    $N\subseteq\carrier[M]$ by \obv{submodule_subseteq}.
    $\carrier[R]\times N
        \subseteq\carrier[R]\times\carrier[M]$ by
        \obv{times_subseteq_right}.
    $f\in\funs{\carrier[R]\times\carrier[M]}{\carrier[M]}$ by
        \obv{module_scalar_graph}.
    Hence $\carrier[R]\times N\subseteq\dom{f}$ by \auto{funs}.
    $s\in\funs{\carrier[R]\times N}{\carrier[M]}$ by
        \auto{funs_restrl}.
    Hence $s$ is a function and
        $\dom{s}=\carrier[R]\times N$ by \obv{funs_elim}.
    Show that $s$ is a function to $N$.
    \begin{subproof}
        Show that for every $p\in\dom{s}$ we have $s(p)\in N$.
        \begin{subproof}
            Fix $p\in\dom{s}$.
            Hence $p\in\carrier[R]\times N$.
            Take $a,x$ such that $a\in\carrier[R]$ and $x\in N$ and
                $p=(a,x)$ by \obv{times_elem_is_tuple}.
            $f$ is a function by \obv{funs_is_function}.
            $s(p)=f(p)$ by \auto{restrl_of_function_apply}.
            Hence $s(p)=\scalarMul[M](a,x)$.
            Follows by \auto{submodule_scalar}.
        \end{subproof}
    \end{subproof}
    Follows by \auto{funs_intro}.
\end{proof}

\begin{proposition}\label{induced_submodule_scalar_apply}
    Suppose $N$ is a submodule of $M$ over $R$.
    Let $a\in\carrier[R]$ and $x\in N$.
    Then
        $\scalarMul[\InducedSubmodule{R}{M}{N}](a,x)
            =\scalarMul[M](a,x)$.
\end{proposition}
\begin{proof}
    Let $P=\InducedSubmodule{R}{M}{N}$ and $f=\scalarMul[M]$.
    Let $s=\restrl{f}{\carrier[R]\times N}$.
    $f$ is a function by
        \auto{submodule_module,module_scalar_graph,funs_is_function}.
    $N\subseteq\carrier[M]$ by \obv{submodule_subseteq}.
    $\carrier[R]\times N
        \subseteq\carrier[R]\times\carrier[M]$ by
        \obv{times_subseteq_right}.
    Hence $\carrier[R]\times N\subseteq\dom{f}$ by
        \auto{submodule_module,module_scalar_graph,funs}.
    $(a,x)\in\carrier[R]\times N$ by \obv{times_tuple_intro}.
    Hence $s((a,x))=f((a,x))$ by
        \auto{restrl_of_function_apply}.
    $\scalarMul[P]=s$ by \auto{induced_submodule_fields}.
\end{proof}

\begin{theorem}\label{induced_submodule_is_exact}
    Suppose $N$ is a submodule of $M$ over $R$.
    Then $\InducedSubmodule{R}{M}{N}$ is an exact left module over $R$.
\end{theorem}
\begin{proof}
    Let $G=\ModuleAdditiveGroup{M}$.
    Let $K=\subgroupStructure{G}{N}$.
    Let $P=\InducedSubmodule{R}{M}{N}$.
    $M$ is a left module over $R$ and $\subgroup{N}{G}$ by
        \obv{submodule_module,submodule_additive_subgroup}.
    $N\subseteq\carrier[M]$ by \obv{submodule_subseteq}.
    $G$ is an exact commutative group by
        \auto{module_additive_group_is_exact_commutative_group}.
    $K$ is an exact group by
        \auto{subgroup_structure_is_exact_group}.
    Hence $K$ is a group and $K$ is a monoid and
        $K$ is a unital magma and $K$ is a magma by
        \obv{group_is_monoid,monoid_is_unital_magma,
            monoid_is_magma}.
    Show that for all $x,y\in\carrier[K]$ we have
        $\mul[K](x,y)=\mul[K](y,x)$.
    \begin{subproof}
        Fix $x,y\in\carrier[K]$.
        Hence $x,y\in N$ by \auto{subgroup_structure_carrier}.
        Hence $x,y\in\carrier[G]$ by
            \auto{subgroup_subseteq,subseteq}.
        $\mul[K](x,y)=\mul[G](x,y)$ and
            $\mul[K](y,x)=\mul[G](y,x)$ by
            \auto{subgroup_structure_mul_apply}.
        Follows by \obv{commutative_group_commutative}.
    \end{subproof}
    $\carrier[P]=\carrier[K]$ and $\mul[P]=\mul[K]$ and
        $\neutral[P]=\neutral[K]$ and $\groupInv[P]=\groupInv[K]$ by
        \auto{induced_submodule_fields}.
    $\mul[P]\in\funs{\carrier[P]\times\carrier[P]}{\carrier[P]}$ by
        \auto{magma_mul_graph}.
    $\neutral[P]\in\carrier[P]$ by \auto{unitalmagma_type}.
    $\groupInv[P]\in\funs{\carrier[P]}{\carrier[P]}$ by
        \auto{group_inverse_graph}.
    For all $x,y,z\in\carrier[P]$ we have
        $\mul[P](x,\mul[P](y,z))=\mul[P](\mul[P](x,y),z)$ by
        \auto{monoid_assoc}.
    For every $x\in\carrier[P]$ we have
        $\mul[P](x,\neutral[P])=x$ and
        $\mul[P](\neutral[P],x)=x$ by
        \auto{unitalmagma_right,unitalmagma_left}.
    For every $x\in\carrier[P]$ we have
        $\mul[P](x,\groupInv[P](x))=\neutral[P]$ and
        $\mul[P](\groupInv[P](x),x)=\neutral[P]$ by
        \auto{group_inverse_right,group_inverse_left}.
    Hence $P$ is a group by
        \auto{induced_submodule,left_moduleaggregateexact,
            magma,unitalmagma,monoid,group}.
    For all $x,y\in\carrier[P]$ we have
        $\mul[P](x,y)=\mul[P](y,x)$ by \auto.
    Hence $P$ is a commutative group by
        \auto{group,commutative_group}.
    $\scalarMul[P]\in
        \funs{\carrier[R]\times\carrier[P]}{\carrier[P]}$ by
        \auto{induced_submodule_fields,subgroup_structure_carrier,
            induced_submodule_scalar_graph}.

    Show that for all $a,x,y$ such that $a\in\carrier[R]$ and
        $x,y\in\carrier[P]$ we have
        $\scalarMul[P](a,\mul[P](x,y))
            =\mul[P](\scalarMul[P](a,x),\scalarMul[P](a,y))$.
    \begin{subproof}
        Fix $a,x,y$.
        Assume $a\in\carrier[R]$ and $x,y\in\carrier[P]$.
        Hence $x,y\in N$ by \auto{subgroup_structure_carrier}.
        Hence $x,y\in\carrier[M]$ by \obv{subseteq}.
        $\mul[G](x,y)\in N$ by \obv{subgroup_mul}.
        Hence $\mul[M](x,y)\in N$ by
            \auto{module_additive_group_fields}.
        $\mul[P](x,y)=\mul[M](x,y)$ by
            \auto{subgroup_structure_mul_apply,
                module_additive_group_fields}.
        Hence $\mul[P](x,y)\in N$.
        $\scalarMul[M](a,x),\scalarMul[M](a,y)\in N$ by
            \obv{submodule_scalar}.
        $\scalarMul[P](a,x),\scalarMul[P](a,y)\in N$ by
            \auto{induced_submodule_scalar_apply}.
        $\scalarMul[P](a,\mul[P](x,y))
            =\scalarMul[M](a,\mul[M](x,y))$ by
            \auto{induced_submodule_scalar_apply}.
        $\mul[P](\scalarMul[P](a,x),\scalarMul[P](a,y))
            =\mul[M](\scalarMul[M](a,x),\scalarMul[M](a,y))$ by
            \auto{induced_submodule_scalar_apply,
                subgroup_structure_mul_apply,
                module_additive_group_fields}.
        Follows by \auto{module_scalar_vector_distributive}.
    \end{subproof}
    Show that for all $a,b,x$ such that $a,b\in\carrier[R]$ and
        $x\in\carrier[P]$ we have
        $\scalarMul[P](\ringAdd[R](a,b),x)
            =\mul[P](\scalarMul[P](a,x),\scalarMul[P](b,x))$.
    \begin{subproof}
        Fix $a,b,x$.
        Assume $a,b\in\carrier[R]$ and $x\in\carrier[P]$.
        Hence $x\in N$ by \auto{subgroup_structure_carrier}.
        Hence $x\in\carrier[M]$ by \obv{subseteq}.
        $\ringAdd[R](a,b)\in\carrier[R]$ by \obv{ring_add_type}.
        $\scalarMul[P](a,x),\scalarMul[P](b,x)\in N$ by
            \auto{induced_submodule_scalar_apply,submodule_scalar}.
        Follows by
            \auto{induced_submodule_scalar_apply,
                subgroup_structure_mul_apply,
                module_additive_group_fields,
                module_scalar_add_distributive}.
    \end{subproof}
    Show that for all $a,b,x$ such that $a,b\in\carrier[R]$ and
        $x\in\carrier[P]$ we have
        $\scalarMul[P](a,\scalarMul[P](b,x))
            =\scalarMul[P](\ringMul[R](a,b),x)$.
    \begin{subproof}
        Fix $a,b,x$.
        Assume $a,b\in\carrier[R]$ and $x\in\carrier[P]$.
        Hence $x\in N$ by \auto{subgroup_structure_carrier}.
        Hence $x\in\carrier[M]$ by \obv{subseteq}.
        $\ringMul[R](a,b)\in\carrier[R]$ by \obv{ring_mul_type}.
        $\scalarMul[P](b,x)\in N$ by
            \auto{induced_submodule_scalar_apply,submodule_scalar}.
        Follows by
            \auto{induced_submodule_scalar_apply,
                module_scalar_associative}.
    \end{subproof}
    Show that for every $x\in\carrier[P]$ we have
        $\scalarMul[P](\ringOne[R],x)=x$.
    \begin{subproof}
        Fix $x\in\carrier[P]$.
        Hence $x\in N$ by \auto{subgroup_structure_carrier}.
        Hence $x\in\carrier[M]$ by \obv{subseteq}.
        $\ringOne[R]\in\carrier[R]$ by \obv{ring_one_type}.
        Follows by
            \auto{induced_submodule_scalar_apply,module_scalar_one}.
    \end{subproof}
    $P$ is a left module over $R$ by
        \auto{induced_submodule,left_moduleaggregateexact,left_module}.
    Follows by
        \auto{induced_submodule,left_moduleaggregateexact}.
\end{proof}

\begin{theorem}\label{induced_submodule_additive_group}
    Suppose $N$ is a submodule of $M$ over $R$.
    Then
        $\ModuleAdditiveGroup{\InducedSubmodule{R}{M}{N}}
            =\subgroupStructure{\ModuleAdditiveGroup{M}}{N}$.
\end{theorem}
\begin{proof}
    Let $P=\InducedSubmodule{R}{M}{N}$ and
        $K=\subgroupStructure{\ModuleAdditiveGroup{M}}{N}$.
    $P$ is an exact left module over $R$ by
        \auto{induced_submodule_is_exact}.
    Hence $\ModuleAdditiveGroup{P}$ is an exact commutative group by
        \obv{module_additive_group_is_exact_commutative_group}.
    Hence $\ModuleAdditiveGroup{P}$ is an exact group by
        \obv{commutative_group_is_group}.
    $K$ is an exact group by
        \auto{submodule_additive_subgroup,
            subgroup_structure_is_exact_group}.
    $\carrier[\ModuleAdditiveGroup{P}]=\carrier[K]$ and
        $\mul[\ModuleAdditiveGroup{P}]=\mul[K]$ and
        $\neutral[\ModuleAdditiveGroup{P}]=\neutral[K]$ and
        $\groupInv[\ModuleAdditiveGroup{P}]=\groupInv[K]$ by
        \auto{module_additive_group_fields,induced_submodule_fields}.
    Follows by \auto{exact_group_extensionality}.
\end{proof}

\begin{theorem}\label{submodule_of_induced_submodule}
    Let $R$ be a ring.
    Let $M$ be a left module over $R$.
    Suppose $A$ is a submodule of $M$ over $R$.
    Suppose $B$ is a submodule of $M$ over $R$ and $B\subseteq A$.
    Then $B$ is a submodule of $\InducedSubmodule{R}{M}{A}$ over $R$.
\end{theorem}
\begin{proof}
    Let $P=\InducedSubmodule{R}{M}{A}$ and
        $G=\ModuleAdditiveGroup{M}$.
    $P$ is a left module over $R$ by
        \auto{induced_submodule_is_exact}.
    $\subgroup{A}{G}$ and $\subgroup{B}{G}$ by
        \obv{submodule_additive_subgroup}.
    Hence $\subgroup{B}{\subgroupStructure{G}{A}}$ by
        \obv{subgroup_of_induced_group_iff}.
    $\ModuleAdditiveGroup{P}=\subgroupStructure{G}{A}$ by
        \auto{induced_submodule_additive_group}.
    Hence $\subgroup{B}{\ModuleAdditiveGroup{P}}$.
    Show that for all $a,x$ such that $a\in\carrier[R]$ and $x\in B$
        we have $\scalarMul[P](a,x)\in B$.
    \begin{subproof}
        Fix $a,x$.
        Assume $a\in\carrier[R]$ and $x\in B$.
        Hence $x\in A$ by \obv{subseteq}.
        $\scalarMul[P](a,x)=\scalarMul[M](a,x)$ by
            \auto{induced_submodule_scalar_apply}.
        $\scalarMul[M](a,x)\in B$ by \obv{submodule_scalar}.
    \end{subproof}
    Follows by \auto{submodule}.
\end{proof}

\begin{theorem}\label{iterated_induced_submodule}
    Let $R$ be a ring.
    Let $M$ be a left module over $R$.
    Suppose $A$ is a submodule of $M$ over $R$.
    Suppose $B$ is a submodule of $M$ over $R$ and $B\subseteq A$.
    Then
        $\InducedSubmodule{R}{\InducedSubmodule{R}{M}{A}}{B}
            =\InducedSubmodule{R}{M}{B}$.
\end{theorem}
\begin{proof}
    Let $G=\ModuleAdditiveGroup{M}$ and
        $P=\InducedSubmodule{R}{M}{A}$.
    Let $S=\InducedSubmodule{R}{P}{B}$ and
        $T=\InducedSubmodule{R}{M}{B}$.
    $B$ is a submodule of $P$ over $R$ by
        \auto{submodule_of_induced_submodule}.
    Hence $S$ is an exact left module over $R$ and
        $T$ is an exact left module over $R$ by
        \auto{induced_submodule_is_exact}.
    $\subgroup{A}{G}$ and $\subgroup{B}{G}$ by
        \obv{submodule_additive_subgroup}.
    $\ModuleAdditiveGroup{P}=\subgroupStructure{G}{A}$ by
        \auto{induced_submodule_additive_group}.
    $\ModuleAdditiveGroup{S}
        =\subgroupStructure{\ModuleAdditiveGroup{P}}{B}$ by
        \auto{induced_submodule_additive_group}.
    $\subgroupStructure{\subgroupStructure{G}{A}}{B}
        =\subgroupStructure{G}{B}$ by
        \obv{iterated_subgroup_structure}.
    $\ModuleAdditiveGroup{T}=\subgroupStructure{G}{B}$ by
        \auto{induced_submodule_additive_group}.
    Hence $\ModuleAdditiveGroup{S}=\ModuleAdditiveGroup{T}$.
    Hence $\carrier[S]=\carrier[T]$ and
        $\mul[S]=\mul[T]$ and
        $\neutral[S]=\neutral[T]$ and
        $\groupInv[S]=\groupInv[T]$ by
        \auto{module_additive_group_fields}.
    $\scalarMul[P]
        =\restrl{\scalarMul[M]}{\carrier[R]\times A}$ by
        \auto{induced_submodule_fields}.
    $\scalarMul[S]
        =\restrl{\scalarMul[P]}{\carrier[R]\times B}$ by
        \auto{induced_submodule_fields}.
    $\carrier[R]\times B\subseteq\carrier[R]\times A$ by
        \obv{times_subseteq_right}.
    Hence
        $\restrl{\restrl{\scalarMul[M]}{\carrier[R]\times A}}{
            \carrier[R]\times B}
        =\restrl{\scalarMul[M]}{\carrier[R]\times B}$ by
        \obv{restrl_restrl}.
    Hence $\scalarMul[S]
        =\restrl{\scalarMul[M]}{\carrier[R]\times B}$.
    $\scalarMul[T]
        =\restrl{\scalarMul[M]}{\carrier[R]\times B}$ by
        \auto{induced_submodule_fields}.
    Hence $\scalarMul[S]=\scalarMul[T]$.
    Follows by \auto{exact_left_module_extensionality}.
\end{proof}

\begin{proposition}\label{induced_submodule_inclusion_type}
    Suppose $N$ is a submodule of $M$ over $R$.
    Then
        $\subgroupInclusion{\ModuleAdditiveGroup{M}}{N}
            \in\funs{\carrier[\InducedSubmodule{R}{M}{N}]}{
                \carrier[M]}$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{submodule_additive_subgroup,subgroup_inclusion_type,
            subgroup_structure_carrier,module_additive_group_fields,
            induced_submodule_fields}.
\end{proof}

\subsection{Substructure records}

The submodule predicate speaks only about a carrier.  The following predicate
records when a module structure on that carrier is literally the one inherited
from the ambient module.

\begin{definition}\label{module_substructure}
    $\moduleSubstructure{R}{N}{M}$ iff
        $N$ is a left module over $R$ and
        $\carrier[N]$ is a submodule of $M$ over $R$ and
        $\ModuleAdditiveGroup{N}
            =\subgroupStructure{\ModuleAdditiveGroup{M}}{\carrier[N]}$ and
        $\scalarMul[N]
            =\restrl{\scalarMul[M]}{
                \carrier[R]\times\carrier[N]}$.
\end{definition}

\begin{proposition}\label{module_substructure_is_module}
    Suppose $\moduleSubstructure{R}{N}{M}$.
    Then $N$ is a left module over $R$.
\end{proposition}
\begin{proof}
    Follows by \obv{module_substructure}.
\end{proof}

\begin{proposition}\label{module_substructure_ambient_module}
    Suppose $\moduleSubstructure{R}{N}{M}$.
    Then $M$ is a left module over $R$.
\end{proposition}
\begin{proof}
    $\carrier[N]$ is a submodule of $M$ over $R$ by
        \obv{module_substructure}.
    Follows by \obv{submodule_module}.
\end{proof}

\begin{proposition}\label{module_substructure_carrier_submodule}
    Suppose $\moduleSubstructure{R}{N}{M}$.
    Then $\carrier[N]$ is a submodule of $M$ over $R$.
\end{proposition}
\begin{proof}
    Follows by \obv{module_substructure}.
\end{proof}

\begin{corollary}\label{module_substructure_carrier_subseteq}
    Suppose $\moduleSubstructure{R}{N}{M}$.
    Then $\carrier[N]\subseteq\carrier[M]$.
\end{corollary}
\begin{proof}
    Follows by
        \auto{module_substructure_carrier_submodule,
            submodule_subseteq}.
\end{proof}

\begin{proposition}\label{module_substructure_additive_group}
    Suppose $\moduleSubstructure{R}{N}{M}$.
    Then $\ModuleAdditiveGroup{N}
        =\subgroupStructure{\ModuleAdditiveGroup{M}}{\carrier[N]}$.
\end{proposition}
\begin{proof}
    Follows by \obv{module_substructure}.
\end{proof}

\begin{theorem}\label{module_substructure_is_additive_group_substructure}
    Suppose $\moduleSubstructure{R}{N}{M}$.
    Then $\groupSubstructure{\ModuleAdditiveGroup{N}}{
        \ModuleAdditiveGroup{M}}$.
\end{theorem}
\begin{proof}
    Let $G=\ModuleAdditiveGroup{N}$ and
        $H=\ModuleAdditiveGroup{M}$.
    $N$ is a left module over $R$ by
        \obv{module_substructure_is_module}.
    Hence $G$ is an exact commutative group by
        \auto{module_additive_group_is_exact_commutative_group}.
    Hence $G$ is an exact group by
        \obv{commutative_group_is_group}.
    $\carrier[N]$ is a submodule of $M$ over $R$ by
        \obv{module_substructure_carrier_submodule}.
    Hence $\subgroup{\carrier[N]}{H}$ by
        \obv{submodule_additive_subgroup}.
    Hence $\subgroup{\carrier[G]}{H}$ by
        \auto{module_additive_group_fields}.
    $G=\subgroupStructure{H}{\carrier[N]}$ by
        \auto{module_substructure_additive_group}.
    Hence $G=\subgroupStructure{H}{\carrier[G]}$ by
        \auto{module_additive_group_fields}.
    $\reduct[group]{G}=G$ by \obv{exact_group_reduct_eq}.
    Hence $\reduct[group]{G}
        =\subgroupStructure{H}{\carrier[G]}$.
    Follows by \auto{group_substructure_intro}.
\end{proof}

\begin{proposition}\label{module_substructure_scalar_graph}
    Suppose $\moduleSubstructure{R}{N}{M}$.
    Then $\scalarMul[N]
        =\restrl{\scalarMul[M]}{
            \carrier[R]\times\carrier[N]}$.
\end{proposition}
\begin{proof}
    Follows by \obv{module_substructure}.
\end{proof}

\begin{proposition}\label{module_substructure_scalar_apply}
    Suppose $\moduleSubstructure{R}{N}{M}$.
    Let $a\in\carrier[R]$ and $x\in\carrier[N]$.
    Then $\scalarMul[N](a,x)=\scalarMul[M](a,x)$.
\end{proposition}
\begin{proof}
    $\carrier[N]$ is a submodule of $M$ over $R$ by
        \obv{module_substructure_carrier_submodule}.
    Hence $M$ is a left module over $R$ and
        $\carrier[N]\subseteq\carrier[M]$ by
        \obv{submodule_module,submodule_subseteq}.
    $\scalarMul[M]$ is a function and
        $\dom{\scalarMul[M]}
            =\carrier[R]\times\carrier[M]$ by
        \auto{module_scalar_graph,funs_elim}.
    $\carrier[R]\times\carrier[N]
        \subseteq\dom{\scalarMul[M]}$ by
        \auto{times_subseteq_right}.
    $(a,x)\in\carrier[R]\times\carrier[N]$ by
        \obv{times_tuple_intro}.
    Follows by
        \auto{module_substructure_scalar_graph,
            restrl_of_function_apply}.
\end{proof}

\begin{proposition}\label{module_substructure_intro}
    Suppose $N$ is a left module over $R$ and
        $\carrier[N]$ is a submodule of $M$ over $R$.
    Suppose $\ModuleAdditiveGroup{N}
        =\subgroupStructure{\ModuleAdditiveGroup{M}}{\carrier[N]}$.
    Suppose $\scalarMul[N]
        =\restrl{\scalarMul[M]}{
            \carrier[R]\times\carrier[N]}$.
    Then $\moduleSubstructure{R}{N}{M}$.
\end{proposition}
\begin{proof}
    Follows by \obv{module_substructure}.
\end{proof}

\begin{theorem}\label{induced_submodule_is_module_substructure}
    Suppose $A$ is a submodule of $M$ over $R$.
    Then $\moduleSubstructure{R}{\InducedSubmodule{R}{M}{A}}{M}$.
\end{theorem}
\begin{proof}
    Let $N=\InducedSubmodule{R}{M}{A}$.
    $N$ is an exact left module over $R$ by
        \auto{induced_submodule_is_exact}.
    Hence $N$ is a left module over $R$.
    $\carrier[N]=A$ by \auto{induced_submodule_carrier}.
    Hence $\carrier[N]$ is a submodule of $M$ over $R$.
    $\ModuleAdditiveGroup{N}
        =\subgroupStructure{\ModuleAdditiveGroup{M}}{A}$ by
        \auto{induced_submodule_additive_group}.
    Hence $\ModuleAdditiveGroup{N}
        =\subgroupStructure{\ModuleAdditiveGroup{M}}{\carrier[N]}$.
    $\scalarMul[N]
        =\restrl{\scalarMul[M]}{\carrier[R]\times A}$ by
        \auto{induced_submodule_fields}.
    Hence $\scalarMul[N]
        =\restrl{\scalarMul[M]}{
            \carrier[R]\times\carrier[N]}$.
    Follows by \auto{module_substructure_intro}.
\end{proof}

\begin{theorem}\label{full_induced_submodule_eq}
    Let $M$ be a set.
    Suppose $M$ is an exact left module over $R$.
    Then $\InducedSubmodule{R}{M}{\carrier[M]}=M$.
\end{theorem}
\begin{proof}
    Let $A=\carrier[M]$ and
        $N=\InducedSubmodule{R}{M}{A}$.
    $M$ is a left module over $R$.
    Hence $R$ is a ring by \obv{module_scalar_ring}.
    Hence $A$ is a submodule of $M$ over $R$ by
        \auto{module_carrier_submodule}.
    Hence $N$ is an exact left module over $R$ by
        \auto{induced_submodule_is_exact}.
    Let $G=\ModuleAdditiveGroup{M}$ and
        $K=\subgroupStructure{G}{A}$.
    $G$ is an exact commutative group by
        \auto{module_additive_group_is_exact_commutative_group}.
    Hence $G$ is an exact group by
        \obv{commutative_group_is_group}.
    $A=\carrier[G]$ by \auto{module_additive_group_fields}.
    $K=\reduct[group]{G}$ by
        \auto{full_subgroup_structure_eq_group_reduct}.
    $\reduct[group]{G}=G$ by \obv{exact_group_reduct_eq}.
    Hence $K=G$.
    $\carrier[N]=\carrier[M]$ and
        $\mul[N]=\mul[M]$ and
        $\neutral[N]=\neutral[M]$ and
        $\groupInv[N]=\groupInv[M]$ by
        \auto{induced_submodule_fields,module_additive_group_fields}.
    $\dom{\scalarMul[M]}=\carrier[R]\times A$ by
        \auto{module_scalar_graph,funs}.
    $\scalarMul[M]$ is a relation by
        \auto{module_scalar_graph,funs_is_relation}.
    Hence $\restrl{\scalarMul[M]}{\carrier[R]\times A}
        =\scalarMul[M]$ by \auto{restrl_by_dom}.
    Hence $\scalarMul[N]=\scalarMul[M]$ by
        \auto{induced_submodule_fields}.
    Follows by \auto{exact_left_module_extensionality}.
\end{proof}
