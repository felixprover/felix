\section{Linear maps}

A linear map is an additive-group homomorphism that also preserves scalar
multiplication.  Both module endpoints and the scalar ring remain explicit in
the predicate.

\begin{definition}\label{linear_map}
    $f$ is a linear map from $M$ to $N$ over $R$ iff
        $M$ is a left module over $R$ and
        $N$ is a left module over $R$ and
        $\groupHom{f}{\ModuleAdditiveGroup{M}}{
            \ModuleAdditiveGroup{N}}$ and
        for all $a,x$ such that
            $a\in\carrier[R]$ and $x\in\carrier[M]$ we have
            $f(\scalarMul[M](a,x))=\scalarMul[N](a,f(x))$.
\end{definition}

\begin{theorem}\label{linear_map_expanded_iff}
    $f$ is a linear map from $M$ to $N$ over $R$ iff
        $M$ is a left module over $R$ and
        $N$ is a left module over $R$ and
        $f\in\funs{\carrier[M]}{\carrier[N]}$ and
        (for all $x,y\in\carrier[M]$ we have
            $f(\mul[M](x,y))=\mul[N](f(x),f(y))$) and
        (for all $a,x$ such that
            $a\in\carrier[R]$ and $x\in\carrier[M]$ we have
            $f(\scalarMul[M](a,x))=\scalarMul[N](a,f(x))$).
\end{theorem}
\begin{proof}
    Show that if $f$ is a linear map from $M$ to $N$ over $R$, then
        $M$ is a left module over $R$ and
        $N$ is a left module over $R$ and
        $f\in\funs{\carrier[M]}{\carrier[N]}$ and
        (for all $x,y\in\carrier[M]$ we have
            $f(\mul[M](x,y))=\mul[N](f(x),f(y))$) and
        (for all $a,x$ such that $a\in\carrier[R]$ and
            $x\in\carrier[M]$ we have
            $f(\scalarMul[M](a,x))=\scalarMul[N](a,f(x))$).
    \begin{subproof}
        Assume $f$ is a linear map from $M$ to $N$ over $R$.
        $M$ is a left module over $R$ and
            $N$ is a left module over $R$ and
            $\groupHom{f}{\ModuleAdditiveGroup{M}}{
                \ModuleAdditiveGroup{N}}$ by \obv{linear_map}.
        $f\in\funs{\carrier[M]}{\carrier[N]}$ by
            \auto{group_hom_type,module_additive_group_fields}.
        For all $x,y\in\carrier[M]$ we have
            $f(\mul[M](x,y))=\mul[N](f(x),f(y))$ by
            \auto{group_hom_mul,module_additive_group_fields}.
        Follows by \obv{linear_map}.
    \end{subproof}
    Show that if
        $M$ is a left module over $R$ and
        $N$ is a left module over $R$ and
        $f\in\funs{\carrier[M]}{\carrier[N]}$ and
        (for all $x,y\in\carrier[M]$ we have
            $f(\mul[M](x,y))=\mul[N](f(x),f(y))$) and
        (for all $a,x$ such that $a\in\carrier[R]$ and
            $x\in\carrier[M]$ we have
            $f(\scalarMul[M](a,x))=\scalarMul[N](a,f(x))$),
        then $f$ is a linear map from $M$ to $N$ over $R$.
    \begin{subproof}
        Assume
            $M$ is a left module over $R$ and
            $N$ is a left module over $R$ and
            $f\in\funs{\carrier[M]}{\carrier[N]}$ and
            (for all $x,y\in\carrier[M]$ we have
                $f(\mul[M](x,y))=\mul[N](f(x),f(y))$) and
            (for all $a,x$ such that $a\in\carrier[R]$ and
                $x\in\carrier[M]$ we have
                $f(\scalarMul[M](a,x))=\scalarMul[N](a,f(x))$).
        $\ModuleAdditiveGroup{M}$ is a group and
            $\ModuleAdditiveGroup{N}$ is a group by
            \obv{module_additive_group_is_group}.
        $f\in\funs{\carrier[\ModuleAdditiveGroup{M}]}{
            \carrier[\ModuleAdditiveGroup{N}]}$ by
            \auto{module_additive_group_fields}.
        For all $x,y\in\carrier[\ModuleAdditiveGroup{M}]$ we have
            $f(\mul[\ModuleAdditiveGroup{M}](x,y))
                =\mul[\ModuleAdditiveGroup{N}](f(x),f(y))$ by
            \auto{module_additive_group_fields}.
        Hence $\groupHom{f}{\ModuleAdditiveGroup{M}}{
            \ModuleAdditiveGroup{N}}$ by \obv{group_hom}.
        Follows by \obv{linear_map}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{proposition}\label{linear_map_source_module}
    Suppose $f$ is a linear map from $M$ to $N$ over $R$.
    Then $M$ is a left module over $R$.
\end{proposition}
\begin{proof}
    Follows by \obv{linear_map}.
\end{proof}

\begin{proposition}\label{linear_map_target_module}
    Suppose $f$ is a linear map from $M$ to $N$ over $R$.
    Then $N$ is a left module over $R$.
\end{proposition}
\begin{proof}
    Follows by \obv{linear_map}.
\end{proof}

\begin{proposition}\label{linear_map_scalar_ring}
    Suppose $f$ is a linear map from $M$ to $N$ over $R$.
    Then $R$ is a ring.
\end{proposition}
\begin{proof}
    Follows by
        \obv{linear_map_source_module,module_scalar_ring}.
\end{proof}

\begin{proposition}\label{linear_map_is_additive_group_hom}
    Suppose $f$ is a linear map from $M$ to $N$ over $R$.
    Then $\groupHom{f}{\ModuleAdditiveGroup{M}}{
        \ModuleAdditiveGroup{N}}$.
\end{proposition}
\begin{proof}
    Follows by \obv{linear_map}.
\end{proof}

\begin{proposition}\label{linear_map_type}
    Suppose $f$ is a linear map from $M$ to $N$ over $R$.
    Then $f\in\funs{\carrier[M]}{\carrier[N]}$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{linear_map_is_additive_group_hom,group_hom_type,
            module_additive_group_fields}.
\end{proof}

\begin{corollary}\label{linear_map_is_function}
    Suppose $f$ is a linear map from $M$ to $N$ over $R$.
    Then $f$ is a function.
\end{corollary}
\begin{proof}
    Follows by \auto{linear_map_type,funs_is_function}.
\end{proof}

\begin{corollary}\label{linear_map_apply_type}
    Suppose $f$ is a linear map from $M$ to $N$ over $R$.
    Let $x\in\carrier[M]$.
    Then $f(x)\in\carrier[N]$.
\end{corollary}
\begin{proof}
    Follows by \auto{linear_map_type,funs_type_apply}.
\end{proof}

\begin{proposition}\label{linear_map_addition}
    Suppose $f$ is a linear map from $M$ to $N$ over $R$.
    Let $x,y\in\carrier[M]$.
    Then $f(\mul[M](x,y))=\mul[N](f(x),f(y))$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{linear_map_is_additive_group_hom,group_hom_mul,
            module_additive_group_fields}.
\end{proof}

\begin{proposition}\label{linear_map_scalar}
    Suppose $f$ is a linear map from $M$ to $N$ over $R$.
    Let $a\in\carrier[R]$ and $x\in\carrier[M]$.
    Then $f(\scalarMul[M](a,x))=\scalarMul[N](a,f(x))$.
\end{proposition}
\begin{proof}
    Follows by \obv{linear_map}.
\end{proof}

\begin{theorem}\label{linear_map_zero}
    Suppose $f$ is a linear map from $M$ to $N$ over $R$.
    Then $f(\neutral[M])=\neutral[N]$.
\end{theorem}
\begin{proof}
    Follows by
        \auto{linear_map_is_additive_group_hom,group_hom_neutral,
            module_additive_group_fields}.
\end{proof}

\begin{theorem}\label{linear_map_negation}
    Suppose $f$ is a linear map from $M$ to $N$ over $R$.
    Let $x\in\carrier[M]$.
    Then $f(\groupInv[M](x))=\groupInv[N](f(x))$.
\end{theorem}
\begin{proof}
    Follows by
        \auto{linear_map_is_additive_group_hom,group_hom_inverse,
            module_additive_group_fields}.
\end{proof}

\begin{theorem}\label{linear_map_subtraction}
    Suppose $f$ is a linear map from $M$ to $N$ over $R$.
    Let $x,y\in\carrier[M]$.
    Then
        $f(\mul[M](x,\groupInv[M](y)))
            =\mul[N](f(x),\groupInv[N](f(y)))$.
\end{theorem}
\begin{proof}
    $\groupInv[M](y)\in\carrier[M]$ by
        \obv{linear_map_source_module,module_is_group,
            group_inverse_type}.
    Follows by \auto{linear_map_addition,linear_map_negation}.
\end{proof}

\begin{theorem}\label{linear_map_finite_sum}
    Suppose $f$ is a linear map from $M$ to $N$ over $R$.
    Suppose $A$ is finite and $g\in\funs{A}{\carrier[M]}$.
    Then
        $f(\finiteSum{\ModuleAdditiveGroup{M}}{A}{g})
            =\finiteSum{\ModuleAdditiveGroup{N}}{A}{f\circ g}$.
\end{theorem}
\begin{proof}
    Let $G=\ModuleAdditiveGroup{M}$ and
        $H=\ModuleAdditiveGroup{N}$.
    $G$ is an exact commutative group and
        $H$ is an exact commutative group by
        \auto{linear_map_source_module,linear_map_target_module,
            module_additive_group_is_exact_commutative_group}.
    For all $x,y\in\carrier[G]$ we have
        $\mul[G](x,y)=\mul[G](y,x)$ by
        \obv{commutative_group_commutative}.
    For all $x,y\in\carrier[H]$ we have
        $\mul[H](x,y)=\mul[H](y,x)$ by
        \obv{commutative_group_commutative}.
    $g\in\funs{A}{\carrier[G]}$ by
        \auto{module_additive_group_fields}.
    $f$ is a monoid homomorphism from $G$ to $H$ by
        \obv{linear_map_is_additive_group_hom,
            group_hom_is_monoid_hom}.
    $f(\finiteFold{G}{A}{g})
        =\finiteFold{H}{A}{f\circ g}$ by
        \obv{monoid_hom_preserves_finite_fold}.
    Follows by \auto{finite_sum}.
\end{proof}

\subsection{Identity, composition, and extensionality}

\begin{proposition}\label{identity_linear_map}
    Let $M$ be a left module over $R$.
    Then $\identity{\carrier[M]}$ is a linear map from $M$ to $M$ over $R$.
\end{proposition}
\begin{proof}
    Let $G=\ModuleAdditiveGroup{M}$.
    $G$ is a group by \obv{module_additive_group_is_group}.
    $R$ is a ring by \obv{module_scalar_ring}.
    $\carrier[G]=\carrier[M]$ by
        \auto{module_additive_group_fields}.
    Hence $\groupHom{\identity{\carrier[M]}}{G}{G}$ by
        \auto{identity_group_hom}.
    Show that for all $a,x$ such that $a\in\carrier[R]$ and
        $x\in\carrier[M]$ we have
        $(\identity{\carrier[M]})(\scalarMul[M](a,x))
            =\scalarMul[M](a,(\identity{\carrier[M]})(x))$.
    \begin{subproof}
        Follows by \auto{module_scalar_type,id_apply}.
    \end{subproof}
    Follows by \obv{linear_map}.
\end{proof}

\begin{theorem}\label{linear_map_composition}
    Suppose $f$ is a linear map from $M$ to $N$ over $R$ and
        $g$ is a linear map from $N$ to $P$ over $R$.
    Then $g\circ f$ is a linear map from $M$ to $P$ over $R$.
\end{theorem}
\begin{proof}
    $M$ is a left module over $R$ and
        $P$ is a left module over $R$ by
        \obv{linear_map_source_module,linear_map_target_module}.
    $R$ is a ring by \obv{linear_map_scalar_ring}.
    $\groupHom{f}{\ModuleAdditiveGroup{M}}{
        \ModuleAdditiveGroup{N}}$ and
        $\groupHom{g}{\ModuleAdditiveGroup{N}}{
        \ModuleAdditiveGroup{P}}$ by
        \obv{linear_map_is_additive_group_hom}.
    Hence $\groupHom{g\circ f}{\ModuleAdditiveGroup{M}}{
        \ModuleAdditiveGroup{P}}$ by
        \obv{group_hom_composition}.
    $f\in\funs{\carrier[M]}{\carrier[N]}$ and
        $g\in\funs{\carrier[N]}{\carrier[P]}$ by
        \obv{linear_map_type}.
    Hence $f$ is a function and $g$ is a function and
        $\dom{f}=\carrier[M]$ and $\dom{g}=\carrier[N]$ and
        $g$ is composable with $f$ by
        \auto{funs_is_function,funs_elim,funs_ran,funs}.
    Show that for all $a,x$ such that $a\in\carrier[R]$ and
        $x\in\carrier[M]$ we have
        $(g\circ f)(\scalarMul[M](a,x))
            =\scalarMul[P](a,(g\circ f)(x))$.
    \begin{subproof}
        Fix $a,x$.
        Assume $a\in\carrier[R]$ and $x\in\carrier[M]$.
        $\scalarMul[M](a,x)\in\carrier[M]$ by
            \obv{module_scalar_type}.
        Hence $\scalarMul[M](a,x)\in\dom{f}$ and $x\in\dom{f}$.
        $f(x)\in\carrier[N]$ by \obv{linear_map_apply_type}.
        $(g\circ f)(\scalarMul[M](a,x))
            =g(f(\scalarMul[M](a,x)))$ by \obv{circ_apply}.
        Hence $(g\circ f)(\scalarMul[M](a,x))
            =g(\scalarMul[N](a,f(x)))$ by
            \obv{linear_map_scalar}.
        Hence $(g\circ f)(\scalarMul[M](a,x))
            =\scalarMul[P](a,g(f(x)))$ by
            \obv{linear_map_scalar}.
        $(g\circ f)(x)=g(f(x))$ by \obv{circ_apply}.
    \end{subproof}
    Follows by \obv{linear_map}.
\end{proof}

\begin{theorem}\label{linear_map_extensionality}
    Suppose $f$ is a linear map from $M$ to $N$ over $R$ and
        $g$ is a linear map from $M$ to $N$ over $R$.
    Suppose for every $x\in\carrier[M]$ we have $f(x)=g(x)$.
    Then $f=g$.
\end{theorem}
\begin{proof}
    $\groupHom{f}{\ModuleAdditiveGroup{M}}{
        \ModuleAdditiveGroup{N}}$ and
        $\groupHom{g}{\ModuleAdditiveGroup{M}}{
        \ModuleAdditiveGroup{N}}$ by
        \obv{linear_map_is_additive_group_hom}.
    For every $x\in\carrier[\ModuleAdditiveGroup{M}]$ we have
        $f(x)=g(x)$ by \auto{module_additive_group_fields}.
    Follows by \obv{group_hom_extensionality}.
\end{proof}

\subsection{Changing endpoints}

\begin{theorem}\label{linear_map_domain_restriction}
    Suppose $f$ is a linear map from $M$ to $N$ over $R$.
    Suppose $A$ is a submodule of $M$ over $R$.
    Then $\restrl{f}{A}$ is a linear map from
        $\InducedSubmodule{R}{M}{A}$ to $N$ over $R$.
\end{theorem}
\begin{proof}
    Let $P=\InducedSubmodule{R}{M}{A}$ and
        $r=\restrl{f}{A}$.
    $P$ is an exact left module over $R$ by
        \auto{induced_submodule_is_exact}.
    Hence $P$ is a left module over $R$.
    $N$ is a left module over $R$ by
        \obv{linear_map_target_module}.
    $\groupHom{f}{\ModuleAdditiveGroup{M}}{
        \ModuleAdditiveGroup{N}}$ by
        \obv{linear_map_is_additive_group_hom}.
    Hence $\groupHom{r}{
        \subgroupStructure{\ModuleAdditiveGroup{M}}{A}}{
        \ModuleAdditiveGroup{N}}$ by
        \auto{submodule_additive_subgroup,
            group_hom_domain_restriction}.
    $\ModuleAdditiveGroup{P}
        =\subgroupStructure{\ModuleAdditiveGroup{M}}{A}$ by
        \auto{induced_submodule_additive_group}.
    Hence $\groupHom{r}{\ModuleAdditiveGroup{P}}{
        \ModuleAdditiveGroup{N}}$.
    $f$ is a function and $A\subseteq\dom{f}$ by
        \auto{linear_map_type,funs_is_function,funs,
            submodule_subseteq}.
    Show that for all $a,x$ such that $a\in\carrier[R]$ and
        $x\in\carrier[P]$ we have
        $r(\scalarMul[P](a,x))=\scalarMul[N](a,r(x))$.
    \begin{subproof}
        Fix $a,x$.
        Assume $a\in\carrier[R]$ and $x\in\carrier[P]$.
        Hence $x\in A$ by \auto{induced_submodule_carrier}.
        Hence $x\in\carrier[M]$ by
            \auto{submodule_subseteq,subseteq}.
        $\scalarMul[M](a,x)\in A$ by \obv{submodule_scalar}.
        $\scalarMul[P](a,x)=\scalarMul[M](a,x)$ by
            \auto{induced_submodule_scalar_apply}.
        Hence $\scalarMul[P](a,x)\in A$.
        $r(\scalarMul[P](a,x))
            =f(\scalarMul[P](a,x))$ and $r(x)=f(x)$ by
            \auto{restrl_of_function_apply}.
        $f(\scalarMul[P](a,x))=f(\scalarMul[M](a,x))$.
        $f(\scalarMul[M](a,x))=\scalarMul[N](a,f(x))$ by
            \obv{linear_map_scalar}.
    \end{subproof}
    Follows by \auto{linear_map}.
\end{proof}

\begin{theorem}\label{linear_map_codomain_enlargement}
    Suppose $f$ is a linear map from $M$ to $N$ over $R$.
    Suppose $\moduleSubstructure{R}{N}{P}$.
    Then $f$ is a linear map from $M$ to $P$ over $R$.
\end{theorem}
\begin{proof}
    $M$ is a left module over $R$ by
        \obv{linear_map_source_module}.
    $P$ is a left module over $R$ by
        \obv{module_substructure_ambient_module}.
    $\groupHom{f}{\ModuleAdditiveGroup{M}}{
        \ModuleAdditiveGroup{N}}$ by
        \obv{linear_map_is_additive_group_hom}.
    $\groupSubstructure{\ModuleAdditiveGroup{N}}{
        \ModuleAdditiveGroup{P}}$ by
        \obv{module_substructure_is_additive_group_substructure}.
    Hence $\groupHom{f}{\ModuleAdditiveGroup{M}}{
        \ModuleAdditiveGroup{P}}$ by
        \obv{group_hom_codomain_enlargement}.
    Show that for all $a,x$ such that $a\in\carrier[R]$ and
        $x\in\carrier[M]$ we have
        $f(\scalarMul[M](a,x))=\scalarMul[P](a,f(x))$.
    \begin{subproof}
        Fix $a,x$.
        Assume $a\in\carrier[R]$ and $x\in\carrier[M]$.
        $f(x)\in\carrier[N]$ by \obv{linear_map_apply_type}.
        $f(\scalarMul[M](a,x))=\scalarMul[N](a,f(x))$ by
            \obv{linear_map_scalar}.
        Follows by \obv{module_substructure_scalar_apply}.
    \end{subproof}
    Follows by \obv{linear_map}.
\end{proof}

\begin{theorem}\label{linear_map_codomain_corestriction}
    Suppose $f$ is a linear map from $M$ to $N$ over $R$.
    Suppose $A$ is a submodule of $N$ over $R$.
    Suppose $\img{f}{\carrier[M]}\subseteq A$.
    Then $f$ is a linear map from $M$ to
        $\InducedSubmodule{R}{N}{A}$ over $R$.
\end{theorem}
\begin{proof}
    Let $P=\InducedSubmodule{R}{N}{A}$.
    $M$ is a left module over $R$ by
        \obv{linear_map_source_module}.
    $P$ is an exact left module over $R$ by
        \auto{induced_submodule_is_exact}.
    Hence $P$ is a left module over $R$.
    $\groupHom{f}{\ModuleAdditiveGroup{M}}{
        \ModuleAdditiveGroup{N}}$ by
        \obv{linear_map_is_additive_group_hom}.
    $\img{f}{\carrier[\ModuleAdditiveGroup{M}]}\subseteq A$ by
        \auto{module_additive_group_fields}.
    Hence $\groupHom{f}{\ModuleAdditiveGroup{M}}{
        \subgroupStructure{\ModuleAdditiveGroup{N}}{A}}$ by
        \obv{submodule_additive_subgroup,
            group_hom_codomain_corestriction}.
    $\ModuleAdditiveGroup{P}
        =\subgroupStructure{\ModuleAdditiveGroup{N}}{A}$ by
        \auto{induced_submodule_additive_group}.
    Hence $\groupHom{f}{\ModuleAdditiveGroup{M}}{
        \ModuleAdditiveGroup{P}}$.
    $f$ is a function and $\dom{f}=\carrier[M]$ by
        \auto{linear_map_type,funs_elim}.
    Show that for all $a,x$ such that $a\in\carrier[R]$ and
        $x\in\carrier[M]$ we have
        $f(\scalarMul[M](a,x))=\scalarMul[P](a,f(x))$.
    \begin{subproof}
        Fix $a,x$.
        Assume $a\in\carrier[R]$ and $x\in\carrier[M]$.
        $(x,f(x))\in f$ by \obv{function_apply_elim}.
        Hence $f(x)\in\img{f}{\carrier[M]}$ by \auto{img_iff}.
        Hence $f(x)\in A$ by \obv{subseteq}.
        $f(\scalarMul[M](a,x))=\scalarMul[N](a,f(x))$ by
            \obv{linear_map_scalar}.
        Follows by \auto{induced_submodule_scalar_apply}.
    \end{subproof}
    Follows by \auto{linear_map}.
\end{proof}

\subsection{Embeddings, surjections, and isomorphisms}

\begin{definition}\label{module_embedding}
    $f$ is a module embedding from $M$ to $N$ over $R$ iff
        $f$ is a linear map from $M$ to $N$ over $R$ and
        $f$ is injective.
\end{definition}

\begin{definition}\label{module_surjection}
    $f$ is a module surjection from $M$ to $N$ over $R$ iff
        $f$ is a linear map from $M$ to $N$ over $R$ and
        $f\in\surj{\carrier[M]}{\carrier[N]}$.
\end{definition}

\begin{definition}\label{module_isomorphism}
    $f$ is a module isomorphism from $M$ to $N$ over $R$ iff
        $f$ is a linear map from $M$ to $N$ over $R$ and
        $f\in\bijections{\carrier[M]}{\carrier[N]}$.
\end{definition}

\begin{proposition}\label{module_embedding_is_linear_map}
    Suppose $f$ is a module embedding from $M$ to $N$ over $R$.
    Then $f$ is a linear map from $M$ to $N$ over $R$.
\end{proposition}
\begin{proof}
    Follows by \obv{module_embedding}.
\end{proof}

\begin{proposition}\label{module_embedding_injective}
    Suppose $f$ is a module embedding from $M$ to $N$ over $R$.
    Then $f$ is injective.
\end{proposition}
\begin{proof}
    Follows by \obv{module_embedding}.
\end{proof}

\begin{proposition}\label{module_surjection_is_linear_map}
    Suppose $f$ is a module surjection from $M$ to $N$ over $R$.
    Then $f$ is a linear map from $M$ to $N$ over $R$.
\end{proposition}
\begin{proof}
    Follows by \obv{module_surjection}.
\end{proof}

\begin{proposition}\label{module_surjection_is_surj}
    Suppose $f$ is a module surjection from $M$ to $N$ over $R$.
    Then $f\in\surj{\carrier[M]}{\carrier[N]}$.
\end{proposition}
\begin{proof}
    Follows by \obv{module_surjection}.
\end{proof}

\begin{proposition}\label{module_isomorphism_is_linear_map}
    Suppose $f$ is a module isomorphism from $M$ to $N$ over $R$.
    Then $f$ is a linear map from $M$ to $N$ over $R$.
\end{proposition}
\begin{proof}
    Follows by \obv{module_isomorphism}.
\end{proof}

\begin{proposition}\label{module_isomorphism_is_bijection}
    Suppose $f$ is a module isomorphism from $M$ to $N$ over $R$.
    Then $f\in\bijections{\carrier[M]}{\carrier[N]}$.
\end{proposition}
\begin{proof}
    Follows by \obv{module_isomorphism}.
\end{proof}

\begin{corollary}\label{module_isomorphism_is_embedding}
    Suppose $f$ is a module isomorphism from $M$ to $N$ over $R$.
    Then $f$ is a module embedding from $M$ to $N$ over $R$.
\end{corollary}
\begin{proof}
    $f$ is injective by
        \auto{module_isomorphism_is_bijection,bijections}.
    Follows by
        \obv{module_embedding,module_isomorphism_is_linear_map}.
\end{proof}

\begin{corollary}\label{module_isomorphism_is_surjection}
    Suppose $f$ is a module isomorphism from $M$ to $N$ over $R$.
    Then $f$ is a module surjection from $M$ to $N$ over $R$.
\end{corollary}
\begin{proof}
    $f\in\surj{\carrier[M]}{\carrier[N]}$ by
        \auto{module_isomorphism_is_bijection,bijections}.
    Follows by
        \obv{module_surjection,module_isomorphism_is_linear_map}.
\end{proof}

\begin{theorem}\label{module_embedding_composition}
    Suppose $f$ is a module embedding from $M$ to $N$ over $R$ and
        $g$ is a module embedding from $N$ to $P$ over $R$.
    Then $g\circ f$ is a module embedding from $M$ to $P$ over $R$.
\end{theorem}
\begin{proof}
    $f$ is a linear map from $M$ to $N$ over $R$ and
        $g$ is a linear map from $N$ to $P$ over $R$ by
        \obv{module_embedding_is_linear_map}.
    Hence $g\circ f$ is a linear map from $M$ to $P$ over $R$ by
        \obv{linear_map_composition}.
    $f$ and $g$ are injective by
        \obv{module_embedding_injective}.
    Hence $g\circ f$ is injective by \obv{circ_injective}.
    Follows by \obv{module_embedding}.
\end{proof}

\subsection{Inverse linear maps}

\begin{theorem}\label{bijective_linear_map_converse}
    Suppose $f$ is a linear map from $M$ to $N$ over $R$.
    Suppose $f\in\bijections{\carrier[M]}{\carrier[N]}$.
    Then $\converse{f}$ is a linear map from $N$ to $M$ over $R$.
\end{theorem}
\begin{proof}
    Let $G=\ModuleAdditiveGroup{M}$ and
        $H=\ModuleAdditiveGroup{N}$.
    Let $g=\converse{f}$.
    $M$ is a left module over $R$ and
        $N$ is a left module over $R$ by
        \obv{linear_map_source_module,linear_map_target_module}.
    Hence $R$ is a ring by \obv{module_scalar_ring}.
    $\groupHom{f}{G}{H}$ by
        \obv{linear_map_is_additive_group_hom}.
    $f\in\bijections{\carrier[G]}{\carrier[H]}$ by
        \auto{module_additive_group_fields}.
    Hence $\groupIsomorphism{f}{G}{H}$ by
        \obv{group_isomorphism}.
    Hence $\groupHom{g}{H}{G}$ by
        \auto{group_isomorphism_converse_hom}.
    $g\in\funs{\carrier[N]}{\carrier[M]}$ by
        \auto{bijective_converse_are_funs}.
    Hence $g$ is a function and $\dom{g}=\carrier[N]$ by
        \obv{funs_elim}.
    $f$ is a function and $\dom{f}=\carrier[M]$ and
        $f$ is injective by
        \auto{linear_map_type,funs_elim,bijections}.
    Show that for all $a,y$ such that $a\in\carrier[R]$ and
        $y\in\carrier[N]$ we have
        $g(\scalarMul[N](a,y))=\scalarMul[M](a,g(y))$.
    \begin{subproof}
        Fix $a,y$.
        Assume $a\in\carrier[R]$ and $y\in\carrier[N]$.
        Let $x=g(y)$ and $z=\scalarMul[N](a,y)$.
        Let $u=g(z)$.
        $x\in\carrier[M]$ by \auto{funs_type_apply}.
        $z\in\carrier[N]$ by \auto{module_scalar_type}.
        Hence $u\in\carrier[M]$ by \auto{funs_type_apply}.
        $(y,x)\in g$ and $(z,u)\in g$ by
            \auto{function_apply_elim}.
        Hence $(x,y)\in f$ and $(u,z)\in f$ by
            \obv{converse_iff}.
        Hence $f(x)=y$ and $f(u)=z$ by
            \obv{function_apply_intro}.
        Let $v=\scalarMul[M](a,x)$.
        $v\in\carrier[M]$ by \auto{module_scalar_type}.
        $f(v)=\scalarMul[N](a,f(x))$ by
            \auto{linear_map_scalar}.
        Hence $f(v)=z$.
        $(u,z)\in f$ and $(v,z)\in f$ by
            \auto{function_apply_elim}.
        Hence $u=v$ by \obv{injective}.
    \end{subproof}
    Follows by \auto{linear_map}.
\end{proof}

\begin{theorem}\label{module_isomorphism_converse}
    Suppose $f$ is a module isomorphism from $M$ to $N$ over $R$.
    Then $\converse{f}$ is a module isomorphism from $N$ to $M$ over $R$.
\end{theorem}
\begin{proof}
    $f$ is a linear map from $M$ to $N$ over $R$ and
        $f\in\bijections{\carrier[M]}{\carrier[N]}$ by
        \obv{module_isomorphism_is_linear_map,
            module_isomorphism_is_bijection}.
    Hence $\converse{f}$ is a linear map from $N$ to $M$ over $R$ by
        \obv{bijective_linear_map_converse}.
    $\converse{f}\in\bijections{\carrier[N]}{\carrier[M]}$ by
        \obv{bijection_converse_is_bijection}.
    Follows by \obv{module_isomorphism}.
\end{proof}

\subsection{Linear-map sets}

\begin{definition}\label{linear_map_set}
    Let $R$ be a ring.
    Let $M$ be a left module over $R$.
    Let $N$ be a left module over $R$.
    $\linearMapSet{R}{M}{N}=
        \{f\in\funs{\carrier[M]}{\carrier[N]}\mid
            \text{$f$ is a linear map from $M$ to $N$ over $R$}\}$.
\end{definition}

\begin{proposition}\label{linear_map_set_iff}
    Let $R$ be a ring.
    Let $M$ be a left module over $R$.
    Let $N$ be a left module over $R$.
    Then $f\in\linearMapSet{R}{M}{N}$ iff
        $f$ is a linear map from $M$ to $N$ over $R$.
\end{proposition}
\begin{proof}
    Follows by \auto{linear_map_set,linear_map_type}.
\end{proof}

\begin{definition}\label{module_endomorphism_set}
    Let $R$ be a ring.
    Let $M$ be a left module over $R$.
    $\moduleEndomorphismSet{R}{M}=\linearMapSet{R}{M}{M}$.
\end{definition}

\begin{proposition}\label{module_endomorphism_set_iff}
    Let $R$ be a ring.
    Let $M$ be a left module over $R$.
    Then $f\in\moduleEndomorphismSet{R}{M}$ iff
        $f$ is a linear map from $M$ to $M$ over $R$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{module_endomorphism_set,linear_map_set_iff}.
\end{proof}
