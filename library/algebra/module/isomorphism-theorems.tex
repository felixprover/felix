\section{Isomorphism theorems for modules}

\subsection{The first isomorphism theorem}

For a linear map, the additive first-isomorphism graph is already the right
set-coded map.  The source and target below are the exact quotient and induced
modules on its additive source and image, so only scalar compatibility remains
to be proved.

\begin{theorem}\label{module_first_isomorphism_map_linear}
    Suppose $f$ is a linear map from $M$ to $P$ over $R$.
    Let $K=\moduleKernel{f}{M}$ and
        $J=\moduleImage{f}{M}$.
    Let $G=\ModuleAdditiveGroup{M}$ and
        $H=\ModuleAdditiveGroup{P}$.
    Then $\firstIsomorphismMap{f}{G}{H}$ is a linear map from
        $\QuotientModule{R}{M}{K}$ to
        $\InducedSubmodule{R}{P}{J}$ over $R$.
\end{theorem}
\begin{proof}
    Let $A=\QuotientModule{R}{M}{K}$ and
        $B=\InducedSubmodule{R}{P}{J}$.
    Let $g=\firstIsomorphismMap{f}{G}{H}$ and
        $E=\subgroupCosetRelation{G}{K}$.
    $R$ is a ring by \obv{linear_map_scalar_ring}.
    $M$ is a left module over $R$ and
        $P$ is a left module over $R$ by
        \obv{linear_map_source_module,linear_map_target_module}.
    $K$ is a submodule of $M$ over $R$ and
        $J$ is a submodule of $P$ over $R$ by
        \auto{module_kernel_is_submodule,module_image_is_submodule}.
    Hence $A$ is a left module over $R$ and
        $B$ is a left module over $R$ by
        \auto{quotient_module_is_exact,induced_submodule_is_exact}.
    $\groupHom{f}{G}{H}$ by
        \obv{linear_map_is_additive_group_hom}.
    $K=\groupKernel{f}{G}$ by \auto{module_kernel}.
    $J=\img{f}{\carrier[G]}$ by
        \auto{module_image,module_additive_group_fields}.
    $\ModuleAdditiveGroup{A}=\quotientGroup{G}{K}$ by
        \auto{quotient_module_additive_group_exact}.
    $\ModuleAdditiveGroup{B}=\subgroupStructure{H}{J}$ by
        \auto{induced_submodule_additive_group}.
    $\groupHom{g}{\quotientGroup{G}{K}}{
        \subgroupStructure{H}{J}}$ by
        \auto{first_isomorphism_map_hom}.
    Hence $\groupHom{g}{\ModuleAdditiveGroup{A}}{
        \ModuleAdditiveGroup{B}}$.
    Show that for all $a,U$ such that $a\in\carrier[R]$ and
        $U\in\carrier[A]$ we have
        $g(\scalarMul[A](a,U))=\scalarMul[B](a,g(U))$.
    \begin{subproof}
        Fix $a,U$.
        Assume $a\in\carrier[R]$ and $U\in\carrier[A]$.
        $\carrier[A]=\groupQuotientCarrier{G}{K}$ by
            \auto{quotient_module_carrier}.
        Hence $U\in\groupQuotientCarrier{G}{K}$.
        Take $x\in\carrier[G]$ such that
            $U=\equivalenceClass{E}{x}$ by
            \auto{group_quotient_carrier,quotient}.
        Hence $x\in\carrier[M]$ by
            \auto{module_additive_group_fields}.
        $\scalarMul[M](a,x)\in\carrier[M]$ by
            \obv{module_scalar_type}.
        Hence $\scalarMul[M](a,x)\in\carrier[G]$ by
            \auto{module_additive_group_fields}.
        $\scalarMul[A](a,U)=
            \equivalenceClass{E}{\scalarMul[M](a,x)}$ by
            \auto{quotient_module_scalar_apply}.
        Hence $g(\scalarMul[A](a,U))
            =f(\scalarMul[M](a,x))$ by
            \auto{first_isomorphism_map_apply,module_kernel}.
        $g(U)=f(x)$ by
            \auto{first_isomorphism_map_apply,module_kernel}.
        $f(x)\in J$ by \auto{module_image_iff}.
        Hence $\scalarMul[B](a,g(U))=\scalarMul[P](a,f(x))$ by
            \auto{induced_submodule_scalar_apply}.
        $f(\scalarMul[M](a,x))=\scalarMul[P](a,f(x))$ by
            \obv{linear_map_scalar}.
    \end{subproof}
    Follows by \auto{linear_map}.
\end{proof}


\begin{theorem}\label{module_first_isomorphism_theorem}
    Suppose $f$ is a linear map from $M$ to $P$ over $R$.
    Let $K=\moduleKernel{f}{M}$ and
        $J=\moduleImage{f}{M}$.
    Let $G=\ModuleAdditiveGroup{M}$ and
        $H=\ModuleAdditiveGroup{P}$.
    Then $\firstIsomorphismMap{f}{G}{H}$ is a module isomorphism from
        $\QuotientModule{R}{M}{K}$ to
        $\InducedSubmodule{R}{P}{J}$ over $R$.
\end{theorem}
\begin{proof}
    Let $A=\QuotientModule{R}{M}{K}$ and
        $B=\InducedSubmodule{R}{P}{J}$.
    Let $g=\firstIsomorphismMap{f}{G}{H}$.
    $R$ is a ring by \obv{linear_map_scalar_ring}.
    $M$ is a left module over $R$ and
        $P$ is a left module over $R$ by
        \obv{linear_map_source_module,linear_map_target_module}.
    $g$ is a linear map from $A$ to $B$ over $R$ by
        \auto{module_first_isomorphism_map_linear}.
    $K$ is a submodule of $M$ over $R$ and
        $J$ is a submodule of $P$ over $R$ by
        \auto{module_kernel_is_submodule,module_image_is_submodule}.
    $\groupHom{f}{G}{H}$ by
        \obv{linear_map_is_additive_group_hom}.
    $K=\groupKernel{f}{G}$ by \auto{module_kernel}.
    $J=\img{f}{\carrier[G]}$ by
        \auto{module_image,module_additive_group_fields}.
    $g\in\bijections{\groupQuotientCarrier{G}{K}}{J}$ by
        \auto{first_isomorphism_map_bijection}.
    $\carrier[A]=\groupQuotientCarrier{G}{K}$ by
        \auto{quotient_module_carrier}.
    $\carrier[B]=J$ by \auto{induced_submodule_carrier}.
    Hence $g\in\bijections{\carrier[A]}{\carrier[B]}$.
    Follows by \obv{module_isomorphism}.
\end{proof}

\begin{proposition}\label{module_first_isomorphism_map_apply}
    Suppose $f$ is a linear map from $M$ to $P$ over $R$.
    Let $K=\moduleKernel{f}{M}$.
    Let $G=\ModuleAdditiveGroup{M}$ and
        $H=\ModuleAdditiveGroup{P}$.
    Let $x\in\carrier[M]$.
    Then
        $\apply{\firstIsomorphismMap{f}{G}{H}}{
            \equivalenceClass{\subgroupCosetRelation{G}{K}}{x}}
        =f(x)$.
\end{proposition}
\begin{proof}
    $\groupHom{f}{G}{H}$ by
        \obv{linear_map_is_additive_group_hom}.
    $K=\groupKernel{f}{G}$ by \auto{module_kernel}.
    $x\in\carrier[G]$ by \auto{module_additive_group_fields}.
    Follows by \auto{first_isomorphism_map_apply}.
\end{proof}

\begin{theorem}\label{module_first_isomorphism_factorization}
    Suppose $f$ is a linear map from $M$ to $P$ over $R$.
    Let $G=\ModuleAdditiveGroup{M}$ and
        $H=\ModuleAdditiveGroup{P}$.
    Let $K=\moduleKernel{f}{M}$.
    Then
        $\firstIsomorphismMap{f}{G}{H}\circ
            \groupQuotientProjection{G}{K}=f$.
\end{theorem}
\begin{proof}
    $\groupHom{f}{G}{H}$ by
        \obv{linear_map_is_additive_group_hom}.
    $K=\groupKernel{f}{G}$ by \auto{module_kernel}.
    Follows by \auto{first_isomorphism_map_factorization}.
\end{proof}

\subsection{The second isomorphism theorem}

\begin{theorem}\label{module_second_isomorphism_intersection_submodule}
    Let $R$ be a ring.
    Let $M$ be a left module over $R$.
    Suppose $N$ is a submodule of $M$ over $R$ and
        $P$ is a submodule of $M$ over $R$.
    Then $N\inter P$ is a submodule of
        $\InducedSubmodule{R}{M}{N}$ over $R$.
\end{theorem}
\begin{proof}
    $N\inter P$ is a submodule of $M$ over $R$ by
        \obv{submodule_intersection}.
    $N\inter P\subseteq N$ by \obv{inter_lower_left}.
    Follows by \obv{submodule_of_induced_submodule}.
\end{proof}

\begin{theorem}\label{module_second_isomorphism_summand_submodule}
    Let $R$ be a ring.
    Let $M$ be a left module over $R$.
    Suppose $N$ is a submodule of $M$ over $R$ and
        $P$ is a submodule of $M$ over $R$.
    Let $S=\submoduleSum{R}{M}{N}{P}$.
    Then $P$ is a submodule of $\InducedSubmodule{R}{M}{S}$ over $R$.
\end{theorem}
\begin{proof}
    $S$ is a submodule of $M$ over $R$ by
        \auto{submodule_sum_is_submodule}.
    $P\subseteq S$ by \obv{submodule_sum_right}.
    Follows by \obv{submodule_of_induced_submodule}.
\end{proof}

\begin{theorem}\label{module_second_isomorphism_restriction_linear}
    Let $R$ be a ring.
    Let $M$ be a left module over $R$.
    Suppose $N$ is a submodule of $M$ over $R$ and
        $P$ is a submodule of $M$ over $R$.
    Let $S=\submoduleSum{R}{M}{N}{P}$.
    Let $A=\InducedSubmodule{R}{M}{N}$ and
        $B=\InducedSubmodule{R}{M}{S}$.
    Let $q=\groupQuotientProjection{\ModuleAdditiveGroup{B}}{P}$ and
        $r=\restrl{q}{N}$.
    Then $r$ is a linear map from $A$ to
        $\QuotientModule{R}{B}{P}$ over $R$.
\end{theorem}
\begin{proof}
    Let $C=\QuotientModule{R}{B}{P}$.
    $S$ is a submodule of $M$ over $R$ by
        \auto{submodule_sum_is_submodule}.
    $N\subseteq S$ by \obv{submodule_sum_left}.
    $P\subseteq S$ by \obv{submodule_sum_right}.
    Hence $N$ is a submodule of $B$ over $R$ by
        \auto{submodule_of_induced_submodule}.
    $P$ is a submodule of $B$ over $R$ by
        \auto{submodule_of_induced_submodule}.
    $B$ is an exact left module over $R$ by
        \auto{induced_submodule_is_exact}.
    $q$ is a linear map from $B$ to $C$ over $R$ by
        \auto{quotient_module_projection_linear}.
    Hence $r$ is a linear map from
        $\InducedSubmodule{R}{B}{N}$ to $C$ over $R$ by
        \auto{linear_map_domain_restriction}.
    $\InducedSubmodule{R}{B}{N}=A$ by
        \auto{iterated_induced_submodule}.
\end{proof}

\begin{theorem}\label{module_second_isomorphism_restriction_kernel}
    Let $R$ be a ring.
    Let $M$ be a left module over $R$.
    Suppose $N$ is a submodule of $M$ over $R$ and
        $P$ is a submodule of $M$ over $R$.
    Let $K=N\inter P$ and $S=\submoduleSum{R}{M}{N}{P}$.
    Let $A=\InducedSubmodule{R}{M}{N}$ and
        $B=\InducedSubmodule{R}{M}{S}$.
    Let $C=\QuotientModule{R}{B}{P}$.
    Let $q=\groupQuotientProjection{\ModuleAdditiveGroup{B}}{P}$ and
        $r=\restrl{q}{N}$.
    Then $\moduleKernel{r}{A}=K$.
\end{theorem}
\begin{proof}
    $S$ is a submodule of $M$ over $R$ by
        \auto{submodule_sum_is_submodule}.
    $N\subseteq S$ and $P\subseteq S$ by
        \obv{submodule_sum_left,submodule_sum_right}.
    Hence $N$ is a submodule of $B$ over $R$ by
        \auto{submodule_of_induced_submodule}.
    $P$ is a submodule of $B$ over $R$ by
        \auto{submodule_of_induced_submodule}.
    $B$ is an exact left module over $R$ by
        \auto{induced_submodule_is_exact}.
    $r$ is a linear map from $A$ to $C$ over $R$ by
        \auto{module_second_isomorphism_restriction_linear}.
    $q$ is a linear map from $B$ to $C$ over $R$ by
        \auto{quotient_module_projection_linear}.
    $\moduleKernel{q}{B}=P$ by
        \auto{quotient_module_projection_kernel}.
    $q$ is a function and $N\subseteq\dom{q}$ by
        \auto{linear_map_type,funs_is_function,funs,
            submodule_subseteq}.
    Show that every element of $\moduleKernel{r}{A}$
        is an element of $K$.
    \begin{subproof}
        Fix $x\in\moduleKernel{r}{A}$.
        Hence $x\in\carrier[A]$ and $r(x)=\neutral[C]$ by
            \auto{module_kernel_iff}.
        Hence $x\in N$ by \auto{induced_submodule_carrier}.
        Hence $r(x)=q(x)$ by \auto{restrl_of_function_apply}.
        $x\in\carrier[B]$ by
            \auto{submodule_subseteq,subseteq}.
        Hence $x\in\moduleKernel{q}{B}$ by
            \auto{module_kernel_iff}.
        Hence $x\in P$.
        Follows by \auto{inter_intro}.
    \end{subproof}
    Show that every element of $K$ is an element of
        $\moduleKernel{r}{A}$.
    \begin{subproof}
        Fix $x\in K$.
        Hence $x\in N$ and $x\in P$ by
            \auto{inter_elim_left,inter_elim_right}.
        Hence $x\in\carrier[A]$ by
            \auto{induced_submodule_carrier}.
        $x\in\carrier[B]$ by
            \auto{submodule_subseteq,subseteq}.
        Hence $x\in\moduleKernel{q}{B}$.
        Hence $q(x)=\neutral[C]$ by \auto{module_kernel_iff}.
        $r(x)=q(x)$ by \auto{restrl_of_function_apply}.
        Follows by \auto{module_kernel_iff}.
    \end{subproof}
    Follows by \obv{setext}.
\end{proof}

\begin{theorem}\label{module_second_isomorphism_restriction_surjective}
    Let $R$ be a ring.
    Let $M$ be a left module over $R$.
    Suppose $N$ is a submodule of $M$ over $R$ and
        $P$ is a submodule of $M$ over $R$.
    Let $S=\submoduleSum{R}{M}{N}{P}$.
    Let $A=\InducedSubmodule{R}{M}{N}$ and
        $B=\InducedSubmodule{R}{M}{S}$.
    Let $C=\QuotientModule{R}{B}{P}$.
    Let $q=\groupQuotientProjection{\ModuleAdditiveGroup{B}}{P}$ and
        $r=\restrl{q}{N}$.
    Then $r\in\surj{\carrier[A]}{\carrier[C]}$.
\end{theorem}
\begin{proof}
    Let $F=\ModuleAdditiveGroup{M}$ and
        $G=\ModuleAdditiveGroup{B}$.
    Let $Q=\quotientGroup{G}{P}$.
    $S$ is a submodule of $M$ over $R$ by
        \auto{submodule_sum_is_submodule}.
    $N\subseteq S$ and $P\subseteq S$ by
        \obv{submodule_sum_left,submodule_sum_right}.
    Hence $N$ is a submodule of $B$ over $R$ by
        \auto{submodule_of_induced_submodule}.
    $P$ is a submodule of $B$ over $R$ by
        \auto{submodule_of_induced_submodule}.
    $B$ is an exact left module over $R$ by
        \auto{induced_submodule_is_exact}.
    $G$ is a commutative group by
        \obv{induced_submodule_is_exact,
            module_additive_group_is_exact_commutative_group}.
    $\subgroup{P}{G}$ by \obv{submodule_additive_subgroup}.
    Hence $\normalSubgroup{P}{G}$ by
        \obv{commutative_group_subgroup_is_normal}.
    $Q$ is a group by \obv{quotient_group_is_group}.
    Hence $Q$ is a monoid by \obv{group_is_monoid}.
    $\groupHom{q}{G}{Q}$ by
        \auto{group_quotient_projection_hom}.
    $\groupKernel{q}{G}=P$ by
        \auto{group_quotient_projection_kernel}.
    $q$ is a function and $\dom{q}=\carrier[G]$ by
        \auto{group_hom_type,funs_is_function,funs}.
    $N\subseteq\carrier[B]$ by \obv{submodule_subseteq}.
    $\carrier[B]=\carrier[G]$ by
        \auto{module_additive_group_fields}.
    Hence $N\subseteq\dom{q}$.
    $\carrier[A]=N$ by \auto{induced_submodule_carrier}.
    $\carrier[C]=\groupQuotientCarrier{G}{P}$ by
        \auto{quotient_module_carrier}.
    $r\in\funs{N}{\carrier[C]}$ by
        \auto{group_quotient_projection_type,funs_restrl}.
    Show that for every $U\in\carrier[C]$ there exists $n\in N$
        such that $r(n)=U$.
    \begin{subproof}
        Fix $U\in\carrier[C]$.
        Take $s\in\carrier[G]$ such that $q(s)=U$ by
            \auto{group_quotient_projection_surjective,surj}.
        $\carrier[G]=\carrier[B]=S$ by
            \auto{module_additive_group_fields,
                induced_submodule_carrier}.
        Hence $s\in S$.
        Take $n\in N$ such that there exists $p\in P$ such that
            $s=\mul[M](n,p)$ by \auto{submodule_sum_iff}.
        Take $p\in P$ such that $s=\mul[M](n,p)$.
        $n,p\in S$ by \obv{subseteq}.
        Hence $n,p\in\carrier[B]$ by
            \obv{induced_submodule_carrier}.
        Hence $n,p\in\carrier[G]$ by
            \obv{module_additive_group_fields}.
        $\mul[G](n,p)=\mul[B](n,p)$ by
            \auto{module_additive_group_fields}.
        $\mul[B]=\mul[\subgroupStructure{F}{S}]$ by
            \auto{induced_submodule_fields}.
        $\subgroup{S}{F}$ by \obv{submodule_additive_subgroup}.
        Hence $\mul[\subgroupStructure{F}{S}](n,p)=\mul[F](n,p)$ by
            \obv{subgroup_structure_mul_apply}.
        $\mul[F](n,p)=\mul[M](n,p)$ by
            \auto{module_additive_group_fields}.
        Hence $\mul[B](n,p)=\mul[M](n,p)$.
        Hence $s=\mul[G](n,p)$.
        $q(s)=\mul[Q](q(n),q(p))$ by \auto{group_hom_mul}.
        $p\in\groupKernel{q}{G}$.
        Hence $q(p)=\neutral[Q]$ by \auto{group_kernel_iff}.
        $q(n)\in\carrier[Q]$ by \obv{group_hom_apply_type}.
        $\mul[Q](q(n),\neutral[Q])=q(n)$ by
            \obv{monoid_is_unital_magma,unitalmagma_right}.
        Hence $q(s)=q(n)$.
        $r(n)=q(n)$ by \auto{restrl_of_function_apply}.
        Follows by \auto.
    \end{subproof}
    Hence $r\in\surj{N}{\carrier[C]}$ by \obv{surj}.
\end{proof}

\begin{theorem}\label{module_second_isomorphism_theorem}
    Let $R$ be a ring.
    Let $M$ be a left module over $R$.
    Suppose $N$ is a submodule of $M$ over $R$ and
        $P$ is a submodule of $M$ over $R$.
    Let $K=N\inter P$ and $S=\submoduleSum{R}{M}{N}{P}$.
    Let $A=\InducedSubmodule{R}{M}{N}$ and
        $B=\InducedSubmodule{R}{M}{S}$.
    Let $C=\QuotientModule{R}{B}{P}$.
    Let $q=\groupQuotientProjection{\ModuleAdditiveGroup{B}}{P}$ and
        $r=\restrl{q}{N}$.
    Let $G=\ModuleAdditiveGroup{A}$ and
        $H=\ModuleAdditiveGroup{C}$.
    Then $\firstIsomorphismMap{r}{G}{H}$ is a module isomorphism from
        $\QuotientModule{R}{A}{K}$ to $C$ over $R$.
\end{theorem}
\begin{proof}
    Let $f=\firstIsomorphismMap{r}{G}{H}$.
    $S$ is a submodule of $M$ over $R$ by
        \auto{submodule_sum_is_submodule}.
    Hence $B$ is an exact left module over $R$ by
        \auto{induced_submodule_is_exact}.
    $K$ is a submodule of $A$ over $R$ by
        \auto{module_second_isomorphism_intersection_submodule}.
    $P$ is a submodule of $B$ over $R$ by
        \auto{module_second_isomorphism_summand_submodule}.
    Hence $C$ is an exact left module over $R$ by
        \auto{quotient_module_is_exact}.
    $r$ is a linear map from $A$ to $C$ over $R$ by
        \auto{module_second_isomorphism_restriction_linear}.
    $\moduleKernel{r}{A}=K$ by
        \obv{module_second_isomorphism_restriction_kernel}.
    $r\in\surj{\carrier[A]}{\carrier[C]}$ by
        \obv{module_second_isomorphism_restriction_surjective}.
    Hence $\moduleImage{r}{A}=\carrier[C]$ by
        \obv{linear_map_surjective_iff_image_full}.
    $f$ is a module isomorphism from
        $\QuotientModule{R}{A}{\moduleKernel{r}{A}}$ to
        $\InducedSubmodule{R}{C}{\moduleImage{r}{A}}$
        over $R$ by \auto{module_first_isomorphism_theorem}.
    $\InducedSubmodule{R}{C}{\carrier[C]}=C$ by
        \obv{full_induced_submodule_eq}.
\end{proof}

\begin{proposition}\label{module_second_isomorphism_map_apply}
    Let $R$ be a ring.
    Let $M$ be a left module over $R$.
    Suppose $N$ is a submodule of $M$ over $R$ and
        $P$ is a submodule of $M$ over $R$.
    Let $K=N\inter P$ and $S=\submoduleSum{R}{M}{N}{P}$.
    Let $A=\InducedSubmodule{R}{M}{N}$ and
        $B=\InducedSubmodule{R}{M}{S}$.
    Let $C=\QuotientModule{R}{B}{P}$.
    Let $q=\groupQuotientProjection{\ModuleAdditiveGroup{B}}{P}$ and
        $r=\restrl{q}{N}$.
    Let $G=\ModuleAdditiveGroup{A}$ and
        $H=\ModuleAdditiveGroup{C}$.
    Let $x\in N$.
    Then
        $\apply{\firstIsomorphismMap{r}{G}{H}}{
            \equivalenceClass{\subgroupCosetRelation{G}{K}}{x}}
        =q(x)$.
\end{proposition}
\begin{proof}
    $S$ is a submodule of $M$ over $R$ by
        \auto{submodule_sum_is_submodule}.
    $N\subseteq S$ by \obv{submodule_sum_left}.
    Hence $N$ is a submodule of $B$ over $R$ by
        \auto{submodule_of_induced_submodule}.
    $P$ is a submodule of $B$ over $R$ by
        \auto{module_second_isomorphism_summand_submodule}.
    $B$ is an exact left module over $R$ by
        \auto{induced_submodule_is_exact}.
    $r$ is a linear map from $A$ to $C$ over $R$ by
        \auto{module_second_isomorphism_restriction_linear}.
    $\moduleKernel{r}{A}=K$ by
        \obv{module_second_isomorphism_restriction_kernel}.
    $x\in\carrier[A]$ by \auto{induced_submodule_carrier}.
    Hence
        $\apply{\firstIsomorphismMap{r}{G}{H}}{
            \equivalenceClass{\subgroupCosetRelation{G}{K}}{x}}
        =r(x)$ by \obv{module_first_isomorphism_map_apply}.
    $q$ is a linear map from $B$ to $C$ over $R$ by
        \auto{quotient_module_projection_linear}.
    Hence $q$ is a function and $\dom{q}=\carrier[B]$ by
        \auto{linear_map_type,funs_is_function,funs}.
    $N\subseteq\carrier[B]$ by \obv{submodule_subseteq}.
    Hence $N\subseteq\dom{q}$.
    $r(x)=q(x)$ by \auto{restrl_of_function_apply}.
\end{proof}

\begin{theorem}\label{module_second_isomorphism_factorization}
    Let $R$ be a ring.
    Let $M$ be a left module over $R$.
    Suppose $N$ is a submodule of $M$ over $R$ and
        $P$ is a submodule of $M$ over $R$.
    Let $K=N\inter P$ and $S=\submoduleSum{R}{M}{N}{P}$.
    Let $A=\InducedSubmodule{R}{M}{N}$ and
        $B=\InducedSubmodule{R}{M}{S}$.
    Let $C=\QuotientModule{R}{B}{P}$.
    Let $q=\groupQuotientProjection{\ModuleAdditiveGroup{B}}{P}$ and
        $r=\restrl{q}{N}$.
    Let $G=\ModuleAdditiveGroup{A}$ and
        $H=\ModuleAdditiveGroup{C}$.
    Then
        $\firstIsomorphismMap{r}{G}{H}\circ
            \groupQuotientProjection{G}{K}=r$.
\end{theorem}
\begin{proof}
    $r$ is a linear map from $A$ to $C$ over $R$ by
        \auto{module_second_isomorphism_restriction_linear}.
    $\moduleKernel{r}{A}=K$ by
        \obv{module_second_isomorphism_restriction_kernel}.
    Follows by \obv{module_first_isomorphism_factorization}.
\end{proof}

\subsection{The third isomorphism theorem}

\begin{theorem}\label{module_third_isomorphism_image_submodule}
    Let $R$ be a ring.
    Let $M$ be a left module over $R$.
    Suppose $N$ is a submodule of $M$ over $R$ and
        $P$ is a submodule of $M$ over $R$.
    Suppose $N\subseteq P$.
    Let $G=\ModuleAdditiveGroup{M}$ and

\begin{theorem}\label{module_quotient_comparison_linear}
    Let $R$ be a ring.
    Let $M$ be a left module over $R$.
    Suppose $N$ is a submodule of $M$ over $R$ and
        $P$ is a submodule of $M$ over $R$.
    Suppose $N\subseteq P$.
    Let $G=\ModuleAdditiveGroup{M}$.
    Let $q_P=\groupQuotientProjection{G}{P}$ and
        $E_N=\subgroupCosetRelation{G}{N}$.
    Let $d=\quotientDescent{q_P}{\carrier[M]}{E_N}$.
    Then $d$ is a linear map from $\QuotientModule{R}{M}{N}$ to
        $\QuotientModule{R}{M}{P}$ over $R$.
\end{theorem}
\begin{proof}
    Let $A=\QuotientModule{R}{M}{N}$ and
        $B=\QuotientModule{R}{M}{P}$.
    Let $Q_N=\quotientGroup{G}{N}$ and
        $Q_P=\quotientGroup{G}{P}$.
    $A$ is a left module over $R$ by
        \auto{quotient_module_is_exact}.
    $B$ is a left module over $R$ by
        \auto{quotient_module_is_exact}.
    $G$ is a commutative group by
        \obv{module_additive_group_is_exact_commutative_group}.
    $\subgroup{N}{G}$ and $\subgroup{P}{G}$ by
        \auto{submodule_additive_subgroup}.
    $\normalSubgroup{N}{G}$ and $\normalSubgroup{P}{G}$ by
        \obv{commutative_group_subgroup_is_normal}.
    $Q_N$ is a group by \obv{quotient_group_is_group}.
    $Q_P$ is a group by \auto{quotient_group_is_group}.
    $d\in\funs{\carrier[A]}{\carrier[B]}$ by
        \auto{module_quotient_comparison_type}.
    $\ModuleAdditiveGroup{A}=Q_N$ and
        $\ModuleAdditiveGroup{B}=Q_P$ by
        \auto{quotient_module_additive_group_exact}.
    Hence $d\in\funs{\carrier[Q_N]}{\carrier[Q_P]}$ by
        \auto{module_additive_group_fields}.
    Show that for all $U,V\in\carrier[Q_N]$ we have
        $d(\mul[Q_N](U,V))=\mul[Q_P](d(U),d(V))$.
    \begin{subproof}
        Fix $U,V\in\carrier[Q_N]$.
        Hence $U,V\in\carrier[A]$ by
            \auto{module_additive_group_fields}.
        Take $x\in\carrier[G]$ such that
            $U=\equivalenceClass{E_N}{x}$ by
            \auto{quotient_group_carrier,group_quotient_carrier,quotient}.
        Take $y\in\carrier[G]$ such that
            $V=\equivalenceClass{E_N}{y}$ by
            \auto{quotient_group_carrier,group_quotient_carrier,quotient}.
        Hence $x,y\in\carrier[M]$ by
            \auto{module_additive_group_fields}.
        $\mul[G](x,y)\in\carrier[G]$ by
            \obv{commutative_group_is_group,group_is_monoid,
                monoid_is_magma,magma_welldef}.
        Hence $\mul[G](x,y)\in\carrier[M]$ by
            \auto{module_additive_group_fields}.
        $\mul[Q_N](U,V)=
            \equivalenceClass{E_N}{\mul[G](x,y)}$ by
            \auto{quotient_group_mul_apply}.
        Hence $d(\mul[Q_N](U,V))=q_P(\mul[G](x,y))$ by
            \auto{module_quotient_comparison_apply}.
        $\groupHom{q_P}{G}{Q_P}$ by
            \auto{group_quotient_projection_hom}.
        Hence $q_P(\mul[G](x,y))=\mul[Q_P](q_P(x),q_P(y))$ by
            \obv{group_hom_mul}.
        $d(U)=q_P(x)$ and $d(V)=q_P(y)$ by
            \auto{module_quotient_comparison_apply}.
    \end{subproof}
    Hence $\groupHom{d}{Q_N}{Q_P}$ by \obv{group_hom}.
    Hence $\groupHom{d}{\ModuleAdditiveGroup{A}}{
        \ModuleAdditiveGroup{B}}$ by \auto.
    Show that for all $a,U$ such that $a\in\carrier[R]$ and
        $U\in\carrier[A]$ we have
        $d(\scalarMul[A](a,U))=\scalarMul[B](a,d(U))$.
    \begin{subproof}
        Fix $a,U$.
        Assume $a\in\carrier[R]$ and $U\in\carrier[A]$.
        $\carrier[A]=\groupQuotientCarrier{G}{N}$ by
            \auto{quotient_module_carrier}.
        Take $x\in\carrier[G]$ such that
            $U=\equivalenceClass{E_N}{x}$ by
            \auto{group_quotient_carrier,quotient}.
        Hence $x\in\carrier[M]$ by
            \auto{module_additive_group_fields}.
        $\scalarMul[M](a,x)\in\carrier[M]$ by
            \obv{module_scalar_type}.
        $\scalarMul[A](a,U)=
            \equivalenceClass{E_N}{\scalarMul[M](a,x)}$ by
            \auto{quotient_module_scalar_apply}.
        Hence $d(\scalarMul[A](a,U))
            =q_P(\scalarMul[M](a,x))$ by
            \auto{module_quotient_comparison_apply}.
        $d(U)=q_P(x)$ by
            \auto{module_quotient_comparison_apply}.
        $q_P$ is a linear map from $M$ to $B$ over $R$ by
            \auto{quotient_module_projection_linear}.
        Hence $q_P(\scalarMul[M](a,x))
            =\scalarMul[B](a,q_P(x))$ by
            \obv{linear_map_scalar}.
    \end{subproof}
    Follows by \auto{linear_map}.
\end{proof}

\begin{proof}

        \auto{module_quotient_comparison_linear}.

        \auto{quotient_module_projection_linear}.

        \auto{quotient_module_projection_kernel}.

        \auto{quotient_module_projection_linear,
            linear_map_is_function}.

        is an element of $S$.

            \auto{quotient_module_carrier}.

            $U=\equivalenceClass{E_N}{x}$ by
            \auto{group_quotient_carrier,quotient}.

            \auto{module_additive_group_fields}.
        $d(U)=q_P(x)$ by
            \auto{module_quotient_comparison_apply}.

            \auto{module_kernel_iff}.

                commutative_group_subgroup_is_normal,

            \auto{quotient_module_projection_linear,linear_map_type,funs}.
        Hence $(x,U)\in q_N$ by \auto{function_apply_elim}.
        Follows by \auto{img_iff}.
    \end{subproof}
    Show that every element of $S$ is an element of
        $\moduleKernel{d}{A}$.
    \begin{subproof}
        Fix $U\in S$.
        Take $x\in P$ such that $(x,U)\in q_N$ by \auto{img_iff}.
        $q_N(x)=U$ by \obv{function_apply_intro}.
        $x\in\carrier[M]$ by \auto{submodule_subseteq,subseteq}.
        Hence $x\in\carrier[G]$ by
            \auto{module_additive_group_fields}.
        $q_N(x)=\equivalenceClass{E_N}{x}$ by
            \auto{submodule_additive_subgroup,
                module_additive_group_is_exact_commutative_group,
                commutative_group_subgroup_is_normal,
                group_quotient_projection_apply}.
        Hence $U=\equivalenceClass{E_N}{x}$.
        Hence $U\in\carrier[A]$ by
            \auto{quotient_module_carrier,group_quotient_carrier,
                quotient}.
        $x\in\moduleKernel{q_P}{M}$.
        Hence $q_P(x)=\neutral[B]$ by \auto{module_kernel_iff}.
        $d(U)=q_P(x)$ by
            \auto{module_quotient_comparison_apply}.
        Follows by \auto{module_kernel_iff}.
    \end{subproof}
    Follows by \auto{setext}.
\end{proof}

        \auto{module_quotient_comparison_linear}.
    $\moduleKernel{d}{A}=S$ by
        \obv{module_quotient_comparison_kernel}.
    Hence
        $\firstIsomorphismMap{d}{H}{K}\circ
            \groupQuotientProjection{H}{S}=d$ by
        \obv{module_first_isomorphism_factorization}.
    $d\circ q_N=q_P$ by
        \obv{module_quotient_comparison_factorization}.
\end{proof}

\subsection{The submodule correspondence theorem}

\begin{theorem}\label{module_quotient_correspondence_preimage_submodule}
    Let $R$ be a ring.
    Let $M$ be a left module over $R$.
    Suppose $N$ is a submodule of $M$ over $R$.
    Let $G=\ModuleAdditiveGroup{M}$ and
        $q=\groupQuotientProjection{G}{N}$.
    Let $A=\QuotientModule{R}{M}{N}$.
    Suppose $Q$ is a submodule of $A$ over $R$.
    Then $\preimg{q}{Q}$ is a submodule of $M$ over $R$.
\end{theorem}
\begin{proof}
    $q$ is a linear map from $M$ to $A$ over $R$ by
        \auto{quotient_module_projection_linear}.
    Follows by \obv{linear_map_submodule_preimage}.
\end{proof}

\begin{theorem}\label{module_quotient_correspondence_preimage_contains_kernel}
    Let $R$ be a ring.
    Let $M$ be a left module over $R$.
    Suppose $N$ is a submodule of $M$ over $R$.
    Let $G=\ModuleAdditiveGroup{M}$ and
        $q=\groupQuotientProjection{G}{N}$.
    Let $A=\QuotientModule{R}{M}{N}$.
    Suppose $Q$ is a submodule of $A$ over $R$.
    Then $N\subseteq\preimg{q}{Q}$.
\end{theorem}
\begin{proof}
    $q$ is a linear map from $M$ to $A$ over $R$ by
        \auto{quotient_module_projection_linear}.
    $\moduleKernel{q}{M}=N$ by
        \auto{quotient_module_projection_kernel}.
    Show that for every $x\in N$ we have $x\in\preimg{q}{Q}$.
    \begin{subproof}
        Fix $x\in N$.
        Hence $x\in\carrier[M]$ and $q(x)=\neutral[A]$ by
            \auto{module_kernel_iff}.
        $\neutral[A]\in Q$ by \obv{submodule_zero}.
        $\groupHom{q}{G}{\ModuleAdditiveGroup{A}}$ by
            \auto{linear_map_is_additive_group_hom}.
        Hence $x\in\carrier[G]$ by
            \auto{module_additive_group_fields}.
        Follows by
            \auto{group_hom_preimage_iff,module_additive_group_fields}.
    \end{subproof}
    Follows by \obv{subseteq}.
\end{proof}

\begin{theorem}\label{module_quotient_correspondence_preimage_image}
    Let $R$ be a ring.
    Let $M$ be a left module over $R$.
    Suppose $N$ is a submodule of $M$ over $R$ and
        $P$ is a submodule of $M$ over $R$.
    Suppose $N\subseteq P$.
    Let $G=\ModuleAdditiveGroup{M}$ and
        $q=\groupQuotientProjection{G}{N}$.
    Then $\preimg{q}{\img{q}{P}}=P$.
\end{theorem}
\begin{proof}
    Let $A=\QuotientModule{R}{M}{N}$ and
        $H=\ModuleAdditiveGroup{A}$.
    $q$ is a linear map from $M$ to $A$ over $R$ by
        \auto{quotient_module_projection_linear}.
    Hence $\groupHom{q}{G}{H}$ by
        \auto{linear_map_is_additive_group_hom}.
    $\subgroup{P}{G}$ by \obv{submodule_additive_subgroup}.
    $\groupKernel{q}{G}=\moduleKernel{q}{M}$ by
        \obv{module_kernel}.
    Hence $\groupKernel{q}{G}=N$ by
        \auto{quotient_module_projection_kernel}.
    Hence $\groupKernel{q}{G}\subseteq P$.
    Follows by \obv{group_hom_preimage_image_eq}.
\end{proof}

\begin{theorem}\label{module_quotient_correspondence_image_preimage}
    Let $R$ be a ring.
    Let $M$ be a left module over $R$.
    Suppose $N$ is a submodule of $M$ over $R$.
    Let $G=\ModuleAdditiveGroup{M}$ and
        $q=\groupQuotientProjection{G}{N}$.
    Let $A=\QuotientModule{R}{M}{N}$.
    Suppose $Q$ is a submodule of $A$ over $R$.
    Then $\img{q}{\preimg{q}{Q}}=Q$.
\end{theorem}
\begin{proof}
    Let $H=\ModuleAdditiveGroup{A}$.
    $q$ is a linear map from $M$ to $A$ over $R$ by
        \auto{quotient_module_projection_linear}.
    Hence $\groupHom{q}{G}{H}$ by
        \auto{linear_map_is_additive_group_hom}.
    $q\in\surj{\carrier[M]}{\carrier[A]}$ by
        \auto{quotient_module_projection_surjective}.
    Hence $q\in\surj{\carrier[G]}{\carrier[H]}$ by
        \auto{module_additive_group_fields}.
    $Q\subseteq\carrier[A]$ by \obv{submodule_subseteq}.
    Hence $Q\subseteq\carrier[H]$ by
        \auto{module_additive_group_fields}.
    Follows by \obv{surjective_group_hom_image_preimage_eq}.
\end{proof}

\begin{theorem}\label{module_quotient_correspondence_image_inclusion}
    Let $R$ be a ring.
    Let $M$ be a left module over $R$.
    Suppose $N$ is a submodule of $M$ over $R$.
    Suppose $P$ is a submodule of $M$ over $R$ and
        $S$ is a submodule of $M$ over $R$.
    Suppose $N\subseteq P$ and $N\subseteq S$.
    Let $q=\groupQuotientProjection{\ModuleAdditiveGroup{M}}{N}$.
    Then $P\subseteq S$ iff $\img{q}{P}\subseteq\img{q}{S}$.
\end{theorem}
\begin{proof}
    Show that if $P\subseteq S$, then
        $\img{q}{P}\subseteq\img{q}{S}$.
    \begin{subproof}
        Assume $P\subseteq S$.
        Follows by \obv{img_subseteq}.
    \end{subproof}
    Show that if $\img{q}{P}\subseteq\img{q}{S}$, then
        $P\subseteq S$.
    \begin{subproof}
        Assume $\img{q}{P}\subseteq\img{q}{S}$.
        Hence
            $\preimg{q}{\img{q}{P}}
                \subseteq\preimg{q}{\img{q}{S}}$ by
            \obv{preimg_subseteq}.
        $\preimg{q}{\img{q}{P}}=P$ and
            $\preimg{q}{\img{q}{S}}=S$ by
            \auto{module_quotient_correspondence_preimage_image}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{theorem}\label{module_quotient_correspondence_image_transport}
    Let $R$ be a ring.
    Let $M$ be a left module over $R$.
    Suppose $N$ is a submodule of $M$ over $R$ and
        $P$ is a submodule of $M$ over $R$.
    Suppose $N\subseteq P$.
    Let $G=\ModuleAdditiveGroup{M}$ and
        $q=\groupQuotientProjection{G}{N}$.
    Let $A=\QuotientModule{R}{M}{N}$.
    Let $B=\InducedSubmodule{R}{M}{P}$ and
        $C=\InducedSubmodule{R}{A}{\img{q}{P}}$.
    Then $\restrl{q}{P}$ is a module surjection from $B$ to $C$ over $R$.
\end{theorem}
\begin{proof}
    Let $r=\restrl{q}{P}$.
    $q$ is a linear map from $M$ to $A$ over $R$ by
        \auto{quotient_module_projection_linear}.
    Hence $r$ is a linear map from $B$ to $A$ over $R$ by
        \auto{linear_map_domain_restriction}.
    $\img{q}{P}$ is a submodule of $A$ over $R$ by
        \auto{module_third_isomorphism_image_submodule}.
    $\carrier[B]=P$ by \auto{induced_submodule_carrier}.
    $P\subseteq\dom{q}$ by
        \auto{submodule_subseteq,linear_map_type,funs}.
    $\img{r}{\carrier[B]}=\img{q}{P\inter P}$ by
        \auto{restrl_img}.
    Hence $\img{r}{\carrier[B]}=\img{q}{P}$ by
        \auto{inter_idempotent}.
    Hence $r$ is a linear map from $B$ to $C$ over $R$ by
        \auto{linear_map_codomain_corestriction,subseteq_refl}.
    $\moduleImage{r}{B}=\img{q}{P}$ by
        \obv{module_image}.
    $\carrier[C]=\img{q}{P}$ by
        \auto{induced_submodule_carrier}.
    Hence $r\in\surj{\carrier[B]}{\carrier[C]}$ by
        \auto{linear_map_surjective_iff_image_full}.
    Follows by \auto{module_surjection}.
\end{proof}

\begin{theorem}\label{module_quotient_correspondence_preimage_transport}
    Let $R$ be a ring.
    Let $M$ be a left module over $R$.
    Suppose $N$ is a submodule of $M$ over $R$.
    Let $G=\ModuleAdditiveGroup{M}$ and
        $q=\groupQuotientProjection{G}{N}$.
    Let $A=\QuotientModule{R}{M}{N}$.
    Suppose $Q$ is a submodule of $A$ over $R$.
    Let $P=\preimg{q}{Q}$.
    Then $\restrl{q}{P}$ is a module surjection from
        $\InducedSubmodule{R}{M}{P}$ to
        $\InducedSubmodule{R}{A}{Q}$ over $R$.
\end{theorem}
\begin{proof}
    $P$ is a submodule of $M$ over $R$ by
        \auto{module_quotient_correspondence_preimage_submodule}.
    $N\subseteq P$ by
        \obv{module_quotient_correspondence_preimage_contains_kernel}.
    $\restrl{q}{P}$ is a module surjection from
        $\InducedSubmodule{R}{M}{P}$ to
        $\InducedSubmodule{R}{A}{\img{q}{P}}$ over $R$ by
        \obv{module_quotient_correspondence_image_transport}.
    $\img{q}{P}=Q$ by
        \auto{module_quotient_correspondence_image_preimage}.
\end{proof}

\begin{theorem}\label{module_quotient_correspondence_restriction_kernel}
    Let $R$ be a ring.
    Let $M$ be a left module over $R$.
    Suppose $N$ is a submodule of $M$ over $R$ and
        $P$ is a submodule of $M$ over $R$.
    Suppose $N\subseteq P$.
    Let $G=\ModuleAdditiveGroup{M}$ and
        $q=\groupQuotientProjection{G}{N}$.
    Let $A=\QuotientModule{R}{M}{N}$.
    Let $B=\InducedSubmodule{R}{M}{P}$ and
        $C=\InducedSubmodule{R}{A}{\img{q}{P}}$.
    Let $r=\restrl{q}{P}$.
    Then $\moduleKernel{r}{B}=N$.
\end{theorem}
\begin{proof}
    $r$ is a linear map from $B$ to $C$ over $R$ by
        \auto{module_quotient_correspondence_image_transport,
            module_surjection_is_linear_map}.
    $q$ is a linear map from $M$ to $A$ over $R$ by
        \auto{quotient_module_projection_linear}.
    $\moduleKernel{q}{M}=N$ by
        \auto{quotient_module_projection_kernel}.
    $q$ is a function and $P\subseteq\dom{q}$ by
        \auto{linear_map_type,funs_is_function,funs,
            submodule_subseteq}.
    $\img{q}{P}$ is a submodule of $A$ over $R$ by
        \auto{module_third_isomorphism_image_submodule}.
    $\neutral[C]=\neutral[A]$ by
        \auto{induced_submodule_additive_group,
            module_additive_group_fields,subgroup_structure_neutral}.
    Show that every element of $\moduleKernel{r}{B}$
        is an element of $N$.
    \begin{subproof}
        Fix $x\in\moduleKernel{r}{B}$.
        Hence $x\in\carrier[B]$ and $r(x)=\neutral[C]$ by
            \auto{module_kernel_iff}.
        Hence $x\in P$ by \auto{induced_submodule_carrier}.
        Hence $x\in\carrier[M]$ by
            \auto{submodule_subseteq,subseteq}.
        $r(x)=q(x)$ by \auto{restrl_of_function_apply}.
        Hence $q(x)=\neutral[A]$.
        Hence $x\in\moduleKernel{q}{M}$ by
            \auto{module_kernel_iff}.
    \end{subproof}
    Show that every element of $N$ is an element of
        $\moduleKernel{r}{B}$.
    \begin{subproof}
        Fix $x\in N$.
        Hence $x\in P$ by \obv{subseteq}.
        Hence $x\in\carrier[B]$ by
            \auto{induced_submodule_carrier}.
        $x\in\carrier[M]$ and $q(x)=\neutral[A]$ by
            \auto{module_kernel_iff}.
        $r(x)=q(x)$ by \auto{restrl_of_function_apply}.
        Hence $r(x)=\neutral[C]$.
        Follows by \auto{module_kernel_iff}.
    \end{subproof}
    Follows by \obv{setext}.
\end{proof}

\begin{theorem}\label{module_quotient_correspondence_induced_isomorphism}
    Let $R$ be a ring.
    Let $M$ be a left module over $R$.
    Suppose $N$ is a submodule of $M$ over $R$ and
        $P$ is a submodule of $M$ over $R$.
    Suppose $N\subseteq P$.
    Let $G=\ModuleAdditiveGroup{M}$ and
        $q=\groupQuotientProjection{G}{N}$.
    Let $A=\QuotientModule{R}{M}{N}$.
    Let $B=\InducedSubmodule{R}{M}{P}$ and
        $C=\InducedSubmodule{R}{A}{\img{q}{P}}$.
    Let $r=\restrl{q}{P}$.
    Let $H=\ModuleAdditiveGroup{B}$ and
        $K=\ModuleAdditiveGroup{C}$.
    Then $\firstIsomorphismMap{r}{H}{K}$ is a module isomorphism from
        $\QuotientModule{R}{B}{N}$ to $C$ over $R$.
\end{theorem}
\begin{proof}
    $r$ is a module surjection from $B$ to $C$ over $R$ by
        \auto{module_quotient_correspondence_image_transport}.
    Hence $r$ is a linear map from $B$ to $C$ over $R$ by
        \obv{module_surjection_is_linear_map}.
    $\moduleKernel{r}{B}=N$ by
        \obv{module_quotient_correspondence_restriction_kernel}.
    $r\in\surj{\carrier[B]}{\carrier[C]}$ by
        \obv{module_surjection_is_surj}.
    Hence $\moduleImage{r}{B}=\carrier[C]$ by
        \obv{linear_map_surjective_iff_image_full}.
    $\firstIsomorphismMap{r}{H}{K}$ is a module isomorphism from
        $\QuotientModule{R}{B}{\moduleKernel{r}{B}}$ to
        $\InducedSubmodule{R}{C}{\moduleImage{r}{B}}$
        over $R$ by \obv{module_first_isomorphism_theorem}.
    $C$ is an exact left module over $R$ by
        \auto{module_third_isomorphism_image_submodule,
            induced_submodule_is_exact}.
    Hence $\InducedSubmodule{R}{C}{\carrier[C]}=C$ by
        \obv{full_induced_submodule_eq}.
\end{proof}

\begin{theorem}\label{module_submodule_correspondence_bijection}
    Let $R$ be a ring.
    Let $M$ be a left module over $R$.
    Suppose $N$ is a submodule of $M$ over $R$.
    Let $G=\ModuleAdditiveGroup{M}$ and
        $q=\groupQuotientProjection{G}{N}$.
    Let $A=\QuotientModule{R}{M}{N}$.
    Let
        $C=\{P\in\pow{\carrier[M]}\mid
            \text{$P$ is a submodule of $M$ over $R$ and
                $N\subseteq P$}\}$.
    Let
        $D=\{Q\in\pow{\carrier[A]}\mid
            \text{$Q$ is a submodule of $A$ over $R$}\}$.
    Let
        $F=\Apply{\graphOnDomainOp}
            {C,\fun{P:\Set}{\img{q}{P}}}$.
    Then $F$ is a bijection from $C$ to $D$.
\end{theorem}
\begin{proof}
    Show that for every $P\in C$ we have $\img{q}{P}\in D$.
    \begin{subproof}
        Fix $P\in C$.
        Hence $P$ is a submodule of $M$ over $R$ and
            $N\subseteq P$ by \auto.
        Hence $\img{q}{P}$ is a submodule of $A$ over $R$ by
            \auto{module_third_isomorphism_image_submodule}.
        Hence $\img{q}{P}\subseteq\carrier[A]$ by
            \obv{submodule_subseteq}.
        Hence $\img{q}{P}\in\pow{\carrier[A]}$ by
            \obv{powerset_intro}.
        Follows by \auto.
    \end{subproof}
    Hence $F\in\funs{C}{D}$ by
        \auto{logical_graph_on_domain_type}.
    Hence $F$ is a function and $\dom{F}=C$ by
        \auto{funs_elim}.
    Show that $F\in\surj{C}{D}$.
    \begin{subproof}
        Show that for every $Q\in D$ there exists $P\in C$ such that
            $F(P)=Q$.
        \begin{subproof}
            Fix $Q\in D$.
            Hence $Q$ is a submodule of $A$ over $R$ by \auto.
            Let $P=\preimg{q}{Q}$.
            $P$ is a submodule of $M$ over $R$ by
                \auto{module_quotient_correspondence_preimage_submodule}.
            Hence $P\subseteq\carrier[M]$ by
                \obv{submodule_subseteq}.
            Hence $P\in\pow{\carrier[M]}$ by \obv{powerset_intro}.
            $N\subseteq P$ by
                \obv{module_quotient_correspondence_preimage_contains_kernel}.
            Hence $P\in C$ by \auto.
            $F(P)=\img{q}{P}$ by
                \auto{logical_graph_on_domain_apply}.
            $\img{q}{P}=Q$ by
                \auto{module_quotient_correspondence_image_preimage}.
            Follows by \auto.
        \end{subproof}
        Follows by \obv{surj}.
    \end{subproof}
    Show that $F$ is injective.
    \begin{subproof}
        Show that for all $P,S\in\dom{F}$ such that $F(P)=F(S)$
            we have $P=S$.
        \begin{subproof}
            Fix $P,S\in\dom{F}$.
            Assume $F(P)=F(S)$.
            Hence $P,S\in C$.
            Hence $P$ is a submodule of $M$ over $R$ and
                $N\subseteq P$ by \auto.
            Hence $S$ is a submodule of $M$ over $R$ and
                $N\subseteq S$ by \auto.
            $F(P)=\img{q}{P}$ and $F(S)=\img{q}{S}$ by
                \auto{logical_graph_on_domain_apply}.
            Hence $\img{q}{P}=\img{q}{S}$.
            $\preimg{q}{\img{q}{P}}=P$ and
                $\preimg{q}{\img{q}{S}}=S$ by
                \obv{module_quotient_correspondence_preimage_image}.
        \end{subproof}
        Follows by \auto{injective_function}.
    \end{subproof}
    Follows by \obv{bijections}.
\end{proof}

\begin{theorem}\label{module_submodule_correspondence_inverse_graph}
    Let $R$ be a ring.
    Let $M$ be a left module over $R$.
    Suppose $N$ is a submodule of $M$ over $R$.
    Let $G=\ModuleAdditiveGroup{M}$ and
        $q=\groupQuotientProjection{G}{N}$.
    Let $A=\QuotientModule{R}{M}{N}$.
    Let
        $C=\{P\in\pow{\carrier[M]}\mid
            \text{$P$ is a submodule of $M$ over $R$ and
                $N\subseteq P$}\}$.
    Let
        $D=\{Q\in\pow{\carrier[A]}\mid
            \text{$Q$ is a submodule of $A$ over $R$}\}$.
    Let
        $F=\Apply{\graphOnDomainOp}
            {C,\fun{P:\Set}{\img{q}{P}}}$.
    Let
        $J=\Apply{\graphOnDomainOp}
            {D,\fun{Q:\Set}{\preimg{q}{Q}}}$.
    Then $\converse{F}=J$ and $J$ is a bijection from $D$ to $C$.
\end{theorem}
\begin{proof}
    $F$ is a bijection from $C$ to $D$ by
        \obv{module_submodule_correspondence_bijection}.
    Hence $\converse{F}\in\funs{D}{C}$ by
        \obv{bijective_converse_are_funs}.
    Show that for every $Q\in D$ we have $\preimg{q}{Q}\in C$.
    \begin{subproof}
        Fix $Q\in D$.
        Hence $Q$ is a submodule of $A$ over $R$ by \auto.
        $\preimg{q}{Q}$ is a submodule of $M$ over $R$ by
            \obv{module_quotient_correspondence_preimage_submodule}.
        Hence $\preimg{q}{Q}\subseteq\carrier[M]$ by
            \obv{submodule_subseteq}.
        Hence $\preimg{q}{Q}\in\pow{\carrier[M]}$ by
            \obv{powerset_intro}.
        $N\subseteq\preimg{q}{Q}$ by
            \obv{module_quotient_correspondence_preimage_contains_kernel}.
        Follows by \auto.
    \end{subproof}
    Hence $J\in\funs{D}{C}$ by
        \auto{logical_graph_on_domain_type}.
    Hence $\converse{F}$ and $J$ are functions and
        $\dom{\converse{F}}=D=\dom{J}$ by
        \auto{funs_elim}.
    Show that for every $Q\in\dom{\converse{F}}$ we have
        $(\converse{F})(Q)=J(Q)$.
    \begin{subproof}
        Fix $Q\in\dom{\converse{F}}$.
        Hence $Q\in D$.
        Hence $Q$ is a submodule of $A$ over $R$ by \auto.
        Let $P=\preimg{q}{Q}$.
        Hence $P\in C$ by \auto.
        $F(P)=\img{q}{P}$ by
            \auto{logical_graph_on_domain_apply}.
        $\img{q}{P}=Q$ by
            \auto{module_quotient_correspondence_image_preimage}.
        Hence $F(P)=Q$.
        Hence $(P,Q)\in F$ by
            \auto{function_apply_elim,bijection_is_function,
                bijections_dom}.
        Hence $(Q,P)\in\converse{F}$ by \auto{converse_iff}.
        Hence $(\converse{F})(Q)=P$ by
            \obv{function_apply_intro}.
        $J(Q)=P$ by \auto{logical_graph_on_domain_apply}.
    \end{subproof}
    Hence $\converse{F}=J$ by \obv{funext_on_domain}.
    $\converse{F}$ is a bijection from $D$ to $C$ by
        \obv{bijection_converse_is_bijection}.
\end{proof}
