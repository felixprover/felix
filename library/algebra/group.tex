\import{algebra/monoid.tex}

\section{Groups}

\begin{struct}\label{group}
    A group $G$ is a monoid equipped with
    \begin{enumerate}
        \item\label{group_inverse_graph}
            an inverse map
            $\groupInv[G]:\carrier[G]\to\carrier[G]$;
    \end{enumerate}
    such that
    \begin{enumerate}
        \item\label{group_inverse_right}
            for all $x\in\carrier[G]$ we have
                $\mul[G](x,\groupInv[G](x))=\neutral[G]$.
        \item\label{group_inverse_left}
            for all $x\in\carrier[G]$ we have
                $\mul[G](\groupInv[G](x),x)=\neutral[G]$.
    \end{enumerate}
\end{struct}

\begin{theorem}\label{exact_group_extensionality}
    Let $A,B$ be sets.
    Suppose $A$ is an exact group and $B$ is an exact group.
    Suppose $\carrier[A]=\carrier[B]$.
    Suppose $\mul[A]=\mul[B]$.
    Suppose $\neutral[A]=\neutral[B]$.
    Suppose $\groupInv[A]=\groupInv[B]$.
    Then $A=B$.
\end{theorem}
\begin{proof}
    Follows by structure extensionality.
\end{proof}

\begin{theorem}\label{exact_group_reduct_eq}
    Let $G$ be a set.
    Suppose $G$ is an exact group.
    Then $\reduct[group]{G}=G$.
\end{theorem}
\begin{proof}
    Follows by \obv{groupreductidentity}.
\end{proof}

\begin{theorem}\label{group_reduct_is_exact_group}
    Let $G$ be a group.
    Then $\reduct[group]{G}$ is an exact group.
\end{theorem}
\begin{proof}
    $G$ is a monoid by \obv{group}.
    Hence $G$ is a unital magma and $G$ is a magma by
        \obv{monoid_is_unital_magma,monoid_is_magma}.
    $\carrier[\reduct[group]{G}]=\carrier[G]$ by
        \obv{groupreductprojection_carrier}.
    $\mul[\reduct[group]{G}]=\mul[G]$ by
        \obv{groupreductprojection_mul}.
    $\neutral[\reduct[group]{G}]=\neutral[G]$ by
        \obv{groupreductprojection_neutral}.
    $\groupInv[\reduct[group]{G}]=\groupInv[G]$ by
        \obv{groupreductprojection_groupInv}.
    $\mul[\reduct[group]{G}]\in
        \funs{\carrier[\reduct[group]{G}]\times
            \carrier[\reduct[group]{G}]}{
            \carrier[\reduct[group]{G}]}$ by
        \obv{magma_mul_graph}.
    $\neutral[\reduct[group]{G}]\in
        \carrier[\reduct[group]{G}]$ by
        \obv{unitalmagma_type}.
    For every $x\in\carrier[\reduct[group]{G}]$ we have
        $\mul[\reduct[group]{G}](x,
            \neutral[\reduct[group]{G}])=x$ by
        \auto{unitalmagma_right}.
    For every $x\in\carrier[\reduct[group]{G}]$ we have
        $\mul[\reduct[group]{G}](
            \neutral[\reduct[group]{G}],x)=x$ by
        \auto{unitalmagma_left}.
    For all $x,y,z\in\carrier[\reduct[group]{G}]$ we have
        $\mul[\reduct[group]{G}](x,
            \mul[\reduct[group]{G}](y,z))=
            \mul[\reduct[group]{G}](
                \mul[\reduct[group]{G}](x,y),z)$ by
        \auto{monoid_assoc}.
    $\groupInv[\reduct[group]{G}]\in
        \funs{\carrier[\reduct[group]{G}]}{
            \carrier[\reduct[group]{G}]}$ by
        \obv{group_inverse_graph}.
    For every $x\in\carrier[\reduct[group]{G}]$ we have
        $\mul[\reduct[group]{G}](x,
            \groupInv[\reduct[group]{G}](x))=
            \neutral[\reduct[group]{G}]$ by
        \auto{group_inverse_right}.
    For every $x\in\carrier[\reduct[group]{G}]$ we have
        $\mul[\reduct[group]{G}](
            \groupInv[\reduct[group]{G}](x),x)=
            \neutral[\reduct[group]{G}]$ by
        \auto{group_inverse_left}.
    Hence $\reduct[group]{G}$ is a group by
        \auto{group}.
    Follows by \obv{groupshape,groupreductexact}.
\end{proof}

\begin{theorem}\label{exact_group_eq_exact_group_reduct}
    Let $A,G$ be sets.
    Suppose $A$ is an exact group.
    Suppose $\reduct[group]{G}$ is an exact group.
    Suppose $\carrier[A]=\carrier[\reduct[group]{G}]$.
    Suppose $\mul[A]=\mul[\reduct[group]{G}]$.
    Suppose $\neutral[A]=\neutral[\reduct[group]{G}]$.
    Suppose $\groupInv[A]=\groupInv[\reduct[group]{G}]$.
    Then $A=\reduct[group]{G}$.
\end{theorem}
\begin{proof}
    Follows by structure extensionality.
\end{proof}

\begin{theorem}\label{exact_group_eq_group_reduct}
    Let $A$ be a set.
    Let $G$ be a group.
    Suppose $A$ is an exact group.
    Suppose $\carrier[A]=\carrier[G]$.
    Suppose $\mul[A]=\mul[G]$.
    Suppose $\neutral[A]=\neutral[G]$.
    Suppose $\groupInv[A]=\groupInv[G]$.
    Then $A=\reduct[group]{G}$.
\end{theorem}
\begin{proof}
    $\reduct[group]{G}$ is an exact group by
        \obv{group_reduct_is_exact_group}.
    $\carrier[A]=\carrier[\reduct[group]{G}]$ by
        \obv{groupreductprojection_carrier}.
    $\mul[A]=\mul[\reduct[group]{G}]$ by
        \obv{groupreductprojection_mul}.
    $\neutral[A]=\neutral[\reduct[group]{G}]$ by
        \obv{groupreductprojection_neutral}.
    $\groupInv[A]=\groupInv[\reduct[group]{G}]$ by
        \obv{groupreductprojection_groupInv}.
    Follows by structure extensionality.
\end{proof}

\begin{proposition}\label{group_inverse_type}
    Let $G$ be a group.
    Let $x\in\carrier[G]$.
    Then $\groupInv[G](x)\in\carrier[G]$.
\end{proposition}
\begin{proof}
    Follows by \obv{group_inverse_graph,funs_type_apply}.
\end{proof}

\begin{struct}\label{commutative_group}
    A commutative group $G$ is a group such that
    \begin{enumerate}
        \item\label{commutative_group_commutative}
            for all $x,y\in\carrier[G]$ we have
                $\mul[G](x,y)=\mul[G](y,x)$.
    \end{enumerate}
\end{struct}

\begin{corollary}\label{group_is_monoid}
    Let $G$ be a group.
    Then $G$ is a monoid.
\end{corollary}
\begin{proof}
    Follows by \obv{group}.
\end{proof}

\begin{corollary}\label{commutative_group_is_group}
    Let $G$ be a commutative group.
    Then $G$ is a group.
\end{corollary}
\begin{proof}
    Follows by \obv{commutative_group}.
\end{proof}

\begin{theorem}\label{group_cancel_left}
    Let $G$ be a group.
    Let $a,b,c\in\carrier[G]$.
    Suppose $\mul[G](a,b)=\mul[G](a,c)$.
    Then $b=c$.
\end{theorem}
\begin{proof}
    $\groupInv[G](a)\in\carrier[G]$
        by \obv{group_inverse_type}.
    $\mul[G](\groupInv[G](a),\mul[G](a,b))
        =\mul[G](\groupInv[G](a),\mul[G](a,c))$.
    Hence
    $\mul[G](\mul[G](\groupInv[G](a),a),b)
        =\mul[G](\mul[G](\groupInv[G](a),a),c)$
        by \obv{group_is_monoid,monoid_is_magma,monoid_assoc,magma_welldef}.
    Follows by
        \auto{group_is_monoid,monoid_is_unital_magma,group_inverse_left,unitalmagma_left}.
\end{proof}

\begin{theorem}\label{group_cancel_right}
    Let $G$ be a group.
    Let $a,b,c\in\carrier[G]$.
    Suppose $\mul[G](a,c)=\mul[G](b,c)$.
    Then $a=b$.
\end{theorem}
\begin{proof}
    $\groupInv[G](c)\in\carrier[G]$
        by \obv{group_inverse_type}.
    $\mul[G](\mul[G](a,c),\groupInv[G](c))
        =\mul[G](\mul[G](b,c),\groupInv[G](c))$.
    Hence
    $\mul[G](a,\mul[G](c,\groupInv[G](c)))
        =\mul[G](b,\mul[G](c,\groupInv[G](c)))$
        by \obv{group_is_monoid,monoid_is_magma,monoid_assoc,magma_welldef}.
    Follows by
        \auto{group_is_monoid,monoid_is_unital_magma,group_inverse_right,unitalmagma_right}.
\end{proof}

\begin{theorem}\label{group_inverse_unique_right}
    Let $G$ be a group.
    Let $x,y\in\carrier[G]$.
    Suppose $\mul[G](x,y)=\neutral[G]$.
    Then $y=\groupInv[G](x)$.
\end{theorem}
\begin{proof}
    $\groupInv[G](x)\in\carrier[G]$
        by \obv{group_inverse_type}.
    $\mul[G](x,\groupInv[G](x))=\neutral[G]$
        by \obv{group_inverse_right}.
    Follows by \obv{group_cancel_left}.
\end{proof}

\begin{theorem}\label{group_inverse_unique_left}
    Let $G$ be a group.
    Let $x,y\in\carrier[G]$.
    Suppose $\mul[G](y,x)=\neutral[G]$.
    Then $y=\groupInv[G](x)$.
\end{theorem}
\begin{proof}
    $\groupInv[G](x)\in\carrier[G]$
        by \obv{group_inverse_type}.
    $\mul[G](\groupInv[G](x),x)=\neutral[G]$
        by \obv{group_inverse_left}.
    Follows by \obv{group_cancel_right}.
\end{proof}

\begin{proposition}\label{group_inverse_neutral}
    Let $G$ be a group.
    Then $\groupInv[G](\neutral[G])=\neutral[G]$.
\end{proposition}
\begin{proof}
    $\neutral[G]\in\carrier[G]$ by \obv{unitalmagma_type}.
    $\groupInv[G](\neutral[G])\in\carrier[G]$
        by \obv{group_inverse_type}.
    $\mul[G](\neutral[G],\groupInv[G](\neutral[G]))
        =\groupInv[G](\neutral[G])$
        by \obv{unitalmagma_left}.
    $\mul[G](\neutral[G],\groupInv[G](\neutral[G]))
        =\neutral[G]$
        by \obv{group_inverse_right}.
\end{proof}

\begin{theorem}\label{group_inverse_involutive}
    Let $G$ be a group.
    Let $x\in\carrier[G]$.
    Then $\groupInv[G](\groupInv[G](x))=x$.
\end{theorem}
\begin{proof}
    $\groupInv[G](x)\in\carrier[G]$
        by \obv{group_inverse_type}.
    $\mul[G](\groupInv[G](x),x)=\neutral[G]$
        by \obv{group_inverse_left}.
    Follows by \obv{group_inverse_unique_right}.
\end{proof}

\begin{theorem}\label{group_inverse_product}
    Let $G$ be a group.
    Let $x,y\in\carrier[G]$.
    Then
        $\groupInv[G](\mul[G](x,y))
            =\mul[G](\groupInv[G](y),\groupInv[G](x))$.
\end{theorem}
\begin{proof}
    $\mul[G](x,y)\in\carrier[G]$
        by \obv{group_is_monoid,monoid_is_magma,magma_welldef}.
    $\groupInv[G](x),\groupInv[G](y)\in\carrier[G]$
        by \obv{group_inverse_type}.
    $\mul[G](\groupInv[G](y),\groupInv[G](x))\in\carrier[G]$
        by \obv{group_is_monoid,monoid_is_magma,magma_welldef}.
    $\mul[G]
        (\mul[G](x,y),
         \mul[G](\groupInv[G](y),\groupInv[G](x)))
        =\neutral[G]$
        by
        \auto{group_is_monoid,monoid_is_unital_magma,monoid_is_magma,monoid_assoc,group_inverse_right,unitalmagma_left,unitalmagma_right,magma_welldef}.
    Follows by \obv{group_inverse_unique_right}.
\end{proof}

\begin{corollary}\label{commutative_group_inverse_product}
    Let $G$ be a commutative group.
    Let $x,y\in\carrier[G]$.
    Then
        $\groupInv[G](\mul[G](x,y))
            =\mul[G](\groupInv[G](x),\groupInv[G](y))$.
\end{corollary}
\begin{proof}
    $G$ is a group by \obv{commutative_group_is_group}.
    $\groupInv[G](x),\groupInv[G](y)\in\carrier[G]$
        by \obv{group_inverse_type}.
    Follows by
        \auto{group_inverse_product,commutative_group_commutative}.
\end{proof}

\begin{theorem}\label{group_solve_left}
    Let $G$ be a group.
    Let $a,b,x\in\carrier[G]$.
    Then $\mul[G](a,x)=b$ iff
        $x=\mul[G](\groupInv[G](a),b)$.
\end{theorem}
\begin{proof}
    $\groupInv[G](a)\in\carrier[G]$
        by \obv{group_inverse_type}.
    Follows by
        \auto{group_is_monoid,monoid_is_unital_magma,monoid_is_magma,monoid_assoc,group_inverse_left,unitalmagma_left,group_cancel_left,magma_welldef}.
\end{proof}

\begin{theorem}\label{group_solve_right}
    Let $G$ be a group.
    Let $a,b,x\in\carrier[G]$.
    Then $\mul[G](x,a)=b$ iff
        $x=\mul[G](b,\groupInv[G](a))$.
\end{theorem}
\begin{proof}
    $\groupInv[G](a)\in\carrier[G]$
        by \obv{group_inverse_type}.
    Follows by
        \auto{group_is_monoid,monoid_is_unital_magma,monoid_is_magma,monoid_assoc,group_inverse_right,unitalmagma_right,group_cancel_right,magma_welldef}.
\end{proof}
