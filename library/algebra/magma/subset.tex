\section{Pointwise multiplication of subsets}

\begin{definition}\label{pointwise_product}
    $\pointwiseProduct{M}{A}{B}
        =\{\mul[M](a,b)\mid a\in A,b\in B\}$.
\end{definition}

\begin{proposition}\label{pointwise_product_iff}
    $x\in\pointwiseProduct{M}{A}{B}$ iff there exist $a,b$ such that
        $a\in A$ and $b\in B$ and $x=\mul[M](a,b)$.
\end{proposition}
\begin{proof}
    Follows by \auto{pointwise_product}.
\end{proof}

\begin{classification}\label{pointwise_product_type}
    Let $M$ be a magma.
    Suppose $A,B\subseteq\carrier[M]$.
    Then $\pointwiseProduct{M}{A}{B}\subseteq\carrier[M]$.
\end{classification}
\begin{proof}
    Show that every element of $\pointwiseProduct{M}{A}{B}$ is an element of
        $\carrier[M]$.
    \begin{subproof}
        Fix $x\in\pointwiseProduct{M}{A}{B}$.
        Take $a,b$ such that $a\in A$ and $b\in B$ and
            $x=\mul[M](a,b)$ by \obv{pointwise_product_iff}.
        $a,b\in\carrier[M]$ by \obv{subseteq}.
        Hence $\mul[M](a,b)\in\carrier[M]$ by \obv{magma_welldef}.
    \end{subproof}
    Follows by \obv{subseteq}.
\end{proof}

\begin{proposition}\label{pointwise_product_monotone}
    Suppose $A\subseteq A'$ and $B\subseteq B'$.
    Then $\pointwiseProduct{M}{A}{B}
        \subseteq\pointwiseProduct{M}{A'}{B'}$.
\end{proposition}
\begin{proof}
    Show that every element of $\pointwiseProduct{M}{A}{B}$ is an element of
        $\pointwiseProduct{M}{A'}{B'}$.
    \begin{subproof}
        Fix $x\in\pointwiseProduct{M}{A}{B}$.
        Take $a,b$ such that $a\in A$ and $b\in B$ and
            $x=\mul[M](a,b)$ by \obv{pointwise_product_iff}.
        $a\in A'$ and $b\in B'$ by \obv{subseteq}.
        Follows by \auto{pointwise_product_iff}.
    \end{subproof}
    Follows by \obv{subseteq}.
\end{proof}

\begin{proposition}\label{pointwise_product_singletons}
    $\pointwiseProduct{M}{\{x\}}{\{y\}}
        =\{\mul[M](x,y)\}$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{pointwise_product_iff,singleton_iff,setext}.
\end{proof}

\begin{proposition}\label{pointwise_product_neutral_singleton_right}
    Let $M$ be a unital magma.
    Suppose $A\subseteq\carrier[M]$.
    Then $\pointwiseProduct{M}{A}{\{\neutral[M]\}}=A$.
\end{proposition}
\begin{proof}
    Show that every element of
        $\pointwiseProduct{M}{A}{\{\neutral[M]\}}$ is an element of $A$.
    \begin{subproof}
        Fix $z\in\pointwiseProduct{M}{A}{\{\neutral[M]\}}$.
        Take $a,b$ such that $a\in A$ and $b\in\{\neutral[M]\}$ and
            $z=\mul[M](a,b)$ by \auto{pointwise_product_iff}.
        Hence $b=\neutral[M]$ by \obv{singleton_iff}.
        $a\in\carrier[M]$ by \obv{subseteq}.
        Hence $\mul[M](a,\neutral[M])=a$ by
            \obv{unitalmagma_right}.
        Thus $z=a$.
    \end{subproof}
    Hence $\pointwiseProduct{M}{A}{\{\neutral[M]\}}\subseteq A$ by
        \obv{subseteq}.

    Show that every element of $A$ is an element of
        $\pointwiseProduct{M}{A}{\{\neutral[M]\}}$.
    \begin{subproof}
        Fix $a\in A$.
        $a\in\carrier[M]$ by \obv{subseteq}.
        $\mul[M](a,\neutral[M])=a$ by \obv{unitalmagma_right}.
        $\neutral[M]\in\{\neutral[M]\}$ by \obv{singleton_intro}.
        Follows by \obv{pointwise_product_iff}.
    \end{subproof}
    Hence $A\subseteq\pointwiseProduct{M}{A}{\{\neutral[M]\}}$ by
        \obv{subseteq}.
    Follows by \obv{subseteq_antisymmetric}.
\end{proof}

\begin{proposition}\label{pointwise_product_neutral_singleton_left}
    Let $M$ be a unital magma.
    Suppose $A\subseteq\carrier[M]$.
    Then $\pointwiseProduct{M}{\{\neutral[M]\}}{A}=A$.
\end{proposition}
\begin{proof}
    Show that every element of
        $\pointwiseProduct{M}{\{\neutral[M]\}}{A}$ is an element of $A$.
    \begin{subproof}
        Fix $z\in\pointwiseProduct{M}{\{\neutral[M]\}}{A}$.
        Take $a,b$ such that $a\in\{\neutral[M]\}$ and $b\in A$ and
            $z=\mul[M](a,b)$ by \auto{pointwise_product_iff}.
        Hence $a=\neutral[M]$ by \obv{singleton_iff}.
        $b\in\carrier[M]$ by \obv{subseteq}.
        Hence $\mul[M](\neutral[M],b)=b$ by
            \obv{unitalmagma_left}.
        Thus $z=b$.
    \end{subproof}
    Hence $\pointwiseProduct{M}{\{\neutral[M]\}}{A}\subseteq A$ by
        \obv{subseteq}.

    Show that every element of $A$ is an element of
        $\pointwiseProduct{M}{\{\neutral[M]\}}{A}$.
    \begin{subproof}
        Fix $b\in A$.
        $b\in\carrier[M]$ by \obv{subseteq}.
        $\mul[M](\neutral[M],b)=b$ by \obv{unitalmagma_left}.
        $\neutral[M]\in\{\neutral[M]\}$ by \obv{singleton_intro}.
        Follows by \obv{pointwise_product_iff}.
    \end{subproof}
    Hence $A\subseteq\pointwiseProduct{M}{\{\neutral[M]\}}{A}$ by
        \obv{subseteq}.
    Follows by \obv{subseteq_antisymmetric}.
\end{proof}

\begin{theorem}\label{pointwise_product_associative}
    Let $M$ be a semigroup.
    Suppose $A,B,C\subseteq\carrier[M]$.
    Then
        $\pointwiseProduct{M}{\pointwiseProduct{M}{A}{B}}{C}
            =\pointwiseProduct{M}{A}{\pointwiseProduct{M}{B}{C}}$.
\end{theorem}
\begin{proof}
    Show that every element of
        $\pointwiseProduct{M}{\pointwiseProduct{M}{A}{B}}{C}$ is an element of
        $\pointwiseProduct{M}{A}{\pointwiseProduct{M}{B}{C}}$.
    \begin{subproof}
        Fix $x\in\pointwiseProduct{M}{\pointwiseProduct{M}{A}{B}}{C}$.
        Take $d,c$ such that $d\in\pointwiseProduct{M}{A}{B}$ and
            $c\in C$ and $x=\mul[M](d,c)$
            by \auto{pointwise_product_iff}.
        Take $a,b$ such that $a\in A$ and $b\in B$ and
            $d=\mul[M](a,b)$ by \obv{pointwise_product_iff}.
        $a,b,c\in\carrier[M]$ by \obv{subseteq}.
        $\mul[M](b,c)\in\pointwiseProduct{M}{B}{C}$
            by \obv{pointwise_product_iff}.
        $\mul[M](\mul[M](a,b),c)=\mul[M](a,\mul[M](b,c))$
            by \obv{semigroup_assoc}.
        Hence $x=\mul[M](a,\mul[M](b,c))$.
        Follows by \auto{pointwise_product_iff}.
    \end{subproof}

    Show that every element of
        $\pointwiseProduct{M}{A}{\pointwiseProduct{M}{B}{C}}$ is an element of
        $\pointwiseProduct{M}{\pointwiseProduct{M}{A}{B}}{C}$.
    \begin{subproof}
        Fix $x\in\pointwiseProduct{M}{A}{\pointwiseProduct{M}{B}{C}}$.
        Take $a,d$ such that $a\in A$ and
            $d\in\pointwiseProduct{M}{B}{C}$ and
            $x=\mul[M](a,d)$ by \auto{pointwise_product_iff}.
        Take $b,c$ such that $b\in B$ and $c\in C$ and
            $d=\mul[M](b,c)$ by \obv{pointwise_product_iff}.
        $a,b,c\in\carrier[M]$ by \obv{subseteq}.
        $\mul[M](a,b)\in\pointwiseProduct{M}{A}{B}$
            by \obv{pointwise_product_iff}.
        $\mul[M](a,\mul[M](b,c))=\mul[M](\mul[M](a,b),c)$
            by \obv{semigroup_assoc}.
        Hence $x=\mul[M](\mul[M](a,b),c)$.
        Follows by \obv{pointwise_product_iff}.
    \end{subproof}
    Follows by set extensionality.
\end{proof}

\subsection{The powerset magma}

\begin{definition}\label{powerset_magma}
    $\PowersetMagma{M}=\structure[magma]{
        \carrier=\pow{\carrier[M]},
        \mul=\Apply{\binaryGraphOnOp}{
            \pow{\carrier[M]},\pow{\carrier[M]},
            \fun{A,B:\Set}{\pointwiseProduct{M}{A}{B}}}
    }$.
\end{definition}

\begin{proposition}\label{powerset_magma_carrier}
    $\carrier[\PowersetMagma{M}]=\pow{\carrier[M]}$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{powerset_magma,magmaaggregateprojection_carrier}.
\end{proof}

\begin{proposition}\label{powerset_magma_multiplication}
    $\mul[\PowersetMagma{M}]
        =\Apply{\binaryGraphOnOp}{
            \pow{\carrier[M]},\pow{\carrier[M]},
            \fun{A,B:\Set}{\pointwiseProduct{M}{A}{B}}}$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{powerset_magma,magmaaggregateprojection_mul}.
\end{proof}

\begin{proposition}\label{powerset_magma_mul_apply}
    Suppose $A,B\subseteq\carrier[M]$.
    Then $\mul[\PowersetMagma{M}](A,B)
        =\pointwiseProduct{M}{A}{B}$.
\end{proposition}
\begin{proof}
    $A,B\in\pow{\carrier[M]}$ by \obv{powerset_intro}.
    Follows by
        \auto{powerset_magma_multiplication,
            logical_binary_graph_on_apply}.
\end{proof}

\begin{theorem}\label{powerset_magma_is_exact}
    Let $M$ be a magma.
    Then $\PowersetMagma{M}$ is an exact magma.
\end{theorem}
\begin{proof}
    Let $P=\PowersetMagma{M}$.
    $\carrier[P]=\pow{\carrier[M]}$ by
        \auto{powerset_magma_carrier}.
    $\mul[P]=\Apply{\binaryGraphOnOp}{
        \pow{\carrier[M]},\pow{\carrier[M]},
        \fun{A,B:\Set}{\pointwiseProduct{M}{A}{B}}}$ by
        \auto{powerset_magma_multiplication}.
    Show that for all $A,B\in\pow{\carrier[M]}$ we have
        $\pointwiseProduct{M}{A}{B}\in\pow{\carrier[M]}$.
    \begin{subproof}
        Fix $A,B\in\pow{\carrier[M]}$.
        $A,B\subseteq\carrier[M]$ by \obv{powerset_elim}.
        $\pointwiseProduct{M}{A}{B}\subseteq\carrier[M]$
            by \obv{pointwise_product_type}.
        Follows by \obv{powerset_intro}.
    \end{subproof}
    Hence $\mul[P]\in
        \funs{\carrier[P]\times\carrier[P]}{\carrier[P]}$ by
        \auto{logical_binary_graph_on_function}.
    Hence $P$ is a magma by
        \auto{powerset_magma,magmaaggregateexact,magma}.
    Follows by \auto{powerset_magma,magmaaggregateexact}.
\end{proof}

\begin{theorem}\label{powerset_magma_is_semigroup}
    Let $M$ be a semigroup.
    Then $\PowersetMagma{M}$ is a semigroup.
\end{theorem}
\begin{proof}
    Let $P=\PowersetMagma{M}$.
    Hence $P$ is a magma by \obv{powerset_magma_is_exact}.
    Show that for all $A,B,C\in\carrier[P]$ we have
        $\mul[P](A,\mul[P](B,C))
            =\mul[P](\mul[P](A,B),C)$.
    \begin{subproof}
        Fix $A,B,C\in\carrier[P]$.
        $\carrier[P]=\pow{\carrier[M]}$ by
            \auto{powerset_magma_carrier}.
        Hence $A,B,C\in\pow{\carrier[M]}$.
        Hence $A,B,C\subseteq\carrier[M]$ by \obv{powerset_elim}.
        $\pointwiseProduct{M}{A}{B}\subseteq\carrier[M]$ and
            $\pointwiseProduct{M}{B}{C}\subseteq\carrier[M]$ by
            \obv{pointwise_product_type}.
        $\mul[P](A,B)=\pointwiseProduct{M}{A}{B}$ and
            $\mul[P](B,C)=\pointwiseProduct{M}{B}{C}$ by
            \auto{powerset_magma_mul_apply}.
        $\mul[P](A,\pointwiseProduct{M}{B}{C})
            =\pointwiseProduct{M}{A}{\pointwiseProduct{M}{B}{C}}$ and
        $\mul[P](\pointwiseProduct{M}{A}{B},C)
            =\pointwiseProduct{M}{\pointwiseProduct{M}{A}{B}}{C}$ by
            \auto{powerset_magma_mul_apply}.
        Follows by \obv{pointwise_product_associative}.
    \end{subproof}
    Follows by
        \auto{powerset_magma,magmaaggregateexact,magma,semigroup}.
\end{proof}
