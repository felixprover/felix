\section{Monoid homomorphisms}

A monoid homomorphism is a magma homomorphism that preserves the displayed
neutral elements.  Its source and target remain explicit.

\begin{definition}\label{monoid_hom}
    $f$ is a monoid homomorphism from $M$ to $N$ iff
        $M$ is a monoid and $N$ is a monoid and
        $f$ is a magma homomorphism from $M$ to $N$ and
        $f(\neutral[M])=\neutral[N]$.
\end{definition}

\begin{proposition}\label{monoid_hom_source_monoid}
    Suppose $f$ is a monoid homomorphism from $M$ to $N$.
    Then $M$ is a monoid.
\end{proposition}
\begin{proof}
    Follows by \obv{monoid_hom}.
\end{proof}

\begin{proposition}\label{monoid_hom_target_monoid}
    Suppose $f$ is a monoid homomorphism from $M$ to $N$.
    Then $N$ is a monoid.
\end{proposition}
\begin{proof}
    Follows by \obv{monoid_hom}.
\end{proof}

\begin{proposition}\label{monoid_hom_is_magma_hom}
    Suppose $f$ is a monoid homomorphism from $M$ to $N$.
    Then $f$ is a magma homomorphism from $M$ to $N$.
\end{proposition}
\begin{proof}
    Follows by \obv{monoid_hom}.
\end{proof}

\begin{proposition}\label{monoid_hom_type}
    Suppose $f$ is a monoid homomorphism from $M$ to $N$.
    Then $f\in\funs{\carrier[M]}{\carrier[N]}$.
\end{proposition}
\begin{proof}
    Follows by
        \obv{monoid_hom_is_magma_hom,magma_hom_type}.
\end{proof}

\begin{corollary}\label{monoid_hom_is_function}
    Suppose $f$ is a monoid homomorphism from $M$ to $N$.
    Then $f$ is a function.
\end{corollary}
\begin{proof}
    Follows by \auto{monoid_hom_type,funs_is_function}.
\end{proof}

\begin{corollary}\label{monoid_hom_apply_type}
    Suppose $f$ is a monoid homomorphism from $M$ to $N$.
    Let $x\in\carrier[M]$.
    Then $f(x)\in\carrier[N]$.
\end{corollary}
\begin{proof}
    Follows by \auto{monoid_hom_type,funs_type_apply}.
\end{proof}

\begin{proposition}\label{monoid_hom_mul}
    Suppose $f$ is a monoid homomorphism from $M$ to $N$.
    Let $x,y\in\carrier[M]$.
    Then $f(\mul[M](x,y))=\mul[N](f(x),f(y))$.
\end{proposition}
\begin{proof}
    Follows by
        \obv{monoid_hom_is_magma_hom,magma_hom_mul}.
\end{proof}

\begin{proposition}\label{monoid_hom_neutral}
    Suppose $f$ is a monoid homomorphism from $M$ to $N$.
    Then $f(\neutral[M])=\neutral[N]$.
\end{proposition}
\begin{proof}
    Follows by \obv{monoid_hom}.
\end{proof}

\begin{proposition}\label{identity_monoid_hom}
    Let $M$ be a monoid.
    Then $\identity{\carrier[M]}$ is a monoid homomorphism from $M$ to $M$.
\end{proposition}
\begin{proof}
    $\identity{\carrier[M]}$ is a magma homomorphism from $M$ to $M$ by
        \obv{monoid_is_magma,identity_magma_hom}.
    $\neutral[M]\in\carrier[M]$ by
        \obv{monoid_is_unital_magma,unitalmagma_type}.
    $(\identity{\carrier[M]})(\neutral[M])=\neutral[M]$ by
        \obv{id_apply}.
    Follows by \obv{monoid_hom}.
\end{proof}

\begin{theorem}\label{monoid_hom_composition}
    Suppose $f$ is a monoid homomorphism from $M$ to $N$ and
        $g$ is a monoid homomorphism from $N$ to $P$.
    Then $g\circ f$ is a monoid homomorphism from $M$ to $P$.
\end{theorem}
\begin{proof}
    $f$ is a magma homomorphism from $M$ to $N$ and
        $g$ is a magma homomorphism from $N$ to $P$ by
        \obv{monoid_hom_is_magma_hom}.
    Hence $g\circ f$ is a magma homomorphism from $M$ to $P$ by
        \obv{magma_hom_composition}.
    $f\in\funs{\carrier[M]}{\carrier[N]}$ and
        $g\in\funs{\carrier[N]}{\carrier[P]}$ by
        \obv{monoid_hom_type}.
    Hence $f$ and $g$ are functions and $g$ is composable with $f$ by
        \auto{funs_is_function,funs_ran,funs}.
    $\neutral[M]\in\carrier[M]$ by
        \obv{monoid_is_unital_magma,unitalmagma_type}.
    $(g\circ f)(\neutral[M])=g(f(\neutral[M]))$ by
        \obv{circ_apply,funs}.
    Hence $(g\circ f)(\neutral[M])=g(\neutral[N])$ by
        \obv{monoid_hom_neutral}.
    Hence $(g\circ f)(\neutral[M])=\neutral[P]$ by
        \obv{monoid_hom_neutral}.
    Follows by \obv{monoid_hom}.
\end{proof}

\begin{theorem}\label{monoid_hom_extensionality}
    Suppose $f$ is a monoid homomorphism from $M$ to $N$ and
        $g$ is a monoid homomorphism from $M$ to $N$.
    Suppose for every $x\in\carrier[M]$ we have $f(x)=g(x)$.
    Then $f=g$.
\end{theorem}
\begin{proof}
    $f$ is a magma homomorphism from $M$ to $N$ and
        $g$ is a magma homomorphism from $M$ to $N$ by
        \obv{monoid_hom_is_magma_hom}.
    Follows by \obv{magma_hom_extensionality}.
\end{proof}

\subsection{Embeddings, surjections, and isomorphisms}

\begin{definition}\label{monoid_embedding}
    $f$ is a monoid embedding from $M$ to $N$ iff
        $f$ is a monoid homomorphism from $M$ to $N$ and
        $f$ is injective.
\end{definition}

\begin{definition}\label{monoid_surjection}
    $f$ is a monoid surjection from $M$ to $N$ iff
        $f$ is a monoid homomorphism from $M$ to $N$ and
        $f\in\surj{\carrier[M]}{\carrier[N]}$.
\end{definition}

\begin{definition}\label{monoid_isomorphism}
    $f$ is a monoid isomorphism from $M$ to $N$ iff
        $f$ is a monoid homomorphism from $M$ to $N$ and
        $f\in\bijections{\carrier[M]}{\carrier[N]}$.
\end{definition}

\begin{proposition}\label{monoid_embedding_is_hom}
    Suppose $f$ is a monoid embedding from $M$ to $N$.
    Then $f$ is a monoid homomorphism from $M$ to $N$.
\end{proposition}
\begin{proof}
    Follows by \obv{monoid_embedding}.
\end{proof}

\begin{proposition}\label{monoid_embedding_injective}
    Suppose $f$ is a monoid embedding from $M$ to $N$.
    Then $f$ is injective.
\end{proposition}
\begin{proof}
    Follows by \obv{monoid_embedding}.
\end{proof}

\begin{proposition}\label{monoid_surjection_is_hom}
    Suppose $f$ is a monoid surjection from $M$ to $N$.
    Then $f$ is a monoid homomorphism from $M$ to $N$.
\end{proposition}
\begin{proof}
    Follows by \obv{monoid_surjection}.
\end{proof}

\begin{proposition}\label{monoid_surjection_is_surjection}
    Suppose $f$ is a monoid surjection from $M$ to $N$.
    Then $f\in\surj{\carrier[M]}{\carrier[N]}$.
\end{proposition}
\begin{proof}
    Follows by \obv{monoid_surjection}.
\end{proof}

\begin{proposition}\label{monoid_isomorphism_is_hom}
    Suppose $f$ is a monoid isomorphism from $M$ to $N$.
    Then $f$ is a monoid homomorphism from $M$ to $N$.
\end{proposition}
\begin{proof}
    Follows by \obv{monoid_isomorphism}.
\end{proof}

\begin{proposition}\label{monoid_isomorphism_is_bijection}
    Suppose $f$ is a monoid isomorphism from $M$ to $N$.
    Then $f\in\bijections{\carrier[M]}{\carrier[N]}$.
\end{proposition}
\begin{proof}
    Follows by \obv{monoid_isomorphism}.
\end{proof}

\begin{corollary}\label{monoid_isomorphism_is_embedding}
    Suppose $f$ is a monoid isomorphism from $M$ to $N$.
    Then $f$ is a monoid embedding from $M$ to $N$.
\end{corollary}
\begin{proof}
    $f$ is injective by
        \auto{monoid_isomorphism_is_bijection,bijections}.
    Follows by
        \obv{monoid_embedding,monoid_isomorphism_is_hom}.
\end{proof}

\begin{corollary}\label{monoid_isomorphism_is_surjection}
    Suppose $f$ is a monoid isomorphism from $M$ to $N$.
    Then $f$ is a monoid surjection from $M$ to $N$.
\end{corollary}
\begin{proof}
    $f\in\surj{\carrier[M]}{\carrier[N]}$ by
        \auto{monoid_isomorphism_is_bijection,bijections}.
    Follows by
        \obv{monoid_surjection,monoid_isomorphism_is_hom}.
\end{proof}

\begin{proposition}\label{identity_monoid_isomorphism}
    Let $M$ be a monoid.
    Then $\identity{\carrier[M]}$ is a monoid isomorphism from $M$ to $M$.
\end{proposition}
\begin{proof}
    $\identity{\carrier[M]}$ is a monoid homomorphism from $M$ to $M$ by
        \obv{identity_monoid_hom}.
    $\identity{\carrier[M]}\in
        \bijections{\carrier[M]}{\carrier[M]}$ by
        \obv{id_elem_bijections}.
    Follows by \obv{monoid_isomorphism}.
\end{proof}

\begin{theorem}\label{monoid_embedding_composition}
    Suppose $f$ is a monoid embedding from $M$ to $N$ and
        $g$ is a monoid embedding from $N$ to $P$.
    Then $g\circ f$ is a monoid embedding from $M$ to $P$.
\end{theorem}
\begin{proof}
    $f$ is a monoid homomorphism from $M$ to $N$ and
        $g$ is a monoid homomorphism from $N$ to $P$ by
        \obv{monoid_embedding_is_hom}.
    Hence $g\circ f$ is a monoid homomorphism from $M$ to $P$ by
        \obv{monoid_hom_composition}.
    $f$ and $g$ are injective by
        \obv{monoid_embedding_injective}.
    Hence $g\circ f$ is injective by \obv{circ_injective}.
    Follows by \obv{monoid_embedding}.
\end{proof}

\begin{theorem}\label{monoid_surjection_composition}
    Suppose $f$ is a monoid surjection from $M$ to $N$ and
        $g$ is a monoid surjection from $N$ to $P$.
    Then $g\circ f$ is a monoid surjection from $M$ to $P$.
\end{theorem}
\begin{proof}
    $f$ is a monoid homomorphism from $M$ to $N$ and
        $g$ is a monoid homomorphism from $N$ to $P$ by
        \obv{monoid_surjection_is_hom}.
    Hence $g\circ f$ is a monoid homomorphism from $M$ to $P$ by
        \obv{monoid_hom_composition}.
    $f\in\surj{\carrier[M]}{\carrier[N]}$ and
        $g\in\surj{\carrier[N]}{\carrier[P]}$ by
        \obv{monoid_surjection_is_surjection}.
    Hence $g\circ f\in\surj{\carrier[M]}{\carrier[P]}$ by
        \obv{surjections_circ}.
    Follows by \obv{monoid_surjection}.
\end{proof}

\begin{theorem}\label{monoid_isomorphism_composition}
    Suppose $f$ is a monoid isomorphism from $M$ to $N$ and
        $g$ is a monoid isomorphism from $N$ to $P$.
    Then $g\circ f$ is a monoid isomorphism from $M$ to $P$.
\end{theorem}
\begin{proof}
    $f$ is a monoid homomorphism from $M$ to $N$ and
        $g$ is a monoid homomorphism from $N$ to $P$ by
        \obv{monoid_isomorphism_is_hom}.
    Hence $g\circ f$ is a monoid homomorphism from $M$ to $P$ by
        \obv{monoid_hom_composition}.
    $f\in\bijections{\carrier[M]}{\carrier[N]}$ and
        $g\in\bijections{\carrier[N]}{\carrier[P]}$ by
        \obv{monoid_isomorphism_is_bijection}.
    Hence $g\circ f\in\bijections{\carrier[M]}{\carrier[P]}$ by
        \obv{bijection_circ}.
    Follows by \obv{monoid_isomorphism}.
\end{proof}

\begin{theorem}\label{monoid_isomorphism_converse_hom}
    Suppose $f$ is a monoid isomorphism from $M$ to $N$.
    Then $\converse{f}$ is a monoid homomorphism from $N$ to $M$.
\end{theorem}
\begin{proof}
    $f$ is a monoid homomorphism from $M$ to $N$ by
        \obv{monoid_isomorphism_is_hom}.
    Hence $M$ is a monoid and $N$ is a monoid by
        \obv{monoid_hom_source_monoid,monoid_hom_target_monoid}.
    $f$ is a magma homomorphism from $M$ to $N$ by
        \obv{monoid_hom_is_magma_hom}.
    $f\in\bijections{\carrier[M]}{\carrier[N]}$ by
        \obv{monoid_isomorphism_is_bijection}.
    Hence $f$ is a magma isomorphism from $M$ to $N$ by
        \obv{magma_isomorphism}.
    Hence $\converse{f}$ is a magma homomorphism from $N$ to $M$ by
        \obv{magma_isomorphism_converse_hom}.
    $\converse{f}\in\funs{\carrier[N]}{\carrier[M]}$ by
        \obv{bijective_converse_are_funs}.
    Hence $\converse{f}$ is a function and
        $\dom{\converse{f}}=\carrier[N]$ by \obv{funs_elim}.
    $f$ is a function by \obv{monoid_hom_is_function}.
    $f(\neutral[M])=\neutral[N]$ by \obv{monoid_hom_neutral}.
    Hence $(\neutral[M],\neutral[N])\in f$ by
        \auto{monoid_is_unital_magma,unitalmagma_type,
            monoid_hom_type,funs,function_apply_elim}.
    Hence $(\neutral[N],\neutral[M])\in\converse{f}$ by
        \obv{converse_iff}.
    Hence $(\converse{f})(\neutral[N])=\neutral[M]$ by
        \obv{function_apply_intro}.
    Follows by \obv{monoid_hom}.
\end{proof}

\begin{theorem}\label{monoid_isomorphism_converse}
    Suppose $f$ is a monoid isomorphism from $M$ to $N$.
    Then $\converse{f}$ is a monoid isomorphism from $N$ to $M$.
\end{theorem}
\begin{proof}
    $\converse{f}$ is a monoid homomorphism from $N$ to $M$ by
        \obv{monoid_isomorphism_converse_hom}.
    $f\in\bijections{\carrier[M]}{\carrier[N]}$ by
        \obv{monoid_isomorphism_is_bijection}.
    Hence $\converse{f}\in\bijections{\carrier[N]}{\carrier[M]}$ by
        \obv{bijection_converse_is_bijection}.
    Follows by \obv{monoid_isomorphism}.
\end{proof}

\subsection{Homomorphism sets}

\begin{definition}\label{monoid_hom_set}
    Let $M$ be a monoid.
    Let $N$ be a monoid.
    $\monoidHomSet{M}{N}=
        \{f\in\funs{\carrier[M]}{\carrier[N]}\mid
            \text{$f$ is a monoid homomorphism from $M$ to $N$}\}$.
\end{definition}

\begin{proposition}\label{monoid_hom_set_iff}
    Let $M$ be a monoid.
    Let $N$ be a monoid.
    Then $f\in\monoidHomSet{M}{N}$ iff
        $f$ is a monoid homomorphism from $M$ to $N$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{monoid_hom_set,monoid_hom,monoid_hom_type}.
\end{proof}

\begin{definition}\label{monoid_endomorphism_set}
    Let $M$ be a monoid.
    $\monoidEndomorphismSet{M}=\monoidHomSet{M}{M}$.
\end{definition}

\begin{proposition}\label{monoid_endomorphism_set_iff}
    Let $M$ be a monoid.
    Then $f\in\monoidEndomorphismSet{M}$ iff
        $f$ is a monoid homomorphism from $M$ to $M$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{monoid_endomorphism_set,monoid_hom_set_iff}.
\end{proof}
