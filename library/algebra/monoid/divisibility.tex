\import{algebra/monoid.tex}

\section{Sided divisibility in monoids}

Sidedness is part of every statement.  A left divisor is displayed on the
left of its cofactor, while a right divisor is displayed on the right.

\begin{definition}\label{left_monoid_divisibility}
    $a$ left-divides $b$ in $M$ iff
        $M$ is a monoid and $a,b\in\carrier[M]$ and
        there exists $c\in\carrier[M]$ such that
            $b=\mul[M](a,c)$.
\end{definition}

\begin{definition}\label{right_monoid_divisibility}
    $a$ right-divides $b$ in $M$ iff
        $M$ is a monoid and $a,b\in\carrier[M]$ and
        there exists $c\in\carrier[M]$ such that
            $b=\mul[M](c,a)$.
\end{definition}

\begin{proposition}\label{left_monoid_divisibility_guards}
    Suppose $a$ left-divides $b$ in $M$.
    Then $M$ is a monoid and $a,b\in\carrier[M]$.
\end{proposition}
\begin{proof}
    Follows by \obv{left_monoid_divisibility}.
\end{proof}

\begin{proposition}\label{right_monoid_divisibility_guards}
    Suppose $a$ right-divides $b$ in $M$.
    Then $M$ is a monoid and $a,b\in\carrier[M]$.
\end{proposition}
\begin{proof}
    Follows by \obv{right_monoid_divisibility}.
\end{proof}

\begin{proposition}\label{left_monoid_divisibility_reflexive}
    Let $M$ be a monoid and $a\in\carrier[M]$.
    Then $a$ left-divides $a$ in $M$.
\end{proposition}
\begin{proof}
    $\neutral[M]\in\carrier[M]$ by \obv{unitalmagma_type}.
    $a=\mul[M](a,\neutral[M])$ by \obv{unitalmagma_right}.
    Follows by \obv{left_monoid_divisibility}.
\end{proof}

\begin{proposition}\label{right_monoid_divisibility_reflexive}
    Let $M$ be a monoid and $a\in\carrier[M]$.
    Then $a$ right-divides $a$ in $M$.
\end{proposition}
\begin{proof}
    $\neutral[M]\in\carrier[M]$ by \obv{unitalmagma_type}.
    $a=\mul[M](\neutral[M],a)$ by \obv{unitalmagma_left}.
    Follows by \obv{right_monoid_divisibility}.
\end{proof}

\begin{theorem}\label{left_monoid_divisibility_transitive}
    Suppose $a$ left-divides $b$ in $M$ and
        $b$ left-divides $d$ in $M$.
    Then $a$ left-divides $d$ in $M$.
\end{theorem}
\begin{proof}
    $M$ is a monoid and $a,b,d\in\carrier[M]$ by
        \obv{left_monoid_divisibility_guards}.
    Take $x,y\in\carrier[M]$ such that
        $b=\mul[M](a,x)$ and $d=\mul[M](b,y)$ by
        \auto{left_monoid_divisibility}.
    $\mul[M](x,y)\in\carrier[M]$ by
        \obv{monoid_is_magma,magma_welldef}.
    $d=\mul[M](a,\mul[M](x,y))$ by
        \auto{monoid_assoc}.
    Follows by \obv{left_monoid_divisibility}.
\end{proof}

\begin{theorem}\label{right_monoid_divisibility_transitive}
    Suppose $a$ right-divides $b$ in $M$ and
        $b$ right-divides $d$ in $M$.
    Then $a$ right-divides $d$ in $M$.
\end{theorem}
\begin{proof}
    $M$ is a monoid and $a,b,d\in\carrier[M]$ by
        \obv{right_monoid_divisibility_guards}.
    Take $x,y\in\carrier[M]$ such that
        $b=\mul[M](x,a)$ and $d=\mul[M](y,b)$ by
        \auto{right_monoid_divisibility}.
    $\mul[M](y,x)\in\carrier[M]$ by
        \obv{monoid_is_magma,magma_welldef}.
    $d=\mul[M](\mul[M](y,x),a)$ by
        \auto{monoid_assoc}.
    Follows by \obv{right_monoid_divisibility}.
\end{proof}

\begin{proposition}\label{left_monoid_divides_product}
    Let $M$ be a monoid and $a,c\in\carrier[M]$.
    Then $a$ left-divides $\mul[M](a,c)$ in $M$.
\end{proposition}
\begin{proof}
    $\mul[M](a,c)\in\carrier[M]$ by
        \obv{monoid_is_magma,magma_welldef}.
    Follows by \obv{left_monoid_divisibility}.
\end{proof}

\begin{proposition}\label{right_monoid_divides_product}
    Let $M$ be a monoid and $a,c\in\carrier[M]$.
    Then $a$ right-divides $\mul[M](c,a)$ in $M$.
\end{proposition}
\begin{proof}
    $\mul[M](c,a)\in\carrier[M]$ by
        \obv{monoid_is_magma,magma_welldef}.
    Follows by \obv{right_monoid_divisibility}.
\end{proof}

\begin{theorem}\label{left_monoid_divisibility_mul_left}
    Suppose $a$ left-divides $b$ in $M$.
    Let $d\in\carrier[M]$.
    Then $\mul[M](d,a)$ left-divides $\mul[M](d,b)$ in $M$.
\end{theorem}
\begin{proof}
    $M$ is a monoid and $a,b\in\carrier[M]$ by
        \obv{left_monoid_divisibility_guards}.
    Take $c\in\carrier[M]$ such that $b=\mul[M](a,c)$ by
        \obv{left_monoid_divisibility}.
    $\mul[M](d,a),\mul[M](d,b)\in\carrier[M]$ by
        \obv{monoid_is_magma,magma_welldef}.
    $\mul[M](d,b)=\mul[M](\mul[M](d,a),c)$ by
        \auto{monoid_assoc}.
    Follows by \obv{left_monoid_divisibility}.
\end{proof}

\begin{theorem}\label{right_monoid_divisibility_mul_right}
    Suppose $a$ right-divides $b$ in $M$.
    Let $d\in\carrier[M]$.
    Then $\mul[M](a,d)$ right-divides $\mul[M](b,d)$ in $M$.
\end{theorem}
\begin{proof}
    $M$ is a monoid and $a,b\in\carrier[M]$ by
        \obv{right_monoid_divisibility_guards}.
    Take $c\in\carrier[M]$ such that $b=\mul[M](c,a)$ by
        \obv{right_monoid_divisibility}.
    $\mul[M](a,d),\mul[M](b,d)\in\carrier[M]$ by
        \obv{monoid_is_magma,magma_welldef}.
    $\mul[M](b,d)=\mul[M](c,\mul[M](a,d))$ by
        \auto{monoid_assoc}.
    Follows by \obv{right_monoid_divisibility}.
\end{proof}

\begin{theorem}\label{sided_monoid_divisibility_equivalent_if_commutative}
    Let $M$ be a monoid.
    Suppose for all $x,y\in\carrier[M]$ we have
        $\mul[M](x,y)=\mul[M](y,x)$.
    Let $a,b\in\carrier[M]$.
    Then $a$ left-divides $b$ in $M$ iff
        $a$ right-divides $b$ in $M$.
\end{theorem}
\begin{proof}
    Show that if $a$ left-divides $b$ in $M$, then
        $a$ right-divides $b$ in $M$.
    \begin{subproof}
        Assume $a$ left-divides $b$ in $M$.
        Take $c\in\carrier[M]$ such that $b=\mul[M](a,c)$ by
            \obv{left_monoid_divisibility}.
        $\mul[M](a,c)=\mul[M](c,a)$.
        Hence $b=\mul[M](c,a)$.
        Follows by \obv{right_monoid_divisibility}.
    \end{subproof}
    Show that if $a$ right-divides $b$ in $M$, then
        $a$ left-divides $b$ in $M$.
    \begin{subproof}
        Assume $a$ right-divides $b$ in $M$.
        Take $c\in\carrier[M]$ such that $b=\mul[M](c,a)$ by
            \obv{right_monoid_divisibility}.
        $\mul[M](c,a)=\mul[M](a,c)$.
        Hence $b=\mul[M](a,c)$.
        Follows by \obv{left_monoid_divisibility}.
    \end{subproof}
    Follows by \auto.
\end{proof}
