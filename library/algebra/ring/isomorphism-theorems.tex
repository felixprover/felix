\section{Isomorphism theorems for rings}

\subsection{The first isomorphism theorem}

For a ring homomorphism, the universal quotient lift at its kernel is the
additive first-isomorphism graph.  Corestricting the original homomorphism to
its image therefore equips this graph with the desired ring structure without
introducing a second factorization map.

\begin{proposition}\label{ring_first_isomorphism_map_eq_ring_quotient_lift}
    Suppose $f$ is a ring homomorphism from $R$ to $S$.
    Let $K=\ringKernel{f}{R}$, $G=\AdditiveGroup{R}$, and
        $H=\AdditiveGroup{S}$.
    Then
        $\firstIsomorphismMap{f}{G}{H}
            =\ringQuotientLift{f}{R}{K}$.
\end{proposition}
\begin{proof}
    $K=\groupKernel{f}{G}$ by
        \auto{ring_kernel}.
    $\firstIsomorphismMap{f}{G}{H}
        =\groupQuotientLift{f}{G}{K}$ by
        \obv{first_isomorphism_map_eq_group_quotient_lift}.
    $\ringQuotientLift{f}{R}{K}
        =\groupQuotientLift{f}{G}{K}$ by
        \obv{ring_quotient_lift}.
\end{proof}

\begin{theorem}\label{ring_first_isomorphism_map_hom}
    Suppose $f$ is a ring homomorphism from $R$ to $S$.
    Let $K=\ringKernel{f}{R}$ and $J=\ringImage{f}{R}$.
    Let $G=\AdditiveGroup{R}$ and $H=\AdditiveGroup{S}$.
    Then $\firstIsomorphismMap{f}{G}{H}$ is a ring homomorphism from
        $\QuotientRing{R}{K}$ to $\InducedSubring{S}{J}$.
\end{theorem}
\begin{proof}
    Let $A=\QuotientRing{R}{K}$ and $B=\InducedSubring{S}{J}$.
    Let $g=\firstIsomorphismMap{f}{G}{H}$.
    $K$ is a two-sided ideal of $R$ by
        \auto{ring_kernel_is_two_sided_ideal}.
    $J$ is a subring of $S$ by \auto{ring_image_is_subring}.
    $J\subseteq J$ by \obv{subseteq_refl}.
    Hence $f$ is a ring homomorphism from $R$ to $B$ by
        \obv{ring_hom_codomain_corestriction}.
    $K\subseteq\ringKernel{f}{R}$ by \obv{subseteq_refl}.
    Hence $\ringQuotientLift{f}{R}{K}$ is a ring homomorphism
        from $A$ to $B$ by
        \auto{ring_quotient_lift_hom}.
    $g=\ringQuotientLift{f}{R}{K}$ by
        \obv{ring_first_isomorphism_map_eq_ring_quotient_lift}.
\end{proof}

\begin{theorem}\label{ring_first_isomorphism_theorem}
    Suppose $f$ is a ring homomorphism from $R$ to $S$.
    Let $K=\ringKernel{f}{R}$ and $J=\ringImage{f}{R}$.
    Let $G=\AdditiveGroup{R}$ and $H=\AdditiveGroup{S}$.
    Then $\firstIsomorphismMap{f}{G}{H}$ is a ring isomorphism from
        $\QuotientRing{R}{K}$ to $\InducedSubring{S}{J}$.
\end{theorem}
\begin{proof}
    Let $A=\QuotientRing{R}{K}$ and $B=\InducedSubring{S}{J}$.
    Let $g=\firstIsomorphismMap{f}{G}{H}$.
    $K$ is a two-sided ideal of $R$ by
        \auto{ring_kernel_is_two_sided_ideal}.
    $J$ is a subring of $S$ by \auto{ring_image_is_subring}.
    $J\subseteq J$ by \obv{subseteq_refl}.
    Hence $f$ is a ring homomorphism from $R$ to $B$ by
        \obv{ring_hom_codomain_corestriction}.
    $K\subseteq\ringKernel{f}{R}$ by \obv{subseteq_refl}.
    $g$ is a ring homomorphism from $A$ to $B$ by
        \auto{ring_first_isomorphism_map_hom}.
    $\ringQuotientLift{f}{R}{K}$ is injective by
        \auto{ring_quotient_lift_injective_iff}.
    $\ringImage{f}{R}=\img{f}{\carrier[R]}$ by \obv{ring_image}.
    $\carrier[B]=J$ by \auto{induced_subring_agreement}.
    $\dom{f}=\carrier[R]$ by
        \auto{ring_hom_type,funs_elim}.
    Hence $\ran{f}=\img{f}{\carrier[R]}$ by \auto{img_dom}.
    $f\in\surj{\carrier[R]}{\ran{f}}$ by
        \auto{ring_hom_type,fun_to_surj}.
    Hence $f\in\surj{\carrier[R]}{\carrier[B]}$.
    Hence $\ringQuotientLift{f}{R}{K}\in
        \surj{\carrier[A]}{\carrier[B]}$ by
        \auto{ring_quotient_lift_surjective_iff}.
    $g=\ringQuotientLift{f}{R}{K}$ by
        \obv{ring_first_isomorphism_map_eq_ring_quotient_lift}.
    Hence $g\in\bijections{\carrier[A]}{\carrier[B]}$ by
        \obv{ring_hom_type,bijections}.
    Follows by \obv{ring_isomorphism}.
\end{proof}

\begin{proposition}\label{ring_first_isomorphism_map_apply}
    Suppose $f$ is a ring homomorphism from $R$ to $S$.
    Let $K=\ringKernel{f}{R}$.
    Let $G=\AdditiveGroup{R}$ and $H=\AdditiveGroup{S}$.
    Let $x\in\carrier[R]$.
    Then
        $\apply{\firstIsomorphismMap{f}{G}{H}}{
            \equivalenceClass{\subgroupCosetRelation{G}{K}}{x}}
        =f(x)$.
\end{proposition}
\begin{proof}
    $K$ is a two-sided ideal of $R$ by
        \auto{ring_kernel_is_two_sided_ideal}.
    $K\subseteq\ringKernel{f}{R}$ by \obv{subseteq_refl}.
    $\idealCongruence{R}{K}=\subgroupCosetRelation{G}{K}$ by
        \obv{ideal_congruence}.
    $\firstIsomorphismMap{f}{G}{H}
        =\ringQuotientLift{f}{R}{K}$ by
        \obv{ring_first_isomorphism_map_eq_ring_quotient_lift}.
    Follows by \auto{ring_quotient_lift_apply}.
\end{proof}

\begin{theorem}\label{ring_first_isomorphism_factorization}
    Suppose $f$ is a ring homomorphism from $R$ to $S$.
    Let $G=\AdditiveGroup{R}$ and $H=\AdditiveGroup{S}$.
    Let $K=\ringKernel{f}{R}$.
    Then
        $\firstIsomorphismMap{f}{G}{H}\circ
            \groupQuotientProjection{G}{K}=f$.
\end{theorem}
\begin{proof}
    $K$ is a two-sided ideal of $R$ by
        \auto{ring_kernel_is_two_sided_ideal}.
    $K\subseteq\ringKernel{f}{R}$ by \obv{subseteq_refl}.
    $\firstIsomorphismMap{f}{G}{H}
        =\ringQuotientLift{f}{R}{K}$ by
        \obv{ring_first_isomorphism_map_eq_ring_quotient_lift}.
    Follows by \auto{ring_quotient_lift_factorization}.
\end{proof}

\subsection{The second isomorphism theorem}

\begin{theorem}\label{ring_second_isomorphism_restriction_hom}
    Let $R$ be a ring.
    Suppose $A$ is a subring of $R$ and
        $I$ is a two-sided ideal of $R$.
    Let $T=\subringIdealSum{R}{A}{I}$.
    Let $D=\InducedSubring{R}{A}$ and $B=\InducedSubring{R}{T}$.
    Let $C=\QuotientRing{B}{I}$.
    Let $q=\groupQuotientProjection{\AdditiveGroup{B}}{I}$ and
        $r=\restrl{q}{A}$.
    Then $r$ is a ring homomorphism from $D$ to $C$.
\end{theorem}
\begin{proof}
    $A\subseteq\carrier[R]$ by \obv{subring_subseteq_carrier}.
    $I\subseteq\carrier[R]$ by
        \obv{two_sided_ideal_subseteq_carrier}.
    $T$ is a subring of $R$ by \auto{subring_ideal_sum_is_subring}.
    $A\subseteq T$ by \obv{subring_ideal_sum_contains_subring}.
    Hence $A$ is a subring of $B$ by
        \obv{subring_of_induced_subring}.
    $I$ is a two-sided ideal of $B$ by
        \obv{ideal_in_subring_ideal_sum}.
    Hence $q$ is a ring homomorphism from $B$ to $C$ by
        \auto{quotient_ring_projection_hom}.
    Hence $r$ is a ring homomorphism from
        $\InducedSubring{B}{A}$ to $C$ by
        \auto{ring_hom_domain_restriction}.
    $\InducedSubring{B}{A}=D$ by
        \auto{iterated_induced_subring}.
\end{proof}

\begin{theorem}\label{ring_second_isomorphism_restriction_kernel}
    Let $R$ be a ring.
    Suppose $A$ is a subring of $R$ and
        $I$ is a two-sided ideal of $R$.
    Let $K=A\inter I$ and $T=\subringIdealSum{R}{A}{I}$.
    Let $D=\InducedSubring{R}{A}$ and $B=\InducedSubring{R}{T}$.
    Let $C=\QuotientRing{B}{I}$.
    Let $q=\groupQuotientProjection{\AdditiveGroup{B}}{I}$ and
        $r=\restrl{q}{A}$.
    Then $\ringKernel{r}{D}=K$.
\end{theorem}
\begin{proof}
    $A\subseteq\carrier[R]$ by \obv{subring_subseteq_carrier}.
    $I\subseteq\carrier[R]$ by
        \obv{two_sided_ideal_subseteq_carrier}.
    $T$ is a subring of $R$ by \auto{subring_ideal_sum_is_subring}.
    $A\subseteq T$ and $I\subseteq T$ by
        \obv{subring_ideal_sum_contains_subring,
            subring_ideal_sum_contains_ideal}.
    $B$ is an exact ring by \auto{induced_subring_is_exact}.
    $I$ is a two-sided ideal of $B$ by
        \obv{ideal_in_subring_ideal_sum}.
    $r$ is a ring homomorphism from $D$ to $C$ by
        \obv{ring_second_isomorphism_restriction_hom}.
    $q$ is a ring homomorphism from $B$ to $C$ by
        \auto{quotient_ring_projection_hom}.
    $\ringKernel{q}{B}=I$ by
        \auto{quotient_ring_projection_kernel}.
    $q$ is a function and $A\subseteq\dom{q}$ by
        \auto{ring_hom_type,funs_is_function,funs,
            induced_subring_agreement,subseteq_transitive}.
    Show that every element of $\ringKernel{r}{D}$ is an element
        of $K$.
    \begin{subproof}
        Fix $x\in\ringKernel{r}{D}$.
        Hence $x\in\carrier[D]$ and $r(x)=\ringZero[C]$ by
            \auto{ring_kernel_iff}.
        Hence $x\in A$ by \auto{induced_subring_agreement}.
        Hence $r(x)=q(x)$ by \auto{restrl_of_function_apply}.
        $x\in\carrier[B]$ by
            \auto{induced_subring_agreement,subseteq}.
        Hence $x\in\ringKernel{q}{B}$ by
            \auto{ring_kernel_iff}.
        Hence $x\in I$.
        Follows by \auto{inter_intro}.
    \end{subproof}
    Show that every element of $K$ is an element of
        $\ringKernel{r}{D}$.
    \begin{subproof}
        Fix $x\in K$.
        Hence $x\in A$ and $x\in I$ by
            \auto{inter_elim_left,inter_elim_right}.
        Hence $x\in\carrier[D]$ by \auto{induced_subring_agreement}.
        $x\in\carrier[B]$ by
            \auto{induced_subring_agreement,subseteq}.
        Hence $x\in\ringKernel{q}{B}$.
        Hence $q(x)=\ringZero[C]$ by \auto{ring_kernel_iff}.
        $r(x)=q(x)$ by \auto{restrl_of_function_apply}.
        Follows by \auto{ring_kernel_iff}.
    \end{subproof}
    Follows by \obv{setext}.
\end{proof}

\begin{theorem}\label{ring_second_isomorphism_restriction_surjective}
    Let $R$ be a ring.
    Suppose $A$ is a subring of $R$ and
        $I$ is a two-sided ideal of $R$.
    Let $T=\subringIdealSum{R}{A}{I}$.
    Let $D=\InducedSubring{R}{A}$ and $B=\InducedSubring{R}{T}$.
    Let $C=\QuotientRing{B}{I}$.
    Let $q=\groupQuotientProjection{\AdditiveGroup{B}}{I}$ and
        $r=\restrl{q}{A}$.
    Then $r\in\surj{\carrier[D]}{\carrier[C]}$.
\end{theorem}
\begin{proof}
    $A\subseteq\carrier[R]$ by \obv{subring_subseteq_carrier}.
    $I\subseteq\carrier[R]$ by
        \obv{two_sided_ideal_subseteq_carrier}.
    $T$ is a subring of $R$ by \auto{subring_ideal_sum_is_subring}.
    $A\subseteq T$ and $I\subseteq T$ by
        \obv{subring_ideal_sum_contains_subring,
            subring_ideal_sum_contains_ideal}.
    $B$ is an exact ring by \auto{induced_subring_is_exact}.
    $I$ is a two-sided ideal of $B$ by
        \obv{ideal_in_subring_ideal_sum}.
    $r$ is a ring homomorphism from $D$ to $C$ by
        \obv{ring_second_isomorphism_restriction_hom}.
    $q$ is a ring surjection from $B$ to $C$ by
        \auto{quotient_ring_projection_surjective}.
    Hence $q$ is a ring homomorphism from $B$ to $C$ and
        $q\in\surj{\carrier[B]}{\carrier[C]}$ by
        \obv{ring_surjection_is_hom,
            ring_surjection_is_surjection}.
    $C$ is a ring by \obv{ring_hom_target_ring}.
    $\ringKernel{q}{B}=I$ by
        \auto{quotient_ring_projection_kernel}.
    $r\in\funs{\carrier[D]}{\carrier[C]}$ by
        \obv{ring_hom_type}.
    $q$ is a function and $A\subseteq\dom{q}$ by
        \auto{ring_hom_type,funs_is_function,funs,
            induced_subring_agreement,subseteq_transitive}.
    Show that for every $U\in\carrier[C]$ there exists
        $a\in\carrier[D]$ such that $r(a)=U$.
    \begin{subproof}
        Fix $U\in\carrier[C]$.
        Take $s\in\carrier[B]$ such that $q(s)=U$ by \obv{surj}.
        Hence $s\in T$ by \auto{induced_subring_agreement}.
        Take $a\in A$ such that there exists $i\in I$ such that
            $s=\ringAdd[R](a,i)$ by \auto{subring_ideal_sum_iff}.
        Take $i\in I$ such that $s=\ringAdd[R](a,i)$.
        Hence $a,i\in T$ by \obv{subseteq}.
        Hence $a,i\in\carrier[B]$ by
            \auto{induced_subring_agreement}.
        $\ringAdd[B](a,i)=\ringAdd[R](a,i)$ by
            \auto{induced_subring_agreement}.
        Hence $s=\ringAdd[B](a,i)$.
        Hence $q(s)=\ringAdd[C](q(a),q(i))$ by
            \auto{ring_hom_add}.
        $i\in\ringKernel{q}{B}$.
        Hence $q(i)=\ringZero[C]$ by \auto{ring_kernel_iff}.
        $q(a)\in\carrier[C]$ by \obv{ring_hom_apply_type}.
        Hence $q(s)=q(a)$ by \auto{ring_add_zero_right}.
        $a\in\carrier[D]$ by \auto{induced_subring_agreement}.
        $r(a)=q(a)$ by \auto{restrl_of_function_apply}.
        Follows by \auto.
    \end{subproof}
    Follows by \obv{surj}.
\end{proof}

\begin{theorem}\label{ring_second_isomorphism_theorem}
    Let $R$ be a ring.
    Suppose $A$ is a subring of $R$ and
        $I$ is a two-sided ideal of $R$.
    Let $K=A\inter I$ and $T=\subringIdealSum{R}{A}{I}$.
    Let $D=\InducedSubring{R}{A}$ and $B=\InducedSubring{R}{T}$.
    Let $C=\QuotientRing{B}{I}$.
    Let $q=\groupQuotientProjection{\AdditiveGroup{B}}{I}$ and
        $r=\restrl{q}{A}$.
    Let $G=\AdditiveGroup{D}$ and $H=\AdditiveGroup{C}$.
    Then $\firstIsomorphismMap{r}{G}{H}$ is a ring isomorphism from
        $\QuotientRing{D}{K}$ to $C$.
\end{theorem}
\begin{proof}
    Let $g=\firstIsomorphismMap{r}{G}{H}$.
    $A\subseteq\carrier[R]$ by \obv{subring_subseteq_carrier}.
    $I\subseteq\carrier[R]$ by
        \obv{two_sided_ideal_subseteq_carrier}.
    $T$ is a subring of $R$ by \auto{subring_ideal_sum_is_subring}.
    Hence $B$ is an exact ring by \auto{induced_subring_is_exact}.
    $K$ is a two-sided ideal of $D$ by
        \auto{subring_intersection_ideal_in_induced_subring}.
    $I$ is a two-sided ideal of $B$ by
        \obv{ideal_in_subring_ideal_sum}.
    Hence $C$ is an exact ring by \auto{quotient_ring_is_exact}.
    $r$ is a ring homomorphism from $D$ to $C$ by
        \obv{ring_second_isomorphism_restriction_hom}.
    $\ringKernel{r}{D}=K$ by
        \obv{ring_second_isomorphism_restriction_kernel}.
    $r\in\surj{\carrier[D]}{\carrier[C]}$ by
        \obv{ring_second_isomorphism_restriction_surjective}.
    Hence $\ringImage{r}{D}=\carrier[C]$ by
        \obv{ring_hom_surjective_iff_image_full}.
    $g$ is a ring isomorphism from
        $\QuotientRing{D}{\ringKernel{r}{D}}$ to
        $\InducedSubring{C}{\ringImage{r}{D}}$ by
        \auto{ring_first_isomorphism_theorem}.
    $\InducedSubring{C}{\carrier[C]}=C$ by
        \obv{full_induced_subring_eq}.
\end{proof}

    $T$ is a subring of $R$ by \auto{subring_ideal_sum_is_subring}.

        \obv{subring_ideal_sum_contains_subring,
            subring_ideal_sum_contains_ideal}.
    $h$ is a ring homomorphism from $B$ to $Q$ by
        \obv{ring_second_isomorphism_ambient_restriction_hom}.
    $p$ is a ring homomorphism from $R$ to $Q$ by
        \auto{quotient_ring_projection_hom}.
    Hence $Q$ is a ring by \obv{ring_hom_target_ring}.
    $\ringKernel{p}{R}=I$ by
        \auto{quotient_ring_projection_kernel}.
    $p$ is a function and $T\subseteq\dom{p}$ and
        $A\subseteq\dom{p}$ by
        \auto{ring_hom_type,funs_is_function,funs,
            subring_subseteq_carrier,subseteq_transitive}.
    Show that every element of $\ringImage{h}{B}$ is an element
        of $\img{p}{A}$.
    \begin{subproof}
        Fix $y\in\ringImage{h}{B}$.
        Take $s\in\carrier[B]$ such that $h(s)=y$ by
            \auto{ring_image_iff}.
        Hence $s\in T$ by \auto{induced_subring_agreement}.
        Take $a\in A$ such that there exists $i\in I$ such that
            $s=\ringAdd[R](a,i)$ by \auto{subring_ideal_sum_iff}.
        Take $i\in I$ such that $s=\ringAdd[R](a,i)$.
        Hence $a\in\carrier[R]$ by
            \obv{subring_subseteq_carrier,subseteq}.
        Hence $i\in\carrier[R]$ by
            \obv{two_sided_ideal_subseteq_carrier,subseteq}.
        Hence $s\in\carrier[R]$ by
            \auto{subring_subseteq_carrier,subseteq}.
        $p(s)=\ringAdd[Q](p(a),p(i))$ by \auto{ring_hom_add}.
        $i\in\ringKernel{p}{R}$.
        Hence $p(i)=\ringZero[Q]$ by \auto{ring_kernel_iff}.
        $p(a)\in\carrier[Q]$ by \obv{ring_hom_apply_type}.
        Hence $p(s)=p(a)$ by \auto{ring_add_zero_right}.
        $h(s)=p(s)$ by \auto{restrl_of_function_apply}.
        Hence $y=p(a)$.
        $a\in\dom{p}$ by \obv{subseteq}.
        $a\in\dom{p}\inter A$ by \obv{inter_intro}.
        Follows by \auto{img_of_function_intro}.
    \end{subproof}
    Show that every element of $\img{p}{A}$ is an element of
        $\ringImage{h}{B}$.
    \begin{subproof}
        Fix $y\in\img{p}{A}$.
        Take $a\in A$ such that $(a,y)\in p$ by \obv{img_iff}.
        Hence $p(a)=y$ by \obv{function_apply_intro}.
        Hence $a\in T$ by \obv{subseteq}.
        Hence $a\in\carrier[B]$ by \auto{induced_subring_agreement}.
        $h(a)=p(a)$ by \auto{restrl_of_function_apply}.
        Hence $h(a)=y$.
        Follows by \auto{ring_image_iff}.
    \end{subproof}
    Follows by \obv{setext}.
\end{proof}

\begin{theorem}\label{ring_second_isomorphism_image_form}

\end{proof}

\subsection{The third isomorphism theorem}

When $I\subseteq J$, the projection modulo $J$ is constant on the
$I$-cosets.  Its descent through $R/I$ has kernel equal to the image of $J$,
so the first isomorphism map gives the canonical quotient-of-a-quotient
isomorphism without choosing representatives.

\begin{theorem}\label{ring_third_isomorphism_image_ideal}
    Let $R$ be a ring.
    Suppose $I$ is a two-sided ideal of $R$ and
        $J$ is a two-sided ideal of $R$.
    Suppose $I\subseteq J$.
    Let $q_I=\groupQuotientProjection{\AdditiveGroup{R}}{I}$.
    Then $\img{q_I}{J}$ is a two-sided ideal of
        $\QuotientRing{R}{I}$.
\end{theorem}
\begin{proof}
    $q_I$ is a ring surjection from $R$ to $\QuotientRing{R}{I}$ by
        \auto{quotient_ring_projection_surjective}.
    Follows by \obv{ring_surjection_two_sided_ideal_image}.
\end{proof}

\begin{theorem}\label{ring_third_isomorphism_theorem}
    Let $R$ be a ring.
    Suppose $I$ is a two-sided ideal of $R$ and
        $J$ is a two-sided ideal of $R$.
    Suppose $I\subseteq J$.
    Let $G=\AdditiveGroup{R}$.
    Let $q_I=\groupQuotientProjection{G}{I}$ and
        $q_J=\groupQuotientProjection{G}{J}$.
    Let $E_I=\subgroupCosetRelation{G}{I}$ and
        $d=\quotientDescent{q_J}{\carrier[R]}{E_I}$.
    Let $A=\QuotientRing{R}{I}$ and $B=\QuotientRing{R}{J}$.
    Let $P=\img{q_I}{J}$.
    Let $H=\AdditiveGroup{A}$ and $K=\AdditiveGroup{B}$.
    Then $\firstIsomorphismMap{d}{H}{K}$ is a ring isomorphism from
        $\QuotientRing{A}{P}$ to $B$.
\end{theorem}
\begin{proof}
    Let $h=\firstIsomorphismMap{d}{H}{K}$.
    $P$ is a two-sided ideal of $A$ by
        \auto{ring_third_isomorphism_image_ideal}.
    Hence $\QuotientRing{A}{P}$ is an exact ring by
        \obv{quotient_ring_is_exact}.
    $d$ is a ring homomorphism from $A$ to $B$ by
        \auto{ring_quotient_comparison_hom}.
    $\ringKernel{d}{A}=P$ by
        \obv{ring_quotient_comparison_kernel}.
    $d\in\surj{\carrier[A]}{\carrier[B]}$ by
        \obv{ring_quotient_comparison_surjective}.
    Hence $\ringImage{d}{A}=\carrier[B]$ by
        \obv{ring_hom_surjective_iff_image_full}.
    $h$ is a ring isomorphism from
        $\QuotientRing{A}{\ringKernel{d}{A}}$ to
        $\InducedSubring{B}{\ringImage{d}{A}}$ by
        \auto{ring_first_isomorphism_theorem}.
    $B$ is an exact ring by \auto{quotient_ring_is_exact}.
    Hence $\InducedSubring{B}{\carrier[B]}=B$ by
        \obv{full_induced_subring_eq}.
\end{proof}

\subsection{The ideal correspondence theorem}

The same quotient projection identifies the two-sided ideals above its kernel
with the two-sided ideals of its codomain.  We first state the two inverse
equations and their order preservation, then package the assignments as
ordinary set-coded inverse bijections.

\begin{theorem}\label{ring_quotient_correspondence_preimage_ideal}
    Let $R$ be a ring.
    Suppose $I$ is a two-sided ideal of $R$.
    Let $G=\AdditiveGroup{R}$ and
        $q=\groupQuotientProjection{G}{I}$.
    Let $A=\QuotientRing{R}{I}$.
    Suppose $Q$ is a two-sided ideal of $A$.
    Then $\preimg{q}{Q}$ is a two-sided ideal of $R$.
\end{theorem}
\begin{proof}
    $q$ is a ring homomorphism from $R$ to $A$ by
        \auto{quotient_ring_projection_hom}.
    Follows by \obv{ring_hom_two_sided_ideal_preimage}.
\end{proof}

\begin{theorem}\label{ring_quotient_correspondence_preimage_contains_kernel}
    Let $R$ be a ring.
    Suppose $I$ is a two-sided ideal of $R$.
    Let $G=\AdditiveGroup{R}$ and
        $q=\groupQuotientProjection{G}{I}$.
    Let $A=\QuotientRing{R}{I}$.
    Suppose $Q$ is a two-sided ideal of $A$.
    Then $I\subseteq\preimg{q}{Q}$.
\end{theorem}
\begin{proof}
    $q$ is a ring homomorphism from $R$ to $A$ by
        \auto{quotient_ring_projection_hom}.
    $\ringKernel{q}{R}=I$ by
        \auto{quotient_ring_projection_kernel}.
    $Q$ is a left ideal of $A$ by \obv{two_sided_ideal}.
    Hence $\subgroup{Q}{\AdditiveGroup{A}}$ by
        \obv{left_ideal_additive_subgroup}.
    Hence $\ringZero[A]\in Q$ by
        \auto{subgroup_neutral,additive_group_fields}.
    Show that for every $x\in I$ we have $x\in\preimg{q}{Q}$.
    \begin{subproof}
        Fix $x\in I$.
        Hence $x\in\carrier[R]$ and $q(x)=\ringZero[A]$ by
            \auto{ring_kernel_iff}.
        Hence $x\in\carrier[G]$ by \auto{additive_group_fields}.
        $\groupHom{q}{G}{\AdditiveGroup{A}}$ by
            \auto{ring_hom_is_additive_group_hom}.
        Follows by
            \auto{group_hom_preimage_iff,additive_group_fields}.
    \end{subproof}
    Follows by \obv{subseteq}.
\end{proof}

\begin{theorem}\label{ring_quotient_correspondence_preimage_image}
    Let $R$ be a ring.
    Suppose $I$ is a two-sided ideal of $R$ and
        $J$ is a two-sided ideal of $R$.
    Suppose $I\subseteq J$.
    Let $G=\AdditiveGroup{R}$ and
        $q=\groupQuotientProjection{G}{I}$.
    Then $\preimg{q}{\img{q}{J}}=J$.
\end{theorem}
\begin{proof}
    Let $A=\QuotientRing{R}{I}$ and
        $H=\AdditiveGroup{A}$.
    $q$ is a ring homomorphism from $R$ to $A$ by
        \auto{quotient_ring_projection_hom}.
    Hence $\groupHom{q}{G}{H}$ by
        \auto{ring_hom_is_additive_group_hom}.
    $\subgroup{J}{G}$ by
        \auto{two_sided_ideal,left_ideal_additive_subgroup}.
    $\groupKernel{q}{G}=\ringKernel{q}{R}$ by
        \obv{ring_kernel}.
    Hence $\groupKernel{q}{G}=I$ by
        \auto{quotient_ring_projection_kernel}.
    Hence $\groupKernel{q}{G}\subseteq J$.
    Follows by \obv{group_hom_preimage_image_eq}.
\end{proof}

\begin{theorem}\label{ring_quotient_correspondence_image_preimage}
    Let $R$ be a ring.
    Suppose $I$ is a two-sided ideal of $R$.
    Let $G=\AdditiveGroup{R}$ and
        $q=\groupQuotientProjection{G}{I}$.
    Let $A=\QuotientRing{R}{I}$.
    Suppose $Q$ is a two-sided ideal of $A$.
    Then $\img{q}{\preimg{q}{Q}}=Q$.
\end{theorem}
\begin{proof}
    Let $H=\AdditiveGroup{A}$.
    $q$ is a ring homomorphism from $R$ to $A$ by
        \auto{quotient_ring_projection_hom}.
    Hence $\groupHom{q}{G}{H}$ by
        \auto{ring_hom_is_additive_group_hom}.
    $q\in\surj{\carrier[R]}{\carrier[A]}$ by
        \auto{quotient_ring_projection_surjective,
            ring_surjection_is_surjection}.
    Hence $q\in\surj{\carrier[G]}{\carrier[H]}$ by
        \auto{additive_group_fields}.
    $Q\subseteq\carrier[A]$ by
        \obv{two_sided_ideal_subseteq_carrier}.
    Hence $Q\subseteq\carrier[H]$ by
        \auto{additive_group_fields}.
    Follows by \obv{surjective_group_hom_image_preimage_eq}.
\end{proof}

\begin{theorem}\label{ring_quotient_correspondence_image_inclusion}
    Let $R$ be a ring.
    Suppose $I$ is a two-sided ideal of $R$.
    Suppose $J$ is a two-sided ideal of $R$ and
        $K$ is a two-sided ideal of $R$.
    Suppose $I\subseteq J$ and $I\subseteq K$.
    Let $q=\groupQuotientProjection{\AdditiveGroup{R}}{I}$.
    Then $J\subseteq K$ iff
        $\img{q}{J}\subseteq\img{q}{K}$.
\end{theorem}
\begin{proof}
    Show that if $J\subseteq K$, then
        $\img{q}{J}\subseteq\img{q}{K}$.
    \begin{subproof}
        Assume $J\subseteq K$.
        Follows by \obv{img_subseteq}.
    \end{subproof}
    Show that if $\img{q}{J}\subseteq\img{q}{K}$, then
        $J\subseteq K$.
    \begin{subproof}
        Assume $\img{q}{J}\subseteq\img{q}{K}$.
        Hence
            $\preimg{q}{\img{q}{J}}
                \subseteq\preimg{q}{\img{q}{K}}$ by
            \obv{preimg_subseteq}.
        $\preimg{q}{\img{q}{J}}=J$ and
            $\preimg{q}{\img{q}{K}}=K$ by
            \auto{ring_quotient_correspondence_preimage_image}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{theorem}\label{ring_ideal_correspondence_bijection}
    Let $R$ be a ring.
    Suppose $I$ is a two-sided ideal of $R$.
    Let $G=\AdditiveGroup{R}$ and
        $q=\groupQuotientProjection{G}{I}$.
    Let $A=\QuotientRing{R}{I}$.
    Let
        $C=\{J\in\pow{\carrier[R]}\mid
            \text{$J$ is a two-sided ideal of $R$ and
                $I\subseteq J$}\}$.
    Let
        $D=\{Q\in\pow{\carrier[A]}\mid
            \text{$Q$ is a two-sided ideal of $A$}\}$.
    Let
        $F=\Apply{\graphOnDomainOp}
            {C,\fun{J:\Set}{\img{q}{J}}}$.
    Then $F$ is a bijection from $C$ to $D$.
\end{theorem}
\begin{proof}
    Show that for every $J\in C$ we have $\img{q}{J}\in D$.
    \begin{subproof}
        Fix $J\in C$.
        Hence $J$ is a two-sided ideal of $R$ and
            $I\subseteq J$ by \auto.
        Hence $\img{q}{J}$ is a two-sided ideal of $A$ by
            \auto{ring_third_isomorphism_image_ideal}.
        Hence $\img{q}{J}\subseteq\carrier[A]$ by
            \obv{two_sided_ideal_subseteq_carrier}.
        Hence $\img{q}{J}\in\pow{\carrier[A]}$ by
            \obv{powerset_intro}.
        Follows by \auto.
    \end{subproof}
    Hence $F\in\funs{C}{D}$ by
        \auto{logical_graph_on_domain_type}.
    Hence $F$ is a function and $\dom{F}=C$ by
        \auto{funs_elim}.
    Show that $F\in\surj{C}{D}$.
    \begin{subproof}
        Show that for every $Q\in D$ there exists $J\in C$ such that
            $F(J)=Q$.
        \begin{subproof}
            Fix $Q\in D$.
            Hence $Q$ is a two-sided ideal of $A$ by \auto.
            Let $J=\preimg{q}{Q}$.
            $J$ is a two-sided ideal of $R$ by
                \auto{ring_quotient_correspondence_preimage_ideal}.
            Hence $J\subseteq\carrier[R]$ by
                \obv{two_sided_ideal_subseteq_carrier}.
            Hence $J\in\pow{\carrier[R]}$ by \obv{powerset_intro}.
            $I\subseteq J$ by
                \obv{ring_quotient_correspondence_preimage_contains_kernel}.
            Hence $J\in C$ by \auto.
            $F(J)=\img{q}{J}$ by
                \auto{logical_graph_on_domain_apply}.
            $\img{q}{J}=Q$ by
                \auto{ring_quotient_correspondence_image_preimage}.
            Follows by \auto.
        \end{subproof}
        Follows by \obv{surj}.
    \end{subproof}
    Show that $F$ is injective.
    \begin{subproof}
        Show that for all $J,K\in\dom{F}$ such that $F(J)=F(K)$
            we have $J=K$.
        \begin{subproof}
            Fix $J,K\in\dom{F}$.
            Assume $F(J)=F(K)$.
            Hence $J,K\in C$.
            Hence $J$ is a two-sided ideal of $R$ and
                $I\subseteq J$ by \auto.
            Hence $K$ is a two-sided ideal of $R$ and
                $I\subseteq K$ by \auto.
            $F(J)=\img{q}{J}$ and $F(K)=\img{q}{K}$ by
                \auto{logical_graph_on_domain_apply}.
            Hence $\img{q}{J}=\img{q}{K}$.
            $\preimg{q}{\img{q}{J}}=J$ and
                $\preimg{q}{\img{q}{K}}=K$ by
                \obv{ring_quotient_correspondence_preimage_image}.
        \end{subproof}
        Follows by \auto{injective_function}.
    \end{subproof}
    Follows by \obv{bijections}.
\end{proof}

\begin{theorem}\label{ring_ideal_correspondence_inverse_graph}
    Let $R$ be a ring.
    Suppose $I$ is a two-sided ideal of $R$.
    Let $G=\AdditiveGroup{R}$ and
        $q=\groupQuotientProjection{G}{I}$.
    Let $A=\QuotientRing{R}{I}$.
    Let
        $C=\{J\in\pow{\carrier[R]}\mid
            \text{$J$ is a two-sided ideal of $R$ and
                $I\subseteq J$}\}$.
    Let
        $D=\{Q\in\pow{\carrier[A]}\mid
            \text{$Q$ is a two-sided ideal of $A$}\}$.
    Let
        $F=\Apply{\graphOnDomainOp}
            {C,\fun{J:\Set}{\img{q}{J}}}$.
    Let
        $L=\Apply{\graphOnDomainOp}
            {D,\fun{Q:\Set}{\preimg{q}{Q}}}$.
    Then $\converse{F}=L$ and
        $L$ is a bijection from $D$ to $C$.
\end{theorem}
\begin{proof}
    $F$ is a bijection from $C$ to $D$ by
        \obv{ring_ideal_correspondence_bijection}.
    Hence $\converse{F}\in\funs{D}{C}$ by
        \obv{bijective_converse_are_funs}.
    Show that for every $Q\in D$ we have $\preimg{q}{Q}\in C$.
    \begin{subproof}
        Fix $Q\in D$.
        Hence $Q$ is a two-sided ideal of $A$ by \auto.
        $\preimg{q}{Q}$ is a two-sided ideal of $R$ by
            \obv{ring_quotient_correspondence_preimage_ideal}.
        Hence $\preimg{q}{Q}\subseteq\carrier[R]$ by
            \obv{two_sided_ideal_subseteq_carrier}.
        Hence $\preimg{q}{Q}\in\pow{\carrier[R]}$ by
            \obv{powerset_intro}.
        $I\subseteq\preimg{q}{Q}$ by
            \obv{ring_quotient_correspondence_preimage_contains_kernel}.
        Follows by \auto.
    \end{subproof}
    Hence $L\in\funs{D}{C}$ by
        \auto{logical_graph_on_domain_type}.
    Hence $\converse{F}$ and $L$ are functions and
        $\dom{\converse{F}}=D=\dom{L}$ by
        \auto{funs_elim}.
    Show that for every $Q\in\dom{\converse{F}}$ we have
        $(\converse{F})(Q)=L(Q)$.
    \begin{subproof}
        Fix $Q\in\dom{\converse{F}}$.
        Hence $Q\in D$.
        Hence $Q$ is a two-sided ideal of $A$ by \auto.
        Let $J=\preimg{q}{Q}$.
        Hence $J\in C$ by \auto.
        $F(J)=\img{q}{J}$ by
            \auto{logical_graph_on_domain_apply}.
        $\img{q}{J}=Q$ by
            \auto{ring_quotient_correspondence_image_preimage}.
        Hence $F(J)=Q$.
        Hence $(J,Q)\in F$ by
            \auto{function_apply_elim,bijection_is_function,
                bijections_dom}.
        Hence $(Q,J)\in\converse{F}$ by \auto{converse_iff}.
        Hence $(\converse{F})(Q)=J$ by
            \obv{function_apply_intro}.
        $L(Q)=J$ by \auto{logical_graph_on_domain_apply}.
    \end{subproof}
    Hence $\converse{F}=L$ by \obv{funext_on_domain}.
    $\converse{F}$ is a bijection from $D$ to $C$ by
        \obv{bijection_converse_is_bijection}.
\end{proof}
