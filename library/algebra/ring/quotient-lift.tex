\section{Homomorphisms induced on quotient rings}

The ring quotient lift is the universal group quotient lift on the additive
reduct.  Its graph records no target ring; target data enters only through
the endpoint-sensitive theorems below.

\begin{definition}\label{ring_quotient_lift}
    $\ringQuotientLift{f}{R}{I}=
        \groupQuotientLift{f}{\AdditiveGroup{R}}{I}$.
\end{definition}

\begin{proposition}\label{ring_quotient_lift_unfold}
    $\ringQuotientLift{f}{R}{I}=
        \quotientDescent
            {f}{\carrier[R]}{\idealCongruence{R}{I}}$.
\end{proposition}
\begin{proof}
    $\carrier[\AdditiveGroup{R}]=\carrier[R]$ by
        \obv{additive_group_fields}.
    $\idealCongruence{R}{I}=
        \subgroupCosetRelation{\AdditiveGroup{R}}{I}$ by
        \obv{ideal_congruence}.
    Follows by
        \obv{ring_quotient_lift,group_quotient_lift}.
\end{proof}

\begin{proposition}\label{ring_quotient_lift_compatible}
    Suppose $f$ is a ring homomorphism from $R$ to $S$.
    Suppose $I$ is a two-sided ideal of $R$.
    Suppose $I\subseteq\ringKernel{f}{R}$.
    Then $\quotientCompatible
        {f}{\carrier[R]}{\idealCongruence{R}{I}}$.
\end{proposition}
\begin{proof}
    Let $G=\AdditiveGroup{R}$ and $H=\AdditiveGroup{S}$.
    $\groupHom{f}{G}{H}$ by
        \obv{ring_hom_is_additive_group_hom}.
    $G$ is a commutative group by
        \obv{ring_hom_source_ring,
            additive_group_is_exact_commutative_group}.
    $\subgroup{I}{G}$ by
        \obv{two_sided_ideal,left_ideal_additive_subgroup}.
    Hence $\normalSubgroup{I}{G}$ by
        \obv{commutative_group_subgroup_is_normal}.
    $\ringKernel{f}{R}=\groupKernel{f}{G}$ by \obv{ring_kernel}.
    Hence $I\subseteq\groupKernel{f}{G}$.
    $\quotientCompatible
        {f}{\carrier[G]}{\subgroupCosetRelation{G}{I}}$ by
        \obv{group_quotient_lift_compatible}.
    $\carrier[G]=\carrier[R]$ and
        $\idealCongruence{R}{I}=\subgroupCosetRelation{G}{I}$ by
        \auto{additive_group_fields,ideal_congruence}.
\end{proof}

\begin{proposition}\label{ring_quotient_lift_type}
    Suppose $f$ is a ring homomorphism from $R$ to $S$.
    Suppose $I$ is a two-sided ideal of $R$.
    Suppose $I\subseteq\ringKernel{f}{R}$.
    Then $\ringQuotientLift{f}{R}{I}\in
        \funs{\carrier[\QuotientRing{R}{I}]}{\carrier[S]}$.
\end{proposition}
\begin{proof}
    Let $G=\AdditiveGroup{R}$ and $H=\AdditiveGroup{S}$.
    Let $Q=\quotientGroup{G}{I}$.
    $\groupHom{f}{G}{H}$ by
        \obv{ring_hom_is_additive_group_hom}.
    $G$ is a commutative group by
        \obv{ring_hom_source_ring,
            additive_group_is_exact_commutative_group}.
    $\subgroup{I}{G}$ by
        \obv{two_sided_ideal,left_ideal_additive_subgroup}.
    Hence $\normalSubgroup{I}{G}$ by
        \obv{commutative_group_subgroup_is_normal}.
    $\ringKernel{f}{R}=\groupKernel{f}{G}$ by \obv{ring_kernel}.
    Hence $I\subseteq\groupKernel{f}{G}$.
    $\groupQuotientLift{f}{G}{I}\in
        \funs{\carrier[Q]}{\carrier[H]}$ by
        \auto{group_quotient_lift_type}.
    $\carrier[Q]=\groupQuotientCarrier{G}{I}$ and
        $\carrier[\QuotientRing{R}{I}]=\groupQuotientCarrier{G}{I}$ and
        $\carrier[H]=\carrier[S]$ by
        \auto{quotient_group_carrier,quotient_ring_carrier,
            additive_group_fields}.
    $\ringQuotientLift{f}{R}{I}=\groupQuotientLift{f}{G}{I}$ by
        \obv{ring_quotient_lift}.
\end{proof}

\begin{proposition}\label{ring_quotient_lift_apply}
    Suppose $f$ is a ring homomorphism from $R$ to $S$.
    Suppose $I$ is a two-sided ideal of $R$.
    Suppose $I\subseteq\ringKernel{f}{R}$.
    Let $x\in\carrier[R]$.
    Then
        $\apply{\ringQuotientLift{f}{R}{I}}
            {\equivalenceClass{\idealCongruence{R}{I}}{x}}=f(x)$.
\end{proposition}
\begin{proof}

    $\groupHom{f}{G}{H}$ by
        \obv{ring_hom_is_additive_group_hom}.
    $G$ is a commutative group by
        \obv{ring_hom_source_ring,
            additive_group_is_exact_commutative_group}.
    $\subgroup{I}{G}$ by
        \obv{two_sided_ideal,left_ideal_additive_subgroup}.
    Hence $\normalSubgroup{I}{G}$ by
        \obv{commutative_group_subgroup_is_normal}.
    $\ringKernel{f}{R}=\groupKernel{f}{G}$ by \obv{ring_kernel}.

    $\apply{\groupQuotientLift{f}{G}{I}}

    $\ringQuotientLift{f}{R}{I}=\groupQuotientLift{f}{G}{I}$ and

\begin{theorem}\label{ring_quotient_lift_factorization}
    Suppose $f$ is a ring homomorphism from $R$ to $S$.
    Suppose $I$ is a two-sided ideal of $R$.
    Suppose $I\subseteq\ringKernel{f}{R}$.
    Then
        $\ringQuotientLift{f}{R}{I}\circ
            \groupQuotientProjection{\AdditiveGroup{R}}{I}=f$.
\end{theorem}
\begin{proof}
    Let $G=\AdditiveGroup{R}$ and $H=\AdditiveGroup{S}$.
    $\groupHom{f}{G}{H}$ by
        \obv{ring_hom_is_additive_group_hom}.
    $G$ is a commutative group by
        \obv{ring_hom_source_ring,
            additive_group_is_exact_commutative_group}.
    $\subgroup{I}{G}$ by
        \obv{two_sided_ideal,left_ideal_additive_subgroup}.
    Hence $\normalSubgroup{I}{G}$ by
        \obv{commutative_group_subgroup_is_normal}.
    $\ringKernel{f}{R}=\groupKernel{f}{G}$ by \obv{ring_kernel}.
    Hence $I\subseteq\groupKernel{f}{G}$.
    $\groupQuotientLift{f}{G}{I}\circ
        \groupQuotientProjection{G}{I}=f$ by
        \obv{group_quotient_lift_factorization}.
    $\ringQuotientLift{f}{R}{I}=\groupQuotientLift{f}{G}{I}$ by
        \obv{ring_quotient_lift}.
\end{proof}

\begin{theorem}\label{ring_quotient_lift_unique}
    Suppose $f$ is a ring homomorphism from $R$ to $S$.
    Suppose $I$ is a two-sided ideal of $R$.
    Suppose $I\subseteq\ringKernel{f}{R}$.
    Let $h\in\funs{\carrier[\QuotientRing{R}{I}]}{\carrier[S]}$.
    Suppose
        $h\circ\groupQuotientProjection{\AdditiveGroup{R}}{I}=f$.
    Then $h=\ringQuotientLift{f}{R}{I}$.
\end{theorem}
\begin{proof}
    Let $G=\AdditiveGroup{R}$ and $H=\AdditiveGroup{S}$.
    Let $Q=\quotientGroup{G}{I}$.
    $\groupHom{f}{G}{H}$ by
        \obv{ring_hom_is_additive_group_hom}.
    $G$ is a commutative group by
        \obv{ring_hom_source_ring,
            additive_group_is_exact_commutative_group}.
    $\subgroup{I}{G}$ by
        \obv{two_sided_ideal,left_ideal_additive_subgroup}.
    Hence $\normalSubgroup{I}{G}$ by
        \obv{commutative_group_subgroup_is_normal}.
    $\ringKernel{f}{R}=\groupKernel{f}{G}$ by \obv{ring_kernel}.
    Hence $I\subseteq\groupKernel{f}{G}$.
    $\carrier[Q]=\groupQuotientCarrier{G}{I}$ and
        $\carrier[\QuotientRing{R}{I}]=\groupQuotientCarrier{G}{I}$ and
        $\carrier[H]=\carrier[S]$ by
        \auto{quotient_group_carrier,quotient_ring_carrier,
            additive_group_fields}.
    Hence $h\in\funs{\carrier[Q]}{\carrier[H]}$.
    Hence $h=\groupQuotientLift{f}{G}{I}$ by
        \auto{group_quotient_lift_unique}.
    $\ringQuotientLift{f}{R}{I}=\groupQuotientLift{f}{G}{I}$ by
        \obv{ring_quotient_lift}.
\end{proof}

\begin{theorem}\label{ring_quotient_lift_additive_hom}
    Suppose $f$ is a ring homomorphism from $R$ to $S$.
    Suppose $I$ is a two-sided ideal of $R$.
    Suppose $I\subseteq\ringKernel{f}{R}$.
    Then
        $\groupHom
            {\ringQuotientLift{f}{R}{I}}
            {\AdditiveGroup{\QuotientRing{R}{I}}}
            {\AdditiveGroup{S}}$.
\end{theorem}
\begin{proof}
    Let $G=\AdditiveGroup{R}$ and $H=\AdditiveGroup{S}$.
    $\groupHom{f}{G}{H}$ by
        \obv{ring_hom_is_additive_group_hom}.
    $G$ is a commutative group by
        \obv{ring_hom_source_ring,
            additive_group_is_exact_commutative_group}.
    $\subgroup{I}{G}$ by
        \obv{two_sided_ideal,left_ideal_additive_subgroup}.
    Hence $\normalSubgroup{I}{G}$ by
        \obv{commutative_group_subgroup_is_normal}.
    $\ringKernel{f}{R}=\groupKernel{f}{G}$ by \obv{ring_kernel}.
    Hence $I\subseteq\groupKernel{f}{G}$.

\end{proof}

\begin{theorem}\label{ring_quotient_lift_hom}
    Suppose $f$ is a ring homomorphism from $R$ to $S$.
    Suppose $I$ is a two-sided ideal of $R$.
    Suppose $I\subseteq\ringKernel{f}{R}$.
    Then $\ringQuotientLift{f}{R}{I}$ is a ring homomorphism from
        $\QuotientRing{R}{I}$ to $S$.
\end{theorem}
\begin{proof}
    Let $A=\QuotientRing{R}{I}$ and $G=\AdditiveGroup{R}$.
    Let $E=\idealCongruence{R}{I}$ and
        $g=\ringQuotientLift{f}{R}{I}$.
    $A$ is an exact ring by \auto{quotient_ring_is_exact}.
    $\groupHom{g}{\AdditiveGroup{A}}{\AdditiveGroup{S}}$ by
        \auto{ring_quotient_lift_additive_hom}.
    Let $M=\MultiplicativeMonoid{A}$ and
        $N=\MultiplicativeMonoid{S}$.
    $M$ is a monoid and $N$ is a monoid by
        \obv{multiplicative_monoid_is_exact_monoid}.
    $g\in\funs{\carrier[M]}{\carrier[N]}$ by
        \auto{ring_quotient_lift_type,
            multiplicative_monoid_fields}.
    Show that for all $U,V\in\carrier[M]$ we have
        $g(\mul[M](U,V))=\mul[N](g(U),g(V))$.
    \begin{subproof}
        Fix $U,V\in\carrier[M]$.
        Hence $U,V\in\carrier[A]$ by
            \auto{multiplicative_monoid_fields}.
        Take $x\in\carrier[G]$ such that
            $U=\equivalenceClass{E}{x}$ by
            \auto{quotient_ring_carrier,group_quotient_carrier,
                ideal_congruence,quotient}.
        Take $y\in\carrier[G]$ such that
            $V=\equivalenceClass{E}{y}$ by
            \auto{quotient_ring_carrier,group_quotient_carrier,
                ideal_congruence,quotient}.
        Hence $x,y\in\carrier[R]$ by \auto{additive_group_fields}.
        $\ringMul[R](x,y)\in\carrier[R]$ by \obv{ring_mul_type}.
        $\ringMul[A](U,V)=
            \equivalenceClass{E}{\ringMul[R](x,y)}$ by
            \auto{quotient_ring_mul_apply,ideal_congruence}.
        Hence $g(\ringMul[A](U,V))=f(\ringMul[R](x,y))$ by
            \auto{ring_quotient_lift_apply}.
        $g(U)=f(x)$ and $g(V)=f(y)$ by
            \auto{ring_quotient_lift_apply}.
        $f(\ringMul[R](x,y))=\ringMul[S](f(x),f(y))$ by
            \obv{ring_hom_mul}.
        Follows by \auto{multiplicative_monoid_fields}.
    \end{subproof}
    Hence $g$ is a magma homomorphism from $M$ to $N$ by
        \obv{monoid_is_magma,magma_hom}.
    $\ringOne[A]=\equivalenceClass{E}{\ringOne[R]}$ by
        \auto{quotient_ring_one,ideal_congruence}.
    $\ringOne[R]\in\carrier[R]$ by
        \obv{ring_hom_source_ring,ring_one_type}.
    Hence $g(\ringOne[A])=f(\ringOne[R])$ by
        \auto{ring_quotient_lift_apply}.
    Hence $g(\ringOne[A])=\ringOne[S]$ by \obv{ring_hom_one}.
    Hence $g(\neutral[M])=\neutral[N]$ by
        \auto{multiplicative_monoid_fields}.
    Hence $g$ is a monoid homomorphism from $M$ to $N$ by
        \obv{monoid_hom}.
    Follows by \auto{ring_hom}.
\end{proof}

\begin{theorem}\label{ring_quotient_lift_range}
    Suppose $f$ is a ring homomorphism from $R$ to $S$.
    Suppose $I$ is a two-sided ideal of $R$.

\end{theorem}
\begin{proof}

    $\groupHom{f}{G}{H}$ by
        \obv{ring_hom_is_additive_group_hom}.
    $G$ is a commutative group by
        \obv{ring_hom_source_ring,
            additive_group_is_exact_commutative_group}.
    $\subgroup{I}{G}$ by
        \obv{two_sided_ideal,left_ideal_additive_subgroup}.

        \obv{commutative_group_subgroup_is_normal}.

        \obv{ring_quotient_lift}.
\end{proof}

\begin{theorem}\label{ring_quotient_lift_kernel}
    Suppose $f$ is a ring homomorphism from $R$ to $S$.
    Suppose $I$ is a two-sided ideal of $R$.
    Suppose $I\subseteq\ringKernel{f}{R}$.
    Let $q=\groupQuotientProjection{\AdditiveGroup{R}}{I}$.
    Then
        $\ringKernel{\ringQuotientLift{f}{R}{I}}
            {\QuotientRing{R}{I}}=
        \img{q}{\ringKernel{f}{R}}$.
\end{theorem}
\begin{proof}
    Let $G=\AdditiveGroup{R}$ and $H=\AdditiveGroup{S}$.
    Let $A=\QuotientRing{R}{I}$ and $Q=\quotientGroup{G}{I}$.
    $\groupHom{f}{G}{H}$ by
        \obv{ring_hom_is_additive_group_hom}.
    $G$ is a commutative group by
        \obv{ring_hom_source_ring,
            additive_group_is_exact_commutative_group}.
    $\subgroup{I}{G}$ by
        \obv{two_sided_ideal,left_ideal_additive_subgroup}.
    Hence $\normalSubgroup{I}{G}$ by

    $\ringKernel{f}{R}=\groupKernel{f}{G}$ by \obv{ring_kernel}.

    $\groupKernel{\groupQuotientLift{f}{G}{I}}{Q}=

        \auto{group_quotient_lift_kernel}.
    $\AdditiveGroup{A}=Q$ by
        \auto{quotient_ring_additive_group}.
    $\ringKernel{\ringQuotientLift{f}{R}{I}}{A}=
        \groupKernel{\ringQuotientLift{f}{R}{I}}{\AdditiveGroup{A}}$ by

    $\ringQuotientLift{f}{R}{I}=\groupQuotientLift{f}{G}{I}$ by

\end{proof}

\begin{theorem}\label{ring_quotient_lift_injective_iff}
    Suppose $f$ is a ring homomorphism from $R$ to $S$.
    Suppose $I$ is a two-sided ideal of $R$.
    Suppose $I\subseteq\ringKernel{f}{R}$.

        $\ringKernel{f}{R}=I$.
\end{theorem}
\begin{proof}

    $\groupHom{f}{G}{H}$ by
        \obv{ring_hom_is_additive_group_hom}.

        \obv{ring_hom_source_ring,
            additive_group_is_exact_commutative_group}.

        \obv{two_sided_ideal,left_ideal_additive_subgroup}.
