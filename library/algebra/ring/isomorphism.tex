\section{Ring isomorphisms}

A ring isomorphism is a bijective ring-homomorphism graph.  The source and
target rings remain part of the predicate, just as they do for
homomorphisms.

\begin{definition}\label{ring_isomorphism}
    $f$ is a ring isomorphism from $R$ to $S$ iff
        $f$ is a ring homomorphism from $R$ to $S$ and
        $f\in\bijections{\carrier[R]}{\carrier[S]}$.
\end{definition}

\begin{proposition}\label{ring_isomorphism_is_hom}
    Suppose $f$ is a ring isomorphism from $R$ to $S$.
    Then $f$ is a ring homomorphism from $R$ to $S$.
\end{proposition}
\begin{proof}
    Follows by \obv{ring_isomorphism}.
\end{proof}

\begin{proposition}\label{ring_isomorphism_is_bijection}
    Suppose $f$ is a ring isomorphism from $R$ to $S$.
    Then $f\in\bijections{\carrier[R]}{\carrier[S]}$.
\end{proposition}
\begin{proof}
    Follows by \obv{ring_isomorphism}.
\end{proof}

\begin{corollary}\label{ring_isomorphism_is_embedding}
    Suppose $f$ is a ring isomorphism from $R$ to $S$.
    Then $f$ is a ring embedding from $R$ to $S$.
\end{corollary}
\begin{proof}
    $f$ is injective by
        \auto{ring_isomorphism_is_bijection,bijections}.
    Follows by
        \obv{ring_embedding,ring_isomorphism_is_hom}.
\end{proof}

\begin{corollary}\label{ring_isomorphism_is_surjection}
    Suppose $f$ is a ring isomorphism from $R$ to $S$.
    Then $f$ is a ring surjection from $R$ to $S$.
\end{corollary}
\begin{proof}
    $f\in\surj{\carrier[R]}{\carrier[S]}$ by
        \auto{ring_isomorphism_is_bijection,bijections}.
    Follows by
        \obv{ring_surjection,ring_isomorphism_is_hom}.
\end{proof}

\begin{proposition}\label{ring_isomorphism_is_additive_group_isomorphism}
    Suppose $f$ is a ring isomorphism from $R$ to $S$.
    Then $\groupIsomorphism{f}{\AdditiveGroup{R}}{\AdditiveGroup{S}}$.
\end{proposition}
\begin{proof}
    $\groupHom{f}{\AdditiveGroup{R}}{\AdditiveGroup{S}}$ by
        \obv{ring_isomorphism_is_hom,
            ring_hom_is_additive_group_hom}.
    $f\in\bijections{\carrier[\AdditiveGroup{R}]}{
        \carrier[\AdditiveGroup{S}]}$ by
        \auto{ring_isomorphism_is_bijection,additive_group_fields}.
    Follows by \obv{group_isomorphism}.
\end{proof}

\begin{proposition}\label{ring_isomorphism_is_multiplicative_monoid_isomorphism}
    Suppose $f$ is a ring isomorphism from $R$ to $S$.
    Then $f$ is a monoid isomorphism from
        $\MultiplicativeMonoid{R}$ to $\MultiplicativeMonoid{S}$.
\end{proposition}
\begin{proof}
    $f$ is a monoid homomorphism from $\MultiplicativeMonoid{R}$ to
        $\MultiplicativeMonoid{S}$ by
        \obv{ring_isomorphism_is_hom,
            ring_hom_is_multiplicative_monoid_hom}.
    $f\in\bijections{\carrier[\MultiplicativeMonoid{R}]}{
        \carrier[\MultiplicativeMonoid{S}]}$ by
        \auto{ring_isomorphism_is_bijection,
            multiplicative_monoid_fields}.
    Follows by \obv{monoid_isomorphism}.
\end{proof}

\begin{proposition}\label{identity_ring_isomorphism}
    Let $R$ be a ring.
    Then $\identity{\carrier[R]}$ is a ring isomorphism from $R$ to $R$.
\end{proposition}
\begin{proof}
    $\identity{\carrier[R]}$ is a ring homomorphism from $R$ to $R$ by
        \obv{identity_ring_hom}.
    $\identity{\carrier[R]}\in
        \bijections{\carrier[R]}{\carrier[R]}$ by
        \obv{id_elem_bijections}.
    Follows by \obv{ring_isomorphism}.
\end{proof}

\begin{theorem}\label{ring_isomorphism_composition}
    Suppose $f$ is a ring isomorphism from $R$ to $S$ and
        $g$ is a ring isomorphism from $S$ to $T$.
    Then $g\circ f$ is a ring isomorphism from $R$ to $T$.
\end{theorem}
\begin{proof}
    $f$ is a ring homomorphism from $R$ to $S$ and
        $g$ is a ring homomorphism from $S$ to $T$ by
        \obv{ring_isomorphism_is_hom}.
    Hence $g\circ f$ is a ring homomorphism from $R$ to $T$ by
        \obv{ring_hom_composition}.
    $f\in\bijections{\carrier[R]}{\carrier[S]}$ and
        $g\in\bijections{\carrier[S]}{\carrier[T]}$ by
        \obv{ring_isomorphism_is_bijection}.
    Hence $g\circ f\in\bijections{\carrier[R]}{\carrier[T]}$ by
        \obv{bijection_circ}.
    Follows by \obv{ring_isomorphism}.
\end{proof}

\subsection{Inverse maps}

\begin{theorem}\label{ring_isomorphism_converse_hom}
    Suppose $f$ is a ring isomorphism from $R$ to $S$.
    Then $\converse{f}$ is a ring homomorphism from $S$ to $R$.
\end{theorem}
\begin{proof}
    Follows by
        \obv{ring_isomorphism_is_hom,
            ring_isomorphism_is_bijection,
            bijective_ring_hom_converse}.
\end{proof}

\begin{theorem}\label{ring_isomorphism_converse}
    Suppose $f$ is a ring isomorphism from $R$ to $S$.
    Then $\converse{f}$ is a ring isomorphism from $S$ to $R$.
\end{theorem}
\begin{proof}
    $\converse{f}$ is a ring homomorphism from $S$ to $R$ by
        \obv{ring_isomorphism_converse_hom}.
    $f\in\bijections{\carrier[R]}{\carrier[S]}$ by
        \obv{ring_isomorphism_is_bijection}.
    Hence $\converse{f}\in\bijections{\carrier[S]}{\carrier[R]}$ by
        \obv{bijection_converse_is_bijection}.
    Follows by \obv{ring_isomorphism}.
\end{proof}

\begin{theorem}\label{ring_isomorphism_converse_circ}
    Suppose $f$ is a ring isomorphism from $R$ to $S$.
    Then $\converse{f}\circ f=\identity{\carrier[R]}$.
\end{theorem}
\begin{proof}
    Follows by
        \obv{ring_isomorphism_is_bijection,
            bijection_converse_circ}.
\end{proof}

\begin{theorem}\label{ring_isomorphism_circ_converse}
    Suppose $f$ is a ring isomorphism from $R$ to $S$.
    Then $f\circ\converse{f}=\identity{\carrier[S]}$.
\end{theorem}
\begin{proof}
    Follows by
        \obv{ring_isomorphism_is_bijection,
            bijection_circ_converse}.
\end{proof}

\begin{theorem}\label{ring_hom_two_sided_inverse_isomorphism}
    Suppose $f$ is a ring homomorphism from $R$ to $S$ and
        $g$ is a ring homomorphism from $S$ to $R$.
    Suppose $g\circ f=\identity{\carrier[R]}$ and
        $f\circ g=\identity{\carrier[S]}$.
    Then $f$ is a ring isomorphism from $R$ to $S$.
\end{theorem}
\begin{proof}
    $\groupHom{f}{\AdditiveGroup{R}}{\AdditiveGroup{S}}$ and
        $\groupHom{g}{\AdditiveGroup{S}}{\AdditiveGroup{R}}$ by
        \obv{ring_hom_is_additive_group_hom}.
    $g\circ f=\identity{\carrier[\AdditiveGroup{R}]}$ and
        $f\circ g=\identity{\carrier[\AdditiveGroup{S}]}$ by
        \auto{additive_group_fields}.
    Hence $\groupIsomorphism{f}{\AdditiveGroup{R}}{\AdditiveGroup{S}}$ by
        \obv{group_hom_two_sided_inverse_isomorphism}.
    Hence $f\in\bijections{\carrier[\AdditiveGroup{R}]}{
        \carrier[\AdditiveGroup{S}]}$ by
        \obv{group_isomorphism_is_bijection}.
    Hence $f\in\bijections{\carrier[R]}{\carrier[S]}$ by
        \auto{additive_group_fields}.
    Follows by \obv{ring_isomorphism}.
\end{proof}

\begin{theorem}\label{ring_isomorphism_between_opposite_rings}
    Suppose $f$ is a ring isomorphism from $R$ to $S$.

        $\OppositeRing{R}$ to $\OppositeRing{S}$.
\end{theorem}
\begin{proof}

        \carrier[\OppositeRing{S}]}$ by
        \auto{ring_isomorphism_is_bijection,opposite_ring_fields}.

\end{proof}
