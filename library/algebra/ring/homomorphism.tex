\section{Ring homomorphisms}

A ring homomorphism preserves both canonical reducts.  In particular, it is
unital, and its source and target rings remain explicit parameters of the
predicate.

\begin{definition}\label{ring_hom}
    $f$ is a ring homomorphism from $R$ to $S$ iff
        $R$ is a ring and $S$ is a ring and
        $\groupHom{f}{\AdditiveGroup{R}}{\AdditiveGroup{S}}$ and
        $f$ is a monoid homomorphism from
            $\MultiplicativeMonoid{R}$ to $\MultiplicativeMonoid{S}$.
\end{definition}

\begin{proposition}\label{ring_hom_source_ring}
    Suppose $f$ is a ring homomorphism from $R$ to $S$.
    Then $R$ is a ring.
\end{proposition}
\begin{proof}
    Follows by \obv{ring_hom}.
\end{proof}

\begin{proposition}\label{ring_hom_target_ring}
    Suppose $f$ is a ring homomorphism from $R$ to $S$.
    Then $S$ is a ring.
\end{proposition}
\begin{proof}
    Follows by \obv{ring_hom}.
\end{proof}

\begin{proposition}\label{ring_hom_is_additive_group_hom}
    Suppose $f$ is a ring homomorphism from $R$ to $S$.
    Then $\groupHom{f}{\AdditiveGroup{R}}{\AdditiveGroup{S}}$.
\end{proposition}
\begin{proof}
    Follows by \obv{ring_hom}.
\end{proof}

\begin{proposition}\label{ring_hom_is_multiplicative_monoid_hom}
    Suppose $f$ is a ring homomorphism from $R$ to $S$.
    Then $f$ is a monoid homomorphism from
        $\MultiplicativeMonoid{R}$ to $\MultiplicativeMonoid{S}$.
\end{proposition}
\begin{proof}
    Follows by \obv{ring_hom}.
\end{proof}

\begin{proposition}\label{ring_hom_type}
    Suppose $f$ is a ring homomorphism from $R$ to $S$.
    Then $f\in\funs{\carrier[R]}{\carrier[S]}$.
\end{proposition}
\begin{proof}
    $f\in\funs{\carrier[\AdditiveGroup{R}]}{
        \carrier[\AdditiveGroup{S}]}$ by
        \obv{ring_hom_is_additive_group_hom,group_hom_type}.
    Follows by \auto{additive_group_fields}.
\end{proof}

\begin{proposition}\label{ring_hom_one}
    Suppose $f$ is a ring homomorphism from $R$ to $S$.
    Then $f(\ringOne[R])=\ringOne[S]$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{ring_hom_is_multiplicative_monoid_hom,
            monoid_hom_neutral,multiplicative_monoid_fields}.
\end{proof}

\subsection{Kernels, images, and homomorphism sets}

\begin{definition}\label{ring_kernel}
    $\ringKernel{f}{R}=\groupKernel{f}{\AdditiveGroup{R}}$.
\end{definition}

\begin{proposition}\label{ring_kernel_iff}
    Suppose $f$ is a ring homomorphism from $R$ to $S$.
    Then $x\in\ringKernel{f}{R}$ iff
        $x\in\carrier[R]$ and $f(x)=\ringZero[S]$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{ring_kernel,ring_hom_is_additive_group_hom,
            group_kernel_iff,additive_group_fields}.
\end{proof}

\begin{theorem}\label{ring_kernel_is_additive_subgroup}
    Suppose $f$ is a ring homomorphism from $R$ to $S$.
    Then $\subgroup{\ringKernel{f}{R}}{\AdditiveGroup{R}}$.
\end{theorem}
\begin{proof}
    Follows by
        \auto{ring_kernel,ring_hom_is_additive_group_hom,
            group_kernel_subgroup}.
\end{proof}

\begin{definition}\label{ring_image}
    $\ringImage{f}{R}=\img{f}{\carrier[R]}$.
\end{definition}

\begin{proposition}\label{ring_image_iff}
    Suppose $f$ is a ring homomorphism from $R$ to $S$.
    Then $y\in\ringImage{f}{R}$ iff
        there exists $x\in\carrier[R]$ such that $f(x)=y$.
\end{proposition}
\begin{proof}
    $\groupHom{f}{\AdditiveGroup{R}}{\AdditiveGroup{S}}$ by
        \obv{ring_hom_is_additive_group_hom}.
    $\carrier[R]\subseteq\carrier[\AdditiveGroup{R}]$ by
        \auto{additive_group_fields,subseteq_refl}.
    Follows by
        \auto{ring_image,group_hom_image_iff}.
\end{proof}

\begin{theorem}\label{ring_image_is_additive_subgroup}
    Suppose $f$ is a ring homomorphism from $R$ to $S$.
    Then $\subgroup{\ringImage{f}{R}}{\AdditiveGroup{S}}$.
\end{theorem}
\begin{proof}
    Follows by
        \auto{ring_image,ring_hom_is_additive_group_hom,
            group_hom_image_subgroup,additive_group_fields}.
\end{proof}

\begin{theorem}\label{ring_hom_surjective_iff_image_full}
    Suppose $f$ is a ring homomorphism from $R$ to $S$.
    Then $f\in\surj{\carrier[R]}{\carrier[S]}$ iff
        $\ringImage{f}{R}=\carrier[S]$.
\end{theorem}
\begin{proof}
    $f\in\funs{\carrier[R]}{\carrier[S]}$ by \obv{ring_hom_type}.
    Hence $\dom{f}=\carrier[R]$ by \obv{funs}.
    Hence $\ran{f}=\img{f}{\carrier[R]}$ by \auto{img_dom}.
    $f\in\surj{\carrier[R]}{\carrier[S]}$ iff
        $\ran{f}=\carrier[S]$ by \obv{funs_surj_iff}.
    Follows by \auto{ring_image}.
\end{proof}

\begin{definition}\label{ring_hom_set}
    Let $R$ be a ring.
    Let $S$ be a ring.
    $\ringHomSet{R}{S}=
        \{f\in\funs{\carrier[R]}{\carrier[S]}\mid
            \text{$f$ is a ring homomorphism from $R$ to $S$}\}$.
\end{definition}

\begin{proposition}\label{ring_hom_set_iff}
    Let $R$ be a ring.
    Let $S$ be a ring.
    Then $f\in\ringHomSet{R}{S}$ iff
        $f$ is a ring homomorphism from $R$ to $S$.
\end{proposition}
\begin{proof}
    Follows by \auto{ring_hom_set,ring_hom_type}.
\end{proof}

\begin{definition}\label{ring_endomorphism_set}
    Let $R$ be a ring.
    $\ringEndomorphismSet{R}=\ringHomSet{R}{R}$.
\end{definition}

\begin{proposition}\label{ring_endomorphism_set_iff}
    Let $R$ be a ring.
    Then $f\in\ringEndomorphismSet{R}$ iff
        $f$ is a ring homomorphism from $R$ to $R$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{ring_endomorphism_set,ring_hom_set_iff}.
\end{proof}

\subsection{Natural elements and powers}

\begin{theorem}\label{ring_hom_preserves_natural}
    Suppose $f$ is a ring homomorphism from $R$ to $S$.
    Let $n$ be a natural number.
    Then $f(\ringNatural{R}{n})=\ringNatural{S}{n}$.
\end{theorem}
\begin{proof}
    $\groupHom{f}{\AdditiveGroup{R}}{\AdditiveGroup{S}}$ by
        \obv{ring_hom_is_additive_group_hom}.
    Hence $f$ is a monoid homomorphism from $\AdditiveGroup{R}$ to
        $\AdditiveGroup{S}$ by \obv{group_hom_is_monoid_hom}.
    $\ringOne[R]\in\carrier[R]$ by
        \obv{ring_hom_source_ring,ring_one_type}.
    Hence $\ringOne[R]\in\carrier[\AdditiveGroup{R}]$ by
        \auto{additive_group_fields}.
    $f(\monoidPower{\AdditiveGroup{R}}{\ringOne[R]}{n})=
        \monoidPower{\AdditiveGroup{S}}{f(\ringOne[R])}{n}$ by
        \obv{monoid_hom_preserves_power}.
    $f(\ringOne[R])=\ringOne[S]$ by \obv{ring_hom_one}.
    Follows by \auto{ring_natural}.
\end{proof}

\begin{theorem}\label{ring_hom_preserves_power}
    Suppose $f$ is a ring homomorphism from $R$ to $S$.
    Let $x\in\carrier[R]$.
    Let $n$ be a natural number.
    Then $f(\ringPower{R}{x}{n})=\ringPower{S}{f(x)}{n}$.
\end{theorem}
\begin{proof}
    $f$ is a monoid homomorphism from $\MultiplicativeMonoid{R}$ to
        $\MultiplicativeMonoid{S}$ by
        \obv{ring_hom_is_multiplicative_monoid_hom}.
    $x\in\carrier[\MultiplicativeMonoid{R}]$ by
        \auto{multiplicative_monoid_fields}.
    Follows by
        \auto{ring_power,monoid_hom_preserves_power}.
\end{proof}

\begin{corollary}\label{ring_hom_is_function}
    Suppose $f$ is a ring homomorphism from $R$ to $S$.
    Then $f$ is a function.
\end{corollary}
\begin{proof}
    Follows by \auto{ring_hom_type,funs_is_function}.
\end{proof}

\begin{corollary}\label{ring_hom_apply_type}
    Suppose $f$ is a ring homomorphism from $R$ to $S$.
    Let $x\in\carrier[R]$.
    Then $f(x)\in\carrier[S]$.
\end{corollary}
\begin{proof}
    Follows by \auto{ring_hom_type,funs_type_apply}.
\end{proof}

\begin{proposition}\label{ring_hom_add}
    Suppose $f$ is a ring homomorphism from $R$ to $S$.
    Let $x,y\in\carrier[R]$.
    Then $f(\ringAdd[R](x,y))=\ringAdd[S](f(x),f(y))$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{ring_hom_is_additive_group_hom,group_hom_mul,
            additive_group_fields}.
\end{proof}

\begin{proposition}\label{ring_hom_zero}
    Suppose $f$ is a ring homomorphism from $R$ to $S$.
    Then $f(\ringZero[R])=\ringZero[S]$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{ring_hom_is_additive_group_hom,group_hom_neutral,
            additive_group_fields}.
\end{proof}

\begin{proposition}\label{ring_hom_neg}
    Suppose $f$ is a ring homomorphism from $R$ to $S$.
    Let $x\in\carrier[R]$.
    Then $f(\ringNeg[R](x))=\ringNeg[S](f(x))$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{ring_hom_is_additive_group_hom,group_hom_inverse,
            additive_group_fields}.
\end{proof}

\begin{proposition}\label{ring_hom_mul}
    Suppose $f$ is a ring homomorphism from $R$ to $S$.
    Let $x,y\in\carrier[R]$.
    Then $f(\ringMul[R](x,y))=\ringMul[S](f(x),f(y))$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{ring_hom_is_multiplicative_monoid_hom,
            monoid_hom_mul,multiplicative_monoid_fields}.
\end{proof}

\begin{proposition}\label{identity_ring_hom}
    Let $R$ be a ring.
    Then $\identity{\carrier[R]}$ is a ring homomorphism from $R$ to $R$.
\end{proposition}
\begin{proof}
    $\AdditiveGroup{R}$ is an exact commutative group by
        \obv{additive_group_is_exact_commutative_group}.
    $\MultiplicativeMonoid{R}$ is a monoid by
        \obv{multiplicative_monoid_is_exact_monoid}.
    $\groupHom{\identity{\carrier[\AdditiveGroup{R}]}}{
        \AdditiveGroup{R}}{\AdditiveGroup{R}}$ by
        \obv{identity_group_hom}.
    Hence $\groupHom{\identity{\carrier[R]}}{
        \AdditiveGroup{R}}{\AdditiveGroup{R}}$ by
        \auto{additive_group_fields}.
    $\identity{\carrier[\MultiplicativeMonoid{R}]}$
        is a monoid homomorphism from $\MultiplicativeMonoid{R}$ to
        $\MultiplicativeMonoid{R}$ by
        \obv{identity_monoid_hom}.
    Hence $\identity{\carrier[R]}$ is a monoid homomorphism from
        $\MultiplicativeMonoid{R}$ to $\MultiplicativeMonoid{R}$ by
        \auto{multiplicative_monoid_fields}.
    Follows by \obv{ring_hom}.
\end{proof}

\begin{theorem}\label{ring_hom_composition}
    Suppose $f$ is a ring homomorphism from $R$ to $S$ and
        $g$ is a ring homomorphism from $S$ to $T$.
    Then $g\circ f$ is a ring homomorphism from $R$ to $T$.
\end{theorem}
\begin{proof}
    $\groupHom{f}{\AdditiveGroup{R}}{\AdditiveGroup{S}}$ and
        $\groupHom{g}{\AdditiveGroup{S}}{\AdditiveGroup{T}}$ by
        \obv{ring_hom_is_additive_group_hom}.
    Hence $\groupHom{g\circ f}{\AdditiveGroup{R}}{
        \AdditiveGroup{T}}$ by \obv{group_hom_composition}.
    $f$ is a monoid homomorphism from $\MultiplicativeMonoid{R}$ to
        $\MultiplicativeMonoid{S}$ and
        $g$ is a monoid homomorphism from $\MultiplicativeMonoid{S}$ to
        $\MultiplicativeMonoid{T}$ by
        \obv{ring_hom_is_multiplicative_monoid_hom}.
    Hence $g\circ f$ is a monoid homomorphism from
        $\MultiplicativeMonoid{R}$ to $\MultiplicativeMonoid{T}$ by
        \obv{monoid_hom_composition}.
    Follows by \obv{ring_hom}.
\end{proof}

\begin{theorem}\label{ring_hom_extensionality}
    Suppose $f$ is a ring homomorphism from $R$ to $S$ and
        $g$ is a ring homomorphism from $R$ to $S$.
    Suppose for every $x\in\carrier[R]$ we have $f(x)=g(x)$.
    Then $f=g$.
\end{theorem}
\begin{proof}
    $\groupHom{f}{\AdditiveGroup{R}}{\AdditiveGroup{S}}$ and
        $\groupHom{g}{\AdditiveGroup{R}}{\AdditiveGroup{S}}$ by
        \obv{ring_hom_is_additive_group_hom}.
    For every $x\in\carrier[\AdditiveGroup{R}]$ we have
        $f(x)=g(x)$ by \auto{additive_group_fields}.
    Follows by \obv{group_hom_extensionality}.
\end{proof}

\subsection{Embeddings and surjections}

\begin{definition}\label{ring_embedding}
    $f$ is a ring embedding from $R$ to $S$ iff
        $f$ is a ring homomorphism from $R$ to $S$ and
        $f$ is injective.
\end{definition}

\begin{definition}\label{ring_surjection}
    $f$ is a ring surjection from $R$ to $S$ iff
        $f$ is a ring homomorphism from $R$ to $S$ and
        $f\in\surj{\carrier[R]}{\carrier[S]}$.
\end{definition}

\begin{proposition}\label{ring_embedding_is_hom}
    Suppose $f$ is a ring embedding from $R$ to $S$.
    Then $f$ is a ring homomorphism from $R$ to $S$.
\end{proposition}
\begin{proof}
    Follows by \obv{ring_embedding}.
\end{proof}

\begin{proposition}\label{ring_embedding_injective}
    Suppose $f$ is a ring embedding from $R$ to $S$.
    Then $f$ is injective.
\end{proposition}
\begin{proof}
    Follows by \obv{ring_embedding}.
\end{proof}

\begin{proposition}\label{ring_surjection_is_hom}
    Suppose $f$ is a ring surjection from $R$ to $S$.
    Then $f$ is a ring homomorphism from $R$ to $S$.
\end{proposition}
\begin{proof}
    Follows by \obv{ring_surjection}.
\end{proof}

\begin{proposition}\label{ring_surjection_is_surjection}
    Suppose $f$ is a ring surjection from $R$ to $S$.
    Then $f\in\surj{\carrier[R]}{\carrier[S]}$.
\end{proposition}
\begin{proof}
    Follows by \obv{ring_surjection}.
\end{proof}

\begin{proposition}\label{identity_ring_embedding}
    Let $R$ be a ring.
    Then $\identity{\carrier[R]}$ is a ring embedding from $R$ to $R$.
\end{proposition}
\begin{proof}
    $\identity{\carrier[R]}$ is a ring homomorphism from $R$ to $R$ by
        \obv{identity_ring_hom}.
    $\identity{\carrier[R]}$ is injective by
        \obv{identity_injective}.
    Follows by \obv{ring_embedding}.
\end{proof}

\begin{theorem}\label{ring_embedding_composition}
    Suppose $f$ is a ring embedding from $R$ to $S$ and
        $g$ is a ring embedding from $S$ to $T$.
    Then $g\circ f$ is a ring embedding from $R$ to $T$.
\end{theorem}
\begin{proof}
    $f$ is a ring homomorphism from $R$ to $S$ and
        $g$ is a ring homomorphism from $S$ to $T$ by
        \obv{ring_embedding_is_hom}.
    Hence $g\circ f$ is a ring homomorphism from $R$ to $T$ by
        \obv{ring_hom_composition}.
    $f$ and $g$ are injective by \obv{ring_embedding_injective}.
    Hence $g\circ f$ is injective by \obv{circ_injective}.
    Follows by \obv{ring_embedding}.
\end{proof}

\begin{theorem}\label{ring_surjection_composition}
    Suppose $f$ is a ring surjection from $R$ to $S$ and
        $g$ is a ring surjection from $S$ to $T$.
    Then $g\circ f$ is a ring surjection from $R$ to $T$.
\end{theorem}
\begin{proof}
    $f$ is a ring homomorphism from $R$ to $S$ and
        $g$ is a ring homomorphism from $S$ to $T$ by
        \obv{ring_surjection_is_hom}.
    Hence $g\circ f$ is a ring homomorphism from $R$ to $T$ by
        \obv{ring_hom_composition}.
    $f\in\surj{\carrier[R]}{\carrier[S]}$ and
        $g\in\surj{\carrier[S]}{\carrier[T]}$ by
        \obv{ring_surjection_is_surjection}.
    Hence $g\circ f\in\surj{\carrier[R]}{\carrier[T]}$ by
        \obv{surjections_circ}.
    Follows by \obv{ring_surjection}.
\end{proof}

\subsection{Inverse and opposite homomorphisms}

\begin{theorem}\label{bijective_ring_hom_converse}
    Suppose $f$ is a ring homomorphism from $R$ to $S$.
    Suppose $f\in\bijections{\carrier[R]}{\carrier[S]}$.
    Then $\converse{f}$ is a ring homomorphism from $S$ to $R$.
\end{theorem}
\begin{proof}
    Let $A=\AdditiveGroup{R}$ and $B=\AdditiveGroup{S}$.
    $R$ is a ring and $S$ is a ring by
        \obv{ring_hom_source_ring,ring_hom_target_ring}.
    $A$ is an exact commutative group and
        $B$ is an exact commutative group by
        \auto{additive_group_is_exact_commutative_group}.
    $\groupHom{f}{A}{B}$ by
        \obv{ring_hom_is_additive_group_hom}.
    Hence $f$ is a monoid homomorphism from $A$ to $B$ by
        \obv{group_hom_is_monoid_hom}.
    Hence $f$ is a magma homomorphism from $A$ to $B$ by
        \obv{monoid_hom_is_magma_hom}.
    $f\in\bijections{\carrier[A]}{\carrier[B]}$ by
        \auto{additive_group_fields}.
    Hence $f$ is a magma isomorphism from $A$ to $B$ by
        \obv{magma_isomorphism}.
    Hence $\converse{f}$ is a magma homomorphism from $B$ to $A$ by
        \obv{magma_isomorphism_converse_hom}.
    Hence $\groupHom{\converse{f}}{B}{A}$ by
        \obv{magma_hom,group_hom}.

    Let $M=\MultiplicativeMonoid{R}$ and
        $N=\MultiplicativeMonoid{S}$.
    $f$ is a monoid homomorphism from $M$ to $N$ by
        \obv{ring_hom_is_multiplicative_monoid_hom}.
    $f\in\bijections{\carrier[M]}{\carrier[N]}$ by
        \auto{multiplicative_monoid_fields}.
    Hence $f$ is a monoid isomorphism from $M$ to $N$ by
        \obv{monoid_isomorphism}.
    Hence $\converse{f}$ is a monoid homomorphism from $N$ to $M$ by
        \obv{monoid_isomorphism_converse_hom}.
    Follows by \auto{ring_hom}.
\end{proof}

\begin{theorem}\label{opposite_ring_additive_group}
    Let $R$ be a ring.
    Then $\AdditiveGroup{\OppositeRing{R}}=\AdditiveGroup{R}$.
\end{theorem}
\begin{proof}
    Let $S=\OppositeRing{R}$.
    $S$ is an exact ring by \auto{opposite_ring_is_exact}.
    $\AdditiveGroup{S}$ is an exact commutative group and
        $\AdditiveGroup{R}$ is an exact commutative group by
        \obv{additive_group_is_exact_commutative_group}.
    Hence $\AdditiveGroup{S}$ is an exact group by
        \obv{commutative_group}.
    $\AdditiveGroup{R}$ is an exact group by
        \obv{commutative_group}.
    $\carrier[\AdditiveGroup{S}]=\carrier[\AdditiveGroup{R}]$ and
        $\mul[\AdditiveGroup{S}]=\mul[\AdditiveGroup{R}]$ and
        $\neutral[\AdditiveGroup{S}]=\neutral[\AdditiveGroup{R}]$ and
        $\groupInv[\AdditiveGroup{S}]=
            \groupInv[\AdditiveGroup{R}]$ by
        \auto{additive_group_fields,opposite_ring_fields}.
    Follows by \auto{exact_group_extensionality}.
\end{proof}

\begin{theorem}\label{ring_hom_between_opposite_rings}
    Suppose $f$ is a ring homomorphism from $R$ to $S$.
    Then $f$ is a ring homomorphism from
        $\OppositeRing{R}$ to $\OppositeRing{S}$.
\end{theorem}
\begin{proof}
    Let $R'=\OppositeRing{R}$ and $S'=\OppositeRing{S}$.
    $R$ is a ring and $S$ is a ring by
        \obv{ring_hom_source_ring,ring_hom_target_ring}.
    $R'$ is an exact ring and $S'$ is an exact ring by
        \auto{opposite_ring_is_exact}.
    $\AdditiveGroup{R'}=\AdditiveGroup{R}$ and
        $\AdditiveGroup{S'}=\AdditiveGroup{S}$ by
        \auto{opposite_ring_additive_group}.
    Hence $\groupHom{f}{\AdditiveGroup{R'}}{\AdditiveGroup{S'}}$ by
        \obv{ring_hom_is_additive_group_hom}.

    Let $M=\MultiplicativeMonoid{R'}$ and
        $N=\MultiplicativeMonoid{S'}$.
    $M$ is a monoid and $N$ is a monoid by
        \obv{multiplicative_monoid_is_exact_monoid}.
    Hence $M$ is a magma and $N$ is a magma by
        \obv{monoid_is_magma}.
    $f\in\funs{\carrier[M]}{\carrier[N]}$ by
        \auto{ring_hom_type,multiplicative_monoid_fields,
            opposite_ring_fields}.
    Show that for all $x,y\in\carrier[M]$ we have
        $f(\mul[M](x,y))=\mul[N](f(x),f(y))$.
    \begin{subproof}
        Fix $x,y\in\carrier[M]$.
        Hence $x,y\in\carrier[R]$ by
            \auto{multiplicative_monoid_fields,opposite_ring_fields}.
        $\mul[M](x,y)=\ringMul[R](y,x)$ and
            $\mul[N](f(x),f(y))=\ringMul[S](f(y),f(x))$ by
            \auto{multiplicative_monoid_fields,
                opposite_ring_mul_apply,ring_hom_apply_type}.
        $f(\ringMul[R](y,x))=\ringMul[S](f(y),f(x))$ by
            \obv{ring_hom_mul}.
    \end{subproof}
    Hence $f$ is a magma homomorphism from $M$ to $N$ by
        \obv{magma_hom}.
    $f(\neutral[M])=\neutral[N]$ by
        \auto{multiplicative_monoid_fields,opposite_ring_fields,
            ring_hom_one}.
    Hence $f$ is a monoid homomorphism from $M$ to $N$ by
        \obv{monoid_hom}.
    Follows by \auto{ring_hom}.
\end{proof}
