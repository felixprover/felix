\section{Units of a ring}

The units of a ring are the units of its multiplicative monoid.  We keep the
generic unit carrier and exact unit group as the canonical terms rather than
introducing synonymous ring-specific constructions.

\begin{theorem}\label{ring_unit_iff}
    Let $R$ be a ring.
    Then $u$ is a unit of $\MultiplicativeMonoid{R}$ iff
        $u\in\carrier[R]$ and
        there exists $v\in\carrier[R]$ such that
            $\ringMul[R](u,v)=\ringOne[R]$ and
            $\ringMul[R](v,u)=\ringOne[R]$.
\end{theorem}
\begin{proof}
    $\MultiplicativeMonoid{R}$ is a monoid by
        \obv{multiplicative_monoid_is_exact_monoid}.
    Follows by
        \auto{monoid_unit,multiplicative_monoid_fields}.
\end{proof}

\begin{theorem}\label{ring_units_iff}
    Let $R$ be a ring.
    Then $u\in\monoidUnits{\MultiplicativeMonoid{R}}$ iff
        $u\in\carrier[R]$ and
        there exists $v\in\carrier[R]$ such that
            $\ringMul[R](u,v)=\ringOne[R]$ and
            $\ringMul[R](v,u)=\ringOne[R]$.
\end{theorem}
\begin{proof}
    Follows by
        \auto{monoid_units_iff,ring_unit_iff,
            multiplicative_monoid_fields}.
\end{proof}

\begin{proposition}\label{ring_units_subseteq_carrier}
    Let $R$ be a ring.
    Then $\monoidUnits{\MultiplicativeMonoid{R}}
        \subseteq\carrier[R]$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{monoid_units_subseteq_carrier,
            multiplicative_monoid_fields}.
\end{proof}

\begin{proposition}\label{ring_one_is_unit}
    Let $R$ be a ring.
    Then $\ringOne[R]$ is a unit of $\MultiplicativeMonoid{R}$.
\end{proposition}
\begin{proof}
    $\MultiplicativeMonoid{R}$ is a monoid by
        \obv{multiplicative_monoid_is_exact_monoid}.
    $\neutral[\MultiplicativeMonoid{R}]=\ringOne[R]$ by
        \obv{multiplicative_monoid_fields}.
    Follows by \auto{monoid_neutral_is_unit}.
\end{proof}

\begin{theorem}\label{ring_units_closed_under_multiplication}
    Let $R$ be a ring.
    Let $u,v\in\monoidUnits{\MultiplicativeMonoid{R}}$.
    Then $\ringMul[R](u,v)\in
        \monoidUnits{\MultiplicativeMonoid{R}}$.
\end{theorem}
\begin{proof}
    $\MultiplicativeMonoid{R}$ is a monoid by
        \obv{multiplicative_monoid_is_exact_monoid}.
    $\mul[\MultiplicativeMonoid{R}]=\ringMul[R]$ by
        \obv{multiplicative_monoid_fields}.
    Follows by \auto{monoid_units_closed_under_product}.
\end{proof}

\begin{theorem}\label{ring_unit_power}
    Let $R$ be a ring.
    Suppose $u$ is a unit of $\MultiplicativeMonoid{R}$.
    Let $n$ be a natural number.
    Then $\ringPower{R}{u}{n}$ is a unit of
        $\MultiplicativeMonoid{R}$.
\end{theorem}
\begin{proof}
    $\monoidPower{\MultiplicativeMonoid{R}}{u}{n}$ is a unit of
        $\MultiplicativeMonoid{R}$ by \obv{monoid_unit_power}.
    Follows by \obv{ring_power}.
\end{proof}

\begin{theorem}\label{ring_unit_group_is_exact_group}
    Let $R$ be a ring.
    Then $\MonoidUnitGroup{\MultiplicativeMonoid{R}}$
        is an exact group.
\end{theorem}
\begin{proof}
    $\MultiplicativeMonoid{R}$ is a monoid by
        \obv{multiplicative_monoid_is_exact_monoid}.
    Follows by \obv{monoid_unit_group_is_exact_group}.
\end{proof}

\begin{proposition}\label{ring_unit_group_carrier}
    Let $R$ be a ring.
    Then
        $\carrier[\MonoidUnitGroup{\MultiplicativeMonoid{R}}]
            =\monoidUnits{\MultiplicativeMonoid{R}}$.
\end{proposition}
\begin{proof}
    Follows by \obv{monoid_unit_group_fields}.
\end{proof}

\begin{proposition}\label{ring_unit_group_mul_apply}
    Let $R$ be a ring.
    Let $u,v\in\monoidUnits{\MultiplicativeMonoid{R}}$.
    Then
        $\mul[\MonoidUnitGroup{\MultiplicativeMonoid{R}}](u,v)
            =\ringMul[R](u,v)$.
\end{proposition}
\begin{proof}
    $\MultiplicativeMonoid{R}$ is a monoid by
        \obv{multiplicative_monoid_is_exact_monoid}.
    Follows by
        \auto{monoid_unit_group_mul_apply,
            multiplicative_monoid_fields}.
\end{proof}
