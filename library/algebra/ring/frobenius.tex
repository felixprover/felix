\section{Prime powers modulo ideals}

The binomial theorem first gives the elementary modular form of the
Freshman's dream.  The correction is kept as the finite sum of the interior
binomial terms; no coefficient or correction term is selected globally.

\begin{theorem}\label{freshmans_dream_mod_principal_ideal}
    Let $R$ be a commutative ring and $x,y\in\carrier[R]$.
    Let $p$ be a prime natural number.
    Let $P=\principalIdeal{R}{\ringNatural{R}{p}}$.
    Then
        $\ringPower{R}{\ringAdd[R](x,y)}{p}
            \mathrel{\idealCongruence{R}{P}}
        \ringAdd[R](\ringPower{R}{x}{p},\ringPower{R}{y}{p})$.
\end{theorem}
\begin{proof}
    Let $G=\AdditiveGroup{R}$ and
        $A=p\setminus\{\emptyset\}$.
    Let $f=\Apply{\graphOnOp}{
        \suc{p},\carrier[R],
        \fun{k:\Set}{
            \ringMul[R](
                \ringNatural{R}{\naturalChoose{p}{k}},
                \ringMul[R](
                    \ringPower{R}{x}{\naturalDifference{p}{k}},
                    \ringPower{R}{y}{k}))}}$.
    Let $g=\restrl{f}{A}$ and $s=\finiteSum{G}{A}{g}$.
    Let $u=\ringPower{R}{x}{p}$ and
        $v=\ringPower{R}{y}{p}$.

    $R$ is a ring by \obv{commutative_ring_is_ring}.
    $p$ is a natural number and $\naturalOne\in p$ by
        \obv{prime_natural}.
    Hence $p\neq\emptyset$ by \auto{emptyset}.
    $p$ is finite and $\suc{p}$ is finite by
        \obv{natural_number_is_finite,omega_successor}.
    $A\subseteq p$ by \auto{setminus_subseteq}.
    Hence $A$ is finite by \obv{subseteq_finite_is_finite}.
    $\ringNatural{R}{p}\in\carrier[R]$ by
        \obv{ring_natural_type}.
    $\principalIdeal{R}{\ringNatural{R}{p}}$
        is an ideal of $R$ by
        \obv{principal_ideal_is_ideal}.
    Hence $\principalIdeal{R}{\ringNatural{R}{p}}$
        is a two-sided ideal of $R$ by \obv{ideal}.
    Hence $P$ is a two-sided ideal of $R$.
    $G$ is a commutative group by
        \obv{additive_group_is_exact_commutative_group}.
    Hence $G$ is a monoid by
        \obv{commutative_group_is_group,group_is_monoid}.
    For all $a,b\in\carrier[G]$ we have
        $\mul[G](a,b)=\mul[G](b,a)$ by
        \obv{commutative_group_commutative}.

    $f\in\funs{\suc{p}}{\carrier[R]}$ by
        \auto{ring_binomial_summand_graph_type}.
    $p\subseteq\suc{p}$ by \obv{subseteq_self_suc_intro}.
    Hence $A\subseteq\suc{p}$ by \obv{subseteq_transitive}.
    Hence $g\in\funs{A}{\carrier[R]}$ by
        \auto{funs_restrl}.
    Show that for every $k\in A$ we have $g(k)\in P$.
    \begin{subproof}
        Fix $k\in A$.
        Hence $k\in p$ and $k\neq\emptyset$ by
            \auto{setminus,singleton_iff}.
        $k$ is a natural number by
            \auto{naturals_is_transitive_set,
                transitiveset_iff_subseteq,subseteq}.
        Hence $k$ is an ordinal and $\emptyset\in k$ by
            \obv{natural_number_is_ordinal,
                ordinal_empty_or_emptyset_elem}.
        $p$ divides $\naturalChoose{p}{k}$ as natural numbers by
            \obv{prime_divides_interior_natural_choose}.
        Hence $\ringNatural{R}{p}$ divides
            $\ringNatural{R}{\naturalChoose{p}{k}}$ in $R$ by
            \obv{commutative_ring_natural_divisibility}.
        $\naturalChoose{p}{k}$ is a natural number by
            \obv{natural_choose_type}.
        $\ringNatural{R}{p},
            \ringNatural{R}{\naturalChoose{p}{k}}\in\carrier[R]$ by
            \obv{ring_natural_type}.
        Hence $\ringNatural{R}{\naturalChoose{p}{k}}\in P$ by
            \obv{principal_ideal_membership_iff_divisibility}.
        $k\subseteq p$ by
            \obv{natural_number_is_ordinal,
                ordinal_elem_implies_subseteq}.
        $\naturalDifference{p}{k}$ is a natural number by
            \auto{natural_difference_specification}.
        $\ringPower{R}{x}{\naturalDifference{p}{k}},
            \ringPower{R}{y}{k}\in\carrier[R]$ by
            \obv{ring_power_type}.
        Hence
            $\ringMul[R](
                \ringPower{R}{x}{\naturalDifference{p}{k}},
                \ringPower{R}{y}{k})\in\carrier[R]$ by
            \obv{ring_mul_type}.
        Hence
            $\ringMul[R](
                \ringNatural{R}{\naturalChoose{p}{k}},
                \ringMul[R](
                    \ringPower{R}{x}{\naturalDifference{p}{k}},
                    \ringPower{R}{y}{k}))\in P$ by
            \auto{two_sided_ideal,right_ideal_expanded_iff}.
        $k\in\suc{p}$ by \obv{elem_subseteq}.
        $\apply{\Apply{\graphOnOp}{
            \suc{p},\carrier[R],
            \fun{j:\Set}{
                \ringMul[R](
                    \ringNatural{R}{\naturalChoose{p}{j}},
                    \ringMul[R](
                        \ringPower{R}{x}{\naturalDifference{p}{j}},
                        \ringPower{R}{y}{j}))}}}{k}=
            \ringMul[R](
                \ringNatural{R}{\naturalChoose{p}{k}},
                \ringMul[R](
                    \ringPower{R}{x}{\naturalDifference{p}{k}},
                    \ringPower{R}{y}{k}))$ by
            \auto{logical_graph_on_apply}.
        Hence $f(k)=
            \ringMul[R](
                \ringNatural{R}{\naturalChoose{p}{k}},
                \ringMul[R](
                    \ringPower{R}{x}{\naturalDifference{p}{k}},
                    \ringPower{R}{y}{k}))$.
        $f$ is a function and $\dom{f}=\suc{p}$ by
            \obv{funs_elim}.
        Hence $A\subseteq\dom{f}$.
        $\apply{\restrl{f}{A}}{k}=f(k)$ by
            \obv{restrl_of_function_apply}.
        Hence $g(k)=f(k)$.
    \end{subproof}
    $g$ is a function and $\dom{g}=A$ by
        \obv{funs_elim}.
    Hence $g$ is a function to $P$.
    Hence $g\in\funs{A}{P}$ by \obv{funs_intro}.
    Hence $s\in P$ by \auto{ideal_finite_sum}.

    $f\in\funs{\suc{p}}{\carrier[G]}$ by
        \auto{additive_group_fields}.
    Let $h=\restrl{f}{p}$.
    $h\in\funs{p}{\carrier[G]}$ by
        \auto{funs_restrl,subseteq_self_suc_intro}.
    $p\in\suc{p}$ and $p\notin p$ by
        \obv{suc_intro_self,in_irrefl}.
    $\suc{p}=\cons{p}{p}$ by \obv{suc}.
    Hence
        $\finiteSum{G}{\suc{p}}{f}
            =\ringAdd[R](\finiteSum{G}{p}{h},f(p))$ by
        \auto{finite_sum,monoid_finite_fold_fresh_adjoin,
            additive_group_fields}.

    $\emptyset\in p$ by
        \obv{natural_nonzero_contains_zero}.
    $\emptyset\notin A$ by
        \auto{setminus,singleton_iff}.
    $\cons{\emptyset}{A}=p$ by
        \auto{cons_remove}.
    $\restrl{\restrl{f}{p}}{A}=\restrl{f}{A}$ by
        \obv{restrl_restrl}.
    Hence $\restrl{h}{A}=g$.
    Hence
        $\finiteSum{G}{p}{h}
            =\ringAdd[R](s,h(\emptyset))$ by
        \auto{finite_sum,monoid_finite_fold_fresh_adjoin,
            additive_group_fields}.
    $\emptyset\in\suc{p}$ by \obv{elem_subseteq}.
    $\apply{\Apply{\graphOnOp}{
        \suc{p},\carrier[R],
        \fun{k:\Set}{
            \ringMul[R](
                \ringNatural{R}{\naturalChoose{p}{k}},
                \ringMul[R](
                    \ringPower{R}{x}{\naturalDifference{p}{k}},
                    \ringPower{R}{y}{k}))}}}{\emptyset}=
        \ringMul[R](
            \ringNatural{R}{\naturalChoose{p}{\emptyset}},
            \ringMul[R](
                \ringPower{R}{x}{\naturalDifference{p}{\emptyset}},
                \ringPower{R}{y}{\emptyset}))$ by
        \auto{logical_graph_on_apply}.
    $\ringMul[R](
        \ringNatural{R}{\naturalChoose{p}{\emptyset}},
        \ringMul[R](
            \ringPower{R}{x}{\naturalDifference{p}{\emptyset}},
            \ringPower{R}{y}{\emptyset}))=u$ by
        \obv{ring_binomial_first_term}.
    Hence $f(\emptyset)=u$.
    $f$ is a function and $\dom{f}=\suc{p}$ by
        \obv{funs_elim}.
    Hence $p\subseteq\dom{f}$.
    $\apply{\restrl{f}{p}}{\emptyset}=f(\emptyset)$ by
        \obv{restrl_of_function_apply}.
    Hence $h(\emptyset)=u$.

    $\apply{\Apply{\graphOnOp}{
        \suc{p},\carrier[R],
        \fun{k:\Set}{
            \ringMul[R](
                \ringNatural{R}{\naturalChoose{p}{k}},
                \ringMul[R](
                    \ringPower{R}{x}{\naturalDifference{p}{k}},
                    \ringPower{R}{y}{k}))}}}{p}=
        \ringMul[R](
            \ringNatural{R}{\naturalChoose{p}{p}},
            \ringMul[R](
                \ringPower{R}{x}{\naturalDifference{p}{p}},
                \ringPower{R}{y}{p}))$ by
        \auto{logical_graph_on_apply}.
    $\ringMul[R](
        \ringNatural{R}{\naturalChoose{p}{p}},
        \ringMul[R](
            \ringPower{R}{x}{\naturalDifference{p}{p}},
            \ringPower{R}{y}{p}))=v$ by
        \obv{ring_binomial_last_term}.
    Hence $f(p)=v$.
    Hence
        $\finiteSum{G}{\suc{p}}{f}
            =\ringAdd[R](\ringAdd[R](s,u),v)$.
    $\ringBinomialExpansion{R}{x}{y}{p}
        =\finiteSum{G}{\suc{p}}{f}$ by
        \obv{ring_binomial_expansion}.
    $\ringPower{R}{\ringAdd[R](x,y)}{p}
        =\ringBinomialExpansion{R}{x}{y}{p}$ by
        \obv{ring_binomial_theorem}.
    Hence $\ringPower{R}{\ringAdd[R](x,y)}{p}
        =\ringAdd[R](\ringAdd[R](s,u),v)$.

    $s,u,v\in\carrier[R]$ by
        \auto{two_sided_ideal_subseteq_carrier,subseteq,
            ring_power_type}.
    $\ringAdd[R](u,v)\in\carrier[R]$ by \obv{ring_add_type}.
    $\ringAdd[R](
        \ringPower{R}{\ringAdd[R](x,y)}{p},
        \ringNeg[R](\ringAdd[R](u,v)))=s$ by
        \auto{ring_add_associative,ring_add_commutative,
            ring_add_inverse_right,ring_add_zero_right,
            ring_add_type,ring_neg_type}.
    Hence
        $\ringAdd[R](
            \ringPower{R}{\ringAdd[R](x,y)}{p},
            \ringNeg[R](\ringAdd[R](u,v)))\in P$.
    $\ringAdd[R](x,y)\in\carrier[R]$ by \obv{ring_add_type}.
    Hence $\ringPower{R}{\ringAdd[R](x,y)}{p}\in\carrier[R]$ by
        \obv{ring_power_type}.
    Hence
        $\ringPower{R}{\ringAdd[R](x,y)}{p}
            \mathrel{\idealCongruence{R}{P}}
        \ringAdd[R](u,v)$ by
        \obv{ideal_congruence_iff_add_neg}.
\end{proof}

\begin{corollary}\label{freshmans_dream_mod_ideal}
    Let $R$ be a commutative ring and $x,y\in\carrier[R]$.
    Let $p$ be a prime natural number.
    Suppose $J$ is a two-sided ideal of $R$ and
        $\ringNatural{R}{p}\in J$.
    Then
        $\ringPower{R}{\ringAdd[R](x,y)}{p}
            \mathrel{\idealCongruence{R}{J}}
        \ringAdd[R](\ringPower{R}{x}{p},\ringPower{R}{y}{p})$.
\end{corollary}
\begin{proof}
    Let $P=\principalIdeal{R}{\ringNatural{R}{p}}$.
    $R$ is a ring by \obv{commutative_ring_is_ring}.
    $p$ is a natural number by \obv{prime_natural_guard}.
    $\ringNatural{R}{p}\in\carrier[R]$ by
        \obv{ring_natural_type}.
    $\principalIdeal{R}{\ringNatural{R}{p}}$ is an ideal of $R$ by
        \obv{principal_ideal_is_ideal}.
    Hence $\principalIdeal{R}{\ringNatural{R}{p}}$
        is a two-sided ideal of $R$ by \obv{ideal}.
    Hence $P$ is a two-sided ideal of $R$.
    $J$ is an ideal of $R$ by \obv{ideal}.
    $\principalIdeal{R}{\ringNatural{R}{p}}\subseteq J$ by
        \obv{principal_ideal_universal_property}.
    Hence $P\subseteq J$.
    Let $a=\ringPower{R}{\ringAdd[R](x,y)}{p}$ and
        $b=\ringAdd[R](\ringPower{R}{x}{p},\ringPower{R}{y}{p})$.
    $a,b\in\carrier[R]$ by
        \auto{ring_add_type,ring_power_type}.
    $\ringPower{R}{\ringAdd[R](x,y)}{p}
        \mathrel{\idealCongruence{R}{
            \principalIdeal{R}{\ringNatural{R}{p}}}}
        \ringAdd[R](\ringPower{R}{x}{p},\ringPower{R}{y}{p})$ by
        \obv{freshmans_dream_mod_principal_ideal}.
    Hence $a\mathrel{\idealCongruence{R}{P}}b$.
    Hence $\ringAdd[R](a,\ringNeg[R](b))\in P$ by
        \obv{ideal_congruence_iff_add_neg}.
    Hence $\ringAdd[R](a,\ringNeg[R](b))\in J$ by
        \obv{elem_subseteq}.
    Follows by \obv{ideal_congruence_iff_add_neg}.
\end{proof}

\begin{theorem}\label{prime_power_ideal_congruence}
    Let $R$ be a commutative ring.
    Let $p$ be a prime natural number.
    Suppose $I$ is a two-sided ideal of $R$ and
        $J$ is a two-sided ideal of $R$.
    Suppose $\ringNatural{R}{p}\in J$.
    Suppose for every $d\in I$ we have
        $\ringPower{R}{d}{p}\in J$.
    Let $x,y\in\carrier[R]$.
    Suppose $x\mathrel{\idealCongruence{R}{I}}y$.
    Then
        $\ringPower{R}{x}{p}
            \mathrel{\idealCongruence{R}{J}}
        \ringPower{R}{y}{p}$.
\end{theorem}
\begin{proof}
    Let $d=\ringAdd[R](\ringNeg[R](x),y)$.
    $R$ is a ring by \obv{commutative_ring_is_ring}.
    $p$ is a natural number by \obv{prime_natural_guard}.
    $d\in I$ by \auto{ideal_congruence_iff_neg_add}.
    Hence $d\in\carrier[R]$ by
        \auto{two_sided_ideal_subseteq_carrier,subseteq}.
    $\ringAdd[R](x,d)=y$ by
        \auto{ring_add_associative,ring_add_inverse_right,
            ring_add_zero_left,ring_add_commutative,ring_neg_type}.
    $\ringPower{R}{\ringAdd[R](x,d)}{p}
        \mathrel{\idealCongruence{R}{J}}
        \ringAdd[R](\ringPower{R}{x}{p},\ringPower{R}{d}{p})$ by
        \obv{freshmans_dream_mod_ideal}.
    Hence $\ringPower{R}{y}{p}
        \mathrel{\idealCongruence{R}{J}}
        \ringAdd[R](\ringPower{R}{x}{p},\ringPower{R}{d}{p})$.
    $\ringPower{R}{d}{p}\in J$.
    Hence $\ringPower{R}{d}{p}\in\carrier[R]$ by
        \auto{two_sided_ideal_subseteq_carrier,subseteq}.
    $\ringNeg[R](\ringZero[R])=\ringZero[R]$ by
        \obv{ring_neg_zero}.
    Hence $\ringAdd[R](\ringNeg[R](\ringZero[R]),
        \ringPower{R}{d}{p})=\ringPower{R}{d}{p}$ by
        \auto{ring_add_zero_left}.
    Hence $\ringAdd[R](\ringNeg[R](\ringZero[R]),
        \ringPower{R}{d}{p})\in J$.
    $\ringZero[R]\mathrel{\idealCongruence{R}{J}}
        \ringPower{R}{d}{p}$ iff
        $\ringAdd[R](\ringNeg[R](\ringZero[R]),
            \ringPower{R}{d}{p})\in J$ by
        \obv{ideal_congruence_iff_neg_add,
            ring_zero_type,ring_power_type}.
    Hence
        $\ringZero[R]\mathrel{\idealCongruence{R}{J}}
            \ringPower{R}{d}{p}$.
    $\ringPower{R}{x}{p}
        \mathrel{\idealCongruence{R}{J}}
        \ringPower{R}{x}{p}$ by
        \obv{ideal_congruence_reflexive,ring_power_type}.
    Hence
        $\ringAdd[R](\ringPower{R}{x}{p},\ringZero[R])
            \mathrel{\idealCongruence{R}{J}}
        \ringAdd[R](\ringPower{R}{x}{p},\ringPower{R}{d}{p})$ by
        \obv{ideal_congruence_addition_congruent,
            ring_power_type,ring_zero_type}.
    $\ringAdd[R](\ringPower{R}{x}{p},\ringZero[R])
        =\ringPower{R}{x}{p}$ by
        \obv{ring_add_zero_right,ring_power_type}.
    Hence
        $\ringPower{R}{x}{p}
            \mathrel{\idealCongruence{R}{J}}
        \ringAdd[R](\ringPower{R}{x}{p},\ringPower{R}{d}{p})$.
    Hence
        $\ringAdd[R](\ringPower{R}{x}{p},\ringPower{R}{d}{p})
            \mathrel{\idealCongruence{R}{J}}
        \ringPower{R}{x}{p}$ by
        \obv{ideal_congruence_symmetric,ring_add_type,
            ring_power_type}.
    Hence $\ringPower{R}{y}{p}
        \mathrel{\idealCongruence{R}{J}}
        \ringPower{R}{x}{p}$ by
        \auto{ideal_congruence_transitive,ring_add_type,
            ring_power_type}.
    Follows by
        \auto{ideal_congruence_symmetric,ring_power_type}.
\end{proof}

\section{Frobenius between quotient rings}

The representative map raises an element to the prime power and then applies
the existing quotient projection.  Its graph records the source ring, the
exponent, and the target ideal, but no quotient of the source.

\begin{definition}\label{quotient_frobenius_representative}
    $\quotientFrobeniusRepresentative{R}{p}{J}=
        \Apply{\graphOnOp}{
            \carrier[R],\carrier[\QuotientRing{R}{J}],
            \fun{x:\Set}{
                \apply{
                    \groupQuotientProjection{\AdditiveGroup{R}}{J}}{
                    \ringPower{R}{x}{p}}}}$.
\end{definition}

\begin{proposition}\label{quotient_frobenius_representative_type}
    Let $R$ be a ring and $p$ be a natural number.
    Suppose $J$ is a two-sided ideal of $R$.
    Then $\quotientFrobeniusRepresentative{R}{p}{J}\in
        \funs{\carrier[R]}{\carrier[\QuotientRing{R}{J}]}$.
\end{proposition}
\begin{proof}
    Let $Q=\QuotientRing{R}{J}$ and
        $q=\groupQuotientProjection{\AdditiveGroup{R}}{J}$.
    $q$ is a ring homomorphism from $R$ to $Q$ by
        \auto{quotient_ring_projection_hom}.
    Show that for every $x\in\carrier[R]$ we have
        $q(\ringPower{R}{x}{p})\in\carrier[Q]$.
    \begin{subproof}
        Fix $x\in\carrier[R]$.
        $\ringPower{R}{x}{p}\in\carrier[R]$ by
            \obv{ring_power_type}.
        Follows by \auto{ring_hom_apply_type}.
    \end{subproof}
    $\Apply{\graphOnOp}{
        \carrier[R],\carrier[Q],
        \fun{x:\Set}{q(\ringPower{R}{x}{p})}}
        \in\funs{\carrier[R]}{\carrier[Q]}$ by
        \auto{logical_graph_on_function}.
    Follows by \auto{quotient_frobenius_representative}.
\end{proof}

\begin{proposition}\label{quotient_frobenius_representative_apply}
    Let $R$ be a ring and $p$ be a natural number.
    Suppose $J$ is a two-sided ideal of $R$.
    Let $x\in\carrier[R]$.
    Let $q=\groupQuotientProjection{\AdditiveGroup{R}}{J}$.
    Then
        $\apply{\quotientFrobeniusRepresentative{R}{p}{J}}{x}
            =q(\ringPower{R}{x}{p})$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{quotient_frobenius_representative,
            logical_graph_on_apply}.
\end{proof}

\begin{theorem}\label{quotient_frobenius_representative_hom}
    Let $R$ be a commutative ring and $p$ be a prime natural number.
    Suppose $J$ is a two-sided ideal of $R$ and
        $\ringNatural{R}{p}\in J$.
    Then $\quotientFrobeniusRepresentative{R}{p}{J}$
        is a ring homomorphism from $R$ to $\QuotientRing{R}{J}$.
\end{theorem}
\begin{proof}
    Let $Q=\QuotientRing{R}{J}$ and
        $q=\groupQuotientProjection{\AdditiveGroup{R}}{J}$.
    Let $f=\quotientFrobeniusRepresentative{R}{p}{J}$.
    Let $G=\AdditiveGroup{R}$ and $H=\AdditiveGroup{Q}$.
    $R$ is a ring by \obv{commutative_ring_is_ring}.
    $p$ is a natural number by \obv{prime_natural_guard}.
    $Q$ is a commutative ring by
        \auto{quotient_ring_is_commutative}.
    Hence $Q$ is a ring by \obv{commutative_ring_is_ring}.
    $q$ is a ring homomorphism from $R$ to $Q$ by
        \auto{quotient_ring_projection_hom}.
    $f\in\funs{\carrier[R]}{\carrier[Q]}$ by
        \auto{quotient_frobenius_representative_type}.
    $G$ is a group and $H$ is a group by
        \auto{additive_group_is_exact_commutative_group,
            commutative_group_is_group}.
    $f\in\funs{\carrier[G]}{\carrier[H]}$ by
        \auto{additive_group_fields}.
    Show that for all $x,y\in\carrier[G]$ we have
        $f(\mul[G](x,y))=\mul[H](f(x),f(y))$.
    \begin{subproof}
        Fix $x,y\in\carrier[G]$.
        Hence $x,y\in\carrier[R]$ by
            \auto{additive_group_fields}.
        $\ringAdd[R](x,y)\in\carrier[R]$ by \obv{ring_add_type}.
        $\ringPower{R}{x}{p},\ringPower{R}{y}{p}\in\carrier[R]$ by
            \obv{ring_power_type}.
        $\ringPower{R}{\ringAdd[R](x,y)}{p}
            \mathrel{\idealCongruence{R}{J}}
            \ringAdd[R](\ringPower{R}{x}{p},\ringPower{R}{y}{p})$ by
            \auto{freshmans_dream_mod_ideal}.
        Hence
            $q(\ringPower{R}{\ringAdd[R](x,y)}{p})
                =q(\ringAdd[R](
                    \ringPower{R}{x}{p},\ringPower{R}{y}{p}))$ by
            \auto{quotient_ring_projection_eq_iff_ideal_congruence,
                ring_power_type,ring_add_type}.
        $q(\ringAdd[R](\ringPower{R}{x}{p},\ringPower{R}{y}{p}))
            =\ringAdd[Q](q(\ringPower{R}{x}{p}),
                q(\ringPower{R}{y}{p}))$ by
            \auto{ring_hom_add}.
        $f(\ringAdd[R](x,y))
            =q(\ringPower{R}{\ringAdd[R](x,y)}{p})$ and
            $f(x)=q(\ringPower{R}{x}{p})$ and
            $f(y)=q(\ringPower{R}{y}{p})$ by
            \auto{quotient_frobenius_representative_apply}.
        Follows by \auto{additive_group_fields}.
    \end{subproof}
    Hence $\groupHom{f}{G}{H}$ by \auto{group_hom}.

    Let $M=\MultiplicativeMonoid{R}$ and
        $N=\MultiplicativeMonoid{Q}$.
    $M$ is a monoid and $N$ is a monoid by
        \auto{multiplicative_monoid_is_exact_monoid}.
    $f\in\funs{\carrier[M]}{\carrier[N]}$ by
        \auto{quotient_frobenius_representative_type,
            multiplicative_monoid_fields}.
    Show that for all $x,y\in\carrier[M]$ we have
        $f(\mul[M](x,y))=\mul[N](f(x),f(y))$.
    \begin{subproof}
        Fix $x,y\in\carrier[M]$.
        Hence $x,y\in\carrier[R]$ by
            \auto{multiplicative_monoid_fields}.
        $\ringMul[R](x,y)\in\carrier[R]$ by \obv{ring_mul_type}.
        $\ringPower{R}{\ringMul[R](x,y)}{p}
            =\ringMul[R](\ringPower{R}{x}{p},
                \ringPower{R}{y}{p})$ by
            \auto{commutative_ring_power_of_product}.
        $q(\ringMul[R](\ringPower{R}{x}{p},
                \ringPower{R}{y}{p}))
            =\ringMul[Q](q(\ringPower{R}{x}{p}),
                q(\ringPower{R}{y}{p}))$ by
            \auto{ring_hom_mul,ring_power_type}.
        $f(\ringMul[R](x,y))
            =q(\ringPower{R}{\ringMul[R](x,y)}{p})$ and
            $f(x)=q(\ringPower{R}{x}{p})$ and
            $f(y)=q(\ringPower{R}{y}{p})$ by
            \auto{quotient_frobenius_representative_apply}.
        Follows by \auto{multiplicative_monoid_fields}.
    \end{subproof}
    Hence $f$ is a magma homomorphism from $M$ to $N$ by

\end{proof}

\begin{theorem}\label{quotient_frobenius_representative_kernel}

        $q=\groupQuotientProjection{\AdditiveGroup{R}}{J}$.

    $p$ is a natural number by \obv{prime_natural_guard}.
    $f$ is a ring homomorphism from $R$ to $Q$ by
        \auto{quotient_frobenius_representative_hom}.
    $q$ is a ring homomorphism from $R$ to $Q$ and
        $\ringKernel{q}{R}=J$ by
        \auto{quotient_ring_projection_hom,
            quotient_ring_projection_kernel}.
    Show that every element of $\ringKernel{f}{R}$ is an element of $K$.
    \begin{subproof}
        Fix $x\in\ringKernel{f}{R}$.
        Hence $x\in\carrier[R]$ and $f(x)=\ringZero[Q]$ by
            \auto{ring_kernel_iff}.
        $f(x)=q(\ringPower{R}{x}{p})$ by
            \auto{quotient_frobenius_representative_apply}.
        $\ringPower{R}{x}{p}\in\carrier[R]$ by
            \obv{ring_power_type}.
        Hence $\ringPower{R}{x}{p}\in\ringKernel{q}{R}$ by
            \auto{ring_kernel_iff}.
        Hence $\ringPower{R}{x}{p}\in J$.
        Follows by \auto.
    \end{subproof}
    Show that every element of $K$ is an element of $\ringKernel{f}{R}$.
    \begin{subproof}
        Fix $x\in K$.
        Hence $x\in\carrier[R]$ and $\ringPower{R}{x}{p}\in J$.
        Hence $\ringPower{R}{x}{p}\in\ringKernel{q}{R}$.
        Hence $q(\ringPower{R}{x}{p})=\ringZero[Q]$ by
            \auto{ring_kernel_iff}.
        $f(x)=q(\ringPower{R}{x}{p})$ by
            \auto{quotient_frobenius_representative_apply}.
        Follows by \auto{ring_kernel_iff}.
    \end{subproof}
    $\ringKernel{f}{R}=K$ by \auto{setext}.
    Follows by \auto.
\end{proof}

\begin{theorem}\label{quotient_frobenius_injective_iff_congruence_reflection}
    Let $R$ be a commutative ring.
    Let $p$ be a prime natural number.
    Suppose $I$ is a two-sided ideal of $R$ and
        $J$ is a two-sided ideal of $R$.
    Suppose $\ringNatural{R}{p}\in J$.
    Suppose for every $d\in I$ we have
        $\ringPower{R}{d}{p}\in J$.
    Then $\quotientFrobenius{R}{p}{I}{J}$ is injective iff
        for all $x,y\in\carrier[R]$ such that
            $\ringPower{R}{x}{p}
                \mathrel{\idealCongruence{R}{J}}
            \ringPower{R}{y}{p}$ we have
                $x\mathrel{\idealCongruence{R}{I}}y$.
\end{theorem}
\begin{proof}
    Let $A=\QuotientRing{R}{I}$ and $B=\QuotientRing{R}{J}$.
    Let $F=\quotientFrobenius{R}{p}{I}{J}$.
    Let $q_I=\groupQuotientProjection{\AdditiveGroup{R}}{I}$ and
        $q_J=\groupQuotientProjection{\AdditiveGroup{R}}{J}$.
    $R$ is a ring and $p$ is a natural number by
        \obv{commutative_ring_is_ring,prime_natural_guard}.
    $F\in\funs{\carrier[A]}{\carrier[B]}$ by
        \auto{quotient_frobenius_type}.
    Hence $F$ is a function and $\dom{F}=\carrier[A]$ by
        \auto{funs_is_function,funs_elim}.
    $q_I\in\surj{\carrier[R]}{\carrier[A]}$ and
        $q_J\in\surj{\carrier[R]}{\carrier[B]}$ by
        \auto{quotient_ring_projection_surjective,
            ring_surjection_is_surjection}.

    Show that if $F$ is injective, then for all
        $x,y\in\carrier[R]$ such that
            $\ringPower{R}{x}{p}
                \mathrel{\idealCongruence{R}{J}}
            \ringPower{R}{y}{p}$ we have
                $x\mathrel{\idealCongruence{R}{I}}y$.
    \begin{subproof}
        Assume $F$ is injective.
        Fix $x,y\in\carrier[R]$.
        Assume
            $\ringPower{R}{x}{p}
                \mathrel{\idealCongruence{R}{J}}
            \ringPower{R}{y}{p}$.
        $\ringPower{R}{x}{p},\ringPower{R}{y}{p}\in\carrier[R]$ by
            \auto{ring_power_type}.
        Hence
            $q_J(\ringPower{R}{x}{p})
                =q_J(\ringPower{R}{y}{p})$ by
            \auto{quotient_ring_projection_eq_iff_ideal_congruence}.
        $F(q_I(x))=q_J(\ringPower{R}{x}{p})$ and
            $F(q_I(y))=q_J(\ringPower{R}{y}{p})$ by
            \auto{quotient_frobenius_apply}.
        Hence $F(q_I(x))=F(q_I(y))$.
        $q_I(x),q_I(y)\in\carrier[A]$ by
            \auto{surj,funs_type_apply}.
        Hence $q_I(x)=q_I(y)$ by \auto{injective_function}.
        Follows by
            \auto{quotient_ring_projection_eq_iff_ideal_congruence}.
    \end{subproof}

    Show that if for all $x,y\in\carrier[R]$ such that
            $\ringPower{R}{x}{p}
                \mathrel{\idealCongruence{R}{J}}
            \ringPower{R}{y}{p}$ we have
                $x\mathrel{\idealCongruence{R}{I}}y$,
        then $F$ is injective.
    \begin{subproof}
        Assume for all $x,y\in\carrier[R]$ such that
                $\ringPower{R}{x}{p}
                    \mathrel{\idealCongruence{R}{J}}
                \ringPower{R}{y}{p}$ we have
                    $x\mathrel{\idealCongruence{R}{I}}y$.
        Show that for all $U,V\in\dom{F}$ we have
            $F(U)=F(V)\implies U=V$.
        \begin{subproof}
            Fix $U,V\in\dom{F}$.
            Assume $F(U)=F(V)$.
            Hence $U,V\in\carrier[A]$.
            Take $x\in\carrier[R]$ such that $q_I(x)=U$ by
                \auto{surj}.
            Take $y\in\carrier[R]$ such that $q_I(y)=V$ by
                \auto{surj}.
            $F(q_I(x))=q_J(\ringPower{R}{x}{p})$ and
                $F(q_I(y))=q_J(\ringPower{R}{y}{p})$ by
                \auto{quotient_frobenius_apply}.
            Hence
                $q_J(\ringPower{R}{x}{p})
                    =q_J(\ringPower{R}{y}{p})$.
            $\ringPower{R}{x}{p},\ringPower{R}{y}{p}\in\carrier[R]$ by
                \auto{ring_power_type}.
            Hence
                $\ringPower{R}{x}{p}
                    \mathrel{\idealCongruence{R}{J}}
                \ringPower{R}{y}{p}$ by
                \auto{quotient_ring_projection_eq_iff_ideal_congruence}.
            Hence $x\mathrel{\idealCongruence{R}{I}}y$ by
                \auto.
            Hence $q_I(x)=q_I(y)$ by
                \auto{quotient_ring_projection_eq_iff_ideal_congruence}.
            Follows by \auto.
        \end{subproof}
        Follows by \auto{injective_function}.
    \end{subproof}
    Follows by \auto.
\end{proof}

                $x\mathrel{\idealCongruence{R}{I}}y$) and
        (for every $y\in\carrier[R]$ there exists $x\in\carrier[R]$
            such that
            $y\mathrel{\idealCongruence{R}{J}}
                \ringPower{R}{x}{p}$).
\end{theorem}
\begin{proof}
    Let $A=\QuotientRing{R}{I}$ and $B=\QuotientRing{R}{J}$.
    Let $F=\quotientFrobenius{R}{p}{I}{J}$.
    $F\in\bijections{\carrier[A]}{\carrier[B]}$ iff
        $F$ is injective and
        $F\in\surj{\carrier[A]}{\carrier[B]}$ by
        \auto{bijections}.
    $F$ is injective iff
        for all $x,y\in\carrier[R]$ such that
            $\ringPower{R}{x}{p}
                \mathrel{\idealCongruence{R}{J}}
            \ringPower{R}{y}{p}$ we have
                $x\mathrel{\idealCongruence{R}{I}}y$ by
        \auto{quotient_frobenius_injective_iff_congruence_reflection}.
    $F\in\surj{\carrier[A]}{\carrier[B]}$ iff
        for every $y\in\carrier[R]$ there exists $x\in\carrier[R]$
            such that
            $y\mathrel{\idealCongruence{R}{J}}
                \ringPower{R}{x}{p}$ by
        \auto{quotient_frobenius_surjective_iff}.
    Follows by \auto.
\end{proof}

\begin{theorem}\label{quotient_frobenius_isomorphism_iff_bijective}
    Let $R$ be a commutative ring.
    Let $p$ be a prime natural number.
    Suppose $I$ is a two-sided ideal of $R$ and
        $J$ is a two-sided ideal of $R$.
    Suppose $\ringNatural{R}{p}\in J$.
    Suppose for every $d\in I$ we have
        $\ringPower{R}{d}{p}\in J$.
    Then $\quotientFrobenius{R}{p}{I}{J}$ is a ring isomorphism from
        $\QuotientRing{R}{I}$ to $\QuotientRing{R}{J}$ iff
        $\quotientFrobenius{R}{p}{I}{J}\in
            \bijections{\carrier[\QuotientRing{R}{I}]}{
                \carrier[\QuotientRing{R}{J}]}$.
\end{theorem}
\begin{proof}
    $\quotientFrobenius{R}{p}{I}{J}$ is a ring homomorphism from
        $\QuotientRing{R}{I}$ to $\QuotientRing{R}{J}$ by
        \auto{quotient_frobenius_hom}.
    Follows by \auto{ring_isomorphism}.
\end{proof}


\section{The principal quotient Frobenius}

For a principal source ideal, principal power containment supplies the
descent condition.  The remaining hypothesis says exactly that the target
ideal contains the image of the prime.

\begin{theorem}\label{principal_frobenius_compatibility}
    Let $R$ be a commutative ring and $p$ be a prime natural number.
    Let $a\in\carrier[R]$.
    Suppose $\ringPower{R}{a}{p}$ divides $\ringNatural{R}{p}$ in $R$.
    Then
        $\ringNatural{R}{p}
            \in\principalIdeal{R}{\ringPower{R}{a}{p}}$ and
        for every $d\in\principalIdeal{R}{a}$ we have
            $\ringPower{R}{d}{p}
                \in\principalIdeal{R}{\ringPower{R}{a}{p}}$.
\end{theorem}
\begin{proof}
    $R$ is a ring and $p$ is a natural number by
        \obv{commutative_ring_is_ring,prime_natural_guard}.
    $\ringPower{R}{a}{p},\ringNatural{R}{p}\in\carrier[R]$ by
        \auto{ring_power_type,ring_natural_type}.
    Hence
        $\ringNatural{R}{p}
            \in\principalIdeal{R}{\ringPower{R}{a}{p}}$ by
        \auto{principal_ideal_membership_iff_divisibility}.
    Show that for every $d\in\principalIdeal{R}{a}$ we have
        $\ringPower{R}{d}{p}
            \in\principalIdeal{R}{\ringPower{R}{a}{p}}$.
    \begin{subproof}
        Fix $d\in\principalIdeal{R}{a}$.
        $\principalIdeal{R}{a}$ is an ideal of $R$ by
            \obv{principal_ideal_is_ideal}.
        Hence $\principalIdeal{R}{a}$ is a two-sided ideal of $R$ by
            \auto{ideal}.
        Hence $d\in\carrier[R]$ by
            \auto{two_sided_ideal_subseteq_carrier,subseteq}.
        Follows by \auto{principal_ideal_power_containment}.
    \end{subproof}
\end{proof}

\begin{definition}\label{principal_quotient_frobenius}
    Let $R$ be a ring.
    $\principalQuotientFrobenius{R}{p}{a}=
        \quotientFrobenius{R}{p}{\principalIdeal{R}{a}}{
            \principalIdeal{R}{\ringPower{R}{a}{p}}}$.
\end{definition}

\begin{theorem}\label{principal_quotient_frobenius_injective_iff}
    Let $R$ be a commutative ring and $p$ be a prime natural number.
    Let $a\in\carrier[R]$.
    Suppose $\ringPower{R}{a}{p}$ divides $\ringNatural{R}{p}$ in $R$.
    Then $\principalQuotientFrobenius{R}{p}{a}$ is injective iff
        for every $x\in\carrier[R]$ such that
            $\ringPower{R}{x}{p}
                \in\principalIdeal{R}{\ringPower{R}{a}{p}}$
        we have $x\in\principalIdeal{R}{a}$.
\end{theorem}
\begin{proof}
    Let $I=\principalIdeal{R}{a}$ and
        $J=\principalIdeal{R}{\ringPower{R}{a}{p}}$.
    Let $K=\{x\in\carrier[R]\mid\ringPower{R}{x}{p}\in J\}$ and
        $F=\principalQuotientFrobenius{R}{p}{a}$.
    $R$ is a ring and $p$ is a natural number by
        \obv{commutative_ring_is_ring,prime_natural_guard}.
    $\ringPower{R}{a}{p}\in\carrier[R]$ by \obv{ring_power_type}.
    $\principalIdeal{R}{a}$ and
        $\principalIdeal{R}{\ringPower{R}{a}{p}}$
        are ideals of $R$ by \auto{principal_ideal_is_ideal}.
    Hence $\principalIdeal{R}{a}$ and
        $\principalIdeal{R}{\ringPower{R}{a}{p}}$
        are two-sided ideals of $R$ by \auto{ideal}.
    Hence $I$ and $J$ are two-sided ideals of $R$ by \auto.
    $\ringNatural{R}{p}
            \in\principalIdeal{R}{\ringPower{R}{a}{p}}$ and
        for every $d\in\principalIdeal{R}{a}$ we have
            $\ringPower{R}{d}{p}
                \in\principalIdeal{R}{\ringPower{R}{a}{p}}$ by
        \auto{principal_frobenius_compatibility}.
    Hence $\ringNatural{R}{p}\in J$ and
        for every $d\in I$ we have $\ringPower{R}{d}{p}\in J$ by
        \auto.

    $I\subseteq\carrier[R]$ by
        \obv{two_sided_ideal_subseteq_carrier}.
    Show that $I\subseteq K$.
    \begin{subproof}
        Show that every element of $I$ is an element of $K$.
        \begin{subproof}
            Fix $d\in I$.
            Hence $d\in\carrier[R]$ by \obv{elem_subseteq}.
            $\ringPower{R}{d}{p}\in J$ by \auto.
            Follows by \auto.
        \end{subproof}
        Follows by \obv{subseteq}.
    \end{subproof}

    $\quotientFrobenius{R}{p}{I}{J}$ is injective iff $K=I$ by
        \auto{quotient_frobenius_injective_iff_kernel}.
    Hence $F$ is injective iff $K=I$ by
        \auto{principal_quotient_frobenius}.

    Show that if $K=I$, then for every $x\in\carrier[R]$ such that
        $\ringPower{R}{x}{p}\in J$ we have $x\in I$.
    \begin{subproof}
        Assume $K=I$.
        Fix $x\in\carrier[R]$.
        Assume $\ringPower{R}{x}{p}\in J$.
        Hence $x\in K$ by \auto.
        Follows by \auto.
    \end{subproof}
    Show that if for every $x\in\carrier[R]$ such that
            $\ringPower{R}{x}{p}\in J$ we have $x\in I$,
        then $K=I$.
    \begin{subproof}
        Assume for every $x\in\carrier[R]$ such that
            $\ringPower{R}{x}{p}\in J$ we have $x\in I$.
        Show that $K\subseteq I$.
        \begin{subproof}
            Show that every element of $K$ is an element of $I$.
            \begin{subproof}
                Fix $x\in K$.
                Hence $x\in\carrier[R]$ and
                    $\ringPower{R}{x}{p}\in J$ by \auto.
                Follows by \auto.
            \end{subproof}
            Follows by \obv{subseteq}.
        \end{subproof}
        Follows by \obv{subseteq_antisymmetric}.
    \end{subproof}
    Follows by \auto.
\end{proof}
