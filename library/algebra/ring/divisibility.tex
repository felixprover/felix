\section{Sided divisibility in rings}

The sided ring predicates are thin views of divisibility in the exact
multiplicative monoid.  They introduce no second cofactor notion.

\begin{definition}\label{left_ring_divisibility}
    $a$ left-divides $b$ in the ring $R$ iff
        $R$ is a ring and $a,b\in\carrier[R]$ and
        $a$ left-divides $b$ in $\MultiplicativeMonoid{R}$.
\end{definition}

\begin{definition}\label{right_ring_divisibility}
    $a$ right-divides $b$ in the ring $R$ iff
        $R$ is a ring and $a,b\in\carrier[R]$ and
        $a$ right-divides $b$ in $\MultiplicativeMonoid{R}$.
\end{definition}

\begin{proposition}\label{left_ring_divisibility_guards}
    Suppose $a$ left-divides $b$ in the ring $R$.
    Then $R$ is a ring and $a,b\in\carrier[R]$.
\end{proposition}
\begin{proof}
    Follows by \obv{left_ring_divisibility}.
\end{proof}

\begin{proposition}\label{right_ring_divisibility_guards}
    Suppose $a$ right-divides $b$ in the ring $R$.
    Then $R$ is a ring and $a,b\in\carrier[R]$.
\end{proposition}
\begin{proof}
    Follows by \obv{right_ring_divisibility}.
\end{proof}

\begin{proposition}\label{left_ring_divisibility_reflexive}
    Let $R$ be a ring and $a\in\carrier[R]$.
    Then $a$ left-divides $a$ in the ring $R$.
\end{proposition}
\begin{proof}
    Let $M=\MultiplicativeMonoid{R}$.
    $M$ is a monoid by \obv{multiplicative_monoid_is_monoid}.
    $\carrier[M]=\carrier[R]$ by
        \auto{multiplicative_monoid_fields}.
    Hence $a\in\carrier[M]$.
    Hence $a$ left-divides $a$ in $M$ by
        \obv{left_monoid_divisibility_reflexive}.
    Follows by \obv{left_ring_divisibility}.
\end{proof}

\begin{proposition}\label{right_ring_divisibility_reflexive}
    Let $R$ be a ring and $a\in\carrier[R]$.
    Then $a$ right-divides $a$ in the ring $R$.
\end{proposition}
\begin{proof}
    Let $M=\MultiplicativeMonoid{R}$.
    $M$ is a monoid by \obv{multiplicative_monoid_is_monoid}.
    $\carrier[M]=\carrier[R]$ by
        \auto{multiplicative_monoid_fields}.
    Hence $a\in\carrier[M]$.
    Hence $a$ right-divides $a$ in $M$ by
        \obv{right_monoid_divisibility_reflexive}.
    Follows by \obv{right_ring_divisibility}.
\end{proof}

\begin{theorem}\label{left_ring_divisibility_transitive}
    Suppose $a$ left-divides $b$ in the ring $R$ and
        $b$ left-divides $d$ in the ring $R$.
    Then $a$ left-divides $d$ in the ring $R$.
\end{theorem}
\begin{proof}
    $R$ is a ring and $a,b,d\in\carrier[R]$ by
        \obv{left_ring_divisibility_guards}.
    $a$ left-divides $b$ in $\MultiplicativeMonoid{R}$ and
        $b$ left-divides $d$ in $\MultiplicativeMonoid{R}$ by
        \obv{left_ring_divisibility}.
    Hence $a$ left-divides $d$ in $\MultiplicativeMonoid{R}$ by
        \obv{left_monoid_divisibility_transitive}.
    Follows by \obv{left_ring_divisibility}.
\end{proof}

\begin{theorem}\label{right_ring_divisibility_transitive}
    Suppose $a$ right-divides $b$ in the ring $R$ and
        $b$ right-divides $d$ in the ring $R$.
    Then $a$ right-divides $d$ in the ring $R$.
\end{theorem}
\begin{proof}

        \obv{right_ring_divisibility_guards}.
    $a$ right-divides $b$ in $\MultiplicativeMonoid{R}$ and
        $b$ right-divides $d$ in $\MultiplicativeMonoid{R}$ by
        \obv{right_ring_divisibility}.
    Hence $a$ right-divides $d$ in $\MultiplicativeMonoid{R}$ by
        \obv{right_monoid_divisibility_transitive}.
    Follows by \obv{right_ring_divisibility}.
\end{proof}

\begin{proposition}\label{left_ring_divides_product}
    Let $R$ be a ring and $a,c\in\carrier[R]$.
    Then $a$ left-divides $\ringMul[R](a,c)$ in the ring $R$.
\end{proposition}
\begin{proof}
    Let $M=\MultiplicativeMonoid{R}$.
    $M$ is a monoid by \obv{multiplicative_monoid_is_monoid}.
    $\carrier[M]=\carrier[R]$ and $\mul[M]=\ringMul[R]$ by
        \auto{multiplicative_monoid_fields}.
    Hence $a,c\in\carrier[M]$.
    Hence $a$ left-divides $\mul[M](a,c)$ in $M$ by
        \obv{left_monoid_divides_product}.
    $\ringMul[R](a,c)\in\carrier[R]$ by \obv{ring_mul_type}.
    Follows by \obv{left_ring_divisibility}.
\end{proof}

\begin{proposition}\label{right_ring_divides_product}
    Let $R$ be a ring and $a,c\in\carrier[R]$.
    Then $a$ right-divides $\ringMul[R](c,a)$ in the ring $R$.
\end{proposition}
\begin{proof}

    Hence $a,c\in\carrier[M]$.
    Hence $a$ right-divides $\mul[M](c,a)$ in $M$ by

    $\ringMul[R](c,a)\in\carrier[R]$ by \obv{ring_mul_type}.
    Follows by \obv{right_ring_divisibility}.
\end{proof}

\begin{theorem}\label{left_ring_divisibility_mul_left}
    Suppose $a$ left-divides $b$ in the ring $R$.
    Let $d\in\carrier[R]$.
    Then $\ringMul[R](d,a)$ left-divides $\ringMul[R](d,b)$
        in the ring $R$.
\end{theorem}
\begin{proof}
    Let $M=\MultiplicativeMonoid{R}$.
    $R$ is a ring and $a,b\in\carrier[R]$ by
        \obv{left_ring_divisibility_guards}.
    $\carrier[M]=\carrier[R]$ and $\mul[M]=\ringMul[R]$ by
        \auto{multiplicative_monoid_fields}.
    $a$ left-divides $b$ in $\MultiplicativeMonoid{R}$ by
        \obv{left_ring_divisibility}.
    Hence $a$ left-divides $b$ in $M$.
    $d\in\carrier[M]$.
    Hence $\mul[M](d,a)$ left-divides $\mul[M](d,b)$ in $M$ by
        \obv{left_monoid_divisibility_mul_left}.
    $\ringMul[R](d,a),\ringMul[R](d,b)\in\carrier[R]$ by
        \obv{ring_mul_type}.
    Follows by \obv{left_ring_divisibility}.
\end{proof}

\begin{theorem}\label{right_ring_divisibility_mul_right}

\begin{theorem}\label{ring_unit_left_divides}
    Let $R$ be a ring.
    Suppose $u$ is a unit of $\MultiplicativeMonoid{R}$.
    Let $a\in\carrier[R]$.
    Then $u$ left-divides $a$ in the ring $R$.
\end{theorem}
\begin{proof}
    $u\in\carrier[\MultiplicativeMonoid{R}]$ by
        \obv{monoid_unit_in_carrier}.
    $\carrier[\MultiplicativeMonoid{R}]=\carrier[R]$ by
        \obv{multiplicative_monoid_fields}.
    Hence $u\in\carrier[R]$ and
        $a\in\carrier[\MultiplicativeMonoid{R}]$.
    $u$ left-divides $a$ in $\MultiplicativeMonoid{R}$ by
        \obv{monoid_unit_left_divides}.
    Follows by \obv{left_ring_divisibility}.
\end{proof}

\begin{theorem}\label{ring_unit_right_divides}
    Let $R$ be a ring.

\begin{theorem}\label{sided_ring_divisibility_equivalent}
    Let $R$ be a commutative ring and $a,b\in\carrier[R]$.
    Then $a$ left-divides $b$ in the ring $R$ iff
        $a$ right-divides $b$ in the ring $R$.
\end{theorem}
\begin{proof}
    Let $M=\MultiplicativeMonoid{R}$.
    $R$ is a ring by \obv{commutative_ring_is_ring}.
    $M$ is a monoid by \obv{multiplicative_monoid_is_monoid}.
    $\carrier[M]=\carrier[R]$ and $\mul[M]=\ringMul[R]$ by
        \auto{multiplicative_monoid_fields}.
    Hence $a,b\in\carrier[M]$.
    For all $x,y\in\carrier[M]$ we have
        $\mul[M](x,y)=\mul[M](y,x)$ by
        \auto{commutative_ring_mul_commutative}.
    Hence $a$ left-divides $b$ in $M$ iff
        $a$ right-divides $b$ in $M$ by
        \obv{sided_monoid_divisibility_equivalent_if_commutative}.
    Follows by
        \auto{left_ring_divisibility,right_ring_divisibility}.
\end{proof}

\section{Divisibility in commutative rings}

\begin{definition}\label{ring_divisibility}
    $a$ divides $b$ in $R$ iff
        $R$ is a commutative ring and $a,b\in\carrier[R]$ and
        $a$ left-divides $b$ in the ring $R$.
\end{definition}

\begin{proposition}\label{ring_divisibility_guards}
    Suppose $a$ divides $b$ in $R$.
    Then $R$ is a commutative ring and $a,b\in\carrier[R]$.
\end{proposition}
\begin{proof}
    Follows by \obv{ring_divisibility}.
\end{proof}

\begin{theorem}\label{ring_divisibility_iff_left}
    Let $R$ be a commutative ring and $a,b\in\carrier[R]$.
    Then $a$ divides $b$ in $R$ iff
        $a$ left-divides $b$ in the ring $R$.
\end{theorem}
\begin{proof}
    Follows by \auto{ring_divisibility}.
\end{proof}

\begin{theorem}\label{ring_divisibility_iff_right}
    Let $R$ be a commutative ring and $a,b\in\carrier[R]$.
    Then $a$ divides $b$ in $R$ iff
        $a$ right-divides $b$ in the ring $R$.
\end{theorem}
\begin{proof}
    Follows by
        \auto{ring_divisibility,sided_ring_divisibility_equivalent}.
\end{proof}

\section{Natural divisibility in a ring}

\begin{theorem}\label{ring_natural_left_divisibility}
    Let $R$ be a ring.
    Suppose $m$ divides $n$ as natural numbers.
    Then $\ringNatural{R}{m}$ left-divides $\ringNatural{R}{n}$
        in the ring $R$.
\end{theorem}
\begin{proof}
    $m,n\in\naturals$ by
        \obv{natural_divisibility_guards}.
    Take a natural number $k$ such that
        $n=\naturalMul{m}{k}$ by
        \obv{natural_divisibility}.
    $\ringNatural{R}{m},\ringNatural{R}{n},
        \ringNatural{R}{k}\in\carrier[R]$ by
        \obv{ring_natural_type}.
    $\ringNatural{R}{\naturalMul{m}{k}}
        =\ringMul[R](\ringNatural{R}{m},\ringNatural{R}{k})$ by
        \obv{ring_natural_multiplication}.
    Hence $\ringNatural{R}{n}
        =\ringMul[R](\ringNatural{R}{m},\ringNatural{R}{k})$.
    Let $M=\MultiplicativeMonoid{R}$.
    $M$ is a monoid by \obv{multiplicative_monoid_is_monoid}.
    $\carrier[M]=\carrier[R]$ and $\mul[M]=\ringMul[R]$ by
        \auto{multiplicative_monoid_fields}.
    Hence $\ringNatural{R}{m}$ left-divides $\ringNatural{R}{n}$ in $M$ by
        \auto{left_monoid_divisibility}.
    Follows by \obv{left_ring_divisibility}.
\end{proof}

\begin{theorem}\label{ring_natural_right_divisibility}
    Let $R$ be a ring.
    Suppose $m$ divides $n$ as natural numbers.
    Then $\ringNatural{R}{m}$ right-divides $\ringNatural{R}{n}$
        in the ring $R$.
\end{theorem}
\begin{proof}
    $m,n\in\naturals$ by
        \obv{natural_divisibility_guards}.
    Take a natural number $k$ such that
        $n=\naturalMul{m}{k}$ by
        \obv{natural_divisibility}.
    $\ringNatural{R}{m},\ringNatural{R}{n},
        \ringNatural{R}{k}\in\carrier[R]$ by
        \obv{ring_natural_type}.
    $\ringNatural{R}{\naturalMul{m}{k}}
        =\ringMul[R](\ringNatural{R}{m},\ringNatural{R}{k})$ by
        \obv{ring_natural_multiplication}.
    $\ringMul[R](\ringNatural{R}{m},\ringNatural{R}{k})
        =\ringMul[R](\ringNatural{R}{k},\ringNatural{R}{m})$ by
        \obv{ring_natural_central}.
    Hence $\ringNatural{R}{n}
        =\ringMul[R](\ringNatural{R}{k},\ringNatural{R}{m})$.
    Let $M=\MultiplicativeMonoid{R}$.
    $M$ is a monoid by \obv{multiplicative_monoid_is_monoid}.
    $\carrier[M]=\carrier[R]$ and $\mul[M]=\ringMul[R]$ by
        \auto{multiplicative_monoid_fields}.
    Hence $\ringNatural{R}{m}$ right-divides $\ringNatural{R}{n}$ in $M$ by
        \auto{right_monoid_divisibility}.
    Follows by \obv{right_ring_divisibility}.
\end{proof}

\begin{corollary}\label{commutative_ring_natural_divisibility}
    Let $R$ be a commutative ring.
    Suppose $m$ divides $n$ as natural numbers.
    Then $\ringNatural{R}{m}$ divides $\ringNatural{R}{n}$ in $R$.
\end{corollary}
\begin{proof}
    $R$ is a ring by \obv{commutative_ring_is_ring}.
    $m,n\in\naturals$ by
        \obv{natural_divisibility_guards}.
    $\ringNatural{R}{m},\ringNatural{R}{n}\in\carrier[R]$ by
        \obv{ring_natural_type}.
    $\ringNatural{R}{m}$ left-divides $\ringNatural{R}{n}$
        in the ring $R$ by
        \obv{ring_natural_left_divisibility}.
    Follows by \obv{ring_divisibility}.
\end{proof}
