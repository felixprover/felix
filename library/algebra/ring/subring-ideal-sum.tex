\section{The sum of a subring and an ideal}

Let $A$ be a subring of a ring $R$ and let $I$ be a two-sided ideal of
$R$.  The carrier $A+I$ is defined once, inside $R$, as the set of sums
$a+i$.  This raw carrier is the common ambient object used by the second ring
isomorphism theorem.

\begin{definition}\label{subring_ideal_sum}
    $\subringIdealSum{R}{A}{I}
        =\{\ringAdd[R](a,i)\mid a\in A,i\in I\}$.
\end{definition}

\begin{proposition}\label{subring_ideal_sum_iff}
    Let $R$ be a ring.
    Let $A,I$ be subsets of $\carrier[R]$.
    Then $x\in\subringIdealSum{R}{A}{I}$ iff
        there exists $a\in A$ such that there exists $i\in I$ such that
            $x=\ringAdd[R](a,i)$.
\end{proposition}
\begin{proof}
    Follows by \obv{subring_ideal_sum}.
\end{proof}

\begin{proposition}\label{subring_ideal_sum_subseteq_carrier}
    Let $R$ be a ring.
    Let $A,I$ be subsets of $\carrier[R]$.
    Then $\subringIdealSum{R}{A}{I}\subseteq\carrier[R]$.
\end{proposition}
\begin{proof}
    Show that for every $x\in\subringIdealSum{R}{A}{I}$ we have
        $x\in\carrier[R]$.
    \begin{subproof}
        Fix $x\in\subringIdealSum{R}{A}{I}$.
        Take $a\in A$ such that there exists $i\in I$ such that
            $x=\ringAdd[R](a,i)$ by \obv{subring_ideal_sum_iff}.
        Take $i\in I$ such that $x=\ringAdd[R](a,i)$.
        $a,i\in\carrier[R]$ by \obv{subseteq}.
        Follows by \auto{ring_add_type}.
    \end{subproof}
    Follows by \obv{subseteq}.
\end{proof}

\begin{proposition}\label{subring_ideal_sum_contains_subring}
    Let $R$ be a ring.
    Let $A,I$ be subsets of $\carrier[R]$.
    Suppose $A$ is a subring of $R$ and $I$ is a two-sided ideal of $R$.
    Then $A\subseteq\subringIdealSum{R}{A}{I}$.
\end{proposition}
\begin{proof}
    $\ringZero[R]\in I$ by
        \auto{two_sided_ideal,left_ideal_additive_subgroup,
            subgroup_neutral,additive_group_fields}.
    Show that for every $a\in A$ we have
        $a\in\subringIdealSum{R}{A}{I}$.
    \begin{subproof}
        Fix $a\in A$.
        Hence $a\in\carrier[R]$ by \obv{subseteq}.
        $a=\ringAdd[R](a,\ringZero[R])$ by
            \obv{ring_add_zero_right}.
        Follows by \obv{subring_ideal_sum_iff}.
    \end{subproof}
    Follows by \obv{subseteq}.
\end{proof}

\begin{proposition}\label{subring_ideal_sum_contains_ideal}
    Let $R$ be a ring.
    Let $A,I$ be subsets of $\carrier[R]$.
    Suppose $A$ is a subring of $R$ and $I$ is a two-sided ideal of $R$.
    Then $I\subseteq\subringIdealSum{R}{A}{I}$.
\end{proposition}
\begin{proof}
    $\ringZero[R]\in A$ by
        \auto{subring_additive_subgroup,subgroup_neutral,
            additive_group_fields}.
    Show that for every $i\in I$ we have
        $i\in\subringIdealSum{R}{A}{I}$.
    \begin{subproof}
        Fix $i\in I$.
        Hence $i\in\carrier[R]$ by \obv{subseteq}.
        $i=\ringAdd[R](\ringZero[R],i)$ by
            \obv{ring_add_zero_left}.
        Follows by \obv{subring_ideal_sum_iff}.
    \end{subproof}
    Follows by \obv{subseteq}.
\end{proof}

\begin{theorem}\label{subring_ideal_sum_is_subring}
    Let $R$ be a ring.
    Let $A,I$ be subsets of $\carrier[R]$.
    Suppose $A$ is a subring of $R$ and $I$ is a two-sided ideal of $R$.
    Then $\subringIdealSum{R}{A}{I}$ is a subring of $R$.
\end{theorem}
\begin{proof}
    Let $T=\subringIdealSum{R}{A}{I}$ and $G=\AdditiveGroup{R}$.
    $G$ is a commutative group by
        \obv{additive_group_is_exact_commutative_group}.
    Hence $G$ is a group and $G$ is a monoid and $G$ is a unital magma by
        \obv{commutative_group_is_group,group_is_monoid,
            monoid_is_unital_magma}.
    $\subgroup{A}{G}$ and $\subgroup{I}{G}$ by
        \obv{subring_additive_subgroup,two_sided_ideal,
            left_ideal_additive_subgroup}.
    $T\subseteq\carrier[R]$ by
        \auto{subring_ideal_sum_subseteq_carrier}.
    Hence $T\subseteq\carrier[G]$ by \auto{additive_group_fields}.
    $\ringZero[R]\in A$ and $\ringZero[R]\in I$ by
        \auto{subgroup_neutral,additive_group_fields}.
    $\ringZero[R]\in\carrier[R]$ by \obv{ring_zero_type}.
    $\ringAdd[R](\ringZero[R],\ringZero[R])=\ringZero[R]$ by
        \obv{ring_add_zero_left}.
    Hence $\ringZero[R]\in T$ by \obv{subring_ideal_sum_iff}.
    Hence $\neutral[G]\in T$ by \auto{additive_group_fields}.
    Show that for all $x,y\in T$ we have $\mul[G](x,y)\in T$.
    \begin{subproof}
        Fix $x,y\in T$.
        Take $a\in A$ such that there exists $i\in I$ such that
            $x=\ringAdd[R](a,i)$ by \obv{subring_ideal_sum_iff}.
        Take $i\in I$ such that $x=\ringAdd[R](a,i)$.
        Take $b\in A$ such that there exists $j\in I$ such that
            $y=\ringAdd[R](b,j)$ by \obv{subring_ideal_sum_iff}.
        Take $j\in I$ such that $y=\ringAdd[R](b,j)$.
        $a,b,i,j\in\carrier[G]$ by
            \auto{subgroup_subseteq,subseteq}.
        $\mul[G](a,i),\mul[G](b,j),
            \mul[G](a,b),\mul[G](i,j)\in\carrier[G]$ by
            \obv{monoid_is_magma,magma_welldef}.
        $\mul[G](a,b)\in A$ and $\mul[G](i,j)\in I$ by
            \obv{subgroup_mul}.
        $\mul[G](\mul[G](a,i),b)
            =\mul[G](\mul[G](a,b),i)$ by
            \obv{monoid_commutative_monoid_exchange,
                commutative_group_commutative}.
        $\mul[G](\mul[G](a,i),\mul[G](b,j))
            =\mul[G](\mul[G](\mul[G](a,i),b),j)$ by
            \obv{monoid_assoc}.
        Hence $\mul[G](\mul[G](a,i),\mul[G](b,j))
            =\mul[G](\mul[G](\mul[G](a,b),i),j)$.
        $\mul[G](\mul[G](\mul[G](a,b),i),j)
            =\mul[G](\mul[G](a,b),\mul[G](i,j))$ by
            \obv{monoid_assoc}.
        Hence $\mul[G](\mul[G](a,i),\mul[G](b,j))
            =\mul[G](\mul[G](a,b),\mul[G](i,j))$.
        $\mul[G](a,i)=\ringAdd[R](a,i)$ and
            $\mul[G](b,j)=\ringAdd[R](b,j)$ and
            $\mul[G](\mul[G](a,b),\mul[G](i,j))
                =\ringAdd[R](\mul[G](a,b),\mul[G](i,j))$ by
            \auto{additive_group_fields}.
        Hence $\mul[G](x,y)
            =\ringAdd[R](\mul[G](a,b),\mul[G](i,j))$.
        Follows by \auto{subring_ideal_sum_iff}.
    \end{subproof}
    Show that for every $x\in T$ we have $\groupInv[G](x)\in T$.
    \begin{subproof}
        Fix $x\in T$.
        Take $a\in A$ such that there exists $i\in I$ such that
            $x=\ringAdd[R](a,i)$ by \obv{subring_ideal_sum_iff}.
        Take $i\in I$ such that $x=\ringAdd[R](a,i)$.
        $a,i\in\carrier[G]$ by \auto{subgroup_subseteq,subseteq}.
        $\groupInv[G](a)\in A$ and $\groupInv[G](i)\in I$ by
            \obv{subgroup_inv}.
        $\groupInv[G](\mul[G](a,i))
            =\mul[G](\groupInv[G](a),\groupInv[G](i))$ by
            \obv{commutative_group_inverse_product}.
        Hence $\groupInv[G](x)
            =\ringAdd[R](\groupInv[G](a),\groupInv[G](i))$ by
            \auto{additive_group_fields}.
        Follows by \auto{subring_ideal_sum_iff}.
    \end{subproof}
    Hence $\subgroup{T}{G}$ by \obv{subgroup}.
    $\ringOne[R]\in A$ by \obv{subring_one}.
    Hence $\ringOne[R]\in T$ by
        \auto{subring_ideal_sum_contains_subring,subseteq}.
    Show that for all $x,y\in T$ we have $\ringMul[R](x,y)\in T$.
    \begin{subproof}
        Fix $x,y\in T$.
        Hence $x,y\in\carrier[R]$ by \obv{subseteq}.
        Take $a\in A$ such that there exists $i\in I$ such that
            $x=\ringAdd[R](a,i)$ by \obv{subring_ideal_sum_iff}.
        Take $i\in I$ such that $x=\ringAdd[R](a,i)$.
        Take $b\in A$ such that there exists $j\in I$ such that
            $y=\ringAdd[R](b,j)$ by \obv{subring_ideal_sum_iff}.
        Take $j\in I$ such that $y=\ringAdd[R](b,j)$.
        $a,b,i,j\in\carrier[R]$ by
            \obv{subring_subseteq_carrier,
                two_sided_ideal_subseteq_carrier,subseteq}.
        Let $p=\ringMul[R](a,b)$.
        Let $u=\ringMul[R](a,j)$.
        Let $v=\ringMul[R](i,b)$.
        Let $w=\ringMul[R](i,j)$.
        $p\in A$ by \auto{subring_mul}.
        $u,v,w\in I$ by
            \auto{two_sided_ideal,left_ideal_expanded_iff,
                right_ideal_expanded_iff}.
        $u,v,w\in\carrier[R]$ by
            \obv{two_sided_ideal_subseteq_carrier,subseteq}.
        $\ringAdd[R](v,w)\in I$ and
            $\ringAdd[R](u,\ringAdd[R](v,w))\in I$ by
            \auto{left_ideal_additive_subgroup,subgroup_mul,
                additive_group_fields,two_sided_ideal}.
        Let $q=\ringAdd[R](u,\ringAdd[R](v,w))$.
        $p,u,\ringAdd[R](v,w),q\in\carrier[R]$ by
            \auto{subring_subseteq_carrier,
                two_sided_ideal_subseteq_carrier,subseteq}.
        $\ringMul[R](a,y)=\ringAdd[R](p,u)$ by
            \auto{ring_distributive_left}.
        $\ringMul[R](i,y)=\ringAdd[R](v,w)$ by
            \auto{ring_distributive_left}.
        $\ringMul[R](\ringAdd[R](a,i),y)
            =\ringAdd[R](\ringMul[R](a,y),\ringMul[R](i,y))$ by
            \obv{ring_distributive_right}.
        Hence $\ringMul[R](x,y)
            =\ringAdd[R](\ringMul[R](a,y),\ringMul[R](i,y))$.
        $\ringAdd[R](p,q)
            =\ringAdd[R](\ringAdd[R](p,u),\ringAdd[R](v,w))$ by
            \auto{ring_add_associative}.
        $\ringMul[R](x,y)
            =\ringAdd[R](p,q)$.
        Follows by \auto{subring_ideal_sum_iff}.
    \end{subproof}
    Follows by \auto{subring}.
\end{proof}

\begin{theorem}\label{subring_ideal_sum_universal_property}
    Let $R$ be a ring.
    Let $A,I$ be subsets of $\carrier[R]$.
    Suppose $A$ is a subring of $R$ and $I$ is a two-sided ideal of $R$.
    Suppose $B$ is a subring of $R$.
    Then $\subringIdealSum{R}{A}{I}\subseteq B$ iff
        $A\subseteq B$ and $I\subseteq B$.
\end{theorem}
\begin{proof}
    Let $T=\subringIdealSum{R}{A}{I}$.
    $A\subseteq T$ and $I\subseteq T$ by
        \obv{subring_ideal_sum_contains_subring,
            subring_ideal_sum_contains_ideal}.
    Show that if $A\subseteq B$ and $I\subseteq B$, then $T\subseteq B$.
    \begin{subproof}
        Assume $A\subseteq B$ and $I\subseteq B$.
        Show that for every $x\in T$ we have $x\in B$.
        \begin{subproof}
            Fix $x\in T$.
            Take $a\in A$ such that there exists $i\in I$ such that
                $x=\ringAdd[R](a,i)$ by \obv{subring_ideal_sum_iff}.
            Take $i\in I$ such that $x=\ringAdd[R](a,i)$.
            Hence $a,i\in B$ by \obv{subseteq}.
            Follows by
                \auto{subring_additive_subgroup,subgroup_mul,
                    additive_group_fields}.
        \end{subproof}
        Follows by \obv{subseteq}.
    \end{subproof}
    Follows by \auto{subseteq_transitive}.
\end{proof}

\begin{theorem}\label{ideal_in_subring_ideal_sum}
    Let $R$ be a ring.
    Let $A,I$ be subsets of $\carrier[R]$.
    Suppose $A$ is a subring of $R$ and $I$ is a two-sided ideal of $R$.
    Let $T=\subringIdealSum{R}{A}{I}$.
    Then $I$ is a two-sided ideal of $\InducedSubring{R}{T}$.
\end{theorem}
\begin{proof}
    Let $S=\InducedSubring{R}{T}$ and $G=\AdditiveGroup{R}$.
    $T$ is a subring of $R$ by \auto{subring_ideal_sum_is_subring}.
    Hence $S$ is a ring by \obv{induced_subring_is_exact}.
    $I\subseteq T$ by \obv{subring_ideal_sum_contains_ideal}.
    $\subgroup{T}{G}$ by \obv{subring_additive_subgroup}.
    $\subgroup{I}{G}$ by
        \obv{two_sided_ideal,left_ideal_additive_subgroup}.
    Hence $\subgroup{I}{\subgroupStructure{G}{T}}$ by
        \obv{subgroup_of_induced_group_iff}.
    $\AdditiveGroup{S}=\subgroupStructure{G}{T}$ by
        \auto{induced_subring_additive_group}.
    Hence $\subgroup{I}{\AdditiveGroup{S}}$.
    Show that for all $s,i$ such that $s\in\carrier[S]$ and $i\in I$
        we have $\ringMul[S](s,i)\in I$.
    \begin{subproof}
        Fix $s,i$.
        Assume $s\in\carrier[S]$ and $i\in I$.
        Hence $s\in T$ by \auto{induced_subring_agreement}.
        Hence $i\in T$ by \obv{subseteq}.
        Hence $s\in\carrier[R]$ by
            \auto{subring_ideal_sum_subseteq_carrier,subseteq}.
        Hence $\ringMul[R](s,i)\in I$ by
            \auto{two_sided_ideal,left_ideal_expanded_iff}.
        Follows by \auto{induced_subring_agreement}.
    \end{subproof}
    Hence $I$ is a left ideal of $S$ by
        \auto{left_ideal_expanded_iff}.
    Show that for all $i,s$ such that $i\in I$ and $s\in\carrier[S]$
        we have $\ringMul[S](i,s)\in I$.
    \begin{subproof}
        Fix $i,s$.
        Assume $i\in I$ and $s\in\carrier[S]$.
        Hence $s\in T$ by \auto{induced_subring_agreement}.
        Hence $i\in T$ by \obv{subseteq}.
        Hence $s\in\carrier[R]$ by
            \auto{subring_ideal_sum_subseteq_carrier,subseteq}.
        Hence $\ringMul[R](i,s)\in I$ by
            \auto{two_sided_ideal,right_ideal_expanded_iff}.
        Follows by \auto{induced_subring_agreement}.
    \end{subproof}
    Hence $I$ is a right ideal of $S$ by
        \auto{right_ideal_expanded_iff}.
    Follows by \obv{two_sided_ideal}.
\end{proof}

\begin{theorem}\label{subring_intersection_ideal_in_induced_subring}
    Let $R$ be a ring.
    Let $A,I$ be subsets of $\carrier[R]$.
    Suppose $A$ is a subring of $R$ and $I$ is a two-sided ideal of $R$.
    Then $A\inter I$ is a two-sided ideal of $\InducedSubring{R}{A}$.
\end{theorem}
\begin{proof}
    Let $S=\InducedSubring{R}{A}$ and $G=\AdditiveGroup{R}$.
    $S$ is a ring by \obv{induced_subring_is_exact}.
    $\subgroup{A}{G}$ by \obv{subring_additive_subgroup}.
    $\subgroup{I}{G}$ by
        \obv{two_sided_ideal,left_ideal_additive_subgroup}.
    Hence $\subgroup{A\inter I}{G}$ by \obv{subgroup_intersection}.
    $A\inter I\subseteq A$ by \obv{subseteq,inter_elim_left}.
    Hence $\subgroup{A\inter I}{\subgroupStructure{G}{A}}$ by
        \obv{subgroup_of_induced_group_iff}.
    $\AdditiveGroup{S}=\subgroupStructure{G}{A}$ by
        \auto{induced_subring_additive_group}.
    Hence $\subgroup{A\inter I}{\AdditiveGroup{S}}$.
    Show that for all $a,x$ such that $a\in\carrier[S]$ and
        $x\in A\inter I$ we have $\ringMul[S](a,x)\in A\inter I$.
    \begin{subproof}
        Fix $a,x$.
        Assume $a\in\carrier[S]$ and $x\in A\inter I$.
        Hence $a\in A$ and $x\in A$ and $x\in I$ by
            \auto{induced_subring_agreement,inter_elim_left,
                inter_elim_right}.
        Hence $a,x\in\carrier[R]$ by
            \obv{subring_subseteq_carrier,subseteq}.
        $\ringMul[R](a,x)\in A$ by \obv{subring_mul}.
        $\ringMul[R](a,x)\in I$ by
            \auto{two_sided_ideal,left_ideal_expanded_iff}.
        Hence $\ringMul[R](a,x)\in A\inter I$ by \obv{inter_intro}.
        Follows by \auto{induced_subring_agreement}.
    \end{subproof}
    Hence $A\inter I$ is a left ideal of $S$ by
        \auto{left_ideal_expanded_iff}.
    Show that for all $x,a$ such that $x\in A\inter I$ and
        $a\in\carrier[S]$ we have $\ringMul[S](x,a)\in A\inter I$.
    \begin{subproof}
        Fix $x,a$.
        Assume $x\in A\inter I$ and $a\in\carrier[S]$.
        Hence $x\in A$ and $x\in I$ and $a\in A$ by
            \auto{inter_elim_left,inter_elim_right,
                induced_subring_agreement}.
        Hence $x,a\in\carrier[R]$ by
            \obv{subring_subseteq_carrier,subseteq}.
        $\ringMul[R](x,a)\in A$ by \obv{subring_mul}.
        $\ringMul[R](x,a)\in I$ by
            \auto{two_sided_ideal,right_ideal_expanded_iff}.
        Hence $\ringMul[R](x,a)\in A\inter I$ by \obv{inter_intro}.
        Follows by \auto{induced_subring_agreement}.
    \end{subproof}
    Hence $A\inter I$ is a right ideal of $S$ by
        \auto{right_ideal_expanded_iff}.
    Follows by \auto{two_sided_ideal}.
\end{proof}
