\section{Generated subrings}

The subring generated by a set is the closure-system hull of that set among
the subring carriers of the ambient ring.  Thus the construction is literally
the intersection, bounded by the ambient carrier, of all subrings containing
the generators.

\begin{proposition}\label{subring_carriers_form_closure_system}
    Let $R$ be a ring.
    Let
        $C=\{B\in\pow{\carrier[R]}\mid
            \text{$B$ is a subring of $R$}\}$.
    Then $C$ is a closure system on $\carrier[R]$.
\end{proposition}
\begin{proof}
    Show that $C\subseteq\pow{\carrier[R]}$.
    \begin{subproof}
        Show that for every $B\in C$ we have
            $B\in\pow{\carrier[R]}$.
        \begin{subproof}
            Follows by \auto.
        \end{subproof}
        Follows by \obv{subseteq}.
    \end{subproof}
    Show that for every $D\subseteq C$ we have
        $\boundedIntersection{\carrier[R]}{D}\in C$.
    \begin{subproof}
        Fix $D\subseteq C$.
        Let $P=\boundedIntersection{\carrier[R]}{D}$.
        $P\subseteq\carrier[R]$ by
            \auto{bounded_intersection_subseteq}.
        Let $G=\AdditiveGroup{R}$.
        $G$ is a group by
            \obv{additive_group_is_exact_commutative_group,
                commutative_group_is_group}.
        $P\subseteq\carrier[G]$ by \auto{additive_group_fields}.
        Show that $\neutral[G]\in P$.
        \begin{subproof}
            $\neutral[G]\in\carrier[G]$ by
                \obv{group_is_monoid,monoid_is_unital_magma,
                    unitalmagma_type}.
            Hence $\neutral[G]\in\carrier[R]$ by
                \auto{additive_group_fields}.
            Show that for every $B\in D$ we have $\neutral[G]\in B$.
            \begin{subproof}
                Fix $B\in D$.
                Hence $B\in C$ by \obv{subseteq}.
                Hence $B$ is a subring of $R$ by \auto.
                Hence $\subgroup{B}{G}$ by
                    \obv{subring_additive_subgroup}.
                Follows by \obv{subgroup_neutral}.
            \end{subproof}
            Follows by \obv{bounded_intersection_iff}.
        \end{subproof}
        Show that for all $x,y\in P$ we have $\mul[G](x,y)\in P$.
        \begin{subproof}
            Fix $x,y\in P$.
            Hence $x,y\in\carrier[G]$ by \obv{subseteq}.
            Hence $\mul[G](x,y)\in\carrier[G]$ by
                \obv{group_is_monoid,monoid_is_magma,magma_welldef}.
            Hence $\mul[G](x,y)\in\carrier[R]$ by
                \auto{additive_group_fields}.
            Show that for every $B\in D$ we have $\mul[G](x,y)\in B$.
            \begin{subproof}
                Fix $B\in D$.
                Hence $B\in C$ by \obv{subseteq}.
                Hence $B$ is a subring of $R$ by \auto.
                Hence $\subgroup{B}{G}$ by
                    \obv{subring_additive_subgroup}.
                $x,y\in B$ by \auto{bounded_intersection_iff}.
                Follows by \obv{subgroup_mul}.
            \end{subproof}
            Follows by \obv{bounded_intersection_iff}.
        \end{subproof}
        Show that for every $x\in P$ we have $\groupInv[G](x)\in P$.
        \begin{subproof}
            Fix $x\in P$.
            Hence $x\in\carrier[G]$ by \obv{subseteq}.
            Hence $\groupInv[G](x)\in\carrier[G]$ by
                \obv{group_inverse_type}.
            Hence $\groupInv[G](x)\in\carrier[R]$ by
                \auto{additive_group_fields}.
            Show that for every $B\in D$ we have $\groupInv[G](x)\in B$.
            \begin{subproof}
                Fix $B\in D$.
                Hence $B\in C$ by \obv{subseteq}.
                Hence $B$ is a subring of $R$ by \auto.
                Hence $\subgroup{B}{G}$ by
                    \obv{subring_additive_subgroup}.
                $x\in B$ by \auto{bounded_intersection_iff}.
                Follows by \obv{subgroup_inv}.
            \end{subproof}
            Follows by \obv{bounded_intersection_iff}.
        \end{subproof}
        Hence $\subgroup{P}{G}$ by \obv{subgroup}.
        Show that $\ringOne[R]\in P$.
        \begin{subproof}
            $\ringOne[R]\in\carrier[R]$ by \obv{ring_one_type}.
            Show that for every $B\in D$ we have $\ringOne[R]\in B$.
            \begin{subproof}
                Fix $B\in D$.
                Hence $B\in C$ by \obv{subseteq}.
                Hence $B$ is a subring of $R$ by \auto.
                Follows by \obv{subring_one}.
            \end{subproof}
            Follows by \obv{bounded_intersection_iff}.
        \end{subproof}
        Show that for all $x,y\in P$ we have $\ringMul[R](x,y)\in P$.
        \begin{subproof}
            Fix $x,y\in P$.
            Hence $x,y\in\carrier[R]$ by \obv{subseteq}.
            Hence $\ringMul[R](x,y)\in\carrier[R]$ by
                \obv{ring_mul_type}.
            Show that for every $B\in D$ we have
                $\ringMul[R](x,y)\in B$.
            \begin{subproof}
                Fix $B\in D$.
                Hence $B\in C$ by \obv{subseteq}.
                Hence $B$ is a subring of $R$ by \auto.
                $x,y\in B$ by \auto{bounded_intersection_iff}.
                Follows by \obv{subring_mul}.
            \end{subproof}
            Follows by \obv{bounded_intersection_iff}.
        \end{subproof}
        Hence $P$ is a subring of $R$ by \obv{subring}.
        $P\in\pow{\carrier[R]}$ by \obv{powerset_intro}.
        Hence $P\in C$ by \auto.
    \end{subproof}
    Follows by \obv{closure_system}.
\end{proof}

\begin{definition}\label{generated_subring}
    Let $R$ be a ring.
    Let $A$ be a subset of $\carrier[R]$.
    $\generatedSubring{R}{A}=
        \closureHull{\carrier[R]}
            {\{B\in\pow{\carrier[R]}\mid
                \text{$B$ is a subring of $R$}\}}
            {A}$.
\end{definition}

\begin{proposition}\label{generated_subring_subseteq_carrier}
    Let $R$ be a ring.
    Let $A$ be a subset of $\carrier[R]$.
    Then $\generatedSubring{R}{A}\subseteq\carrier[R]$.
\end{proposition}
\begin{proof}
    Follows by \auto{generated_subring,closure_hull_subseteq}.
\end{proof}

\begin{proposition}\label{generated_subring_contains_generators}
    Let $R$ be a ring.
    Let $A$ be a subset of $\carrier[R]$.
    Then $A\subseteq\generatedSubring{R}{A}$.
\end{proposition}
\begin{proof}
    Let
        $C=\{B\in\pow{\carrier[R]}\mid
            \text{$B$ is a subring of $R$}\}$.
    $C$ is a closure system on $\carrier[R]$ by
        \obv{subring_carriers_form_closure_system}.
    $A\subseteq\closureHull{\carrier[R]}{C}{A}$ by
        \obv{closure_hull_extensive}.
    Follows by \obv{generated_subring}.
\end{proof}

\begin{theorem}\label{generated_subring_is_subring}
    Let $R$ be a ring.
    Let $A$ be a subset of $\carrier[R]$.
    Then $\generatedSubring{R}{A}$ is a subring of $R$.
\end{theorem}
\begin{proof}
    Let
        $C=\{B\in\pow{\carrier[R]}\mid
            \text{$B$ is a subring of $R$}\}$.
    $C$ is a closure system on $\carrier[R]$ by
        \obv{subring_carriers_form_closure_system}.
    $\closureHull{\carrier[R]}{C}{A}\in C$ by
        \obv{closure_hull_closed}.
    Follows by \auto{generated_subring}.
\end{proof}

\begin{theorem}\label{generated_subring_least}
    Let $R$ be a ring.
    Let $A$ be a subset of $\carrier[R]$.
    Suppose $B$ is a subring of $R$ and $A\subseteq B$.
    Then $\generatedSubring{R}{A}\subseteq B$.
\end{theorem}
\begin{proof}
    Let
        $C=\{K\in\pow{\carrier[R]}\mid
            \text{$K$ is a subring of $R$}\}$.
    $C$ is a closure system on $\carrier[R]$ by
        \obv{subring_carriers_form_closure_system}.
    $B\subseteq\carrier[R]$ by \obv{subring_subseteq_carrier}.
    Hence $B\in\pow{\carrier[R]}$ by \obv{powerset_intro}.
    Hence $B\in C$.
    $\closureHull{\carrier[R]}{C}{A}\subseteq B$ by
        \obv{closure_hull_least}.
    Follows by \obv{generated_subring}.
\end{proof}

\begin{theorem}\label{generated_subring_iff}
    Let $R$ be a ring.
    Let $A$ be a subset of $\carrier[R]$.
    Then $x\in\generatedSubring{R}{A}$ iff for every $B$ such that
        $B$ is a subring of $R$ and $A\subseteq B$ we have $x\in B$.
\end{theorem}
\begin{proof}
    Show that if $x\in\generatedSubring{R}{A}$, then for every $B$ such that
        $B$ is a subring of $R$ and $A\subseteq B$ we have $x\in B$.
    \begin{subproof}
        Assume $x\in\generatedSubring{R}{A}$.
        Fix $B$.
        Assume $B$ is a subring of $R$ and $A\subseteq B$.
        $\generatedSubring{R}{A}\subseteq B$ by
            \obv{generated_subring_least}.
        Follows by \obv{subseteq}.
    \end{subproof}
    Show that if for every $B$ such that
        $B$ is a subring of $R$ and $A\subseteq B$ we have $x\in B$, then
        $x\in\generatedSubring{R}{A}$.
    \begin{subproof}
        Assume that for every $B$ such that
            $B$ is a subring of $R$ and $A\subseteq B$ we have $x\in B$.
        $\generatedSubring{R}{A}$ is a subring of $R$ by
            \obv{generated_subring_is_subring}.
        $A\subseteq\generatedSubring{R}{A}$ by
            \obv{generated_subring_contains_generators}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{theorem}\label{generated_subring_universal_property}
    Let $R$ be a ring.
    Let $A$ be a subset of $\carrier[R]$.
    Suppose $B$ is a subring of $R$.
    Then $\generatedSubring{R}{A}\subseteq B$ iff $A\subseteq B$.
\end{theorem}
\begin{proof}
    $A\subseteq\generatedSubring{R}{A}$ by
        \obv{generated_subring_contains_generators}.
    Follows by \auto{generated_subring_least,subseteq_transitive}.
\end{proof}

\begin{corollary}\label{generated_subring_of_subring}
    Let $R$ be a ring.
    Let $A$ be a subset of $\carrier[R]$.
    Suppose $A$ is a subring of $R$.
    Then $\generatedSubring{R}{A}=A$.
\end{corollary}
\begin{proof}
    $R$ is a ring and $A\subseteq\carrier[R]$ by
        \obv{subring_ambient_ring,subring_subseteq_carrier}.
    $A\subseteq\generatedSubring{R}{A}$ by
        \obv{generated_subring_contains_generators}.
    $\generatedSubring{R}{A}\subseteq A$ by
        \obv{generated_subring_least,subseteq_refl}.
    Follows by \obv{subseteq_antisymmetric}.
\end{proof}

\begin{proposition}\label{generated_subring_monotone}
    Let $R$ be a ring.
    Let $A,B\subseteq\carrier[R]$.
    Suppose $A\subseteq B$.
    Then $\generatedSubring{R}{A}\subseteq\generatedSubring{R}{B}$.
\end{proposition}
\begin{proof}
    $A\subseteq\carrier[R]$ by \obv{subseteq_transitive}.
    $\generatedSubring{R}{B}$ is a subring of $R$ by
        \obv{generated_subring_is_subring}.
    $B\subseteq\generatedSubring{R}{B}$ by
        \obv{generated_subring_contains_generators}.
    Hence $A\subseteq\generatedSubring{R}{B}$ by
        \obv{subseteq_transitive}.
    Follows by \obv{generated_subring_least}.
\end{proof}

\subsection{Homomorphisms and generators}

\begin{theorem}\label{ring_hom_agreement_on_generated_subring}
    Let $R$ be a ring.
    Suppose $f$ is a ring homomorphism from $R$ to $S$ and
        $g$ is a ring homomorphism from $R$ to $S$.
    Let $A$ be a subset of $\carrier[R]$.
    Suppose for every $a\in A$ we have $f(a)=g(a)$.
    Then for every $x\in\generatedSubring{R}{A}$ we have $f(x)=g(x)$.
\end{theorem}
\begin{proof}
    Let $E=\{x\in\carrier[R]\mid f(x)=g(x)\}$.
    Let $G=\AdditiveGroup{R}$ and $H=\AdditiveGroup{S}$.
    $\groupHom{f}{G}{H}$ and $\groupHom{g}{G}{H}$ by
        \obv{ring_hom_is_additive_group_hom}.
    $G$ is a group and $H$ is a group by
        \obv{additive_group_is_exact_commutative_group,
            commutative_group_is_group}.
    $E\subseteq\carrier[G]$ by
        \auto{subseteq,additive_group_fields}.
    $f(\neutral[G])=\neutral[H]$ and
        $g(\neutral[G])=\neutral[H]$ by \obv{group_hom_neutral}.
    $\neutral[G]\in\carrier[G]$ by
        \obv{group_is_monoid,monoid_is_unital_magma,
            unitalmagma_type}.
    Hence $\neutral[G]\in\carrier[R]$ by
        \auto{additive_group_fields}.
    Hence $\neutral[G]\in E$.
    Show that for all $x,y\in E$ we have $\mul[G](x,y)\in E$.
    \begin{subproof}
        Fix $x,y\in E$.
        Hence $x,y\in\carrier[R]$ and
            $f(x)=g(x)$ and $f(y)=g(y)$.
        Hence $x,y\in\carrier[G]$ by \obv{subseteq}.
        $\mul[G](x,y)\in\carrier[G]$ by
            \obv{group_is_monoid,monoid_is_magma,magma_welldef}.
        Hence $\mul[G](x,y)\in\carrier[R]$ by
            \auto{additive_group_fields}.
        $f(\mul[G](x,y))=\mul[H](f(x),f(y))$ and
            $g(\mul[G](x,y))=\mul[H](g(x),g(y))$ by
            \obv{group_hom_mul}.
        Hence $f(\mul[G](x,y))=g(\mul[G](x,y))$.
    \end{subproof}
    Show that for every $x\in E$ we have $\groupInv[G](x)\in E$.
    \begin{subproof}
        Fix $x\in E$.
        Hence $x\in\carrier[R]$ and $f(x)=g(x)$.
        Hence $x\in\carrier[G]$ by \obv{subseteq}.
        $\groupInv[G](x)\in\carrier[G]$ by
            \obv{group_inverse_type}.
        Hence $\groupInv[G](x)\in\carrier[R]$ by
            \auto{additive_group_fields}.
        $f(\groupInv[G](x))=\groupInv[H](f(x))$ and
            $g(\groupInv[G](x))=\groupInv[H](g(x))$ by
            \obv{group_hom_inverse}.
        Hence $f(\groupInv[G](x))=g(\groupInv[G](x))$.
    \end{subproof}
    Hence $\subgroup{E}{G}$ by \obv{subgroup}.
    $f(\ringOne[R])=\ringOne[S]$ and
        $g(\ringOne[R])=\ringOne[S]$ by \obv{ring_hom_one}.
    $\ringOne[R]\in\carrier[R]$ by \obv{ring_one_type}.
    Hence $\ringOne[R]\in E$.
    Show that for all $x,y\in E$ we have $\ringMul[R](x,y)\in E$.
    \begin{subproof}
        Fix $x,y\in E$.
        Hence $x,y\in\carrier[R]$ and
            $f(x)=g(x)$ and $f(y)=g(y)$.
        $\ringMul[R](x,y)\in\carrier[R]$ by
            \obv{ring_mul_type}.
        $f(\ringMul[R](x,y))=\ringMul[S](f(x),f(y))$ and
            $g(\ringMul[R](x,y))=\ringMul[S](g(x),g(y))$ by
            \obv{ring_hom_mul}.
        Hence $f(\ringMul[R](x,y))=g(\ringMul[R](x,y))$.
    \end{subproof}
    Hence $E$ is a subring of $R$ by \obv{subring}.
    Show that $A\subseteq E$.
    \begin{subproof}
        Show that for every $a\in A$ we have $a\in E$.
        \begin{subproof}
            Fix $a\in A$.
            Hence $a\in\carrier[R]$ by \obv{subseteq}.
            $f(a)=g(a)$.
        \end{subproof}
        Follows by \obv{subseteq}.
    \end{subproof}
    Hence $\generatedSubring{R}{A}\subseteq E$ by
        \obv{generated_subring_least}.
    Follows by \auto{subseteq}.
\end{proof}

\begin{theorem}\label{ring_hom_equality_from_generated_subring}
    Let $R$ be a ring.
    Let $A$ be a subset of $\carrier[R]$.
    Suppose $\generatedSubring{R}{A}=\carrier[R]$.
    Suppose $f$ is a ring homomorphism from $R$ to $S$ and
        $g$ is a ring homomorphism from $R$ to $S$.
    Suppose for every $a\in A$ we have $f(a)=g(a)$.
    Then $f=g$.
\end{theorem}
\begin{proof}
    For every $x\in\generatedSubring{R}{A}$ we have $f(x)=g(x)$ by
        \obv{ring_hom_agreement_on_generated_subring}.
    Hence for every $x\in\carrier[R]$ we have $f(x)=g(x)$.
    Follows by \obv{ring_hom_extensionality}.
\end{proof}
