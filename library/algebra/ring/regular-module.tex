\section{Regular modules}

The left regular module uses the additive group of a ring literally and lets
the ring act by its own multiplication.  The right regular module is the same
construction over the opposite ring, so there is only one module
construction and one stored scalar-action convention.

\begin{definition}\label{left_regular_module}
    $\LeftRegularModule{R}=
        \structure[left module over R]{
            \carrier=\carrier[\AdditiveGroup{R}],
            \mul=\mul[\AdditiveGroup{R}],
            \neutral=\neutral[\AdditiveGroup{R}],
            \groupInv=\groupInv[\AdditiveGroup{R}],
            \scalarMul=\ringMul[R]
        }$.
\end{definition}

\begin{proposition}\label{left_regular_module_fields}
    Let $G=\AdditiveGroup{R}$.
    Then
        $\carrier[\LeftRegularModule{R}]=\carrier[G]$ and
        $\mul[\LeftRegularModule{R}]=\mul[G]$ and
        $\neutral[\LeftRegularModule{R}]=\neutral[G]$ and
        $\groupInv[\LeftRegularModule{R}]=\groupInv[G]$ and
        $\scalarMul[\LeftRegularModule{R}]=\ringMul[R]$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{left_regular_module,
            left_moduleaggregateprojection_carrier,
            left_moduleaggregateprojection_mul,
            left_moduleaggregateprojection_neutral,
            left_moduleaggregateprojection_groupInv,
            left_moduleaggregateprojection_scalarMul}.
\end{proof}

\begin{proposition}\label{left_regular_module_carrier}
    $\carrier[\LeftRegularModule{R}]=\carrier[R]$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{left_regular_module_fields,additive_group_fields}.
\end{proof}

\begin{proposition}\label{left_regular_scalar_apply}
    Let $R$ be a ring and $a,x\in\carrier[R]$.
    Then
        $\scalarMul[\LeftRegularModule{R}](a,x)
            =\ringMul[R](a,x)$.
\end{proposition}
\begin{proof}
    Follows by \auto{left_regular_module_fields}.
\end{proof}

\begin{theorem}\label{left_regular_module_is_exact}
    Let $R$ be a ring.
    Then $\LeftRegularModule{R}$ is an exact left module over $R$.
\end{theorem}
\begin{proof}
    Let $G=\AdditiveGroup{R}$ and $M=\LeftRegularModule{R}$.
    $G$ is an exact commutative group by
        \auto{additive_group_is_exact_commutative_group}.
    Hence $G$ is a group and $G$ is a monoid and
        $G$ is a unital magma and $G$ is a magma by
        \obv{commutative_group_is_group,group_is_monoid,
            monoid_is_unital_magma,monoid_is_magma}.
    $\carrier[M]=\carrier[G]$ and $\mul[M]=\mul[G]$ and
        $\neutral[M]=\neutral[G]$ and
        $\groupInv[M]=\groupInv[G]$ and
        $\scalarMul[M]=\ringMul[R]$ by
        \auto{left_regular_module_fields}.
    $\mul[M]\in
        \funs{\carrier[M]\times\carrier[M]}{\carrier[M]}$ by
        \auto{magma_mul_graph}.
    $\neutral[M]\in\carrier[M]$ by \auto{unitalmagma_type}.
    $\groupInv[M]\in\funs{\carrier[M]}{\carrier[M]}$ by
        \auto{group_inverse_graph}.
    For all $x,y,z\in\carrier[M]$ we have
        $\mul[M](x,\mul[M](y,z))=\mul[M](\mul[M](x,y),z)$ by
        \auto{monoid_assoc}.
    For every $x\in\carrier[M]$ we have
        $\mul[M](x,\neutral[M])=x$ and
        $\mul[M](\neutral[M],x)=x$ by
        \auto{unitalmagma_right,unitalmagma_left}.
    For every $x\in\carrier[M]$ we have
        $\mul[M](x,\groupInv[M](x))=\neutral[M]$ and
        $\mul[M](\groupInv[M](x),x)=\neutral[M]$ by
        \auto{group_inverse_right,group_inverse_left}.
    For all $x,y\in\carrier[M]$ we have
        $\mul[M](x,y)=\mul[M](y,x)$ by
        \auto{commutative_group_commutative}.
    $\scalarMul[M]\in
        \funs{\carrier[R]\times\carrier[M]}{\carrier[M]}$ by
        \auto{ring_mul_graph,additive_group_fields}.
    For all $a,x,y$ such that
        $a\in\carrier[R]$ and $x,y\in\carrier[M]$ we have
        $\scalarMul[M](a,\mul[M](x,y))
            =\mul[M](\scalarMul[M](a,x),\scalarMul[M](a,y))$ by
        \auto{additive_group_fields,ring_distributive_left}.
    For all $a,b,x$ such that
        $a,b\in\carrier[R]$ and $x\in\carrier[M]$ we have
        $\scalarMul[M](\ringAdd[R](a,b),x)
            =\mul[M](\scalarMul[M](a,x),\scalarMul[M](b,x))$ by
        \auto{additive_group_fields,ring_distributive_right}.
    For all $a,b,x$ such that
        $a,b\in\carrier[R]$ and $x\in\carrier[M]$ we have
        $\scalarMul[M](a,\scalarMul[M](b,x))
            =\scalarMul[M](\ringMul[R](a,b),x)$ by
        \auto{additive_group_fields,ring_mul_associative}.
    For every $x\in\carrier[M]$ we have
        $\scalarMul[M](\ringOne[R],x)=x$ by
        \auto{additive_group_fields,ring_mul_one_left}.
    It is not the case that it is not the case that
        $M$ is a left module over $R$ by
        \auto{left_regular_module,left_moduleaggregateexact,left_module}.
    Hence $M$ is a left module over $R$.
    Follows by
        \auto{left_regular_module,left_moduleaggregateexact}.
\end{proof}

\begin{theorem}\label{left_regular_module_additive_group}
    Let $R$ be a ring.
    Then
        $\ModuleAdditiveGroup{\LeftRegularModule{R}}
            =\AdditiveGroup{R}$.
\end{theorem}
\begin{proof}
    Let $M=\LeftRegularModule{R}$ and $G=\AdditiveGroup{R}$.
    $M$ is an exact left module over $R$ by
        \auto{left_regular_module_is_exact}.
    Hence $\ModuleAdditiveGroup{M}$ is an exact group by
        \obv{module_additive_group_is_exact_commutative_group,
            commutative_group_is_group}.
    $G$ is an exact group by
        \auto{additive_group_is_exact_commutative_group,
            commutative_group_is_group}.
    $\carrier[\ModuleAdditiveGroup{M}]=\carrier[G]$ and
        $\mul[\ModuleAdditiveGroup{M}]=\mul[G]$ and
        $\neutral[\ModuleAdditiveGroup{M}]=\neutral[G]$ and
        $\groupInv[\ModuleAdditiveGroup{M}]=\groupInv[G]$ by
        \auto{module_additive_group_fields,left_regular_module_fields}.
    Follows by \auto{exact_group_extensionality}.
\end{proof}

\begin{definition}\label{right_regular_module}
    $\RightRegularModule{R}=\LeftRegularModule{\OppositeRing{R}}$.
\end{definition}

\begin{theorem}\label{right_regular_module_is_exact}
    Let $R$ be a ring.
    Then $\RightRegularModule{R}$ is an exact left module over
        $\OppositeRing{R}$.
\end{theorem}
\begin{proof}
    $\OppositeRing{R}$ is an exact ring by
        \obv{opposite_ring_is_exact}.
    Hence $\OppositeRing{R}$ is a ring.
    Follows by
        \auto{right_regular_module,left_regular_module_is_exact}.
\end{proof}

\begin{proposition}\label{right_regular_module_is_right_module}
    Let $R$ be a ring.
    Then $\RightRegularModule{R}$ is a right module over $R$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{right_module,right_regular_module_is_exact}.
\end{proof}

\begin{proposition}\label{right_regular_module_carrier}
    $\carrier[\RightRegularModule{R}]=\carrier[R]$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{right_regular_module,left_regular_module_carrier,
            opposite_ring_fields}.
\end{proof}

\begin{proposition}\label{right_regular_scalar_apply}
    Let $R$ be a ring and $x,a\in\carrier[R]$.
    Then
        $\rightScalarMul{\RightRegularModule{R}}{x}{a}
            =\ringMul[R](x,a)$.
\end{proposition}
\begin{proof}
    $\OppositeRing{R}$ is an exact ring by
        \obv{opposite_ring_is_exact}.
    $x,a\in\carrier[\OppositeRing{R}]$ by
        \auto{opposite_ring_fields}.
    $\rightScalarMul{\RightRegularModule{R}}{x}{a}
        =\scalarMul[\RightRegularModule{R}](a,x)$ by
        \obv{right_scalar_multiplication}.
    $\scalarMul[\RightRegularModule{R}](a,x)
        =\scalarMul[\LeftRegularModule{\OppositeRing{R}}](a,x)$ by
        \auto{right_regular_module}.
    Hence $\scalarMul[\RightRegularModule{R}](a,x)
        =\ringMul[\OppositeRing{R}](a,x)$ by
        \obv{left_regular_scalar_apply}.
    $\ringMul[\OppositeRing{R}](a,x)=\ringMul[R](x,a)$ by
        \obv{opposite_ring_mul_apply}.
\end{proof}

\begin{theorem}\label{right_regular_module_additive_group}
    Let $R$ be a ring.
    Then
        $\ModuleAdditiveGroup{\RightRegularModule{R}}
            =\AdditiveGroup{R}$.
\end{theorem}
\begin{proof}
    $\OppositeRing{R}$ is an exact ring by
        \obv{opposite_ring_is_exact}.
    $\ModuleAdditiveGroup{\RightRegularModule{R}}
        =\AdditiveGroup{\OppositeRing{R}}$ by
        \auto{right_regular_module,left_regular_module_additive_group}.
    $\AdditiveGroup{\OppositeRing{R}}$ is an exact group by
        \obv{opposite_ring_is_exact,
            additive_group_is_exact_commutative_group,
            commutative_group_is_group}.
    $\AdditiveGroup{R}$ is an exact group by
        \obv{additive_group_is_exact_commutative_group,
            commutative_group_is_group}.
    $\carrier[\AdditiveGroup{\OppositeRing{R}}]
        =\carrier[\AdditiveGroup{R}]$ and
        $\mul[\AdditiveGroup{\OppositeRing{R}}]
        =\mul[\AdditiveGroup{R}]$ and
        $\neutral[\AdditiveGroup{\OppositeRing{R}}]
        =\neutral[\AdditiveGroup{R}]$ and
        $\groupInv[\AdditiveGroup{\OppositeRing{R}}]
        =\groupInv[\AdditiveGroup{R}]$ by
        \auto{additive_group_fields,opposite_ring_fields}.
    Hence $\AdditiveGroup{\OppositeRing{R}}=\AdditiveGroup{R}$ by
        \obv{exact_group_extensionality}.
\end{proof}
