\section{Ordered rings and fields}

The primitive order carried by every structure in this section is non-strict.
The corresponding strict relation is obtained by deleting its diagonal.  This
keeps the record interface uniform with ordered sets and avoids storing two
order relations whose agreement would have to be maintained separately.

\begin{definition}\label{ordered_algebra_strict_relation}
    $\ringStrictOrder[R]=\tostrictorder{\leq[R]}$.
\end{definition}

\begin{struct}\label{ordered_ring}
    An ordered ring $R$ is a ring and an ordered set such that
    \begin{enumerate}
        \item\label{ordered_ring_total}
            for all $x,y\in\carrier[R]$ we have
                $x\mathrel{\leq[R]}y$ or $y\mathrel{\leq[R]}x$.
        \item\label{ordered_ring_add_mono}
            for all $x,y,z\in\carrier[R]$ such that
                $x\mathrel{\leq[R]}y$ we have
                $\ringAdd[R](x,z)
                    \mathrel{\leq[R]}\ringAdd[R](y,z)$.
        \item\label{ordered_ring_mul_nonnegative}
            for all $x,y\in\carrier[R]$ such that
                $\ringZero[R]\mathrel{\leq[R]}x$ and
                $\ringZero[R]\mathrel{\leq[R]}y$ we have
                $\ringZero[R]\mathrel{\leq[R]}\ringMul[R](x,y)$.
    \end{enumerate}
\end{struct}

\begin{struct}\label{ordered_integral_domain}
    An ordered integral domain $R$ is an integral domain and an ordered ring.
\end{struct}

\begin{struct}\label{ordered_field}
    An ordered field $F$ is a field and an ordered ring.
\end{struct}

\begin{struct}\label{archimedean_ordered_field}
    An Archimedean ordered field $F$ is an ordered field such that
    \begin{enumerate}
        \item\label{archimedean_ordered_field_law}
            for all $x,y\in\carrier[F]$ such that
                $\ringZero[F]\mathrel{\ringStrictOrder[F]}x$ and
                $\ringZero[F]\mathrel{\leq[F]}y$
            there exists a natural number $n$ such that
                $y\mathrel{\leq[F]}
                    \ringMul[F](\ringNatural{F}{n},x)$.
    \end{enumerate}
\end{struct}

\begin{proposition}\label{archimedean_ordered_field_intro}
    Let $F$ be an ordered field.
    Suppose for all $x,y\in\carrier[F]$ such that
        $\ringZero[F]\mathrel{\ringStrictOrder[F]}x$ and
        $\ringZero[F]\mathrel{\leq[F]}y$
        there exists a natural number $n$ such that
        $y\mathrel{\leq[F]}\ringMul[F](\ringNatural{F}{n},x)$.
    Then $F$ is an Archimedean ordered field.
\end{proposition}
\begin{proof}
    Follows by definition.
\end{proof}

\begin{struct}\label{complete_ordered_field}
    A complete ordered field $F$ is an ordered field such that
    \begin{enumerate}
        \item\label{complete_ordered_field_law}
            for every $A\subseteq\carrier[F]$ such that $A$ is inhabited
            and there exists $b\in\carrier[F]$ such that for every $a\in A$
            we have $a\mathrel{\leq[F]}b$, there exists $s\in\carrier[F]$
            such that
            (for every $a\in A$ we have $a\mathrel{\leq[F]}s$) and
            (for every $u\in\carrier[F]$ such that for every $r\in A$
                we have $r\mathrel{\leq[F]}u$ we have
                $s\mathrel{\leq[F]}u$).
    \end{enumerate}
\end{struct}

\begin{proposition}\label{complete_ordered_field_intro}
    Let $F$ be an ordered field.
    Suppose for every $A\subseteq\carrier[F]$ such that $A$ is inhabited
        and there exists $b\in\carrier[F]$ such that for every $a\in A$
        we have $a\mathrel{\leq[F]}b$, there exists $s\in\carrier[F]$
        such that
        (for every $a\in A$ we have $a\mathrel{\leq[F]}s$) and
        (for every $u\in\carrier[F]$ such that for every $r\in A$
            we have $r\mathrel{\leq[F]}u$ we have
            $s\mathrel{\leq[F]}u$).
    Then $F$ is a complete ordered field.
\end{proposition}
\begin{proof}
    Follows by definition.
\end{proof}

\begin{struct}\label{complete_archimedean_ordered_field}
    A complete Archimedean ordered field $F$ is an Archimedean ordered field
    and a complete ordered field.
\end{struct}

\begin{proposition}\label{complete_archimedean_ordered_field_intro}
    Let $F$ be an Archimedean ordered field.
    Suppose $F$ is a complete ordered field.
    Then $F$ is a complete Archimedean ordered field.
\end{proposition}
\begin{proof}
    Follows by definition.
\end{proof}

\begin{corollary}\label{ordered_integral_domain_is_integral_domain}
    Let $R$ be an ordered integral domain.
    Then $R$ is an integral domain.
\end{corollary}
\begin{proof}
    Follows by \obv{ordered_integral_domain}.
\end{proof}

\begin{corollary}\label{ordered_field_is_field}
    Let $F$ be an ordered field.
    Then $F$ is a field.
\end{corollary}
\begin{proof}
    Follows by \obv{ordered_field}.
\end{proof}

\begin{corollary}\label{complete_archimedean_ordered_field_is_archimedean}
    Let $F$ be a complete Archimedean ordered field.
    Then $F$ is an Archimedean ordered field.
\end{corollary}
\begin{proof}
    Follows by \obv{complete_archimedean_ordered_field}.
\end{proof}

\begin{corollary}\label{complete_archimedean_ordered_field_is_complete}
    Let $F$ be a complete Archimedean ordered field.
    Then $F$ is a complete ordered field.
\end{corollary}
\begin{proof}
    Follows by \obv{complete_archimedean_ordered_field}.
\end{proof}

\begin{theorem}\label{ordered_field_is_linearly_ordered_set}
    Let $F$ be an ordered field.
    Then $F$ is a linearly ordered set.
\end{theorem}
\begin{proof}
    $F$ is an ordered ring by \obv{ordered_field}.
    For all $x,y\in\carrier[F]$ we have
        $x\mathrel{\leq[F]}y$ or $y\mathrel{\leq[F]}x$ by
        \obv{ordered_ring_total}.
    Follows by \obv{linearly_ordered_set}.
\end{proof}

\begin{theorem}\label{complete_ordered_field_is_conditionally_complete}
    Let $F$ be a complete ordered field.
    Then $F$ is a conditionally complete linearly ordered set.
\end{theorem}
\begin{proof}
    Show that for every $S\subseteq\carrier[F]$ such that
        $S$ is inhabited and $S$ is bounded above in $F$ we have
        $\hasSupremum{F}{S}$.
    \begin{subproof}
        Fix $S\subseteq\carrier[F]$.
        Assume $S$ is inhabited and $S$ is bounded above in $F$.
        Take $b\in\carrier[F]$ such that for every $s\in S$ we have
            $s\mathrel{\leq[F]}b$ by
            \obv{bounded_above_in_ordered_set}.
        Take $u\in\carrier[F]$ such that
            (for every $s\in S$ we have $s\mathrel{\leq[F]}u$) and
            (for every $v\in\carrier[F]$ such that for every $s\in S$
                we have $s\mathrel{\leq[F]}v$ we have
                $u\mathrel{\leq[F]}v$) by
            \obv{complete_ordered_field_law}.
        $\leq[F]$ is an order on $\carrier[F]$ by
            \obv{ordered_set_order_on_carrier}.
        Hence $u$ is a least upper bound of $S$ in $\carrier[F]$
            with respect to $\leq[F]$ by
            \auto{least_upper_bound_iff_order_on}.
        Thus $u$ is a supremum of $S$ in $F$ by
            \obv{supremum_element}.
        Follows by \obv{has_supremum_iff}.
    \end{subproof}
    Follows by \obv{conditionally_complete_linear_order}.
\end{proof}
