\import{algebra/monoid.tex}

\section{Monoid actions}

Actions are ordinary set-coded function graphs.  The acting monoid and the
acted-on set remain explicit parameters, so the same graph can be used in
different compatible contexts without hidden endpoint data.

\begin{abbreviation}\label{action_apply}
    $\actionApply{p}{a}{x}=p((a,x))$.
\end{abbreviation}

\begin{definition}\label{monoid_action}
    $\monoidAction{p}{M}{X}$ iff
        $M$ is a monoid and
        $p\in\funs{\carrier[M]\times X}{X}$ and
        (for every $x\in X$ we have
            $\actionApply{p}{\neutral[M]}{x}=x$) and
        (for all $a,b\in\carrier[M]$ we have for every $x\in X$
            $\actionApply{p}{\mul[M](a,b)}{x}=
                \actionApply{p}{a}{\actionApply{p}{b}{x}}$).
\end{definition}

\begin{proposition}\label{monoid_action_monoid}
    Suppose $\monoidAction{p}{M}{X}$.
    Then $M$ is a monoid.
\end{proposition}
\begin{proof}
    Follows by \obv{monoid_action}.
\end{proof}

\begin{proposition}\label{monoid_action_type}
    Suppose $\monoidAction{p}{M}{X}$.
    Then $p\in\funs{\carrier[M]\times X}{X}$.
\end{proposition}
\begin{proof}
    Follows by \obv{monoid_action}.
\end{proof}

\begin{proposition}\label{monoid_action_apply_type}
    Suppose $\monoidAction{p}{M}{X}$.
    Let $a\in\carrier[M]$.
    Let $x\in X$.
    Then $\actionApply{p}{a}{x}\in X$.
\end{proposition}
\begin{proof}
    $(a,x)\in\carrier[M]\times X$ by
        \obv{times_tuple_intro}.
    Follows by
        \obv{monoid_action_type,funs_type_apply}.
\end{proof}

\begin{proposition}\label{monoid_action_neutral}
    Suppose $\monoidAction{p}{M}{X}$.
    Let $x\in X$.
    Then $\actionApply{p}{\neutral[M]}{x}=x$.
\end{proposition}
\begin{proof}
    Follows by \obv{monoid_action}.
\end{proof}

\begin{proposition}\label{monoid_action_mul}
    Suppose $\monoidAction{p}{M}{X}$.
    Let $a,b\in\carrier[M]$.
    Let $x\in X$.
    Then
        $\actionApply{p}{\mul[M](a,b)}{x}=
            \actionApply{p}{a}{\actionApply{p}{b}{x}}$.
\end{proposition}
\begin{proof}
    Follows by \obv{monoid_action}.
\end{proof}

\begin{definition}\label{action_orbit}
    $\actionOrbit{p}{M}{X}{x}=
        \{y\in X\mid\text{$x\in X$ and there exists
            $a\in\carrier[M]$ such that
            $\actionApply{p}{a}{x}=y$}\}$.
\end{definition}

\begin{proposition}\label{action_orbit_iff}
    Suppose $\monoidAction{p}{M}{X}$ and $x\in X$.
    Then $y\in\actionOrbit{p}{M}{X}{x}$ iff
        there exists $a\in\carrier[M]$ such that
            $\actionApply{p}{a}{x}=y$.
\end{proposition}
\begin{proof}
    Show that if $y\in\actionOrbit{p}{M}{X}{x}$, then
        there exists $a\in\carrier[M]$ such that
            $\actionApply{p}{a}{x}=y$.
    \begin{subproof}
        Assume $y\in\actionOrbit{p}{M}{X}{x}$.
        Follows by \obv{action_orbit}.
    \end{subproof}
    Show that if there exists $a\in\carrier[M]$ such that
        $\actionApply{p}{a}{x}=y$, then
        $y\in\actionOrbit{p}{M}{X}{x}$.
    \begin{subproof}
        Assume that there exists $a\in\carrier[M]$ such that
            $\actionApply{p}{a}{x}=y$.
        Take $a\in\carrier[M]$ such that
            $\actionApply{p}{a}{x}=y$.
        $\actionApply{p}{a}{x}\in X$ by
            \obv{monoid_action_apply_type}.
        Follows by \obv{action_orbit}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{proposition}\label{action_orbit_subseteq}
    $\actionOrbit{p}{M}{X}{x}\subseteq X$.
\end{proposition}
\begin{proof}
    Follows by \obv{action_orbit,subseteq}.
\end{proof}

\begin{proposition}\label{action_orbit_contains_point}
    Suppose $\monoidAction{p}{M}{X}$ and $x\in X$.
    Then $x\in\actionOrbit{p}{M}{X}{x}$.
\end{proposition}
\begin{proof}
    $\neutral[M]\in\carrier[M]$ by
        \obv{monoid_action_monoid,monoid_is_unital_magma,unitalmagma_type}.
    $\actionApply{p}{\neutral[M]}{x}=x$ by
        \obv{monoid_action_neutral}.
    Follows by \obv{action_orbit_iff}.
\end{proof}

\begin{proposition}\label{action_orbit_off_carrier}
    Suppose $x\notin X$.
    Then $\actionOrbit{p}{M}{X}{x}=\emptyset$.
\end{proposition}
\begin{proof}
    Follows by \auto{action_orbit,setext}.
\end{proof}
