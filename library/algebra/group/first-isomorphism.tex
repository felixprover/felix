\section{The first isomorphism theorem}

The induced graph is the universal quotient lift at the kernel of the
homomorphism.  Its target is the canonical exact group on the actual image
carrier, so no ambient codomain is silently recovered from the graph.

\begin{definition}\label{first_isomorphism_map}
    $\firstIsomorphismMap{f}{G}{H}=
        \quotientDescent
            {f}{\carrier[G]}
            {\subgroupCosetRelation{G}{\groupKernel{f}{G}}}$.
\end{definition}

\begin{proposition}\label{first_isomorphism_map_eq_group_quotient_lift}
    $\firstIsomorphismMap{f}{G}{H}=
        \groupQuotientLift{f}{G}{\groupKernel{f}{G}}$.
\end{proposition}
\begin{proof}
    Follows by
        \obv{first_isomorphism_map,group_quotient_lift}.
\end{proof}

\begin{proposition}\label{group_hom_image_type}
    Suppose $\groupHom{f}{G}{H}$.
    Then $f\in
        \funs{\carrier[G]}{\img{f}{\carrier[G]}}$.
\end{proposition}
\begin{proof}
    $f\in\funs{\carrier[G]}{\carrier[H]}$ by
        \obv{group_hom_type}.
    Hence $f\in\surj{\carrier[G]}{\ran{f}}$ by
        \obv{fun_to_surj}.
    $\dom{f}=\carrier[G]$ by \obv{group_hom_type,funs}.
    Hence $\ran{f}=\img{f}{\carrier[G]}$ by \auto{img_dom}.
    Follows by \obv{surj_to_fun}.
\end{proof}

\begin{proposition}\label{first_isomorphism_map_type}
    Suppose $\groupHom{f}{G}{H}$.
    Then $\firstIsomorphismMap{f}{G}{H}\in
        \funs{
            \groupQuotientCarrier{G}{\groupKernel{f}{G}}}
            {\img{f}{\carrier[G]}}$.
\end{proposition}
\begin{proof}
    Let $N=\groupKernel{f}{G}$ and
        $I=\img{f}{\carrier[G]}$.
    Let $Q=\quotientGroup{G}{N}$ and
        $J=\subgroupStructure{H}{I}$.
    $\normalSubgroup{N}{G}$ by \auto{group_kernel_normal_subgroup}.
    $\subgroup{I}{H}$ by \auto{group_hom_image_subgroup}.
    $I\subseteq I$ by \obv{subseteq_refl}.
    Hence $\groupHom{f}{G}{J}$ by
        \obv{group_hom_codomain_corestriction}.
    $N\subseteq\groupKernel{f}{G}$ by \obv{subseteq_refl}.
    Hence $\groupQuotientLift{f}{G}{N}\in
        \funs{\carrier[Q]}{\carrier[J]}$ by
        \auto{group_quotient_lift_type}.
    $\carrier[Q]=\groupQuotientCarrier{G}{N}$ and
        $\carrier[J]=I$ by
        \auto{quotient_group_carrier,subgroup_structure_carrier}.
    $\firstIsomorphismMap{f}{G}{H}=
        \groupQuotientLift{f}{G}{N}$ by
        \obv{first_isomorphism_map_eq_group_quotient_lift}.
\end{proof}

\begin{proposition}\label{first_isomorphism_map_apply}
    Suppose $\groupHom{f}{G}{H}$.
    Let $a\in\carrier[G]$.
    Then
        $\apply{\firstIsomorphismMap{f}{G}{H}}
            {\equivalenceClass
                {\subgroupCosetRelation{G}{\groupKernel{f}{G}}}
                {a}}=f(a)$.
\end{proposition}
\begin{proof}
    Let $N=\groupKernel{f}{G}$.
    $\normalSubgroup{N}{G}$ by \auto{group_kernel_normal_subgroup}.
    $N\subseteq\groupKernel{f}{G}$ by \obv{subseteq_refl}.
    $\apply{\groupQuotientLift{f}{G}{N}}
        {\equivalenceClass{\subgroupCosetRelation{G}{N}}{a}}
        =f(a)$ by \obv{group_quotient_lift_apply}.
    $\firstIsomorphismMap{f}{G}{H}=
        \groupQuotientLift{f}{G}{N}$ by
        \obv{first_isomorphism_map_eq_group_quotient_lift}.
\end{proof}

\begin{theorem}\label{first_isomorphism_map_factorization}
    Suppose $\groupHom{f}{G}{H}$.
    Then
        $\firstIsomorphismMap{f}{G}{H}\circ
            \groupQuotientProjection
                {G}{\groupKernel{f}{G}}=f$.
\end{theorem}
\begin{proof}
    Let $N=\groupKernel{f}{G}$.
    $\normalSubgroup{N}{G}$ by \auto{group_kernel_normal_subgroup}.
    $N\subseteq\groupKernel{f}{G}$ by \obv{subseteq_refl}.
    $\groupQuotientLift{f}{G}{N}\circ
        \groupQuotientProjection{G}{N}=f$ by
        \obv{group_quotient_lift_factorization}.
    $\firstIsomorphismMap{f}{G}{H}=
        \groupQuotientLift{f}{G}{N}$ by
        \obv{first_isomorphism_map_eq_group_quotient_lift}.
\end{proof}

\begin{theorem}\label{first_isomorphism_map_unique}
    Suppose $\groupHom{f}{G}{H}$.
    Let
        $h\in
            \funs{
                \groupQuotientCarrier{G}{\groupKernel{f}{G}}}
                {\img{f}{\carrier[G]}}$.
    Suppose
        $h\circ
            \groupQuotientProjection
                {G}{\groupKernel{f}{G}}=f$.
    Then $h=\firstIsomorphismMap{f}{G}{H}$.
\end{theorem}
\begin{proof}
    Let $N=\groupKernel{f}{G}$ and
        $I=\img{f}{\carrier[G]}$.
    Let $Q=\quotientGroup{G}{N}$ and
        $J=\subgroupStructure{H}{I}$.
    $\normalSubgroup{N}{G}$ by \auto{group_kernel_normal_subgroup}.
    $\subgroup{I}{H}$ by \auto{group_hom_image_subgroup}.
    $I\subseteq I$ by \obv{subseteq_refl}.
    Hence $\groupHom{f}{G}{J}$ by
        \obv{group_hom_codomain_corestriction}.
    $N\subseteq\groupKernel{f}{G}$ by \obv{subseteq_refl}.
    $\carrier[Q]=\groupQuotientCarrier{G}{N}$ and
        $\carrier[J]=I$ by
        \auto{quotient_group_carrier,subgroup_structure_carrier}.
    Hence $h\in\funs{\carrier[Q]}{\carrier[J]}$.
    Hence $h=\groupQuotientLift{f}{G}{N}$ by
        \auto{group_quotient_lift_unique}.
    $\firstIsomorphismMap{f}{G}{H}=
        \groupQuotientLift{f}{G}{N}$ by
        \obv{first_isomorphism_map_eq_group_quotient_lift}.
\end{proof}

\begin{theorem}\label{first_isomorphism_map_hom}
    Suppose $\groupHom{f}{G}{H}$.
    Then
        $\groupHom
            {\firstIsomorphismMap{f}{G}{H}}
            {\quotientGroup{G}{\groupKernel{f}{G}}}
            {\subgroupStructure{H}{\img{f}{\carrier[G]}}}$.
\end{theorem}
\begin{proof}
    Let $N=\groupKernel{f}{G}$ and
        $I=\img{f}{\carrier[G]}$.
    Let $J=\subgroupStructure{H}{I}$.
    $\normalSubgroup{N}{G}$ by \auto{group_kernel_normal_subgroup}.
    $\subgroup{I}{H}$ by \auto{group_hom_image_subgroup}.
    $I\subseteq I$ by \obv{subseteq_refl}.
    Hence $\groupHom{f}{G}{J}$ by
        \obv{group_hom_codomain_corestriction}.
    $N\subseteq\groupKernel{f}{G}$ by \obv{subseteq_refl}.
    Hence
        $\groupHom
            {\groupQuotientLift{f}{G}{N}}
            {\quotientGroup{G}{N}}{J}$ by
        \obv{group_quotient_lift_hom}.
    $\firstIsomorphismMap{f}{G}{H}=
        \groupQuotientLift{f}{G}{N}$ by
        \obv{first_isomorphism_map_eq_group_quotient_lift}.
\end{proof}

\begin{theorem}\label{first_isomorphism_map_bijection}
    Suppose $\groupHom{f}{G}{H}$.
    Then $\firstIsomorphismMap{f}{G}{H}\in
        \bijections{
            \groupQuotientCarrier{G}{\groupKernel{f}{G}}}
            {\img{f}{\carrier[G]}}$.
\end{theorem}
\begin{proof}
    Let $N=\groupKernel{f}{G}$ and
        $I=\img{f}{\carrier[G]}$.
    Let $Q=\quotientGroup{G}{N}$ and
        $J=\subgroupStructure{H}{I}$.
    Let $g=\groupQuotientLift{f}{G}{N}$.
    $\normalSubgroup{N}{G}$ by \auto{group_kernel_normal_subgroup}.
    $\subgroup{I}{H}$ by \auto{group_hom_image_subgroup}.
    $I\subseteq I$ by \obv{subseteq_refl}.
    Hence $\groupHom{f}{G}{J}$ by
        \obv{group_hom_codomain_corestriction}.
    $N\subseteq\groupKernel{f}{G}$ by \obv{subseteq_refl}.
    Hence $g$ is injective by
        \auto{group_quotient_lift_injective_iff}.
    $\dom{f}=\carrier[G]$ by \obv{group_hom_type,funs_elim}.
    Hence $\ran{f}=I$ by \auto{img_dom}.
    $\carrier[J]=I$ by \auto{subgroup_structure_carrier}.
    Hence $f\in\surj{\carrier[G]}{\carrier[J]}$ by
        \auto{group_hom_type,funs_surj_iff}.
    Hence $g\in\surj{\carrier[Q]}{\carrier[J]}$ by
        \auto{group_quotient_lift_surjective_iff}.
    $g\in\funs{\carrier[Q]}{\carrier[J]}$ by
        \obv{group_quotient_lift_type}.
    $\carrier[Q]=\groupQuotientCarrier{G}{N}$ by
        \auto{quotient_group_carrier}.
    $\firstIsomorphismMap{f}{G}{H}=g$ by
        \obv{first_isomorphism_map_eq_group_quotient_lift}.
    Follows by \auto{bijections}.
\end{proof}

\begin{theorem}\label{first_isomorphism_theorem}
    Suppose $\groupHom{f}{G}{H}$.
    Then
        $\groupIsomorphism
            {\firstIsomorphismMap{f}{G}{H}}
            {\quotientGroup{G}{\groupKernel{f}{G}}}
            {\subgroupStructure{H}{\img{f}{\carrier[G]}}}$.
\end{theorem}
\begin{proof}
    Let $K=\quotientGroup{G}{\groupKernel{f}{G}}$.
    Let $J=\subgroupStructure{H}{\img{f}{\carrier[G]}}$.
    Let $g=\firstIsomorphismMap{f}{G}{H}$.
    $\groupHom{g}{K}{J}$ by \auto{first_isomorphism_map_hom}.
    $\carrier[K]=
        \groupQuotientCarrier{G}{\groupKernel{f}{G}}$ and
        $\carrier[J]=\img{f}{\carrier[G]}$ by
        \auto{quotient_group_carrier,subgroup_structure_carrier}.
    Hence $g\in\bijections{\carrier[K]}{\carrier[J]}$ by
        \auto{first_isomorphism_map_bijection}.
    Follows by \obv{group_isomorphism}.
\end{proof}
