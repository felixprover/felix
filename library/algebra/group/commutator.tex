\section{Commutators and the commutator subgroup}

We use the convention
\( [x,y]=x^{-1}y^{-1}xy \).  The commutator subgroup is the subgroup
generated by all such elements in one explicit ambient group.  Thus the
derived subgroup is textbook terminology for this same carrier, not a second
construction.

\begin{definition}\label{group_commutator}
    $\groupCommutator{G}{x}{y}=
        \mul[G](
            \mul[G](
                \mul[G](\groupInv[G](x),\groupInv[G](y)),x),y)$.
\end{definition}

\begin{proposition}\label{group_commutator_type}
    Let $G$ be a group.
    Let $x,y\in\carrier[G]$.
    Then $\groupCommutator{G}{x}{y}\in\carrier[G]$.
\end{proposition}
\begin{proof}
    $\groupInv[G](x),\groupInv[G](y)\in\carrier[G]$ by
        \obv{group_inverse_type}.
    Follows by
        \auto{group_commutator,group_is_monoid,monoid_is_magma,
            magma_welldef}.
\end{proof}

\begin{proposition}\label{group_commutator_eq_inverse_mul_conjugate}
    Let $G$ be a group.
    Let $x,y\in\carrier[G]$.
    Then
        $\groupCommutator{G}{x}{y}=
            \mul[G](\groupInv[G](x),
                \conjugate{G}{\groupInv[G](y)}{x})$.
\end{proposition}
\begin{proof}
    Let $u=\groupInv[G](x)$ and $v=\groupInv[G](y)$.
    $u,v\in\carrier[G]$ by \auto{group_inverse_type}.
    $\groupInv[G](\groupInv[G](y))=y$ by
        \obv{group_inverse_involutive}.
    Hence
        $\conjugate{G}{v}{x}=\mul[G](\mul[G](v,x),y)$ by
        \obv{group_conjugate}.
    $\mul[G](v,x),\mul[G](u,v)\in\carrier[G]$ by
        \obv{group_is_monoid,monoid_is_magma,magma_welldef}.
    Hence
        $\mul[G](u,\conjugate{G}{v}{x})=
            \mul[G](\mul[G](\mul[G](u,v),x),y)$ by
        \auto{group_is_monoid,monoid_assoc}.
    Follows by \obv{group_commutator}.
\end{proof}

\begin{corollary}\label{group_conjugate_inverse_eq_mul_commutator}
    Let $G$ be a group.
    Let $x,y\in\carrier[G]$.
    Then
        $\conjugate{G}{\groupInv[G](y)}{x}=
            \mul[G](x,\groupCommutator{G}{x}{y})$.
\end{corollary}
\begin{proof}
    Let $u=\groupInv[G](x)$ and
        $c=\conjugate{G}{\groupInv[G](y)}{x}$.
    $u,c,\groupCommutator{G}{x}{y}\in\carrier[G]$ by
        \auto{group_inverse_type,group_conjugate_type,
            group_commutator_type}.
    $\groupCommutator{G}{x}{y}=\mul[G](u,c)$ by
        \auto{group_commutator_eq_inverse_mul_conjugate}.
    Hence
        $c=\mul[G](\groupInv[G](u),
            \groupCommutator{G}{x}{y})$ by
        \auto{group_solve_left}.
    $\groupInv[G](u)=x$ by
        \auto{group_inverse_involutive}.
\end{proof}

\begin{theorem}\label{group_commutator_inverse}
    Let $G$ be a group.
    Let $x,y\in\carrier[G]$.
    Then
        $\groupInv[G](\groupCommutator{G}{x}{y})=
            \groupCommutator{G}{y}{x}$.
\end{theorem}
\begin{proof}
    Let $u=\groupInv[G](x)$ and $v=\groupInv[G](y)$.
    $u,v\in\carrier[G]$ by \auto{group_inverse_type}.
    $\mul[G](u,v),\mul[G](\mul[G](u,v),x)\in\carrier[G]$ by
        \obv{group_is_monoid,monoid_is_magma,magma_welldef}.
    $\groupInv[G](\mul[G](u,v))=
        \mul[G](\groupInv[G](v),\groupInv[G](u))$ by
        \obv{group_inverse_product}.
    Hence $\groupInv[G](\mul[G](u,v))=\mul[G](y,x)$ by
        \auto{group_inverse_involutive}.
    $\groupInv[G](\mul[G](\mul[G](u,v),x))=
        \mul[G](\groupInv[G](x),\mul[G](y,x))$ by
        \auto{group_inverse_product}.
    Hence
        $\groupInv[G](\groupCommutator{G}{x}{y})=
            \mul[G](v,
                \mul[G](u,\mul[G](y,x)))$ by
        \auto{group_commutator,group_inverse_product}.
    $\mul[G](y,x),\mul[G](u,\mul[G](y,x)),
        \mul[G](v,u),\mul[G](\mul[G](v,u),y)\in\carrier[G]$ by
        \obv{group_is_monoid,monoid_is_magma,magma_welldef}.
    $\mul[G](v,\mul[G](u,\mul[G](y,x)))=
        \mul[G](\mul[G](\mul[G](v,u),y),x)$ by
        \auto{group_is_monoid,monoid_assoc}.
    Follows by \obv{group_commutator}.
\end{proof}

\begin{theorem}\label{group_commutator_mul_left}
    Let $G$ be a group.
    Let $x,y,z\in\carrier[G]$.
    Then
        $\groupCommutator{G}{\mul[G](x,y)}{z}=
            \mul[G](
                \conjugate{G}{\groupInv[G](y)}{
                    \groupCommutator{G}{x}{z}},
                \groupCommutator{G}{y}{z})$.
\end{theorem}
\begin{proof}
    $\groupInv[G](x),\groupInv[G](y),\groupInv[G](z)\in\carrier[G]$
        by \obv{group_inverse_type}.
    $\mul[G](x,y)\in\carrier[G]$ by
        \obv{group_is_monoid,monoid_is_magma,magma_welldef}.
    $\groupInv[G](\mul[G](x,y))=
        \mul[G](\groupInv[G](y),\groupInv[G](x))$ by
        \obv{group_inverse_product}.
    $\groupInv[G](\groupInv[G](y))=y$ by
        \obv{group_inverse_involutive}.
    $\mul[G](z,\groupInv[G](z))=\neutral[G]$ and
        $\mul[G](y,\groupInv[G](y))=\neutral[G]$ by
        \obv{group_inverse_right}.
    \begin{align*}
        \groupCommutator{G}{\mul[G](x,y)}{z}
            &= \mul[G](\groupInv[G](y),
                \mul[G](\groupInv[G](x),
                    \mul[G](\groupInv[G](z),
                        \mul[G](x,\mul[G](y,z)))))
                \explanation{by
                    \auto{group_commutator,group_is_monoid,
                        monoid_assoc,monoid_is_magma,magma_welldef}}
            \\
            &= \mul[G](\groupInv[G](y),
                \mul[G](\groupInv[G](x),
                    \mul[G](\groupInv[G](z),
                        \mul[G](x,
                            \mul[G](z,
                                \mul[G](\groupInv[G](z),
                                    \mul[G](y,z)))))))
                \explanation{by
                    \auto{group_is_monoid,monoid_assoc,
                        monoid_is_magma,magma_welldef,
                        monoid_is_unital_magma,unitalmagma_left,
                        unitalmagma_right}}
            \\
            &= \mul[G](\groupInv[G](y),
                \mul[G](\groupInv[G](x),
                    \mul[G](\groupInv[G](z),
                        \mul[G](x,
                            \mul[G](z,
                                \mul[G](y,
                                    \mul[G](\groupInv[G](y),
                                        \mul[G](\groupInv[G](z),
                                            \mul[G](y,z)))))))))
                \explanation{by
                    \auto{group_is_monoid,monoid_assoc,
                        monoid_is_magma,magma_welldef,
                        monoid_is_unital_magma,unitalmagma_left,
                        unitalmagma_right}}
            \\
            &= \mul[G](
                \conjugate{G}{\groupInv[G](y)}{
                    \groupCommutator{G}{x}{z}},
                \groupCommutator{G}{y}{z})
                \explanation{by
                    \auto{group_conjugate,group_commutator,
                        group_is_monoid,monoid_assoc,
                        monoid_is_magma,magma_welldef}}
    \end{align*}
\end{proof}

\begin{theorem}\label{group_commutator_mul_right}
    Let $G$ be a group.
    Let $x,y,z\in\carrier[G]$.
    Then
        $\groupCommutator{G}{x}{\mul[G](y,z)}=
            \mul[G](
                \groupCommutator{G}{x}{z},
                \conjugate{G}{\groupInv[G](z)}{
                    \groupCommutator{G}{x}{y}})$.
\end{theorem}
\begin{proof}
    $\groupInv[G](x),\groupInv[G](y),\groupInv[G](z)\in\carrier[G]$
        by \obv{group_inverse_type}.
    $\mul[G](y,z)\in\carrier[G]$ by
        \obv{group_is_monoid,monoid_is_magma,magma_welldef}.
    $\groupInv[G](\mul[G](y,z))=
        \mul[G](\groupInv[G](z),\groupInv[G](y))$ by
        \obv{group_inverse_product}.
    $\groupInv[G](\groupInv[G](z))=z$ by
        \obv{group_inverse_involutive}.
    $\mul[G](x,\groupInv[G](x))=\neutral[G]$ and
        $\mul[G](z,\groupInv[G](z))=\neutral[G]$ by
        \obv{group_inverse_right}.
    \begin{align*}
        \groupCommutator{G}{x}{\mul[G](y,z)}
            &= \mul[G](\groupInv[G](x),
                \mul[G](\groupInv[G](z),
                    \mul[G](\groupInv[G](y),
                        \mul[G](x,\mul[G](y,z)))))
                \explanation{by
                    \auto{group_commutator,group_is_monoid,
                        monoid_assoc,monoid_is_magma,magma_welldef}}
            \\
            &= \mul[G](\groupInv[G](x),
                \mul[G](\groupInv[G](z),
                    \mul[G](x,
                        \mul[G](\groupInv[G](x),
                            \mul[G](\groupInv[G](y),
                                \mul[G](x,\mul[G](y,z)))))))
                \explanation{by
                    \auto{group_is_monoid,monoid_assoc,
                        monoid_is_magma,magma_welldef,
                        monoid_is_unital_magma,unitalmagma_left,
                        unitalmagma_right}}
            \\
            &= \mul[G](\groupInv[G](x),
                \mul[G](\groupInv[G](z),
                    \mul[G](x,
                        \mul[G](z,
                            \mul[G](\groupInv[G](z),
                                \mul[G](\groupInv[G](x),
                                    \mul[G](\groupInv[G](y),
                                        \mul[G](x,
                                            \mul[G](y,z)))))))))
                \explanation{by
                    \auto{group_is_monoid,monoid_assoc,
                        monoid_is_magma,magma_welldef,
                        monoid_is_unital_magma,unitalmagma_left,
                        unitalmagma_right}}
            \\
            &= \mul[G](
                \groupCommutator{G}{x}{z},
                \conjugate{G}{\groupInv[G](z)}{
                    \groupCommutator{G}{x}{y}})
                \explanation{by
                    \auto{group_conjugate,group_commutator,
                        group_is_monoid,monoid_assoc,
                        monoid_is_magma,magma_welldef}}
    \end{align*}
\end{proof}

\begin{theorem}\label{group_hom_commutator}
    Suppose $\groupHom{f}{G}{H}$.
    Let $x,y\in\carrier[G]$.
    Then
        $f(\groupCommutator{G}{x}{y})=
            \groupCommutator{H}{f(x)}{f(y)}$.
\end{theorem}
\begin{proof}
    $\groupInv[G](x),\groupInv[G](y)\in\carrier[G]$ by
        \obv{group_inverse_type}.
    $\mul[G](\groupInv[G](x),\groupInv[G](y))\in\carrier[G]$ and
        $\mul[G](
            \mul[G](\groupInv[G](x),\groupInv[G](y)),x)
            \in\carrier[G]$ by
        \obv{group_is_monoid,monoid_is_magma,magma_welldef}.
    $f(\groupInv[G](x))=\groupInv[H](f(x))$ and
        $f(\groupInv[G](y))=\groupInv[H](f(y))$ by
        \obv{group_hom_inverse}.
    Follows by \auto{group_commutator,group_hom_mul}.
\end{proof}

\begin{theorem}\label{group_commutator_conjugate}
    Let $G$ be a group.
    Let $a,x,y\in\carrier[G]$.
    Then
        $\conjugate{G}{a}{\groupCommutator{G}{x}{y}}=
            \groupCommutator{G}
                {\conjugate{G}{a}{x}}{\conjugate{G}{a}{y}}$.
\end{theorem}
\begin{proof}
    Let $f=\ConjugationMap{G}{a}$.
    $\groupHom{f}{G}{G}$ by \auto{conjugation_map_group_hom}.
    $\groupCommutator{G}{x}{y}\in\carrier[G]$ by
        \obv{group_commutator_type}.
    $f(\groupCommutator{G}{x}{y})=
        \groupCommutator{G}{f(x)}{f(y)}$ by
        \obv{group_hom_commutator}.
    $f(x)=\conjugate{G}{a}{x}$ and
        $f(y)=\conjugate{G}{a}{y}$ and
        $f(\groupCommutator{G}{x}{y})=
            \conjugate{G}{a}{\groupCommutator{G}{x}{y}}$ by
        \auto{conjugation_map_apply}.
\end{proof}

\begin{theorem}\label{group_commutator_neutral_iff_commutes}
    Let $G$ be a group.
    Let $x,y\in\carrier[G]$.
    Then $\groupCommutator{G}{x}{y}=\neutral[G]$ iff
        $\commutesIn{G}{x}{y}$.
\end{theorem}
\begin{proof}
    Let $u=\groupInv[G](x)$ and $v=\groupInv[G](y)$.
    $u,v\in\carrier[G]$ by \auto{group_inverse_type}.
    Show that if $\groupCommutator{G}{x}{y}=\neutral[G]$, then
        $\commutesIn{G}{x}{y}$.
    \begin{subproof}
        Assume $\groupCommutator{G}{x}{y}=\neutral[G]$.
        $\mul[G](x,\groupCommutator{G}{x}{y})=x$ by
            \auto{group_commutator_type,group_is_monoid,
                monoid_is_unital_magma,unitalmagma_right}.
        $\mul[G](u,v),\mul[G](\mul[G](u,v),x),
            \mul[G](x,y),\mul[G](v,\mul[G](x,y))\in\carrier[G]$ by
            \obv{group_is_monoid,monoid_is_magma,magma_welldef}.
        $\groupCommutator{G}{x}{y}=
            \mul[G](u,\mul[G](v,\mul[G](x,y)))$ by
            \auto{group_commutator,group_is_monoid,monoid_assoc}.
        $\mul[G](x,\groupCommutator{G}{x}{y})=
            \mul[G](v,\mul[G](x,y))$ by
            \auto{group_is_monoid,monoid_assoc,
                group_inverse_right,monoid_is_unital_magma,
                unitalmagma_left}.
        Hence $\mul[G](v,\mul[G](x,y))=x$.
        Hence $\mul[G](y,\mul[G](v,\mul[G](x,y)))=
            \mul[G](y,x)$.
        $\mul[G](y,\mul[G](v,\mul[G](x,y)))=
            \mul[G](x,y)$ by
            \auto{group_is_monoid,monoid_assoc,group_inverse_right,
                monoid_is_unital_magma,unitalmagma_left}.
        Hence $\mul[G](x,y)=\mul[G](y,x)$.
        Follows by \obv{commutes_in}.
    \end{subproof}
    Show that if $\commutesIn{G}{x}{y}$, then
        $\groupCommutator{G}{x}{y}=\neutral[G]$.
    \begin{subproof}
        Assume $\commutesIn{G}{x}{y}$.
        Hence $\mul[G](x,y)=\mul[G](y,x)$ by \obv{commutes_in}.
        $\mul[G](u,v),\mul[G](\mul[G](u,v),x),
            \mul[G](x,y),\mul[G](y,x),
            \mul[G](v,\mul[G](x,y)),
            \mul[G](v,\mul[G](y,x))\in\carrier[G]$ by
            \obv{group_is_monoid,monoid_is_magma,magma_welldef}.
        $\groupCommutator{G}{x}{y}=
            \mul[G](u,\mul[G](v,\mul[G](x,y)))$ by
            \auto{group_commutator,group_is_monoid,monoid_assoc}.
        Hence $\groupCommutator{G}{x}{y}=
            \mul[G](u,\mul[G](v,\mul[G](y,x)))$.
        $\mul[G](u,\mul[G](v,\mul[G](y,x)))=
            \neutral[G]$ by
            \auto{group_is_monoid,monoid_assoc,group_inverse_left,
                monoid_is_unital_magma,unitalmagma_left}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{definition}\label{group_commutator_carrier}
    $\groupCommutatorCarrier{G}=
        \{z\in\carrier[G]\mid
            \text{there exists $x\in\carrier[G]$ such that there exists
                $y\in\carrier[G]$ such that
                $z=\groupCommutator{G}{x}{y}$}\}$.
\end{definition}

\begin{proposition}\label{group_commutator_carrier_iff}
    $z\in\groupCommutatorCarrier{G}$ iff
        $z\in\carrier[G]$ and there exists $x\in\carrier[G]$ such that
            there exists $y\in\carrier[G]$ such that
                $z=\groupCommutator{G}{x}{y}$.
\end{proposition}
\begin{proof}
    Follows by \obv{group_commutator_carrier}.
\end{proof}

\begin{proposition}\label{group_commutator_carrier_subseteq}
    $\groupCommutatorCarrier{G}\subseteq\carrier[G]$.
\end{proposition}
\begin{proof}
    Follows by \obv{group_commutator_carrier,subseteq}.
\end{proof}

\begin{classification}\label{group_commutator_carrier_is_subset_in_group}
    Let $G$ be a group.
    Then $\groupCommutatorCarrier{G}\subseteq\carrier[G]$.
\end{classification}
\begin{proof}
    Follows by \obv{group_commutator_carrier_subseteq}.
\end{proof}

\begin{definition}\label{commutator_subgroup}
    Let $G$ be a group.
    $\commutatorSubgroup{G}=
        \generatedSubgroup{\groupCommutatorCarrier{G}}{G}$.
\end{definition}

\begin{proposition}\label{commutator_subgroup_contains_commutators}
    Let $G$ be a group.
    Let $x,y\in\carrier[G]$.
    Then $\groupCommutator{G}{x}{y}\in\commutatorSubgroup{G}$.
\end{proposition}
\begin{proof}
    Let $C=\groupCommutatorCarrier{G}$.
    $\groupCommutator{G}{x}{y}\in\carrier[G]$ by
        \obv{group_commutator_type}.
    Hence $\groupCommutator{G}{x}{y}\in C$ by
        \obv{group_commutator_carrier_iff}.
    $C\subseteq\carrier[G]$ by
        \auto{group_commutator_carrier_subseteq}.
    $C\subseteq\generatedSubgroup{C}{G}$ by
        \obv{generated_subgroup_contains_generators,subseteq}.
    Follows by \obv{commutator_subgroup,subseteq}.
\end{proof}

\begin{proposition}\label{commutator_subgroup_is_subgroup}
    Let $G$ be a group.
    Then $\subgroup{\commutatorSubgroup{G}}{G}$.
\end{proposition}
\begin{proof}
    Let $C=\groupCommutatorCarrier{G}$.
    $C\subseteq\carrier[G]$ by
        \auto{group_commutator_carrier_subseteq}.
    Follows by
        \auto{commutator_subgroup,generated_subgroup_is_subgroup}.
\end{proof}

\begin{corollary}\label{commutator_subgroup_subseteq}
    Let $G$ be a group.
    Then $\commutatorSubgroup{G}\subseteq\carrier[G]$.
\end{corollary}
\begin{proof}
    Follows by
        \obv{commutator_subgroup_is_subgroup,subgroup_subseteq}.
\end{proof}

\begin{theorem}\label{commutator_subgroup_least}
    Let $G$ be a group.
    Suppose $\subgroup{K}{G}$ and for all $x,y\in\carrier[G]$ we have
        $\groupCommutator{G}{x}{y}\in K$.
    Then $\commutatorSubgroup{G}\subseteq K$.
\end{theorem}
\begin{proof}
    Let $C=\groupCommutatorCarrier{G}$.
    Show that $C\subseteq K$.
    \begin{subproof}
        Show that for every $z\in C$ we have $z\in K$.
        \begin{subproof}
            Fix $z\in C$.
            Take $x\in\carrier[G]$ such that there exists
                $y\in\carrier[G]$ such that
                $z=\groupCommutator{G}{x}{y}$ by
                \auto{group_commutator_carrier_iff}.
            Take $y\in\carrier[G]$ such that
                $z=\groupCommutator{G}{x}{y}$.
            Follows by \auto.
        \end{subproof}
        Follows by \obv{subseteq}.
    \end{subproof}
    $C\subseteq\carrier[G]$ by
        \auto{group_commutator_carrier_subseteq}.
    Follows by
        \auto{commutator_subgroup,generated_subgroup_least}.
\end{proof}

\begin{theorem}\label{automorphism_image_commutator_generators}
    Let $G$ be a group.
    Let $f\in\automorphismSet{G}$.
    Let $C=\groupCommutatorCarrier{G}$.
    Then $\img{f}{C}=C$.
\end{theorem}
\begin{proof}
    $\groupHom{f}{G}{G}$ by \obv{automorphism_set_is_hom}.
    $f\in\bijections{\carrier[G]}{\carrier[G]}$ by
        \obv{automorphism_set_type}.
    Hence $f\in\surj{\carrier[G]}{\carrier[G]}$ by
        \obv{bijections}.
    Show that $\img{f}{C}\subseteq C$.
    \begin{subproof}
        Show that for every $z\in\img{f}{C}$ we have $z\in C$.
        \begin{subproof}
            Fix $z\in\img{f}{C}$.
            $C\subseteq\carrier[G]$ by
                \auto{group_commutator_carrier_subseteq}.
            Take $c\in C$ such that $f(c)=z$ by
                \auto{group_hom_image_iff}.
            Take $x\in\carrier[G]$ such that there exists
                $y\in\carrier[G]$ such that
                $c=\groupCommutator{G}{x}{y}$ by
                \auto{group_commutator_carrier_iff}.
            Take $y\in\carrier[G]$ such that
                $c=\groupCommutator{G}{x}{y}$.
            $f(x),f(y)\in\carrier[G]$ by
                \obv{group_hom_apply_type}.
            $f(c)=\groupCommutator{G}{f(x)}{f(y)}$ by
                \auto{group_hom_commutator}.
            Hence $z=\groupCommutator{G}{f(x)}{f(y)}$.
            $z\in\carrier[G]$ by
                \auto{group_commutator_type}.
            Follows by \auto{group_commutator_carrier_iff}.
        \end{subproof}
        Follows by \obv{subseteq}.
    \end{subproof}
    Show that $C\subseteq\img{f}{C}$.
    \begin{subproof}
        Show that for every $z\in C$ we have $z\in\img{f}{C}$.
        \begin{subproof}
            Fix $z\in C$.
            Take $x\in\carrier[G]$ such that there exists
                $y\in\carrier[G]$ such that
                $z=\groupCommutator{G}{x}{y}$ by
                \auto{group_commutator_carrier_iff}.
            Take $y\in\carrier[G]$ such that
                $z=\groupCommutator{G}{x}{y}$.
            Take $u\in\carrier[G]$ such that $f(u)=x$ by
                \obv{surj}.
            Take $v\in\carrier[G]$ such that $f(v)=y$ by
                \obv{surj}.
            Let $c=\groupCommutator{G}{u}{v}$.
            $c\in\carrier[G]$ by \auto{group_commutator_type}.
            Hence $c\in C$ by
                \auto{group_commutator_carrier_iff}.
            $f(c)=\groupCommutator{G}{f(u)}{f(v)}$ by
                \auto{group_hom_commutator}.
            Hence $f(c)=z$.
            $C\subseteq\carrier[G]$ by
                \auto{group_commutator_carrier_subseteq}.
            Follows by \auto{group_hom_image_iff}.
        \end{subproof}
        Follows by \obv{subseteq}.
    \end{subproof}
    Follows by \obv{subseteq_antisymmetric}.
\end{proof}

\begin{theorem}\label{commutator_subgroup_characteristic}
    Let $G$ be a group.
    Then $\characteristicSubgroup{\commutatorSubgroup{G}}{G}$.
\end{theorem}
\begin{proof}
    Let $C=\groupCommutatorCarrier{G}$.
    $C\subseteq\carrier[G]$ by
        \auto{group_commutator_carrier_subseteq}.
    $\subgroup{\commutatorSubgroup{G}}{G}$ by
        \obv{commutator_subgroup_is_subgroup}.
    Show that for every $f\in\automorphismSet{G}$ we have
        $\img{f}{\commutatorSubgroup{G}}=\commutatorSubgroup{G}$.
    \begin{subproof}
        Fix $f\in\automorphismSet{G}$.
        $\groupHom{f}{G}{G}$ by \obv{automorphism_set_is_hom}.
        $f\in\funs{\carrier[G]}{\carrier[G]}$ by
            \obv{group_hom_type}.
        $\img{f}{C}=C$ by
            \obv{automorphism_image_commutator_generators}.
        $\img{f}{\generatedSubgroup{C}{G}}=
            \generatedSubgroup{\img{f}{C}}{G}$ by
            \obv{group_hom_image_generated_subgroup}.
        Follows by \auto{commutator_subgroup}.
    \end{subproof}
    Follows by \obv{characteristic_subgroup}.
\end{proof}

\begin{corollary}\label{commutator_subgroup_normal}
    Let $G$ be a group.
    Then $\normalSubgroup{\commutatorSubgroup{G}}{G}$.
\end{corollary}
\begin{proof}
    Follows by
        \obv{commutator_subgroup_characteristic,
            characteristic_subgroup_is_normal}.
\end{proof}

\begin{theorem}\label{quotient_group_commutative_iff_commutator_subgroup}
    Let $G$ be a group.
    Suppose $\normalSubgroup{N}{G}$.
    Then $\quotientGroup{G}{N}$ is a commutative group iff
        $\commutatorSubgroup{G}\subseteq N$.
\end{theorem}
\begin{proof}
    Let $Q=\quotientGroup{G}{N}$ and
        $q=\groupQuotientProjection{G}{N}$.
    $Q$ is a group by \obv{quotient_group_is_group}.
    $\groupHom{q}{G}{Q}$ by \auto{group_quotient_projection_hom}.
    $q\in\surj{\carrier[G]}{\carrier[Q]}$ by
        \auto{group_quotient_projection_surjective,
            quotient_group_carrier}.
    $\groupKernel{q}{G}=N$ by
        \auto{group_quotient_projection_kernel}.
    Show that if $Q$ is a commutative group, then
        $\commutatorSubgroup{G}\subseteq N$.
    \begin{subproof}
        Assume $Q$ is a commutative group.
        Show that for all $x,y\in\carrier[G]$ we have
            $\groupCommutator{G}{x}{y}\in N$.
        \begin{subproof}
            Fix $x,y\in\carrier[G]$.
            $q(x),q(y)\in\carrier[Q]$ by
                \obv{group_hom_apply_type}.
            $\commutesIn{Q}{q(x)}{q(y)}$ by
                \obv{commutes_in,commutative_group_commutative}.
            Hence $\groupCommutator{Q}{q(x)}{q(y)}=\neutral[Q]$ by
                \obv{group_commutator_neutral_iff_commutes}.
            $q(\groupCommutator{G}{x}{y})=
                \groupCommutator{Q}{q(x)}{q(y)}$ by
                \obv{group_hom_commutator}.
            Hence $\groupCommutator{G}{x}{y}\in
                \groupKernel{q}{G}$ by
                \auto{group_commutator_type,group_kernel_iff}.
        \end{subproof}
        $\subgroup{N}{G}$ by \obv{normal_subgroup_is_subgroup}.
        Follows by \obv{commutator_subgroup_least}.
    \end{subproof}
    Show that if $\commutatorSubgroup{G}\subseteq N$, then
        $Q$ is a commutative group.
    \begin{subproof}
        Assume $\commutatorSubgroup{G}\subseteq N$.
        Show that for all $U,V\in\carrier[Q]$ we have
            $\mul[Q](U,V)=\mul[Q](V,U)$.
        \begin{subproof}
            Fix $U,V\in\carrier[Q]$.
            Take $x\in\carrier[G]$ such that $q(x)=U$ by
                \obv{surj}.
            Take $y\in\carrier[G]$ such that $q(y)=V$ by
                \obv{surj}.
            $\groupCommutator{G}{x}{y}\in
                \commutatorSubgroup{G}$ by
                \obv{commutator_subgroup_contains_commutators}.
            Hence $\groupCommutator{G}{x}{y}\in N$ by
                \obv{subseteq}.
            Hence $\groupCommutator{G}{x}{y}\in
                \groupKernel{q}{G}$.
            Hence $q(\groupCommutator{G}{x}{y})=\neutral[Q]$ by
                \auto{group_kernel_iff}.
            $q(\groupCommutator{G}{x}{y})=
                \groupCommutator{Q}{q(x)}{q(y)}$ by
                \obv{group_hom_commutator}.
            Hence $\commutesIn{Q}{U}{V}$ by
                \auto{group_commutator_neutral_iff_commutes}.
            Follows by \obv{commutes_in}.
        \end{subproof}
        $Q$ is a group by \obv{quotient_group_is_group}.
        Follows by \obv{group,commutative_group}.
    \end{subproof}
    Follows by \auto.
\end{proof}
