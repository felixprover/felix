\section{Normal subgroups}

Normality is a relation between a subgroup carrier and an explicit ambient
group.  Its primary definition says that every ambient conjugation fixes the
carrier as a set; the closure formulation below is the usual elimination and
introduction interface.

\begin{definition}\label{normal_subgroup}
    $\normalSubgroup{N}{G}$ iff
        $\subgroup{N}{G}$ and
        for every $a\in\carrier[G]$ we have
            $\conjugateSubset{G}{a}{N}=N$.
\end{definition}

\begin{proposition}\label{normal_subgroup_is_subgroup}
    Suppose $\normalSubgroup{N}{G}$.
    Then $\subgroup{N}{G}$.
\end{proposition}
\begin{proof}
    Follows by \obv{normal_subgroup}.
\end{proof}

\begin{corollary}\label{normal_subgroup_ambient_group}
    Suppose $\normalSubgroup{N}{G}$.
    Then $G$ is a group.
\end{corollary}
\begin{proof}
    Follows by
        \obv{normal_subgroup_is_subgroup,subgroup_ambient_group}.
\end{proof}

\begin{corollary}\label{normal_subgroup_subseteq}
    Suppose $\normalSubgroup{N}{G}$.
    Then $N\subseteq\carrier[G]$.
\end{corollary}
\begin{proof}
    Follows by
        \obv{normal_subgroup_is_subgroup,subgroup_subseteq}.
\end{proof}

\begin{proposition}\label{normal_subgroup_fixed_conjugate}
    Suppose $\normalSubgroup{N}{G}$.
    Let $a\in\carrier[G]$.
    Then $\conjugateSubset{G}{a}{N}=N$.
\end{proposition}
\begin{proof}
    Follows by \obv{normal_subgroup}.
\end{proof}

\begin{proposition}\label{normal_subgroup_closed_under_conjugation}
    Suppose $\normalSubgroup{N}{G}$.
    Let $a\in\carrier[G]$.
    Let $x\in N$.
    Then $\conjugate{G}{a}{x}\in N$.
\end{proposition}
\begin{proof}
    $G$ is a group by \obv{normal_subgroup_ambient_group}.
    $x\in\carrier[G]$ by
        \auto{normal_subgroup_subseteq,subseteq}.
    Hence $x\in N\inter\carrier[G]$ by \obv{inter_intro}.
    Hence $\conjugate{G}{a}{x}\in\conjugateSubset{G}{a}{N}$ by
        \auto{conjugate_subset_iff}.
    Follows by \obv{normal_subgroup_fixed_conjugate}.
\end{proof}

\begin{theorem}\label{normal_subgroup_iff_closed_under_conjugation}
    Suppose $\subgroup{N}{G}$.
    Then $\normalSubgroup{N}{G}$ iff
        for all $a\in\carrier[G]$ we have for every $x\in N$ we have
            $\conjugate{G}{a}{x}\in N$.
\end{theorem}
\begin{proof}
    Show that if $\normalSubgroup{N}{G}$, then
        for all $a\in\carrier[G]$ we have for every $x\in N$ we have
            $\conjugate{G}{a}{x}\in N$.
    \begin{subproof}
        Assume $\normalSubgroup{N}{G}$.
        Follows by \obv{normal_subgroup_closed_under_conjugation}.
    \end{subproof}
    Show that if for all $a\in\carrier[G]$ we have for every $x\in N$
        we have $\conjugate{G}{a}{x}\in N$, then
        $\normalSubgroup{N}{G}$.
    \begin{subproof}
        Assume that for all $a\in\carrier[G]$ we have for every $x\in N$
            we have $\conjugate{G}{a}{x}\in N$.
        $G$ is a group and $N\subseteq\carrier[G]$ by
            \obv{subgroup_ambient_group,subgroup_subseteq}.
        Show that for every $a\in\carrier[G]$ we have
            $\conjugateSubset{G}{a}{N}=N$.
        \begin{subproof}
            Fix $a\in\carrier[G]$.
            Show that $\conjugateSubset{G}{a}{N}\subseteq N$.
            \begin{subproof}
                Show that for every $y\in\conjugateSubset{G}{a}{N}$
                    we have $y\in N$.
                \begin{subproof}
                    Fix $y\in\conjugateSubset{G}{a}{N}$.
                    Take $x\in N\inter\carrier[G]$ such that
                        $y=\conjugate{G}{a}{x}$ by
                        \auto{conjugate_subset_iff}.
                    $x\in N$ by \obv{inter_elim_left}.
                    Follows by \auto.
                \end{subproof}
                Follows by \obv{subseteq}.
            \end{subproof}
            Show that $N\subseteq\conjugateSubset{G}{a}{N}$.
            \begin{subproof}
                Show that for every $y\in N$ we have
                    $y\in\conjugateSubset{G}{a}{N}$.
                \begin{subproof}
                    Fix $y\in N$.
                    $y\in\carrier[G]$ by \obv{subseteq}.
                    $\groupInv[G](a)\in\carrier[G]$ by
                        \obv{group_inverse_type}.
                    Let $x=\conjugate{G}{\groupInv[G](a)}{y}$.
                    $x\in N$ by \auto.
                    $x\in\carrier[G]$ by
                        \obv{group_conjugate_type}.
                    Hence $x\in N\inter\carrier[G]$ by
                        \obv{inter_intro}.
                    $\conjugate{G}{a}{x}=y$ by
                        \auto{group_conjugate_inverse_right}.
                    Follows by \auto{conjugate_subset_iff}.
                \end{subproof}
                Follows by \obv{subseteq}.
            \end{subproof}
            Follows by \obv{subseteq_antisymmetric}.
        \end{subproof}
        Follows by \obv{normal_subgroup}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{theorem}\label{commutative_group_subgroup_is_normal}
    Let $G$ be a commutative group.
    Suppose $\subgroup{N}{G}$.
    Then $\normalSubgroup{N}{G}$.
\end{theorem}
\begin{proof}
    $G$ is a group by \obv{commutative_group_is_group}.
    Show that for all $a\in\carrier[G]$ we have for every $x\in N$ we have
        $\conjugate{G}{a}{x}\in N$.
    \begin{subproof}
        Fix $a\in\carrier[G]$.
        Fix $x\in N$.
        $x\in\carrier[G]$ by \auto{subgroup_subseteq,subseteq}.
        $\mul[G](a,x)=\mul[G](x,a)$ by
            \obv{commutative_group_commutative}.
        Hence $\commutesIn{G}{a}{x}$ by \obv{commutes_in}.
        Hence $\conjugate{G}{a}{x}=x$ by
            \obv{commutes_in_iff_conjugate_fixed}.
    \end{subproof}
    Follows by \obv{normal_subgroup_iff_closed_under_conjugation}.
\end{proof}

\begin{proposition}\label{full_carrier_normal_subgroup}
    Let $G$ be a group.
    Then $\normalSubgroup{\carrier[G]}{G}$.
\end{proposition}
\begin{proof}
    $\subgroup{\carrier[G]}{G}$ by
        \obv{full_carrier_subgroup}.
    For every $a\in\carrier[G]$ we have
        $\conjugateSubset{G}{a}{\carrier[G]}=\carrier[G]$ by
        \obv{conjugate_full_carrier}.
    Follows by \obv{normal_subgroup}.
\end{proof}

\begin{proposition}\label{trivial_carrier_normal_subgroup}
    Let $G$ be a group.
    Then $\normalSubgroup{\trivialSubgroupCarrier{G}}{G}$.
\end{proposition}
\begin{proof}
    Let $N=\trivialSubgroupCarrier{G}$.
    $\subgroup{N}{G}$ by \auto{trivial_carrier_subgroup}.
    Show that for all $a\in\carrier[G]$ we have for every $x\in N$
        we have $\conjugate{G}{a}{x}\in N$.
    \begin{subproof}
        Fix $a\in\carrier[G]$.
        Fix $x\in N$.
        $x=\neutral[G]$ by
            \auto{trivial_subgroup_carrier,singleton_iff}.
        Follows by
            \auto{group_conjugate_neutral,trivial_subgroup_carrier,singleton_iff}.
    \end{subproof}
    Follows by \auto{normal_subgroup_iff_closed_under_conjugation}.
\end{proof}

\subsection{Normality and homomorphisms}

\begin{theorem}\label{group_kernel_normal_subgroup}
    Suppose $\groupHom{f}{G}{H}$.
    Then $\normalSubgroup{\groupKernel{f}{G}}{G}$.
\end{theorem}
\begin{proof}
    Let $N=\groupKernel{f}{G}$.
    $\subgroup{N}{G}$ by \auto{group_kernel_subgroup}.
    Show that for all $a\in\carrier[G]$ we have for every $x\in N$
        we have $\conjugate{G}{a}{x}\in N$.
    \begin{subproof}
        Fix $a\in\carrier[G]$.
        Fix $x\in N$.
        $x\in\carrier[G]$ and $f(x)=\neutral[H]$ by
            \obv{group_kernel_iff}.
        $f(a)\in\carrier[H]$ by \obv{group_hom_apply_type}.
        $f(\conjugate{G}{a}{x})=
            \conjugate{H}{f(a)}{f(x)}$ by
            \obv{group_hom_conjugate}.
        Hence $f(\conjugate{G}{a}{x})=\neutral[H]$ by
            \auto{group_hom_target_group,group_conjugate_neutral}.
        $\conjugate{G}{a}{x}\in\carrier[G]$ by
            \obv{group_hom_source_group,group_conjugate_type}.
        Follows by \auto{group_kernel_iff}.
    \end{subproof}
    Follows by \auto{normal_subgroup_iff_closed_under_conjugation}.
\end{proof}

\begin{theorem}\label{group_hom_normal_preimage}
    Suppose $\groupHom{f}{G}{H}$ and $\normalSubgroup{N}{H}$.
    Then $\normalSubgroup{\preimg{f}{N}}{G}$.
\end{theorem}
\begin{proof}
    Let $P=\preimg{f}{N}$.
    $\subgroup{N}{H}$ by \obv{normal_subgroup_is_subgroup}.
    Hence $\subgroup{P}{G}$ by
        \auto{group_hom_subgroup_preimage}.
    Show that for all $a\in\carrier[G]$ we have for every $x\in P$
        we have $\conjugate{G}{a}{x}\in P$.
    \begin{subproof}
        Fix $a\in\carrier[G]$.
        Fix $x\in P$.
        Hence $x\in\carrier[G]$ and $f(x)\in N$ by
            \auto{group_hom_preimage_iff}.
        $f(a)\in\carrier[H]$ by \obv{group_hom_apply_type}.
        Hence $\conjugate{H}{f(a)}{f(x)}\in N$ by
            \obv{normal_subgroup_closed_under_conjugation}.
        $f(\conjugate{G}{a}{x})=
            \conjugate{H}{f(a)}{f(x)}$ by
            \obv{group_hom_conjugate}.
        $\conjugate{G}{a}{x}\in\carrier[G]$ by
            \obv{group_hom_source_group,group_conjugate_type}.
        Follows by \auto{group_hom_preimage_iff}.
    \end{subproof}
    Follows by \auto{normal_subgroup_iff_closed_under_conjugation}.
\end{proof}

\begin{lemma}\label{subgroup_structure_conjugate_apply}
    Suppose $\subgroup{I}{H}$.
    Let $a,x\in I$.
    Then
        $\conjugate{\subgroupStructure{H}{I}}{a}{x}=
            \conjugate{H}{a}{x}$.
\end{lemma}
\begin{proof}
    Let $T=\subgroupStructure{H}{I}$.
    $\groupInv[T](a)=\groupInv[H](a)$ by
        \auto{subgroup_structure_inv_apply}.
    $\mul[T](a,x)=\mul[H](a,x)$ by
        \auto{subgroup_structure_mul_apply}.
    $\mul[H](a,x)\in I$ by \obv{subgroup_mul}.
    Hence
        $\mul[T](\mul[T](a,x),\groupInv[T](a))=
            \mul[H](\mul[H](a,x),\groupInv[H](a))$ by
        \auto{subgroup_inv,subgroup_structure_mul_apply}.
    Follows by \obv{group_conjugate}.
\end{proof}

\begin{theorem}\label{group_hom_normal_image_in_induced_image}
    Suppose $\groupHom{f}{G}{H}$ and $\normalSubgroup{N}{G}$.
    Then
        $\normalSubgroup
            {\img{f}{N}}
            {\subgroupStructure{H}{\img{f}{\carrier[G]}}}$.
\end{theorem}
\begin{proof}
    Let $I=\img{f}{\carrier[G]}$.
    Let $J=\img{f}{N}$.
    Let $T=\subgroupStructure{H}{I}$.
    $\subgroup{I}{H}$ by \auto{group_hom_image_subgroup}.
    $N\subseteq\carrier[G]$ by \obv{normal_subgroup_subseteq}.
    Hence $J\subseteq I$ by \auto{img_subseteq}.
    $\subgroup{N}{G}$ by \obv{normal_subgroup_is_subgroup}.
    Hence $\subgroup{J}{H}$ by \auto{group_hom_subgroup_image}.
    Hence $\subgroup{J}{T}$ by
        \obv{subgroup_of_induced_group_iff}.
    Show that for all $a\in\carrier[T]$ we have for every $x\in J$
        we have $\conjugate{T}{a}{x}\in J$.
    \begin{subproof}
        Fix $a\in\carrier[T]$.
        Fix $x\in J$.
        $\carrier[T]=I$ by \auto{subgroup_structure_carrier}.
        Hence $a\in I$.
        Take $b\in\carrier[G]$ such that $f(b)=a$ by
            \auto{group_hom_image_iff,subseteq_refl}.
        Take $y\in N$ such that $f(y)=x$ by
            \auto{group_hom_image_iff,normal_subgroup_subseteq}.
        $y\in\carrier[G]$ by
            \obv{normal_subgroup_subseteq,subseteq}.
        $\conjugate{G}{b}{y}\in N$ by
            \obv{normal_subgroup_closed_under_conjugation}.
        Hence $f(\conjugate{G}{b}{y})\in J$ by
            \auto{group_hom_image_iff,normal_subgroup_subseteq}.
        $f(\conjugate{G}{b}{y})=
            \conjugate{H}{f(b)}{f(y)}$ by
            \obv{group_hom_conjugate}.
        Hence $\conjugate{H}{a}{x}\in J$.
        $x\in I$ by \obv{subseteq}.
        Follows by \auto{subgroup_structure_conjugate_apply}.
    \end{subproof}
    Follows by \auto{normal_subgroup_iff_closed_under_conjugation}.
\end{proof}

\begin{theorem}\label{group_isomorphism_normal_image}
    Suppose $\groupIsomorphism{f}{G}{H}$ and
        $\normalSubgroup{N}{G}$.
    Then $\normalSubgroup{\img{f}{N}}{H}$.
\end{theorem}
\begin{proof}
    $\groupHom{f}{G}{H}$ by
        \obv{group_isomorphism_is_hom}.
    Let $J=\img{f}{N}$.
    $\subgroup{N}{G}$ by \obv{normal_subgroup_is_subgroup}.
    Hence $\subgroup{J}{H}$ by
        \auto{group_hom_subgroup_image}.
    Show that for all $a\in\carrier[H]$ we have for every $x\in J$
        we have $\conjugate{H}{a}{x}\in J$.
    \begin{subproof}
        Fix $a\in\carrier[H]$.
        Fix $x\in J$.
        $f\in\surj{\carrier[G]}{\carrier[H]}$ by
            \auto{group_isomorphism_is_bijection,bijections}.
        Take $b\in\carrier[G]$ such that $f(b)=a$ by
            \obv{surj}.
        Take $y\in N$ such that $f(y)=x$ by
            \auto{group_hom_image_iff,normal_subgroup_subseteq}.
        $y\in\carrier[G]$ by
            \auto{normal_subgroup_subseteq,subseteq}.
        $\conjugate{G}{b}{y}\in N$ by
            \obv{normal_subgroup_closed_under_conjugation}.
        Hence $f(\conjugate{G}{b}{y})\in J$ by
            \auto{group_hom_image_iff,normal_subgroup_subseteq}.
        $f(\conjugate{G}{b}{y})=
            \conjugate{H}{f(b)}{f(y)}$ by
            \obv{group_hom_conjugate}.
    \end{subproof}
    Follows by \auto{normal_subgroup_iff_closed_under_conjugation}.
\end{proof}
