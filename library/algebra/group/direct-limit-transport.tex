\section{Coherent transport of group direct systems}

A family of componentwise isomorphisms transports a direct system only when
it commutes with every bond.  The family is raw morphism data with explicit
endpoint systems, not a carrier-bearing structure.

\begin{definition}\label{group_direct_system_isomorphism}
    $\groupDirectSystemIsomorphism{E}{S}{T}$ iff
        $S$ is a group direct system and
        $T$ is a group direct system and
        $\carrier[S]=\carrier[T]$ and
        $\leq[S]=\leq[T]$ and
        $E\in\funs{\carrier[S]}{\ran{E}}$ and
        for every $i\in\carrier[S]$ we have
            $\groupIsomorphism{\apply{E}{i}}
                {\directSystemGroup{S}{i}}
                {\directSystemGroup{T}{i}}$ and
        for all $a,b\in\carrier[S]$ such that
            $a\mathrel{\leq[S]}b$ we have
            $\apply{E}{b}\circ\directSystemBond{S}{a}{b}
                =\directSystemBond{T}{a}{b}\circ\apply{E}{a}$.
\end{definition}

\begin{definition}\label{group_direct_system_component_isomorphisms}
    $\groupDirectSystemComponentIsomorphisms{E}{S}{T}$ iff
        for every $i\in\carrier[S]$ we have
            $\groupIsomorphism{\apply{E}{i}}
                {\directSystemGroup{S}{i}}
                {\directSystemGroup{T}{i}}$.
\end{definition}

\begin{proposition}\label{group_direct_system_component_isomorphisms_intro}
    Suppose for every $i\in\carrier[S]$ we have
        $\groupIsomorphism{\apply{E}{i}}
            {\directSystemGroup{S}{i}}
            {\directSystemGroup{T}{i}}$.
    Then $\groupDirectSystemComponentIsomorphisms{E}{S}{T}$.
\end{proposition}
\begin{proof}
    Follows by \obv{group_direct_system_component_isomorphisms}.
\end{proof}

\begin{definition}\label{group_direct_system_isomorphism_squares}
    $\groupDirectSystemIsomorphismSquares{E}{S}{T}$ iff
        for all $a,b\in\carrier[S]$ such that
            $a\mathrel{\leq[S]}b$ we have
            $\apply{E}{b}\circ\directSystemBond{S}{a}{b}=
                \directSystemBond{T}{a}{b}\circ\apply{E}{a}$.
\end{definition}

\begin{proposition}\label{group_direct_system_isomorphism_squares_intro}
    Suppose for all $a,b\in\carrier[S]$ such that
        $a\mathrel{\leq[S]}b$ we have
        $\apply{E}{b}\circ\directSystemBond{S}{a}{b}=
            \directSystemBond{T}{a}{b}\circ\apply{E}{a}$.
    Then $\groupDirectSystemIsomorphismSquares{E}{S}{T}$.
\end{proposition}
\begin{proof}
    Follows by \obv{group_direct_system_isomorphism_squares}.
\end{proof}

\begin{definition}\label{group_direct_system_common_shape}
    $\groupDirectSystemCommonShape{S}{T}$ iff
        $S$ is a group direct system and
        $T$ is a group direct system and
        $\carrier[S]=\carrier[T]$ and
        $\leq[S]=\leq[T]$.
\end{definition}

\begin{proposition}\label{group_direct_system_common_shape_intro}
    Let $S$ be a group direct system.
    Let $T$ be a group direct system.
    Suppose $\carrier[S]=\carrier[T]$ and
        $\leq[S]=\leq[T]$.
    Then $\groupDirectSystemCommonShape{S}{T}$.
\end{proposition}
\begin{proof}
    Follows by \obv{group_direct_system_common_shape}.
\end{proof}

\begin{proposition}\label{group_direct_system_isomorphism_from_parts}
    Suppose $\groupDirectSystemCommonShape{S}{T}$.
    Let $E\in\funs{\carrier[S]}{\ran{E}}$.
    Suppose $\groupDirectSystemComponentIsomorphisms{E}{S}{T}$ and
        $\groupDirectSystemIsomorphismSquares{E}{S}{T}$.
    Then $\groupDirectSystemIsomorphism{E}{S}{T}$.
\end{proposition}
\begin{proof}
    $S$ is a group direct system by
        \obv{group_direct_system_common_shape}.
    $T$ is a group direct system by
        \obv{group_direct_system_common_shape}.
    $\carrier[S]=\carrier[T]$ and $\leq[S]=\leq[T]$ by
        \obv{group_direct_system_common_shape}.
    For every $i\in\carrier[S]$ we have
        $\groupIsomorphism{\apply{E}{i}}
            {\directSystemGroup{S}{i}}
            {\directSystemGroup{T}{i}}$ by
        \obv{group_direct_system_component_isomorphisms}.
    For all $a,b\in\carrier[S]$ such that
        $a\mathrel{\leq[S]}b$ we have
        $\apply{E}{b}\circ\directSystemBond{S}{a}{b}=
            \directSystemBond{T}{a}{b}\circ\apply{E}{a}$ by
        \obv{group_direct_system_isomorphism_squares}.
    Follows by \obv{group_direct_system_isomorphism}.
\end{proof}

\begin{proposition}\label{group_direct_system_isomorphism_intro}
    Let $S$ be a group direct system.
    Let $T$ be a group direct system.
    Suppose $\carrier[S]=\carrier[T]$ and
        $\leq[S]=\leq[T]$.
    Let $E\in\funs{\carrier[S]}{\ran{E}}$.
    Suppose for every $i\in\carrier[S]$ we have
        $\groupIsomorphism{\apply{E}{i}}
            {\directSystemGroup{S}{i}}{\directSystemGroup{T}{i}}$.

        $a\mathrel{\leq[S]}b$ we have

\end{proposition}
\begin{proof}
    Follows by \obv{group_direct_system_isomorphism}.
\end{proof}

\begin{proposition}\label{group_direct_system_isomorphism_component}
    Suppose $\groupDirectSystemIsomorphism{E}{S}{T}$.
    Let $i\in\carrier[S]$.
    Then $\groupIsomorphism{\apply{E}{i}}
        {\directSystemGroup{S}{i}}{\directSystemGroup{T}{i}}$.
\end{proposition}
\begin{proof}
    Follows by \obv{group_direct_system_isomorphism}.
\end{proof}

\begin{proposition}\label{group_direct_system_isomorphism_square}
    Suppose $\groupDirectSystemIsomorphism{E}{S}{T}$.
    Let $i,j\in\carrier[S]$.
    Suppose $i\mathrel{\leq[S]}j$.
    Then $\apply{E}{j}\circ\directSystemBond{S}{i}{j}
        =\directSystemBond{T}{i}{j}\circ\apply{E}{i}$.
\end{proposition}
\begin{proof}
    Follows by \auto{group_direct_system_isomorphism}.
\end{proof}

\begin{theorem}\label{group_direct_system_isomorphism_extensionality}
    Suppose $\groupDirectSystemIsomorphism{E}{S}{T}$ and
        $\groupDirectSystemIsomorphism{D}{S}{T}$.
    Suppose for every $i\in\carrier[S]$ we have
        $\apply{E}{i}=\apply{D}{i}$.
    Then $E=D$.
\end{theorem}
\begin{proof}
    $E\in\funs{\carrier[S]}{\ran{E}}$ and
        $D\in\funs{\carrier[S]}{\ran{D}}$ by
        \obv{group_direct_system_isomorphism_family_type}.
    Hence $E$ and $D$ are functions and
        $\dom{E}=\carrier[S]=\dom{D}$ by
        \obv{funs_elim,funs_is_function}.
    Follows by \obv{funext_on_domain}.
\end{proof}

\subsection{Identity and composition}

\begin{definition}\label{direct_system_identity_isomorphism_family}
    $\directSystemIdentityIsomorphismFamily{S}=
        \{(i,\identity{\carrier[\directSystemGroup{S}{i}]})
            \mid i\in\carrier[S]\}$.
\end{definition}

\begin{proposition}\label{direct_system_identity_isomorphism_family_iff}
    $(i,f)\in\directSystemIdentityIsomorphismFamily{S}$ iff
        $i\in\carrier[S]$ and
        $f=\identity{\carrier[\directSystemGroup{S}{i}]}$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{direct_system_identity_isomorphism_family,pair_eq_iff}.
\end{proof}

\begin{proposition}\label{direct_system_identity_isomorphism_family_type}
    $\directSystemIdentityIsomorphismFamily{S}\in
        \funs{\carrier[S]}
            {\ran{\directSystemIdentityIsomorphismFamily{S}}}$.
\end{proposition}
\begin{proof}
    Let $E=\directSystemIdentityIsomorphismFamily{S}$.
    Show that $E$ is a relation.
    \begin{subproof}
        Follows by
            \auto{relation,direct_system_identity_isomorphism_family,pair_eq_iff}.
    \end{subproof}
    Show that $E$ is right-unique.
    \begin{subproof}
        Follows by
            \auto{rightunique,direct_system_identity_isomorphism_family_iff}.
    \end{subproof}
    $\dom{E}=\carrier[S]$ by
        \auto{setext,dom_iff,direct_system_identity_isomorphism_family_iff}.
    For every $i\in\dom{E}$ we have $\apply{E}{i}\in\ran{E}$ by
        \auto{fun_ran_iff}.
    Hence $E$ is a function to $\ran{E}$.
    Follows by \auto{funs_intro}.
\end{proof}

\begin{proposition}\label{direct_system_identity_isomorphism_family_apply}
    Let $i\in\carrier[S]$.
    Then $\apply{\directSystemIdentityIsomorphismFamily{S}}{i}
        =\identity{\carrier[\directSystemGroup{S}{i}]}$.
\end{proposition}
\begin{proof}
    $\directSystemIdentityIsomorphismFamily{S}$ is a function by
        \auto{direct_system_identity_isomorphism_family_type,funs_is_function}.
    Follows by
        \obv{function_apply_intro,direct_system_identity_isomorphism_family_iff}.
\end{proof}

\begin{theorem}\label{identity_group_direct_system_isomorphism}
    Let $S$ be a group direct system.
    Then $\groupDirectSystemIsomorphism
        {\directSystemIdentityIsomorphismFamily{S}}{S}{S}$.
\end{theorem}
\begin{proof}
    Let $E=\directSystemIdentityIsomorphismFamily{S}$.
    $E\in\funs{\carrier[S]}{\ran{E}}$ by
        \auto{direct_system_identity_isomorphism_family_type}.
    Show that for every $i\in\carrier[S]$ we have
        $\groupIsomorphism{\apply{E}{i}}
            {\directSystemGroup{S}{i}}{\directSystemGroup{S}{i}}$.
    \begin{subproof}
        Fix $i\in\carrier[S]$.
        $\directSystemGroup{S}{i}$ is a group by
            \obv{direct_system_group_is_group}.
        $\apply{E}{i}=\identity{\carrier[\directSystemGroup{S}{i}]}$ by
            \auto{direct_system_identity_isomorphism_family_apply}.
        Follows by \auto{identity_group_isomorphism}.
    \end{subproof}
    Show that for all $i,j\in\carrier[S]$ such that
        $i\mathrel{\leq[S]}j$ we have
        $\apply{E}{j}\circ\directSystemBond{S}{i}{j}
            =\directSystemBond{S}{i}{j}\circ\apply{E}{i}$.
    \begin{subproof}
        Fix $i,j\in\carrier[S]$ such that $i\mathrel{\leq[S]}j$.
        Let $b=\directSystemBond{S}{i}{j}$.
        $\groupHom{b}{\directSystemGroup{S}{i}}
            {\directSystemGroup{S}{j}}$ by
            \auto{direct_system_bond_hom}.
        Hence $b\in
            \funs{\carrier[\directSystemGroup{S}{i}]}
                {\carrier[\directSystemGroup{S}{j}]}$ by
            \obv{group_hom_type}.
        $\apply{E}{i}=\identity{\carrier[\directSystemGroup{S}{i}]}$ and
            $\apply{E}{j}=\identity{\carrier[\directSystemGroup{S}{j}]}$ by
            \auto{direct_system_identity_isomorphism_family_apply}.
        $\identity{\carrier[\directSystemGroup{S}{j}]}\circ b=b$ and
            $b\circ\identity{\carrier[\directSystemGroup{S}{i}]}=b$ by
            \obv{funs_identity_circ_left,funs_circ_identity_right}.
    \end{subproof}
    Follows by \auto{group_direct_system_isomorphism}.
\end{proof}

\begin{theorem}\label{group_direct_system_isomorphism_composition}
    Suppose $\groupDirectSystemIsomorphism{E}{S}{T}$ and
        $\groupDirectSystemIsomorphism{D}{T}{U}$.
    Let $F=\pointwiseCompositionFamily{\carrier[S]}{D}{E}$.
    Then $\groupDirectSystemIsomorphism{F}{S}{U}$.
\end{theorem}
\begin{proof}
    $S$ is a group direct system and $T$ is a group direct system by
        \obv{group_direct_system_isomorphism_source,group_direct_system_isomorphism_target}.
    $U$ is a group direct system by
        \obv{group_direct_system_isomorphism_target}.
    $\carrier[S]=\carrier[T]=\carrier[U]$ and
        $\leq[S]=\leq[T]=\leq[U]$ by
        \obv{group_direct_system_isomorphism_carrier,group_direct_system_isomorphism_order}.
    $F\in\funs{\carrier[S]}{\ran{F}}$ by
        \auto{pointwise_composition_family_type}.
    Show that for every $i\in\carrier[S]$ we have
        $\groupIsomorphism{\apply{F}{i}}
            {\directSystemGroup{S}{i}}{\directSystemGroup{U}{i}}$.
    \begin{subproof}
        Fix $i\in\carrier[S]$.
        Hence $i\in\carrier[T]$.
        $\apply{F}{i}=\apply{D}{i}\circ\apply{E}{i}$ by
            \auto{pointwise_composition_family_apply}.
        $\groupIsomorphism{\apply{E}{i}}
            {\directSystemGroup{S}{i}}{\directSystemGroup{T}{i}}$ and
            $\groupIsomorphism{\apply{D}{i}}
            {\directSystemGroup{T}{i}}{\directSystemGroup{U}{i}}$ by
            \obv{group_direct_system_isomorphism_component}.
        Follows by \auto{group_isomorphism_composition}.
    \end{subproof}
    Show that for all $i,j\in\carrier[S]$ such that
        $i\mathrel{\leq[S]}j$ we have
        $\apply{F}{j}\circ\directSystemBond{S}{i}{j}
            =\directSystemBond{U}{i}{j}\circ\apply{F}{i}$.
    \begin{subproof}
        Fix $i,j\in\carrier[S]$ such that $i\mathrel{\leq[S]}j$.
        Hence $i,j\in\carrier[T]$ and
            $i\mathrel{\leq[T]}j$.
        $\apply{E}{j}\circ\directSystemBond{S}{i}{j}
            =\directSystemBond{T}{i}{j}\circ\apply{E}{i}$ and
            $\apply{D}{j}\circ\directSystemBond{T}{i}{j}
            =\directSystemBond{U}{i}{j}\circ\apply{D}{i}$ by
            \obv{group_direct_system_isomorphism_square}.
        $\apply{F}{i}=\apply{D}{i}\circ\apply{E}{i}$ and
            $\apply{F}{j}=\apply{D}{j}\circ\apply{E}{j}$ by
            \auto{pointwise_composition_family_apply}.
        $(\apply{D}{j}\circ\apply{E}{j})\circ
            \directSystemBond{S}{i}{j}
            =\apply{D}{j}\circ
                (\apply{E}{j}\circ\directSystemBond{S}{i}{j})$ by
            \obv{circ_assoc}.
        Hence $(\apply{D}{j}\circ\apply{E}{j})\circ
            \directSystemBond{S}{i}{j}
            =\apply{D}{j}\circ
                (\directSystemBond{T}{i}{j}\circ\apply{E}{i})$.
        $\apply{D}{j}\circ
            (\directSystemBond{T}{i}{j}\circ\apply{E}{i})
            =(\apply{D}{j}\circ\directSystemBond{T}{i}{j})
                \circ\apply{E}{i}$ by
            \obv{circ_assoc}.
        Hence $\apply{D}{j}\circ
            (\directSystemBond{T}{i}{j}\circ\apply{E}{i})
            =(\directSystemBond{U}{i}{j}\circ\apply{D}{i})
                \circ\apply{E}{i}$.
        $(\directSystemBond{U}{i}{j}\circ\apply{D}{i})
                \circ\apply{E}{i}
            =\directSystemBond{U}{i}{j}\circ
                (\apply{D}{i}\circ\apply{E}{i})$ by
            \obv{circ_assoc}.
    \end{subproof}
    Follows by \obv{group_direct_system_isomorphism}.
\end{proof}

\begin{theorem}\label{group_direct_system_isomorphism_converse}
    Suppose $\groupDirectSystemIsomorphism{E}{S}{T}$.
    Let $C=\pointwiseConverseFamily{\carrier[T]}{E}$.
    Then $\groupDirectSystemIsomorphism{C}{T}{S}$.
\end{theorem}
\begin{proof}
    $S$ is a group direct system and $T$ is a group direct system by
        \obv{group_direct_system_isomorphism_source,group_direct_system_isomorphism_target}.
    $\carrier[S]=\carrier[T]$ and $\leq[S]=\leq[T]$ by
        \obv{group_direct_system_isomorphism_carrier,group_direct_system_isomorphism_order}.
    $C\in\funs{\carrier[T]}{\ran{C}}$ by
        \auto{pointwise_converse_family_type}.
    Show that for every $i\in\carrier[T]$ we have
        $\groupIsomorphism{\apply{C}{i}}
            {\directSystemGroup{T}{i}}{\directSystemGroup{S}{i}}$.
    \begin{subproof}
        Fix $i\in\carrier[T]$.
        Hence $i\in\carrier[S]$.
        $\apply{C}{i}=\converse{\apply{E}{i}}$ by
            \auto{pointwise_converse_family_apply}.
        $\groupIsomorphism{\apply{E}{i}}
            {\directSystemGroup{S}{i}}{\directSystemGroup{T}{i}}$ by
            \obv{group_direct_system_isomorphism_component}.
        Follows by \auto{group_isomorphism_converse}.
    \end{subproof}
    Show that for all $i,j\in\carrier[T]$ such that
        $i\mathrel{\leq[T]}j$ we have
        $\apply{C}{j}\circ\directSystemBond{T}{i}{j}
            =\directSystemBond{S}{i}{j}\circ\apply{C}{i}$.
    \begin{subproof}
        Fix $i,j\in\carrier[T]$ such that $i\mathrel{\leq[T]}j$.
        Hence $i,j\in\carrier[S]$ and
            $i\mathrel{\leq[S]}j$.
        Let $e=\apply{E}{i}$.
        Let $d=\apply{E}{j}$.
        Let $b=\directSystemBond{S}{i}{j}$.
        Let $c=\directSystemBond{T}{i}{j}$.
        Let $q=\converse{d}\circ c$.
        $\groupIsomorphism{e}
            {\directSystemGroup{S}{i}}{\directSystemGroup{T}{i}}$ and
            $\groupIsomorphism{d}
            {\directSystemGroup{S}{j}}{\directSystemGroup{T}{j}}$ by
            \auto{group_direct_system_isomorphism_component}.
        Hence $\groupIsomorphism{\converse{d}}
            {\directSystemGroup{T}{j}}{\directSystemGroup{S}{j}}$ by
            \obv{group_isomorphism_converse}.
        $\groupHom{c}{\directSystemGroup{T}{i}}
            {\directSystemGroup{T}{j}}$ and
            $\groupHom{b}{\directSystemGroup{S}{i}}
            {\directSystemGroup{S}{j}}$ by
            \auto{direct_system_bond_hom}.
        Hence $\groupHom{q}{\directSystemGroup{T}{i}}
            {\directSystemGroup{S}{j}}$ by
            \auto{group_isomorphism_is_hom,group_hom_composition}.
        Hence $q\in
            \funs{\carrier[\directSystemGroup{T}{i}]}
                {\carrier[\directSystemGroup{S}{j}]}$ and
            $b\in
            \funs{\carrier[\directSystemGroup{S}{i}]}
                {\carrier[\directSystemGroup{S}{j}]}$ by
            \obv{group_hom_type}.
        $d\circ b=c\circ e$ by
            \auto{group_direct_system_isomorphism_square}.
        $d\in\bijections
            {\carrier[\directSystemGroup{S}{j}]}
            {\carrier[\directSystemGroup{T}{j}]}$ by
            \obv{group_isomorphism_is_bijection}.
        Hence $\converse{d}\circ d
            =\identity{\carrier[\directSystemGroup{S}{j}]}$ by
            \obv{bijection_converse_circ}.
        \begin{align*}
            q\circ e
                &=(\converse{d}\circ c)\circ e
                \\
                &=\converse{d}\circ(c\circ e)
                    \explanation{by \obv{circ_assoc}}
                \\
                &=\converse{d}\circ(d\circ b)
                \\
                &=(\converse{d}\circ d)\circ b
                    \explanation{by \obv{circ_assoc}}
                \\
                &=\identity{\carrier[\directSystemGroup{S}{j}]}\circ b
                \\
                &=b
                    \explanation{by \obv{funs_identity_circ_left}}
        \end{align*}
        $e\in\bijections
            {\carrier[\directSystemGroup{S}{i}]}
            {\carrier[\directSystemGroup{T}{i}]}$ by
            \obv{group_isomorphism_is_bijection}.
        Hence $e\circ\converse{e}
            =\identity{\carrier[\directSystemGroup{T}{i}]}$ by
            \obv{bijection_circ_converse}.
        \begin{align*}
            b\circ\converse{e}
                &=(q\circ e)\circ\converse{e}
                \\
                &=q\circ(e\circ\converse{e})
                    \explanation{by \obv{circ_assoc}}
                \\
                &=q\circ\identity{\carrier[\directSystemGroup{T}{i}]}
                \\
                &=q
                    \explanation{by \obv{funs_circ_identity_right}}
        \end{align*}
        $\apply{C}{i}=\converse{e}$ and
            $\apply{C}{j}=\converse{d}$ by
            \auto{pointwise_converse_family_apply}.
    \end{subproof}
    Follows by \obv{group_direct_system_isomorphism}.
\end{proof}

\subsection{The transported system}

\begin{definition}\label{direct_system_transport_bond_family}
    $\directSystemTransportBondFamily{S}{E}=
        \{(p,
            \apply{E}{\snd{p}}\circ
                (\directSystemBond{S}{\fst{p}}{\snd{p}}\circ
                    \converse{\apply{E}{\fst{p}}}))
            \mid p\in\leq[S]\}$.
\end{definition}

\begin{proposition}\label{direct_system_transport_bond_family_iff}
    $(p,f)\in\directSystemTransportBondFamily{S}{E}$ iff
        $p\in\leq[S]$ and
        $f=\apply{E}{\snd{p}}\circ
            (\directSystemBond{S}{\fst{p}}{\snd{p}}\circ
                \converse{\apply{E}{\fst{p}}})$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{direct_system_transport_bond_family,pair_eq_iff}.
\end{proof}

\begin{proposition}\label{direct_system_transport_bond_family_type}
    $\directSystemTransportBondFamily{S}{E}\in
        \funs{\leq[S]}{\ran{\directSystemTransportBondFamily{S}{E}}}$.
\end{proposition}
\begin{proof}
    Let $\Phi=\directSystemTransportBondFamily{S}{E}$.
    Show that $\Phi$ is a relation.
    \begin{subproof}
        Follows by
            \auto{relation,direct_system_transport_bond_family,pair_eq_iff}.
    \end{subproof}
    Show that $\Phi$ is right-unique.
    \begin{subproof}
        Follows by
            \auto{rightunique,direct_system_transport_bond_family_iff}.
    \end{subproof}
    $\dom{\Phi}=\leq[S]$ by
        \auto{setext,dom_iff,direct_system_transport_bond_family_iff}.
    For every $p\in\dom{\Phi}$ we have
        $\apply{\Phi}{p}\in\ran{\Phi}$ by
        \auto{fun_ran_iff}.
    Hence $\Phi$ is a function to $\ran{\Phi}$.
    Follows by \auto{funs_intro}.
\end{proof}

\begin{proposition}\label{direct_system_transport_bond_family_apply}
    Let $i,j\in\carrier[S]$.
    Suppose $i\mathrel{\leq[S]}j$.
    Then
        $\apply{\directSystemTransportBondFamily{S}{E}}{(i,j)}
            =\apply{E}{j}\circ
                (\directSystemBond{S}{i}{j}\circ

\begin{theorem}\label{transported_group_direct_system_is_exact}
    Let $S$ be a group direct system.
    Let $H\in\funs{\carrier[S]}{\ran{H}}$.
    Suppose for every $i\in\carrier[S]$ we have
        $\apply{H}{i}$ is a group.
    Let $E\in\funs{\carrier[S]}{\ran{E}}$.
    Suppose for every $i\in\carrier[S]$ we have
        $\groupIsomorphism{\apply{E}{i}}
            {\directSystemGroup{S}{i}}{\apply{H}{i}}$.
    Then $\transportedGroupDirectSystem{S}{H}{E}$ is an exact
        group direct system.
\end{theorem}
\begin{proof}
    Let $T=\transportedGroupDirectSystem{S}{H}{E}$.
    Let $\Phi=\directSystemTransportBondFamily{S}{E}$.
    $S$ is a directed quasiordered set by
        \obv{group_direct_system_is_directed}.
    $\Phi\in\funs{\leq[S]}{\ran{\Phi}}$ by
        \auto{direct_system_transport_bond_family_type}.
    Show that for all $i,j\in\carrier[S]$ such that
        $i\mathrel{\leq[S]}j$ we have
        $\groupHom{\apply{\Phi}{(i,j)}}
            {\apply{H}{i}}{\apply{H}{j}}$.
    \begin{subproof}
        Fix $i,j\in\carrier[S]$ such that $i\mathrel{\leq[S]}j$.
        $\groupIsomorphism{\apply{E}{i}}
            {\directSystemGroup{S}{i}}{\apply{H}{i}}$ and
            $\groupIsomorphism{\apply{E}{j}}
            {\directSystemGroup{S}{j}}{\apply{H}{j}}$.
        Hence $\groupHom{\converse{\apply{E}{i}}}
            {\apply{H}{i}}{\directSystemGroup{S}{i}}$ and
            $\groupHom{\apply{E}{j}}
            {\directSystemGroup{S}{j}}{\apply{H}{j}}$ by
            \obv{group_isomorphism_converse_hom,group_isomorphism_is_hom}.
        $\groupHom{\directSystemBond{S}{i}{j}}
            {\directSystemGroup{S}{i}}{\directSystemGroup{S}{j}}$ by
            \obv{direct_system_bond_hom}.
        Hence $\groupHom
            {\directSystemBond{S}{i}{j}\circ
                \converse{\apply{E}{i}}}
            {\apply{H}{i}}{\directSystemGroup{S}{j}}$ by
            \obv{group_hom_composition}.
        Hence $\groupHom
            {\apply{E}{j}\circ
                (\directSystemBond{S}{i}{j}\circ
                    \converse{\apply{E}{i}})}
            {\apply{H}{i}}{\apply{H}{j}}$ by
            \obv{group_hom_composition}.
        $\apply{\Phi}{(i,j)}=
            \apply{E}{j}\circ
                (\directSystemBond{S}{i}{j}\circ
                    \converse{\apply{E}{i}})$ by
            \auto{direct_system_transport_bond_family_apply}.
    \end{subproof}
    Show that for every $i\in\carrier[S]$ we have
        $\apply{\Phi}{(i,i)}=\identity{\carrier[\apply{H}{i}]}$.
    \begin{subproof}
        Fix $i\in\carrier[S]$.
        Let $e=\apply{E}{i}$.
        Let $G=\directSystemGroup{S}{i}$.
        Let $K=\apply{H}{i}$.
        $\groupIsomorphism{e}{G}{K}$ by \auto.
        Hence $e\in\bijections{\carrier[G]}{\carrier[K]}$ by
            \obv{group_isomorphism_is_bijection}.
        Hence $\converse{e}\in\funs{\carrier[K]}{\carrier[G]}$ by
            \obv{bijective_converse_are_funs}.
        $\directSystemBond{S}{i}{i}=\identity{\carrier[G]}$ by
            \auto{direct_system_bond_identity}.
        $\identity{\carrier[G]}\circ\converse{e}=\converse{e}$ by
            \obv{funs_identity_circ_left}.
        $e\circ\converse{e}=\identity{\carrier[K]}$ by
            \obv{bijection_circ_converse}.
        $\apply{\Phi}{(i,i)}=
            e\circ(\directSystemBond{S}{i}{i}\circ\converse{e})$ by
            \auto{direct_system_transport_bond_family_apply,directed_order_reflexive}.
    \end{subproof}
    Show that for all $i,j,k\in\carrier[S]$ such that
        $i\mathrel{\leq[S]}j$ and $j\mathrel{\leq[S]}k$ we have
        $\apply{\Phi}{(i,k)}=
            \apply{\Phi}{(j,k)}\circ\apply{\Phi}{(i,j)}$.
    \begin{subproof}
        Fix $i,j,k\in\carrier[S]$.
        Assume $i\mathrel{\leq[S]}j$ and
            $j\mathrel{\leq[S]}k$.
        $i\mathrel{\leq[S]}k$ by
            \obv{directed_order_transitive}.
        Let $e=\apply{E}{i}$.
        Let $d=\apply{E}{j}$.
        Let $q=\apply{E}{k}$.
        Let $b=\directSystemBond{S}{i}{j}$.
        Let $c=\directSystemBond{S}{j}{k}$.
        Let $a=\directSystemBond{S}{i}{k}$.
        Let $x=b\circ\converse{e}$.
        Let $p=c\circ\converse{d}$.
        $\groupIsomorphism{e}
            {\directSystemGroup{S}{i}}{\apply{H}{i}}$ and
            $\groupIsomorphism{d}
            {\directSystemGroup{S}{j}}{\apply{H}{j}}$ by \auto.
        Hence $\groupHom{\converse{e}}
            {\apply{H}{i}}{\directSystemGroup{S}{i}}$ and
            $\groupHom{\converse{d}}
            {\apply{H}{j}}{\directSystemGroup{S}{j}}$ by
            \obv{group_isomorphism_converse_hom}.
        $\groupHom{b}{\directSystemGroup{S}{i}}
            {\directSystemGroup{S}{j}}$ and
            $\groupHom{c}{\directSystemGroup{S}{j}}
            {\directSystemGroup{S}{k}}$ by
            \auto{direct_system_bond_hom}.
        Hence $\groupHom{x}{\apply{H}{i}}
            {\directSystemGroup{S}{j}}$ and
            $\groupHom{p}{\apply{H}{j}}
            {\directSystemGroup{S}{k}}$ by
            \auto{group_hom_composition}.
        Hence $x\in\funs{\carrier[\apply{H}{i}]}
            {\carrier[\directSystemGroup{S}{j}]}$ by
            \obv{group_hom_type}.
        $d\in\bijections
            {\carrier[\directSystemGroup{S}{j}]}
            {\carrier[\apply{H}{j}]}$ by
            \obv{group_isomorphism_is_bijection}.
        Hence $\converse{d}\circ d
            =\identity{\carrier[\directSystemGroup{S}{j}]}$ by
            \obv{bijection_converse_circ}.
        $\identity{\carrier[\directSystemGroup{S}{j}]}\circ x=x$ by
            \obv{funs_identity_circ_left}.
        $p\circ(d\circ x)=c\circ(\converse{d}\circ(d\circ x))$ by
            \auto{circ_assoc}.
        $\converse{d}\circ(d\circ x)
            =(\converse{d}\circ d)\circ x$ by
            \obv{circ_assoc}.
        Hence $p\circ(d\circ x)=c\circ x$.
        $c\circ x=c\circ(b\circ\converse{e})$.
        $c\circ(b\circ\converse{e})
            =(c\circ b)\circ\converse{e}$ by
            \obv{circ_assoc}.
        $a=c\circ b$ by
            \auto{direct_system_bond_composition}.
        Hence $p\circ(d\circ x)=a\circ\converse{e}$.
        $(q\circ p)\circ(d\circ x)
            =q\circ(p\circ(d\circ x))$ by
            \obv{circ_assoc}.
        Hence $(q\circ p)\circ(d\circ x)
            =q\circ(a\circ\converse{e})$.
        $\apply{\Phi}{(i,j)}=d\circ x$ and
            $\apply{\Phi}{(j,k)}=q\circ p$ and
            $\apply{\Phi}{(i,k)}=q\circ(a\circ\converse{e})$ by
            \auto{direct_system_transport_bond_family_apply}.
    \end{subproof}
    $\groupDirectSystemStructure{S}{H}{\Phi}$ is an exact
        group direct system by
        \obv{group_direct_system_structure_is_system}.
    Follows by \auto{transported_group_direct_system}.
\end{proof}

\begin{theorem}\label{transported_group_direct_system_isomorphism}
    Let $S$ be a group direct system.
    Let $H\in\funs{\carrier[S]}{\ran{H}}$.
    Suppose for every $i\in\carrier[S]$ we have
        $\apply{H}{i}$ is a group.
    Let $E\in\funs{\carrier[S]}{\ran{E}}$.
    Suppose for every $i\in\carrier[S]$ we have
        $\groupIsomorphism{\apply{E}{i}}
            {\directSystemGroup{S}{i}}{\apply{H}{i}}$.
    Then $\groupDirectSystemIsomorphism
        {E}{S}{\transportedGroupDirectSystem{S}{H}{E}}$.
\end{theorem}
\begin{proof}
    Let $T=\transportedGroupDirectSystem{S}{H}{E}$.
    $T$ is a group direct system by
        \auto{transported_group_direct_system_is_exact}.
    $\carrier[T]=\carrier[S]$ and $\leq[T]=\leq[S]$ by
        \auto{transported_group_direct_system_carrier,transported_group_direct_system_order}.
    For every $i\in\carrier[S]$ we have
        $\directSystemGroup{T}{i}=\apply{H}{i}$ by
        \auto{transported_group_direct_system_group}.
    Hence for every $i\in\carrier[S]$ we have
        $\groupIsomorphism{\apply{E}{i}}
            {\directSystemGroup{S}{i}}{\directSystemGroup{T}{i}}$ by
        \auto.
    For all $i,j\in\carrier[S]$ such that
        $i\mathrel{\leq[S]}j$ we have
        $\directSystemBond{T}{i}{j}
            =\apply{E}{j}\circ
                (\directSystemBond{S}{i}{j}\circ
                    \converse{\apply{E}{i}})$ by
        \auto{transported_group_direct_system_bond}.
    Show that for all $i,j\in\carrier[S]$ such that
        $i\mathrel{\leq[S]}j$ we have
        $\apply{E}{j}\circ\directSystemBond{S}{i}{j}
            =\directSystemBond{T}{i}{j}\circ\apply{E}{i}$.
    \begin{subproof}
        Fix $i,j\in\carrier[S]$ such that $i\mathrel{\leq[S]}j$.
        Let $e=\apply{E}{i}$.
        Let $d=\apply{E}{j}$.
        Let $b=\directSystemBond{S}{i}{j}$.
        $\groupIsomorphism{e}
            {\directSystemGroup{S}{i}}{\apply{H}{i}}$ by \auto.
        Hence $e\in\bijections
            {\carrier[\directSystemGroup{S}{i}]}
            {\carrier[\apply{H}{i}]}$ by
            \obv{group_isomorphism_is_bijection}.
        Hence $\converse{e}\circ e
            =\identity{\carrier[\directSystemGroup{S}{i}]}$ by
            \obv{bijection_converse_circ}.
        $\groupHom{b}{\directSystemGroup{S}{i}}
            {\directSystemGroup{S}{j}}$ by
            \auto{direct_system_bond_hom}.
        Hence $b\in
            \funs{\carrier[\directSystemGroup{S}{i}]}
                {\carrier[\directSystemGroup{S}{j}]}$ by
            \obv{group_hom_type}.
        $b\circ\identity{\carrier[\directSystemGroup{S}{i}]}=b$ by
            \obv{funs_circ_identity_right}.
        $(b\circ\converse{e})\circ e
            =b\circ(\converse{e}\circ e)$ by
            \obv{circ_assoc}.
        Hence $(b\circ\converse{e})\circ e=b$ by \auto.
        $(d\circ(b\circ\converse{e}))\circ e
            =d\circ((b\circ\converse{e})\circ e)$ by
            \obv{circ_assoc}.
        Hence $(d\circ(b\circ\converse{e}))\circ e=d\circ b$.
        $\directSystemBond{T}{i}{j}=d\circ(b\circ\converse{e})$ by
            \auto{transported_group_direct_system_bond}.
    \end{subproof}
    Follows by \auto{group_direct_system_isomorphism}.
\end{proof}

\begin{theorem}\label{group_cocone_pullback_along_system_isomorphism}
    Suppose $\groupDirectSystemIsomorphism{E}{S}{T}$ and
        $\groupCocone{T}{H}{\kappa}$.
    Then $\groupCocone
        {S}{H}{\pulledGroupCoconeFamily{S}{E}{\kappa}}$.
\end{theorem}
\begin{proof}
    Let $P=\pulledGroupCoconeFamily{S}{E}{\kappa}$.
    $S$ is a group direct system by
        \obv{group_direct_system_isomorphism_source}.
    $T$ is a group direct system by
        \obv{group_direct_system_isomorphism_target}.
    $\carrier[S]=\carrier[T]$ and $\leq[S]=\leq[T]$ by
        \obv{group_direct_system_isomorphism_carrier,group_direct_system_isomorphism_order}.
    $P\in\funs{\carrier[S]}{\ran{P}}$ by
        \auto{pulled_group_cocone_family_type}.
    Show that for every $i\in\carrier[S]$ we have
        $\groupHom{\apply{P}{i}}{\directSystemGroup{S}{i}}{H}$.
    \begin{subproof}
        Fix $i\in\carrier[S]$.
        Hence $i\in\carrier[T]$.
        $\groupIsomorphism{\apply{E}{i}}
            {\directSystemGroup{S}{i}}{\directSystemGroup{T}{i}}$ by
            \obv{group_direct_system_isomorphism_component}.
        Hence $\groupHom{\apply{E}{i}}
            {\directSystemGroup{S}{i}}{\directSystemGroup{T}{i}}$ by
            \obv{group_isomorphism_is_hom}.
        $\groupHom{\apply{\kappa}{i}}
            {\directSystemGroup{T}{i}}{H}$ by
            \obv{group_cocone_leg_hom}.
        Hence $\groupHom{\apply{\kappa}{i}\circ\apply{E}{i}}
            {\directSystemGroup{S}{i}}{H}$ by
            \obv{group_hom_composition}.
        $\apply{P}{i}=\apply{\kappa}{i}\circ\apply{E}{i}$ by
            \auto{pulled_group_cocone_family_apply}.
    \end{subproof}
    Show that for all $i,j\in\carrier[S]$ such that
        $i\mathrel{\leq[S]}j$ we have
        $\apply{P}{j}\circ\directSystemBond{S}{i}{j}
            =\apply{P}{i}$.
    \begin{subproof}
        Fix $i,j\in\carrier[S]$ such that $i\mathrel{\leq[S]}j$.
        Hence $i,j\in\carrier[T]$ and
            $i\mathrel{\leq[T]}j$.
        $\apply{E}{j}\circ\directSystemBond{S}{i}{j}
            =\directSystemBond{T}{i}{j}\circ\apply{E}{i}$ by
            \obv{group_direct_system_isomorphism_square}.
        $\apply{\kappa}{j}\circ\directSystemBond{T}{i}{j}
            =\apply{\kappa}{i}$ by
            \obv{group_cocone_compatibility}.
        $\apply{P}{i}=\apply{\kappa}{i}\circ\apply{E}{i}$ and
            $\apply{P}{j}=\apply{\kappa}{j}\circ\apply{E}{j}$ by
            \auto{pulled_group_cocone_family_apply}.
        $(\apply{\kappa}{j}\circ\apply{E}{j})\circ
            \directSystemBond{S}{i}{j}
            =\apply{\kappa}{j}\circ
                (\apply{E}{j}\circ\directSystemBond{S}{i}{j})$ by
            \obv{circ_assoc}.
        Hence $(\apply{\kappa}{j}\circ\apply{E}{j})\circ
            \directSystemBond{S}{i}{j}
            =\apply{\kappa}{j}\circ
                (\directSystemBond{T}{i}{j}\circ\apply{E}{i})$.
        $\apply{\kappa}{j}\circ
            (\directSystemBond{T}{i}{j}\circ\apply{E}{i})
            =(\apply{\kappa}{j}\circ\directSystemBond{T}{i}{j})
                \circ\apply{E}{i}$ by
            \obv{circ_assoc}.
    \end{subproof}
    Follows by \auto{group_cocone}.
\end{proof}

\subsection{Semantic transport of limits}

\begin{definition}\label{group_direct_system_limit_isomorphism}
    $\groupDirectSystemLimitIsomorphism
        {E}{S}{T}{L}{\iota}{M}{\kappa}{u}$ iff
        $\groupIsomorphism{u}{L}{M}$ and
        for every $i\in\carrier[S]$ we have
            $u\circ\apply{\iota}{i}
                =\apply{\kappa}{i}\circ\apply{E}{i}$.
\end{definition}

\begin{proposition}\label{group_direct_system_limit_isomorphism_is_iso}
    Suppose $\groupDirectSystemLimitIsomorphism
        {E}{S}{T}{L}{\iota}{M}{\kappa}{u}$.
    Then $\groupIsomorphism{u}{L}{M}$.
\end{proposition}
\begin{proof}
    Follows by \obv{group_direct_system_limit_isomorphism}.
\end{proof}

\begin{proposition}\label{group_direct_system_limit_isomorphism_compatibility}
    Suppose $\groupDirectSystemLimitIsomorphism
        {E}{S}{T}{L}{\iota}{M}{\kappa}{u}$.
    Let $i\in\carrier[S]$.
    Then $u\circ\apply{\iota}{i}
        =\apply{\kappa}{i}\circ\apply{E}{i}$.
\end{proposition}
\begin{proof}
    Follows by \obv{group_direct_system_limit_isomorphism}.
\end{proof}

\begin{lemma}\label{system_isomorphism_limit_mediators_are_inverse}
    Suppose $\groupDirectSystemIsomorphism{E}{S}{T}$.
    Suppose $\directLimitCocone{S}{L}{\iota}$ and
        $\directLimitCocone{T}{M}{\kappa}$.
    Let $C=\pointwiseConverseFamily{\carrier[T]}{E}$.
    Let $P=\pulledGroupCoconeFamily{S}{E}{\kappa}$.
    Let $Q=\pulledGroupCoconeFamily{T}{C}{\iota}$.
    Suppose $\groupCoconeMediator{S}{L}{\iota}{M}{P}{u}$ and
        $\groupCoconeMediator{T}{M}{\kappa}{L}{Q}{v}$.
    Then $v\circ u=\identity{\carrier[L]}$ and
        $u\circ v=\identity{\carrier[M]}$.
\end{lemma}
\begin{proof}
    $\groupCocone{S}{L}{\iota}$ and
        $\groupCocone{T}{M}{\kappa}$ by
        \obv{direct_limit_cocone_is_cocone}.
    $\carrier[S]=\carrier[T]$ by
        \obv{group_direct_system_isomorphism_carrier}.
    $\groupHom{u}{L}{M}$ and $\groupHom{v}{M}{L}$ by
        \obv{group_cocone_mediator_hom}.
    Hence $\groupHom{v\circ u}{L}{L}$ and
        $\groupHom{u\circ v}{M}{M}$ by
        \obv{group_hom_composition}.
    Show that for every $i\in\carrier[S]$ we have
        $(v\circ u)\circ\apply{\iota}{i}=\apply{\iota}{i}$.
    \begin{subproof}
        Fix $i\in\carrier[S]$.
        Hence $i\in\carrier[T]$.
        Let $e=\apply{E}{i}$.
        Let $c=\apply{C}{i}$.
        $u\circ\apply{\iota}{i}=\apply{P}{i}$ and
            $v\circ\apply{\kappa}{i}=\apply{Q}{i}$ by
            \obv{group_cocone_mediator_compatibility}.
        $\apply{P}{i}=\apply{\kappa}{i}\circ e$ and
            $\apply{Q}{i}=\apply{\iota}{i}\circ c$ by
            \auto{pulled_group_cocone_family_apply}.
        $c=\converse{e}$ by
            \auto{pointwise_converse_family_apply}.
        $\groupIsomorphism{e}
            {\directSystemGroup{S}{i}}{\directSystemGroup{T}{i}}$ by
            \auto{group_direct_system_isomorphism_component}.
        Hence $e\in\bijections
            {\carrier[\directSystemGroup{S}{i}]}
            {\carrier[\directSystemGroup{T}{i}]}$ by
            \obv{group_isomorphism_is_bijection}.
        Hence $c\circ e
            =\identity{\carrier[\directSystemGroup{S}{i}]}$ by
            \auto{bijection_converse_circ}.
        $\groupHom{\apply{\iota}{i}}
            {\directSystemGroup{S}{i}}{L}$ by
            \obv{group_cocone_leg_hom}.
        Hence $\apply{\iota}{i}\in
            \funs{\carrier[\directSystemGroup{S}{i}]}{\carrier[L]}$ by
            \obv{group_hom_type}.
        Hence $\apply{\iota}{i}\circ
            \identity{\carrier[\directSystemGroup{S}{i}]}
            =\apply{\iota}{i}$ by
            \obv{funs_circ_identity_right}.
        $(v\circ u)\circ\apply{\iota}{i}
            =v\circ(u\circ\apply{\iota}{i})$ by
            \obv{circ_assoc}.
        Hence $(v\circ u)\circ\apply{\iota}{i}
            =v\circ(\apply{\kappa}{i}\circ e)$.
        $v\circ(\apply{\kappa}{i}\circ e)
            =(v\circ\apply{\kappa}{i})\circ e$ by
            \obv{circ_assoc}.
        Hence $v\circ(\apply{\kappa}{i}\circ e)
            =(\apply{\iota}{i}\circ c)\circ e$.
        $(\apply{\iota}{i}\circ c)\circ e
            =\apply{\iota}{i}\circ(c\circ e)$ by
            \obv{circ_assoc}.
    \end{subproof}
    Hence $\groupCoconeMediator
        {S}{L}{\iota}{L}{\iota}{v\circ u}$ by
        \obv{group_cocone_mediator}.
    $\groupHom{\identity{\carrier[L]}}{L}{L}$ by
        \obv{identity_group_hom}.
    Show that for every $i\in\carrier[S]$ we have
        $\identity{\carrier[L]}\circ\apply{\iota}{i}
            =\apply{\iota}{i}$.
    \begin{subproof}
        Fix $i\in\carrier[S]$.
        $\groupHom{\apply{\iota}{i}}
            {\directSystemGroup{S}{i}}{L}$ by
            \obv{group_cocone_leg_hom}.
        Hence $\apply{\iota}{i}\in
            \funs{\carrier[\directSystemGroup{S}{i}]}{\carrier[L]}$ by
            \obv{group_hom_type}.
        Follows by \obv{funs_identity_circ_left}.
    \end{subproof}
    Hence $\groupCoconeMediator
        {S}{L}{\iota}{L}{\iota}{\identity{\carrier[L]}}$ by
        \obv{group_cocone_mediator}.
    Hence $v\circ u=\identity{\carrier[L]}$ by
        \obv{direct_limit_cocone_mediator_unique}.
    Show that for every $i\in\carrier[T]$ we have
        $(u\circ v)\circ\apply{\kappa}{i}=\apply{\kappa}{i}$.
    \begin{subproof}
        Fix $i\in\carrier[T]$.
        Hence $i\in\carrier[S]$.
        Let $e=\apply{E}{i}$.
        Let $c=\apply{C}{i}$.
        $u\circ\apply{\iota}{i}=\apply{P}{i}$ and
            $v\circ\apply{\kappa}{i}=\apply{Q}{i}$ by
            \obv{group_cocone_mediator_compatibility}.
        $\apply{P}{i}=\apply{\kappa}{i}\circ e$ and
            $\apply{Q}{i}=\apply{\iota}{i}\circ c$ by
            \auto{pulled_group_cocone_family_apply}.
        $c=\converse{e}$ by
            \auto{pointwise_converse_family_apply}.
        $\groupIsomorphism{e}
            {\directSystemGroup{S}{i}}{\directSystemGroup{T}{i}}$ by
            \auto{group_direct_system_isomorphism_component}.
        Hence $e\in\bijections
            {\carrier[\directSystemGroup{S}{i}]}
            {\carrier[\directSystemGroup{T}{i}]}$ by
            \obv{group_isomorphism_is_bijection}.
        Hence $e\circ c
            =\identity{\carrier[\directSystemGroup{T}{i}]}$ by
            \auto{bijection_circ_converse}.
        $\groupHom{\apply{\kappa}{i}}
            {\directSystemGroup{T}{i}}{M}$ by
            \obv{group_cocone_leg_hom}.
        Hence $\apply{\kappa}{i}\in
            \funs{\carrier[\directSystemGroup{T}{i}]}{\carrier[M]}$ by
            \obv{group_hom_type}.
        Hence $\apply{\kappa}{i}\circ
            \identity{\carrier[\directSystemGroup{T}{i}]}
            =\apply{\kappa}{i}$ by
            \obv{funs_circ_identity_right}.
        $(u\circ v)\circ\apply{\kappa}{i}
            =u\circ(v\circ\apply{\kappa}{i})$ by
            \obv{circ_assoc}.
        Hence $(u\circ v)\circ\apply{\kappa}{i}
            =u\circ(\apply{\iota}{i}\circ c)$.
        $u\circ(\apply{\iota}{i}\circ c)
            =(u\circ\apply{\iota}{i})\circ c$ by
            \obv{circ_assoc}.
        Hence $u\circ(\apply{\iota}{i}\circ c)
            =(\apply{\kappa}{i}\circ e)\circ c$.
        $(\apply{\kappa}{i}\circ e)\circ c
            =\apply{\kappa}{i}\circ(e\circ c)$ by
            \obv{circ_assoc}.
    \end{subproof}
    Hence $\groupCoconeMediator
        {T}{M}{\kappa}{M}{\kappa}{u\circ v}$ by
        \obv{group_cocone_mediator}.
    $\groupHom{\identity{\carrier[M]}}{M}{M}$ by
        \obv{identity_group_hom}.
    Show that for every $i\in\carrier[T]$ we have
        $\identity{\carrier[M]}\circ\apply{\kappa}{i}
            =\apply{\kappa}{i}$.
    \begin{subproof}
        Fix $i\in\carrier[T]$.
        $\groupHom{\apply{\kappa}{i}}
            {\directSystemGroup{T}{i}}{M}$ by
            \obv{group_cocone_leg_hom}.
        Hence $\apply{\kappa}{i}\in
            \funs{\carrier[\directSystemGroup{T}{i}]}{\carrier[M]}$ by
            \obv{group_hom_type}.
        Follows by \obv{funs_identity_circ_left}.
    \end{subproof}
    Hence $\groupCoconeMediator
        {T}{M}{\kappa}{M}{\kappa}{\identity{\carrier[M]}}$ by
        \obv{group_cocone_mediator}.
    Hence $u\circ v=\identity{\carrier[M]}$ by
        \obv{direct_limit_cocone_mediator_unique}.
\end{proof}

\begin{theorem}\label{group_direct_system_limit_isomorphism_exists_unique}
    Suppose $\groupDirectSystemIsomorphism{E}{S}{T}$.
    Suppose $\directLimitCocone{S}{L}{\iota}$ and
        $\directLimitCocone{T}{M}{\kappa}$.
    Then there exists $u$ such that
        $\groupDirectSystemLimitIsomorphism
            {E}{S}{T}{L}{\iota}{M}{\kappa}{u}$ and
        for every $w$ such that
            $\groupDirectSystemLimitIsomorphism
                {E}{S}{T}{L}{\iota}{M}{\kappa}{w}$
        we have $w=u$.
\end{theorem}
\begin{proof}
    Let $C=\pointwiseConverseFamily{\carrier[T]}{E}$.
    Let $P=\pulledGroupCoconeFamily{S}{E}{\kappa}$.
    Let $Q=\pulledGroupCoconeFamily{T}{C}{\iota}$.
    $\groupDirectSystemIsomorphism{C}{T}{S}$ by
        \obv{group_direct_system_isomorphism_converse}.
    $\groupCocone{T}{M}{\kappa}$ and
        $\groupCocone{S}{L}{\iota}$ by
        \obv{direct_limit_cocone_is_cocone}.
    Hence $\groupCocone{S}{M}{P}$ and
        $\groupCocone{T}{L}{Q}$ by
        \obv{group_cocone_pullback_along_system_isomorphism}.
    Take $u$ such that
        $\groupCoconeMediator{S}{L}{\iota}{M}{P}{u}$ and
        for every $w$ such that
            $\groupCoconeMediator{S}{L}{\iota}{M}{P}{w}$
        we have $w=u$ by
        \obv{direct_limit_cocone_universal}.
    Take $v$ such that
        $\groupCoconeMediator{T}{M}{\kappa}{L}{Q}{v}$ and
        for every $w$ such that
            $\groupCoconeMediator{T}{M}{\kappa}{L}{Q}{w}$
        we have $w=v$ by
        \obv{direct_limit_cocone_universal}.
    $\groupHom{u}{L}{M}$ and $\groupHom{v}{M}{L}$ by
        \obv{group_cocone_mediator_hom}.
    $v\circ u=\identity{\carrier[L]}$ and
        $u\circ v=\identity{\carrier[M]}$ by
        \obv{system_isomorphism_limit_mediators_are_inverse}.
    Hence $\groupIsomorphism{u}{L}{M}$ by
        \obv{group_hom_two_sided_inverse_isomorphism}.
    Show that for every $i\in\carrier[S]$ we have
        $u\circ\apply{\iota}{i}
            =\apply{\kappa}{i}\circ\apply{E}{i}$.
    \begin{subproof}
        Fix $i\in\carrier[S]$.
        $u\circ\apply{\iota}{i}=\apply{P}{i}$ by
            \obv{group_cocone_mediator_compatibility}.
        $\apply{P}{i}=\apply{\kappa}{i}\circ\apply{E}{i}$ by
            \auto{pulled_group_cocone_family_apply}.
    \end{subproof}
    Hence $\groupDirectSystemLimitIsomorphism
        {E}{S}{T}{L}{\iota}{M}{\kappa}{u}$ by
        \obv{group_direct_system_limit_isomorphism}.
    Show that for every $w$ such that
        $\groupDirectSystemLimitIsomorphism
            {E}{S}{T}{L}{\iota}{M}{\kappa}{w}$
        we have $w=u$.
    \begin{subproof}
        Fix $w$.
        Assume $\groupDirectSystemLimitIsomorphism
            {E}{S}{T}{L}{\iota}{M}{\kappa}{w}$.
        Hence $\groupHom{w}{L}{M}$ by
            \obv{group_direct_system_limit_isomorphism_is_iso,group_isomorphism_is_hom}.
        Show that for every $i\in\carrier[S]$ we have
            $w\circ\apply{\iota}{i}=\apply{P}{i}$.
        \begin{subproof}
            Fix $i\in\carrier[S]$.
            $w\circ\apply{\iota}{i}
                =\apply{\kappa}{i}\circ\apply{E}{i}$ by
                \obv{group_direct_system_limit_isomorphism_compatibility}.
            $\apply{P}{i}=\apply{\kappa}{i}\circ\apply{E}{i}$ by
                \auto{pulled_group_cocone_family_apply}.
        \end{subproof}
        Hence $\groupCoconeMediator{S}{L}{\iota}{M}{P}{w}$ by
            \obv{group_cocone_mediator}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\subsection{The chosen comparison of concrete limits}

\begin{definition}\label{direct_system_limit_transport}
    $\directSystemLimitTransport{E}{S}{T}=
        \directLimitMediator
            {S}{\groupDirectLimit{T}}
            {\pulledGroupCoconeFamily
                {S}{E}{\directLimitCanonicalFamily{T}}}$.
\end{definition}

\begin{theorem}\label{direct_system_limit_transport_is_limit_isomorphism}
    Suppose $\groupDirectSystemIsomorphism{E}{S}{T}$.
    Then $\groupDirectSystemLimitIsomorphism
        {E}{S}{T}
        {\groupDirectLimit{S}}{\directLimitCanonicalFamily{S}}
        {\groupDirectLimit{T}}{\directLimitCanonicalFamily{T}}
        {\directSystemLimitTransport{E}{S}{T}}$.
\end{theorem}
\begin{proof}
    Let $L=\groupDirectLimit{S}$.
    Let $M=\groupDirectLimit{T}$.
    Let $\iota=\directLimitCanonicalFamily{S}$.
    Let $\kappa=\directLimitCanonicalFamily{T}$.
    Let $P=\pulledGroupCoconeFamily{S}{E}{\kappa}$.
    Let $u=\directSystemLimitTransport{E}{S}{T}$.
    $S$ is a group direct system and $T$ is a group direct system by
        \obv{group_direct_system_isomorphism_source,group_direct_system_isomorphism_target}.
    $\directLimitCocone{S}{L}{\iota}$ and
        $\directLimitCocone{T}{M}{\kappa}$ by
        \auto{group_direct_limit_is_direct_limit_cocone}.
    $\groupCocone{T}{M}{\kappa}$ by
        \obv{direct_limit_cocone_is_cocone}.
    Hence $\groupCocone{S}{M}{P}$ by
        \obv{group_cocone_pullback_along_system_isomorphism}.
    $u=\directLimitMediator{S}{M}{P}$ by
        \auto{direct_system_limit_transport}.
    Hence $\groupCoconeMediator{S}{L}{\iota}{M}{P}{u}$ by
        \auto{direct_limit_mediator_is_mediator}.
    Take $v$ such that
        $\groupDirectSystemLimitIsomorphism
            {E}{S}{T}{L}{\iota}{M}{\kappa}{v}$ and
        for every $w$ such that
            $\groupDirectSystemLimitIsomorphism
                {E}{S}{T}{L}{\iota}{M}{\kappa}{w}$
        we have $w=v$ by
        \obv{group_direct_system_limit_isomorphism_exists_unique}.
    $\groupHom{v}{L}{M}$ by
        \obv{group_direct_system_limit_isomorphism_is_iso,group_isomorphism_is_hom}.
    Show that for every $i\in\carrier[S]$ we have
        $v\circ\apply{\iota}{i}=\apply{P}{i}$.
    \begin{subproof}
        Fix $i\in\carrier[S]$.
        $v\circ\apply{\iota}{i}
            =\apply{\kappa}{i}\circ\apply{E}{i}$ by
            \obv{group_direct_system_limit_isomorphism_compatibility}.
        $\apply{P}{i}=\apply{\kappa}{i}\circ\apply{E}{i}$ by
            \auto{pulled_group_cocone_family_apply}.
    \end{subproof}
    Hence $\groupCoconeMediator{S}{L}{\iota}{M}{P}{v}$ by
        \obv{group_cocone_mediator}.
    Hence $v=\directLimitMediator{S}{M}{P}$ by
        \auto{direct_limit_mediator_unique}.
    Hence $v=u$.
    Follows by \auto.
\end{proof}

\begin{corollary}\label{direct_system_limit_transport_isomorphism}
    Suppose $\groupDirectSystemIsomorphism{E}{S}{T}$.
    Then $\groupIsomorphism
        {\directSystemLimitTransport{E}{S}{T}}
        {\groupDirectLimit{S}}{\groupDirectLimit{T}}$.
\end{corollary}
\begin{proof}
    Follows by
        \auto{direct_system_limit_transport_is_limit_isomorphism,group_direct_system_limit_isomorphism_is_iso}.
\end{proof}

\begin{corollary}\label{direct_system_limit_transport_compatibility}
    Suppose $\groupDirectSystemIsomorphism{E}{S}{T}$.
    Let $i\in\carrier[S]$.
    Then $\directSystemLimitTransport{E}{S}{T}\circ
        \directSystemCanonicalMap{S}{i}
        =\directSystemCanonicalMap{T}{i}\circ\apply{E}{i}$.
\end{corollary}
\begin{proof}
    $\apply{\directLimitCanonicalFamily{S}}{i}
        =\directSystemCanonicalMap{S}{i}$ by
        \obv{direct_limit_canonical_family_apply}.
    $\carrier[S]=\carrier[T]$ by
        \obv{group_direct_system_isomorphism_carrier}.
    Hence $i\in\carrier[T]$.
    $\apply{\directLimitCanonicalFamily{T}}{i}
        =\directSystemCanonicalMap{T}{i}$ by
        \obv{direct_limit_canonical_family_apply}.
    Follows by
        \auto{direct_system_limit_transport_is_limit_isomorphism,group_direct_system_limit_isomorphism_compatibility}.
\end{proof}

\begin{theorem}\label{direct_system_limit_transport_unique}
    Suppose $\groupDirectSystemIsomorphism{E}{S}{T}$.
    Suppose $\groupDirectSystemLimitIsomorphism
        {E}{S}{T}
        {\groupDirectLimit{S}}{\directLimitCanonicalFamily{S}}
        {\groupDirectLimit{T}}{\directLimitCanonicalFamily{T}}{u}$.
    Then $u=\directSystemLimitTransport{E}{S}{T}$.
\end{theorem}
\begin{proof}
    $\groupDirectSystemLimitIsomorphism
        {E}{S}{T}
        {\groupDirectLimit{S}}{\directLimitCanonicalFamily{S}}
        {\groupDirectLimit{T}}{\directLimitCanonicalFamily{T}}
        {\directSystemLimitTransport{E}{S}{T}}$ by
        \obv{direct_system_limit_transport_is_limit_isomorphism}.
    Take $v$ such that
        $\groupDirectSystemLimitIsomorphism
            {E}{S}{T}
            {\groupDirectLimit{S}}{\directLimitCanonicalFamily{S}}
            {\groupDirectLimit{T}}{\directLimitCanonicalFamily{T}}{v}$ and
        for every $w$ such that
            $\groupDirectSystemLimitIsomorphism
                {E}{S}{T}
                {\groupDirectLimit{S}}{\directLimitCanonicalFamily{S}}
                {\groupDirectLimit{T}}{\directLimitCanonicalFamily{T}}{w}$
        we have $w=v$ by
        \obv{group_direct_system_limit_isomorphism_exists_unique,group_direct_limit_is_direct_limit_cocone,group_direct_system_isomorphism_source,group_direct_system_isomorphism_target}.
    $u=v$ and $\directSystemLimitTransport{E}{S}{T}=v$.
\end{proof}
