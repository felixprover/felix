\section{Complete groups}

A complete group is centreless and every automorphism is inner.  Both clauses
are retained explicitly: surjectivity of the inner map alone would not rule
out a nontrivial centre.

\begin{definition}\label{complete_group}
    $\completeGroup{G}$ iff
        $\centrelessGroup{G}$ and
        $\innerAutomorphismCarrier{G}=\automorphismSet{G}$.
\end{definition}

\begin{proposition}\label{complete_group_is_centreless}
    Suppose $\completeGroup{G}$.
    Then $\centrelessGroup{G}$.
\end{proposition}
\begin{proof}
    Follows by \obv{complete_group}.
\end{proof}

\begin{corollary}\label{complete_group_is_group}
    Suppose $\completeGroup{G}$.
    Then $G$ is a group.
\end{corollary}
\begin{proof}
    Follows by
        \obv{complete_group_is_centreless,centreless_group_is_group}.
\end{proof}

\begin{proposition}\label{complete_group_all_automorphisms_inner}
    Suppose $\completeGroup{G}$.
    Then $\innerAutomorphismCarrier{G}=\automorphismSet{G}$.
\end{proposition}
\begin{proof}
    Follows by \obv{complete_group}.
\end{proof}

\begin{theorem}\label{complete_group_iff_inner_map_isomorphism}
    $\completeGroup{G}$ iff
        $\groupIsomorphism
            {\innerMap{G}}
            {G}
            {\automorphismGroup{G}}$.
\end{theorem}
\begin{proof}
    Show that if $\completeGroup{G}$, then
        $\groupIsomorphism
            {\innerMap{G}}
            {G}
            {\automorphismGroup{G}}$.
    \begin{subproof}
        Assume $\completeGroup{G}$.
        Hence $G$ is a group by \obv{complete_group_is_group}.
        Let $A=\automorphismGroup{G}$.
        $\groupHom{\innerMap{G}}{G}{A}$ by
            \obv{inner_map_group_hom}.
        $\groupCentre{G}=\trivialSubgroupCarrier{G}$ by
            \obv{complete_group_is_centreless,centreless_group_centre}.
        Hence $\innerMap{G}$ is injective by
            \obv{inner_map_injective_iff_centre_trivial}.
        $\innerMap{G}\in\funs{\carrier[G]}{\automorphismSet{G}}$ by
            \obv{inner_map_type}.
        Hence $\dom{\innerMap{G}}=\carrier[G]$ by
            \obv{funs_elim}.
        $\img{\innerMap{G}}{\carrier[G]}=
            \ran{\innerMap{G}}$ by \auto{img_dom}.
        Hence $\ran{\innerMap{G}}=\automorphismSet{G}$ by
            \auto{inner_automorphism_carrier,complete_group_all_automorphisms_inner}.
        Hence $\innerMap{G}\in
            \surj{\carrier[G]}{\automorphismSet{G}}$ by
            \obv{funs_surj_iff}.
        Hence $\innerMap{G}\in
            \bijections{\carrier[G]}{\automorphismSet{G}}$ by
            \obv{bijections}.
        $\carrier[A]=\automorphismSet{G}$ by
            \auto{automorphism_group_carrier}.
        Follows by \obv{group_isomorphism}.
    \end{subproof}
    Show that if
        $\groupIsomorphism
            {\innerMap{G}}
            {G}
            {\automorphismGroup{G}}$, then
        $\completeGroup{G}$.
    \begin{subproof}
        Assume
            $\groupIsomorphism
                {\innerMap{G}}
                {G}
                {\automorphismGroup{G}}$.
        Hence $\groupHom{\innerMap{G}}{G}{\automorphismGroup{G}}$ by
            \obv{group_isomorphism_is_hom}.
        $\innerMap{G}$ is injective by
            \auto{group_isomorphism_is_bijection,bijections}.
        Hence
            $\groupKernel{\innerMap{G}}{G}
                =\trivialSubgroupCarrier{G}$ by
            \obv{group_hom_injective_iff_kernel_trivial}.
        $\groupKernel{\innerMap{G}}{G}
            =\groupCentre{G}$ by
            \obv{inner_map_kernel_is_centre}.
        Hence $\groupCentre{G}=\trivialSubgroupCarrier{G}$.
        Hence $\centrelessGroup{G}$ by
            \obv{centreless_group}.
        $\innerMap{G}\in
            \bijections{\carrier[G]}{
                \carrier[\automorphismGroup{G}]}$ by
            \obv{group_isomorphism_is_bijection}.
        Hence $\innerMap{G}\in
            \surj{\carrier[G]}{\carrier[\automorphismGroup{G}]}$ by
            \obv{bijections}.
        Hence
            $\ran{\innerMap{G}}=\carrier[\automorphismGroup{G}]$ by
            \obv{ran_of_surj}.
        $\dom{\innerMap{G}}=\carrier[G]$ by
            \obv{group_hom_type,funs_elim}.
        Hence
            $\img{\innerMap{G}}{\carrier[G]}=
                \ran{\innerMap{G}}$ by
            \auto{img_dom}.
        $\carrier[\automorphismGroup{G}]=\automorphismSet{G}$ by
            \obv{automorphism_group_carrier}.
        Hence
            $\innerAutomorphismCarrier{G}=\automorphismSet{G}$ by
            \obv{inner_automorphism_carrier}.
        Follows by \obv{complete_group}.
    \end{subproof}
    Follows by \auto.
\end{proof}
