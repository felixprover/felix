\section{Canonical transport between coherent tower segments}

Fixing a base isomorphism determines the stage comparison recursively.  At a
successor the comparison is automorphism transport.  At a limit it is the
unique comparison supplied by the semantic direct-limit property.  Logical
selection names that unique set-coded graph; all mathematical laws below are
guarded by coherent segment hypotheses.

\begin{definition}\label{tower_limit_transport}
    $\towerLimitTransport{T}{U}{\lambda}{p}=
        \Select[\Set]{\fun{u:\Set}{
            \groupDirectSystemLimitIsomorphism
                {p}
                {\towerEarlierSystem{T}{\lambda}}
                {\towerEarlierSystem{U}{\lambda}}
                {\towerStage{T}{\lambda}}
                {\towerIncomingFamily{T}{\lambda}}
                {\towerStage{U}{\lambda}}
                {\towerIncomingFamily{U}{\lambda}}
                {u}}}$.
\end{definition}

\begin{theorem}\label{tower_limit_transport_specification}
    Suppose $\coherentTowerSegment{T}{G}{\theta}$ and
        $\coherentTowerSegment{U}{H}{\theta}$.
    Let $\lambda\in\theta$.
    Suppose $\lambda$ is a limit ordinal.
    Suppose $\groupDirectSystemIsomorphism{p}
        {\towerEarlierSystem{T}{\lambda}}
        {\towerEarlierSystem{U}{\lambda}}$.
    Then $\groupDirectSystemLimitIsomorphism
        {p}
        {\towerEarlierSystem{T}{\lambda}}
        {\towerEarlierSystem{U}{\lambda}}
        {\towerStage{T}{\lambda}}
        {\towerIncomingFamily{T}{\lambda}}
        {\towerStage{U}{\lambda}}
        {\towerIncomingFamily{U}{\lambda}}
        {\towerLimitTransport{T}{U}{\lambda}{p}}$.
\end{theorem}
\begin{proof}
    $\directLimitCocone
        {\towerEarlierSystem{T}{\lambda}}
        {\towerStage{T}{\lambda}}
        {\towerIncomingFamily{T}{\lambda}}$ and
    $\directLimitCocone
        {\towerEarlierSystem{U}{\lambda}}
        {\towerStage{U}{\lambda}}
        {\towerIncomingFamily{U}{\lambda}}$ by
        \obv{coherent_tower_segment_limit}.
    Hence there exists $u$ such that
        $\groupDirectSystemLimitIsomorphism
            {p}
            {\towerEarlierSystem{T}{\lambda}}
            {\towerEarlierSystem{U}{\lambda}}
            {\towerStage{T}{\lambda}}
            {\towerIncomingFamily{T}{\lambda}}
            {\towerStage{U}{\lambda}}
            {\towerIncomingFamily{U}{\lambda}}
            {u}$ by
        \auto{group_direct_system_limit_isomorphism_exists_unique}.
    Follows by \auto{tower_limit_transport}.
\end{proof}

\begin{definition}\label{tower_transport_successor_step}
    $\towerTransportSuccessorStep{T}{U}{\beta}{p}=
        \automorphismTransport{\apply{p}{\beta}}
            {\towerStage{T}{\beta}}{\towerStage{U}{\beta}}$.
\end{definition}

\begin{definition}\label{tower_transport_nonzero_step}
    $\towerTransportNonzeroStep{T}{U}{\alpha}{p}=
        \Apply{\logicalIfSetOp}
            {\towerSuccessorCase{\alpha},\towerTransportSuccessorStep{T}{U}{\unions{\alpha}}{p},\towerLimitTransport{T}{U}{\alpha}{p}}$.
\end{definition}

\begin{definition}\label{tower_transport_step_op}
    $\towerTransportStepOp:
        \Set\to\Set\to\Set\to\Set\to\Set\to\Set=
        \fun{e,T,U,\alpha,p:\Set}{
                \Apply{\logicalIfSetOp}
                    {(\alpha=\emptyset),e,\towerTransportNonzeroStep{T}{U}{\alpha}{p}}}$.
\end{definition}

\begin{abbreviation}\label{tower_transport_step}
    $\towerTransportStep{e}{T}{U}{\alpha}{p}=
        \Apply{\towerTransportStepOp}{e,T,U,\alpha,p}$.
\end{abbreviation}

\begin{proposition}\label{tower_transport_step_zero}
    $\towerTransportStep{e}{T}{U}{\emptyset}{p}=e$.
\end{proposition}
\begin{proof}
    Follows by
        \obv{tower_transport_step_op,logical_if_set_true}.
\end{proof}

\begin{proposition}\label{tower_transport_step_nonzero}
    Suppose $\alpha\neq\emptyset$.
    Then $\towerTransportStep{e}{T}{U}{\alpha}{p}=
        \towerTransportNonzeroStep{T}{U}{\alpha}{p}$.
\end{proposition}
\begin{proof}
    Follows by
        \obv{tower_transport_step_op,logical_if_set_false}.
\end{proof}

\begin{proposition}\label{tower_transport_nonzero_step_successor}
    Suppose $\towerSuccessorCase{\alpha}$.
    Then $\towerTransportNonzeroStep{T}{U}{\alpha}{p}=
        \towerTransportSuccessorStep{T}{U}{\unions{\alpha}}{p}$.
\end{proposition}
\begin{proof}
    Follows by
        \obv{tower_transport_nonzero_step,logical_if_set_true}.
\end{proof}

\begin{proposition}\label{tower_transport_nonzero_step_limit}
    Suppose $\lnot\towerSuccessorCase{\alpha}$.
    Then $\towerTransportNonzeroStep{T}{U}{\alpha}{p}=
        \towerLimitTransport{T}{U}{\alpha}{p}$.
\end{proposition}
\begin{proof}
    Follows by
        \obv{tower_transport_nonzero_step,logical_if_set_false}.
\end{proof}

\begin{proposition}\label{tower_transport_step_successor_ordinal}
    Let $\alpha$ be a successor ordinal.
    Then $\towerTransportStep{e}{T}{U}{\alpha}{p}=
        \towerTransportSuccessorStep{T}{U}{\unions{\alpha}}{p}$.
\end{proposition}
\begin{proof}
    $\alpha\neq\emptyset$ by \obv{zero_not_successorordinal}.
    $\towerSuccessorCase{\alpha}$ by
        \obv{tower_successor_case}.
    $\towerTransportStep{e}{T}{U}{\alpha}{p}=
        \towerTransportNonzeroStep{T}{U}{\alpha}{p}$ by
        \obv{tower_transport_step_nonzero}.
    $\towerTransportNonzeroStep{T}{U}{\alpha}{p}=
        \towerTransportSuccessorStep{T}{U}{\unions{\alpha}}{p}$ by
        \obv{tower_transport_nonzero_step_successor}.
\end{proof}

\begin{corollary}\label{tower_transport_step_successor}
    Let $\beta$ be an ordinal.
    Then $\towerTransportStep{e}{T}{U}{\suc{\beta}}{p}=
        \towerTransportSuccessorStep{T}{U}{\beta}{p}$.
\end{corollary}
\begin{proof}
    Follows by
        \auto{ordinal_successor_data,
            tower_transport_step_successor_ordinal}.
\end{proof}

\begin{proposition}\label{tower_transport_step_limit}
    Let $\lambda$ be a limit ordinal.
    Then $\towerTransportStep{e}{T}{U}{\lambda}{p}=
        \towerLimitTransport{T}{U}{\lambda}{p}$.
\end{proposition}
\begin{proof}
    $\lambda\neq\emptyset$ by \obv{zero_not_limitordinal}.
    $\lambda$ is not a successor ordinal by
        \obv{limit_ordinal_is_not_successor_ordinal}.
    Hence $\lnot\towerSuccessorCase{\lambda}$ by
        \obv{tower_successor_case}.
    Follows by
        \auto{tower_transport_step_nonzero,
            tower_transport_nonzero_step_limit}.
\end{proof}

\begin{definition}\label{tower_transport_family}
    $\towerTransportFamily{e}{T}{U}=
        \Apply{\unboundedOrdinalRecursionOp}
            {\carrier[T],\Apply{\towerTransportStepOp}{e,T,U}}$.
\end{definition}

\begin{proposition}\label{tower_transport_family_function}
    Suppose $\carrier[T]$ is an ordinal.
    Then $\towerTransportFamily{e}{T}{U}$ is a function.
\end{proposition}
\begin{proof}
    Let $f=\Apply{\unboundedOrdinalRecursionOp}
        {\carrier[T],\Apply{\towerTransportStepOp}{e,T,U}}$.
    $\Apply{\recursiveSetFunctionOp}{f}$ by
        \obv{unbounded_ordinal_recursion_function}.
    $f=\towerTransportFamily{e}{T}{U}$ by
        \obv{tower_transport_family}.
    Hence $\Apply{\recursiveSetFunctionOp}{
        \towerTransportFamily{e}{T}{U}}$.
    Hence $\recursiveSetFunction{\towerTransportFamily{e}{T}{U}}$ by
        \obv{recursive_set_function_op}.
    Follows by \obv{recursive_set_function}.
\end{proof}

\begin{proposition}\label{tower_transport_family_domain}
    Suppose $\carrier[T]$ is an ordinal.
    Then $\dom{\towerTransportFamily{e}{T}{U}}=\carrier[T]$.
\end{proposition}
\begin{proof}
    Let $f=\Apply{\unboundedOrdinalRecursionOp}
        {\carrier[T],\Apply{\towerTransportStepOp}{e,T,U}}$.
    $\dom{f}=\carrier[T]$ by
        \auto{unbounded_ordinal_recursion_domain}.
    $f=\towerTransportFamily{e}{T}{U}$ by
        \obv{tower_transport_family}.
\end{proof}

\begin{proposition}\label{tower_transport_family_type}
    Suppose $\carrier[T]$ is an ordinal.
    Then $\towerTransportFamily{e}{T}{U}\in
        \funs{\carrier[T]}{\ran{\towerTransportFamily{e}{T}{U}}}$.
\end{proposition}
\begin{proof}
    Let $F=\towerTransportFamily{e}{T}{U}$.
    $F$ is a function and $\dom{F}=\carrier[T]$ by
        \auto{tower_transport_family_function,
            tower_transport_family_domain}.
    $F\in\funs{\dom{F}}{\ran{F}}$ by
        \obv{function_exact_type_to_range}.
\end{proof}

\begin{proposition}\label{tower_transport_family_equation}
    Suppose $\carrier[T]$ is an ordinal.
    Let $\alpha\in\carrier[T]$.
    Then $\apply{\towerTransportFamily{e}{T}{U}}{\alpha}=
        \towerTransportStep{e}{T}{U}{\alpha}{
            \restrl{\towerTransportFamily{e}{T}{U}}{\alpha}}$.
\end{proposition}
\begin{proof}
    Let $f=\Apply{\unboundedOrdinalRecursionOp}
        {\carrier[T],\Apply{\towerTransportStepOp}{e,T,U}}$.
    $\apply{f}{\alpha}=
        \Apply{\towerTransportStepOp}{e,T,U,\alpha,\restrl{f}{\alpha}}$ by
        \auto{unbounded_ordinal_recursion_equation}.
    $f=\towerTransportFamily{e}{T}{U}$ by
        \obv{tower_transport_family}.
    $\Apply{\towerTransportStepOp}{e,T,U,\alpha,\restrl{f}{\alpha}}=
        \towerTransportStep{e}{T}{U}{\alpha}{\restrl{f}{\alpha}}$ by
        \obv{tower_transport_step_op}.
\end{proof}

\subsection{The bounded transport invariant}

\begin{definition}\label{tower_transport_at}
    $\towerTransportAt{E}{T}{U}{\alpha}$ iff
        $\groupIsomorphism{\apply{E}{\alpha}}
            {\towerStage{T}{\alpha}}{\towerStage{U}{\alpha}}$ and
        for every $\beta\in\alpha$ we have
            $\apply{E}{\alpha}\circ\towerBond{T}{\beta}{\alpha}=
                \towerBond{U}{\beta}{\alpha}\circ\apply{E}{\beta}$.
\end{definition}

\begin{definition}\label{tower_transport_below}
    $\towerTransportBelow{E}{T}{U}{\alpha}$ iff
        for every $\gamma\in\alpha$ we have
            $\towerTransportAt{E}{T}{U}{\gamma}$.
\end{definition}

\begin{proposition}\label{tower_transport_at_component}
    Suppose $\towerTransportAt{E}{T}{U}{\alpha}$.
    Then $\groupIsomorphism{\apply{E}{\alpha}}
        {\towerStage{T}{\alpha}}{\towerStage{U}{\alpha}}$.
\end{proposition}
\begin{proof}
    Follows by \obv{tower_transport_at}.
\end{proof}

\begin{proposition}\label{tower_transport_at_strict_square}
    Suppose $\towerTransportAt{E}{T}{U}{\alpha}$.
    Let $\beta\in\alpha$.
    Then $\apply{E}{\alpha}\circ\towerBond{T}{\beta}{\alpha}=
        \towerBond{U}{\beta}{\alpha}\circ\apply{E}{\beta}$.
\end{proposition}
\begin{proof}
    Follows by \obv{tower_transport_at}.
\end{proof}

\begin{proposition}\label{tower_transport_below_at}
    Suppose $\towerTransportBelow{E}{T}{U}{\alpha}$.
    Let $\gamma\in\alpha$.
    Then $\towerTransportAt{E}{T}{U}{\gamma}$.
\end{proposition}
\begin{proof}
    Follows by \obv{tower_transport_below}.
\end{proof}

\begin{theorem}\label{tower_transport_bounded_induction}
    Let $\theta$ be an ordinal.
    Suppose for every $\alpha\in\theta$ such that
        $\towerTransportBelow{E}{T}{U}{\alpha}$ we have
        $\towerTransportAt{E}{T}{U}{\alpha}$.
    Then $\towerTransportBelow{E}{T}{U}{\theta}$.
\end{theorem}
\begin{proof}
    Let $P:\Set\to\Prop=\fun{\alpha:\Set}{
        \towerTransportAt{E}{T}{U}{\alpha}}$.
    Show that $\forall \alpha:\Set.\ \alpha\in\theta\implies
        ((\forall \gamma:\Set.\ \gamma\in\alpha\implies
            \Apply{P}{\gamma})
        \implies\Apply{P}{\alpha})$.
    \begin{subproof}
        Fix $\alpha\in\theta$.
        Assume $\forall \gamma:\Set.\ \gamma\in\alpha\implies
            \Apply{P}{\gamma}$.
        Show that $\towerTransportBelow{E}{T}{U}{\alpha}$.
        \begin{subproof}
            Show that for every $\gamma\in\alpha$ we have
                $\towerTransportAt{E}{T}{U}{\gamma}$.
            \begin{subproof}
                Fix $\gamma\in\alpha$.
                $\Apply{P}{\gamma}$.
            \end{subproof}
            Follows by \obv{tower_transport_below}.
        \end{subproof}
        Hence $\towerTransportAt{E}{T}{U}{\alpha}$.
    \end{subproof}
    Hence $\forall \alpha:\Set.\ \alpha\in\theta\implies
        \Apply{P}{\alpha}$ by
        \obv{bounded_transfinite_induction}.
    Show that for every $\alpha\in\theta$ we have
        $\towerTransportAt{E}{T}{U}{\alpha}$.
    \begin{subproof}
        Fix $\alpha\in\theta$.
        $\Apply{P}{\alpha}$.
    \end{subproof}
    Follows by \obv{tower_transport_below}.
\end{proof}

\begin{proposition}\label{tower_transport_self_square}
    Suppose $\towerSegmentDiagram{T}{G}{\theta}$ and
        $\towerSegmentDiagram{U}{H}{\theta}$.
    Let $\alpha\in\theta$.
    Suppose $\groupIsomorphism{e}
        {\towerStage{T}{\alpha}}{\towerStage{U}{\alpha}}$.
    Then $e\circ\towerBond{T}{\alpha}{\alpha}=
        \towerBond{U}{\alpha}{\alpha}\circ e$.
\end{proposition}
\begin{proof}
    $\towerBond{T}{\alpha}{\alpha}=
        \identity{\carrier[\towerStage{T}{\alpha}]}$ and
    $\towerBond{U}{\alpha}{\alpha}=
        \identity{\carrier[\towerStage{U}{\alpha}]}$ by
        \obv{tower_segment_bond_identity}.
    $e\in\funs{\carrier[\towerStage{T}{\alpha}]}
        {\carrier[\towerStage{U}{\alpha}]}$ by
        \obv{group_isomorphism_is_hom,group_hom_type}.
    $e\circ\identity{\carrier[\towerStage{T}{\alpha}]}=e$ and
        $\identity{\carrier[\towerStage{U}{\alpha}]}\circ e=e$ by
        \obv{funs_circ_identity_right,funs_identity_circ_left}.
\end{proof}

\begin{proposition}\label{tower_transport_at_nonstrict_square}
    Suppose $\towerSegmentDiagram{T}{G}{\theta}$ and
        $\towerSegmentDiagram{U}{H}{\theta}$.
    Let $\alpha\in\theta$.
    Suppose $\towerTransportAt{E}{T}{U}{\alpha}$.
    Let $\beta\in\suc{\alpha}$.
    Then $\apply{E}{\alpha}\circ\towerBond{T}{\beta}{\alpha}=
        \towerBond{U}{\beta}{\alpha}\circ\apply{E}{\beta}$.
\end{proposition}
\begin{proof}
    If $\beta\in\alpha$, then
        $\apply{E}{\alpha}\circ\towerBond{T}{\beta}{\alpha}=
            \towerBond{U}{\beta}{\alpha}\circ\apply{E}{\beta}$ by
        \obv{tower_transport_at_strict_square}.
    If $\beta=\alpha$, then
        $\groupIsomorphism{\apply{E}{\alpha}}
            {\towerStage{T}{\alpha}}{\towerStage{U}{\alpha}}$ by
        \obv{tower_transport_at_component}.
    If $\beta=\alpha$, then
        $\apply{E}{\alpha}\circ\towerBond{T}{\beta}{\alpha}=
            \towerBond{U}{\beta}{\alpha}\circ\apply{E}{\beta}$ by
        \auto{tower_transport_self_square}.
    Follows by \auto{suc_iff}.
\end{proof}

\begin{proposition}\label{tower_transport_below_nonstrict_square}
    Suppose $\towerSegmentDiagram{T}{G}{\theta}$ and
        $\towerSegmentDiagram{U}{H}{\theta}$.
    Let $\lambda\in\theta$.
    Suppose $\towerTransportBelow{E}{T}{U}{\lambda}$.
    Let $\alpha,\beta\in\lambda$.
    Suppose $\alpha\subseteq\beta$.
    Then $\apply{E}{\beta}\circ\towerBond{T}{\alpha}{\beta}=
        \towerBond{U}{\alpha}{\beta}\circ\apply{E}{\alpha}$.
\end{proposition}
\begin{proof}
    $\towerTransportAt{E}{T}{U}{\beta}$ by
        \obv{tower_transport_below_at}.
    Hence $\alpha$ and $\beta$ are ordinals by
        \obv{elem_of_ordinal_is_ordinal}.
    Hence $\alpha\in\beta$ or $\alpha=\beta$ by
        \obv{ordinal_subseteq_implies_elem_or_eq}.
    Hence $\alpha\in\suc{\beta}$ by \auto{suc_iff}.
    $\beta\in\theta$ by \obv{ordinal_transitivity}.
    Follows by \obv{tower_transport_at_nonstrict_square}.
\end{proof}

\begin{proposition}\label{tower_transport_restriction_family_type}

\end{proof}

\begin{proposition}\label{tower_transport_earlier_system_context}

        \obv{tower_earlier_system_is_system}.

        \carrier[\towerEarlierSystem{U}{\lambda}]$ and
        $\leq[\towerEarlierSystem{T}{\lambda}]=\subseteqrel{\lambda}=
        \leq[\towerEarlierSystem{U}{\lambda}]$ by
        \obv{tower_earlier_system_carrier,

\begin{proposition}\label{tower_transport_restriction_component}

        $\towerSegmentDiagram{U}{H}{\theta}$ by

        \obv{elem_of_ordinal_is_ordinal}.
    Hence $\lambda\subseteq\theta$ by
        \obv{ordinal_elem_implies_subseteq}.
    $\apply{p}{\alpha}=\apply{E}{\alpha}$ by
        \auto{restrl_of_function_apply}.
    $\towerTransportAt{E}{T}{U}{\alpha}$ by
        \obv{tower_transport_below_at}.

        {\towerStage{T}{\alpha}}{\towerStage{U}{\alpha}}$ by

\begin{proposition}\label{tower_transport_restriction_square}

        \obv{tower_earlier_system_indices_ordered}.
    $\apply{E}{\beta}\circ\towerBond{T}{\alpha}{\beta}=

    Let $\lambda\in\theta$.

        {\towerStage{T}{\lambda}}{\towerStage{U}{\lambda}}$ by
        \obv{tower_transport_family_limit_component}.
    Show that for every $\beta\in\lambda$ we have
        $\apply{F}{\lambda}\circ\towerBond{T}{\beta}{\lambda}=
        \towerBond{U}{\beta}{\lambda}\circ\apply{F}{\beta}$.
    \begin{subproof}
        Fix $\beta\in\lambda$.

    \end{subproof}

\begin{proposition}\label{coherent_tower_length_is_ordinal}
    Suppose $\coherentTowerSegment{T}{G}{\theta}$.

\end{proposition}

\begin{proposition}\label{coherent_tower_index_is_ordinal}

\end{proposition}
\begin{proof}

        \obv{tower_segment_diagram_data}.

\begin{proposition}\label{coherent_tower_successor_predecessor_context}
    Suppose $\coherentTowerSegment{T}{G}{\theta}$.
    Let $\alpha\in\theta$.
    Suppose $\alpha$ is a successor ordinal.
    Then there exists an ordinal $\delta$ such that
        $\alpha=\suc{\delta}$ and $\delta\in\theta$.
\end{proposition}
\begin{proof}
    Take an ordinal $\delta$ such that $\alpha=\suc{\delta}$ by
        \obv{successor_ordinal}.
    $\delta\in\alpha$ by \obv{suc_intro_self}.
    $\theta$ is an ordinal by
        \obv{coherent_tower_length_is_ordinal}.
    Hence $\delta\in\theta$ by \obv{ordinal_transitivity}.
\end{proof}

\begin{proposition}\label{tower_transport_successor_from_below}
    Suppose $\coherentTowerSegment{T}{G}{\theta}$ and
        $\coherentTowerSegment{U}{H}{\theta}$.
    Let $\alpha\in\theta$.
    Suppose $\alpha$ is a successor ordinal and
        $\towerTransportBelow{F}{T}{U}{\alpha}$.
    Suppose $F=\towerTransportFamily{e}{T}{U}$.
    Then $\towerTransportAt{F}{T}{U}{\alpha}$.
\end{proposition}
\begin{proof}
    Take an ordinal $\delta$ such that
        $\alpha=\suc{\delta}$ and $\delta\in\theta$ by
        \obv{coherent_tower_successor_predecessor_context}.
    $\delta\in\alpha$ by \obv{suc_intro_self}.
    Hence $\towerTransportAt{F}{T}{U}{\delta}$ by
        \obv{tower_transport_below_at}.
    $\towerSegmentDiagram{T}{G}{\theta}$ and
        $\towerSegmentDiagram{U}{H}{\theta}$ by
        \obv{coherent_tower_segment_diagram}.
    $\apply{F}{\suc{\delta}}=
        \automorphismTransport{\apply{F}{\delta}}
            {\towerStage{T}{\delta}}{\towerStage{U}{\delta}}$ by
        \obv{tower_transport_family_successor_equation}.
    Hence $\towerTransportAt{F}{T}{U}{\suc{\delta}}$ by
        \obv{tower_transport_at_successor}.
\end{proof}

\begin{proposition}\label{tower_transport_at_from_ordinal_cases}
    Let $\alpha$ be an ordinal.
    Suppose $\alpha=\emptyset\implies
        \towerTransportAt{F}{T}{U}{\alpha}$.
    Suppose if $\alpha$ is a limit ordinal then
        $\towerTransportAt{F}{T}{U}{\alpha}$.
    Suppose if $\alpha$ is a successor ordinal then
        $\towerTransportAt{F}{T}{U}{\alpha}$.
    Then $\towerTransportAt{F}{T}{U}{\alpha}$.
\end{proposition}
\begin{proof}
    $\alpha=\emptyset$ or $\emptyset\in\alpha$ by
        \obv{ordinal_empty_or_emptyset_elem}.
    If $\emptyset\in\alpha$, then $\emptyset\precedes\alpha$.
    Hence if $\emptyset\in\alpha$, then
        $\alpha$ is a limit ordinal or
        $\alpha$ is a successor ordinal by
        \obv{positive_ordinal_is_limit_or_successor}.
\end{proof}

\begin{proposition}\label{coherent_tower_transport_zero_case}
    Suppose $\coherentTowerSegment{T}{G}{\theta}$ and
        $\coherentTowerSegment{U}{H}{\theta}$ and
        $\groupIsomorphism{e}{G}{H}$.
    Let $\alpha\in\theta$.
    Suppose $\alpha=\emptyset$ and
        $F=\towerTransportFamily{e}{T}{U}$.
    Then $\towerTransportAt{F}{T}{U}{\alpha}$.
\end{proposition}
\begin{proof}
    $\emptyset\in\theta$.
    Follows by \obv{tower_transport_at_zero}.
\end{proof}

\begin{proposition}\label{coherent_tower_transport_limit_case}
    Suppose $\coherentTowerSegment{T}{G}{\theta}$ and
        $\coherentTowerSegment{U}{H}{\theta}$.
    Let $\alpha\in\theta$.
    Suppose $\alpha$ is a limit ordinal and
        $\towerTransportBelow{F}{T}{U}{\alpha}$ and
        $F=\towerTransportFamily{e}{T}{U}$.
    Then $\towerTransportAt{F}{T}{U}{\alpha}$.
\end{proposition}
\begin{proof}
    Follows by \obv{tower_transport_at_limit}.
\end{proof}

\begin{proposition}\label{coherent_tower_transport_step}
    Suppose $\coherentTowerSegment{T}{G}{\theta}$ and
        $\coherentTowerSegment{U}{H}{\theta}$ and
        $\groupIsomorphism{e}{G}{H}$.
    Let $\alpha\in\theta$.
    Suppose $\towerTransportBelow{F}{T}{U}{\alpha}$ and
        $F=\towerTransportFamily{e}{T}{U}$.
    Then $\towerTransportAt{F}{T}{U}{\alpha}$.
\end{proposition}
\begin{proof}
    $\alpha$ is an ordinal by
        \obv{coherent_tower_index_is_ordinal}.
    $\alpha=\emptyset$ or $\emptyset\in\alpha$ by
        \obv{ordinal_empty_or_emptyset_elem}.
    If $\alpha=\emptyset$, then
        $\towerTransportAt{F}{T}{U}{\alpha}$ by
        \obv{coherent_tower_transport_zero_case}.
    If $\emptyset\in\alpha$, then $\emptyset\precedes\alpha$.
    Hence if $\emptyset\in\alpha$, then
        $\alpha$ is a limit ordinal or
        $\alpha$ is a successor ordinal by
        \obv{positive_ordinal_is_limit_or_successor}.
    If $\alpha$ is a limit ordinal, then
        $\towerTransportAt{F}{T}{U}{\alpha}$ by
        \obv{coherent_tower_transport_limit_case}.
    If $\alpha$ is a successor ordinal, then
        $\towerTransportAt{F}{T}{U}{\alpha}$ by
        \obv{tower_transport_successor_from_below}.
\end{proof}

\begin{theorem}\label{coherent_tower_transport_below}
    Suppose $\coherentTowerSegment{T}{G}{\theta}$ and
        $\coherentTowerSegment{U}{H}{\theta}$ and
        $\groupIsomorphism{e}{G}{H}$.
    Let $F=\towerTransportFamily{e}{T}{U}$.
    Then $\towerTransportBelow{F}{T}{U}{\theta}$.
\end{theorem}
\begin{proof}
    $\theta$ is an ordinal by
        \auto{coherent_tower_segment_diagram,
            tower_segment_diagram_data}.
    Show that for every $\alpha\in\theta$ such that
        $\towerTransportBelow{F}{T}{U}{\alpha}$ we have
        $\towerTransportAt{F}{T}{U}{\alpha}$.
    \begin{subproof}
        Fix $\alpha$.
        Assume $\alpha\in\theta$.
        Assume $\towerTransportBelow{F}{T}{U}{\alpha}$.
        Follows by \obv{coherent_tower_transport_step}.
    \end{subproof}
    Follows by \obv{tower_transport_bounded_induction}.
\end{proof}

\begin{theorem}\label{coherent_tower_transport_component}
    Suppose $\coherentTowerSegment{T}{G}{\theta}$ and
        $\coherentTowerSegment{U}{H}{\theta}$ and

    Let $\alpha\in\theta$.

        {\towerStage{T}{\alpha}}{\towerStage{U}{\alpha}}$.

\begin{proof}
    $\towerTransportBelow{F}{T}{U}{\theta}$ by
        \obv{coherent_tower_transport_below}.
    Hence $\towerTransportAt{F}{T}{U}{\alpha}$ by
        \obv{tower_transport_below_at}.
    Follows by \obv{tower_transport_at_component}.
\end{proof}

\begin{theorem}\label{coherent_tower_transport_square}
    Suppose $\coherentTowerSegment{T}{G}{\theta}$ and
        $\coherentTowerSegment{U}{H}{\theta}$ and
        $\groupIsomorphism{e}{G}{H}$.
    Let $F=\towerTransportFamily{e}{T}{U}$.
    Let $\alpha,\beta\in\theta$.
    Suppose $\alpha\subseteq\beta$.
    Then $\apply{F}{\beta}\circ\towerBond{T}{\alpha}{\beta}=
        \towerBond{U}{\alpha}{\beta}\circ\apply{F}{\alpha}$.
\end{theorem}
\begin{proof}
    $\towerTransportBelow{F}{T}{U}{\theta}$ by
        \obv{coherent_tower_transport_below}.
    Hence $\towerTransportAt{F}{T}{U}{\beta}$ by
        \obv{tower_transport_below_at}.
    $\alpha$ is an ordinal by
        \obv{coherent_tower_index_is_ordinal}.
    $\beta$ is an ordinal by
        \obv{coherent_tower_index_is_ordinal}.
    Hence $\alpha\in\beta$ or $\alpha=\beta$ by
        \obv{ordinal_subseteq_implies_elem_or_eq}.
    Hence $\alpha\in\suc{\beta}$ by \auto{suc_iff}.
    $\towerSegmentDiagram{T}{G}{\theta}$ and
        $\towerSegmentDiagram{U}{H}{\theta}$ by
        \obv{coherent_tower_segment_diagram}.
    Follows by \obv{tower_transport_at_nonstrict_square}.
\end{proof}

        $\groupIsomorphism{\apply{F}{\alpha}}
            {\towerStage{T}{\alpha}}{\towerStage{U}{\alpha}}$.

\subsection{Recursive characterization}

\begin{definition}\label{tower_transport_recursion}
    $\towerTransportRecursion{E}{e}{T}{U}$ iff
        $E$ is a function and
        $\dom{E}=\carrier[T]$ and
        for every $\alpha\in\carrier[T]$ we have
            $\apply{E}{\alpha}=
                \towerTransportStep{e}{T}{U}{\alpha}{\restrl{E}{\alpha}}$.
\end{definition}

\begin{proposition}\label{tower_transport_recursion_function}
    Suppose $\towerTransportRecursion{E}{e}{T}{U}$.
    Then $E$ is a function.
\end{proposition}
\begin{proof}
    Follows by \obv{tower_transport_recursion}.
\end{proof}

\begin{proposition}\label{tower_transport_recursion_domain}
    Suppose $\towerTransportRecursion{E}{e}{T}{U}$.
    Then $\dom{E}=\carrier[T]$.
\end{proposition}
\begin{proof}
    Follows by \obv{tower_transport_recursion}.
\end{proof}

\begin{proposition}\label{tower_transport_recursion_equation}
    Suppose $\towerTransportRecursion{E}{e}{T}{U}$.
    Let $\alpha\in\carrier[T]$.
    Then $\apply{E}{\alpha}=
        \towerTransportStep{e}{T}{U}{\alpha}{\restrl{E}{\alpha}}$.
\end{proposition}
\begin{proof}
    Follows by \obv{tower_transport_recursion}.
\end{proof}

\begin{proposition}\label{tower_transport_family_satisfies_recursion}
    Suppose $\carrier[T]$ is an ordinal.
    Then $\towerTransportRecursion
        {\towerTransportFamily{e}{T}{U}}{e}{T}{U}$.
\end{proposition}
\begin{proof}
    $\towerTransportFamily{e}{T}{U}$ is a function by
        \obv{tower_transport_family_function}.
    $\dom{\towerTransportFamily{e}{T}{U}}=\carrier[T]$ by
        \obv{tower_transport_family_domain}.
    For every $\alpha\in\carrier[T]$ we have
        $\apply{\towerTransportFamily{e}{T}{U}}{\alpha}=
        \towerTransportStep{e}{T}{U}{\alpha}{
            \restrl{\towerTransportFamily{e}{T}{U}}{\alpha}}$ by
        \obv{tower_transport_family_equation}.
    Follows by \obv{tower_transport_recursion}.
\end{proof}

\begin{theorem}\label{tower_transport_recursion_unique}
    Suppose $\carrier[T]$ is an ordinal and
        $\towerTransportRecursion{E}{e}{T}{U}$.
    Then $E=\towerTransportFamily{e}{T}{U}$.
\end{theorem}
\begin{proof}
    Hence $\Apply{\recursiveSetFunctionOp}{E}$ by
        \obv{recursive_set_function,recursive_set_function_op}.
    $\dom{E}=\carrier[T]$ by
        \obv{tower_transport_recursion_domain}.
    Show that for every $\alpha\in\carrier[T]$ we have
        $\apply{E}{\alpha}=
        \Apply{\towerTransportStepOp}
                {e,T,U,\alpha,\restrl{E}{\alpha}}$.
    \begin{subproof}
        Fix $\alpha\in\carrier[T]$.
        $\apply{E}{\alpha}=
            \towerTransportStep{e}{T}{U}{\alpha}{\restrl{E}{\alpha}}$ by
            \obv{tower_transport_recursion_equation}.
    \end{subproof}
    Hence $E=
        \Apply{\unboundedOrdinalRecursionOp}
            {\carrier[T],\Apply{\towerTransportStepOp}{e,T,U}}$ by
        \obv{unbounded_ordinal_recursion_characterization}.
    Follows by \obv{tower_transport_family}.
\end{proof}

\begin{corollary}\label{tower_transport_recursive_families_equal}
    Suppose $\carrier[T]$ is an ordinal and
        $\towerTransportRecursion{E}{e}{T}{U}$ and
        $\towerTransportRecursion{D}{e}{T}{U}$.
    Then $E=D$.
\end{corollary}
\begin{proof}
    $E=\towerTransportFamily{e}{T}{U}$ and
        $D=\towerTransportFamily{e}{T}{U}$ by
        \obv{tower_transport_recursion_unique}.
\end{proof}
