\section{Concrete direct limits of groups}

The construction uses tagged component elements, so no disjointness of
component carriers is assumed.  Eventual agreement is expressed by existence
of a common upper index; neither representatives nor upper bounds are selected
in the public construction.

\begin{definition}\label{direct_system_carrier_value_op}
    $\directSystemCarrierValueOp:
        \Set\to\Set\to\Set
        =\fun{S,i:\Set}{
                \carrier[\directSystemGroup{S}{i}]}$.
\end{definition}

\begin{definition}\label{direct_system_carrier_family}
    $\directSystemCarrierFamily{S}=
        \Apply{\graphOnDomainOp}
            {\carrier[S],\Apply{\directSystemCarrierValueOp}{S}}$.
\end{definition}

\begin{proposition}\label{direct_system_carrier_family_type}
    $\directSystemCarrierFamily{S}\in
        \funs{\carrier[S]}{\ran{\directSystemCarrierFamily{S}}}$.
\end{proposition}
\begin{proof}
    $\Apply{\graphOnDomainOp}
        {\carrier[S],\Apply{\directSystemCarrierValueOp}{S}}
        \in\funs{\carrier[S]}{
            \ran{\Apply{\graphOnDomainOp}
                {\carrier[S],\Apply{\directSystemCarrierValueOp}{S}}}}$ by
        \obv{logical_graph_on_domain_function}.
    Follows by \obv{direct_system_carrier_family}.
\end{proof}

\begin{proposition}\label{direct_system_carrier_family_apply}
    Let $i\in\carrier[S]$.
    Then $\apply{\directSystemCarrierFamily{S}}{i}
        =\carrier[\directSystemGroup{S}{i}]$.
\end{proposition}
\begin{proof}
    Follows by \auto{direct_system_carrier_family,direct_system_carrier_value_op,logical_graph_on_domain_apply}.
\end{proof}

\begin{definition}\label{direct_system_tagged_carrier}
    $\directSystemTaggedCarrier{S}=
        \indexedSum{\carrier[S]}{\directSystemCarrierFamily{S}}$.
\end{definition}

\begin{proposition}\label{direct_system_tagged_carrier_iff}
    $(i,x)\in\directSystemTaggedCarrier{S}$ iff
        $i\in\carrier[S]$ and
        $x\in\carrier[\directSystemGroup{S}{i}]$.
\end{proposition}
\begin{proof}
    $\directSystemCarrierFamily{S}$ is a function on $\carrier[S]$ by
        \auto{direct_system_carrier_family_type,funs_elim}.
    Follows by \auto{direct_system_tagged_carrier,indexed_sum_iff,direct_system_carrier_family_apply}.
\end{proof}

\begin{proposition}\label{direct_system_tagged_carrier_projections}
    Suppose $p\in\directSystemTaggedCarrier{S}$.
    Then $\fst{p}\in\carrier[S]$ and
        $\snd{p}\in\carrier[\directSystemGroup{S}{\fst{p}}]$.
\end{proposition}
\begin{proof}
    $p\in\carrier[S]\times\unions{\ran{\directSystemCarrierFamily{S}}}$
        and $\snd{p}\in
            \apply{\directSystemCarrierFamily{S}}{\fst{p}}$ by
        \auto{direct_system_tagged_carrier,indexed_sum}.
    Hence $\fst{p}\in\carrier[S]$ by \obv{fst_type}.
    $\apply{\directSystemCarrierFamily{S}}{\fst{p}}
        =\carrier[\directSystemGroup{S}{\fst{p}}]$ by
        \obv{direct_system_carrier_family_apply}.
\end{proof}

\begin{definition}\label{direct_system_move}
    $\directSystemMove{S}{p}{k}=
        \apply{\directSystemBond{S}{\fst{p}}{k}}{\snd{p}}$.
\end{definition}

\begin{proposition}\label{direct_system_move_type}
    Let $S$ be a group direct system.
    Let $p\in\directSystemTaggedCarrier{S}$.
    Let $k\in\carrier[S]$.
    Suppose $\fst{p}\mathrel{\leq[S]}k$.
    Then $\directSystemMove{S}{p}{k}\in
        \carrier[\directSystemGroup{S}{k}]$.
\end{proposition}
\begin{proof}
    $p\in\carrier[S]\times\unions{\ran{\directSystemCarrierFamily{S}}}$ by
        \auto{direct_system_tagged_carrier,indexed_sum}.
    Take $i,x$ such that $p=(i,x)$ by \auto{times_elem_is_tuple}.
    $\fst{p}=i$ and $\snd{p}=x$ by
        \auto{fst_eq,snd_eq}.
    $i\in\carrier[S]$ and
        $x\in\carrier[\directSystemGroup{S}{i}]$ by
        \auto{direct_system_tagged_carrier_iff}.
    $\groupHom
        {\directSystemBond{S}{i}{k}}
        {\directSystemGroup{S}{i}}
        {\directSystemGroup{S}{k}}$ by
        \obv{direct_system_bond_hom}.
    Hence $\apply{\directSystemBond{S}{i}{k}}{x}\in
        \carrier[\directSystemGroup{S}{k}]$ by
        \obv{group_hom_apply_type}.
    Follows by \auto{direct_system_move}.
\end{proof}

\begin{proposition}\label{direct_system_move_identity}
    Let $S$ be a group direct system.
    Let $p\in\directSystemTaggedCarrier{S}$.
    Then $\directSystemMove{S}{p}{\fst{p}}=\snd{p}$.
\end{proposition}
\begin{proof}
    $p\in\carrier[S]\times\unions{\ran{\directSystemCarrierFamily{S}}}$ by
        \auto{direct_system_tagged_carrier,indexed_sum}.
    Take $i,x$ such that $p=(i,x)$ by \auto{times_elem_is_tuple}.
    $\fst{p}=i$ and $\snd{p}=x$ by
        \auto{fst_eq,snd_eq}.
    $i\in\carrier[S]$ and
        $x\in\carrier[\directSystemGroup{S}{i}]$ by
        \auto{direct_system_tagged_carrier_iff}.
    $\directSystemBond{S}{i}{i}
        =\identity{\carrier[\directSystemGroup{S}{i}]}$ by
        \obv{direct_system_bond_identity}.
    Hence $\apply{\directSystemBond{S}{i}{i}}{x}=x$ by
        \auto{id_apply}.
    Follows by \obv{direct_system_move}.
\end{proof}

\begin{proposition}\label{direct_system_move_composition}
    Let $S$ be a group direct system.
    Let $p\in\directSystemTaggedCarrier{S}$.
    Let $j,k\in\carrier[S]$.
    Suppose $\fst{p}\mathrel{\leq[S]}j$ and
        $j\mathrel{\leq[S]}k$.
    Then $\directSystemMove{S}{p}{k}
        =\apply{\directSystemBond{S}{j}{k}}
            {\directSystemMove{S}{p}{j}}$.
\end{proposition}
\begin{proof}
    $\fst{p}\in\carrier[S]$ and $\snd{p}\in
        \carrier[\directSystemGroup{S}{\fst{p}}]$ by
        \auto{direct_system_tagged_carrier,indexed_sum,fst_type,direct_system_carrier_family_apply}.
    $\directSystemBond{S}{\fst{p}}{k}
        =\directSystemBond{S}{j}{k}\circ
            \directSystemBond{S}{\fst{p}}{j}$ by
        \obv{direct_system_bond_composition}.
    $\groupHom
        {\directSystemBond{S}{\fst{p}}{j}}
        {\directSystemGroup{S}{\fst{p}}}
        {\directSystemGroup{S}{j}}$ by
        \obv{direct_system_bond_hom}.
    $\groupHom
        {\directSystemBond{S}{j}{k}}
        {\directSystemGroup{S}{j}}
        {\directSystemGroup{S}{k}}$ by
        \obv{direct_system_bond_hom}.
    $\directSystemBond{S}{\fst{p}}{j}\in
        \funs{\carrier[\directSystemGroup{S}{\fst{p}}]}
            {\carrier[\directSystemGroup{S}{j}]}$ and
        $\directSystemBond{S}{j}{k}\in
        \funs{\carrier[\directSystemGroup{S}{j}]}
            {\carrier[\directSystemGroup{S}{k}]}$ by
        \obv{direct_system_bond_hom,group_hom_type}.
    Hence $\directSystemBond{S}{\fst{p}}{j}$ and
        $\directSystemBond{S}{j}{k}$ are functions by
        \obv{funs_is_function}.
    $\ran{\directSystemBond{S}{\fst{p}}{j}}\subseteq
        \carrier[\directSystemGroup{S}{j}]$ and
        $\dom{\directSystemBond{S}{j}{k}}=
        \carrier[\directSystemGroup{S}{j}]$ by
        \obv{direct_system_bond_hom,group_hom_type,funs,funs_ran}.
    Hence $\directSystemBond{S}{j}{k}$ is composable with
        $\directSystemBond{S}{\fst{p}}{j}$.
    $\snd{p}\in\dom{\directSystemBond{S}{\fst{p}}{j}}$ by
        \auto{direct_system_tagged_carrier,indexed_sum,fst_type,direct_system_carrier_family_apply,direct_system_bond_hom,group_hom_type,funs}.
    $\directSystemBond{S}{\fst{p}}{j}$ and
        $\directSystemBond{S}{j}{k}$ are functions and
        $\directSystemBond{S}{j}{k}$ is composable with
        $\directSystemBond{S}{\fst{p}}{j}$ and
        $\snd{p}\in\dom{\directSystemBond{S}{\fst{p}}{j}}$.
    Hence $\apply{
            \directSystemBond{S}{j}{k}\circ
            \directSystemBond{S}{\fst{p}}{j}}
            {\snd{p}}
        =\apply{\directSystemBond{S}{j}{k}}
            {\apply{\directSystemBond{S}{\fst{p}}{j}}{\snd{p}}}$ by
        \obv{circ_apply}.
    Hence $\apply{\directSystemBond{S}{\fst{p}}{k}}{\snd{p}}
        =\apply{\directSystemBond{S}{j}{k}}
            {\apply{\directSystemBond{S}{\fst{p}}{j}}{\snd{p}}}$ by
        \obv{direct_system_bond_composition}.
    Follows by \auto{direct_system_move}.
\end{proof}

\begin{definition}\label{direct_system_eventual_relation}
    $\directSystemEventualRelation{S}=
        \{(p,q)\mid
            p\in\directSystemTaggedCarrier{S},
            q\in\directSystemTaggedCarrier{S}\mid
            \text{there exists $k\in\carrier[S]$ such that
                $\fst{p}\mathrel{\leq[S]}k$ and
                $\fst{q}\mathrel{\leq[S]}k$ and
                $\directSystemMove{S}{p}{k}
                    =\directSystemMove{S}{q}{k}$}\}$.
\end{definition}

\begin{proposition}\label{direct_system_eventual_relation_iff}
    $p\mathrel{\directSystemEventualRelation{S}}q$ iff
        $p,q\in\directSystemTaggedCarrier{S}$ and
        there exists $k\in\carrier[S]$ such that
            $\fst{p}\mathrel{\leq[S]}k$ and
            $\fst{q}\mathrel{\leq[S]}k$ and
            $\directSystemMove{S}{p}{k}
                =\directSystemMove{S}{q}{k}$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{direct_system_eventual_relation,times_tuple_elim,pair_eq_iff}.
\end{proof}

\begin{lemma}\label{direct_system_agreement_persists}
    Let $S$ be a group direct system.
    Let $p,q\in\directSystemTaggedCarrier{S}$.
    Let $k,m\in\carrier[S]$.
    Suppose $\fst{p}\mathrel{\leq[S]}k$ and
        $\fst{q}\mathrel{\leq[S]}k$ and
        $k\mathrel{\leq[S]}m$.
    Suppose $\directSystemMove{S}{p}{k}
        =\directSystemMove{S}{q}{k}$.
    Then $\directSystemMove{S}{p}{m}
        =\directSystemMove{S}{q}{m}$.
\end{lemma}
\begin{proof}
    $\directSystemMove{S}{p}{m}
        =\apply{\directSystemBond{S}{k}{m}}
            {\directSystemMove{S}{p}{k}}$ by
        \obv{direct_system_move_composition}.
    $\directSystemMove{S}{q}{m}
        =\apply{\directSystemBond{S}{k}{m}}
            {\directSystemMove{S}{q}{k}}$ by
        \obv{direct_system_move_composition}.
\end{proof}

\begin{proposition}\label{direct_system_eventual_relation_type}
    $\directSystemEventualRelation{S}$ is a binary relation on
        $\directSystemTaggedCarrier{S}$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{direct_system_eventual_relation,subseteq,times_tuple_intro}.
\end{proof}

\begin{proposition}\label{direct_system_eventual_reflexive}
    Let $S$ be a group direct system.
    Then $\directSystemEventualRelation{S}$ is reflexive on
        $\directSystemTaggedCarrier{S}$.
\end{proposition}
\begin{proof}
    Show that for every $p\in\directSystemTaggedCarrier{S}$ we have
        $p\mathrel{\directSystemEventualRelation{S}}p$.
    \begin{subproof}
        Fix $p\in\directSystemTaggedCarrier{S}$.
        $\fst{p}\in\carrier[S]$ by
            \auto{direct_system_tagged_carrier_projections}.
        $S$ is a directed quasiordered set by
            \obv{group_direct_system_is_directed}.
        Hence $\fst{p}\mathrel{\leq[S]}\fst{p}$ by
            \obv{directed_order_reflexive}.
        $\directSystemMove{S}{p}{\fst{p}}=\snd{p}$ by
            \obv{direct_system_move_identity}.
        Follows by
            \obv{direct_system_eventual_relation_iff}.
    \end{subproof}
    Follows by \obv{reflexive_on}.
\end{proof}

\begin{proposition}\label{direct_system_eventual_symmetric}
    Let $S$ be a group direct system.
    Then $\directSystemEventualRelation{S}$ is symmetric.
\end{proposition}
\begin{proof}
    Show that for all $p,q$ such that
        $p\mathrel{\directSystemEventualRelation{S}}q$ we have
        $q\mathrel{\directSystemEventualRelation{S}}p$.
    \begin{subproof}
        Fix $p,q$.
        Assume $p\mathrel{\directSystemEventualRelation{S}}q$.
        Take $k\in\carrier[S]$ such that
            $\fst{p}\mathrel{\leq[S]}k$ and
            $\fst{q}\mathrel{\leq[S]}k$ and
            $\directSystemMove{S}{p}{k}
                =\directSystemMove{S}{q}{k}$ by
            \auto{direct_system_eventual_relation_iff}.
        $p,q\in\directSystemTaggedCarrier{S}$ by
            \auto{direct_system_eventual_relation_iff}.
        $\directSystemMove{S}{q}{k}
            =\directSystemMove{S}{p}{k}$.
        Follows by
            \auto{direct_system_eventual_relation_iff}.
    \end{subproof}
    Follows by \auto{symmetric}.
\end{proof}

\begin{proposition}\label{direct_system_eventual_transitive}
    Let $S$ be a group direct system.
    Then $\directSystemEventualRelation{S}$ is transitive.
\end{proposition}
\begin{proof}
    Show that for all $p,q,r$ such that
        $p\mathrel{\directSystemEventualRelation{S}}q$ and
        $q\mathrel{\directSystemEventualRelation{S}}r$ we have
        $p\mathrel{\directSystemEventualRelation{S}}r$.
    \begin{subproof}
        Fix $p,q,r$.
        Assume $p\mathrel{\directSystemEventualRelation{S}}q$ and
            $q\mathrel{\directSystemEventualRelation{S}}r$.
        Take $k\in\carrier[S]$ such that
            $\fst{p}\mathrel{\leq[S]}k$ and
            $\fst{q}\mathrel{\leq[S]}k$ and
            $\directSystemMove{S}{p}{k}
                =\directSystemMove{S}{q}{k}$ by
            \auto{direct_system_eventual_relation_iff}.
        Take $l\in\carrier[S]$ such that
            $\fst{q}\mathrel{\leq[S]}l$ and
            $\fst{r}\mathrel{\leq[S]}l$ and
            $\directSystemMove{S}{q}{l}
                =\directSystemMove{S}{r}{l}$ by
            \auto{direct_system_eventual_relation_iff}.
        $p,q,r\in\directSystemTaggedCarrier{S}$ by
            \auto{direct_system_eventual_relation_iff}.
        Hence $\fst{p},\fst{q},\fst{r}\in\carrier[S]$ by
            \auto{direct_system_tagged_carrier_projections}.
        $S$ is a directed quasiordered set by
            \obv{group_direct_system_is_directed}.
        Take $m\in\carrier[S]$ such that
            $k\mathrel{\leq[S]}m$ and $l\mathrel{\leq[S]}m$ by
            \obv{directed_common_upper_bound}.
        $\fst{p}\mathrel{\leq[S]}m$ by
            \obv{direct_system_eventual_relation_iff,direct_system_tagged_carrier_projections,group_direct_system_is_directed,directed_order_transitive}.
        $\fst{r}\mathrel{\leq[S]}m$ by
            \obv{direct_system_eventual_relation_iff,direct_system_tagged_carrier_projections,group_direct_system_is_directed,directed_order_transitive}.
        $\directSystemMove{S}{p}{m}
            =\directSystemMove{S}{q}{m}$ by
            \obv{direct_system_agreement_persists}.
        $\directSystemMove{S}{q}{m}
            =\directSystemMove{S}{r}{m}$ by
            \obv{direct_system_agreement_persists}.
        Hence $\directSystemMove{S}{p}{m}
            =\directSystemMove{S}{r}{m}$.
        Follows by
            \auto{direct_system_eventual_relation_iff}.
    \end{subproof}
    Follows by \obv{transitive}.
\end{proof}

\begin{theorem}\label{direct_system_eventual_is_equivalence}
    Let $S$ be a group direct system.
    Then $\directSystemEventualRelation{S}$ is an equivalence on
        $\directSystemTaggedCarrier{S}$.
\end{theorem}
\begin{proof}
    $\directSystemEventualRelation{S}$ is a binary relation on
        $\directSystemTaggedCarrier{S}$ by
        \obv{direct_system_eventual_relation_type}.
    $\directSystemEventualRelation{S}$ is reflexive on
        $\directSystemTaggedCarrier{S}$ by
        \obv{direct_system_eventual_reflexive}.
    $\directSystemEventualRelation{S}$ is transitive by
        \obv{direct_system_eventual_transitive}.
    $\directSystemEventualRelation{S}$ is symmetric by
        \obv{direct_system_eventual_symmetric}.
\end{proof}

\subsection{The quotient carrier}

\begin{definition}\label{group_direct_limit_carrier}
    $\directLimitCarrier{S}=
        \quotient{\directSystemTaggedCarrier{S}}
            {\directSystemEventualRelation{S}}$.
\end{definition}

\begin{definition}\label{direct_system_class}
    $\directSystemClass{S}{p}=
        \equivalenceClass{\directSystemEventualRelation{S}}{p}$.
\end{definition}

\begin{definition}\label{group_direct_limit_projection}
    $\directLimitProjection{S}=
        \quotientProjection
            {\directSystemTaggedCarrier{S}}

\begin{proposition}\label{direct_system_class_type}
    Let $S$ be a group direct system.
    Let $p\in\directSystemTaggedCarrier{S}$.
    Then $\directSystemClass{S}{p}\in\directLimitCarrier{S}$.
\end{proposition}
\begin{proof}
    Follows by
        \obv{group_direct_limit_carrier,direct_system_class,quotient}.
\end{proof}

\begin{proposition}\label{direct_system_class_equality_iff}
    Let $S$ be a group direct system.
    Let $p,q\in\directSystemTaggedCarrier{S}$.
    Then $\directSystemClass{S}{p}=\directSystemClass{S}{q}$ iff
        $p\mathrel{\directSystemEventualRelation{S}}q$.
\end{proposition}
\begin{proof}
    $\directSystemEventualRelation{S}$ is an equivalence on
        $\directSystemTaggedCarrier{S}$ by
        \obv{direct_system_eventual_is_equivalence}.

\end{proof}

\begin{proposition}\label{direct_limit_carrier_representation}

\begin{proof}
    Follows by

\begin{proposition}\label{direct_limit_carrier_coordinates}
    Let $S$ be a group direct system.
    Let $C\in\directLimitCarrier{S}$.

\begin{proof}

        \auto{times_elem_is_tuple}.
    $i\in\carrier[S]$ and
        $x\in\carrier[\directSystemGroup{S}{i}]$ by
        \auto{direct_system_tagged_carrier_iff}.
    Follows by \auto.
\end{proof}

\begin{proposition}\label{direct_limit_projection_type}

\subsection{Representative-free quotient operations}

\begin{definition}\label{direct_system_product_tag}
    $\directSystemProductTag{S}{p}{q}{k}=
        (k,\mul[\directSystemGroup{S}{k}]
            (\directSystemMove{S}{p}{k},
             \directSystemMove{S}{q}{k}))$.
\end{definition}

\begin{proposition}\label{direct_system_product_tag_type}
    Let $S$ be a group direct system.
    Let $p,q\in\directSystemTaggedCarrier{S}$.
    Let $k\in\carrier[S]$.
    Suppose $\fst{p}\mathrel{\leq[S]}k$ and
        $\fst{q}\mathrel{\leq[S]}k$.
    Then $\directSystemProductTag{S}{p}{q}{k}\in
        \directSystemTaggedCarrier{S}$.
\end{proposition}
\begin{proof}
    $\directSystemMove{S}{p}{k},\directSystemMove{S}{q}{k}\in
        \carrier[\directSystemGroup{S}{k}]$ by
        \obv{direct_system_move_type}.
    $\directSystemGroup{S}{k}$ is a group by
        \obv{direct_system_group_is_group}.
    Hence $\mul[\directSystemGroup{S}{k}]
        (\directSystemMove{S}{p}{k},\directSystemMove{S}{q}{k})\in
        \carrier[\directSystemGroup{S}{k}]$ by
        \obv{group_is_monoid,monoid_is_magma,magma_welldef}.
    Follows by
        \auto{direct_system_product_tag,direct_system_tagged_carrier_iff}.
\end{proof}

\begin{proposition}\label{direct_system_product_tag_first}
    $\fst{\directSystemProductTag{S}{p}{q}{k}}=k$.
\end{proposition}
\begin{proof}
    Follows by \auto{direct_system_product_tag,fst_eq}.
\end{proof}

\begin{proposition}\label{direct_system_product_tag_second}
    $\snd{\directSystemProductTag{S}{p}{q}{k}}=
        \mul[\directSystemGroup{S}{k}]
            (\directSystemMove{S}{p}{k},
             \directSystemMove{S}{q}{k})$.
\end{proposition}
\begin{proof}
    Follows by \auto{direct_system_product_tag,snd_eq}.
\end{proof}

\begin{lemma}\label{direct_system_product_tag_moves}
    Let $S$ be a group direct system.
    Let $p,q\in\directSystemTaggedCarrier{S}$.
    Let $k,m\in\carrier[S]$.
    Suppose $\fst{p}\mathrel{\leq[S]}k$ and
        $\fst{q}\mathrel{\leq[S]}k$ and
        $k\mathrel{\leq[S]}m$.
    Then
        $\directSystemMove{S}
            {\directSystemProductTag{S}{p}{q}{k}}{m}
        =\mul[\directSystemGroup{S}{m}]
            (\directSystemMove{S}{p}{m},
             \directSystemMove{S}{q}{m})$.
\end{lemma}
\begin{proof}
    $\directSystemMove{S}{p}{k},\directSystemMove{S}{q}{k}\in
        \carrier[\directSystemGroup{S}{k}]$ by
        \obv{direct_system_move_type}.
    $\groupHom
        {\directSystemBond{S}{k}{m}}
        {\directSystemGroup{S}{k}}
        {\directSystemGroup{S}{m}}$ by
        \obv{direct_system_bond_hom}.
    Hence
        $\apply{\directSystemBond{S}{k}{m}}
            {\mul[\directSystemGroup{S}{k}]
                (\directSystemMove{S}{p}{k},
                 \directSystemMove{S}{q}{k})}
        =\mul[\directSystemGroup{S}{m}]
            (\apply{\directSystemBond{S}{k}{m}}
                {\directSystemMove{S}{p}{k}},
             \apply{\directSystemBond{S}{k}{m}}
                {\directSystemMove{S}{q}{k}})$ by
        \obv{group_hom_mul}.
    $\directSystemMove{S}{p}{m}
        =\apply{\directSystemBond{S}{k}{m}}
            {\directSystemMove{S}{p}{k}}$ and
        $\directSystemMove{S}{q}{m}
        =\apply{\directSystemBond{S}{k}{m}}
            {\directSystemMove{S}{q}{k}}$ by
        \obv{direct_system_move_composition}.
    Follows by

\begin{theorem}\label{direct_system_product_relation_type}
    Let $S$ be a group direct system.
    Then $\directSystemProductRelation{S}\in
        \funs{
            \directSystemTaggedCarrier{S}\times
            \directSystemTaggedCarrier{S}}
            {\directLimitCarrier{S}}$.
\end{theorem}
\begin{proof}
    Let $R=\directSystemProductRelation{S}$.
    Let $A=\directSystemTaggedCarrier{S}\times
        \directSystemTaggedCarrier{S}$.
    Let $Q=\directLimitCarrier{S}$.
    Show that $R$ is a relation.
    \begin{subproof}
        Show that for every $z\in R$ there exist $w,C$ such that
            $z=(w,C)$.
        \begin{subproof}
            Fix $z\in R$.
            Take $w,C$ such that $z=(w,C)$ by
                \auto{direct_system_product_relation}.
            Follows by \auto.
        \end{subproof}
        Follows by \obv{relation}.
    \end{subproof}
    $R$ is right-unique by
        \obv{direct_system_product_relation_right_unique}.
    Show that $\dom{R}=A$.
    \begin{subproof}
        For every $w\in\dom{R}$ we have $w\in A$ by
            \auto{dom_iff,direct_system_product_relation_iff}.
        Show that for every $w\in A$ we have $w\in\dom{R}$.
        \begin{subproof}
            Fix $w\in A$.
            Take $C\in Q$ such that $(w,C)\in R$ by
                \auto{direct_system_product_relation_total}.
            Follows by \auto{dom_iff}.
        \end{subproof}
        Follows by set extensionality.
    \end{subproof}
    Show that $R$ is a function to $Q$.
    \begin{subproof}
        Show that for every $w\in\dom{R}$ we have
            $\apply{R}{w}\in Q$.
        \begin{subproof}
            Fix $w\in\dom{R}$.
            $(w,\apply{R}{w})\in R$ by
                \obv{function_apply_elim}.
            Follows by
                \auto{direct_system_product_relation_iff}.
        \end{subproof}
    \end{subproof}
    Follows by \auto{funs_intro}.
\end{proof}

\begin{theorem}\label{group_direct_limit_is_exact_group}

    Then $\groupDirectLimit{S}$ is an exact group.
\end{theorem}
\begin{proof}
    Let $K=\groupDirectLimit{S}$.
    Let $Q=\directLimitCarrier{S}$.
    $\carrier[K]=Q$ by
        \auto{group_direct_limit_structure_carrier}.
    $\mul[K]=\directLimitMultiplication{S}$ by
        \auto{group_direct_limit_structure_mul}.
    $\neutral[K]=\directLimitNeutral{S}$ by
        \auto{group_direct_limit_structure_neutral}.
    $\groupInv[K]=\directLimitInverse{S}$ by
        \auto{group_direct_limit_structure_inverse}.
    $\mul[K]\in\funs{\carrier[K]\times\carrier[K]}{\carrier[K]}$ by
        \auto{direct_limit_multiplication_type}.
    $\neutral[K]\in\carrier[K]$ by
        \auto{direct_limit_neutral_type}.
    $\groupInv[K]\in\funs{\carrier[K]}{\carrier[K]}$ by
        \auto{direct_limit_inverse_type}.
    Show that for every $U\in\carrier[K]$ we have
        $\mul[K](U,\neutral[K])=U$.
    \begin{subproof}
        Fix $U\in\carrier[K]$.
        Take $i,x$ such that $i\in\carrier[S]$ and
            $x\in\carrier[\directSystemGroup{S}{i}]$ and
            $U=\directSystemClass{S}{(i,x)}$ by
            \obv{direct_limit_carrier_coordinates}.
        $\neutral[K]=\directSystemClass{S}
            {(i,\neutral[\directSystemGroup{S}{i}])}$ by
            \auto{group_direct_limit_structure_neutral,direct_limit_neutral_is_stage_class,direct_system_neutral_tag}.
        $\directSystemGroup{S}{i}$ is a group by
            \obv{direct_system_group_is_group}.
        Hence $\neutral[\directSystemGroup{S}{i}]\in
            \carrier[\directSystemGroup{S}{i}]$ by
            \obv{group_is_monoid,monoid_is_unital_magma,unitalmagma_type}.
        $\apply{\directLimitMultiplication{S}}
            {(\directSystemClass{S}{(i,x)},
              \directSystemClass{S}
                {(i,\neutral[\directSystemGroup{S}{i}])})}=
            \directSystemClass{S}
                {(i,\mul[\directSystemGroup{S}{i}]
                    (x,\neutral[\directSystemGroup{S}{i}]))}$ by
            \obv{direct_limit_multiplication_same_stage}.
        Hence $\mul[\directSystemGroup{S}{i}]
            (x,\neutral[\directSystemGroup{S}{i}])=x$ by
            \obv{group_is_monoid,monoid_is_unital_magma,unitalmagma_right}.
        Follows by
            \obv{group_direct_limit_structure_mul}.
    \end{subproof}
    Show that for every $U\in\carrier[K]$ we have
        $\mul[K](\neutral[K],U)=U$.
    \begin{subproof}
        Fix $U\in\carrier[K]$.
        Take $i,x$ such that $i\in\carrier[S]$ and
            $x\in\carrier[\directSystemGroup{S}{i}]$ and
            $U=\directSystemClass{S}{(i,x)}$ by
            \obv{direct_limit_carrier_coordinates}.
        $\neutral[K]=\directSystemClass{S}
            {(i,\neutral[\directSystemGroup{S}{i}])}$ by
            \auto{group_direct_limit_structure_neutral,direct_limit_neutral_is_stage_class,direct_system_neutral_tag}.
        $\directSystemGroup{S}{i}$ is a group by
            \obv{direct_system_group_is_group}.
        Hence $\neutral[\directSystemGroup{S}{i}]\in
            \carrier[\directSystemGroup{S}{i}]$ by
            \obv{group_is_monoid,monoid_is_unital_magma,unitalmagma_type}.
        $\apply{\directLimitMultiplication{S}}
            {(\directSystemClass{S}
                {(i,\neutral[\directSystemGroup{S}{i}])},
              \directSystemClass{S}{(i,x)})}=
            \directSystemClass{S}
                {(i,\mul[\directSystemGroup{S}{i}]
                    (\neutral[\directSystemGroup{S}{i}],x))}$ by
            \obv{direct_limit_multiplication_same_stage}.
        Hence $\mul[\directSystemGroup{S}{i}]
            (\neutral[\directSystemGroup{S}{i}],x)=x$ by
            \obv{group_is_monoid,monoid_is_unital_magma,unitalmagma_left}.
        Follows by
            \obv{group_direct_limit_structure_mul}.
    \end{subproof}
    Hence $K$ is a unital magma by
        \auto{unitalmagma,group_direct_limit,groupaggregateexact,groupaggregateprojection_carrier,groupaggregateprojection_mul,groupaggregateprojection_neutral}.
    Show that for all $U,V,W\in\carrier[K]$ we have
        $\mul[K](U,\mul[K](V,W))=
        \mul[K](\mul[K](U,V),W)$.
    \begin{subproof}
        Fix $U,V,W\in\carrier[K]$.
        Take $i,x$ such that $i\in\carrier[S]$ and
            $x\in\carrier[\directSystemGroup{S}{i}]$ and
            $U=\directSystemClass{S}{(i,x)}$ by
            \obv{direct_limit_carrier_coordinates}.
        Take $j,y$ such that $j\in\carrier[S]$ and
            $y\in\carrier[\directSystemGroup{S}{j}]$ and
            $V=\directSystemClass{S}{(j,y)}$ by
            \obv{direct_limit_carrier_coordinates}.
        Take $l,z$ such that $l\in\carrier[S]$ and
            $z\in\carrier[\directSystemGroup{S}{l}]$ and
            $W=\directSystemClass{S}{(l,z)}$ by
            \obv{direct_limit_carrier_coordinates}.
        $S$ is a directed quasiordered set by
            \obv{group_direct_system_is_directed}.
        Take $n\in\carrier[S]$ such that
            $i\mathrel{\leq[S]}n$ and $j\mathrel{\leq[S]}n$ by
            \obv{directed_common_upper_bound}.
        Take $k\in\carrier[S]$ such that
            $n\mathrel{\leq[S]}k$ and $l\mathrel{\leq[S]}k$ by
            \obv{directed_common_upper_bound}.
        $i\mathrel{\leq[S]}k$ and $j\mathrel{\leq[S]}k$ by
            \obv{group_direct_system_is_directed,directed_order_transitive}.
        Let $a=\directSystemMove{S}{(i,x)}{k}$.
        Let $b=\directSystemMove{S}{(j,y)}{k}$.
        Let $c=\directSystemMove{S}{(l,z)}{k}$.
        $(i,x),(j,y),(l,z)\in\directSystemTaggedCarrier{S}$ by
            \auto{direct_system_tagged_carrier_iff}.
        $a,b,c\in\carrier[\directSystemGroup{S}{k}]$ by
            \auto{direct_system_move_type}.
        $U=\directSystemClass{S}{(k,a)}$ and
            $V=\directSystemClass{S}{(k,b)}$ and
            $W=\directSystemClass{S}{(k,c)}$ by
            \auto{direct_system_class_move,direct_system_moved_tag}.
        $\directSystemGroup{S}{k}$ is a group by
            \obv{direct_system_group_is_group}.
        Hence $\mul[\directSystemGroup{S}{k}](a,b),
            \mul[\directSystemGroup{S}{k}](b,c)\in
            \carrier[\directSystemGroup{S}{k}]$ by
            \obv{group_is_monoid,monoid_is_magma,magma_welldef}.
        $\mul[K](V,W)=\directSystemClass{S}
            {(k,\mul[\directSystemGroup{S}{k}](b,c))}$ and
            $\mul[K](U,V)=\directSystemClass{S}
            {(k,\mul[\directSystemGroup{S}{k}](a,b))}$ by
            \auto{group_direct_limit_structure_mul,direct_limit_multiplication_same_stage}.
        Hence $\mul[K](U,\mul[K](V,W))=
            \directSystemClass{S}
                {(k,\mul[\directSystemGroup{S}{k}]
                    (a,\mul[\directSystemGroup{S}{k}](b,c)))}$ by
            \auto{group_direct_limit_structure_mul,direct_limit_multiplication_same_stage}.
        $\mul[K](\mul[K](U,V),W)=
            \directSystemClass{S}
                {(k,\mul[\directSystemGroup{S}{k}]
                    (\mul[\directSystemGroup{S}{k}](a,b),c))}$ by
            \auto{group_direct_limit_structure_mul,direct_limit_multiplication_same_stage}.
        $\mul[\directSystemGroup{S}{k}]
            (a,\mul[\directSystemGroup{S}{k}](b,c))
            =\mul[\directSystemGroup{S}{k}]
                (\mul[\directSystemGroup{S}{k}](a,b),c)$ by
            \obv{group_is_monoid,monoid_assoc}.
    \end{subproof}
    Show that for every $U\in\carrier[K]$ we have
        $\mul[K](U,\groupInv[K](U))=\neutral[K]$.
    \begin{subproof}
        Fix $U\in\carrier[K]$.
        Take $i,x$ such that $i\in\carrier[S]$ and
            $x\in\carrier[\directSystemGroup{S}{i}]$ and
            $U=\directSystemClass{S}{(i,x)}$ by
            \obv{direct_limit_carrier_coordinates}.
        $\groupInv[\directSystemGroup{S}{i}](x)\in
            \carrier[\directSystemGroup{S}{i}]$ by
            \obv{direct_system_group_is_group,group_inverse_type}.
        $\groupInv[K](U)=\directSystemClass{S}
            {(i,\groupInv[\directSystemGroup{S}{i}](x))}$ by
            \auto{group_direct_limit_structure_inverse,direct_limit_inverse_same_stage}.
        Hence $\mul[K](U,\groupInv[K](U))=
            \directSystemClass{S}
                {(i,\mul[\directSystemGroup{S}{i}]
                    (x,\groupInv[\directSystemGroup{S}{i}](x)))}$ by
            \auto{group_direct_limit_structure_mul,direct_limit_multiplication_same_stage}.
        $\mul[\directSystemGroup{S}{i}]
            (x,\groupInv[\directSystemGroup{S}{i}](x))
            =\neutral[\directSystemGroup{S}{i}]$ by
            \obv{direct_system_group_is_group,group_inverse_right}.
        $\neutral[K]=\directSystemClass{S}
            {(i,\neutral[\directSystemGroup{S}{i}])}$ by
            \auto{group_direct_limit_structure_neutral,direct_limit_neutral_is_stage_class,direct_system_neutral_tag}.
    \end{subproof}
    Show that for every $U\in\carrier[K]$ we have
        $\mul[K](\groupInv[K](U),U)=\neutral[K]$.
    \begin{subproof}
        Fix $U\in\carrier[K]$.
        Take $i,x$ such that $i\in\carrier[S]$ and
            $x\in\carrier[\directSystemGroup{S}{i}]$ and
            $U=\directSystemClass{S}{(i,x)}$ by
            \obv{direct_limit_carrier_coordinates}.
        $\groupInv[\directSystemGroup{S}{i}](x)\in
            \carrier[\directSystemGroup{S}{i}]$ by
            \obv{direct_system_group_is_group,group_inverse_type}.
        $\groupInv[K](U)=\directSystemClass{S}
            {(i,\groupInv[\directSystemGroup{S}{i}](x))}$ by
            \auto{group_direct_limit_structure_inverse,direct_limit_inverse_same_stage}.
        Hence $\mul[K](\groupInv[K](U),U)=
            \directSystemClass{S}
                {(i,\mul[\directSystemGroup{S}{i}]
                    (\groupInv[\directSystemGroup{S}{i}](x),x))}$ by
            \auto{group_direct_limit_structure_mul,direct_limit_multiplication_same_stage}.
        $\mul[\directSystemGroup{S}{i}]
            (\groupInv[\directSystemGroup{S}{i}](x),x)
            =\neutral[\directSystemGroup{S}{i}]$ by
            \obv{direct_system_group_is_group,group_inverse_left}.
        $\neutral[K]=\directSystemClass{S}
            {(i,\neutral[\directSystemGroup{S}{i}])}$ by
            \auto{group_direct_limit_structure_neutral,direct_limit_neutral_is_stage_class,direct_system_neutral_tag}.
    \end{subproof}
    Hence $K$ is a group by
        \auto{group,group_direct_limit,groupaggregateexact,groupaggregateprojection_carrier,groupaggregateprojection_mul,groupaggregateprojection_neutral,groupaggregateprojection_groupInv}.
    Follows by \auto{group_direct_limit,groupaggregateexact}.
\end{proof}

\begin{definition}\label{direct_system_canonical_map}

\begin{proposition}\label{direct_system_canonical_map_iff}

\end{proposition}

\begin{theorem}\label{direct_system_canonical_map_type}

    Then $\directSystemCanonicalMap{S}{i}\in
        \funs{\carrier[\directSystemGroup{S}{i}]}
            {\directLimitCarrier{S}}$.
\end{theorem}
\begin{proof}
    Let $f=\directSystemCanonicalMap{S}{i}$.
    Let $A=\carrier[\directSystemGroup{S}{i}]$.
    Let $Q=\directLimitCarrier{S}$.
    Show that $f$ is a relation.
    \begin{subproof}
        Show that for every $z\in f$ there exist $x,C$ such that
            $z=(x,C)$.
        \begin{subproof}
            Fix $z\in f$.
            Take $x\in A$ such that
                $z=(x,\directSystemClass{S}{(i,x)})$ by
                \auto{direct_system_canonical_map}.
            Follows by \auto.
        \end{subproof}
        Follows by \obv{relation}.
    \end{subproof}
    Show that $f$ is right-unique.
    \begin{subproof}
        Show that for all $x,C,D$ such that $(x,C),(x,D)\in f$
            we have $C=D$.
        \begin{subproof}
            Fix $x,C,D$.
            Assume $(x,C),(x,D)\in f$.
            $C=\directSystemClass{S}{(i,x)}$ and
                $D=\directSystemClass{S}{(i,x)}$ by
                \auto{direct_system_canonical_map_iff}.
        \end{subproof}
        Follows by \obv{rightunique}.
    \end{subproof}
    Show that $\dom{f}=A$.
    \begin{subproof}
        For every $x\in\dom{f}$ we have $x\in A$ by
            \auto{dom_iff,direct_system_canonical_map_iff}.
        For every $x\in A$ we have $x\in\dom{f}$ by
            \auto{dom_iff,direct_system_canonical_map_iff}.
        Follows by \obv{setext}.
    \end{subproof}
    Show that $f$ is a function to $Q$.
    \begin{subproof}
        Show that for every $x\in\dom{f}$ we have
            $\apply{f}{x}\in Q$.
        \begin{subproof}
            Fix $x\in\dom{f}$.
            $(x,\apply{f}{x})\in f$ by
                \obv{function_apply_elim}.
            Hence $x\in A$ and
                $\apply{f}{x}=\directSystemClass{S}{(i,x)}$ by
                \auto{direct_system_canonical_map_iff}.
            $(i,x)\in\directSystemTaggedCarrier{S}$ by
                \auto{direct_system_tagged_carrier_iff}.
            Follows by \obv{direct_system_class_type}.
        \end{subproof}
    \end{subproof}
    Follows by \auto{funs_intro}.
\end{proof}

\begin{theorem}\label{direct_system_canonical_map_hom}

        {\directSystemGroup{S}{i}}

\end{theorem}
\begin{proof}

    $\groupDirectLimit{S}$ is a group by
        \obv{group_direct_limit_is_group}.
    $\carrier[\groupDirectLimit{S}]=\directLimitCarrier{S}$ by
        \obv{group_direct_limit_structure_carrier}.
    Hence $\directSystemCanonicalMap{S}{i}\in
        \funs{\carrier[\directSystemGroup{S}{i}]}
            {\carrier[\groupDirectLimit{S}]}$ by
        \auto{direct_system_canonical_map_type}.
    Show that for all $x,y\in\carrier[\directSystemGroup{S}{i}]$
        we have
        $\apply{\directSystemCanonicalMap{S}{i}}
            {\mul[\directSystemGroup{S}{i}](x,y)}
        =\mul[\groupDirectLimit{S}]
            (\apply{\directSystemCanonicalMap{S}{i}}{x},
             \apply{\directSystemCanonicalMap{S}{i}}{y})$.
    \begin{subproof}
        Fix $x,y\in\carrier[\directSystemGroup{S}{i}]$.
        $\mul[\directSystemGroup{S}{i}](x,y)\in
            \carrier[\directSystemGroup{S}{i}]$ by

                {(i,\mul[\directSystemGroup{S}{i}](x,y))}$ by

            =\directSystemClass{S}{(i,x)}$ and
            $\apply{\directSystemCanonicalMap{S}{i}}{y}
            =\directSystemClass{S}{(i,y)}$ by

\end{proof}

\begin{theorem}\label{direct_system_canonical_map_compatibility}
    Let $S$ be a group direct system.
    Let $i,j\in\carrier[S]$.
    Suppose $i\mathrel{\leq[S]}j$.
    Then
        $\directSystemCanonicalMap{S}{j}\circ
            \directSystemBond{S}{i}{j}
        =\directSystemCanonicalMap{S}{i}$.
\end{theorem}
\begin{proof}
    Let $f=\directSystemBond{S}{i}{j}$.
    Let $g=\directSystemCanonicalMap{S}{j}$.
    Let $h=\directSystemCanonicalMap{S}{i}$.
    Let $A=\carrier[\directSystemGroup{S}{i}]$.
    Let $B=\carrier[\directSystemGroup{S}{j}]$.
    Let $Q=\directLimitCarrier{S}$.
    $f\in\funs{A}{B}$ by
        \auto{direct_system_bond_hom,group_hom_type}.
    $g\in\funs{B}{Q}$ and $h\in\funs{A}{Q}$ by
        \auto{direct_system_canonical_map_type}.
    Hence $g\circ f\in\funs{A}{Q}$ by
        \obv{funs_circ}.
    $g$ is composable with $f$ by
        \auto{funs,funs_ran,subseteq}.
    $g\circ f$ and $h$ are functions and
        $\dom{g\circ f}=A=\dom{h}$ by
        \auto{funs_elim,funs_is_function}.
    Show that for every $x\in\dom{g\circ f}$ we have
        $\apply{g\circ f}{x}=\apply{h}{x}$.
    \begin{subproof}
        Fix $x\in\dom{g\circ f}$.
        Hence $x\in A$.
        $\apply{f}{x}\in B$ by \obv{funs_type_apply}.
        $\apply{g\circ f}{x}=\apply{g}{\apply{f}{x}}$ by
            \auto{funs_is_function,funs_elim,circ_apply}.
        $\apply{g}{\apply{f}{x}}=
            \directSystemClass{S}{(j,\apply{f}{x})}$ by
            \auto{direct_system_canonical_map_apply}.
        $\apply{h}{x}=\directSystemClass{S}{(i,x)}$ by
            \auto{direct_system_canonical_map_apply}.
        $\directSystemClass{S}{(i,x)}=
            \directSystemClass{S}{(j,\apply{f}{x})}$ by
            \auto{direct_system_class_bond}.
    \end{subproof}
    Follows by \auto{funext_on_domain}.
\end{proof}

            {\groupDirectLimit{S}}$ by
        \auto{direct_limit_canonical_family_apply,direct_system_canonical_map_hom}.

        $i\mathrel{\leq[S]}j$ we have
        $\apply{\directLimitCanonicalFamily{S}}{j}\circ
            \directSystemBond{S}{i}{j}
        =\apply{\directLimitCanonicalFamily{S}}{i}$ by
        \auto{direct_limit_canonical_family_apply,direct_system_canonical_map_compatibility}.

\end{proof}

\begin{theorem}\label{direct_limit_mediator_hom}
    Suppose $\groupCocone{S}{H}{\kappa}$.
    Then $\groupHom
        {\directLimitMediator{S}{H}{\kappa}}
        {\groupDirectLimit{S}}{H}$.
\end{theorem}
\begin{proof}
    Let $u=\directLimitMediator{S}{H}{\kappa}$.
    Let $L=\groupDirectLimit{S}$.
    $S$ is a group direct system by
        \obv{group_cocone_system}.
    $H$ is a group by \obv{group_cocone_target}.
    $L$ is a group by \obv{group_direct_limit_is_group}.
    $\carrier[L]=\directLimitCarrier{S}$ by
        \auto{group_direct_limit_structure_carrier}.
    Hence $u\in\funs{\carrier[L]}{\carrier[H]}$ by
        \auto{direct_limit_mediator_type}.
    Show that for all $U,V\in\carrier[L]$ we have
        $\apply{u}{\mul[L](U,V)}=
        \mul[H](\apply{u}{U},\apply{u}{V})$.
    \begin{subproof}
        Fix $U,V\in\carrier[L]$.
        Take $i,x$ such that $i\in\carrier[S]$ and
            $x\in\carrier[\directSystemGroup{S}{i}]$ and
            $U=\directSystemClass{S}{(i,x)}$ by
            \obv{direct_limit_carrier_coordinates}.
        Take $j,y$ such that $j\in\carrier[S]$ and
            $y\in\carrier[\directSystemGroup{S}{j}]$ and
            $V=\directSystemClass{S}{(j,y)}$ by
            \obv{direct_limit_carrier_coordinates}.
        $S$ is a directed quasiordered set by
            \obv{group_direct_system_is_directed}.
        Take $k\in\carrier[S]$ such that
            $i\mathrel{\leq[S]}k$ and $j\mathrel{\leq[S]}k$ by
            \obv{directed_common_upper_bound}.
        Let $a=\directSystemMove{S}{(i,x)}{k}$.
        Let $b=\directSystemMove{S}{(j,y)}{k}$.
        $(i,x),(j,y)\in\directSystemTaggedCarrier{S}$ by
            \auto{direct_system_tagged_carrier_iff}.
        $a,b\in\carrier[\directSystemGroup{S}{k}]$ by
            \auto{direct_system_move_type}.
        $U=\directSystemClass{S}{(k,a)}$ and
            $V=\directSystemClass{S}{(k,b)}$ by
            \auto{direct_system_class_move,direct_system_moved_tag}.
        $\directSystemGroup{S}{k}$ is a group by
            \obv{direct_system_group_is_group}.
        Hence $\mul[\directSystemGroup{S}{k}](a,b)\in
            \carrier[\directSystemGroup{S}{k}]$ by
            \obv{group_is_monoid,monoid_is_magma,magma_welldef}.
        $\mul[L](U,V)=\directSystemClass{S}
            {(k,\mul[\directSystemGroup{S}{k}](a,b))}$ by
            \auto{group_direct_limit_structure_mul,direct_limit_multiplication_same_stage}.
        Hence $\apply{u}{\mul[L](U,V)}=
            \apply{\apply{\kappa}{k}}
                {\mul[\directSystemGroup{S}{k}](a,b)}$ by
            \auto{direct_limit_mediator_apply}.
        $\apply{u}{U}=\apply{\apply{\kappa}{k}}{a}$ and
            $\apply{u}{V}=\apply{\apply{\kappa}{k}}{b}$ by
            \auto{direct_limit_mediator_apply}.
        $\groupHom{\apply{\kappa}{k}}
            {\directSystemGroup{S}{k}}{H}$ by
            \obv{group_cocone_leg_hom}.
        Hence $\apply{\apply{\kappa}{k}}
                {\mul[\directSystemGroup{S}{k}](a,b)}
            =\mul[H]
                (\apply{\apply{\kappa}{k}}{a},
                 \apply{\apply{\kappa}{k}}{b})$ by
            \obv{group_hom_mul}.
    \end{subproof}
    Follows by \auto{group_hom}.
\end{proof}

\begin{theorem}\label{direct_limit_mediator_factorization}
    Suppose $\groupCocone{S}{H}{\kappa}$.
    Let $i\in\carrier[S]$.
    Then
        $\directLimitMediator{S}{H}{\kappa}\circ
            \directSystemCanonicalMap{S}{i}
        =\apply{\kappa}{i}$.
\end{theorem}
\begin{proof}
    Let $u=\directLimitMediator{S}{H}{\kappa}$.
    Let $f=\directSystemCanonicalMap{S}{i}$.
    Let $h=\apply{\kappa}{i}$.
    Let $A=\carrier[\directSystemGroup{S}{i}]$.
    Let $L=\groupDirectLimit{S}$.
    $S$ is a group direct system by
        \obv{group_cocone_system}.
    $\groupHom{f}{\directSystemGroup{S}{i}}{L}$ by
        \auto{direct_system_canonical_map_hom}.
    $\groupHom{u}{L}{H}$ by
        \auto{direct_limit_mediator_hom}.
    $\groupHom{h}{\directSystemGroup{S}{i}}{H}$ by
        \auto{group_cocone_leg_hom}.
    Hence $f\in\funs{A}{\carrier[L]}$ and
        $u\in\funs{\carrier[L]}{\carrier[H]}$ and
        $h\in\funs{A}{\carrier[H]}$ by
        \obv{group_hom_type}.
    Hence $u\circ f\in\funs{A}{\carrier[H]}$ by
        \obv{funs_circ}.
    $u$ is composable with $f$ by
        \auto{funs,funs_ran,subseteq}.
    $u\circ f$ and $h$ are functions and
        $\dom{u\circ f}=A=\dom{h}$ by
        \auto{funs_elim,funs_is_function}.
    Show that for every $x\in\dom{u\circ f}$ we have
        $\apply{u\circ f}{x}=\apply{h}{x}$.
    \begin{subproof}
        Fix $x\in\dom{u\circ f}$.
        Hence $x\in A$.
        $\apply{u\circ f}{x}=\apply{u}{\apply{f}{x}}$ by
            \auto{funs_is_function,funs_elim,circ_apply}.
        $\apply{f}{x}=\directSystemClass{S}{(i,x)}$ by
            \auto{direct_system_canonical_map_apply}.
        $\apply{u}{\directSystemClass{S}{(i,x)}}=\apply{h}{x}$ by
            \auto{direct_limit_mediator_apply}.
    \end{subproof}
    Follows by \auto{funext_on_domain}.
\end{proof}

\begin{theorem}\label{direct_limit_mediator_is_mediator}

    Then $\groupCoconeMediator

\begin{lemma}\label{canonical_cocone_mediator_apply}
    Suppose $\groupCocone{S}{H}{\kappa}$.
    Suppose $\groupCoconeMediator
        {S}{\groupDirectLimit{S}}{\directLimitCanonicalFamily{S}}
        {H}{\kappa}{v}$.
    Let $i\in\carrier[S]$.
    Let $x\in\carrier[\directSystemGroup{S}{i}]$.
    Then $\apply{v}{\directSystemClass{S}{(i,x)}}
        =\apply{\apply{\kappa}{i}}{x}$.
\end{lemma}
\begin{proof}
    Let $f=\apply{\directLimitCanonicalFamily{S}}{i}$.
    $S$ is a group direct system by
        \obv{group_cocone_system}.
    $\groupHom{v}{\groupDirectLimit{S}}{H}$ by
        \obv{group_cocone_mediator_hom}.
    $f=\directSystemCanonicalMap{S}{i}$ by
        \obv{direct_limit_canonical_family_apply}.
    $\groupHom{f}{\directSystemGroup{S}{i}}
        {\groupDirectLimit{S}}$ by
        \auto{direct_system_canonical_map_hom}.
    $v\circ f=\apply{\kappa}{i}$ by
        \auto{group_cocone_mediator_compatibility}.
    $f\in\funs{\carrier[\directSystemGroup{S}{i}]}
        {\carrier[\groupDirectLimit{S}]}$ and
        $v\in\funs{\carrier[\groupDirectLimit{S}]}{\carrier[H]}$ by
        \obv{group_hom_type}.
    Hence $f$ and $v$ are functions and
        $x\in\dom{f}$ and $v$ is composable with $f$ by
        \auto{funs_elim,funs_is_function,funs_ran,subseteq}.
    Hence $\apply{v\circ f}{x}=\apply{v}{\apply{f}{x}}$ by
        \obv{circ_apply}.
    $\apply{f}{x}=\directSystemClass{S}{(i,x)}$ by
        \auto{direct_system_canonical_map_apply}.
\end{proof}

\begin{theorem}\label{direct_limit_mediator_unique}
    Suppose $\groupCocone{S}{H}{\kappa}$.
    Suppose $\groupCoconeMediator
        {S}{\groupDirectLimit{S}}{\directLimitCanonicalFamily{S}}
        {H}{\kappa}{v}$.
    Then $v=\directLimitMediator{S}{H}{\kappa}$.
\end{theorem}
\begin{proof}
    Let $u=\directLimitMediator{S}{H}{\kappa}$.
    Let $L=\groupDirectLimit{S}$.
    $S$ is a group direct system by
        \obv{group_cocone_system}.
    $\groupHom{v}{L}{H}$ by
        \auto{group_cocone_mediator_hom}.
    $\groupHom{u}{L}{H}$ by
        \auto{direct_limit_mediator_hom}.
    Hence $v$ and $u$ are functions and
        $\dom{v}=\carrier[L]=\dom{u}$ by
        \auto{group_hom_type,funs_elim,funs_is_function}.
    Show that for every $C\in\dom{v}$ we have
        $\apply{v}{C}=\apply{u}{C}$.
    \begin{subproof}
        Fix $C\in\dom{v}$.
        Hence $C\in\carrier[L]=\directLimitCarrier{S}$ by
            \auto{group_direct_limit_structure_carrier}.
        Take $i,x$ such that $i\in\carrier[S]$ and
            $x\in\carrier[\directSystemGroup{S}{i}]$ and
            $C=\directSystemClass{S}{(i,x)}$ by
            \obv{direct_limit_carrier_coordinates}.
        $\apply{v}{C}=\apply{\apply{\kappa}{i}}{x}$ by
            \auto{canonical_cocone_mediator_apply}.
        $\apply{u}{C}=\apply{\apply{\kappa}{i}}{x}$ by
            \auto{direct_limit_mediator_apply}.
    \end{subproof}
    Follows by \auto{funext_on_domain}.
\end{proof}

\begin{lemma}\label{direct_limit_mediator_composite_identity}
    Suppose $\directLimitCocone{S}{L}{\iota}$ and
        $\directLimitCocone{S}{M}{\kappa}$.
    Suppose $\groupCoconeMediator{S}{L}{\iota}{M}{\kappa}{u}$ and
        $\groupCoconeMediator{S}{M}{\kappa}{L}{\iota}{v}$.
    Then $v\circ u=\identity{\carrier[L]}$.
\end{lemma}
\begin{proof}
    $\groupCocone{S}{L}{\iota}$ and
        $\groupCocone{S}{M}{\kappa}$ by
        \obv{direct_limit_cocone_is_cocone}.
    $\groupHom{u}{L}{M}$ and $\groupHom{v}{M}{L}$ by
        \obv{group_cocone_mediator_hom}.
    Hence $\groupHom{v\circ u}{L}{L}$ by
        \obv{group_hom_composition}.
    Show that for every $i\in\carrier[S]$ we have
        $(v\circ u)\circ\apply{\iota}{i}=\apply{\iota}{i}$.
    \begin{subproof}
        Fix $i\in\carrier[S]$.
        $u\circ\apply{\iota}{i}=\apply{\kappa}{i}$ and
            $v\circ\apply{\kappa}{i}=\apply{\iota}{i}$ by
            \obv{group_cocone_mediator_compatibility}.
        $(v\circ u)\circ\apply{\iota}{i}
            =v\circ(u\circ\apply{\iota}{i})$ by
            \obv{circ_assoc}.
    \end{subproof}
    Hence $\groupCoconeMediator
        {S}{L}{\iota}{L}{\iota}{v\circ u}$ by
        \obv{group_cocone_mediator}.
    $\groupHom{\identity{\carrier[L]}}{L}{L}$ by
        \obv{identity_group_hom}.
    Show that for every $i\in\carrier[S]$ we have
        $\identity{\carrier[L]}\circ\apply{\iota}{i}
            =\apply{\iota}{i}$.
    \begin{subproof}
        Fix $i\in\carrier[S]$.
        $\groupHom{\apply{\iota}{i}}
            {\directSystemGroup{S}{i}}{L}$ by
            \obv{group_cocone_leg_hom}.
        Hence $\apply{\iota}{i}\in
            \funs{\carrier[\directSystemGroup{S}{i}]}{\carrier[L]}$ by
            \obv{group_hom_type}.
        Follows by \obv{funs_identity_circ_left}.
    \end{subproof}
    Hence $\groupCoconeMediator
        {S}{L}{\iota}{L}{\iota}{\identity{\carrier[L]}}$ by
        \obv{group_cocone_mediator}.
    Follows by \obv{direct_limit_cocone_mediator_unique}.
\end{proof}

\begin{theorem}\label{direct_limit_cocone_comparison}
    Suppose $\directLimitCocone{S}{L}{\iota}$ and
        $\directLimitCocone{S}{M}{\kappa}$.
    Then there exists $u$ such that
        $\groupCoconeIsomorphism{S}{L}{\iota}{M}{\kappa}{u}$ and
        for every $w$ such that
            $\groupCoconeIsomorphism{S}{L}{\iota}{M}{\kappa}{w}$
        we have $w=u$.
\end{theorem}
\begin{proof}
    $\groupCocone{S}{L}{\iota}$ and
        $\groupCocone{S}{M}{\kappa}$ by
        \obv{direct_limit_cocone_is_cocone}.
    Take $u$ such that
        $\groupCoconeMediator{S}{L}{\iota}{M}{\kappa}{u}$ and
        for every $w$ such that
            $\groupCoconeMediator{S}{L}{\iota}{M}{\kappa}{w}$
        we have $w=u$ by
        \obv{direct_limit_cocone_universal}.
    Take $v$ such that
        $\groupCoconeMediator{S}{M}{\kappa}{L}{\iota}{v}$ and
        for every $w$ such that
            $\groupCoconeMediator{S}{M}{\kappa}{L}{\iota}{w}$
        we have $w=v$ by
        \obv{direct_limit_cocone_universal}.
    $\groupHom{u}{L}{M}$ and $\groupHom{v}{M}{L}$ by
        \obv{group_cocone_mediator_hom}.
    Hence $u\in\funs{\carrier[L]}{\carrier[M]}$ and
        $v\in\funs{\carrier[M]}{\carrier[L]}$ by
        \obv{group_hom_type}.
    Hence $u$ and $v$ are functions and
        $\dom{u}=\carrier[L]$ and $\dom{v}=\carrier[M]$ by
        \obv{funs_elim,funs_is_function}.
    $v\circ u=\identity{\carrier[L]}$ and
        $u\circ v=\identity{\carrier[M]}$ by
        \obv{direct_limit_mediator_composite_identity}.
    Show that for every $x\in\carrier[L]$ we have
        $\apply{v}{\apply{u}{x}}=x$.
    \begin{subproof}
        Fix $x\in\carrier[L]$.
        Hence $x\in\dom{u}$.
        $v$ is composable with $u$ by
            \obv{funs_ran,subseteq,funs_elim}.
        $\apply{v\circ u}{x}=\apply{v}{\apply{u}{x}}$ by
            \obv{funs_is_function,circ_apply}.
        $\apply{\identity{\carrier[L]}}{x}=x$ by
            \obv{id_apply}.
    \end{subproof}
    Hence $v$ is a left inverse of $u$ by
        \obv{leftinverse_from_to_iff}.
    Hence $u$ is injective by
        \obv{leftinverse_implies_injection}.
    Show that for every $y\in\carrier[M]$ we have
        $\apply{u}{\apply{v}{y}}=y$.
    \begin{subproof}
        Fix $y\in\carrier[M]$.
        Hence $y\in\dom{v}$.
        $u$ is composable with $v$ by
            \obv{funs_ran,subseteq,funs_elim}.
        $\apply{u\circ v}{y}=\apply{u}{\apply{v}{y}}$ by
            \obv{funs_is_function,circ_apply}.
        $\apply{\identity{\carrier[M]}}{y}=y$ by
            \obv{id_apply}.
    \end{subproof}
    Hence $v$ is a right inverse of $u$ by
        \obv{rightinverse_from_to_iff}.
    Hence $u\in\surj{\carrier[L]}{\carrier[M]}$ by
        \obv{rightinverse_implies_surjection}.
    Hence $u\in\bijections{\carrier[L]}{\carrier[M]}$ by
        \obv{bijections}.
    Hence $\groupIsomorphism{u}{L}{M}$ by
        \obv{group_isomorphism}.
    For every $i\in\carrier[S]$ we have
        $u\circ\apply{\iota}{i}=\apply{\kappa}{i}$ by
        \obv{group_cocone_mediator_compatibility}.
    Hence $\groupCoconeIsomorphism{S}{L}{\iota}{M}{\kappa}{u}$ by
        \obv{group_cocone_isomorphism}.
    Show that for every $w$ such that
        $\groupCoconeIsomorphism{S}{L}{\iota}{M}{\kappa}{w}$
    we have $w=u$.
    \begin{subproof}
        Fix $w$.
        Assume $\groupCoconeIsomorphism{S}{L}{\iota}{M}{\kappa}{w}$.
        Hence $\groupCoconeMediator{S}{L}{\iota}{M}{\kappa}{w}$ by
            \obv{group_cocone_isomorphism_is_mediator}.
    \end{subproof}
    Follows by \auto.
\end{proof}
