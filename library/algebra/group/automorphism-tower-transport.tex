\section{Coherent isomorphisms of tower segments}

A tower-segment isomorphism is a set-coded family of stage isomorphisms with
commuting bond squares.  The endpoints remain explicit exact diagram data; the
family itself is raw morphism data and has no fictitious carrier.

\begin{definition}\label{group_tower_segment_data}
    $\groupTowerSegmentData{T}$ iff
        $\carrier[T]$ is an ordinal and
        $T=\towerSegmentData{\carrier[T]}{\towerStages[T]}{\towerBonds[T]}$ and
        $\towerStages[T]\in
            \funs{\carrier[T]}{\ran{\towerStages[T]}}$ and
        $\towerBonds[T]\in
            \funs{\subseteqrel{\carrier[T]}}{\ran{\towerBonds[T]}}$ and
        for every $i\in\carrier[T]$ we have
            $\towerStage{T}{i}$ is a group and
        for all $\alpha,\beta\in\carrier[T]$ such that
            $\alpha\subseteq\beta$ we have
            $\groupHom{\towerBond{T}{\alpha}{\beta}}
                {\towerStage{T}{\alpha}}{\towerStage{T}{\beta}}$.
\end{definition}

\begin{proposition}\label{tower_segment_diagram_is_group_data}
    Suppose $\towerSegmentDiagram{T}{G}{\theta}$.
    Then $\groupTowerSegmentData{T}$.
\end{proposition}
\begin{proof}
    $\theta$ is an ordinal and $\carrier[T]=\theta$ and
        $T=\towerSegmentData{\theta}{\towerStages[T]}{\towerBonds[T]}$ and
        $\towerStages[T]\in\funs{\theta}{\ran{\towerStages[T]}}$ and
        $\towerBonds[T]\in
            \funs{\subseteqrel{\theta}}{\ran{\towerBonds[T]}}$ by
        \obv{tower_segment_diagram_data}.
    For every $\alpha\in\carrier[T]$ we have
        $\towerStage{T}{\alpha}$ is a group by
        \obv{tower_segment_stage_is_group}.
    For all $\alpha,\beta\in\carrier[T]$ such that
        $\alpha\subseteq\beta$ we have
        $\groupHom{\towerBond{T}{\alpha}{\beta}}
            {\towerStage{T}{\alpha}}{\towerStage{T}{\beta}}$ by
        \obv{tower_segment_bond_is_hom}.
    Follows by \obv{group_tower_segment_data}.
\end{proof}

\begin{proposition}\label{group_tower_segment_data_stage}
    Suppose $\groupTowerSegmentData{T}$.
    Let $\alpha\in\carrier[T]$.
    Then $\towerStage{T}{\alpha}$ is a group.
\end{proposition}
\begin{proof}
    Follows by \obv{group_tower_segment_data}.
\end{proof}

\begin{proposition}\label{group_tower_segment_data_bond}
    Suppose $\groupTowerSegmentData{T}$.
    Let $\alpha,\beta\in\carrier[T]$.
    Suppose $\alpha\subseteq\beta$.
    Then $\groupHom{\towerBond{T}{\alpha}{\beta}}
        {\towerStage{T}{\alpha}}{\towerStage{T}{\beta}}$.
\end{proposition}
\begin{proof}
    Follows by \auto{group_tower_segment_data}.
\end{proof}

\begin{definition}\label{tower_segment_isomorphism}
    $\towerSegmentIsomorphism{E}{T}{U}$ iff
        $\groupTowerSegmentData{T}$ and
        $\groupTowerSegmentData{U}$ and
        $\carrier[T]=\carrier[U]$ and
        $E\in\funs{\carrier[T]}{\ran{E}}$ and
        for every $i\in\carrier[T]$ we have
            $\groupIsomorphism{\apply{E}{i}}
                {\towerStage{T}{i}}{\towerStage{U}{i}}$ and
        for all $\alpha,\beta\in\carrier[T]$ such that
            $\alpha\subseteq\beta$ we have
            $\apply{E}{\beta}\circ\towerBond{T}{\alpha}{\beta}=
                \towerBond{U}{\alpha}{\beta}\circ\apply{E}{\alpha}$.
\end{definition}

\begin{proposition}\label{tower_segment_isomorphism_source}
    Suppose $\towerSegmentIsomorphism{E}{T}{U}$.
    Then $\groupTowerSegmentData{T}$.
\end{proposition}
\begin{proof}
    Follows by \obv{tower_segment_isomorphism}.
\end{proof}

\begin{proposition}\label{tower_segment_isomorphism_target}
    Suppose $\towerSegmentIsomorphism{E}{T}{U}$.
    Then $\groupTowerSegmentData{U}$.
\end{proposition}
\begin{proof}
    Follows by \obv{tower_segment_isomorphism}.
\end{proof}

\begin{proposition}\label{tower_segment_isomorphism_carrier}
    Suppose $\towerSegmentIsomorphism{E}{T}{U}$.
    Then $\carrier[T]=\carrier[U]$.
\end{proposition}
\begin{proof}
    Follows by \obv{tower_segment_isomorphism}.
\end{proof}

\begin{proposition}\label{tower_segment_isomorphism_family_type}
    Suppose $\towerSegmentIsomorphism{E}{T}{U}$.
    Then $E\in\funs{\carrier[T]}{\ran{E}}$.
\end{proposition}
\begin{proof}
    Follows by \obv{tower_segment_isomorphism}.
\end{proof}

\begin{proposition}\label{tower_segment_isomorphism_component}
    Suppose $\towerSegmentIsomorphism{E}{T}{U}$.
    Let $\alpha\in\carrier[T]$.
    Then $\groupIsomorphism{\apply{E}{\alpha}}
        {\towerStage{T}{\alpha}}{\towerStage{U}{\alpha}}$.
\end{proposition}
\begin{proof}
    Follows by \obv{tower_segment_isomorphism}.
\end{proof}

\begin{proposition}\label{tower_segment_isomorphism_square}
    Suppose $\towerSegmentIsomorphism{E}{T}{U}$.
    Let $\alpha,\beta\in\carrier[T]$.
    Suppose $\alpha\subseteq\beta$.
    Then $\apply{E}{\beta}\circ\towerBond{T}{\alpha}{\beta}=
        \towerBond{U}{\alpha}{\beta}\circ\apply{E}{\alpha}$.
\end{proposition}
\begin{proof}
    Follows by \auto{tower_segment_isomorphism}.
\end{proof}

\begin{theorem}\label{tower_segment_isomorphism_extensionality}
    Suppose $\towerSegmentIsomorphism{E}{T}{U}$ and
        $\towerSegmentIsomorphism{D}{T}{U}$.
    Suppose for every $\alpha\in\carrier[T]$ we have
        $\apply{E}{\alpha}=\apply{D}{\alpha}$.
    Then $E=D$.
\end{theorem}
\begin{proof}
    $E\in\funs{\carrier[T]}{\ran{E}}$ and
        $D\in\funs{\carrier[T]}{\ran{D}}$ by
        \obv{tower_segment_isomorphism_family_type}.
    Follows by \obv{typed_functions_equal_from_agreement}.
\end{proof}

\subsection{Identity, composition, and converse}

\begin{definition}\label{tower_segment_identity_isomorphism_family}
    $\towerSegmentIdentityIsomorphismFamily{T}=
        \directSystemIdentityIsomorphismFamily{\towerSegmentSystem{T}}$.
\end{definition}

\begin{proposition}\label{tower_segment_identity_isomorphism_family_type}
    $\towerSegmentIdentityIsomorphismFamily{T}\in
        \funs{\carrier[T]}{
            \ran{\towerSegmentIdentityIsomorphismFamily{T}}}$.
\end{proposition}
\begin{proof}
    $\carrier[\towerSegmentSystem{T}]=\carrier[T]$ by
        \obv{tower_segment_system_carrier}.
    Follows by
        \auto{tower_segment_identity_isomorphism_family,
            direct_system_identity_isomorphism_family_type}.
\end{proof}

\begin{proposition}\label{tower_segment_identity_isomorphism_family_apply}
    Let $\alpha\in\carrier[T]$.
    Then $\apply{\towerSegmentIdentityIsomorphismFamily{T}}{\alpha}=
        \identity{\carrier[\towerStage{T}{\alpha}]}$.
\end{proposition}
\begin{proof}
    $\alpha\in\carrier[\towerSegmentSystem{T}]$ by
        \obv{tower_segment_system_carrier}.
    $\directSystemGroup{\towerSegmentSystem{T}}{\alpha}=
        \towerStage{T}{\alpha}$ by
        \obv{tower_segment_system_group}.
    Follows by
        \auto{tower_segment_identity_isomorphism_family,
            direct_system_identity_isomorphism_family_apply}.
\end{proof}

\begin{theorem}\label{identity_tower_segment_isomorphism}
    Suppose $\groupTowerSegmentData{T}$.
    Then $\towerSegmentIsomorphism
        {\towerSegmentIdentityIsomorphismFamily{T}}{T}{T}$.
\end{theorem}
\begin{proof}
    Let $E=\towerSegmentIdentityIsomorphismFamily{T}$.
    $E\in\funs{\carrier[T]}{\ran{E}}$ by
        \auto{tower_segment_identity_isomorphism_family_type}.
    Show that for every $\alpha\in\carrier[T]$ we have
        $\groupIsomorphism{\apply{E}{\alpha}}
            {\towerStage{T}{\alpha}}{\towerStage{T}{\alpha}}$.
    \begin{subproof}
        Fix $\alpha\in\carrier[T]$.
        $\towerStage{T}{\alpha}$ is a group by
            \obv{group_tower_segment_data}.
        $\apply{E}{\alpha}=\identity{\carrier[\towerStage{T}{\alpha}]}$
            by \auto{tower_segment_identity_isomorphism_family_apply}.
        Follows by \auto{identity_group_isomorphism}.
    \end{subproof}
    Show that for all $\alpha,\beta\in\carrier[T]$ such that
        $\alpha\subseteq\beta$ we have
        $\apply{E}{\beta}\circ\towerBond{T}{\alpha}{\beta}=
            \towerBond{T}{\alpha}{\beta}\circ\apply{E}{\alpha}$.
    \begin{subproof}
        Fix $\alpha,\beta\in\carrier[T]$ such that $\alpha\subseteq\beta$.
        Let $b=\towerBond{T}{\alpha}{\beta}$.
        $\groupHom{b}{\towerStage{T}{\alpha}}{\towerStage{T}{\beta}}$
            by \auto{group_tower_segment_data}.
        Hence $b\in\funs{\carrier[\towerStage{T}{\alpha}]}
            {\carrier[\towerStage{T}{\beta}]}$ by
            \obv{group_hom_type}.
        $\apply{E}{\alpha}=
            \identity{\carrier[\towerStage{T}{\alpha}]}$ and
        $\apply{E}{\beta}=
            \identity{\carrier[\towerStage{T}{\beta}]}$ by
            \auto{tower_segment_identity_isomorphism_family_apply}.
        $\identity{\carrier[\towerStage{T}{\beta}]}\circ b=b$ and
            $b\circ\identity{\carrier[\towerStage{T}{\alpha}]}=b$ by
            \obv{funs_identity_circ_left,funs_circ_identity_right}.
    \end{subproof}
    Follows by \auto{tower_segment_isomorphism}.
\end{proof}

\begin{theorem}\label{tower_segment_isomorphism_composition}
    Suppose $\towerSegmentIsomorphism{E}{T}{U}$ and
        $\towerSegmentIsomorphism{D}{U}{V}$.
    Let $F=\pointwiseCompositionFamily{\carrier[T]}{D}{E}$.
    Then $\towerSegmentIsomorphism{F}{T}{V}$.
\end{theorem}
\begin{proof}
    $\carrier[T]=\carrier[U]=\carrier[V]$ by
        \obv{tower_segment_isomorphism_carrier}.
    $F\in\funs{\carrier[T]}{\ran{F}}$ by
        \auto{pointwise_composition_family_type}.
    Show that for every $\alpha\in\carrier[T]$ we have
        $\groupIsomorphism{\apply{F}{\alpha}}
            {\towerStage{T}{\alpha}}{\towerStage{V}{\alpha}}$.
    \begin{subproof}
        Fix $\alpha\in\carrier[T]$.
        Hence $\alpha\in\carrier[U]$.
        $\apply{F}{\alpha}=\apply{D}{\alpha}\circ\apply{E}{\alpha}$ by
            \auto{pointwise_composition_family_apply}.
        $\groupIsomorphism{\apply{E}{\alpha}}
            {\towerStage{T}{\alpha}}{\towerStage{U}{\alpha}}$ and
        $\groupIsomorphism{\apply{D}{\alpha}}
            {\towerStage{U}{\alpha}}{\towerStage{V}{\alpha}}$ by
            \obv{tower_segment_isomorphism_component}.
        Follows by \auto{group_isomorphism_composition}.
    \end{subproof}
    Show that for all $\alpha,\beta\in\carrier[T]$ such that
        $\alpha\subseteq\beta$ we have
        $\apply{F}{\beta}\circ\towerBond{T}{\alpha}{\beta}=
            \towerBond{V}{\alpha}{\beta}\circ\apply{F}{\alpha}$.
    \begin{subproof}
        Fix $\alpha,\beta\in\carrier[T]$ such that $\alpha\subseteq\beta$.
        Hence $\alpha,\beta\in\carrier[U]$.
        $\apply{E}{\beta}\circ\towerBond{T}{\alpha}{\beta}=
            \towerBond{U}{\alpha}{\beta}\circ\apply{E}{\alpha}$ and
        $\apply{D}{\beta}\circ\towerBond{U}{\alpha}{\beta}=
            \towerBond{V}{\alpha}{\beta}\circ\apply{D}{\alpha}$ by
            \obv{tower_segment_isomorphism_square}.
        $\apply{F}{\alpha}=\apply{D}{\alpha}\circ\apply{E}{\alpha}$ and
        $\apply{F}{\beta}=\apply{D}{\beta}\circ\apply{E}{\beta}$ by
            \auto{pointwise_composition_family_apply}.
        $(\apply{D}{\beta}\circ\apply{E}{\beta})\circ
            \towerBond{T}{\alpha}{\beta}=
            \apply{D}{\beta}\circ
                (\apply{E}{\beta}\circ\towerBond{T}{\alpha}{\beta})$ by
            \obv{circ_assoc}.
        Hence $(\apply{D}{\beta}\circ\apply{E}{\beta})\circ
            \towerBond{T}{\alpha}{\beta}=
            \apply{D}{\beta}\circ
                (\towerBond{U}{\alpha}{\beta}\circ\apply{E}{\alpha})$.
        $\apply{D}{\beta}\circ
                (\towerBond{U}{\alpha}{\beta}\circ\apply{E}{\alpha})=
            (\apply{D}{\beta}\circ\towerBond{U}{\alpha}{\beta})\circ
                \apply{E}{\alpha}$ by \obv{circ_assoc}.
        Hence $\apply{D}{\beta}\circ
                (\towerBond{U}{\alpha}{\beta}\circ\apply{E}{\alpha})=
            (\towerBond{V}{\alpha}{\beta}\circ\apply{D}{\alpha})\circ
                \apply{E}{\alpha}$.
        $(\towerBond{V}{\alpha}{\beta}\circ\apply{D}{\alpha})\circ
                \apply{E}{\alpha}=
            \towerBond{V}{\alpha}{\beta}\circ
                (\apply{D}{\alpha}\circ\apply{E}{\alpha})$ by
            \obv{circ_assoc}.
    \end{subproof}
    $\groupTowerSegmentData{T}$ by
        \obv{tower_segment_isomorphism_source}.
    $\groupTowerSegmentData{V}$ by
        \obv{tower_segment_isomorphism_target}.
    Follows by \obv{tower_segment_isomorphism}.
\end{proof}

\begin{lemma}\label{tower_segment_isomorphism_converse_square}
    Suppose $\towerSegmentIsomorphism{E}{T}{U}$.
    Let $\alpha,\beta\in\carrier[T]$.
    Suppose $\alpha\subseteq\beta$.
    Then $\converse{\apply{E}{\beta}}\circ
            \towerBond{U}{\alpha}{\beta}=
        \towerBond{T}{\alpha}{\beta}\circ
            \converse{\apply{E}{\alpha}}$.
\end{lemma}
\begin{proof}
    $\carrier[T]=\carrier[U]$ by
        \obv{tower_segment_isomorphism_carrier}.
    Hence $\alpha,\beta\in\carrier[U]$.
    $\groupIsomorphism{\apply{E}{\alpha}}
        {\towerStage{T}{\alpha}}{\towerStage{U}{\alpha}}$ by
        \obv{tower_segment_isomorphism_component}.
    $\groupIsomorphism{\apply{E}{\beta}}
        {\towerStage{T}{\beta}}{\towerStage{U}{\beta}}$ by
        \obv{tower_segment_isomorphism_component}.
    $\groupTowerSegmentData{U}$ by
        \obv{tower_segment_isomorphism_target}.
    $\groupHom{\towerBond{U}{\alpha}{\beta}}
        {\towerStage{U}{\alpha}}{\towerStage{U}{\beta}}$ by
        \obv{group_tower_segment_data_bond}.
    $\groupTowerSegmentData{T}$ by
        \obv{tower_segment_isomorphism_source}.
    $\groupHom{\towerBond{T}{\alpha}{\beta}}
        {\towerStage{T}{\alpha}}{\towerStage{T}{\beta}}$ by
        \obv{group_tower_segment_data_bond}.
    $\apply{E}{\beta}\circ\towerBond{T}{\alpha}{\beta}=
        \towerBond{U}{\alpha}{\beta}\circ\apply{E}{\alpha}$ by
        \obv{tower_segment_isomorphism_square}.
    Follows by \obv{group_isomorphism_square_converse}.
\end{proof}

\begin{theorem}\label{tower_segment_isomorphism_converse}
    Suppose $\towerSegmentIsomorphism{E}{T}{U}$.
    Let $C=\pointwiseConverseFamily{\carrier[U]}{E}$.
    Then $\towerSegmentIsomorphism{C}{U}{T}$.
\end{theorem}
\begin{proof}
    $\carrier[T]=\carrier[U]$ by
        \obv{tower_segment_isomorphism_carrier}.
    $C\in\funs{\carrier[U]}{\ran{C}}$ by
        \auto{pointwise_converse_family_type}.
    Show that for every $\alpha\in\carrier[U]$ we have
        $\groupIsomorphism{\apply{C}{\alpha}}
            {\towerStage{U}{\alpha}}{\towerStage{T}{\alpha}}$.
    \begin{subproof}
        Fix $\alpha\in\carrier[U]$.
        Hence $\alpha\in\carrier[T]$.
        $\apply{C}{\alpha}=\converse{\apply{E}{\alpha}}$ by
            \auto{pointwise_converse_family_apply}.
        $\groupIsomorphism{\apply{E}{\alpha}}
            {\towerStage{T}{\alpha}}{\towerStage{U}{\alpha}}$ by
            \obv{tower_segment_isomorphism_component}.
        Follows by \auto{group_isomorphism_converse}.
    \end{subproof}
    Show that for all $\alpha,\beta\in\carrier[U]$ such that
        $\alpha\subseteq\beta$ we have
        $\apply{C}{\beta}\circ\towerBond{U}{\alpha}{\beta}=
            \towerBond{T}{\alpha}{\beta}\circ\apply{C}{\alpha}$.
    \begin{subproof}
        Fix $\alpha,\beta\in\carrier[U]$ such that $\alpha\subseteq\beta$.
        Hence $\alpha,\beta\in\carrier[T]$.
        $\converse{\apply{E}{\beta}}\circ
                \towerBond{U}{\alpha}{\beta}=
            \towerBond{T}{\alpha}{\beta}\circ
                \converse{\apply{E}{\alpha}}$ by
            \obv{tower_segment_isomorphism_converse_square}.
        $\apply{C}{\alpha}=\converse{\apply{E}{\alpha}}$ and
            $\apply{C}{\beta}=\converse{\apply{E}{\beta}}$ by
            \auto{pointwise_converse_family_apply}.
    \end{subproof}
    $\groupTowerSegmentData{U}$ by
        \obv{tower_segment_isomorphism_target}.
    $\groupTowerSegmentData{T}$ by
        \obv{tower_segment_isomorphism_source}.
    Follows by \obv{tower_segment_isomorphism}.
\end{proof}

\subsection{The direct-system bridge}

\begin{theorem}\label{tower_segment_isomorphism_is_system_isomorphism}
    Suppose $\towerSegmentIsomorphism{E}{T}{U}$.
    Suppose $\towerSegmentSystem{T}$ is a group direct system and
        $\towerSegmentSystem{U}$ is a group direct system.
    Then $\groupDirectSystemIsomorphism{E}
        {\towerSegmentSystem{T}}{\towerSegmentSystem{U}}$.
\end{theorem}
\begin{proof}
    $\carrier[\towerSegmentSystem{T}]=\carrier[T]$ and
        $\carrier[\towerSegmentSystem{U}]=\carrier[U]$ by
        \obv{tower_segment_system_carrier}.
    $\carrier[T]=\carrier[U]$ by
        \obv{tower_segment_isomorphism_carrier}.
    $\leq[\towerSegmentSystem{T}]=\subseteqrel{\carrier[T]}$ and
        $\leq[\towerSegmentSystem{U}]=\subseteqrel{\carrier[U]}$ by
        \obv{tower_segment_system_order}.
    $E\in\funs{\carrier[T]}{\ran{E}}$ by
        \obv{tower_segment_isomorphism_family_type}.
    Show that for every $\alpha\in\carrier[T]$ we have
        $\groupIsomorphism{\apply{E}{\alpha}}
            {\directSystemGroup{\towerSegmentSystem{T}}{\alpha}}
            {\directSystemGroup{\towerSegmentSystem{U}}{\alpha}}$.
    \begin{subproof}
        Fix $\alpha\in\carrier[T]$.
        $\groupIsomorphism{\apply{E}{\alpha}}
            {\towerStage{T}{\alpha}}{\towerStage{U}{\alpha}}$ by
            \obv{tower_segment_isomorphism_component}.
        Follows by \obv{tower_segment_system_group}.
    \end{subproof}
    Show that for all $\alpha,\beta\in\carrier[T]$ such that
        $\alpha\mathrel{\leq[\towerSegmentSystem{T}]}\beta$ we have
        $\apply{E}{\beta}\circ
            \directSystemBond{\towerSegmentSystem{T}}{\alpha}{\beta}=
        \directSystemBond{\towerSegmentSystem{U}}{\alpha}{\beta}\circ
            \apply{E}{\alpha}$.
    \begin{subproof}
        Fix $\alpha,\beta\in\carrier[T]$.
        Assume $\alpha\mathrel{\leq[\towerSegmentSystem{T}]}\beta$.
        Hence $\alpha\subseteq\beta$ by
            \auto{tower_segment_system_order,subseteqrel}.
        $\apply{E}{\beta}\circ\towerBond{T}{\alpha}{\beta}=
            \towerBond{U}{\alpha}{\beta}\circ\apply{E}{\alpha}$ by
            \obv{tower_segment_isomorphism_square}.
        Follows by \auto{tower_segment_system_bond}.
    \end{subproof}
    Follows by \obv{group_direct_system_isomorphism}.
\end{proof}
