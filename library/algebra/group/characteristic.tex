\section{Characteristic subgroups}

Characteristicity is invariance under every set-coded automorphism of one
explicit ambient group.  As with normality, the invariant object is a subgroup
carrier, not an induced group record.

\begin{definition}\label{characteristic_subgroup}
    $\characteristicSubgroup{C}{G}$ iff
        $\subgroup{C}{G}$ and
        for every $f\in\automorphismSet{G}$ we have
            $\img{f}{C}=C$.
\end{definition}

\begin{proposition}\label{characteristic_subgroup_is_subgroup}
    Suppose $\characteristicSubgroup{C}{G}$.
    Then $\subgroup{C}{G}$.
\end{proposition}
\begin{proof}
    Follows by \obv{characteristic_subgroup}.
\end{proof}

\begin{corollary}\label{characteristic_subgroup_ambient_group}
    Suppose $\characteristicSubgroup{C}{G}$.
    Then $G$ is a group.
\end{corollary}
\begin{proof}
    Follows by
        \obv{characteristic_subgroup_is_subgroup,subgroup_ambient_group}.
\end{proof}

\begin{corollary}\label{characteristic_subgroup_subseteq}
    Suppose $\characteristicSubgroup{C}{G}$.
    Then $C\subseteq\carrier[G]$.
\end{corollary}
\begin{proof}
    Follows by
        \obv{characteristic_subgroup_is_subgroup,subgroup_subseteq}.
\end{proof}

\begin{proposition}\label{characteristic_subgroup_fixed_image}
    Suppose $\characteristicSubgroup{C}{G}$.
    Let $f\in\automorphismSet{G}$.
    Then $\img{f}{C}=C$.
\end{proposition}
\begin{proof}
    Follows by \obv{characteristic_subgroup}.
\end{proof}

\begin{theorem}\label{characteristic_subgroup_is_normal}
    Suppose $\characteristicSubgroup{C}{G}$.
    Then $\normalSubgroup{C}{G}$.
\end{theorem}
\begin{proof}
    $\subgroup{C}{G}$ by
        \obv{characteristic_subgroup_is_subgroup}.
    Hence $G$ is a group and $C\subseteq\carrier[G]$ by
        \obv{subgroup_ambient_group,subgroup_subseteq}.
    Show that for every $a\in\carrier[G]$ we have
        $\conjugateSubset{G}{a}{C}=C$.
    \begin{subproof}
        Fix $a\in\carrier[G]$.
        $\ConjugationMap{G}{a}\in\automorphismSet{G}$ by
            \obv{inner_automorphism_closed}.
        Hence $\img{\ConjugationMap{G}{a}}{C}=C$ by
            \obv{characteristic_subgroup_fixed_image}.
        $C\inter\carrier[G]=C$ by
            \obv{inter_absorb_supseteq_right}.
        Follows by \obv{conjugate_subset}.
    \end{subproof}
    Follows by \obv{normal_subgroup}.
\end{proof}

\begin{proposition}\label{automorphism_image_carrier}
    Let $f\in\automorphismSet{G}$.
    Then $\img{f}{\carrier[G]}=\carrier[G]$.
\end{proposition}
\begin{proof}
    $f\in\bijections{\carrier[G]}{\carrier[G]}$ by
        \obv{automorphism_set_type}.
    Hence $f\in\surj{\carrier[G]}{\carrier[G]}$ by
        \obv{bijections}.
    Hence $f\in\funs{\carrier[G]}{\carrier[G]}$ by
        \obv{surj_to_fun}.
    Hence $\dom{f}=\carrier[G]$ by \obv{funs_elim}.
    $\ran{f}=\carrier[G]$ by \obv{ran_of_surj}.
    $\img{f}{\dom{f}}=\ran{f}$ by \obv{img_dom}.
\end{proof}

\begin{proposition}\label{automorphism_image_trivial_carrier}
    Let $G$ be a group.
    Let $f\in\automorphismSet{G}$.
    Then
        $\img{f}{\trivialSubgroupCarrier{G}}=
            \trivialSubgroupCarrier{G}$.
\end{proposition}
\begin{proof}
    $\groupHom{f}{G}{G}$ by \obv{automorphism_set_is_hom}.
    Follows by \obv{group_hom_image_trivial_subgroup}.
\end{proof}

\begin{proposition}\label{full_carrier_characteristic_subgroup}
    Let $G$ be a group.
    Then $\characteristicSubgroup{\carrier[G]}{G}$.
\end{proposition}
\begin{proof}
    $\subgroup{\carrier[G]}{G}$ by
        \obv{full_carrier_subgroup}.
    For every $f\in\automorphismSet{G}$ we have
        $\img{f}{\carrier[G]}=\carrier[G]$ by
        \obv{automorphism_image_carrier}.
    Follows by \obv{characteristic_subgroup}.
\end{proof}

\begin{proposition}\label{trivial_carrier_characteristic_subgroup}
    Let $G$ be a group.
    Then $\characteristicSubgroup{\trivialSubgroupCarrier{G}}{G}$.
\end{proposition}
\begin{proof}
    $\subgroup{\trivialSubgroupCarrier{G}}{G}$ by
        \obv{trivial_carrier_subgroup}.
    For every $f\in\automorphismSet{G}$ we have
        $\img{f}{\trivialSubgroupCarrier{G}}=
            \trivialSubgroupCarrier{G}$ by
        \obv{automorphism_image_trivial_carrier}.
    Follows by \obv{characteristic_subgroup}.
\end{proof}

\begin{theorem}\label{group_centre_characteristic_subgroup}
    Let $G$ be a group.
    Then $\characteristicSubgroup{\groupCentre{G}}{G}$.
\end{theorem}
\begin{proof}
    $\subgroup{\groupCentre{G}}{G}$ by
        \obv{group_centre_subgroup}.
    Show that for every $f\in\automorphismSet{G}$ we have
        $\img{f}{\groupCentre{G}}=\groupCentre{G}$.
    \begin{subproof}
        Fix $f\in\automorphismSet{G}$.
        $\groupIsomorphism{f}{G}{G}$ by
            \obv{automorphism_set_iff}.
        Follows by \obv{group_isomorphism_centre_image}.
    \end{subproof}
    Follows by \obv{characteristic_subgroup}.
\end{proof}

\subsection{Intersections}

\begin{theorem}\label{characteristic_subgroup_indexed_intersection}
    Let $G$ be a group.
    Let $F\in\funs{I}{\pow{\carrier[G]}}$.
    Suppose for every $i\in I$ we have
        $\characteristicSubgroup{F(i)}{G}$.
    Then
        $\characteristicSubgroup
            {\indexedIntersection{I}{F}{\carrier[G]}}{G}$.
\end{theorem}
\begin{proof}
    Let $C=\indexedIntersection{I}{F}{\carrier[G]}$.
    For every $i\in I$ we have $\subgroup{F(i)}{G}$ by
        \obv{characteristic_subgroup_is_subgroup}.
    Hence $\subgroup{C}{G}$ by
        \auto{subgroup_indexed_intersection}.
    Show that for every $f\in\automorphismSet{G}$ we have
        $\img{f}{C}=C$.
    \begin{subproof}
        Fix $f\in\automorphismSet{G}$.
        $\groupHom{f}{G}{G}$ by
            \obv{automorphism_set_is_hom}.
        $f\in\bijections{\carrier[G]}{\carrier[G]}$ by
            \obv{automorphism_set_type}.
        Hence $f\in\funs{\carrier[G]}{\carrier[G]}$ by
            \obv{bijections_to_funs}.
        Hence $f$ is a function and $\dom{f}=\carrier[G]$ by
            \auto{funs_elim}.
        Hence $f$ is injective and
            $f\in\surj{\carrier[G]}{\carrier[G]}$ by
            \obv{bijections}.
        $C\subseteq\carrier[G]$ by
            \obv{indexed_intersection_subseteq}.
        Show that $\img{f}{C}\subseteq C$.
        \begin{subproof}
            Show that for every $y\in\img{f}{C}$ we have $y\in C$.
            \begin{subproof}
                Fix $y\in\img{f}{C}$.
                Take $x\in C$ such that $f(x)=y$ by
                    \auto{group_hom_image_iff}.
                Hence $x\in\carrier[G]$ by \obv{subseteq}.
                Hence $y\in\carrier[G]$ by
                    \auto{group_hom_apply_type}.
                Show that for every $i\in I$ we have $y\in F(i)$.
                \begin{subproof}
                    Fix $i\in I$.
                    $x\in F(i)$ by
                        \auto{indexed_intersection_iff}.
                    $F(i)\subseteq\carrier[G]$ by
                        \obv{characteristic_subgroup_subseteq}.
                    Hence $f(x)\in\img{f}{F(i)}$ by
                        \auto{group_hom_image_iff}.
                    $\img{f}{F(i)}=F(i)$ by
                        \auto{characteristic_subgroup_fixed_image}.
                \end{subproof}
                Follows by \obv{indexed_intersection}.
            \end{subproof}
            Follows by \obv{subseteq}.
        \end{subproof}
        Show that $C\subseteq\img{f}{C}$.
        \begin{subproof}
            Show that for every $y\in C$ we have $y\in\img{f}{C}$.
            \begin{subproof}
                Fix $y\in C$.
                Hence $y\in\carrier[G]$ by \obv{subseteq}.
                Take $x\in\carrier[G]$ such that $f(x)=y$ by
                    \obv{surj}.
                Show that for every $i\in I$ we have $x\in F(i)$.
                \begin{subproof}
                    Fix $i\in I$.
                    $y\in F(i)$ by
                        \auto{indexed_intersection_iff}.
                    $\img{f}{F(i)}=F(i)$ by
                        \auto{characteristic_subgroup_fixed_image}.
                    Hence $y\in\img{f}{F(i)}$.
                    $F(i)\subseteq\carrier[G]$ by
                        \obv{characteristic_subgroup_subseteq}.
                    Take $z\in F(i)$ such that $f(z)=y$ by
                        \auto{group_hom_image_iff}.
                    Hence $z\in\carrier[G]$ by \obv{subseteq}.
                    Hence $x,z\in\dom{f}$.
                    $f(x)=f(z)$.
                    Hence $x=z$ by \auto{injective_function}.
                \end{subproof}
                Hence $x\in C$ by
                    \obv{indexed_intersection}.
                Follows by \auto{group_hom_image_iff}.
            \end{subproof}
            Follows by \obv{subseteq}.
        \end{subproof}
        Follows by \obv{subseteq_antisymmetric}.
    \end{subproof}
    Follows by \auto{characteristic_subgroup}.
\end{proof}

\begin{corollary}\label{empty_indexed_intersection_characteristic_subgroup}
    Let $G$ be a group.
    Then
        $\characteristicSubgroup
            {\indexedIntersection{\emptyset}{F}{\carrier[G]}}{G}$.
\end{corollary}
\begin{proof}
    $\indexedIntersection{\emptyset}{F}{\carrier[G]}=
        \carrier[G]$ by
        \obv{empty_indexed_intersection_is_full_carrier}.
    Follows by \auto{full_carrier_characteristic_subgroup}.
\end{proof}

\subsection{Unique invariant descriptions}

\begin{theorem}\label{unique_automorphism_invariant_subgroup_is_characteristic}
    Let $G$ be a group.
    Then
    \[
        \forall P:\Set\to\Prop.\ \forall C:\Set.
        \left(
            \subgroup{C}{G}\land\Apply{P}{C}\land
            \left(\forall f,K:\Set.
                \left(
                    f\in\automorphismSet{G}\land
                    \subgroup{K}{G}\land\Apply{P}{K}
                \right)
                \implies\Apply{P}{\img{f}{K}}
            \right)\land
            \left(\forall K:\Set.
                \left(\subgroup{K}{G}\land\Apply{P}{K}\right)
                \implies K=C
            \right)
        \right)
        \implies\characteristicSubgroup{C}{G}
    \].
\end{theorem}
\begin{proof}
    Fix $P$.
    Fix $C$.
    Assume
    \[
        \subgroup{C}{G}\land\Apply{P}{C}\land
        \left(\forall f,K:\Set.
            \left(
                f\in\automorphismSet{G}\land
                \subgroup{K}{G}\land\Apply{P}{K}
            \right)
            \implies\Apply{P}{\img{f}{K}}
        \right)\land
        \left(\forall K:\Set.
            \left(\subgroup{K}{G}\land\Apply{P}{K}\right)
            \implies K=C
        \right)
    \].
    $\subgroup{C}{G}$.
    $\Apply{P}{C}$.
    Show that for every $f\in\automorphismSet{G}$ we have
        $\img{f}{C}=C$.
    \begin{subproof}
        Fix $f\in\automorphismSet{G}$.
        $\groupIsomorphism{f}{G}{G}$ by
            \obv{automorphism_set_iff}.
        Hence $\subgroup{\img{f}{C}}{G}$ by
            \obv{group_isomorphism_subgroup_image}.
        $\Apply{P}{\img{f}{C}}$.
    \end{subproof}
    Follows by \obv{characteristic_subgroup}.
\end{proof}
