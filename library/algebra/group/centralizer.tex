\section{Centralizers}

The centralizer is totalized by commuting only with the trace of its argument
on the ambient carrier.  Consequently elements of a supplied set outside the
group have no effect.

\begin{definition}\label{group_centralizer}
    $\groupCentralizer{G}{A}=
        \{x\in\carrier[G]\mid
            \text{for every $a\in A\inter\carrier[G]$ we have
                $\commutesIn{G}{x}{a}$}\}$.
\end{definition}

\begin{proposition}\label{group_centralizer_iff}
    $x\in\groupCentralizer{G}{A}$ iff
        $x\in\carrier[G]$ and
        for every $a\in A\inter\carrier[G]$ we have
            $\commutesIn{G}{x}{a}$.
\end{proposition}
\begin{proof}
    Follows by \obv{group_centralizer}.
\end{proof}

\begin{proposition}\label{group_centralizer_subseteq}
    $\groupCentralizer{G}{A}\subseteq\carrier[G]$.
\end{proposition}
\begin{proof}
    Follows by \obv{group_centralizer,subseteq}.
\end{proof}

\begin{proposition}\label{group_centralizer_trace}
    $\groupCentralizer{G}{A}=
        \groupCentralizer{G}{A\inter\carrier[G]}$.
\end{proposition}
\begin{proof}
    $(A\inter\carrier[G])\inter\carrier[G]
        =A\inter\carrier[G]$ by
        \auto{inter_assoc,inter_idempotent}.
    Show that for every $x$ we have
        $x\in\groupCentralizer{G}{A}$ iff
        $x\in\groupCentralizer{G}{A\inter\carrier[G]}$.
    \begin{subproof}
        Fix $x$.
        Follows by \auto{group_centralizer_iff}.
    \end{subproof}
    Follows by \obv{setext}.
\end{proof}

\begin{theorem}\label{group_centralizer_antitone}
    Let $G$ be a group.
    Suppose $A\subseteq B$.
    Then
        $\groupCentralizer{G}{B}\subseteq
         \groupCentralizer{G}{A}$.
\end{theorem}
\begin{proof}
    Show that for every $x\in\groupCentralizer{G}{B}$ we have
        $x\in\groupCentralizer{G}{A}$.
    \begin{subproof}
        Fix $x\in\groupCentralizer{G}{B}$.
        $x\in\carrier[G]$ by \obv{group_centralizer}.
        Show that for every $a\in A\inter\carrier[G]$ we have
            $\commutesIn{G}{x}{a}$.
        \begin{subproof}
            Fix $a\in A\inter\carrier[G]$.
            $a\in B\inter\carrier[G]$ by
                \obv{inter_intro,inter_elim_left,inter_elim_right,subseteq}.
            Follows by \obv{group_centralizer}.
        \end{subproof}
        Follows by \obv{group_centralizer}.
    \end{subproof}
    Follows by \obv{subseteq}.
\end{proof}

\begin{theorem}\label{group_centralizer_union}
    Let $G$ be a group.
    Then
        $\groupCentralizer{G}{A\union B}=
         \groupCentralizer{G}{A}\inter
         \groupCentralizer{G}{B}$.
\end{theorem}
\begin{proof}
    $A\subseteq A\union B$ and $B\subseteq A\union B$ by
        \obv{union_upper_left,union_upper_right}.
    Hence
        $\groupCentralizer{G}{A\union B}\subseteq
            \groupCentralizer{G}{A}$ and
        $\groupCentralizer{G}{A\union B}\subseteq
            \groupCentralizer{G}{B}$ by
        \obv{group_centralizer_antitone}.
    Hence
        $\groupCentralizer{G}{A\union B}\subseteq
            \groupCentralizer{G}{A}\inter
            \groupCentralizer{G}{B}$ by
        \obv{subseteq,inter_intro}.
    Show that
        $\groupCentralizer{G}{A}\inter
            \groupCentralizer{G}{B}\subseteq
        \groupCentralizer{G}{A\union B}$.
    \begin{subproof}
        Show that for every
            $x\in\groupCentralizer{G}{A}\inter
                \groupCentralizer{G}{B}$ we have
            $x\in\groupCentralizer{G}{A\union B}$.
        \begin{subproof}
            Fix
                $x\in\groupCentralizer{G}{A}\inter
                    \groupCentralizer{G}{B}$.
            $x\in\carrier[G]$ by
                \auto{inter_elim_left,group_centralizer_iff}.
            Show that for every
                $a\in(A\union B)\inter\carrier[G]$ we have
                $\commutesIn{G}{x}{a}$.
            \begin{subproof}
                Fix $a\in(A\union B)\inter\carrier[G]$.
                Follows by
                    \auto{group_centralizer_iff,inter_intro,
                        inter_elim_left,inter_elim_right,union_iff}.
            \end{subproof}
            Follows by \obv{group_centralizer}.
        \end{subproof}
        Follows by \obv{subseteq}.
    \end{subproof}
    Follows by \obv{subseteq_antisymmetric}.
\end{proof}

\begin{theorem}\label{group_centralizer_subgroup}
    Let $G$ be a group.
    Then $\subgroup{\groupCentralizer{G}{A}}{G}$.
\end{theorem}
\begin{proof}
    Let $C=\groupCentralizer{G}{A}$.
    $C\subseteq\carrier[G]$ by
        \auto{group_centralizer_subseteq}.
    $\neutral[G]\in\carrier[G]$ by
        \obv{group_is_monoid,monoid_is_unital_magma,unitalmagma_type}.
    Show that for every $a\in A\inter\carrier[G]$ we have
        $\commutesIn{G}{\neutral[G]}{a}$.
    \begin{subproof}
        Fix $a\in A\inter\carrier[G]$.
        Hence $a\in\carrier[G]$ by \obv{inter_elim_right}.
        Follows by
            \obv{commutes_in,group_is_monoid,monoid_is_unital_magma,unitalmagma_left,unitalmagma_right}.
    \end{subproof}
    Hence $\neutral[G]\in C$ by
        \obv{group_centralizer}.
    Show that for all $x,y\in C$ we have $\mul[G](x,y)\in C$.
    \begin{subproof}
        Fix $x,y\in C$.
        Hence $x,y\in\carrier[G]$ by \obv{subseteq}.
        $\mul[G](x,y)\in\carrier[G]$ by
            \obv{group_is_monoid,monoid_is_magma,magma_welldef}.
        Show that for every $a\in A\inter\carrier[G]$ we have
            $\commutesIn{G}{\mul[G](x,y)}{a}$.
        \begin{subproof}
            Fix $a\in A\inter\carrier[G]$.
            Hence $a\in\carrier[G]$ by \obv{inter_elim_right}.
            $\commutesIn{G}{x}{a}$ and $\commutesIn{G}{y}{a}$ by
                \auto{group_centralizer_iff}.
            Hence $\conjugate{G}{x}{a}=a$ and
                $\conjugate{G}{y}{a}=a$ by
                \obv{commutes_in_iff_conjugate_fixed}.
            $\conjugate{G}{x}{\conjugate{G}{y}{a}}=
                \conjugate{G}{\mul[G](x,y)}{a}$ by
                \obv{group_conjugate_composition}.
            Hence $\conjugate{G}{\mul[G](x,y)}{a}=a$.
            Follows by \obv{commutes_in_iff_conjugate_fixed}.
        \end{subproof}
        Follows by \obv{group_centralizer}.
    \end{subproof}
    Show that for every $x\in C$ we have $\groupInv[G](x)\in C$.
    \begin{subproof}
        Fix $x\in C$.
        Hence $x\in\carrier[G]$ by \obv{subseteq}.
        $\groupInv[G](x)\in\carrier[G]$ by
            \obv{group_inverse_type}.
        Show that for every $a\in A\inter\carrier[G]$ we have
            $\commutesIn{G}{\groupInv[G](x)}{a}$.
        \begin{subproof}
            Fix $a\in A\inter\carrier[G]$.
            Hence $a\in\carrier[G]$ by \obv{inter_elim_right}.
            $\commutesIn{G}{x}{a}$ by
                \auto{group_centralizer_iff}.
            Hence $\conjugate{G}{x}{a}=a$ by
                \obv{commutes_in_iff_conjugate_fixed}.
            $\conjugate{G}{\groupInv[G](x)}{
                \conjugate{G}{x}{a}}=a$ by
                \obv{group_conjugate_inverse_left}.
            Hence $\conjugate{G}{\groupInv[G](x)}{a}=a$.
            Follows by \obv{commutes_in_iff_conjugate_fixed}.
        \end{subproof}
        Follows by \obv{group_centralizer}.
    \end{subproof}
    Follows by \obv{subgroup}.
\end{proof}

\begin{proposition}\label{centralizer_of_emptyset}
    Let $G$ be a group.
    Then $\groupCentralizer{G}{\emptyset}=\carrier[G]$.
\end{proposition}
\begin{proof}
    $\emptyset\inter\carrier[G]=\emptyset$ by
        \auto{inter_comm,inter_emptyset}.
    Follows by \auto{group_centralizer,setext,emptyset}.
\end{proof}
