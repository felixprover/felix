\section{Centreless embedding limits}

The argument is ambient-free at the system level: a central limit element is
represented in one component, and the injective canonical map reflects its
commutation equations back to that component.

\begin{theorem}\label{embedding_direct_limit_centreless}
    Suppose $\embeddingGroupDirectSystem{S}$.
    Suppose for every $i\in\carrier[S]$ we have
        $\centrelessGroup{\directSystemGroup{S}{i}}$.
    Then $\centrelessGroup{\groupDirectLimit{S}}$.
\end{theorem}
\begin{proof}
    $S$ is a group direct system by
        \obv{embedding_group_direct_system_is_system}.
    Let $L=\groupDirectLimit{S}$.
    $L$ is a group by \obv{group_direct_limit_is_group}.
    Show that $\groupCentre{L}\subseteq\trivialSubgroupCarrier{L}$.
    \begin{subproof}
        Show that for every $z\in\groupCentre{L}$ we have
            $z\in\trivialSubgroupCarrier{L}$.
        \begin{subproof}
            Fix $z\in\groupCentre{L}$.
            Hence $z\in\carrier[L]$ by
                \auto{group_centre_subseteq,subseteq}.
            Take $i\in\carrier[S]$ such that
                $z\in\directLimitImage{S}{i}$ by
                \obv{direct_limit_image_coverage}.
            Let $G=\directSystemGroup{S}{i}$.
            Let $f=\directSystemCanonicalMap{S}{i}$.
            $\groupEmbedding{f}{G}{L}$ by
                \auto{embedding_system_canonical_map_embedding}.
            Hence $\groupHom{f}{G}{L}$ by
                \obv{group_embedding_is_hom}.
            Hence $G$ is a group by
                \obv{group_hom_source_group}.
            Take $x\in\carrier[G]$ such that $\apply{f}{x}=z$ by
                \auto{direct_limit_image,group_hom_image_iff,subseteq_refl}.
            Show that $x\in\groupCentre{G}$.
            \begin{subproof}
                Show that for every $y\in\carrier[G]$ we have
                    $\commutesIn{G}{x}{y}$.
                \begin{subproof}
                    Fix $y\in\carrier[G]$.
                    $\apply{f}{y}\in\carrier[L]$ by
                        \obv{group_hom_apply_type}.
                    Hence $\commutesIn{L}{z}{\apply{f}{y}}$ by
                        \auto{group_centre_iff}.
                    Hence $\mul[L](z,\apply{f}{y})
                        =\mul[L](\apply{f}{y},z)$ by
                        \obv{commutes_in}.
                    Let $a=\mul[G](x,y)$.
                    Let $b=\mul[G](y,x)$.
                    $a,b\in\carrier[G]$ by
                        \auto{group_is_monoid,monoid_is_magma,magma_welldef}.
                    $\apply{f}{a}=\mul[L]
                        (\apply{f}{x},\apply{f}{y})$ and
                        $\apply{f}{b}=\mul[L]
                        (\apply{f}{y},\apply{f}{x})$ by
                        \auto{group_hom_mul}.
                    Hence $\apply{f}{a}=\apply{f}{b}$.
                    $f$ is a function and
                        $\dom{f}=\carrier[G]$ by
                        \auto{group_hom_type,funs_elim,funs_is_function}.
                    $f$ is injective by
                        \obv{group_embedding_injective}.
                    Hence $a=b$ by
                        \auto{injective_function}.
                    Follows by \obv{commutes_in}.
                \end{subproof}
                Follows by \obv{group_centre_iff}.
            \end{subproof}
            $\centrelessGroup{G}$.
            Hence $\groupCentre{G}=\trivialSubgroupCarrier{G}$ by
                \obv{centreless_group_centre}.
            Hence $x=\neutral[G]$ by
                \auto{trivial_subgroup_carrier,singleton_iff}.
            $\apply{f}{\neutral[G]}=\neutral[L]$ by
                \obv{group_hom_neutral}.
            Hence $z=\neutral[L]$.
            Follows by
                \obv{trivial_subgroup_carrier,singleton_iff}.
        \end{subproof}
        Follows by \obv{subseteq}.
    \end{subproof}
    $\subgroup{\groupCentre{L}}{L}$ by
        \obv{group_centre_subgroup}.
    Hence $\neutral[L]\in\groupCentre{L}$ by
        \obv{subgroup_neutral}.
    Hence $\trivialSubgroupCarrier{L}\subseteq\groupCentre{L}$ by
        \auto{trivial_subgroup_carrier,singleton_subset_intro}.
    Hence $\groupCentre{L}=\trivialSubgroupCarrier{L}$ by
        \obv{subseteq_antisymmetric}.
    Follows by \obv{centreless_group}.
\end{proof}
