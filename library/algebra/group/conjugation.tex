\section{Conjugation as an automorphism and an action}

The element-level conjugation operation and its unary graph are defined in
\obv{group_conjugate,conjugation_map}.  This chapter supplies their algebraic
laws and graphifies the two-argument operation once to obtain the conjugation
action.  We continue to use the convention $x\mapsto axa^{-1}$.

\begin{theorem}\label{group_conjugate_mul}
    Let $G$ be a group.
    Let $a,x,y\in\carrier[G]$.
    Then
        $\conjugate{G}{a}{\mul[G](x,y)}=
            \mul[G](\conjugate{G}{a}{x},\conjugate{G}{a}{y})$.
\end{theorem}
\begin{proof}
    Let $u=\groupInv[G](a)$.
    $u\in\carrier[G]$ by
        \auto{group_inverse_type}.
    $\mul[G](x,y),\mul[G](a,x),\mul[G](a,y)\in\carrier[G]$ by
        \obv{group_is_monoid,monoid_is_magma,magma_welldef}.
    $\mul[G](\groupInv[G](a),a)=\neutral[G]$ by
        \obv{group_inverse_left}.
    Hence $\mul[G](u,\mul[G](a,y))=y$ by
        \auto{group_is_monoid,monoid_assoc,monoid_is_unital_magma,unitalmagma_left}.
    $\mul[G](\mul[G](a,x),u)\in\carrier[G]$ by
        \obv{group_is_monoid,monoid_is_magma,magma_welldef}.
    Hence
        $\mul[G](\mul[G](\mul[G](a,x),u),\mul[G](a,y))=
            \mul[G](\mul[G](a,x),y)$ by
        \auto{group_is_monoid,monoid_assoc}.
    Hence
        $\mul[G](
            \mul[G](\mul[G](a,x),u),
            \mul[G](\mul[G](a,y),u))=
         \mul[G](\mul[G](\mul[G](a,x),y),u)$ by
        \auto{group_is_monoid,monoid_assoc}.
    $\mul[G](\mul[G](\mul[G](a,x),y),u)=
        \mul[G](\mul[G](a,\mul[G](x,y)),u)$ by
        \obv{group_is_monoid,monoid_assoc}.
    Follows by \auto{group_conjugate}.
\end{proof}

\begin{theorem}\label{conjugation_map_group_hom}
    Let $G$ be a group.
    Let $a\in\carrier[G]$.
    Then $\groupHom{\ConjugationMap{G}{a}}{G}{G}$.
\end{theorem}
\begin{proof}
    Let $f=\ConjugationMap{G}{a}$.
    $f\in\funs{\carrier[G]}{\carrier[G]}$ by
        \auto{conjugation_map_function}.
    Show that for all $x,y\in\carrier[G]$ we have
        $f(\mul[G](x,y))=\mul[G](f(x),f(y))$.
    \begin{subproof}
        Fix $x,y\in\carrier[G]$.
        $\mul[G](x,y)\in\carrier[G]$ by
            \obv{group_is_monoid,monoid_is_magma,magma_welldef}.
        \begin{align*}
            f(\mul[G](x,y))
                &= \conjugate{G}{a}{\mul[G](x,y)}
                    \explanation{by \auto{conjugation_map_apply}}
                \\
                &= \mul[G](\conjugate{G}{a}{x},\conjugate{G}{a}{y})
                    \explanation{by \obv{group_conjugate_mul}}
                \\
                &= \mul[G](f(x),f(y))
                    \explanation{by \auto{conjugation_map_apply}}
        \end{align*}
    \end{subproof}
    Follows by \auto{group_hom}.
\end{proof}

\begin{theorem}\label{conjugation_map_group_isomorphism}
    Let $G$ be a group.
    Let $a\in\carrier[G]$.
    Then $\groupIsomorphism{\ConjugationMap{G}{a}}{G}{G}$.
\end{theorem}
\begin{proof}
    $\groupHom{\ConjugationMap{G}{a}}{G}{G}$ by
        \obv{conjugation_map_group_hom}.
    $\ConjugationMap{G}{a}\in
        \bijections{\carrier[G]}{\carrier[G]}$ by
        \obv{conjugation_map_bijection}.
    Follows by \obv{group_isomorphism}.
\end{proof}

\begin{proposition}\label{group_conjugate_neutral}
    Let $G$ be a group.
    Let $a\in\carrier[G]$.
    Then $\conjugate{G}{a}{\neutral[G]}=\neutral[G]$.
\end{proposition}
\begin{proof}
    $\groupHom{\ConjugationMap{G}{a}}{G}{G}$ by
        \obv{conjugation_map_group_hom}.
    Hence
        $\apply{\ConjugationMap{G}{a}}{\neutral[G]}=\neutral[G]$ by
        \obv{group_hom_neutral}.
    Follows by
        \obv{group_is_monoid,monoid_is_unital_magma,unitalmagma_type,conjugation_map_apply}.
\end{proof}

\begin{theorem}\label{commutes_in_iff_conjugate_fixed}
    Let $G$ be a group.
    Let $x,y\in\carrier[G]$.
    Then $\commutesIn{G}{x}{y}$ iff
        $\conjugate{G}{x}{y}=y$.
\end{theorem}
\begin{proof}
    Let $u=\groupInv[G](x)$.
    $u\in\carrier[G]$ by \auto{group_inverse_type}.
    Show that if $\commutesIn{G}{x}{y}$, then
        $\conjugate{G}{x}{y}=y$.
    \begin{subproof}
        Assume $\commutesIn{G}{x}{y}$.
        Hence $\mul[G](x,y)=\mul[G](y,x)$ by
            \obv{commutes_in}.
        $\mul[G](x,u)=\neutral[G]$ by
            \auto{group_inverse_right}.
        Hence $\mul[G](\mul[G](y,x),u)=y$ by
            \auto{group_is_monoid,monoid_assoc,monoid_is_unital_magma,unitalmagma_right}.
        Hence $\mul[G](\mul[G](x,y),u)=y$.
        Follows by \obv{group_conjugate}.
    \end{subproof}
    Show that if $\conjugate{G}{x}{y}=y$, then
        $\commutesIn{G}{x}{y}$.
    \begin{subproof}
        Assume $\conjugate{G}{x}{y}=y$.
        Hence $\mul[G](\mul[G](x,y),u)=y$ by
            \auto{group_conjugate}.
        $\mul[G](u,x)=\neutral[G]$ by
            \auto{group_inverse_left}.
        $\mul[G](x,y),\mul[G](\mul[G](x,y),u)\in\carrier[G]$ by
            \obv{group_is_monoid,monoid_is_magma,magma_welldef}.
        Hence
            $\mul[G](\mul[G](\mul[G](x,y),u),x)=\mul[G](y,x)$.
        $\mul[G](\mul[G](\mul[G](x,y),u),x)=\mul[G](x,y)$ by
            \auto{group_is_monoid,monoid_assoc,monoid_is_unital_magma,unitalmagma_right}.
        Hence $\mul[G](x,y)=\mul[G](y,x)$.
        Follows by \obv{commutes_in}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{theorem}\label{conjugation_map_composition}
    Let $G$ be a group.
    Let $a,b\in\carrier[G]$.
    Then
        $\ConjugationMap{G}{\mul[G](a,b)}=
            \ConjugationMap{G}{a}\circ\ConjugationMap{G}{b}$.
\end{theorem}
\begin{proof}
    Let $f=\ConjugationMap{G}{\mul[G](a,b)}$.
    Let $g=\ConjugationMap{G}{a}\circ\ConjugationMap{G}{b}$.
    $\mul[G](a,b)\in\carrier[G]$ by
        \obv{group_is_monoid,monoid_is_magma,magma_welldef}.
    Hence $\groupHom{f}{G}{G}$ by
        \auto{conjugation_map_group_hom}.
    $\groupHom{\ConjugationMap{G}{a}}{G}{G}$ and
        $\groupHom{\ConjugationMap{G}{b}}{G}{G}$ by
        \obv{conjugation_map_group_hom}.
    Hence $\ConjugationMap{G}{a}\in
        \funs{\carrier[G]}{\carrier[G]}$ and
        $\ConjugationMap{G}{b}\in
        \funs{\carrier[G]}{\carrier[G]}$ by
        \obv{group_hom_type}.
    $\ConjugationMap{G}{a}$ is a function by
        \obv{funs_is_function}.
    $\ConjugationMap{G}{b}$ is a function and
        $\dom{\ConjugationMap{G}{b}}=\carrier[G]$ by
        \auto{funs_elim}.
    Hence $\ConjugationMap{G}{a}$ is composable with
        $\ConjugationMap{G}{b}$ by
        \auto{funs_ran,funs}.
    Hence $\groupHom{g}{G}{G}$ by
        \auto{group_hom_composition}.
    Show that for every $x\in\carrier[G]$ we have $f(x)=g(x)$.
    \begin{subproof}
        Fix $x\in\carrier[G]$.
        $\conjugate{G}{b}{x}\in\carrier[G]$ by
            \obv{group_conjugate_type}.
        $(\ConjugationMap{G}{a}\circ\ConjugationMap{G}{b})(x)=
            \apply{\ConjugationMap{G}{a}}{
            \apply{\ConjugationMap{G}{b}}{x}}$ by
            \obv{circ_apply}.
        Hence $g(x)=\conjugate{G}{a}{\conjugate{G}{b}{x}}$ by
            \auto{conjugation_map_apply}.
        $f(x)=\conjugate{G}{\mul[G](a,b)}{x}$ by
            \auto{conjugation_map_apply}.
        Follows by \obv{group_conjugate_composition}.
    \end{subproof}
    Follows by \auto{group_hom_extensionality}.
\end{proof}

\begin{definition}\label{conjugation_action_op}
    $\conjugationActionOp:\Set\to\Set\to\Set\to\Set
        =\fun{G,a,x:\Set}{
            \conjugate{G}{a}{x}}$.
\end{definition}

\begin{definition}\label{conjugation_action}
    $\ConjugationAction{G}=
        \Apply{\heteroBinaryGraphOnOp}
            {\carrier[G],\carrier[G],\Apply{\conjugationActionOp}{G}}$.
\end{definition}

\begin{proposition}\label{conjugation_action_type}
    Let $G$ be a group.
    Then $\ConjugationAction{G}\in
        \funs{\carrier[G]\times\carrier[G]}{\carrier[G]}$.
\end{proposition}
\begin{proof}
    Show that $\forall a,x:\Set.
        (a\in\carrier[G]\land x\in\carrier[G])\implies
        \Apply{\conjugationActionOp}{G,a,x}
            \in\carrier[G]$.
    \begin{subproof}
        Fix $a,x:\Set$.
        Assume $a,x\in\carrier[G]$.
        Follows by
            \auto{conjugation_action_op,group_conjugate_type}.
    \end{subproof}
    Follows by
        \auto{conjugation_action,logical_heterogeneous_binary_graph_on_type}.
\end{proof}

\begin{proposition}\label{conjugation_action_apply}
    Let $G$ be a group.
    Let $a,x\in\carrier[G]$.
    Then $\actionApply{\ConjugationAction{G}}{a}{x}
        =\conjugate{G}{a}{x}$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{conjugation_action,conjugation_action_op,logical_heterogeneous_binary_graph_on_apply}.
\end{proof}

\begin{theorem}\label{conjugation_is_group_action}
    Let $G$ be a group.
    Then $\groupAction{\ConjugationAction{G}}{G}{\carrier[G]}$.
\end{theorem}
\begin{proof}
    $\ConjugationAction{G}\in
        \funs{\carrier[G]\times\carrier[G]}{\carrier[G]}$ by
        \obv{conjugation_action_type}.
    $\neutral[G]\in\carrier[G]$ by
        \obv{group_is_monoid,monoid_is_unital_magma,unitalmagma_type}.
    For every $x\in\carrier[G]$ we have
        $\actionApply{\ConjugationAction{G}}{\neutral[G]}{x}=x$ by
        \auto{conjugation_action_apply,group_conjugate_by_neutral}.
    Show that for all $a,b\in\carrier[G]$ we have for every
        $x\in\carrier[G]$ we have
        $\actionApply{\ConjugationAction{G}}{\mul[G](a,b)}{x}=
            \actionApply{\ConjugationAction{G}}{a}{
                \actionApply{\ConjugationAction{G}}{b}{x}}$.
    \begin{subproof}
        Fix $a,b\in\carrier[G]$.
        Fix $x\in\carrier[G]$.
        $\mul[G](a,b)\in\carrier[G]$ by
            \obv{group_is_monoid,monoid_is_magma,magma_welldef}.
        $\conjugate{G}{b}{x}\in\carrier[G]$ by
            \obv{group_conjugate_type}.
        Follows by
            \auto{conjugation_action_apply,group_conjugate_composition}.
    \end{subproof}
    Hence $\monoidAction{\ConjugationAction{G}}{G}{\carrier[G]}$ by
        \obv{monoid_action}.
    Follows by \obv{group_action}.
\end{proof}

\subsection{Conjugating subsets}

Only the trace of a set on the ambient carrier is conjugated.  Thus this is a
total set constructor without assigning group-theoretic meaning to elements
outside the carrier.

\begin{definition}\label{conjugate_subset}
    $\conjugateSubset{G}{a}{A}=
        \img{\ConjugationMap{G}{a}}{A\inter\carrier[G]}$.
\end{definition}

\begin{proposition}\label{conjugate_subset_iff}
    Let $G$ be a group.
    Let $a\in\carrier[G]$.
    Then $y\in\conjugateSubset{G}{a}{A}$ iff
        there exists $x\in A\inter\carrier[G]$ such that
            $y=\conjugate{G}{a}{x}$.
\end{proposition}
\begin{proof}
    Show that if $y\in\conjugateSubset{G}{a}{A}$, then
        there exists $x\in A\inter\carrier[G]$ such that
            $y=\conjugate{G}{a}{x}$.
    \begin{subproof}
        Assume $y\in\conjugateSubset{G}{a}{A}$.
        Take $x\in A\inter\carrier[G]$ such that
            $(x,y)\in\ConjugationMap{G}{a}$ by
            \auto{conjugate_subset,img_iff}.
        $y=\conjugate{G}{a}{x}$ by
            \auto{conjugation_map_iff}.
    \end{subproof}
    Show that if there exists $x\in A\inter\carrier[G]$ such that
        $y=\conjugate{G}{a}{x}$, then
        $y\in\conjugateSubset{G}{a}{A}$.
    \begin{subproof}
        Assume that there exists $x\in A\inter\carrier[G]$ such that
            $y=\conjugate{G}{a}{x}$.
        Take $x\in A\inter\carrier[G]$ such that
            $y=\conjugate{G}{a}{x}$.
        $x\in\carrier[G]$ by \obv{inter_elim_right}.
        $(x,y)\in\ConjugationMap{G}{a}$ by
            \obv{conjugation_map_iff}.
        Follows by \auto{conjugate_subset,img_iff}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{proposition}\label{conjugate_subset_subseteq_carrier}
    Let $G$ be a group.
    Let $a\in\carrier[G]$.
    Then $\conjugateSubset{G}{a}{A}\subseteq\carrier[G]$.
\end{proposition}
\begin{proof}
    Show that for every $y\in\conjugateSubset{G}{a}{A}$ we have
        $y\in\carrier[G]$.
    \begin{subproof}
        Fix $y\in\conjugateSubset{G}{a}{A}$.
        Take $x\in A\inter\carrier[G]$ such that
            $y=\conjugate{G}{a}{x}$ by
            \auto{conjugate_subset_iff}.
        $x\in\carrier[G]$ by \obv{inter_elim_right}.
        Follows by \auto{group_conjugate_type}.
    \end{subproof}
    Follows by \obv{subseteq}.
\end{proof}

\begin{proposition}\label{conjugate_full_carrier}
    Let $G$ be a group.
    Let $a\in\carrier[G]$.
    Then $\conjugateSubset{G}{a}{\carrier[G]}=\carrier[G]$.
\end{proposition}
\begin{proof}
    Let $f=\ConjugationMap{G}{a}$.
    $f\in\bijections{\carrier[G]}{\carrier[G]}$ by
        \auto{conjugation_map_bijection}.
    Hence $f\in\surj{\carrier[G]}{\carrier[G]}$ by
        \obv{bijections}.
    Hence $f\in\funs{\carrier[G]}{\carrier[G]}$ by
        \obv{surj_to_fun}.
    Hence $\dom{f}=\carrier[G]$ by
        \obv{funs_elim}.
    $\ran{f}=\carrier[G]$ by \obv{ran_of_surj}.
    $\carrier[G]\inter\carrier[G]=\carrier[G]$ by
        \obv{inter_idempotent}.
    $\img{f}{\carrier[G]}=\ran{f}$ by
        \auto{img_dom}.
    Follows by \obv{conjugate_subset}.
\end{proof}

\begin{proposition}\label{conjugation_map_neutral}
    Let $G$ be a group.
    Then
        $\ConjugationMap{G}{\neutral[G]}=
            \identity{\carrier[G]}$.
\end{proposition}
\begin{proof}
    $\neutral[G]\in\carrier[G]$ by
        \obv{group_is_monoid,monoid_is_unital_magma,unitalmagma_type}.
    $\groupHom{\ConjugationMap{G}{\neutral[G]}}{G}{G}$ by
        \obv{conjugation_map_group_hom}.
    $\groupHom{\identity{\carrier[G]}}{G}{G}$ by
        \obv{identity_group_hom}.
    Show that for every $x\in\carrier[G]$ we have
        $\apply{\ConjugationMap{G}{\neutral[G]}}{x}=
            \apply{\identity{\carrier[G]}}{x}$.
    \begin{subproof}
        Fix $x\in\carrier[G]$.
        $\apply{\ConjugationMap{G}{\neutral[G]}}{x}=
            \conjugate{G}{\neutral[G]}{x}$ by
            \obv{conjugation_map_apply}.
        $\conjugate{G}{\neutral[G]}{x}=x$ by
            \obv{group_conjugate_by_neutral}.
        $\apply{\identity{\carrier[G]}}{x}=x$ by
            \obv{id_apply}.
    \end{subproof}
    Follows by \obv{group_hom_extensionality}.
\end{proof}

\begin{proposition}\label{conjugate_subset_by_neutral}
    Let $G$ be a group.
    Suppose $A\subseteq\carrier[G]$.
    Then $\conjugateSubset{G}{\neutral[G]}{A}=A$.
\end{proposition}
\begin{proof}
    $A\inter\carrier[G]=A$ by
        \obv{inter_absorb_supseteq_right}.
    $\ConjugationMap{G}{\neutral[G]}=
        \identity{\carrier[G]}$ by
        \obv{conjugation_map_neutral}.
    $\img{\identity{\carrier[G]}}{A}=
        \carrier[G]\inter A$ by \obv{id_img}.
    $\carrier[G]\inter A=A$ by
        \obv{inter_absorb_supseteq_left}.
    Follows by \obv{conjugate_subset}.
\end{proof}

\begin{theorem}\label{conjugate_subset_composition}
    Let $G$ be a group.
    Let $a,b\in\carrier[G]$.
    Suppose $A\subseteq\carrier[G]$.
    Then
        $\conjugateSubset{G}{\mul[G](a,b)}{A}=
            \conjugateSubset{G}{a}{\conjugateSubset{G}{b}{A}}$.
\end{theorem}
\begin{proof}
    Let $f=\ConjugationMap{G}{b}$.
    Let $g=\ConjugationMap{G}{a}$.
    Let $B=\conjugateSubset{G}{b}{A}$.
    $\mul[G](a,b)\in\carrier[G]$ by
        \obv{group_is_monoid,monoid_is_magma,magma_welldef}.
    $A\inter\carrier[G]=A$ by
        \obv{inter_absorb_supseteq_right}.
    $B\subseteq\carrier[G]$ by
        \auto{conjugate_subset_subseteq_carrier}.
    Hence $B\inter\carrier[G]=B$ by
        \obv{inter_absorb_supseteq_right}.
    $f,g\in\funs{\carrier[G]}{\carrier[G]}$ by
        \auto{conjugation_map_function}.
    Hence $f$ and $g$ are functions and
        $\ran{f}\subseteq\carrier[G]$ and
        $\dom{g}=\carrier[G]$ by
        \auto{funs_elim,funs_ran}.
    Hence $\ran{f}\subseteq\dom{g}$.
    $\ConjugationMap{G}{\mul[G](a,b)}=g\circ f$ by
        \auto{conjugation_map_composition}.
    $\img{g\circ f}{A}=\img{g}{\img{f}{A}}$ by
        \obv{img_of_circ}.
    $B=\img{f}{A}$ by \obv{conjugate_subset}.
    Follows by \obv{conjugate_subset}.
\end{proof}

\begin{theorem}\label{conjugate_generated_subgroup}
    Let $G$ be a group.
    Let $a\in\carrier[G]$.
    Let $A$ be a subset of $\carrier[G]$.
    Then
        $\conjugateSubset{G}{a}{\generatedSubgroup{A}{G}}=
            \generatedSubgroup{\conjugateSubset{G}{a}{A}}{G}$.
\end{theorem}
\begin{proof}
    Let $f=\ConjugationMap{G}{a}$ and
        $P=\generatedSubgroup{A}{G}$.
    $\groupHom{f}{G}{G}$ by
        \auto{conjugation_map_group_hom}.
    Hence $f\in\funs{\carrier[G]}{\carrier[G]}$ by
        \obv{group_hom_type}.
    $P\subseteq\carrier[G]$ by
        \auto{generated_subgroup_subseteq_carrier}.
    Hence $P\inter\carrier[G]=P$ by
        \obv{inter_absorb_supseteq_right}.
    $A\inter\carrier[G]=A$ by
        \obv{inter_absorb_supseteq_right}.
    $\conjugateSubset{G}{a}{P}=\img{f}{P}$ and
        $\conjugateSubset{G}{a}{A}=\img{f}{A}$ by
        \obv{conjugate_subset}.
    $\img{f}{P}=\generatedSubgroup{\img{f}{A}}{G}$ by
        \auto{group_hom_image_generated_subgroup}.
\end{proof}

\begin{theorem}\label{group_hom_conjugate}
    Suppose $\groupHom{f}{G}{H}$.
    Let $a,x\in\carrier[G]$.
    Then
        $f(\conjugate{G}{a}{x})=
            \conjugate{H}{f(a)}{f(x)}$.
\end{theorem}
\begin{proof}
    $\groupInv[G](a),\mul[G](a,x)\in\carrier[G]$ by
        \obv{group_inverse_type,group_hom_source_group,group_is_monoid,monoid_is_magma,magma_welldef}.
    $f(a),f(x)\in\carrier[H]$ by
        \obv{group_hom_apply_type}.
    $f(\groupInv[G](a))=\groupInv[H](f(a))$ by
        \obv{group_hom_inverse}.
    Follows by
        \auto{group_conjugate,group_hom_mul}.
\end{proof}
