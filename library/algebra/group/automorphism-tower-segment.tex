\section{Canonical set-sized automorphism towers}

For each ordinal length, the recursive package history is projected once into
an exact stage family and an exact bond family.  Prefix equality is literal:
it is equality of these set-coded graphs, not merely a later comparison
isomorphism.

\subsection{The exact segment constructor}

\begin{definition}\label{automorphism_tower_stages}
    $\automorphismTowerStages{G}{\theta}=
        \towerHistoryStages{\theta}{\towerPackageHistory{G}{\theta}}$.
\end{definition}

\begin{definition}\label{automorphism_tower_bonds}
    $\automorphismTowerBonds{G}{\theta}=
        \towerHistoryBonds{\theta}{\towerPackageHistory{G}{\theta}}$.
\end{definition}

\begin{definition}\label{automorphism_tower_segment}
    $\automorphismTowerSegment{G}{\theta}=
        \towerSegmentData{\theta}

\subsection{Literal prefix stability}

\begin{proposition}\label{automorphism_tower_stage_prefix_value}
    Let $\theta,\eta$ be ordinals.
    Suppose $\theta\subseteq\eta$.
    Let $\beta\in\theta$.
    Then $\apply{\automorphismTowerStages{G}{\theta}}{\beta}=
        \apply{\automorphismTowerStages{G}{\eta}}{\beta}$.
\end{proposition}
\begin{proof}
    $\beta\in\eta$ by \obv{subseteq}.
    \begin{align*}
        \apply{\automorphismTowerStages{G}{\theta}}{\beta}
            &= \towerStageGroup[
                \apply{\towerPackageHistory{G}{\theta}}{\beta}]
                \explanation{by
                    \auto{automorphism_tower_stages,
                        tower_history_stages_apply}}
            \\
            &= \towerStageGroup[
                \apply{\towerPackageHistory{G}{\eta}}{\beta}]
                \explanation{by
                    \obv{tower_package_history_value_stability}}
            \\
            &= \apply{\automorphismTowerStages{G}{\eta}}{\beta}
                \explanation{by
                    \auto{automorphism_tower_stages,
                        tower_history_stages_apply}}
    \end{align*}
\end{proof}

\begin{theorem}\label{automorphism_tower_bonds_restriction}
    Let $\theta,\eta$ be ordinals.
    Suppose $\theta\subseteq\eta$.
    Then $\automorphismTowerBonds{G}{\theta}=
        \restrl{\automorphismTowerBonds{G}{\eta}}
            {\subseteqrel{\theta}}$.
\end{theorem}
\begin{proof}
    $\automorphismTowerBonds{G}{\theta}\in
        \funs{\subseteqrel{\theta}}{
            \ran{\automorphismTowerBonds{G}{\theta}}}$ by
        \obv{automorphism_tower_bonds_type}.
    $\automorphismTowerBonds{G}{\eta}\in
        \funs{\subseteqrel{\eta}}{
            \ran{\automorphismTowerBonds{G}{\eta}}}$ by
        \obv{automorphism_tower_bonds_type}.
    $\subseteqrel{\theta}\subseteq\subseteqrel{\eta}$ by
        \obv{subseteqrel_monotone_for_tower}.
    Show that for every $z\in\subseteqrel{\theta}$ we have
        $\apply{\automorphismTowerBonds{G}{\theta}}{z}=
        \apply{\automorphismTowerBonds{G}{\eta}}{z}$.
    \begin{subproof}
        Fix $z\in\subseteqrel{\theta}$.
        Take $\alpha,\beta\in\theta$ such that
            $z=(\alpha,\beta)$ and $\alpha\subseteq\beta$ by
            \auto{subseteqrel}.
        Follows by
            \auto{automorphism_tower_bond_prefix_value}.
    \end{subproof}
    Follows by \obv{function_eq_restriction_from_agreement}.
\end{proof}

        \obv{automorphism_tower_segment,tower_segment_restriction}.
\end{proof}

                \automorphismTowerSegment{G}{\theta}}{\lambda}}{\beta}=
            \apply{\directLimitCanonicalFamily{\towerEarlierSystem{
                \automorphismTowerSegment{G}{\theta}}{\lambda}}}{\beta}$.
    Then $\towerIncomingFamily{
            \automorphismTowerSegment{G}{\theta}}{\lambda}=
        \directLimitCanonicalFamily{\towerEarlierSystem{
            \automorphismTowerSegment{G}{\theta}}{\lambda}}$.
\end{proposition}
\begin{proof}
    Follows by \obv{typed_functions_equal_from_agreement}.
\end{proof}

\subsection{Secondary stage and map accessors}

These total terms are backed by the least successor-length segment containing
the requested value.  The segment API above remains primary; the following
stability laws ensure that the notation does not choose incompatible local
witnesses.

\begin{definition}\label{automorphism_tower_stage}
    $\automorphismTowerStage{G}{\alpha}=
        \towerStage{\automorphismTowerSegment{G}{\suc{\alpha}}}{\alpha}$.
\end{definition}

\begin{definition}\label{automorphism_tower_map}
    $\automorphismTowerMap{G}{\alpha}{\beta}=
        \towerBond{\automorphismTowerSegment{G}{\suc{\beta}}}
            {\alpha}{\beta}$.
\end{definition}

\begin{proposition}\label{automorphism_tower_stage_on_segment}
    Let $\theta$ be an ordinal.
    Let $\alpha\in\theta$.
    Then $\automorphismTowerStage{G}{\alpha}=
        \towerStage{\automorphismTowerSegment{G}{\theta}}{\alpha}$.
\end{proposition}
\begin{proof}
    $\alpha$ is an ordinal and $\suc{\alpha}\subseteq\theta$ by
        \obv{elem_of_ordinal_is_ordinal,ordinal_suc_subseteq}.
    $\suc{\alpha}$ is an ordinal by \obv{suc_ordinal}.
    $\alpha\in\suc{\alpha}$ by \obv{suc_intro_self}.
    Follows by
        \obv{automorphism_tower_stage,
            automorphism_tower_stage_stability}.
\end{proof}

\begin{proposition}\label{automorphism_tower_map_on_segment}
    Let $\theta$ be an ordinal.
    Let $\alpha,\beta\in\theta$.
    Suppose $\alpha\subseteq\beta$.
    Then $\automorphismTowerMap{G}{\alpha}{\beta}=
        \towerBond{\automorphismTowerSegment{G}{\theta}}
            {\alpha}{\beta}$.
\end{proposition}
\begin{proof}
    $\alpha$ and $\beta$ are ordinals by
        \obv{elem_of_ordinal_is_ordinal}.
    Hence $\suc{\beta}$ is an ordinal and
        $\suc{\beta}\subseteq\theta$ by
        \obv{suc_ordinal,ordinal_suc_subseteq}.
    $\alpha\in\beta$ or $\alpha=\beta$ by
        \obv{ordinal_subseteq_implies_elem_or_eq}.
    Hence $\alpha,\beta\in\suc{\beta}$ by
        \auto{suc_iff}.
    Follows by
        \obv{automorphism_tower_map,
            automorphism_tower_bond_stability}.
\end{proof}

\begin{proposition}\label{automorphism_tower_stage_is_group}
    Let $G$ be a group.
    Let $\alpha$ be an ordinal.
    Then $\automorphismTowerStage{G}{\alpha}$ is a group.
\end{proposition}
\begin{proof}
    $\suc{\alpha}$ is an ordinal and $\alpha\in\suc{\alpha}$ by
        \obv{suc_ordinal,suc_intro_self}.
    Follows by
        \auto{automorphism_tower_stage,
            automorphism_tower_segment_stage_is_group}.
\end{proof}

\begin{proposition}\label{automorphism_tower_map_is_hom}
    Let $G$ be a group.
    Let $\alpha,\beta$ be ordinals.
    Suppose $\alpha\subseteq\beta$.
    Then $\groupHom{\automorphismTowerMap{G}{\alpha}{\beta}}
        {\automorphismTowerStage{G}{\alpha}}
        {\automorphismTowerStage{G}{\beta}}$.
\end{proposition}
\begin{proof}
    $\suc{\beta}$ is an ordinal by \obv{suc_ordinal}.
    $\alpha\in\beta$ or $\alpha=\beta$ by
        \obv{ordinal_subseteq_implies_elem_or_eq}.
    Hence $\alpha,\beta\in\suc{\beta}$ by \auto{suc_iff}.
    $\groupHom{\towerBond{\automorphismTowerSegment{G}{\suc{\beta}}}
            {\alpha}{\beta}}
        {\towerStage{\automorphismTowerSegment{G}{\suc{\beta}}}{\alpha}}
        {\towerStage{\automorphismTowerSegment{G}{\suc{\beta}}}{\beta}}$
        by \obv{automorphism_tower_bond_is_hom}.
    $\automorphismTowerStage{G}{\alpha}=
        \towerStage{\automorphismTowerSegment{G}{\suc{\beta}}}{\alpha}$
        and
        $\automorphismTowerStage{G}{\beta}=
        \towerStage{\automorphismTowerSegment{G}{\suc{\beta}}}{\beta}$
        by \obv{automorphism_tower_stage_on_segment}.
    Follows by \obv{automorphism_tower_map}.
\end{proof}
