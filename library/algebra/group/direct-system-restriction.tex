\import{algebra/group/direct-system.tex}

\section{Restriction of group direct systems}

Restriction is literal precomposition with a directed index reduct.  The
smaller index carrier must itself be directed; this is deliberately stronger
than merely being a subset of the original carrier.

\begin{definition}\label{group_direct_system_restriction}
    $\groupDirectSystemRestriction{S}{J}=
        \groupDirectSystemStructure
            {J}
            {\restrl{\directSystemGroups[S]}{\carrier[J]}}
            {\restrl{\directSystemBonds[S]}{\leq[J]}}$.
\end{definition}

\begin{proposition}\label{group_direct_system_restriction_carrier}
    $\carrier[\groupDirectSystemRestriction{S}{J}]=\carrier[J]$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{group_direct_system_restriction,group_direct_system_structure_carrier}.
\end{proof}

\begin{proposition}\label{group_direct_system_restriction_order}
    $\leq[\groupDirectSystemRestriction{S}{J}]=\leq[J]$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{group_direct_system_restriction,group_direct_system_structure_order}.
\end{proof}

\begin{proposition}\label{group_direct_system_restriction_groups}
    $\directSystemGroups[\groupDirectSystemRestriction{S}{J}]
        =\restrl{\directSystemGroups[S]}{\carrier[J]}$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{group_direct_system_restriction,group_direct_system_structure_groups}.
\end{proof}

\begin{proposition}\label{group_direct_system_restriction_bonds}
    $\directSystemBonds[\groupDirectSystemRestriction{S}{J}]
        =\restrl{\directSystemBonds[S]}{\leq[J]}$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{group_direct_system_restriction,group_direct_system_structure_bonds}.
\end{proof}

\begin{proposition}\label{group_direct_system_restriction_group}
    Let $S$ be a group direct system.
    Suppose $\indexSubstructure{J}{S}$.
    Let $i\in\carrier[J]$.
    Then $\directSystemGroup{\groupDirectSystemRestriction{S}{J}}{i}
        =\directSystemGroup{S}{i}$.
\end{proposition}
\begin{proof}
    $\directSystemGroups[S]\in
        \funs{\carrier[S]}{\ran{\directSystemGroups[S]}}$ by
        \obv{group_direct_system_group_family}.
    Hence $\directSystemGroups[S]$ is a function and
        $\dom{\directSystemGroups[S]}=\carrier[S]$ by
        \obv{funs_is_function,funs_elim}.
    $\carrier[J]\subseteq\carrier[S]$ by
        \obv{index_substructure_carrier_subseteq}.
    Hence $\carrier[J]\subseteq\dom{\directSystemGroups[S]}$.
    Hence
        $\apply{\restrl{\directSystemGroups[S]}{\carrier[J]}}{i}
            =\apply{\directSystemGroups[S]}{i}$ by
        \obv{restrl_of_function_apply}.
    Follows by
        \auto{direct_system_group,group_direct_system_restriction_groups}.
\end{proof}

\begin{proposition}\label{group_direct_system_restriction_bond}
    Let $S$ be a group direct system.
    Suppose $\indexSubstructure{J}{S}$.
    Let $i,j\in\carrier[J]$.
    Suppose $i\mathrel{\leq[J]}j$.
    Then $\directSystemBond{\groupDirectSystemRestriction{S}{J}}{i}{j}
        =\directSystemBond{S}{i}{j}$.
\end{proposition}
\begin{proof}
    $\directSystemBonds[S]\in
        \funs{\leq[S]}{\ran{\directSystemBonds[S]}}$ by
        \obv{group_direct_system_bond_family}.
    Hence $\directSystemBonds[S]$ is a function and
        $\dom{\directSystemBonds[S]}=\leq[S]$ by
        \obv{funs_is_function,funs_elim}.
    $\leq[J]\subseteq\leq[S]$ by
        \obv{index_substructure_order_subseteq}.
    Hence $\leq[J]\subseteq\dom{\directSystemBonds[S]}$.
    $(i,j)\in\leq[J]$.
    Hence
        $\apply{\restrl{\directSystemBonds[S]}{\leq[J]}}{(i,j)}
            =\apply{\directSystemBonds[S]}{(i,j)}$ by
        \obv{restrl_of_function_apply}.
    Follows by
        \auto{direct_system_bond,group_direct_system_restriction_bonds}.
\end{proof}

\begin{theorem}\label{group_direct_system_restriction_is_exact}
    Let $S$ be a group direct system.
    Suppose $\indexSubstructure{J}{S}$.
    Then $\groupDirectSystemRestriction{S}{J}$ is an exact
        group direct system.
\end{theorem}
\begin{proof}
    Let $G=\restrl{\directSystemGroups[S]}{\carrier[J]}$.
    Let $\Phi=\restrl{\directSystemBonds[S]}{\leq[J]}$.
    $J$ is a directed quasiordered set by
        \obv{index_substructure_is_directed}.
    $\carrier[J]\subseteq\carrier[S]$ and
        $\leq[J]\subseteq\leq[S]$ by
        \obv{index_substructure_carrier_subseteq,index_substructure_order_subseteq}.
    $\directSystemGroups[S]\in
        \funs{\carrier[S]}{\ran{\directSystemGroups[S]}}$ and
        $\directSystemBonds[S]\in
        \funs{\leq[S]}{\ran{\directSystemBonds[S]}}$ by
        \obv{group_direct_system_group_family,group_direct_system_bond_family}.
    Hence $G\in\funs{\carrier[J]}{\ran{\directSystemGroups[S]}}$ and
        $\Phi\in\funs{\leq[J]}{\ran{\directSystemBonds[S]}}$ by
        \auto{funs_restrl}.
    Hence $G\in\surj{\carrier[J]}{\ran{G}}$ and
        $\Phi\in\surj{\leq[J]}{\ran{\Phi}}$ by
        \obv{fun_to_surj}.
    Hence $G\in\funs{\carrier[J]}{\ran{G}}$ and
        $\Phi\in\funs{\leq[J]}{\ran{\Phi}}$ by
        \obv{surj_to_fun}.
    Show that for every $i\in\carrier[J]$ we have $\apply{G}{i}$ is a group.
    \begin{subproof}
        Fix $i\in\carrier[J]$.
        $i\in\carrier[S]$ by \obv{subseteq}.
        $\apply{G}{i}=\directSystemGroup{S}{i}$ by
            \auto{group_direct_system_restriction_group,direct_system_group,group_direct_system_restriction_groups}.
        Follows by \obv{direct_system_group_is_group}.
    \end{subproof}
    Show that for all $i,j\in\carrier[J]$ such that
        $i\mathrel{\leq[J]}j$ we have
        $\groupHom{\apply{\Phi}{(i,j)}}
            {\apply{G}{i}}{\apply{G}{j}}$.
    \begin{subproof}
        Fix $i,j\in\carrier[J]$.
        Assume $i\mathrel{\leq[J]}j$.
        Hence $i,j\in\carrier[S]$ and
            $i\mathrel{\leq[S]}j$ by
            \obv{subseteq,index_substructure_order_iff}.
        $\apply{G}{i}=\directSystemGroup{S}{i}$ and
            $\apply{G}{j}=\directSystemGroup{S}{j}$ by
            \auto{group_direct_system_restriction_group,direct_system_group,group_direct_system_restriction_groups}.
        $\apply{\Phi}{(i,j)}=\directSystemBond{S}{i}{j}$ by
            \auto{group_direct_system_restriction_bond,direct_system_bond,group_direct_system_restriction_bonds}.
        Follows by \auto{direct_system_bond_hom}.
    \end{subproof}
    Show that for every $i\in\carrier[J]$ we have
        $\apply{\Phi}{(i,i)}=\identity{\carrier[\apply{G}{i}]}$.
    \begin{subproof}
        Fix $i\in\carrier[J]$.
        Hence $i\in\carrier[S]$ by \obv{subseteq}.
        $i\mathrel{\leq[J]}i$ by
            \obv{directed_order_reflexive}.
        $\apply{G}{i}=\directSystemGroup{S}{i}$ by
            \auto{group_direct_system_restriction_group,direct_system_group,group_direct_system_restriction_groups}.
        $\apply{\Phi}{(i,i)}=\directSystemBond{S}{i}{i}$ by
            \auto{group_direct_system_restriction_bond,direct_system_bond,group_direct_system_restriction_bonds}.
        Follows by \auto{direct_system_bond_identity}.
    \end{subproof}
    Show that for all $i,j,k\in\carrier[J]$ such that
        $i\mathrel{\leq[J]}j$ and $j\mathrel{\leq[J]}k$ we have
        $\apply{\Phi}{(i,k)}=
            \apply{\Phi}{(j,k)}\circ\apply{\Phi}{(i,j)}$.
    \begin{subproof}
        Fix $i,j,k\in\carrier[J]$.
        Assume $i\mathrel{\leq[J]}j$ and
            $j\mathrel{\leq[J]}k$.
        Hence $i,j,k\in\carrier[S]$ and
            $i\mathrel{\leq[S]}j$ and
            $j\mathrel{\leq[S]}k$ by
            \obv{subseteq,index_substructure_order_iff}.
        $\apply{\Phi}{(i,k)}=\directSystemBond{S}{i}{k}$ and
            $\apply{\Phi}{(j,k)}=\directSystemBond{S}{j}{k}$ and
            $\apply{\Phi}{(i,j)}=\directSystemBond{S}{i}{j}$ by
            \auto{group_direct_system_restriction_bond,direct_system_bond,group_direct_system_restriction_bonds,directed_order_transitive}.
        Follows by \auto{direct_system_bond_composition}.
    \end{subproof}
    $\groupDirectSystemStructure{J}{G}{\Phi}$ is an exact
        group direct system by
        \obv{group_direct_system_structure_is_system}.
    Follows by \auto{group_direct_system_restriction}.
\end{proof}

\begin{proposition}\label{ordinal_index_substructure_of_group_system}
    Let $S$ be a group direct system.
    Let $\theta$ be an ordinal such that $\theta\neq\emptyset$.
    Suppose $\carrier[S]=\theta$ and
        $\leq[S]=\subseteqrel{\theta}$.
    Let $\lambda$ be an ordinal such that
        $\lambda\neq\emptyset$ and $\lambda\subseteq\theta$.
    Then $\indexSubstructure
        {\ordinalDirectedQuasiorder{\lambda}}{S}$.
\end{proposition}
\begin{proof}
    Let $J=\ordinalDirectedQuasiorder{\lambda}$.
    Let $I=\ordinalDirectedQuasiorder{\theta}$.
    $\indexSubstructure{J}{I}$ by
        \auto{ordinal_index_substructure}.
    Hence $J$ is a directed quasiordered set and
        $\carrier[J]\subseteq\carrier[I]$ and
        $\leq[J]=\leq[I]\inter
            (\carrier[J]\times\carrier[J])$ by
        \obv{index_substructure_is_directed,
            index_substructure_carrier_subseteq,
            index_substructure_order_reduct}.
    $S$ is a directed quasiordered set by
        \obv{group_direct_system_is_directed}.
    $\carrier[I]=\theta$ and
        $\leq[I]=\subseteqrel{\theta}$ by
        \auto{ordinal_directed_quasiorder_carrier,
            ordinal_directed_quasiorder_relation}.
    Hence $\carrier[J]\subseteq\carrier[S]$ and
        $\leq[J]=\leq[S]\inter
            (\carrier[J]\times\carrier[J])$.
    Follows by \obv{index_substructure}.
\end{proof}

\begin{theorem}\label{ordinal_group_direct_system_restriction_is_exact}
    Let $S$ be a group direct system.
    Let $\theta$ be an ordinal such that $\theta\neq\emptyset$.
    Suppose $\carrier[S]=\theta$ and
        $\leq[S]=\subseteqrel{\theta}$.
    Let $\lambda$ be an ordinal such that
        $\lambda\neq\emptyset$ and $\lambda\subseteq\theta$.
    Then $\groupDirectSystemRestriction
        {S}{\ordinalDirectedQuasiorder{\lambda}}$ is an exact
        group direct system.
\end{theorem}
\begin{proof}
    $\indexSubstructure
        {\ordinalDirectedQuasiorder{\lambda}}{S}$ by
        \obv{ordinal_index_substructure_of_group_system}.
    Follows by \obv{group_direct_system_restriction_is_exact}.
\end{proof}

\begin{corollary}\label{ordinal_group_direct_system_restriction_is_system}
    Let $S$ be a group direct system.
    Let $\theta$ be an ordinal such that $\theta\neq\emptyset$.
    Suppose $\carrier[S]=\theta$ and
        $\leq[S]=\subseteqrel{\theta}$.
    Let $\lambda$ be an ordinal such that
        $\lambda\neq\emptyset$ and $\lambda\subseteq\theta$.
    Then $\groupDirectSystemRestriction
        {S}{\ordinalDirectedQuasiorder{\lambda}}$ is a
        group direct system.
\end{corollary}
\begin{proof}
    Follows by
        \obv{ordinal_group_direct_system_restriction_is_exact}.
\end{proof}

\begin{theorem}\label{embedding_group_direct_system_restriction}
    Suppose $\embeddingGroupDirectSystem{S}$ and
        $\indexSubstructure{J}{S}$.
    Then $\embeddingGroupDirectSystem
        {\groupDirectSystemRestriction{S}{J}}$.
\end{theorem}
\begin{proof}
    $S$ is a group direct system by
        \obv{embedding_group_direct_system_is_system}.
    Let $T=\groupDirectSystemRestriction{S}{J}$.
    $T$ is a group direct system by
        \obv{group_direct_system_restriction_is_exact}.
    $\carrier[T]=\carrier[J]$ and $\leq[T]=\leq[J]$ by
        \auto{group_direct_system_restriction_carrier,group_direct_system_restriction_order}.
    Show that for all $i,j\in\carrier[T]$ such that
        $i\mathrel{\leq[T]}j$ we have
        $\groupEmbedding
            {\directSystemBond{T}{i}{j}}
            {\directSystemGroup{T}{i}}
            {\directSystemGroup{T}{j}}$.
    \begin{subproof}
        Fix $i,j\in\carrier[T]$ such that $i\mathrel{\leq[T]}j$.
        Hence $i,j\in\carrier[J]$ and
            $i\mathrel{\leq[J]}j$.
        Hence $i,j\in\carrier[S]$ and
            $i\mathrel{\leq[S]}j$ by
            \auto{subseteq,index_substructure_carrier_subseteq,index_substructure_order_iff}.
        $\directSystemGroup{T}{i}=\directSystemGroup{S}{i}$ and
            $\directSystemGroup{T}{j}=\directSystemGroup{S}{j}$ by
            \auto{group_direct_system_restriction_group}.
        $\directSystemBond{T}{i}{j}=\directSystemBond{S}{i}{j}$ by
            \auto{group_direct_system_restriction_bond}.
        Follows by \auto{embedding_direct_system_bond}.
    \end{subproof}
    Follows by \auto{embedding_group_direct_system}.
\end{proof}
