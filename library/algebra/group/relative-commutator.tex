\section{Relative commutator subgroups}

The relative commutator subgroup is generated by commutators whose entries
come from two displayed sets.  As with centralizers and conjugated subsets,
only the traces of those sets on the ambient carrier matter.  This makes the
construction total without assigning group-theoretic meaning to elements
outside the group.

\begin{definition}\label{relative_commutator_carrier}
    $\relativeCommutatorCarrier{G}{A}{B}=
        \{z\in\carrier[G]\mid
            \text{there exists $a\in A\inter\carrier[G]$ such that
            there exists $b\in B\inter\carrier[G]$ such that
            $z=\groupCommutator{G}{a}{b}$}\}$.
\end{definition}

\begin{proposition}\label{relative_commutator_carrier_iff}
    $z\in\relativeCommutatorCarrier{G}{A}{B}$ iff
        $z\in\carrier[G]$ and
        there exists $a\in A\inter\carrier[G]$ such that
            there exists $b\in B\inter\carrier[G]$ such that
                $z=\groupCommutator{G}{a}{b}$.
\end{proposition}
\begin{proof}
    Follows by \obv{relative_commutator_carrier}.
\end{proof}

\begin{proposition}\label{relative_commutator_carrier_subseteq}
    $\relativeCommutatorCarrier{G}{A}{B}\subseteq\carrier[G]$.
\end{proposition}
\begin{proof}
    Follows by \obv{relative_commutator_carrier,subseteq}.
\end{proof}

\begin{classification}\label{relative_commutator_carrier_is_subset_in_group}
    Let $G$ be a group.
    Then $\relativeCommutatorCarrier{G}{A}{B}\subseteq\carrier[G]$.
\end{classification}
\begin{proof}
    Follows by \obv{relative_commutator_carrier_subseteq}.
\end{proof}

\begin{proposition}\label{relative_commutator_carrier_trace}
    $\relativeCommutatorCarrier{G}{A}{B}=
        \relativeCommutatorCarrier{G}
            {A\inter\carrier[G]}{B\inter\carrier[G]}$.
\end{proposition}
\begin{proof}
    $(A\inter\carrier[G])\inter\carrier[G]
        =A\inter\carrier[G]$ and
        $(B\inter\carrier[G])\inter\carrier[G]
        =B\inter\carrier[G]$ by
        \auto{inter_assoc,inter_idempotent}.
    Follows by \auto{relative_commutator_carrier_iff,setext}.
\end{proof}

\begin{definition}\label{relative_commutator_subgroup}
    Let $G$ be a group.
    $\relativeCommutatorSubgroup{G}{A}{B}=
        \generatedSubgroup{
            \relativeCommutatorCarrier{G}{A}{B}}{G}$.
\end{definition}

\begin{proposition}\label{relative_commutator_subgroup_trace}
    Let $G$ be a group.
    Then
        $\relativeCommutatorSubgroup{G}{A}{B}=
            \relativeCommutatorSubgroup{G}
                {A\inter\carrier[G]}{B\inter\carrier[G]}$.
\end{proposition}
\begin{proof}
    Follows by
        \obv{relative_commutator_subgroup,
            relative_commutator_carrier_trace}.
\end{proof}

\begin{theorem}\label{relative_commutator_subgroup_is_subgroup}
    Let $G$ be a group.
    Then $\subgroup{\relativeCommutatorSubgroup{G}{A}{B}}{G}$.
\end{theorem}
\begin{proof}
    $\relativeCommutatorCarrier{G}{A}{B}\subseteq\carrier[G]$ by
        \obv{relative_commutator_carrier_subseteq}.
    Follows by
        \auto{relative_commutator_subgroup,
            generated_subgroup_is_subgroup}.
\end{proof}

\begin{proposition}\label{relative_commutator_subgroup_contains_commutators}
    Let $G$ be a group.
    Let $a\in A\inter\carrier[G]$.
    Let $b\in B\inter\carrier[G]$.
    Then
        $\groupCommutator{G}{a}{b}\in
            \relativeCommutatorSubgroup{G}{A}{B}$.
\end{proposition}
\begin{proof}
    $a,b\in\carrier[G]$ by \obv{inter_elim_right}.
    $\groupCommutator{G}{a}{b}\in\carrier[G]$ by
        \obv{group_commutator_type}.
    Hence
        $\groupCommutator{G}{a}{b}\in
            \relativeCommutatorCarrier{G}{A}{B}$ by
        \obv{relative_commutator_carrier_iff}.
    $\relativeCommutatorCarrier{G}{A}{B}\subseteq\carrier[G]$ by
        \obv{relative_commutator_carrier_subseteq}.
    Follows by
        \auto{relative_commutator_subgroup,
            generated_subgroup_contains_generators,subseteq}.
\end{proof}

\begin{theorem}\label{relative_commutator_subgroup_universal_property}
    Let $G$ be a group.
    Suppose $\subgroup{K}{G}$.
    Then
        $\relativeCommutatorSubgroup{G}{A}{B}\subseteq K$ iff
        for every $a\in A\inter\carrier[G]$ we have for every
        $b\in B\inter\carrier[G]$ we have
            $\groupCommutator{G}{a}{b}\in K$.
\end{theorem}
\begin{proof}
    Show that if
        $\relativeCommutatorSubgroup{G}{A}{B}\subseteq K$, then
        for every $a\in A\inter\carrier[G]$ we have for every
        $b\in B\inter\carrier[G]$ we have
            $\groupCommutator{G}{a}{b}\in K$.
    \begin{subproof}
        Assume $\relativeCommutatorSubgroup{G}{A}{B}\subseteq K$.
        Fix $a\in A\inter\carrier[G]$ and
            $b\in B\inter\carrier[G]$.
        $\groupCommutator{G}{a}{b}\in
            \relativeCommutatorSubgroup{G}{A}{B}$ by
            \obv{relative_commutator_subgroup_contains_commutators}.
        Follows by \obv{subseteq}.
    \end{subproof}
    Show that if for every $a\in A\inter\carrier[G]$ we have for every
        $b\in B\inter\carrier[G]$ we have
            $\groupCommutator{G}{a}{b}\in K$, then
        $\relativeCommutatorSubgroup{G}{A}{B}\subseteq K$.
    \begin{subproof}
        Assume that for every $a\in A\inter\carrier[G]$ we have for every
            $b\in B\inter\carrier[G]$ we have
                $\groupCommutator{G}{a}{b}\in K$.
        Show that $\relativeCommutatorCarrier{G}{A}{B}\subseteq K$.
        \begin{subproof}
            Show that for every
                $z\in\relativeCommutatorCarrier{G}{A}{B}$ we have
                $z\in K$.
            \begin{subproof}
                Fix $z\in\relativeCommutatorCarrier{G}{A}{B}$.
                Take $a\in A\inter\carrier[G]$ such that there exists
                    $b\in B\inter\carrier[G]$ such that
                    $z=\groupCommutator{G}{a}{b}$ by
                    \obv{relative_commutator_carrier_iff}.
                Take $b\in B\inter\carrier[G]$ such that
                    $z=\groupCommutator{G}{a}{b}$.
                Follows by \auto.
            \end{subproof}
            Follows by \obv{subseteq}.
        \end{subproof}
        $\relativeCommutatorCarrier{G}{A}{B}\subseteq\carrier[G]$ by
            \obv{relative_commutator_carrier_subseteq}.
        Follows by
            \auto{relative_commutator_subgroup,generated_subgroup_least}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{corollary}\label{relative_commutator_subgroup_least}
    Let $G$ be a group.
    Suppose $\subgroup{K}{G}$.
    Suppose for every $a\in A\inter\carrier[G]$ we have for every
        $b\in B\inter\carrier[G]$ we have
            $\groupCommutator{G}{a}{b}\in K$.
    Then $\relativeCommutatorSubgroup{G}{A}{B}\subseteq K$.
\end{corollary}
\begin{proof}
    Follows by
        \auto{relative_commutator_subgroup_universal_property}.
\end{proof}

\begin{proposition}\label{relative_commutator_subgroup_monotone}
    Let $G$ be a group.
    Suppose $A\subseteq A'$ and $B\subseteq B'$.
    Then
        $\relativeCommutatorSubgroup{G}{A}{B}\subseteq
         \relativeCommutatorSubgroup{G}{A'}{B'}$.
\end{proposition}
\begin{proof}
    $\subgroup{\relativeCommutatorSubgroup{G}{A'}{B'}}{G}$ by
        \obv{relative_commutator_subgroup_is_subgroup}.
    Show that for every $a\in A\inter\carrier[G]$ we have for every
        $b\in B\inter\carrier[G]$ we have
        $\groupCommutator{G}{a}{b}\in
            \relativeCommutatorSubgroup{G}{A'}{B'}$.
    \begin{subproof}
        Fix $a\in A\inter\carrier[G]$ and
            $b\in B\inter\carrier[G]$.
        $a\in A'\inter\carrier[G]$ and
            $b\in B'\inter\carrier[G]$ by
            \obv{inter_intro,inter_elim_left,inter_elim_right,subseteq}.
        Follows by
            \obv{relative_commutator_subgroup_contains_commutators}.
    \end{subproof}
    Follows by \obv{relative_commutator_subgroup_least}.
\end{proof}

\begin{corollary}\label{relative_commutator_subgroup_empty_left}
    Let $G$ be a group.
    Then
        $\relativeCommutatorSubgroup{G}{\emptyset}{B}=
            \trivialSubgroupCarrier{G}$.
\end{corollary}
\begin{proof}
    $\emptyset\inter\carrier[G]=\emptyset$ by
        \auto{inter_comm,inter_emptyset}.
    Hence $\relativeCommutatorCarrier{G}{\emptyset}{B}=\emptyset$ by
        \auto{relative_commutator_carrier_iff,setext,emptyset}.
    Follows by
        \auto{relative_commutator_subgroup,generated_subgroup_emptyset}.
\end{proof}

\begin{corollary}\label{relative_commutator_subgroup_empty_right}
    Let $G$ be a group.
    Then
        $\relativeCommutatorSubgroup{G}{A}{\emptyset}=
            \trivialSubgroupCarrier{G}$.
\end{corollary}
\begin{proof}
    $\emptyset\inter\carrier[G]=\emptyset$ by
        \auto{inter_comm,inter_emptyset}.
    Hence $\relativeCommutatorCarrier{G}{A}{\emptyset}=\emptyset$ by
        \auto{relative_commutator_carrier_iff,setext,emptyset}.
    Follows by
        \auto{relative_commutator_subgroup,generated_subgroup_emptyset}.
\end{proof}

\begin{theorem}\label{relative_commutator_subgroup_full_carrier}
    Let $G$ be a group.
    Then
        $\relativeCommutatorSubgroup{G}{\carrier[G]}{\carrier[G]}=
            \commutatorSubgroup{G}$.
\end{theorem}
\begin{proof}
    $\carrier[G]\inter\carrier[G]=\carrier[G]$ by
        \obv{inter_idempotent}.
    Hence
        $\relativeCommutatorCarrier{G}{\carrier[G]}{\carrier[G]}=
            \groupCommutatorCarrier{G}$ by
        \auto{relative_commutator_carrier_iff,
            group_commutator_carrier_iff,setext}.
    Follows by
        \obv{relative_commutator_subgroup,commutator_subgroup}.
\end{proof}

\begin{theorem}\label{relative_commutator_subgroup_symmetric}
    Let $G$ be a group.
    Then
        $\relativeCommutatorSubgroup{G}{A}{B}=
         \relativeCommutatorSubgroup{G}{B}{A}$.
\end{theorem}
\begin{proof}
    Let $R=\relativeCommutatorSubgroup{G}{A}{B}$ and
        $S=\relativeCommutatorSubgroup{G}{B}{A}$.
    $\subgroup{R}{G}$ by
        \auto{relative_commutator_subgroup_is_subgroup}.
    $\subgroup{S}{G}$ by
        \auto{relative_commutator_subgroup_is_subgroup}.
    Show that $R\subseteq S$.
    \begin{subproof}
        Show that for every $a\in A\inter\carrier[G]$ we have for every
            $b\in B\inter\carrier[G]$ we have
            $\groupCommutator{G}{a}{b}\in S$.
        \begin{subproof}
            Fix $a\in A\inter\carrier[G]$ and
                $b\in B\inter\carrier[G]$.
            $a,b\in\carrier[G]$ by \obv{inter_elim_right}.
            $\groupCommutator{G}{b}{a}\in S$ by
                \obv{relative_commutator_subgroup_contains_commutators}.
            Hence
                $\groupInv[G](\groupCommutator{G}{b}{a})\in S$ by
                \obv{subgroup_inv}.
            $\groupInv[G](\groupCommutator{G}{b}{a})=
                \groupCommutator{G}{a}{b}$ by
                \obv{group_commutator_inverse}.
        \end{subproof}
        Follows by \auto{relative_commutator_subgroup_least}.
    \end{subproof}
    Show that $S\subseteq R$.
    \begin{subproof}
        Show that for every $b\in B\inter\carrier[G]$ we have for every
            $a\in A\inter\carrier[G]$ we have
            $\groupCommutator{G}{b}{a}\in R$.
        \begin{subproof}
            Fix $b\in B\inter\carrier[G]$ and
                $a\in A\inter\carrier[G]$.
            $a,b\in\carrier[G]$ by \obv{inter_elim_right}.
            $\groupCommutator{G}{a}{b}\in R$ by
                \obv{relative_commutator_subgroup_contains_commutators}.
            Hence
                $\groupInv[G](\groupCommutator{G}{a}{b})\in R$ by
                \obv{subgroup_inv}.
            $\groupInv[G](\groupCommutator{G}{a}{b})=
                \groupCommutator{G}{b}{a}$ by
                \obv{group_commutator_inverse}.
        \end{subproof}
        Follows by \auto{relative_commutator_subgroup_least}.
    \end{subproof}
    $R=S$ by \obv{subseteq_antisymmetric}.
\end{proof}

\subsection{Naturality}

\begin{proposition}\label{group_hom_image_relative_commutator_carrier}
    Suppose $\groupHom{f}{G}{H}$.
    Then
        $\img{f}{\relativeCommutatorCarrier{G}{A}{B}}=
            \relativeCommutatorCarrier{H}{\img{f}{A}}{\img{f}{B}}$.
\end{proposition}
\begin{proof}
    $G$ is a group and $H$ is a group by
        \obv{group_hom_source_group,group_hom_target_group}.
    $f$ is a function and $\dom{f}=\carrier[G]$ by
        \obv{group_hom_type,funs_elim}.
    Show that for every $z$ we have
        $z\in\img{f}{\relativeCommutatorCarrier{G}{A}{B}}$ iff
        $z\in\relativeCommutatorCarrier{H}{\img{f}{A}}{\img{f}{B}}$.
    \begin{subproof}
        Fix $z$.
        Show that if
            $z\in\img{f}{\relativeCommutatorCarrier{G}{A}{B}}$, then
            $z\in\relativeCommutatorCarrier{H}{\img{f}{A}}{\img{f}{B}}$.
        \begin{subproof}
            Assume
                $z\in\img{f}{\relativeCommutatorCarrier{G}{A}{B}}$.
            Take
                $c\in\dom{f}\inter
                    \relativeCommutatorCarrier{G}{A}{B}$ such that
                $z=f(c)$ by \obv{img_of_function_elim}.
            $c\in\relativeCommutatorCarrier{G}{A}{B}$ by
                \obv{inter_elim_right}.
            Take $a\in A\inter\carrier[G]$ such that there exists
                $b\in B\inter\carrier[G]$ such that
                $c=\groupCommutator{G}{a}{b}$ by
                \auto{relative_commutator_carrier_iff}.
            Take $b\in B\inter\carrier[G]$ such that
                $c=\groupCommutator{G}{a}{b}$.
            $a,b\in\carrier[G]$ by \obv{inter_elim_right}.
            $a\in\dom{f}\inter A$ and $b\in\dom{f}\inter B$ by
                \obv{inter_intro,inter_elim_left}.
            Hence $f(a)\in\img{f}{A}$ and $f(b)\in\img{f}{B}$ by
                \obv{img_of_function_intro}.
            $f(a),f(b)\in\carrier[H]$ by
                \obv{group_hom_apply_type}.
            Hence
                $f(a)\in\img{f}{A}\inter\carrier[H]$ and
                $f(b)\in\img{f}{B}\inter\carrier[H]$ by
                \obv{inter_intro}.
            $f(c)=\groupCommutator{H}{f(a)}{f(b)}$ by
                \auto{group_hom_commutator}.
            $\groupCommutator{H}{f(a)}{f(b)}\in\carrier[H]$ by
                \obv{group_commutator_type}.
            Follows by \auto{relative_commutator_carrier_iff}.
        \end{subproof}
        Show that if
            $z\in\relativeCommutatorCarrier{H}{\img{f}{A}}{\img{f}{B}}$,
            then $z\in\img{f}{\relativeCommutatorCarrier{G}{A}{B}}$.
        \begin{subproof}
            Assume
                $z\in\relativeCommutatorCarrier{H}{\img{f}{A}}{\img{f}{B}}$.
            Take $p\in\img{f}{A}\inter\carrier[H]$ such that there exists
                $q\in\img{f}{B}\inter\carrier[H]$ such that
                $z=\groupCommutator{H}{p}{q}$ by
                \auto{relative_commutator_carrier_iff}.
            Take $q\in\img{f}{B}\inter\carrier[H]$ such that
                $z=\groupCommutator{H}{p}{q}$.
            $p\in\img{f}{A}$ and $q\in\img{f}{B}$ by
                \obv{inter_elim_left}.
            Take $a\in\dom{f}\inter A$ such that $p=f(a)$ by
                \obv{img_of_function_elim}.
            Take $b\in\dom{f}\inter B$ such that $q=f(b)$ by
                \obv{img_of_function_elim}.
            $a\in A\inter\carrier[G]$ and
                $b\in B\inter\carrier[G]$ by
                \obv{inter_intro,inter_elim_left,inter_elim_right}.
            Hence $a,b\in\carrier[G]$ by \obv{inter_elim_right}.
            Let $c=\groupCommutator{G}{a}{b}$.
            $c\in\carrier[G]$ by \auto{group_commutator_type}.
            Hence $c\in\relativeCommutatorCarrier{G}{A}{B}$ by
                \auto{relative_commutator_carrier_iff}.
            $f(\groupCommutator{G}{a}{b})=
                \groupCommutator{H}{f(a)}{f(b)}$ by
                \obv{group_hom_commutator}.
            Hence $f(c)=\groupCommutator{H}{f(a)}{f(b)}$.
            Hence $f(c)=z$.
            $c\in\dom{f}$.
            $c\in\dom{f}\inter\relativeCommutatorCarrier{G}{A}{B}$ by
                \obv{inter_intro}.
            Hence
                $f(c)\in\img{f}{\relativeCommutatorCarrier{G}{A}{B}}$ by
                \obv{img_of_function_intro}.
        \end{subproof}
        Follows by \auto.
    \end{subproof}
    Follows by \obv{setext}.
\end{proof}

\begin{theorem}\label{group_hom_image_relative_commutator_subgroup}
    Let $G$ be a group.
    Let $H$ be a group.
    Suppose $\groupHom{f}{G}{H}$.
    Then
        $\img{f}{\relativeCommutatorSubgroup{G}{A}{B}}=
            \relativeCommutatorSubgroup{H}{\img{f}{A}}{\img{f}{B}}$.
\end{theorem}
\begin{proof}
    $f\in\funs{\carrier[G]}{\carrier[H]}$ by
        \obv{group_hom_type}.
    $\relativeCommutatorCarrier{G}{A}{B}\subseteq\carrier[G]$ by
        \obv{relative_commutator_carrier_subseteq}.
    $\img{f}{\relativeCommutatorSubgroup{G}{A}{B}}=
        \generatedSubgroup{
            \img{f}{\relativeCommutatorCarrier{G}{A}{B}}}{H}$ by
        \auto{relative_commutator_subgroup,
            group_hom_image_generated_subgroup}.
    Follows by
        \auto{group_hom_image_relative_commutator_carrier,
            relative_commutator_subgroup}.
\end{proof}

\begin{theorem}\label{relative_commutator_subgroup_conjugate}
    Let $G$ be a group.
    Let $g\in\carrier[G]$.
    Then
        $\conjugateSubset{G}{g}{
            \relativeCommutatorSubgroup{G}{A}{B}}=
          \relativeCommutatorSubgroup{G}
            {\conjugateSubset{G}{g}{A}}
            {\conjugateSubset{G}{g}{B}}$.
\end{theorem}
\begin{proof}
    Let $f=\ConjugationMap{G}{g}$ and
        $R=\relativeCommutatorSubgroup{G}{A}{B}$.
    $\groupHom{\ConjugationMap{G}{g}}{G}{G}$ by
        \obv{conjugation_map_group_hom}.
    Hence $\groupHom{f}{G}{G}$.
    $f$ is a function and $\dom{f}=\carrier[G]$ by
        \auto{group_hom_type,funs_elim}.
    $\subgroup{\relativeCommutatorSubgroup{G}{A}{B}}{G}$ by
        \obv{relative_commutator_subgroup_is_subgroup}.
    Hence $\subgroup{R}{G}$.
    Hence $R\subseteq\carrier[G]$ by \obv{subgroup_subseteq}.
    Hence $R\inter\carrier[G]=R$ by
        \obv{inter_absorb_supseteq_right}.
    Hence $\conjugateSubset{G}{g}{R}=\img{f}{R}$ by
        \obv{conjugate_subset}.
    Show that $\img{f}{A}=\conjugateSubset{G}{g}{A}$.
    \begin{subproof}
        Show that for every $y$ we have
            $y\in\img{f}{A}$ iff
            $y\in\conjugateSubset{G}{g}{A}$.
        \begin{subproof}
            Fix $y$.
            Follows by
                \auto{conjugate_subset,img_of_function,
                    inter_comm,inter_assoc,inter_idempotent}.
        \end{subproof}
        Follows by \obv{setext}.
    \end{subproof}
    Show that $\img{f}{B}=\conjugateSubset{G}{g}{B}$.
    \begin{subproof}
        Show that for every $y$ we have
            $y\in\img{f}{B}$ iff
            $y\in\conjugateSubset{G}{g}{B}$.
        \begin{subproof}
            Fix $y$.
            Follows by
                \auto{conjugate_subset,img_of_function,
                    inter_comm,inter_assoc,inter_idempotent}.
        \end{subproof}
        Follows by \obv{setext}.
    \end{subproof}
    Follows by
        \auto{group_hom_image_relative_commutator_subgroup}.
\end{proof}

\subsection{Normalizers and normal subgroups}

\begin{theorem}\label{relative_commutator_subgroup_le_of_subset_normalizer}
    Let $G$ be a group.
    Suppose $\subgroup{A}{G}$ and
        $B\subseteq\groupNormalizer{G}{A}$.
    Then $\relativeCommutatorSubgroup{G}{A}{B}\subseteq A$.
\end{theorem}
\begin{proof}
    $G$ is a group and $A\subseteq\carrier[G]$ by
        \obv{subgroup_ambient_group,subgroup_subseteq}.
    $\subgroup{\groupNormalizer{G}{A}}{G}$ by
        \obv{group_normalizer_subgroup}.
    Show that for every $a\in A\inter\carrier[G]$ we have for every
        $b\in B\inter\carrier[G]$ we have
        $\groupCommutator{G}{a}{b}\in A$.
    \begin{subproof}
        Fix $a\in A\inter\carrier[G]$ and
            $b\in B\inter\carrier[G]$.
        $a\in A$ and $a,b\in\carrier[G]$ by
            \obv{inter_elim_left,inter_elim_right}.
        $b\in\groupNormalizer{G}{A}$ by
            \obv{inter_elim_left,subseteq}.
        Hence $\groupInv[G](b)\in\groupNormalizer{G}{A}$ by
            \obv{subgroup_inv}.
        Hence
            $\conjugateSubset{G}{\groupInv[G](b)}{A}=A$ by
            \obv{group_normalizer}.
        $\groupInv[G](b)\in\carrier[G]$ by
            \obv{group_inverse_type}.
        $a\in A\inter\carrier[G]$ by \obv{inter_intro}.
        Hence
            $\conjugate{G}{\groupInv[G](b)}{a}\in
                \conjugateSubset{G}{\groupInv[G](b)}{A}$ by
            \auto{conjugate_subset_iff}.
        Hence $\conjugate{G}{\groupInv[G](b)}{a}\in A$.
        $\groupInv[G](a)\in A$ by \obv{subgroup_inv}.
        Hence
            $\mul[G](\groupInv[G](a),
                \conjugate{G}{\groupInv[G](b)}{a})\in A$ by
            \obv{subgroup_mul}.
        Follows by \obv{group_commutator_eq_inverse_mul_conjugate}.
    \end{subproof}
    Follows by \obv{relative_commutator_subgroup_least}.
\end{proof}

\begin{theorem}\label{subgroup_subset_normalizer_of_relative_commutator_le}
    Let $G$ be a group.
    Suppose $\subgroup{A}{G}$ and $\subgroup{B}{G}$ and
        $\relativeCommutatorSubgroup{G}{A}{B}\subseteq A$.
    Then $B\subseteq\groupNormalizer{G}{A}$.
\end{theorem}
\begin{proof}
    $G$ is a group and $A,B\subseteq\carrier[G]$ by
        \obv{subgroup_ambient_group,subgroup_subseteq}.
    Show that for every $b\in B$ we have
        $b\in\groupNormalizer{G}{A}$.
    \begin{subproof}
        Fix $b\in B$.
        Hence $b\in\carrier[G]$ by \obv{subseteq}.
        Let $v=\groupInv[G](b)$.
        $\groupInv[G](b)\in B$ by \obv{subgroup_inv}.
        Hence $v\in B$.
        Hence $v\in\carrier[G]$ by \obv{subseteq}.
        $\groupInv[G](v)=b$ by
            \auto{group_inverse_involutive}.
        Show that $\conjugateSubset{G}{b}{A}\subseteq A$.
        \begin{subproof}
            Show that for every $y\in\conjugateSubset{G}{b}{A}$ we have
                $y\in A$.
            \begin{subproof}
                Fix $y\in\conjugateSubset{G}{b}{A}$.
                Take $a\in A\inter\carrier[G]$ such that
                    $y=\conjugate{G}{b}{a}$ by
                    \obv{conjugate_subset_iff}.
                $a\in A$ and $a\in\carrier[G]$ by
                    \obv{inter_elim_left,inter_elim_right}.
                $v\in B\inter\carrier[G]$ by \obv{inter_intro}.
                $\groupCommutator{G}{a}{v}\in
                    \relativeCommutatorSubgroup{G}{A}{B}$ by
                    \obv{relative_commutator_subgroup_contains_commutators}.
                Hence $\groupCommutator{G}{a}{v}\in A$ by
                    \obv{subseteq}.
                Hence
                    $\mul[G](a,\groupCommutator{G}{a}{v})\in A$ by
                    \obv{subgroup_mul}.
                $\conjugate{G}{\groupInv[G](v)}{a}=
                    \mul[G](a,\groupCommutator{G}{a}{v})$ by
                    \obv{group_conjugate_inverse_eq_mul_commutator}.
                Hence $\conjugate{G}{b}{a}=
                    \mul[G](a,\groupCommutator{G}{a}{v})$.
            \end{subproof}
            Follows by \obv{subseteq}.
        \end{subproof}
        Show that $A\subseteq\conjugateSubset{G}{b}{A}$.
        \begin{subproof}
            Show that for every $y\in A$ we have
                $y\in\conjugateSubset{G}{b}{A}$.
            \begin{subproof}
                Fix $y\in A$.
                Hence $y\in\carrier[G]$ by \obv{subseteq}.
                $b\in B\inter\carrier[G]$ by \obv{inter_intro}.
                $y\in A\inter\carrier[G]$ by \obv{inter_intro}.
                $\groupCommutator{G}{y}{b}\in
                    \relativeCommutatorSubgroup{G}{A}{B}$ by
                    \obv{relative_commutator_subgroup_contains_commutators}.
                Hence $\groupCommutator{G}{y}{b}\in A$ by
                    \obv{subseteq}.
                Let $x=\conjugate{G}{v}{y}$.
                $\conjugate{G}{\groupInv[G](b)}{y}=
                    \mul[G](y,\groupCommutator{G}{y}{b})$ by
                    \obv{group_conjugate_inverse_eq_mul_commutator}.
                Hence $x=\mul[G](y,\groupCommutator{G}{y}{b})$.
                $\mul[G](y,\groupCommutator{G}{y}{b})\in A$ by
                    \obv{subgroup_mul}.
                Hence $x\in A$.
                Hence $x\in\carrier[G]$ by \obv{subseteq}.
                $x\in A\inter\carrier[G]$ by \obv{inter_intro}.
                $\conjugate{G}{b}{x}=y$ by
                    \auto{group_conjugate_inverse_right}.
                Follows by \auto{conjugate_subset_iff}.
            \end{subproof}
            Follows by \obv{subseteq}.
        \end{subproof}
        Hence $\conjugateSubset{G}{b}{A}=A$ by
            \obv{subseteq_antisymmetric}.
        Follows by \obv{group_normalizer}.
    \end{subproof}
    Follows by \obv{subseteq}.
\end{proof}

\begin{lemma}\label{left_subgroup_conjugates_relative_commutator_generator}
    Let $G$ be a group.
    Suppose $\subgroup{A}{G}$.
    Let $h,a\in A$.
    Let $b\in B\inter\carrier[G]$.
    Then
        $\conjugate{G}{h}{\groupCommutator{G}{a}{b}}\in
            \relativeCommutatorSubgroup{G}{A}{B}$.
\end{lemma}
\begin{proof}
    Let $R=\relativeCommutatorSubgroup{G}{A}{B}$.
    $A\subseteq\carrier[G]$ by \obv{subgroup_subseteq}.
    Hence $h,a\in\carrier[G]$ by \obv{subseteq}.
    $b\in\carrier[G]$ by \obv{inter_elim_right}.
    Let $u=\groupInv[G](h)$ and $p=\mul[G](a,u)$.
    $\groupInv[G](h)\in A$ by \obv{subgroup_inv}.
    Hence $u\in A$.
    $\mul[G](a,u)\in A$ by \obv{subgroup_mul}.
    Hence $p\in A$.
    Hence $p,u\in\carrier[G]$ by \obv{subseteq}.
    $p,u\in A\inter\carrier[G]$ by \obv{inter_intro}.
    $\subgroup{R}{G}$ by
        \auto{relative_commutator_subgroup_is_subgroup}.
    $\groupCommutator{G}{p}{b}\in R$ and
        $\groupCommutator{G}{u}{b}\in R$ by
        \obv{relative_commutator_subgroup_contains_commutators}.
    Hence $\groupInv[G](\groupCommutator{G}{u}{b})\in R$ by
        \obv{subgroup_inv}.
    Hence
        $\mul[G](
            \groupCommutator{G}{p}{b},
            \groupInv[G](\groupCommutator{G}{u}{b}))\in R$ by
        \obv{subgroup_mul}.
    $\groupInv[G](u)=h$ by
        \auto{group_inverse_involutive}.
    Hence
        $\groupCommutator{G}{p}{b}=
            \mul[G](
                \conjugate{G}{h}{\groupCommutator{G}{a}{b}},
                \groupCommutator{G}{u}{b})$ by
        \auto{group_commutator_mul_left}.
    $\conjugate{G}{h}{\groupCommutator{G}{a}{b}},
        \groupCommutator{G}{u}{b},
        \groupCommutator{G}{p}{b}\in\carrier[G]$ by
        \obv{group_conjugate_type,group_commutator_type}.
    Hence
        $\conjugate{G}{h}{\groupCommutator{G}{a}{b}}=
            \mul[G](
                \groupCommutator{G}{p}{b},
                \groupInv[G](\groupCommutator{G}{u}{b}))$ by
        \auto{group_solve_right}.
\end{proof}

        $A\inter\carrier[G]\subseteq\groupCentralizer{G}{B}$.
\end{theorem}
\begin{proof}
    Let $C=\relativeCommutatorCarrier{G}{A}{B}$ and
        $R=\relativeCommutatorSubgroup{G}{A}{B}$.
    $C\subseteq\carrier[G]$ by
        \auto{relative_commutator_carrier_subseteq}.
    Show that if $R=\trivialSubgroupCarrier{G}$, then
        $A\inter\carrier[G]\subseteq\groupCentralizer{G}{B}$.
    \begin{subproof}
        Assume $R=\trivialSubgroupCarrier{G}$.
        Show that for every $a\in A\inter\carrier[G]$ we have
            $a\in\groupCentralizer{G}{B}$.
        \begin{subproof}
            Fix $a\in A\inter\carrier[G]$.
            Hence $a\in\carrier[G]$ by \obv{inter_elim_right}.
            Show that for every $b\in B\inter\carrier[G]$ we have
                $\commutesIn{G}{a}{b}$.
            \begin{subproof}
                Fix $b\in B\inter\carrier[G]$.
                Hence $b\in\carrier[G]$ by \obv{inter_elim_right}.
                $\groupCommutator{G}{a}{b}\in R$ by
                    \obv{relative_commutator_subgroup_contains_commutators}.
                Hence
                    $\groupCommutator{G}{a}{b}\in
                        \trivialSubgroupCarrier{G}$.
                Hence $\groupCommutator{G}{a}{b}=\neutral[G]$ by
                    \obv{trivial_subgroup_carrier,singleton_iff}.
                Follows by
                    \obv{group_commutator_neutral_iff_commutes}.
            \end{subproof}
            Follows by \obv{group_centralizer}.
        \end{subproof}
        Follows by \obv{subseteq}.
    \end{subproof}
    Show that if
        $A\inter\carrier[G]\subseteq\groupCentralizer{G}{B}$, then
        $R=\trivialSubgroupCarrier{G}$.
    \begin{subproof}
        Assume
            $A\inter\carrier[G]\subseteq\groupCentralizer{G}{B}$.
        Show that $C\subseteq\trivialSubgroupCarrier{G}$.
        \begin{subproof}
            Show that for every $z\in C$ we have
                $z\in\trivialSubgroupCarrier{G}$.
            \begin{subproof}
                Fix $z\in C$.
                Take $a\in A\inter\carrier[G]$ such that there exists
                    $b\in B\inter\carrier[G]$ such that
                    $z=\groupCommutator{G}{a}{b}$ by
                    \auto{relative_commutator_carrier_iff}.
                Take $b\in B\inter\carrier[G]$ such that
                    $z=\groupCommutator{G}{a}{b}$.
                $a\in\groupCentralizer{G}{B}$ by \obv{subseteq}.
                Hence $\commutesIn{G}{a}{b}$ by
                    \obv{group_centralizer}.
                $a,b\in\carrier[G]$ by \obv{inter_elim_right}.
                Hence $\groupCommutator{G}{a}{b}=\neutral[G]$ by
                    \obv{group_commutator_neutral_iff_commutes}.
                Hence $z=\neutral[G]$.
                Follows by
                    \obv{trivial_subgroup_carrier,singleton_iff}.
            \end{subproof}
            Follows by \obv{subseteq}.
        \end{subproof}
        Hence
            $\generatedSubgroup{C}{G}=\trivialSubgroupCarrier{G}$ by
            \obv{generated_subgroup_eq_trivial_iff}.
        $R=\generatedSubgroup{C}{G}$ by
            \obv{relative_commutator_subgroup}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{corollary}\label{relative_commutator_trivial_iff_subset_centralizer}
    Let $G$ be a group.
    Suppose $A\subseteq\carrier[G]$.
    Then
        $\relativeCommutatorSubgroup{G}{A}{B}=
            \trivialSubgroupCarrier{G}$ iff
        $A\subseteq\groupCentralizer{G}{B}$.
\end{corollary}
\begin{proof}
    $A\inter\carrier[G]=A$ by
        \obv{inter_absorb_supseteq_right}.
    Follows by
        \auto{relative_commutator_trivial_iff_trace_subset_centralizer}.
\end{proof}

\begin{theorem}\label{relative_commutator_trivial_iff_right_trace_subset_centralizer}
    Let $G$ be a group.
    Then
        $\relativeCommutatorSubgroup{G}{A}{B}=
            \trivialSubgroupCarrier{G}$ iff
        $B\inter\carrier[G]\subseteq\groupCentralizer{G}{A}$.
\end{theorem}
\begin{proof}
    $\relativeCommutatorSubgroup{G}{A}{B}=
        \relativeCommutatorSubgroup{G}{B}{A}$ by
        \obv{relative_commutator_subgroup_symmetric}.
    $\relativeCommutatorSubgroup{G}{B}{A}=
        \trivialSubgroupCarrier{G}$ iff
        $B\inter\carrier[G]\subseteq\groupCentralizer{G}{A}$ by
        \obv{relative_commutator_trivial_iff_trace_subset_centralizer}.
    Follows by \auto.
\end{proof}
