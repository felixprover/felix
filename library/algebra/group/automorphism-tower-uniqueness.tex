\section{Uniqueness of canonical automorphism-tower segments}

Canonical segments use literal constructors at every branch.  Their
uniqueness is correspondingly literal.  The bounded agreement predicates
below separate the transfinite argument from function and record
extensionality.

\begin{definition}\label{tower_segments_agree_at}
    $\towerSegmentsAgreeAt{T}{U}{\alpha}$ iff
        $\towerStage{T}{\alpha}=\towerStage{U}{\alpha}$ and
        for every $\beta\in\suc{\alpha}$ we have
            $\towerBond{T}{\beta}{\alpha}=
                \towerBond{U}{\beta}{\alpha}$.
\end{definition}

\begin{definition}\label{tower_segments_agree_below}
    $\towerSegmentsAgreeBelow{T}{U}{\alpha}$ iff
        for every $\gamma\in\alpha$ we have
            $\towerSegmentsAgreeAt{T}{U}{\gamma}$.
\end{definition}

\begin{proposition}\label{tower_segments_agree_at_stage}
    Suppose $\towerSegmentsAgreeAt{T}{U}{\alpha}$.
    Then $\towerStage{T}{\alpha}=\towerStage{U}{\alpha}$.
\end{proposition}
\begin{proof}
    Follows by \obv{tower_segments_agree_at}.
\end{proof}

\begin{proposition}\label{tower_segments_agree_at_bond}
    Suppose $\towerSegmentsAgreeAt{T}{U}{\alpha}$.
    Let $\beta\in\suc{\alpha}$.
    Then $\towerBond{T}{\beta}{\alpha}=
        \towerBond{U}{\beta}{\alpha}$.
\end{proposition}
\begin{proof}
    Follows by \obv{tower_segments_agree_at}.
\end{proof}

\begin{proposition}\label{tower_segments_agree_below_at}
    Suppose $\towerSegmentsAgreeBelow{T}{U}{\alpha}$.
    Let $\gamma\in\alpha$.
    Then $\towerSegmentsAgreeAt{T}{U}{\gamma}$.
\end{proposition}
\begin{proof}
    Follows by \obv{tower_segments_agree_below}.
\end{proof}

\subsection{Equality of earlier systems}

\begin{proposition}\label{tower_agreement_stage_restrictions}
    Suppose $\towerSegmentDiagram{T}{G}{\theta}$ and
        $\towerSegmentDiagram{U}{G}{\theta}$.
    Suppose $\alpha\subseteq\theta$.
    Suppose $\towerSegmentsAgreeBelow{T}{U}{\alpha}$.
    Then $\restrl{\towerStages[T]}{\alpha}=
        \restrl{\towerStages[U]}{\alpha}$.
\end{proposition}
\begin{proof}
    $\towerStages[T]$ and $\towerStages[U]$ are functions and
        $\dom{\towerStages[T]}=\theta=\dom{\towerStages[U]}$ by
        \auto{tower_segment_diagram_data,funs_is_function,funs_elim}.
    Show that for every $\gamma\in\alpha$ we have
        $\apply{\towerStages[T]}{\gamma}=
        \apply{\towerStages[U]}{\gamma}$.
    \begin{subproof}
        Fix $\gamma\in\alpha$.
        $\towerSegmentsAgreeAt{T}{U}{\gamma}$ by
            \obv{tower_segments_agree_below_at}.
        Hence $\towerStage{T}{\gamma}=\towerStage{U}{\gamma}$ by
            \obv{tower_segments_agree_at_stage}.
        Follows by \obv{tower_stage}.
    \end{subproof}
    Follows by \auto{restrictions_agree,subseteq,union_iff,inter}.
\end{proof}

\begin{proposition}\label{tower_agreement_bond_value}
    Suppose $\towerSegmentsAgreeBelow{T}{U}{\alpha}$.
    Let $\alpha$ be an ordinal.
    Let $\beta,\gamma\in\alpha$.
    Suppose $\beta\subseteq\gamma$.
    Then $\towerBond{T}{\beta}{\gamma}=
        \towerBond{U}{\beta}{\gamma}$.
\end{proposition}
\begin{proof}
    $\towerSegmentsAgreeAt{T}{U}{\gamma}$ by
        \obv{tower_segments_agree_below_at}.
    $\beta$ and $\gamma$ are ordinals by
        \obv{elem_of_ordinal_is_ordinal}.
    Hence $\beta\in\gamma$ or $\beta=\gamma$ by
        \obv{ordinal_subseteq_implies_elem_or_eq}.
    Hence $\beta\in\suc{\gamma}$ by \auto{suc_iff}.
    Follows by \obv{tower_segments_agree_at_bond}.
\end{proof}

\begin{proposition}\label{tower_agreement_bond_restrictions}
    Suppose $\towerSegmentDiagram{T}{G}{\theta}$ and
        $\towerSegmentDiagram{U}{G}{\theta}$.
    Let $\alpha$ be an ordinal.
    Suppose $\alpha\subseteq\theta$.
    Suppose $\towerSegmentsAgreeBelow{T}{U}{\alpha}$.
    Then $\restrl{\towerBonds[T]}{\subseteqrel{\alpha}}=
        \restrl{\towerBonds[U]}{\subseteqrel{\alpha}}$.
\end{proposition}
\begin{proof}
    $\towerBonds[T]$ and $\towerBonds[U]$ are functions and
        $\dom{\towerBonds[T]}=\subseteqrel{\theta}=
            \dom{\towerBonds[U]}$ by
        \auto{tower_segment_diagram_data,funs_is_function,funs_elim}.
    $\subseteqrel{\alpha}\subseteq\subseteqrel{\theta}$ by
        \obv{subseteqrel_monotone_for_tower}.
    Show that for every $z\in\subseteqrel{\alpha}$ we have
        $\apply{\towerBonds[T]}{z}=\apply{\towerBonds[U]}{z}$.
    \begin{subproof}
        Fix $z\in\subseteqrel{\alpha}$.
        Take $\beta,\gamma\in\alpha$ such that
            $z=(\beta,\gamma)$ and $\beta\subseteq\gamma$ by
            \auto{subseteqrel}.
        $\towerBond{T}{\beta}{\gamma}=
            \towerBond{U}{\beta}{\gamma}$ by
            \obv{tower_agreement_bond_value}.
        Follows by \auto{tower_bond}.
    \end{subproof}
    Follows by \auto{restrictions_agree,subseteq,union_iff,inter}.
\end{proof}

\begin{proposition}\label{tower_earlier_system_congruence}
    Suppose $\restrl{\towerStages[T]}{\alpha}=
        \restrl{\towerStages[U]}{\alpha}$.
    Suppose $\restrl{\towerBonds[T]}{\subseteqrel{\alpha}}=
        \restrl{\towerBonds[U]}{\subseteqrel{\alpha}}$.
    Then $\towerEarlierSystem{T}{\alpha}=
        \towerEarlierSystem{U}{\alpha}$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{group_direct_system_restriction,
            tower_segment_system_groups,
            tower_segment_system_bonds,
            ordinal_directed_quasiorder_carrier,
            ordinal_directed_quasiorder_relation}.
\end{proof}

\begin{theorem}\label{tower_agreement_earlier_systems}
    Suppose $\towerSegmentDiagram{T}{G}{\theta}$ and
        $\towerSegmentDiagram{U}{G}{\theta}$.
    Let $\alpha$ be an ordinal.
    Suppose $\alpha\subseteq\theta$.
    Suppose $\towerSegmentsAgreeBelow{T}{U}{\alpha}$.
    Then $\towerEarlierSystem{T}{\alpha}=
        \towerEarlierSystem{U}{\alpha}$.
\end{theorem}
\begin{proof}
    $\restrl{\towerStages[T]}{\alpha}=
        \restrl{\towerStages[U]}{\alpha}$ by
        \obv{tower_agreement_stage_restrictions}.
    $\restrl{\towerBonds[T]}{\subseteqrel{\alpha}}=
        \restrl{\towerBonds[U]}{\subseteqrel{\alpha}}$ by
        \obv{tower_agreement_bond_restrictions}.
    Follows by \obv{tower_earlier_system_congruence}.
\end{proof}

\begin{proposition}\label{tower_agreement_from_ordinal_cases}
    Let $\alpha$ be an ordinal.
    Suppose $\alpha=\emptyset\implies
        \towerSegmentsAgreeAt{T}{U}{\alpha}$.
    Suppose if $\alpha$ is a limit ordinal then
        $\towerSegmentsAgreeAt{T}{U}{\alpha}$.
    Suppose if $\alpha$ is a successor ordinal then
        $\towerSegmentsAgreeAt{T}{U}{\alpha}$.
    Then $\towerSegmentsAgreeAt{T}{U}{\alpha}$.
\end{proposition}
\begin{proof}
    $\alpha=\emptyset$ or $\emptyset\in\alpha$ by
        \obv{ordinal_empty_or_emptyset_elem}.
    If $\emptyset\in\alpha$ then
        $\emptyset\precedes\alpha$.
    Hence if $\emptyset\in\alpha$ then $\alpha$ is a limit ordinal or
        $\alpha$ is a successor ordinal by
        \obv{positive_ordinal_is_limit_or_successor}.
\end{proof}

\begin{theorem}\label{canonical_tower_segments_agree_at}

        $\canonicalTowerSegment{U}{G}{\theta}$.
    Suppose $\alpha\in\theta$.

\begin{proof}[Proof by transfinite induction]
    $\theta$ is an ordinal by
        \auto{canonical_tower_comparison_context}.
    Show that $\towerSegmentsAgreeBelow{T}{U}{\alpha}$.
    \begin{subproof}

            $\towerSegmentsAgreeAt{T}{U}{\gamma}$.
        \begin{subproof}

                \obv{ordinal_transitivity}.

        Follows by \obv{tower_segments_agree_below}.
    \end{subproof}
    Follows by \obv{canonical_tower_agreement_step}.
\end{proof}
