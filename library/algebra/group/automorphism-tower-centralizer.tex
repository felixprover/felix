\section{Centralizers of adjacent automorphism-tower stages}

At an immediate successor, the preceding stage is represented by its inner
automorphisms.  For a centreless stage this image therefore has trivial
centralizer.  Transporting the adjacent pair into a common later stage gives
the fixed-ambient form used below.

\begin{theorem}\label{segment_image_immediate_successor_centralizer_trivial}
    Suppose $\towerSegmentDiagram{T}{G}{\theta}$.
    Let $\alpha\in\theta$.
    Suppose $\suc{\alpha}\in\theta$ and
        $\centrelessGroup{\towerStage{T}{\alpha}}$.
    Then
        $\groupCentralizer
            {\towerStage{T}{\suc{\alpha}}}
            {\segmentImage{T}{\alpha}{\suc{\alpha}}}
        =\trivialSubgroupCarrier{
            \towerStage{T}{\suc{\alpha}}}$.
\end{theorem}
\begin{proof}
    $\groupCentralizer
        {\automorphismGroup{\towerStage{T}{\alpha}}}
        {\innerAutomorphismCarrier{\towerStage{T}{\alpha}}}
        =\trivialSubgroupCarrier{
            \automorphismGroup{\towerStage{T}{\alpha}}}$ by
        \obv{inner_automorphism_centralizer_trivial}.
    $\towerStage{T}{\suc{\alpha}}=
        \automorphismGroup{\towerStage{T}{\alpha}}$ by
        \auto{tower_segment_successor}.
    $\segmentImage{T}{\alpha}{\suc{\alpha}}=
        \innerAutomorphismCarrier{\towerStage{T}{\alpha}}$ by
        \obv{segment_image_immediate_successor}.
\end{proof}

\begin{theorem}\label{segment_image_successor_centralizer_trivial_in_later_image}
    Suppose $\towerSegmentDiagram{T}{G}{\theta}$.
    Let $\alpha,\gamma\in\theta$.
    Suppose $\suc{\alpha}\subseteq\gamma$.
    Suppose $\centrelessGroup{\towerStage{T}{\alpha}}$.
    Suppose
        $\groupEmbedding
            {\towerBond{T}{\suc{\alpha}}{\gamma}}
            {\towerStage{T}{\suc{\alpha}}}
            {\towerStage{T}{\gamma}}$.
    Then
        $\groupCentralizer
            {\subgroupStructure
                {\towerStage{T}{\gamma}}
                {\segmentImage{T}{\suc{\alpha}}{\gamma}}}
            {\segmentImage{T}{\alpha}{\gamma}}
        =\trivialSubgroupCarrier{
            \subgroupStructure
                {\towerStage{T}{\gamma}}
                {\segmentImage{T}{\suc{\alpha}}{\gamma}}}$.
\end{theorem}
\begin{proof}
    Hence $\alpha$ and $\gamma$ are ordinals by
        \obv{elem_of_ordinal_is_ordinal}.
    Hence $\suc{\alpha}$ is an ordinal by \obv{suc_ordinal}.
    Hence $\suc{\alpha}\in\gamma$ or
        $\suc{\alpha}=\gamma$ by
        \obv{ordinal_subseteq_implies_elem_or_eq}.
    Hence $\suc{\alpha}\in\theta$ by
        \obv{ordinal_transitivity}.
    Let $f=\towerBond{T}{\suc{\alpha}}{\gamma}$.
    Let $K=\subgroupStructure
        {\towerStage{T}{\gamma}}
        {\segmentImage{T}{\suc{\alpha}}{\gamma}}$.
    $\groupCentralizer
        {\towerStage{T}{\suc{\alpha}}}
        {\segmentImage{T}{\alpha}{\suc{\alpha}}}
        =\trivialSubgroupCarrier{
            \towerStage{T}{\suc{\alpha}}}$ by
        \obv{segment_image_immediate_successor_centralizer_trivial}.
    $\groupIsomorphism
        {\towerBond{T}{\suc{\alpha}}{\gamma}}
        {\towerStage{T}{\suc{\alpha}}}
        {\subgroupStructure
            {\towerStage{T}{\gamma}}
            {\img{\towerBond{T}{\suc{\alpha}}{\gamma}}
                {\carrier[\towerStage{T}{\suc{\alpha}}]}}}$ by
        \obv{group_embedding_image_isomorphism}.
    $\segmentImage{T}{\suc{\alpha}}{\gamma}=
        \img{\towerBond{T}{\suc{\alpha}}{\gamma}}
            {\carrier[\towerStage{T}{\suc{\alpha}}]}$ by
        \obv{segment_image}.
    Hence
        $\groupIsomorphism
            {f}
            {\towerStage{T}{\suc{\alpha}}}
            {K}$.
    Hence
        $\groupCentralizer
            {K}
            {\img{f}{\segmentImage{T}{\alpha}{\suc{\alpha}}}}
        =\trivialSubgroupCarrier{K}$ by
        \obv{group_isomorphism_trivial_centralizer_image}.
    $\alpha\subseteq\suc{\alpha}$ by
        \obv{subseteq_self_suc_intro}.
    $\segmentImage{T}{\alpha}{\gamma}=
        \img{f}{\segmentImage{T}{\alpha}{\suc{\alpha}}}$ by
        \auto{segment_image_factorization}.
\end{proof}

\begin{corollary}\label{canonical_segment_image_successor_centralizer_trivial_in_later_image}
    Suppose $\canonicalTowerSegment{T}{G}{\theta}$ and
        $\centrelessGroup{G}$.
    Let $\alpha,\gamma\in\theta$.
    Suppose $\suc{\alpha}\subseteq\gamma$.
    Then
        $\groupCentralizer
            {\subgroupStructure
                {\towerStage{T}{\gamma}}
                {\segmentImage{T}{\suc{\alpha}}{\gamma}}}
            {\segmentImage{T}{\alpha}{\gamma}}
        =\trivialSubgroupCarrier{
            \subgroupStructure
                {\towerStage{T}{\gamma}}
                {\segmentImage{T}{\suc{\alpha}}{\gamma}}}$.
\end{corollary}
\begin{proof}
    $\towerSegmentDiagram{T}{G}{\theta}$ by
        \obv{canonical_tower_segment_diagram}.
    $\theta$ is an ordinal by \obv{tower_segment_diagram_data}.
    Hence $\alpha$ and $\gamma$ are ordinals by
        \obv{elem_of_ordinal_is_ordinal}.
    Hence $\suc{\alpha}$ is an ordinal by \obv{suc_ordinal}.
    Hence $\suc{\alpha}\in\gamma$ or
        $\suc{\alpha}=\gamma$ by
        \obv{ordinal_subseteq_implies_elem_or_eq}.
    Hence $\suc{\alpha}\in\theta$ by
        \obv{ordinal_transitivity}.
    $\centrelessGroup{\towerStage{T}{\alpha}}$ by
        \obv{canonical_tower_stage_centreless}.
    $\groupEmbedding
        {\towerBond{T}{\suc{\alpha}}{\gamma}}
        {\towerStage{T}{\suc{\alpha}}}
        {\towerStage{T}{\gamma}}$ by
        \obv{canonical_tower_bond_embedding}.
    Follows by
        \obv{segment_image_successor_centralizer_trivial_in_later_image}.
\end{proof}
