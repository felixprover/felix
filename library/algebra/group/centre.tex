\import{algebra/group/centralizer.tex}

\section{Centres and centreless groups}

\begin{definition}\label{group_centre}
    $\groupCentre{G}=\groupCentralizer{G}{\carrier[G]}$.
\end{definition}

\begin{proposition}\label{group_centre_iff}
    $x\in\groupCentre{G}$ iff
        $x\in\carrier[G]$ and
        for every $y\in\carrier[G]$ we have
            $\commutesIn{G}{x}{y}$.
\end{proposition}
\begin{proof}
    $\carrier[G]\inter\carrier[G]=\carrier[G]$ by
        \obv{inter_idempotent}.
    Follows by \auto{group_centre,group_centralizer_iff}.
\end{proof}

\begin{proposition}\label{group_centre_subseteq}
    $\groupCentre{G}\subseteq\carrier[G]$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{group_centre,group_centralizer_subseteq}.
\end{proof}

\begin{theorem}\label{group_centre_subgroup}
    Let $G$ be a group.
    Then $\subgroup{\groupCentre{G}}{G}$.
\end{theorem}
\begin{proof}
    Follows by
        \auto{group_centre,group_centralizer_subgroup}.
\end{proof}

\begin{theorem}\label{group_centre_normal_subgroup}
    Let $G$ be a group.
    Then $\normalSubgroup{\groupCentre{G}}{G}$.
\end{theorem}
\begin{proof}
    $\subgroup{\groupCentre{G}}{G}$ by
        \obv{group_centre_subgroup}.
    Show that for all $a\in\carrier[G]$ we have for every
        $x\in\groupCentre{G}$ we have
        $\conjugate{G}{a}{x}\in\groupCentre{G}$.
    \begin{subproof}
        Fix $a\in\carrier[G]$.
        Fix $x\in\groupCentre{G}$.
        $x\in\carrier[G]$ and $\commutesIn{G}{a}{x}$ by
            \auto{group_centre_iff,commutes_in}.
        Hence $\conjugate{G}{a}{x}=x$ by
            \obv{commutes_in_iff_conjugate_fixed}.
    \end{subproof}
    Follows by \obv{normal_subgroup_iff_closed_under_conjugation}.
\end{proof}

\begin{definition}\label{centreless_group}
    $\centrelessGroup{G}$ iff
        $G$ is a group and
        $\groupCentre{G}=\trivialSubgroupCarrier{G}$.
\end{definition}

\begin{proposition}\label{centreless_group_is_group}
    Suppose $\centrelessGroup{G}$.
    Then $G$ is a group.
\end{proposition}
\begin{proof}
    Follows by \obv{centreless_group}.
\end{proof}

\begin{proposition}\label{centreless_group_centre}
    Suppose $\centrelessGroup{G}$.
    Then $\groupCentre{G}=\trivialSubgroupCarrier{G}$.
\end{proposition}
\begin{proof}
    Follows by \obv{centreless_group}.
\end{proof}

\subsection{Transport by a chosen isomorphism}

\begin{theorem}\label{group_isomorphism_centre_image_subseteq}
    Suppose $\groupIsomorphism{f}{G}{H}$.
    Then $\img{f}{\groupCentre{G}}\subseteq\groupCentre{H}$.
\end{theorem}
\begin{proof}
    $\groupHom{f}{G}{H}$ by
        \obv{group_isomorphism_is_hom}.
    $\groupCentre{G}\subseteq\carrier[G]$ by
        \obv{group_centre_subseteq}.
    $f\in\surj{\carrier[G]}{\carrier[H]}$ by
        \auto{group_isomorphism_is_bijection,bijections}.
    Show that for every $z\in\img{f}{\groupCentre{G}}$ we have
        $z\in\groupCentre{H}$.
    \begin{subproof}
        Fix $z\in\img{f}{\groupCentre{G}}$.
        Take $x\in\groupCentre{G}$ such that $f(x)=z$ by
            \auto{group_hom_image_iff}.
        Hence $x\in\carrier[G]$ by
            \obv{group_centre_subseteq,subseteq}.
        $z\in\carrier[H]$ by
            \auto{group_hom_apply_type}.
        Show that for every $w\in\carrier[H]$ we have
            $\commutesIn{H}{z}{w}$.
        \begin{subproof}
            Fix $w\in\carrier[H]$.
            Take $y\in\carrier[G]$ such that $f(y)=w$ by
                \obv{surj}.
            $\commutesIn{G}{x}{y}$ by
                \auto{group_centre_iff}.
            Hence $\mul[G](x,y)=\mul[G](y,x)$ by
                \obv{commutes_in}.
            $f(\mul[G](x,y))=\mul[H](f(x),f(y))$ and
                $f(\mul[G](y,x))=\mul[H](f(y),f(x))$ by
                \obv{group_hom_mul}.
            Hence $\mul[H](z,w)=\mul[H](w,z)$.
            Follows by \obv{commutes_in}.
        \end{subproof}
        Follows by \obv{group_centre_iff}.
    \end{subproof}
    Follows by \obv{subseteq}.
\end{proof}

\begin{theorem}\label{group_isomorphism_centre_image}
    Suppose $\groupIsomorphism{f}{G}{H}$.
    Then $\img{f}{\groupCentre{G}}=\groupCentre{H}$.
\end{theorem}
\begin{proof}
    $\img{f}{\groupCentre{G}}\subseteq\groupCentre{H}$ by
        \obv{group_isomorphism_centre_image_subseteq}.
    $\groupCentre{H}\subseteq\carrier[H]$ by
        \obv{group_centre_subseteq}.
    $\groupIsomorphism{\converse{f}}{H}{G}$ by
        \obv{group_isomorphism_converse}.
    Hence
        $\img{\converse{f}}{\groupCentre{H}}
            \subseteq\groupCentre{G}$ by
        \obv{group_isomorphism_centre_image_subseteq}.
    Show that $\groupCentre{H}\subseteq\img{f}{\groupCentre{G}}$.
    \begin{subproof}
        Show that for every $y\in\groupCentre{H}$ we have
            $y\in\img{f}{\groupCentre{G}}$.
        \begin{subproof}
            Fix $y\in\groupCentre{H}$.
            Hence $y\in\carrier[H]$ by \obv{subseteq}.
            $\converse{f}\in\funs{\carrier[H]}{\carrier[G]}$ by
                \obv{group_isomorphism_is_bijection,bijective_converse_are_funs}.
            Let $x=(\converse{f})(y)$.
            $\converse{f}$ is a function and
                $\dom{\converse{f}}=\carrier[H]$ by
                \obv{funs_elim}.
            Hence $y\in\dom{\converse{f}}\inter\groupCentre{H}$ by
                \obv{inter_intro}.
            $x\in\img{\converse{f}}{\groupCentre{H}}$ by
                \auto{img_of_function_intro}.
            Hence $x\in\groupCentre{G}$ by \obv{subseteq}.
            $(y,x)\in\converse{f}$ by
                \auto{function_apply_elim}.
            Hence $(x,y)\in f$ by \obv{converse_iff}.
            $f$ is a function by
                \obv{group_isomorphism_is_hom,group_hom_is_function}.
            Hence $f(x)=y$ by
                \obv{function_apply_intro}.
            Follows by
                \auto{group_isomorphism_is_hom,group_hom_image_iff,group_centre_subseteq}.
        \end{subproof}
        Follows by \obv{subseteq}.
    \end{subproof}
    Follows by \obv{subseteq_antisymmetric}.
\end{proof}
