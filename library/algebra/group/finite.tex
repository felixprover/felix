\section{Finite groups and Lagrange's theorem}

Finite group theory is expressed through the ordinary carrier and explicit
natural cardinality witnesses.  There is no separate finite-group record and
no total element-order operation.

\begin{proposition}\label{finite_subgroup_coset_quotient}
    Suppose $\subgroup{H}{G}$ and $\carrier[G]$ is finite.
    Then
        $\quotient{\carrier[G]}{\subgroupCosetRelation{G}{H}}$
        is finite.
\end{proposition}
\begin{proof}
    Let $E=\subgroupCosetRelation{G}{H}$ and
        $Q=\quotient{\carrier[G]}{E}$.
    $E$ is an equivalence on $\carrier[G]$ by
        \auto{subgroup_coset_relation_equivalence}.
    Hence $Q$ is a partition of $\carrier[G]$ by
        \auto{quotient_partition_of}.
    Show that $Q\subseteq\pow{\carrier[G]}$.
    \begin{subproof}
        Show that every element of $Q$ is an element of
            $\pow{\carrier[G]}$.
        \begin{subproof}
            Fix $C\in Q$.
            $C\subseteq\carrier[G]$ by
                \obv{partition_block_subseteq}.
            Follows by \obv{pow_iff}.
        \end{subproof}
        Follows by \obv{subseteq}.
    \end{subproof}
    $\pow{\carrier[G]}$ is finite by \obv{finite_powerset}.
    Follows by \obv{subseteq_finite_is_finite}.
\end{proof}

\begin{theorem}[Lagrange's theorem]\label{lagrange_finite_cardinality}
    Suppose $\subgroup{H}{G}$.
    Suppose $\carrier[G]$ has finite cardinality $m$ and
        $H$ has finite cardinality $n$.
    Then $n$ divides $m$ as natural numbers.
\end{theorem}
\begin{proof}
    Let $E=\subgroupCosetRelation{G}{H}$ and
        $Q=\quotient{\carrier[G]}{E}$.
    $\carrier[G]$ is finite by \obv{finite_cardinality,finite}.
    Hence $Q$ is finite by \auto{finite_subgroup_coset_quotient}.
    Take a natural number $k$ such that there exists
        $f\in\bijections{k}{Q}$ by \auto{finite}.
    Take $f$ such that $f\in\bijections{k}{Q}$ by \auto.
    Hence $Q$ has finite cardinality $k$ by
        \obv{finite_cardinality}.
    $E$ is an equivalence on $\carrier[G]$ by
        \auto{subgroup_coset_relation_equivalence}.
    Hence $Q$ is a partition of $\carrier[G]$ by
        \auto{quotient_partition_of}.
    Show that every element of $Q$ has finite cardinality $n$.
    \begin{subproof}
        Fix $C\in Q$.
        $C$ is inhabited by \obv{quotient_elems_inhabited}.
        Take $a$ such that $a\in C$.
        $C\subseteq\carrier[G]$ by
            \obv{partition_block_subseteq}.
        Hence $a\in\carrier[G]$ by \obv{subseteq}.
        $C=\leftCoset{G}{a}{H}$ by
            \auto{subgroup_coset_block_from_member}.
        $H\approx\leftCoset{G}{a}{H}$ by
            \obv{left_coset_equinumerous_subgroup}.
        Hence $H\approx C$.
        Take $g$ such that $g\in\bijections{H}{C}$ by \obv{equinum}.
        Follows by \obv{finite_cardinality_bijection}.
    \end{subproof}
    Hence $\carrier[G]$ has finite cardinality
        $\naturalMul{n}{k}$ by
        \obv{uniform_finite_partition_cardinality}.
    Hence $m=\naturalMul{n}{k}$ by
        \obv{finite_cardinality_unique}.
    Follows by \obv{natural_divisibility}.
\end{proof}

\begin{corollary}\label{cyclic_subgroup_cardinality_divides_group_cardinality}
    Let $G$ be a group and $x\in\carrier[G]$.
    Suppose $\carrier[G]$ has finite cardinality $m$ and
        $\cyclicSubgroup{G}{x}$ has finite cardinality $n$.
    Then $n$ divides $m$ as natural numbers.
\end{corollary}
\begin{proof}
    $\subgroup{\cyclicSubgroup{G}{x}}{G}$ by
        \obv{cyclic_subgroup_is_subgroup}.
    Follows by \obv{lagrange_finite_cardinality}.
\end{proof}

\begin{corollary}\label{cyclic_subgroup_has_divisor_cardinality}
    Let $G$ be a group and $x\in\carrier[G]$.
    Suppose $\carrier[G]$ has finite cardinality $m$.
    Then there exists a natural number $n$ such that
        $\cyclicSubgroup{G}{x}$ has finite cardinality $n$ and
        $n$ divides $m$ as natural numbers.
\end{corollary}
\begin{proof}
    Let $H=\cyclicSubgroup{G}{x}$.
    $\subgroup{H}{G}$ by \auto{cyclic_subgroup_is_subgroup}.
    Hence $H\subseteq\carrier[G]$ by \obv{subgroup_subseteq}.
    $\carrier[G]$ is finite by \obv{finite_cardinality,finite}.
    Hence $H$ is finite by \obv{subseteq_finite_is_finite}.
    Take a natural number $n$ such that there exists
        $f\in\bijections{n}{H}$ by \auto{finite}.
    Take $f$ such that $f\in\bijections{n}{H}$ by \auto.
    Hence $H$ has finite cardinality $n$ by
        \obv{finite_cardinality}.
    Hence $n$ divides $m$ as natural numbers by
        \obv{lagrange_finite_cardinality}.
\end{proof}

\begin{theorem}\label{prime_cardinality_group_is_cyclic}
    Let $G$ be a group.
    Let $p$ be a prime natural number.
    Suppose $\carrier[G]$ has finite cardinality $p$.
    Then $\cyclicGroup{G}$.
\end{theorem}
\begin{proof}
    $\naturalOne=\cardOne$ by
        \obv{natural_one,cardinal_one}.
    $\naturalOne\in p$ by \obv{prime_natural}.
    Hence $p\neq\emptyset$ by \obv{emptyset}.

    Show that there exists $x\in\carrier[G]$ such that
        $x\neq\neutral[G]$.
    \begin{subproof}
        Suppose not.
        Hence every element of $\carrier[G]$ is equal to $\neutral[G]$
            by \auto.
        $\neutral[G]\in\carrier[G]$ by
            \obv{group_is_monoid,monoid_is_unital_magma,
                unitalmagma_neutral_in_carrier}.
        Hence $\carrier[G]=\{\neutral[G]\}$ by
            \auto{setext,singleton_iff}.
        Hence $\carrier[G]$ has finite cardinality $\cardOne$ by
            \auto{finite_cardinality_one_iff_singleton}.
        Hence $p=\cardOne$ by \obv{finite_cardinality_unique}.
        Hence $p=\naturalOne$.
        Thus $\naturalOne\in\naturalOne$.
        Contradiction by \obv{in_irrefl}.
    \end{subproof}
    Take $x\in\carrier[G]$ such that $x\neq\neutral[G]$.

    Let $H=\cyclicSubgroup{G}{x}$.
    $\subgroup{H}{G}$ by \auto{cyclic_subgroup_is_subgroup}.
    Hence $H\subseteq\carrier[G]$ by \obv{subgroup_subseteq}.
    $\carrier[G]$ is finite by \obv{finite_cardinality,finite}.
    Hence $H$ is finite by \obv{subseteq_finite_is_finite}.
    Take a natural number $n$ such that there exists
        $u\in\bijections{n}{H}$ by \auto{finite}.
    Take $u$ such that $u\in\bijections{n}{H}$ by \auto.
    Hence $H$ has finite cardinality $n$ by \obv{finite_cardinality}.
    Hence $n$ divides $p$ as natural numbers by
        \obv{lagrange_finite_cardinality}.
    Hence $n=\naturalOne$ or $n=p$ by \obv{prime_natural}.
    Show that $n\neq\naturalOne$.
    \begin{subproof}
        Suppose $n=\naturalOne$.
        Hence $H$ has finite cardinality $\cardOne$.
        $x\in H$ by \obv{cyclic_subgroup_contains_generator}.
        $\neutral[G]\in H$ by \obv{subgroup_neutral}.
        Hence $x=\neutral[G]$ by
            \obv{finite_cardinality_one_elements_equal}.
        Contradiction.
    \end{subproof}
    Hence $n=p$.
    Hence $H$ has finite cardinality $p$.
    Hence $H=\carrier[G]$ by
        \obv{finite_subset_same_cardinality,prime_natural_guard}.
    Follows by \auto{cyclic_group}.
\end{proof}
