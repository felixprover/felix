\section{Natural powers in groups}

Group powers are the monoid powers of the same element in the same record.
The iteration equation below retains the former right-translation interface.

\begin{definition}\label{group_power}
    $\groupPower{G}{a}{n}=\monoidPower{G}{a}{n}$.
\end{definition}

\begin{proposition}\label{group_power_iteration_eq}
    $\groupPower{G}{a}{n}=
        \iterationAt{\RightTranslation{G}{a}}{\neutral[G]}{n}$.
\end{proposition}
\begin{proof}
    $\monoidRightTranslation{G}{a}=\RightTranslation{G}{a}$ by
        \obv{monoid_right_translation,right_translation,setext}.
    Follows by \obv{group_power,monoid_power}.
\end{proof}

\begin{proposition}\label{group_power_zero}
    $\groupPower{G}{a}{\emptyset}=\neutral[G]$.
\end{proposition}
\begin{proof}
    Follows by \obv{group_power,monoid_power_zero}.
\end{proof}

\begin{proposition}\label{group_power_type}
    Let $G$ be a group.
    Let $a\in\carrier[G]$.
    Let $n$ be a natural number.
    Then $\groupPower{G}{a}{n}\in\carrier[G]$.
\end{proposition}
\begin{proof}
    $G$ is a monoid by \obv{group_is_monoid}.
    Follows by \auto{group_power,monoid_power_type}.
\end{proof}

\begin{proposition}\label{group_power_successor}
    Let $G$ be a group.
    Let $a\in\carrier[G]$.
    Let $n$ be a natural number.
    Then $\groupPower{G}{a}{\suc{n}}=
        \mul[G](\groupPower{G}{a}{n},a)$.
\end{proposition}
\begin{proof}
    $G$ is a monoid by \obv{group_is_monoid}.
    Follows by \auto{group_power,monoid_power_successor}.
\end{proof}

\begin{proposition}\label{group_power_one}
    Let $G$ be a group.
    Let $a\in\carrier[G]$.
    Then $\groupPower{G}{a}{\suc{\emptyset}}=a$.
\end{proposition}
\begin{proof}
    $G$ is a monoid by \obv{group_is_monoid}.
    Follows by \obv{group_power,monoid_power_one}.
\end{proof}

\begin{theorem}\label{group_power_addition}
    Let $G$ be a group.
    Let $a\in\carrier[G]$.
    Let $m,n$ be natural numbers.
    Then $\groupPower{G}{a}{m+n}=
        \mul[G](\groupPower{G}{a}{m},\groupPower{G}{a}{n})$.
\end{theorem}
\begin{proof}
    $G$ is a monoid by \obv{group_is_monoid}.
    Follows by \auto{group_power,monoid_power_addition}.
\end{proof}

\begin{theorem}\label{group_power_in_subgroup}
    Suppose $\subgroup{H}{G}$.
    Let $a\in H$.
    Let $n$ be a natural number.
    Then $\groupPower{G}{a}{n}\in H$.
\end{theorem}
\begin{proof}
    Let $C=\{k\in\naturals\mid \groupPower{G}{a}{k}\in H\}$.
    $\emptyset\in C$ by
        \obv{omega_empty,group_power_zero,subgroup_neutral}.
    Show that for every $k\in C$ we have $\suc{k}\in C$.
    \begin{subproof}
        Fix $k\in C$.
        Hence $k$ is a natural number and
            $\groupPower{G}{a}{k}\in H$.
        $G$ is a group and $a\in\carrier[G]$ by
            \auto{subgroup_ambient_group,subgroup_subseteq,subseteq}.
        $\mul[G](\groupPower{G}{a}{k},a)\in H$ by
            \obv{subgroup_mul}.
        $\groupPower{G}{a}{\suc{k}}=
            \mul[G](\groupPower{G}{a}{k},a)$ by
            \obv{group_power_successor}.
        Hence $\groupPower{G}{a}{\suc{k}}\in H$.
        $\suc{k}$ is a natural number by \obv{omega_successor}.
    \end{subproof}
    Hence $C$ is an inductive set by \obv{inductive_set}.
    Hence every natural number is an element of $C$ by
        \obv{omega_minimal}.
    Hence $n\in C$.
\end{proof}

\begin{theorem}\label{group_hom_preserves_power}
    Suppose $\groupHom{f}{G}{H}$.
    Let $a\in\carrier[G]$.
    Let $n$ be a natural number.
    Then $f(\groupPower{G}{a}{n})=
        \groupPower{H}{f(a)}{n}$.
\end{theorem}
\begin{proof}
    $f$ is a monoid homomorphism from $G$ to $H$ by
        \obv{group_hom_is_monoid_hom}.
    Follows by
        \auto{group_power,monoid_hom_preserves_power}.
\end{proof}
