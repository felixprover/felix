\section{Products of subgroups}

This is the internal product of two subgroup carriers in one ambient group,
not the categorical product of two group structures.  The carrier is bounded
by the ambient group even when its arguments are arbitrary sets.

\begin{definition}\label{subgroup_product}
    $\subgroupProduct{G}{H}{K}=
        \{x\in\carrier[G]\mid
            \text{there exists $h\in H$ such that there exists
                $k\in K$ such that $x=\mul[G](h,k)$}\}$.
\end{definition}

\begin{proposition}\label{subgroup_product_iff}
    $x\in\subgroupProduct{G}{H}{K}$ iff
        $x\in\carrier[G]$ and there exists $h\in H$ such that
            there exists $k\in K$ such that $x=\mul[G](h,k)$.
\end{proposition}
\begin{proof}
    Follows by \obv{subgroup_product}.
\end{proof}

\begin{proposition}\label{subgroup_product_iff_for_subgroups}
    Suppose $\subgroup{H}{G}$ and $\subgroup{K}{G}$.
    Then $x\in\subgroupProduct{G}{H}{K}$ iff
        there exists $h\in H$ such that there exists $k\in K$ such that
            $x=\mul[G](h,k)$.
\end{proposition}
\begin{proof}
    Show that if $x\in\subgroupProduct{G}{H}{K}$, then
        there exists $h\in H$ such that there exists $k\in K$ such that
            $x=\mul[G](h,k)$.
    \begin{subproof}
        Assume $x\in\subgroupProduct{G}{H}{K}$.
        Follows by \obv{subgroup_product_iff}.
    \end{subproof}
    Show that if there exists $h\in H$ such that there exists $k\in K$
        such that $x=\mul[G](h,k)$, then
        $x\in\subgroupProduct{G}{H}{K}$.
    \begin{subproof}
        Assume that there exists $h\in H$ such that there exists $k\in K$
            such that $x=\mul[G](h,k)$.
        Take $h\in H$ such that there exists $k\in K$ such that
            $x=\mul[G](h,k)$.
        Take $k\in K$ such that $x=\mul[G](h,k)$.
        $h,k\in\carrier[G]$ by
            \auto{subgroup_subseteq,subseteq}.
        Hence $x\in\carrier[G]$ by
            \auto{subgroup_ambient_group,group_is_monoid,
                monoid_is_magma,magma_welldef}.
        Follows by \obv{subgroup_product_iff}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{proposition}\label{subgroup_product_subseteq_carrier}
    $\subgroupProduct{G}{H}{K}\subseteq\carrier[G]$.
\end{proposition}
\begin{proof}
    Follows by \obv{subgroup_product,subseteq}.
\end{proof}

\begin{proposition}\label{subgroup_product_contains_left}
    Suppose $\subgroup{H}{G}$ and $\subgroup{K}{G}$.
    Then $H\subseteq\subgroupProduct{G}{H}{K}$.
\end{proposition}
\begin{proof}
    $\neutral[G]\in K$ by \obv{subgroup_neutral}.
    Show that for every $h\in H$ we have
        $h\in\subgroupProduct{G}{H}{K}$.
    \begin{subproof}
        Fix $h\in H$.
        $h\in\carrier[G]$ by
            \auto{subgroup_subseteq,subseteq}.
        $\mul[G](h,\neutral[G])=h$ by
            \obv{subgroup_ambient_group,group_is_monoid,
                monoid_is_unital_magma,unitalmagma_right}.
        Follows by \obv{subgroup_product_iff_for_subgroups}.
    \end{subproof}
    Follows by \obv{subseteq}.
\end{proof}

\begin{proposition}\label{subgroup_product_contains_right}
    Suppose $\subgroup{H}{G}$ and $\subgroup{K}{G}$.
    Then $K\subseteq\subgroupProduct{G}{H}{K}$.
\end{proposition}
\begin{proof}
    $\neutral[G]\in H$ by \obv{subgroup_neutral}.
    Show that for every $k\in K$ we have
        $k\in\subgroupProduct{G}{H}{K}$.
    \begin{subproof}
        Fix $k\in K$.
        $k\in\carrier[G]$ by
            \auto{subgroup_subseteq,subseteq}.
        $\mul[G](\neutral[G],k)=k$ by
            \obv{subgroup_ambient_group,group_is_monoid,
                monoid_is_unital_magma,unitalmagma_left}.
        Follows by \obv{subgroup_product_iff_for_subgroups}.
    \end{subproof}
    Follows by \obv{subseteq}.
\end{proof}

\begin{corollary}\label{subgroup_product_contains_union}
    Suppose $\subgroup{H}{G}$ and $\subgroup{K}{G}$.
    Then $H\union K\subseteq\subgroupProduct{G}{H}{K}$.
\end{corollary}
\begin{proof}
    Follows by
        \auto{subgroup_product_contains_left,
            subgroup_product_contains_right,subseteq,union_iff}.
\end{proof}

\subsection{Rearranging a normal factor}

\begin{lemma}\label{subgroup_product_swap_right_factor}
    Let $G$ be a group.
    Let $h,k\in\carrier[G]$.
    Then
        $\mul[G](h,k)=
            \mul[G](\conjugate{G}{h}{k},h)$.
\end{lemma}
\begin{proof}
    $\groupInv[G](h)\in\carrier[G]$ by
        \obv{group_inverse_type}.
    $\mul[G](h,k)\in\carrier[G]$ by
        \obv{group_is_monoid,monoid_is_magma,magma_welldef}.
    $\mul[G](\groupInv[G](h),h)=\neutral[G]$ by
        \obv{group_inverse_left}.
    Follows by
        \auto{group_conjugate,group_is_monoid,monoid_assoc,
            monoid_is_unital_magma,unitalmagma_right}.
\end{proof}

\begin{lemma}\label{subgroup_product_swap_left_factor}
    Let $G$ be a group.
    Let $h,k\in\carrier[G]$.
    Then
        $\mul[G](k,h)=
            \mul[G]
                (h,\conjugate{G}{\groupInv[G](h)}{k})$.
\end{lemma}
\begin{proof}
    Let $u=\groupInv[G](h)$.
    $u\in\carrier[G]$ by \auto{group_inverse_type}.
    $\groupInv[G](u)=h$ by \auto{group_inverse_involutive}.
    $\mul[G](h,u)=\neutral[G]$ by
        \auto{group_inverse_right}.
    $\mul[G](u,k)\in\carrier[G]$ by
        \obv{group_is_monoid,monoid_is_magma,magma_welldef}.
    Follows by
        \auto{group_conjugate,group_is_monoid,monoid_assoc,
            monoid_is_unital_magma,unitalmagma_left}.
\end{proof}

\begin{lemma}\label{subgroup_product_mul_rearrangement}
    Let $G$ be a group.
    Let $h_1,h_2,k_1,k_2\in\carrier[G]$.
    Then
        $\mul[G](\mul[G](h_1,k_1),\mul[G](h_2,k_2))=
            \mul[G]
                (\mul[G](h_1,h_2),
                 \mul[G]
                    (\conjugate{G}{\groupInv[G](h_2)}{k_1},k_2))$.
\end{lemma}
\begin{proof}
    Let $u=\groupInv[G](h_2)$.
    $u\in\carrier[G]$ by \auto{group_inverse_type}.
    $\groupInv[G](u)=h_2$ by \auto{group_inverse_involutive}.
    $\mul[G](h_2,u)=\neutral[G]$ by
        \auto{group_inverse_right}.
    $\mul[G](u,k_1),\mul[G](h_1,h_2),
        \mul[G](h_1,k_1),\mul[G](h_2,k_2)\in\carrier[G]$ by
        \obv{group_is_monoid,monoid_is_magma,magma_welldef}.
    $\conjugate{G}{u}{k_1}=\mul[G](\mul[G](u,k_1),h_2)$ by
        \obv{group_conjugate}.
    Follows by
        \auto{group_is_monoid,monoid_assoc,
            monoid_is_unital_magma,unitalmagma_left}.
\end{proof}

\begin{lemma}\label{subgroup_product_inverse_rearrangement}
    Let $G$ be a group.
    Let $h,k\in\carrier[G]$.
    Then
        $\groupInv[G](\mul[G](h,k))=
            \mul[G]
                (\groupInv[G](h),
                 \conjugate{G}{h}{\groupInv[G](k)})$.
\end{lemma}
\begin{proof}
    Let $u=\groupInv[G](h)$ and $v=\groupInv[G](k)$.
    $u,v\in\carrier[G]$ by \auto{group_inverse_type}.
    $\mul[G](u,h)=\neutral[G]$ by \auto{group_inverse_left}.
    $\mul[G](h,v)\in\carrier[G]$ by
        \obv{group_is_monoid,monoid_is_magma,magma_welldef}.
    $\groupInv[G](\mul[G](h,k))=\mul[G](v,u)$ by
        \auto{group_inverse_product}.
    Follows by
        \auto{group_conjugate,group_is_monoid,monoid_assoc,
            monoid_is_unital_magma,unitalmagma_left}.
\end{proof}

\subsection{Subgroup and normality laws}

\begin{theorem}\label{subgroup_product_subgroup}
    Suppose $\subgroup{H}{G}$ and $\normalSubgroup{K}{G}$.
    Then $\subgroup{\subgroupProduct{G}{H}{K}}{G}$.
\end{theorem}
\begin{proof}
    Let $P=\subgroupProduct{G}{H}{K}$.
    $\subgroup{K}{G}$ by \obv{normal_subgroup_is_subgroup}.
    $G$ is a group by \obv{subgroup_ambient_group}.
    $P\subseteq\carrier[G]$ by
        \auto{subgroup_product_subseteq_carrier}.
    $\neutral[G]\in H$ and $\neutral[G]\in K$ by
        \obv{subgroup_neutral}.
    $\neutral[G]\in\carrier[G]$ by
        \obv{group_is_monoid,monoid_is_unital_magma,
            unitalmagma_type}.
    $\mul[G](\neutral[G],\neutral[G])=\neutral[G]$ by
        \obv{group_is_monoid,monoid_is_unital_magma,
            unitalmagma_left}.
    Hence $\neutral[G]\in P$ by
        \obv{subgroup_product_iff}.
    Show that for all $x,y\in P$ we have $\mul[G](x,y)\in P$.
    \begin{subproof}
        Fix $x,y\in P$.
        Take $h_1\in H$ such that there exists $k_1\in K$ such that
            $x=\mul[G](h_1,k_1)$ by
            \obv{subgroup_product_iff_for_subgroups}.
        Take $k_1\in K$ such that $x=\mul[G](h_1,k_1)$.
        Take $h_2\in H$ such that there exists $k_2\in K$ such that
            $y=\mul[G](h_2,k_2)$ by
            \obv{subgroup_product_iff_for_subgroups}.
        Take $k_2\in K$ such that $y=\mul[G](h_2,k_2)$.
        $h_1,h_2,k_1,k_2\in\carrier[G]$ by
            \auto{subgroup_subseteq,normal_subgroup_subseteq,subseteq}.
        $\mul[G](h_1,h_2)\in H$ by \obv{subgroup_mul}.
        $\groupInv[G](h_2)\in\carrier[G]$ by
            \obv{group_inverse_type}.
        $\conjugate{G}{\groupInv[G](h_2)}{k_1}\in K$ by
            \obv{normal_subgroup_closed_under_conjugation}.
        Hence
            $\mul[G]
                (\conjugate{G}{\groupInv[G](h_2)}{k_1},k_2)
                \in K$ by \obv{subgroup_mul}.
        $\mul[G](x,y)=
            \mul[G]
                (\mul[G](h_1,h_2),
                 \mul[G]
                    (\conjugate{G}{\groupInv[G](h_2)}{k_1},k_2))$ by
            \auto{subgroup_product_mul_rearrangement}.
        Follows by \auto{subgroup_product_iff_for_subgroups}.
    \end{subproof}
    Show that for every $x\in P$ we have $\groupInv[G](x)\in P$.
    \begin{subproof}
        Fix $x\in P$.
        Take $h\in H$ such that there exists $k\in K$ such that
            $x=\mul[G](h,k)$ by
            \obv{subgroup_product_iff_for_subgroups}.
        Take $k\in K$ such that $x=\mul[G](h,k)$.
        $h,k\in\carrier[G]$ by
            \auto{subgroup_subseteq,normal_subgroup_subseteq,subseteq}.
        $\groupInv[G](h)\in H$ and $\groupInv[G](k)\in K$ by
            \obv{subgroup_inv}.
        $\conjugate{G}{h}{\groupInv[G](k)}\in K$ by
            \obv{normal_subgroup_closed_under_conjugation}.
        $\groupInv[G](x)=
            \mul[G]
                (\groupInv[G](h),
                 \conjugate{G}{h}{\groupInv[G](k)})$ by
            \auto{subgroup_product_inverse_rearrangement}.
        Follows by \auto{subgroup_product_iff_for_subgroups}.
    \end{subproof}
    Follows by \obv{subgroup}.
\end{proof}

\begin{theorem}\label{subgroup_product_commutes_if_right_normal}
    Suppose $\subgroup{H}{G}$ and $\normalSubgroup{K}{G}$.
    Then
        $\subgroupProduct{G}{H}{K}=\subgroupProduct{G}{K}{H}$.
\end{theorem}
\begin{proof}
    $\subgroup{K}{G}$ by \obv{normal_subgroup_is_subgroup}.
    $G$ is a group by \obv{subgroup_ambient_group}.
    Show that $\subgroupProduct{G}{H}{K}\subseteq
        \subgroupProduct{G}{K}{H}$.
    \begin{subproof}
        Show that for every $x\in\subgroupProduct{G}{H}{K}$ we have
            $x\in\subgroupProduct{G}{K}{H}$.
        \begin{subproof}
            Fix $x\in\subgroupProduct{G}{H}{K}$.
            Take $h\in H$ such that there exists $k\in K$ such that
                $x=\mul[G](h,k)$ by
                \obv{subgroup_product_iff_for_subgroups}.
            Take $k\in K$ such that $x=\mul[G](h,k)$.
            $h,k\in\carrier[G]$ by
                \auto{subgroup_subseteq,normal_subgroup_subseteq,subseteq}.
            Let $c=\conjugate{G}{h}{k}$.
            $c\in K$ by
                \auto{normal_subgroup_closed_under_conjugation}.
            $x=\mul[G](c,h)$ by
                \auto{subgroup_product_swap_right_factor}.
            Follows by \obv{subgroup_product_iff_for_subgroups}.
        \end{subproof}
        Follows by \obv{subseteq}.
    \end{subproof}
    Show that $\subgroupProduct{G}{K}{H}\subseteq
        \subgroupProduct{G}{H}{K}$.
    \begin{subproof}
        Show that for every $x\in\subgroupProduct{G}{K}{H}$ we have
            $x\in\subgroupProduct{G}{H}{K}$.
        \begin{subproof}
            Fix $x\in\subgroupProduct{G}{K}{H}$.
            Take $k\in K$ such that there exists $h\in H$ such that
                $x=\mul[G](k,h)$ by
                \obv{subgroup_product_iff_for_subgroups}.
            Take $h\in H$ such that $x=\mul[G](k,h)$.
            $h,k\in\carrier[G]$ by
                \auto{subgroup_subseteq,normal_subgroup_subseteq,subseteq}.
            $\groupInv[G](h)\in\carrier[G]$ by
                \obv{group_inverse_type}.
            Let $c=\conjugate{G}{\groupInv[G](h)}{k}$.
            $c\in K$ by
                \auto{normal_subgroup_closed_under_conjugation}.
            $x=\mul[G](h,c)$ by
                \auto{subgroup_product_swap_left_factor}.
            Follows by \auto{subgroup_product_iff_for_subgroups}.
        \end{subproof}
        Follows by \obv{subseteq}.
    \end{subproof}
    Follows by \obv{subseteq_antisymmetric}.
\end{proof}

\begin{theorem}\label{subgroup_product_right_factor_normal}
    Suppose $\subgroup{H}{G}$ and $\normalSubgroup{K}{G}$.
    Let $P=\subgroupProduct{G}{H}{K}$.
    Then $\normalSubgroup{K}{\subgroupStructure{G}{P}}$.
\end{theorem}
\begin{proof}
    Let $S=\subgroupStructure{G}{P}$.
    $\subgroup{K}{G}$ by \obv{normal_subgroup_is_subgroup}.
    $\subgroup{P}{G}$ by \auto{subgroup_product_subgroup}.
    $K\subseteq P$ by \obv{subgroup_product_contains_right}.
    Hence $\subgroup{K}{S}$ by
        \obv{subgroup_of_induced_group_iff}.
    Show that for all $a\in\carrier[S]$ we have for every $x\in K$
        we have $\conjugate{S}{a}{x}\in K$.
    \begin{subproof}
        Fix $a\in\carrier[S]$.
        Fix $x\in K$.
        $\carrier[S]=P$ by \auto{subgroup_structure_carrier}.
        Hence $a\in P$.
        Hence $a\in\carrier[G]$ by
            \auto{subgroup_product_subseteq_carrier,subseteq}.
        $x\in P$ by \obv{subseteq}.
        $\conjugate{G}{a}{x}\in K$ by
            \obv{normal_subgroup_closed_under_conjugation}.
        Follows by \auto{subgroup_structure_conjugate_apply}.
    \end{subproof}
    Follows by \obv{normal_subgroup_iff_closed_under_conjugation}.
\end{proof}

\begin{theorem}\label{subgroup_product_commutes_if_left_normal}
    Suppose $\normalSubgroup{H}{G}$ and $\subgroup{K}{G}$.
    Then
        $\subgroupProduct{G}{H}{K}=\subgroupProduct{G}{K}{H}$.
\end{theorem}
\begin{proof}
    Follows by
        \obv{normal_subgroup_is_subgroup,
            subgroup_product_commutes_if_right_normal}.
\end{proof}

\begin{theorem}\label{subgroup_product_subgroup_if_left_normal}
    Suppose $\normalSubgroup{H}{G}$ and $\subgroup{K}{G}$.
    Then $\subgroup{\subgroupProduct{G}{H}{K}}{G}$.
\end{theorem}
\begin{proof}
    $\subgroup{\subgroupProduct{G}{K}{H}}{G}$ by
        \obv{subgroup_product_subgroup}.
    Follows by \obv{subgroup_product_commutes_if_left_normal}.
\end{proof}

\begin{theorem}\label{subgroup_product_left_factor_normal}
    Suppose $\normalSubgroup{H}{G}$ and $\subgroup{K}{G}$.
    Let $P=\subgroupProduct{G}{H}{K}$.
    Then $\normalSubgroup{H}{\subgroupStructure{G}{P}}$.
\end{theorem}
\begin{proof}
    Let $Q=\subgroupProduct{G}{K}{H}$.
    $P=Q$ by \obv{subgroup_product_commutes_if_left_normal}.
    $\normalSubgroup{H}{\subgroupStructure{G}{Q}}$ by
        \obv{subgroup_product_right_factor_normal}.
\end{proof}

\begin{theorem}\label{subgroup_product_factors_normal}
    Suppose $\normalSubgroup{H}{G}$ and $\normalSubgroup{K}{G}$.
    Let $P=\subgroupProduct{G}{H}{K}$.
    Then $\normalSubgroup{H}{\subgroupStructure{G}{P}}$ and
        $\normalSubgroup{K}{\subgroupStructure{G}{P}}$.
\end{theorem}
\begin{proof}
    Follows by
        \auto{normal_subgroup_is_subgroup,
            subgroup_product_left_factor_normal,
            subgroup_product_right_factor_normal}.
\end{proof}

\begin{theorem}\label{normal_subgroup_product}
    Suppose $\normalSubgroup{H}{G}$ and $\normalSubgroup{K}{G}$.
    Then $\normalSubgroup{\subgroupProduct{G}{H}{K}}{G}$.
\end{theorem}
\begin{proof}
    Let $P=\subgroupProduct{G}{H}{K}$.
    $\subgroup{H}{G}$ and $\subgroup{K}{G}$ by
        \obv{normal_subgroup_is_subgroup}.
    $\subgroup{P}{G}$ by \auto{subgroup_product_subgroup}.
    Show that for all $a\in\carrier[G]$ we have for every $x\in P$
        we have $\conjugate{G}{a}{x}\in P$.
    \begin{subproof}
        Fix $a\in\carrier[G]$.
        Fix $x\in P$.
        Take $h\in H$ such that there exists $k\in K$ such that
            $x=\mul[G](h,k)$ by
            \obv{subgroup_product_iff_for_subgroups}.
        Take $k\in K$ such that $x=\mul[G](h,k)$.
        $h,k\in\carrier[G]$ by
            \auto{normal_subgroup_subseteq,subseteq}.
        $\conjugate{G}{a}{h}\in H$ and
            $\conjugate{G}{a}{k}\in K$ by
            \obv{normal_subgroup_closed_under_conjugation}.
        $\conjugate{G}{a}{\mul[G](h,k)}=
            \mul[G](\conjugate{G}{a}{h},
                \conjugate{G}{a}{k})$ by
            \obv{group_conjugate_mul}.
        Hence $\conjugate{G}{a}{x}=
            \mul[G](\conjugate{G}{a}{h},
                \conjugate{G}{a}{k})$.
        Follows by \auto{subgroup_product_iff_for_subgroups}.
    \end{subproof}
    Follows by \auto{normal_subgroup_iff_closed_under_conjugation}.
\end{proof}
