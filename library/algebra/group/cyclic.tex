\section{Cyclic subgroups}

The cyclic subgroup generated by an element is defined through the general
generated-subgroup API.  Natural powers include exponent zero.  Their inverse
range is kept separate, so no integer object is imported merely to describe
the elementary power content of a cyclic subgroup.  The inherited intersection
membership law is exposed directly for clients that do not need powers.

\begin{definition}\label{cyclic_subgroup}
    Let $G$ be a group.
    Let $a$ be an element of $\carrier[G]$.
    $\cyclicSubgroup{G}{a}=\generatedSubgroup{\{a\}}{G}$.
\end{definition}

\begin{theorem}\label{cyclic_subgroup_is_subgroup}
    Let $G$ be a group.
    Let $a\in\carrier[G]$.
    Then $\subgroup{\cyclicSubgroup{G}{a}}{G}$.
\end{theorem}
\begin{proof}
    $\{a\}\subseteq\carrier[G]$ by
        \obv{singleton_subset_intro}.
    Follows by
        \auto{cyclic_subgroup,generated_subgroup_is_subgroup}.
\end{proof}

\begin{proposition}\label{cyclic_subgroup_contains_generator}
    Let $G$ be a group.
    Let $a\in\carrier[G]$.
    Then $a\in\cyclicSubgroup{G}{a}$.
\end{proposition}
\begin{proof}
    $\{a\}\subseteq\carrier[G]$ by
        \obv{singleton_subset_intro}.
    Hence $\{a\}\subseteq\generatedSubgroup{\{a\}}{G}$ by
        \obv{generated_subgroup_contains_generators}.
    $a\in\{a\}$ by \obv{singleton_intro}.
    Follows by \obv{cyclic_subgroup,subseteq}.
\end{proof}

\begin{theorem}\label{cyclic_subgroup_universal_property}
    Let $G$ be a group.
    Suppose $\subgroup{H}{G}$.
    Let $a\in\carrier[G]$.
    Then $\cyclicSubgroup{G}{a}\subseteq H$ iff $a\in H$.
\end{theorem}
\begin{proof}
    $a\in\cyclicSubgroup{G}{a}$ by
        \obv{cyclic_subgroup_contains_generator}.
    Show that if $\cyclicSubgroup{G}{a}\subseteq H$, then $a\in H$.
    \begin{subproof}
        Follows by \obv{subseteq}.
    \end{subproof}
    Show that if $a\in H$, then $\cyclicSubgroup{G}{a}\subseteq H$.
    \begin{subproof}
        Assume $a\in H$.
        $\{a\}\subseteq H$ by notion.
        $\{a\}\subseteq\carrier[G]$ by
            \obv{singleton_subset_intro}.
        Follows by
            \auto{cyclic_subgroup,generated_subgroup_least}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{theorem}\label{cyclic_subgroup_iff}
    Let $G$ be a group.
    Let $a\in\carrier[G]$.
    Then $x\in\cyclicSubgroup{G}{a}$ iff for every $H$ such that
        $\subgroup{H}{G}$ and $a\in H$ we have $x\in H$.
\end{theorem}
\begin{proof}
    $\{a\}\subseteq\carrier[G]$ by
        \obv{singleton_subset_intro}.
    Show that if $x\in\cyclicSubgroup{G}{a}$, then for every $H$ such that
        $\subgroup{H}{G}$ and $a\in H$ we have $x\in H$.
    \begin{subproof}
        Assume $x\in\cyclicSubgroup{G}{a}$.
        Hence $x\in\generatedSubgroup{\{a\}}{G}$ by
            \obv{cyclic_subgroup}.
        Fix $H$.
        Assume $\subgroup{H}{G}$ and $a\in H$.
        $\{a\}\subseteq H$ by \obv{singleton_subset_intro}.
        Follows by \auto{generated_subgroup_iff}.
    \end{subproof}
    Show that if for every $H$ such that
        $\subgroup{H}{G}$ and $a\in H$ we have $x\in H$, then
        $x\in\cyclicSubgroup{G}{a}$.
    \begin{subproof}
        Assume that for every $H$ such that
            $\subgroup{H}{G}$ and $a\in H$ we have $x\in H$.
        Show that for every $H$ such that
            $\subgroup{H}{G}$ and $\{a\}\subseteq H$ we have $x\in H$.
        \begin{subproof}
            Fix $H$.
            Assume $\subgroup{H}{G}$ and $\{a\}\subseteq H$.
            $a\in\{a\}$ by \auto{singleton_iff}.
            Hence $a\in H$ by \obv{subseteq}.
        \end{subproof}
        Hence $x\in\generatedSubgroup{\{a\}}{G}$ by
            \auto{generated_subgroup_iff}.
        Follows by \obv{cyclic_subgroup}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{definition}\label{natural_power_carrier}
    $\naturalPowerCarrier{G}{a}=
        \{x\in\carrier[G]\mid
            \text{there exists $n\in\naturals$ such that
                $x=\groupPower{G}{a}{n}$}\}$.
\end{definition}

\begin{proposition}\label{natural_power_carrier_iff}
    $x\in\naturalPowerCarrier{G}{a}$ iff
        $x\in\carrier[G]$ and
        there exists $n\in\naturals$ such that
            $x=\groupPower{G}{a}{n}$.
\end{proposition}
\begin{proof}
    Follows by \obv{natural_power_carrier}.
\end{proof}

\begin{proposition}\label{natural_power_carrier_subseteq}
    $\naturalPowerCarrier{G}{a}\subseteq\carrier[G]$.
\end{proposition}
\begin{proof}
    Follows by \obv{subseteq,natural_power_carrier_iff}.
\end{proof}

\begin{classification}\label{natural_power_carrier_is_subset_in_group}
    Let $G$ be a group.
    Then $\naturalPowerCarrier{G}{a}\subseteq\carrier[G]$.
\end{classification}
\begin{proof}
    Follows by \obv{natural_power_carrier_subseteq}.
\end{proof}

\begin{proposition}\label{natural_power_carrier_contains_generator}
    Let $G$ be a group.
    Let $a\in\carrier[G]$.
    Then $a\in\naturalPowerCarrier{G}{a}$.
\end{proposition}
\begin{proof}
    $\suc{\emptyset}$ is a natural number by
        \obv{omega_empty,omega_successor}.
    $a=\groupPower{G}{a}{\suc{\emptyset}}$ by
        \obv{group_power_one}.
    Follows by \obv{natural_power_carrier_iff}.
\end{proof}

\begin{theorem}\label{natural_power_carrier_subseteq_cyclic_subgroup}
    Let $G$ be a group.
    Let $a\in\carrier[G]$.
    Then
        $\naturalPowerCarrier{G}{a}\subseteq\cyclicSubgroup{G}{a}$.
\end{theorem}
\begin{proof}
    $\subgroup{\cyclicSubgroup{G}{a}}{G}$ by
        \obv{cyclic_subgroup_is_subgroup}.
    $a\in\cyclicSubgroup{G}{a}$ by
        \obv{cyclic_subgroup_contains_generator}.
    Show that for every $x\in\naturalPowerCarrier{G}{a}$ we have
        $x\in\cyclicSubgroup{G}{a}$.
    \begin{subproof}
        Fix $x\in\naturalPowerCarrier{G}{a}$.
        Take $n\in\naturals$ such that
            $x=\groupPower{G}{a}{n}$ by
            \obv{natural_power_carrier_iff}.
        $\groupPower{G}{a}{n}\in\cyclicSubgroup{G}{a}$ by
            \obv{group_power_in_subgroup}.
    \end{subproof}
    Follows by \obv{subseteq}.
\end{proof}

\begin{definition}\label{inverse_natural_power_carrier}
    $\inverseNaturalPowerCarrier{G}{a}=
        \{x\in\carrier[G]\mid
            \text{there exists $n\in\naturals$ such that
                $x=\groupInv[G](\groupPower{G}{a}{n})$}\}$.
\end{definition}

\begin{proposition}\label{inverse_natural_power_carrier_iff}
    $x\in\inverseNaturalPowerCarrier{G}{a}$ iff
        $x\in\carrier[G]$ and
        there exists $n\in\naturals$ such that
            $x=\groupInv[G](\groupPower{G}{a}{n})$.
\end{proposition}
\begin{proof}
    Follows by \obv{inverse_natural_power_carrier}.
\end{proof}

\begin{proposition}\label{inverse_natural_power_carrier_subseteq}
    $\inverseNaturalPowerCarrier{G}{a}\subseteq\carrier[G]$.
\end{proposition}
\begin{proof}
    Follows by
        \obv{subseteq,inverse_natural_power_carrier_iff}.
\end{proof}

\begin{classification}\label{inverse_natural_power_carrier_is_subset_in_group}
    Let $G$ be a group.
    Then $\inverseNaturalPowerCarrier{G}{a}\subseteq\carrier[G]$.
\end{classification}
\begin{proof}
    Follows by \obv{inverse_natural_power_carrier_subseteq}.
\end{proof}

\begin{proposition}\label{inverse_group_power_in_cyclic_subgroup}
    Let $G$ be a group.
    Let $a\in\carrier[G]$.
    Let $n$ be a natural number.
    Then
        $\groupInv[G](\groupPower{G}{a}{n})
            \in\cyclicSubgroup{G}{a}$.
\end{proposition}
\begin{proof}
    $\subgroup{\cyclicSubgroup{G}{a}}{G}$ by
        \obv{cyclic_subgroup_is_subgroup}.
    $a\in\cyclicSubgroup{G}{a}$ by
        \obv{cyclic_subgroup_contains_generator}.
    $\groupPower{G}{a}{n}\in\cyclicSubgroup{G}{a}$ by
        \obv{group_power_in_subgroup}.
    Follows by \obv{subgroup_inv}.
\end{proof}

\begin{theorem}\label{inverse_natural_power_carrier_subseteq_cyclic_subgroup}
    Let $G$ be a group.
    Let $a\in\carrier[G]$.
    Then
        $\inverseNaturalPowerCarrier{G}{a}\subseteq
            \cyclicSubgroup{G}{a}$.
\end{theorem}
\begin{proof}
    Show that for every $x\in\inverseNaturalPowerCarrier{G}{a}$
        we have $x\in\cyclicSubgroup{G}{a}$.
    \begin{subproof}
        Fix $x\in\inverseNaturalPowerCarrier{G}{a}$.
        Take $n\in\naturals$ such that
            $x=\groupInv[G](\groupPower{G}{a}{n})$ by
            \obv{inverse_natural_power_carrier_iff}.
        $\groupInv[G](\groupPower{G}{a}{n})
            \in\cyclicSubgroup{G}{a}$ by
            \obv{inverse_group_power_in_cyclic_subgroup}.
    \end{subproof}
    Follows by \obv{subseteq}.
\end{proof}

\begin{theorem}\label{cyclic_subgroup_generated_by_natural_powers}
    Let $G$ be a group.
    Let $a\in\carrier[G]$.
    Then
        $\cyclicSubgroup{G}{a}=
            \generatedSubgroup{\naturalPowerCarrier{G}{a}}{G}$.
\end{theorem}
\begin{proof}
    Let $P=\naturalPowerCarrier{G}{a}$.
    Let $C=\cyclicSubgroup{G}{a}$.
    $\{a\}\subseteq\carrier[G]$ by
        \obv{singleton_subset_intro}.
    $P\subseteq\carrier[G]$ by
        \auto{natural_power_carrier_subseteq}.
    $a\in P$ by
        \obv{natural_power_carrier_contains_generator}.
    Hence $\{a\}\subseteq P$ by \obv{singleton_subset_intro}.
    Hence $C\subseteq\generatedSubgroup{P}{G}$ by
        \auto{cyclic_subgroup,generated_subgroup_monotone}.
    $P\subseteq C$ by
        \auto{natural_power_carrier_subseteq_cyclic_subgroup}.
    $\subgroup{C}{G}$ by
        \auto{cyclic_subgroup_is_subgroup}.
    Hence $\generatedSubgroup{P}{G}\subseteq C$ by
        \obv{generated_subgroup_least}.
    Follows by \obv{subseteq_antisymmetric}.
\end{proof}

\begin{proposition}\label{cyclic_subgroup_subseteq_generated_signed_natural_powers}
    Let $G$ be a group.
    Let $a\in\carrier[G]$.
    Then
        $\cyclicSubgroup{G}{a}\subseteq
            \generatedSubgroup{
                \naturalPowerCarrier{G}{a}\union
                    \inverseNaturalPowerCarrier{G}{a}}{G}$.
\end{proposition}
\begin{proof}
    Let $P=\naturalPowerCarrier{G}{a}$.
    Let $Q=\inverseNaturalPowerCarrier{G}{a}$.
    Let $C=\cyclicSubgroup{G}{a}$.
    $P\subseteq\carrier[G]$ and $P\subseteq P\union Q$ by
        \auto{natural_power_carrier_subseteq,union_upper_left}.
    $P\union Q\subseteq\carrier[G]$ by
        \auto{natural_power_carrier_subseteq,
            inverse_natural_power_carrier_subseteq,
            union_subsets_is_subset}.
    $C=\generatedSubgroup{P}{G}$ by
        \auto{cyclic_subgroup_generated_by_natural_powers}.
    $\generatedSubgroup{P}{G}\subseteq
        \generatedSubgroup{P\union Q}{G}$ by
        \obv{generated_subgroup_monotone}.
\end{proof}

\begin{proposition}\label{generated_signed_natural_powers_subseteq_cyclic_subgroup}
    Let $G$ be a group.
    Let $a\in\carrier[G]$.
    Then
        $\generatedSubgroup{
            \naturalPowerCarrier{G}{a}\union
                \inverseNaturalPowerCarrier{G}{a}}{G}
            \subseteq\cyclicSubgroup{G}{a}$.
\end{proposition}
\begin{proof}
    Let $P=\naturalPowerCarrier{G}{a}$.
    Let $Q=\inverseNaturalPowerCarrier{G}{a}$.
    Let $C=\cyclicSubgroup{G}{a}$.
    $P,Q\subseteq C$ by
        \auto{natural_power_carrier_subseteq_cyclic_subgroup,
            inverse_natural_power_carrier_subseteq_cyclic_subgroup}.
    Hence $P\union Q\subseteq C$ by
        \obv{union_subsets_is_subset}.
    $\subgroup{C}{G}$ by
        \auto{cyclic_subgroup_is_subgroup}.
    Hence $C\subseteq\carrier[G]$ by \obv{subgroup_subseteq}.
    Hence $P\union Q\subseteq\carrier[G]$ by
        \obv{subseteq_transitive}.
    Hence $\generatedSubgroup{P\union Q}{G}\subseteq C$ by
        \obv{generated_subgroup_least}.
\end{proof}

\begin{theorem}\label{cyclic_subgroup_generated_by_signed_natural_powers}
    Let $G$ be a group.
    Let $a\in\carrier[G]$.
    Then
        $\cyclicSubgroup{G}{a}=
            \generatedSubgroup{
                \naturalPowerCarrier{G}{a}\union
                    \inverseNaturalPowerCarrier{G}{a}}{G}$.
\end{theorem}
\begin{proof}
    $\cyclicSubgroup{G}{a}\subseteq
        \generatedSubgroup{
            \naturalPowerCarrier{G}{a}\union
                \inverseNaturalPowerCarrier{G}{a}}{G}$ by
        \obv{cyclic_subgroup_subseteq_generated_signed_natural_powers}.
    $\generatedSubgroup{
        \naturalPowerCarrier{G}{a}\union
            \inverseNaturalPowerCarrier{G}{a}}{G}
        \subseteq\cyclicSubgroup{G}{a}$ by
        \obv{generated_signed_natural_powers_subseteq_cyclic_subgroup}.
    Follows by \obv{subseteq_antisymmetric}.
\end{proof}

\begin{definition}\label{cyclic_group}
    Let $G$ be a group.
    $\cyclicGroup{G}$ iff
        $G$ is a group and
        there exists $a\in\carrier[G]$ such that
            $\cyclicSubgroup{G}{a}=\carrier[G]$.
\end{definition}

\begin{proposition}\label{cyclic_group_iff_singleton_generates}
    Let $G$ be a group.
    $\cyclicGroup{G}$ iff
        there exists $a\in\carrier[G]$ such that
            $\generatesGroup{\{a\}}{G}$.
\end{proposition}
\begin{proof}
    Show that if $\cyclicGroup{G}$, then
        there exists $a\in\carrier[G]$ such that
            $\generatesGroup{\{a\}}{G}$.
    \begin{subproof}
        Assume $\cyclicGroup{G}$.
        Hence $G$ is a group and there exists $a\in\carrier[G]$
            such that $\cyclicSubgroup{G}{a}=\carrier[G]$ by
            \obv{cyclic_group}.
        Take $a\in\carrier[G]$ such that
            $\cyclicSubgroup{G}{a}=\carrier[G]$.
        $\{a\}\subseteq\carrier[G]$ by
            \obv{singleton_subset_intro}.
        $\generatedSubgroup{\{a\}}{G}=\carrier[G]$ by
            \obv{cyclic_subgroup}.
        Hence $\generatesGroup{\{a\}}{G}$ by
            \obv{generates_group}.
    \end{subproof}
    Show that if there exists $a\in\carrier[G]$ such that
        $\generatesGroup{\{a\}}{G}$, then $\cyclicGroup{G}$.
    \begin{subproof}
        Assume that there exists $a\in\carrier[G]$ such that
            $\generatesGroup{\{a\}}{G}$.
        Take $a\in\carrier[G]$ such that
            $\generatesGroup{\{a\}}{G}$.
        $\{a\}\subseteq\carrier[G]$ by
            \obv{singleton_subset_intro}.
        $\generatedSubgroup{\{a\}}{G}=\carrier[G]$ by
            \obv{generates_group_equation}.
        Hence $\cyclicSubgroup{G}{a}=\carrier[G]$ by
            \obv{cyclic_subgroup}.
        Follows by \obv{cyclic_group}.
    \end{subproof}
    Follows by \auto.
\end{proof}
