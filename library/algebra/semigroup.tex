\import{algebra/magma.tex}

\section{Semigroups}

As for magmas, semigroup carriers may be empty; associativity is then
vacuous. No hidden inhabitation requirement is imposed.

\begin{struct}\label{semigroup}
    A semigroup $A$ is a magma such that
    \begin{enumerate}
        \item\label{semigroup_assoc}
            for all $a,b,c\in\carrier[A]$ we have
                $\mul[A](a,\mul[A](b,c))
                    =\mul[A](\mul[A](a,b),c)$.
    \end{enumerate}
\end{struct}

\subsection{Regular semigroups}

\begin{struct}\label{regularsemigroup}
    A regular semigroup $A$ is a semigroup such that
    \begin{enumerate}
        \item\label{regularsemigroup_regular}
            for every $a\in\carrier[A]$ there exists $b\in\carrier[A]$
                such that $\mul[A](a,\mul[A](b,a))=a$.
    \end{enumerate}
\end{struct}

\subsection{Inverse semigroups}

\begin{struct}\label{inversesemigroup}
    An inverse semigroup $A$ is a regular semigroup such that
    \begin{enumerate}
        \item\label{inversesemigroup_comm}
            for all $a,b\in\idempotents{A}$ we have
                $\mul[A](a,b)=\mul[A](b,a)$.
    \end{enumerate}
\end{struct}

\begin{proposition}\label{inversesemigroup_is_semigroup}
    Suppose $A$ is an inverse semigroup.
    Then $A$ is a semigroup.
\end{proposition}
\begin{proof}
    Follows by \obv{inversesemigroup,regularsemigroup}.
\end{proof}


\begin{proposition}\label{inversesemigroup_is_regularsemigroup}
    Suppose $A$ is an inverse semigroup.
    Then $A$ is a regular semigroup.
\end{proposition}
\begin{proof}
    Follows by \obv{inversesemigroup}.
\end{proof}

\begin{proposition}\label{idempotentelems_eq_iff_orbits_eq}
    Let $A$ be an inverse semigroup.
    Let $e,f$ be idempotent elements of $A$.
    Then $e=f$ iff $\LeftOrb{e}{A}=\LeftOrb{f}{A}$.
\end{proposition}
\begin{proof}
    If $e=f$, then $\LeftOrb{e}{A}=\LeftOrb{f}{A}$.
    Show that if $\LeftOrb{e}{A}=\LeftOrb{f}{A}$, then $e=f$.
    \begin{subproof}
        Assume $\LeftOrb{e}{A}=\LeftOrb{f}{A}$.
        $A$ is a semigroup by
            \obv{inversesemigroup_is_semigroup}.
        $e,f\in\carrier[A]$ and
            $\mul[A](e,e)=e$ and $\mul[A](f,f)=f$ by
            \obv{idempotentelement}.
        Take $x\in\carrier[A]$ such that
            $\mul[A](e,e)=\mul[A](x,f)$
            by \auto{eq_left_orbit_witness,semigroup,cdot_indexed}.
        Take $y\in\carrier[A]$ such that
            $\mul[A](f,f)=\mul[A](y,e)$
            by \auto{eq_left_orbit_witness,semigroup,cdot_indexed}.
        Hence $e=\mul[A](x,f)$ and $f=\mul[A](y,e)$.
        $\mul[A](e,f)=e$ and $\mul[A](f,e)=f$
            by \auto{idempotentelement,semigroup_assoc}.
        $e,f\in\idempotents{A}$ by
            \obv{idempotents,cdot_indexed}.
        $\mul[A](e,f)=\mul[A](f,e)$
            by \obv{inversesemigroup_comm}.
    \end{subproof}
    Follows by \auto.
\end{proof}
