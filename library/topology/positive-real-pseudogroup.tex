\section{The positive real pseudogroup}

\import{topology/positive-real-change.tex}
\import{topology/full-pseudogroup.tex}

The positive pseudogroup retains local order information that the full
topological pseudogroup forgets.

\begin{definition}\label{positive_real_changes}
    $\PositiveRealChanges=\{h\in\openPartialHomeomorphisms{\RealTopology}{\RealTopology}
        \mid\text{$h$ is a positive real coordinate change}\}$.
\end{definition}

\begin{theorem}\label{positive_real_changes_iff}
    Then $h\in\PositiveRealChanges$ iff $h$ is a positive real coordinate change.
\end{theorem}
\begin{proof}
    If $h\in\PositiveRealChanges$, then $h$ is a positive real coordinate change by
        \obv{positive_real_changes}.
    Show that if $h$ is a positive real coordinate change, then $h\in\PositiveRealChanges$.
    \begin{subproof}
        Assume $h$ is a positive real coordinate change.
        $h$ is an open partial homeomorphism from $\RealTopology$ to $\RealTopology$ by
            \obv{positive_real_change}.
        $h\in\openPartialHomeomorphisms{\RealTopology}{\RealTopology}$ by
            \auto{open_partial_homeomorphisms_iff}.
        Follows by \obv{positive_real_changes}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{instance}\label{positive_real_pseudogroup}
    The coordinate pseudogroup $\PositiveRealPseudogroup=
        \structure{\model=\RealTopology,\changes=\PositiveRealChanges}$.
\end{instance}
\begin{proof}
    Let $G=\PositiveRealPseudogroup$.
    $\RealTopology$ is a topological space by \obv{real_topology_is_topological_space}.
    $\model[G]=\RealTopology$ and $\changes[G]=\PositiveRealChanges$ by
        \auto{coordinate_pseudogroupaggregateprojection_model,
            coordinate_pseudogroupaggregateprojection_changes}.
    $G$ is a record by \obv{coordinate_pseudogroupaggregateexact}.
    Every element of $\changes[G]$ is a positive real coordinate change by
        \auto{positive_real_changes_iff}.
    Every element of $\changes[G]$ is an open partial homeomorphism from
        $\model[G]$ to $\model[G]$ by \auto{positive_real_change}.
    $\carrier[\RealTopology]\in\opens[\RealTopology]$ by \obv{carrier_open}.
    $\identity{\carrier[\RealTopology]}$ is a positive real coordinate change by
        \obv{open_identity_positive_real_change}.
    $\identity{\carrier[\model[G]]}\in\changes[G]$ by
        \auto{positive_real_changes_iff}.
    Show that for every $h\in\changes[G]$ we have for every $U\in\opens[\model[G]]$
        we have $\restrl{h}{U}\in\changes[G]$.
    \begin{subproof}
        Fix $h\in\changes[G]$ and $U\in\opens[\model[G]]$.
        $h$ is a positive real coordinate change.
        $\restrl{h}{U}$ is a positive real coordinate change by
            \auto{positive_real_change_restriction}.
        Follows by \auto{positive_real_changes_iff}.
    \end{subproof}
    Show that for every $h\in\changes[G]$ we have $\converse{h}\in\changes[G]$.
    \begin{subproof}
        Fix $h\in\changes[G]$.
        $h$ is a positive real coordinate change.
        $\converse{h}$ is a positive real coordinate change by
            \obv{positive_real_change_converse}.
        Follows by \auto{positive_real_changes_iff}.
    \end{subproof}
    Show that for all $f,g\in\changes[G]$ we have $g\circ f\in\changes[G]$.
    \begin{subproof}
        Fix $f,g\in\changes[G]$.
        $f$ and $g$ are positive real coordinate changes.
        $g\circ f$ is a positive real coordinate change by
            \obv{positive_real_change_composition}.
        Follows by \auto{positive_real_changes_iff}.
    \end{subproof}
    Show that for every $h$ such that $h$ is an open partial homeomorphism
        from $\model[G]$ to $\model[G]$ we have
        for every $B\subseteq\opens[\model[G]]$ such that
            (for every $x\in\dom{h}$ there exists $U\in B$ such that $x\in U$) and
            (for every $U\in B$ we have $\restrl{h}{U}\in\changes[G]$)
        we have $h\in\changes[G]$.
    \begin{subproof}
        Fix $h$ such that $h$ is an open partial homeomorphism
            from $\model[G]$ to $\model[G]$.
        Fix $B\subseteq\opens[\model[G]]$ such that
            (for every $x\in\dom{h}$ there exists $U\in B$ such that $x\in U$) and
            (for every $U\in B$ we have $\restrl{h}{U}\in\changes[G]$).
        For every $U\in B$ we have $\restrl{h}{U}$ is a positive real coordinate change
            by \auto{positive_real_changes_iff}.
        $h$ is a positive real coordinate change by \auto{positive_real_change_locality}.
        Follows by \auto{positive_real_changes_iff}.
    \end{subproof}
    $G$ is a coordinate pseudogroup by \auto{coordinate_pseudogroup}.
    Follows by \auto{coordinate_pseudogroup}.
\end{proof}

\begin{theorem}\label{positive_real_pseudogroup_model}
    Then $\model[\PositiveRealPseudogroup]=\RealTopology$.
\end{theorem}
\begin{proof}
    Follows by \obv{positive_real_pseudogroupprojection_model}.
\end{proof}

\begin{classification}\label{positive_real_pseudogroup_on}
    Then $\PositiveRealPseudogroup$ is a pseudogroup on $\RealTopology$.
\end{classification}
\begin{proof}
    $\PositiveRealPseudogroup$ is a coordinate pseudogroup by \obv{positive_real_pseudogroup}.
    $\model[\PositiveRealPseudogroup]=\RealTopology$ by \obv{positive_real_pseudogroup_model}.
    Follows by \obv{pseudogroup_on}.
\end{proof}

\begin{theorem}\label{positive_real_pseudogroup_changes}
    Then $h\in\changes[\PositiveRealPseudogroup]$ iff
        $h$ is a positive real coordinate change.
\end{theorem}
\begin{proof}
    $\changes[\PositiveRealPseudogroup]=\PositiveRealChanges$ by
        \obv{positive_real_pseudogroupprojection_changes}.
    Follows by \auto{positive_real_changes_iff}.
\end{proof}

\begin{classification}\label{positive_real_pseudogroup_in_full}
    Then $\PositiveRealPseudogroup$ is a subpseudogroup of $\FullPseudogroup{\RealTopology}$.
\end{classification}
\begin{proof}
    $\PositiveRealPseudogroup$ is a coordinate pseudogroup by \obv{positive_real_pseudogroup}.
    $\model[\PositiveRealPseudogroup]=\RealTopology$ by
        \obv{positive_real_pseudogroup_model}.
    Follows by \auto{full_pseudogroup_maximal}.
\end{proof}
