\section{Basic properties of topological manifolds}

Local path connectedness and local compactness come from the Euclidean charts
alone. Regularity, separation, and first countability use the additional
manifold hypotheses.

\begin{theorem}\label{locally_euclidean_is_locally_path_connected}
    Let $n$ be a natural number and $X$ be a topological space.
    Suppose $X$ is locally Euclidean of dimension $n$.
    Then $X$ is locally path-connected.
\end{theorem}
\begin{proof}
    $n$ is finite by \obv{natural_number_is_finite}.
    $\RealCoordinateSpace{n}$ is a locally path-connected topological space by
        \obv{real_coordinate_space_is_locally_path_connected,real_coordinate_space_is_topological_space}.
    Show that for every $x\in\carrier[X]$ there exists $U\in\opens[X]$ such that
        $x\in U$ and $\subspaceTopology{X}{U}$ is locally path-connected.
    \begin{subproof}
        Fix $x\in\carrier[X]$.
        Take $U,V,h$ such that $U\in\opens[X]$ and $x\in U$ and
            $V\in\opens[\RealCoordinateSpace{n}]$ and $h$ is a homeomorphism from
            $\subspaceTopology{X}{U}$ to $\subspaceTopology{\RealCoordinateSpace{n}}{V}$ by
            \auto{locally_euclidean}.
        $\subspaceTopology{\RealCoordinateSpace{n}}{V}$ is locally path-connected by
            \obv{open_subspace_is_locally_path_connected}.
        $\converse{h}$ is a homeomorphism from $\subspaceTopology{\RealCoordinateSpace{n}}{V}$ to
            $\subspaceTopology{X}{U}$ by \obv{homeomorphism_inverse}.
        $\subspaceTopology{X}{U}$ is locally path-connected by
            \auto{homeomorphism_preserves_locally_path_connectedness,subspace_topology_is_topological_space}.
    \end{subproof}
    Follows by \obv{locally_path_connected_from_open_neighbourhoods}.
\end{proof}

\begin{theorem}\label{topological_manifold_is_locally_path_connected}
    Let $n$ be a natural number and $X$ be a topological space.
    Suppose $X$ is a topological manifold of dimension $n$.
    Then $X$ is locally path-connected.
\end{theorem}
\begin{proof}
    $X$ is locally Euclidean of dimension $n$ by \obv{topological_manifold}.
    Follows by \obv{locally_euclidean_is_locally_path_connected}.
\end{proof}

\begin{theorem}\label{locally_euclidean_is_locally_compact}
    Let $n$ be a natural number and $X$ be a topological space.
    Suppose $X$ is locally Euclidean of dimension $n$.
    Then $X$ is locally compact.
\end{theorem}
\begin{proof}
    $n$ is finite by \obv{natural_number_is_finite}.
    Let $E=\RealCoordinateSpace{n}$.
    $E$ is a topological space by \obv{real_coordinate_space_is_topological_space}.
    $E$ is Hausdorff and $E$ is locally compact by
        \obv{real_coordinate_space_is_hausdorff,real_coordinate_space_is_locally_compact}.
    Show that for every $x\in\carrier[X]$ there exists $U\in\opens[X]$ such that
        $x\in U$ and $\subspaceTopology{X}{U}$ is locally compact.
    \begin{subproof}
        Fix $x\in\carrier[X]$.
        Take $U,V,h$ such that $U\in\opens[X]$ and $x\in U$ and
            $V\in\opens[E]$ and $h$ is a homeomorphism from
            $\subspaceTopology{X}{U}$ to $\subspaceTopology{E}{V}$ by
            \auto{locally_euclidean}.
        $\subspaceTopology{E}{V}$ is locally compact by
            \obv{open_subspace_of_locally_compact_hausdorff_is_locally_compact}.
        $\converse{h}$ is a homeomorphism from $\subspaceTopology{E}{V}$ to
            $\subspaceTopology{X}{U}$ by \obv{homeomorphism_inverse}.
        $\subspaceTopology{X}{U}$ is locally compact by
            \auto{homeomorphism_preserves_local_compactness,subspace_topology_is_topological_space}.
        Follows by \auto.
    \end{subproof}
    Follows by \obv{locally_compact_from_open_neighbourhoods}.
\end{proof}

\begin{theorem}\label{topological_manifold_is_locally_compact}

\begin{proof}

    Follows by \obv{locally_euclidean_is_locally_compact}.
\end{proof}

\begin{theorem}\label{topological_manifold_is_regular}
    Let $n$ be a natural number and $X$ be a topological space.
    Suppose $X$ is a topological manifold of dimension $n$.
    Then $X$ is regular.
\end{theorem}
\begin{proof}
    $X$ is locally compact by \obv{topological_manifold_is_locally_compact}.
    $X$ is Hausdorff by \obv{topological_manifold}.
    Follows by \obv{locally_compact_hausdorff_is_regular}.
\end{proof}
