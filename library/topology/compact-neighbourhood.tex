\section{Compact neighbourhoods in Hausdorff spaces}

A point outside a compact subset of a Hausdorff space can be separated from
that entire subset. A finite subcover combines the individual Hausdorff
separations; the empty compact subset is included in the same argument.

\begin{theorem}\label{hausdorff_point_compact_separation}
    Let $X$ be a topological space.
    Suppose $X$ is Hausdorff.
    Suppose $K$ is compact in $X$ and $p\in\carrier[X]$ and $p\notin K$.
    Then there exist $U,V\in\opens[X]$ such that
        $p\in U$ and $K\subseteq V$ and $U\inter V=\emptyset$.
\end{theorem}
\begin{proof}
    Let $S=\subspaceTopology{X}{K}$.
    $K\subseteq\carrier[X]$ and $S$ is compact by \obv{compact_in}.
    $\carrier[S]=K$ by \auto{subspace_topology_carrier_of_subset}.
    Let $C=\{W\in\opens[S]\mid\text{there exist $U,V\in\opens[X]$ such that
        $p\in U$ and $W\subseteq V$ and $U\inter V=\emptyset$}\}$.
    Show that $C$ covers $\carrier[S]$.
    \begin{subproof}
        Fix $q\in\carrier[S]$.
        $q\in\carrier[X]$ and $p\neq q$ by \auto{subseteq}.
        Take $U,V\in\opens[X]$ such that $p\in U$ and $q\in V$ and
            $U\inter V=\emptyset$ by \auto{is_hausdorff,disjoint,inter,emptyset,setext}.
        $V\inter K\in\opens[S]$ by \auto{subspace_open_iff}.
        $V\inter K\subseteq V$ by \obv{inter_lower_left}.
        Hence $V\inter K\in C$ by \auto.
        $q\in V\inter K$ by \auto{inter}.
        Follows by \auto.
    \end{subproof}
    $C\subseteq\opens[S]$ by \auto{subseteq}.
    $C$ is an open cover of $S$ by \obv{open_cover}.
    Take a finite set $D$ such that $D\subseteq C$ and
        $D$ covers $\carrier[S]$ by \obv{is_compact}.
    Let $H=\{B\in\pow{D}\mid\text{there exist $U,V\in\opens[X]$ such that
        $p\in U$ and $\unions{B}\subseteq V$ and $U\inter V=\emptyset$}\}$.
    $\carrier[X]\in\opens[X]$ and $\emptyset\in\opens[X]$ by
        \auto{carrier_open,emptyset_open,is_open}.
    $\emptyset\in H$ by \auto{powerset_bottom,unions_emptyset,emptyset_subseteq,inter_emptyset}.
    Show that for every $B,W$ such that $B\in H$ and $W\in D$ we have
        $\cons{W}{B}\in H$.
    \begin{subproof}
        Fix $B,W$.
        Assume $B\in H$ and $W\in D$.
        $B\in\pow{D}$.
        $B\subseteq D$ by \obv{pow_iff}.
        $\cons{W}{B}\subseteq D$ by \obv{cons_subseteq_intro}.
        $\cons{W}{B}\in\pow{D}$ by \obv{pow_iff}.
        Take $U,V\in\opens[X]$ such that $p\in U$ and
            $\unions{B}\subseteq V$ and $U\inter V=\emptyset$ by \auto.
        $W\in C$ by \obv{subseteq}.
        Take $N,O\in\opens[X]$ such that $p\in N$ and
            $W\subseteq O$ and $N\inter O=\emptyset$ by \auto.
        $U\inter N\in\opens[X]$ by \obv{opens_inter}.
        $V\union O\in\opens[X]$ by \auto{union_open,is_open}.
        $p\in U\inter N$ by \obv{inter}.
        $\unions{\cons{W}{B}}\subseteq V\union O$ by
            \auto{unions_cons,union,subseteq}.
        $(U\inter N)\inter V=\emptyset$ by
            \auto{inter_comm,inter_assoc,inter_emptyset}.
        $(U\inter N)\inter O=\emptyset$ by
            \auto{inter_assoc,inter_emptyset}.
        Hence $(U\inter N)\inter(V\union O)=\emptyset$
            by \auto{inter_distrib_union,union_emptyset}.
        Follows by \auto.
    \end{subproof}
    $D\in H$ by \obv{finite_induction}.
    Take $U,V\in\opens[X]$ such that $p\in U$ and
        $\unions{D}\subseteq V$ and $U\inter V=\emptyset$ by \auto.
    $K\subseteq\unions{D}$ by \auto{unions_intro,subseteq}.
    Follows by \auto{subseteq_transitive}.
\end{proof}

A compact neighbourhood can now be shrunk inside any prescribed open set.
Separate its part outside that open set from the given point, then remove
the separating open set from the original compact neighbourhood.

\begin{theorem}\label{locally_compact_hausdorff_compact_neighbourhood_in_open}
    Let $X$ be a locally compact topological space.
    Suppose $X$ is Hausdorff and $A\in\opens[X]$ and $p\in A$.
    Then there exist $U,K$ such that $U\in\opens[X]$ and
        $p\in U\subseteq K$ and $K\subseteq A$ and $K$ is compact in $X$.
\end{theorem}
\begin{proof}
    $p\in\carrier[X]$ by \auto{opens_type,subseteq}.
    Take $W,L$ such that $W\in\opens[X]$ and $p\in W\subseteq L$ and
        $L$ is compact in $X$ by \obv{is_locally_compact}.
    Let $S=\subspaceTopology{X}{L}$.
    $S$ is a topological space by \obv{subspace_topology_is_topological_space}.
    $L\subseteq\carrier[X]$ and $S$ is compact by \obv{compact_in}.
    $\carrier[S]=L$ by \auto{subspace_topology_carrier_of_subset}.
    $A\inter L\in\opens[S]$ by \auto{subspace_open_iff}.
    $A\inter L\subseteq\carrier[S]$ by \auto{inter_lower_right}.
    $L\setminus A=\carrier[S]\setminus(A\inter L)$ by \auto{setext,setminus,inter}.
    Hence $L\setminus A$ is closed in $S$ by
        \auto{complement_of_open_is_closed,is_open}.
    $L\setminus A$ is compact in $S$ by \obv{closed_subset_of_compact_is_compact}.
    $L\setminus A\subseteq L$ by \obv{setminus_subseteq}.
    $L\setminus A$ is compact in $X$ by \auto{compact_in_subspace_iff}.
    $p\notin L\setminus A$ by \obv{setminus}.
    Take $N,V\in\opens[X]$ such that $p\in N$ and
        $L\setminus A\subseteq V$ and $N\inter V=\emptyset$ by
        \obv{hausdorff_point_compact_separation}.
    Let $K=L\setminus V$ and $U=W\inter N$.
    $V\inter L\in\opens[S]$ by \auto{subspace_open_iff}.
    $V\inter L\subseteq\carrier[S]$ by \auto{inter_lower_right}.
    $K=\carrier[S]\setminus(V\inter L)$ by \auto{setext,setminus,inter}.
    Hence $K$ is closed in $S$ by \auto{complement_of_open_is_closed,is_open}.
    $K$ is compact in $S$ by \obv{closed_subset_of_compact_is_compact}.
    $K\subseteq L$ by \auto{setminus_subseteq}.
    $K$ is compact in $X$ by \auto{compact_in_subspace_iff}.
    $U\in\opens[X]$ by \auto{opens_inter}.
    $p\in U$ by \auto{inter}.
    $U\subseteq K$ and $K\subseteq A$ by \auto{setminus,inter,subseteq,emptyset}.
    Follows by \auto.
\end{proof}

\begin{theorem}\label{open_subspace_of_locally_compact_hausdorff_is_locally_compact}
    Let $X$ be a locally compact topological space.
    Suppose $X$ is Hausdorff and $A\in\opens[X]$.
    Then $\subspaceTopology{X}{A}$ is locally compact.
\end{theorem}
\begin{proof}
    Let $S=\subspaceTopology{X}{A}$.
    $S$ is a topological space by \obv{subspace_topology_is_topological_space}.
    $A\subseteq\carrier[X]$ by \auto{opens_type,subseteq}.
    $\carrier[S]=A$ by \auto{subspace_topology_carrier_of_subset}.
    Show that for every $p\in\carrier[S]$ there exist $U,K$ such that
        $U\in\opens[S]$ and $p\in U\subseteq K$ and $K$ is compact in $S$.
    \begin{subproof}
        Fix $p\in\carrier[S]$.
        Take $U,K$ such that $U\in\opens[X]$ and $p\in U\subseteq K$ and
            $K\subseteq A$ and $K$ is compact in $X$ by
            \auto{locally_compact_hausdorff_compact_neighbourhood_in_open}.
        $U\subseteq A$ by \obv{subseteq_transitive}.
        $U\in\opens[S]$ by \auto{open_subspace_open_iff}.
        $K$ is compact in $S$ by \auto{compact_in_subspace_iff}.
        Follows by \auto.
    \end{subproof}
    Follows by \auto{is_locally_compact}.
\end{proof}

\begin{theorem}\label{locally_compact_hausdorff_is_regular}
    Let $X$ be a locally compact topological space.
    Suppose $X$ is Hausdorff.
    Then $X$ is regular.
\end{theorem}
\begin{proof}
    Show that for every $A\in\opens[X]$ we have for every $p\in A$
        there exists $K\in\neighbourhoods{p}{X}$ such that
            $K\subseteq A$ and $K$ is closed in $X$.
    \begin{subproof}
        Fix $A\in\opens[X]$.
        Fix $p\in A$.
        Take $U,K$ such that $U\in\opens[X]$ and $p\in U\subseteq K$ and
            $K\subseteq A$ and $K$ is compact in $X$ by
            \obv{locally_compact_hausdorff_compact_neighbourhood_in_open}.
        $K$ is closed in $X$ by \obv{compact_subset_of_hausdorff_is_closed}.
        $K\subseteq\carrier[X]$ by \obv{compact_in}.
        $K\in\pow{\carrier[X]}$ by \obv{pow_iff}.
        $K\in\neighbourhoods{p}{X}$ by \auto{neighbourhoods}.
        Follows by \auto.
    \end{subproof}
    Follows by \obv{regular_iff_closed_neighbourhood_in_open}.
\end{proof}
