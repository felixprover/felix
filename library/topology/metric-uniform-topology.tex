\section{Agreement of pseudometric and uniform topologies}

The basic entourage of radius $e$ has precisely the metric ball of radius
$e$ as its section at a carrier point.  Arbitrary metric entourages contain
one such basic entourage, which is exactly the comparison needed in the
reverse direction.

\begin{proposition}\label{metric_entourage_uniform_ball}
    Let $M$ be a pseudometric space and $x\in\carrier[M]$.
    Then
        $\uniformBall{\metricUniformSpace{M}}
            {\metricEntourage{M}{e}}{x}
        =\metricBall{M}{x}{e}$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{setext,uniform_ball_iff,metric_entourage_iff,metric_ball_iff}.
\end{proof}

\begin{proposition}\label{metric_ball_below_uniform_ball}
    Let $M$ be a pseudometric space and $x\in\carrier[M]$.
    Suppose $\metricEntourage{M}{e}\subseteq U$.
    Then
        $\metricBall{M}{x}{e}
            \subseteq\uniformBall{\metricUniformSpace{M}}{U}{x}$.
\end{proposition}
\begin{proof}
    Show that for every $y\in\metricBall{M}{x}{e}$ we have
        $y\in\uniformBall{\metricUniformSpace{M}}{U}{x}$.
    \begin{subproof}
        Fix $y\in\metricBall{M}{x}{e}$.
        $y\in\carrier[M]$ and
            $\surrealLt{\metricDistance[M](x,y)}{e}$
            by \obv{metric_ball}.
        Hence $(x,y)\in\metricEntourage{M}{e}$
            by \auto{metric_entourage_iff}.
        Hence $(x,y)\in U$ by \obv{subseteq}.
        Follows by \obv{uniform_ball_iff}.
    \end{subproof}
    Follows by \obv{subseteq}.
\end{proof}

\begin{theorem}\label{metric_open_iff_induced_uniform_open}
    Let $M$ be a pseudometric space.
    Then
        $\metricOpen{M}{A}$ iff
        $\uniformOpen{\metricUniformSpace{M}}{A}$.
\end{theorem}
\begin{proof}
    Show that if $\metricOpen{M}{A}$, then
        $\uniformOpen{\metricUniformSpace{M}}{A}$.
    \begin{subproof}
        Assume $\metricOpen{M}{A}$.
        $A\subseteq\carrier[M]$ by \obv{metric_open}.
        Hence $A\subseteq\carrier[\metricUniformSpace{M}]$
            by \obv{metric_uniform_space_carrier}.
        Show that for every $x\in A$ there exists
            $U\in\uniformity[\metricUniformSpace{M}]$ such that
            $\uniformBall{\metricUniformSpace{M}}{U}{x}\subseteq A$.
        \begin{subproof}
            Fix $x\in A$.
            Take $e\in\reals$ such that
                $\surrealLt{\surrealZero}{e}$ and
                $\metricBall{M}{x}{e}\subseteq A$
                by \obv{metric_open}.
            $x\in\carrier[M]$ by \obv{subseteq}.
            $\metricEntourage{M}{e}\in\metricUniformity{M}$
                by \obv{metric_entourage_is_entourage}.
            Hence
                $\metricEntourage{M}{e}
                    \in\uniformity[\metricUniformSpace{M}]$
                by \obv{metric_uniform_space_uniformity}.
            $\uniformBall{\metricUniformSpace{M}}
                {\metricEntourage{M}{e}}{x}
                =\metricBall{M}{x}{e}$
                by \obv{metric_entourage_uniform_ball}.
        \end{subproof}
        Follows by \obv{uniform_open}.
    \end{subproof}
    Show that if $\uniformOpen{\metricUniformSpace{M}}{A}$, then
        $\metricOpen{M}{A}$.
    \begin{subproof}
        Assume $\uniformOpen{\metricUniformSpace{M}}{A}$.
        $A\subseteq\carrier[\metricUniformSpace{M}]$
            by \obv{uniform_open}.
        Hence $A\subseteq\carrier[M]$
            by \obv{metric_uniform_space_carrier}.
        Show that for every $x\in A$ there exists $e\in\reals$ such that
            $\surrealLt{\surrealZero}{e}$ and
            $\metricBall{M}{x}{e}\subseteq A$.
        \begin{subproof}
            Fix $x\in A$.
            Take $U\in\uniformity[\metricUniformSpace{M}]$ such that
                $\uniformBall{\metricUniformSpace{M}}{U}{x}\subseteq A$
                by \obv{uniform_open}.
            $U\in\metricUniformity{M}$
                by \obv{metric_uniform_space_uniformity}.
            Take $e\in\reals$ such that
                $\surrealLt{\surrealZero}{e}$ and
                $\metricEntourage{M}{e}\subseteq U$
                by \obv{metric_uniformity}.
            $x\in\carrier[M]$ by \obv{subseteq}.
            $\metricBall{M}{x}{e}
                \subseteq\uniformBall{\metricUniformSpace{M}}{U}{x}$
                by \obv{metric_ball_below_uniform_ball}.
            Hence $\metricBall{M}{x}{e}\subseteq A$
                by \obv{subseteq_transitive}.
        \end{subproof}
        Follows by \obv{metric_open}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{theorem}\label{metric_uniform_opens_agree}
    Let $M$ be a pseudometric space.
    Then
        $\metricOpens{M}
            =\uniformOpens{\metricUniformSpace{M}}$.
\end{theorem}
\begin{proof}
    Follows by
        \auto{setext,metric_opens_iff,uniform_opens_iff,metric_open_iff_induced_uniform_open}.
\end{proof}

\begin{proposition}\label{metric_uniform_topology_carriers_agree}
    $\carrier[\metricTopology{M}]
        =\carrier[\uniformTopology{\metricUniformSpace{M}}]$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{metric_topology_carrier,uniform_topology_carrier,metric_uniform_space_carrier}.
\end{proof}

\begin{theorem}\label{metric_uniform_topology_opens_agree}
    Let $M$ be a pseudometric space.
    Then
        $\opens[\metricTopology{M}]
            =\opens[\uniformTopology{\metricUniformSpace{M}}]$.
\end{theorem}
\begin{proof}
    Follows by
        \auto{metric_topology_opens,uniform_topology_opens,metric_uniform_opens_agree}.
\end{proof}

\begin{theorem}\label{metric_uniform_topologies_agree}
    Let $M$ be a pseudometric space.
    Then
        $\metricTopology{M}
            =\uniformTopology{\metricUniformSpace{M}}$.
\end{theorem}
\begin{proof}
    Follows by
        \obv{metric_topology,uniform_topology,metric_uniform_space_carrier,metric_uniform_opens_agree}.
\end{proof}

\begin{theorem}\label{metric_topology_is_hausdorff}
    Let $M$ be a metric space.
    Then $\metricTopology{M}$ is Hausdorff.
\end{theorem}
\begin{proof}
    $\metricUniformSpace{M}$ is a uniform space by
        \obv{metric_uniform_space_is_uniform_space}.
    $\metricUniformSpace{M}$ is separated uniform by
        \obv{metric_uniform_space_is_separated}.
    Hence $\uniformTopology{\metricUniformSpace{M}}$ is Hausdorff by
        \obv{separated_uniform_iff_uniform_topology_hausdorff}.
    $\metricTopology{M}=\uniformTopology{\metricUniformSpace{M}}$ by
        \obv{metric_uniform_topologies_agree}.
\end{proof}
