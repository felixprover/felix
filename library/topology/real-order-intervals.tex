\section{Order intervals in the metric real line}

These are subsets and subspaces of the metric real line.  No order topology
is introduced.

\subsection{Intervals and rays}

\begin{definition}\label{real_closed_interval}
    $\realClosedInterval{a}{b}=\{x\in\reals\mid
        \surrealLe{a}{x}\land\surrealLe{x}{b}\}$.
\end{definition}

\begin{definition}\label{real_open_interval}
    $\realOpenInterval{a}{b}=\{x\in\reals\mid
        \surrealLt{a}{x}\land\surrealLt{x}{b}\}$.
\end{definition}

\begin{definition}\label{real_closed_lower_ray}
    $\realClosedLowerRay{b}=\{x\in\reals\mid\surrealLe{x}{b}\}$.
\end{definition}

\begin{definition}\label{real_closed_upper_ray}
    $\realClosedUpperRay{a}=\{x\in\reals\mid\surrealLe{a}{x}\}$.
\end{definition}

\begin{definition}\label{real_open_lower_ray}
    $\realOpenLowerRay{b}=\{x\in\reals\mid\surrealLt{x}{b}\}$.
\end{definition}

\begin{definition}\label{real_open_upper_ray}
    $\realOpenUpperRay{a}=\{x\in\reals\mid\surrealLt{a}{x}\}$.
\end{definition}

\begin{definition}\label{real_closed_symmetric_interval}
    $\realClosedSymmetricInterval{r}=
        \realClosedInterval{\surrealNeg{r}}{r}$.
\end{definition}

\begin{definition}\label{real_open_symmetric_interval}
    $\realOpenSymmetricInterval{r}=
        \realOpenInterval{\surrealNeg{r}}{r}$.
\end{definition}

\begin{definition}\label{real_open_symmetric_unit_interval}
    $\RealOpenSymmetricUnitInterval=
        \realOpenInterval{\surrealNeg{\surrealOne}}{\surrealOne}$.
\end{definition}

\begin{proposition}\label{real_closed_interval_member_iff}
    $x\in\realClosedInterval{a}{b}$ iff $x\in\reals$ and
        $\surrealLe{a}{x}$ and $\surrealLe{x}{b}$.
\end{proposition}
\begin{proof}
    Follows by \auto{real_closed_interval}.
\end{proof}

\begin{proposition}\label{real_open_interval_member_iff}
    $x\in\realOpenInterval{a}{b}$ iff $x\in\reals$ and
        $\surrealLt{a}{x}$ and $\surrealLt{x}{b}$.
\end{proposition}
\begin{proof}
    Follows by \auto{real_open_interval}.
\end{proof}

\begin{proposition}\label{real_closed_lower_ray_member_iff}

\end{proposition}

\end{proof}

\end{proof}

\begin{theorem}\label{real_open_rays_cover}
    Let $a,b\in\reals$.
    Suppose $\surrealLt{a}{b}$.
    Then $\realOpenLowerRay{b}\union\realOpenUpperRay{a}=\reals$.
\end{theorem}
\begin{proof}
    $\realOpenLowerRay{b}\union\realOpenUpperRay{a}\subseteq\reals$ by
        \auto{real_open_lower_ray_member_iff,real_open_upper_ray_member_iff,
            union_iff,subseteq}.
    Show that for every $x\in\reals$ we have
        $x\in\realOpenLowerRay{b}\union\realOpenUpperRay{a}$.
    \begin{subproof}
        Fix $x\in\reals$.
        $\surreal{x}$ and $\surreal{a}$ and $\surreal{b}$ by
            \obv{surreal_real_is_surreal}.
        $\surrealLt{x}{a}$ or $x=a$ or $\surrealLt{a}{x}$ by
            \obv{surreal_lt_trichotomy}.
        $\surrealLt{x}{b}$ or $\surrealLt{a}{x}$ by \auto{surreal_lt_transitive}.
        Follows by \auto{real_open_lower_ray_member_iff,real_open_upper_ray_member_iff,
            union_iff}.
    \end{subproof}
    Follows by \auto{subseteq,subseteq_antisymmetric}.
\end{proof}

\begin{proposition}\label{real_closed_interval_subset_reals}
    $\realClosedInterval{a}{b}\subseteq\reals$.
\end{proposition}
\begin{proof}
    Follows by \obv{real_closed_interval,subseteq}.
\end{proof}

\begin{proposition}\label{real_open_interval_subset_reals}

\end{proof}

\begin{proposition}\label{real_ray_subsets_reals}
    $\realClosedLowerRay{b},\realClosedUpperRay{a},
        \realOpenLowerRay{b},\realOpenUpperRay{a}\subseteq\reals$.
\end{proposition}
\begin{proof}
    Follows by
        \obv{real_closed_lower_ray,real_closed_upper_ray,real_open_lower_ray,real_open_upper_ray,subseteq}.
\end{proof}

\subsection{Open and closed subsets of the real line}

\begin{theorem}\label{real_open_lower_ray_is_open}
    Suppose $b\in\reals$.
    Then $\realOpenLowerRay{b}\in\opens[\RealTopology]$.
\end{theorem}
\begin{proof}
    Let $U=\realOpenLowerRay{b}$.
    $U\subseteq\carrier[\RealMetric]$
        by \auto{real_ray_subsets_reals,surreal_real_metric_carrier}.
    Show that for every $x\in U$ there exists $r\in\reals$ such that
        $\surrealLt{\surrealZero}{r}$ and
        $\metricBall{\RealMetric}{x}{r}\subseteq U$.
    \begin{subproof}
        Fix $x\in U$.
        $x\in\reals$ and $\surrealLt{x}{b}$
            by \obv{real_open_lower_ray_member_iff}.
        Let $r=\surrealAdd{b}{\surrealNeg{x}}$.
        $\surrealNeg{x}\in\reals$ by \obv{surreal_real_negation_closed}.
        Hence $r\in\reals$ by \auto{surreal_real_addition_closed}.
        $\surrealLt{\surrealZero}{r}$
            by \obv{surreal_subtraction_positive,surreal_real_is_surreal}.
        Show that $\metricBall{\RealMetric}{x}{r}\subseteq U$.
        \begin{subproof}
            Show that for every $y\in\metricBall{\RealMetric}{x}{r}$
                we have $y\in U$.
            \begin{subproof}
                Fix $y\in\metricBall{\RealMetric}{x}{r}$.
                $y\in\reals$ and
                    $\surrealLt{\metricDistance[\RealMetric](x,y)}{r}$
                    by
                    \obv{metric_ball_iff,surreal_real_metric_carrier}.
                $\metricDistance[\RealMetric](x,y)
                    =\abs{\surrealAdd{x}{\surrealNeg{y}}}$
                    by \obv{surreal_real_metric_distance}.
                Hence
                    $\surrealLt{
                        \abs{\surrealAdd{x}{\surrealNeg{y}}}}{r}$.
                Hence $\surrealLt{y}{\surrealAdd{x}{r}}$
                    by
                    \auto{surreal_abs_difference_lt_iff,surreal_real_is_surreal}.
                $\surrealAdd{x}{r}=b$
                    by
                    \auto{surreal_add_associative,surreal_add_commutative,surreal_add_inverse_right,surreal_add_zero_left,surreal_negation_is_surreal,surreal_real_is_surreal}.
                Hence $\surrealLt{y}{b}$.
                Follows by \obv{real_open_lower_ray}.
            \end{subproof}
            Follows by \obv{subseteq}.
        \end{subproof}
    \end{subproof}
    Hence $\metricOpen{\RealMetric}{U}$ by \obv{metric_open}.
    Hence $U\in\metricOpens{\RealMetric}$ by \obv{metric_opens_iff}.
    $\opens[\RealTopology]=\metricOpens{\RealMetric}$
        by \auto{real_topology,metric_topology_opens}.
\end{proof}

\begin{theorem}\label{real_open_upper_ray_is_open}
    Suppose $a\in\reals$.
    Then $\realOpenUpperRay{a}\in\opens[\RealTopology]$.
\end{theorem}
\begin{proof}
    Let $U=\realOpenUpperRay{a}$.
    $U\subseteq\carrier[\RealMetric]$
        by \auto{real_ray_subsets_reals,surreal_real_metric_carrier}.
    Show that for every $x\in U$ there exists $r\in\reals$ such that
        $\surrealLt{\surrealZero}{r}$ and
        $\metricBall{\RealMetric}{x}{r}\subseteq U$.
    \begin{subproof}
        Fix $x\in U$.
        $x\in\reals$ and $\surrealLt{a}{x}$
            by \obv{real_open_upper_ray_member_iff}.
        Let $r=\surrealAdd{x}{\surrealNeg{a}}$.
        $\surrealNeg{a}\in\reals$ by \obv{surreal_real_negation_closed}.
        Hence $r\in\reals$ by \auto{surreal_real_addition_closed}.
        $\surrealLt{\surrealZero}{r}$
            by \obv{surreal_subtraction_positive,surreal_real_is_surreal}.
        Show that $\metricBall{\RealMetric}{x}{r}\subseteq U$.
        \begin{subproof}
            Show that for every $y\in\metricBall{\RealMetric}{x}{r}$
                we have $y\in U$.
            \begin{subproof}
                Fix $y\in\metricBall{\RealMetric}{x}{r}$.
                $y\in\reals$ and
                    $\surrealLt{\metricDistance[\RealMetric](x,y)}{r}$
                    by
                    \obv{metric_ball_iff,surreal_real_metric_carrier}.
                $\metricDistance[\RealMetric](x,y)
                    =\abs{\surrealAdd{x}{\surrealNeg{y}}}$
                    by \obv{surreal_real_metric_distance}.
                Hence
                    $\surrealLt{
                        \abs{\surrealAdd{x}{\surrealNeg{y}}}}{r}$.
                Hence $\surrealLt{x}{\surrealAdd{y}{r}}$
                    by
                    \auto{surreal_abs_difference_lt_iff,surreal_real_is_surreal}.
                $\surrealAdd{y}{r}
                    =\surrealAdd{\surrealAdd{y}{\surrealNeg{a}}}{x}$
                    by
                    \auto{surreal_add_associative,surreal_add_commutative,surreal_negation_is_surreal,surreal_real_is_surreal}.
                $\surrealAdd{\surrealZero}{x}=x$
                    by
                    \obv{surreal_add_zero_left,surreal_real_is_surreal}.
                Hence
                    $\surrealLt{\surrealAdd{\surrealZero}{x}}
                        {\surrealAdd{
                            \surrealAdd{y}{\surrealNeg{a}}}{x}}$.
                Hence
                    $\surrealLt{\surrealZero}
                        {\surrealAdd{y}{\surrealNeg{a}}}$
                    by
                    \auto{surreal_add_lt_cancel_right,surreal_negation_is_surreal,surreal_add_is_surreal,surreal_real_is_surreal,empty_surreal}.
                Hence $\surrealLt{\surrealAdd{\surrealZero}{a}}{y}$
                    by
                    \obv{surreal_lt_subtract_add,surreal_real_is_surreal,empty_surreal}.
                Hence $\surrealLt{a}{y}$
                    by \obv{surreal_add_zero_left,surreal_real_is_surreal}.
                Follows by \obv{real_open_upper_ray}.
            \end{subproof}
            Follows by \obv{subseteq}.
        \end{subproof}
    \end{subproof}
    Hence $\metricOpen{\RealMetric}{U}$ by \obv{metric_open}.
    Hence $U\in\metricOpens{\RealMetric}$ by \obv{metric_opens_iff}.
    $\opens[\RealTopology]=\metricOpens{\RealMetric}$
        by \auto{real_topology,metric_topology_opens}.
\end{proof}

\begin{proposition}\label{real_closed_symmetric_interval_iff_abs_bound}
    Suppose $r\in\reals$ and $\surrealLe{\surrealZero}{r}$.
    Then $x\in\realClosedSymmetricInterval{r}$ iff
        $x\in\reals$ and $\surrealLe{\abs{x}}{r}$.
\end{proposition}
\begin{proof}
    $\surreal{r}$ by \obv{surreal_real_is_surreal}.
    Show that if $x\in\realClosedSymmetricInterval{r}$, then
        $x\in\reals$ and $\surrealLe{\abs{x}}{r}$.
    \begin{subproof}
        Assume $x\in\realClosedSymmetricInterval{r}$.
        Then $x\in\reals$ and $\surrealLe{\surrealNeg{r}}{x}$ and
            $\surrealLe{x}{r}$
            by
            \auto{real_closed_symmetric_interval,real_closed_interval_member_iff}.
        Hence $\surreal{x}$ by \obv{surreal_real_is_surreal}.
            $\surrealLe{\surrealNeg{x}}
                {\surrealNeg{\surrealNeg{r}}}$
            by
            \obv{surreal_negation_le_reverses,surreal_negation_is_surreal}.
        $\surrealNeg{\surrealNeg{r}}=r$
            by \obv{surreal_negation_involutive}.
        Hence $\surrealLe{\surrealNeg{x}}{r}$.
        Hence $\surrealLe{\abs{x}}{r}$
            by \auto{surreal_abs_le_iff}.
    \end{subproof}
    Show that if $x\in\reals$ and $\surrealLe{\abs{x}}{r}$, then
        $x\in\realClosedSymmetricInterval{r}$.
    \begin{subproof}
        Assume $x\in\reals$ and $\surrealLe{\abs{x}}{r}$.
        Hence $\surreal{x}$ by \obv{surreal_real_is_surreal}.
        $\surrealLe{x}{r}$ and $\surrealLe{\surrealNeg{x}}{r}$
            by \obv{surreal_abs_le_iff}.
        Hence
            $\surrealLe{\surrealNeg{r}}{\surrealNeg{\surrealNeg{x}}}$
            by
            \obv{surreal_negation_le_reverses,surreal_negation_is_surreal}.
        $\surrealNeg{\surrealNeg{x}}=x$
            by \obv{surreal_negation_involutive}.
        Hence $\surrealLe{\surrealNeg{r}}{x}$.
        Follows by
            \obv{real_closed_symmetric_interval,real_closed_interval}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\subsection{Closed-interval subspaces}
