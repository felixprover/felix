\section{Neighbourhood filters of uniform spaces}

\import{topology/uniform-topology.tex}
\import{topology/neighbourhood-filter.tex}
\import{set/filter-basis.tex}

The balls of the entourages at a point form a proper filter basis.  Thus the
ordinary neighbourhood filter of the induced topology is exactly the filter
generated by those balls.

\begin{theorem}\label{uniform_neighbourhood_iff}
    Let $X$ be a uniform space and $x\in\carrier[X]$.
    Then $N\in\neighbourhoods{x}{\uniformTopology{X}}$ iff
        $N\subseteq\carrier[X]$ and there exists $U\in\uniformity[X]$
        such that $\uniformBall{X}{U}{x}\subseteq N$.
\end{theorem}
\begin{proof}
    $\carrier[\uniformTopology{X}]=\carrier[X]$ by
        \obv{uniform_topology_carrier}.
    Hence $x\in\carrier[\uniformTopology{X}]$.

    Show that if $N\in\neighbourhoods{x}{\uniformTopology{X}}$, then
        $N\subseteq\carrier[X]$ and there exists $U\in\uniformity[X]$
        such that $\uniformBall{X}{U}{x}\subseteq N$.
    \begin{subproof}
        Assume $N\in\neighbourhoods{x}{\uniformTopology{X}}$.
        $N\subseteq\carrier[\uniformTopology{X}]$ by
            \obv{neighbourhoods,pow_iff,subseteq}.
        Hence $N\subseteq\carrier[X]$.
        Take $A\in\opens[\uniformTopology{X}]$ such that
            $x\in A\subseteq N$ by \obv{neighbourhoods}.
        $A\in\uniformOpens{X}$ by \obv{uniform_topology_opens}.
        Hence $\uniformOpen{X}{A}$ by \obv{uniform_opens_iff}.
        Take $U\in\uniformity[X]$ such that
            $\uniformBall{X}{U}{x}\subseteq A$ by
            \obv{uniform_open}.
        Hence $\uniformBall{X}{U}{x}\subseteq N$ by
            \obv{subseteq_transitive}.
    \end{subproof}

    Show that if $N\subseteq\carrier[X]$ and there exists
        $U\in\uniformity[X]$ such that
        $\uniformBall{X}{U}{x}\subseteq N$, then
        $N\in\neighbourhoods{x}{\uniformTopology{X}}$.
    \begin{subproof}
        Assume $N\subseteq\carrier[X]$ and there exists
            $U\in\uniformity[X]$ such that
            $\uniformBall{X}{U}{x}\subseteq N$.
        Take $U\in\uniformity[X]$ such that
            $\uniformBall{X}{U}{x}\subseteq N$.
        Take $A\in\opens[\uniformTopology{X}]$ such that
            $x\in A\subseteq\uniformBall{X}{U}{x}$ by
            \obv{uniform_ball_contains_open_neighbourhood}.
        Hence $A\subseteq N$ by \obv{subseteq_transitive}.
        $N\in\pow{\carrier[\uniformTopology{X}]}$ by
            \obv{powerset_intro}.
        Follows by \obv{neighbourhoods}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{theorem}\label{uniform_neighbourhood_filter_is_generated_by_balls}
    Let $X$ be a uniform space and $x\in\carrier[X]$.
    Then
        $\neighbourhoods{x}{\uniformTopology{X}}
        =\generatedFilter{\carrier[X]}{
            \{A\in\pow{\carrier[X]}\mid
                \text{there exists $U\in\uniformity[X]$ such that
                    $A=\uniformBall{X}{U}{x}$}\}}$.
\end{theorem}
\begin{proof}
    Let $B=\{A\in\pow{\carrier[X]}\mid
        \text{there exists $U\in\uniformity[X]$ such that
            $A=\uniformBall{X}{U}{x}$}\}$.

    Show that $B$ is a filter basis on $\carrier[X]$.
    \begin{subproof}
        $B$ is a family of subsets of $\carrier[X]$ by
            \obv{pow_iff,subseteq}.
        Let $T=\carrier[X]\times\carrier[X]$.
        $T\in\uniformity[X]$ by \auto{uniform_top_entourage}.
        $\uniformBall{X}{T}{x}\subseteq\carrier[X]$ by
            \obv{uniform_ball_subset_carrier}.
        Hence $\uniformBall{X}{T}{x}\in\pow{\carrier[X]}$ by
            \obv{powerset_intro}.
        Thus $\uniformBall{X}{T}{x}\in B$.
        Hence $B$ is inhabited.

        Show that for all $A,C$ such that $A\in B$ and $C\in B$ there
            exists $D\in B$ such that $D\subseteq A\inter C$.
        \begin{subproof}
            Fix $A,C$.
            Assume $A\in B$ and $C\in B$.
            Take $U\in\uniformity[X]$ such that
                $A=\uniformBall{X}{U}{x}$.
            Take $V\in\uniformity[X]$ such that
                $C=\uniformBall{X}{V}{x}$.
            Let $W=U\inter V$.
            $U\inter V\in\uniformity[X]$ by
                \obv{uniform_entourage_intersection}.
            Hence $W\in\uniformity[X]$.
            Let $D=\uniformBall{X}{W}{x}$.
            $D\subseteq\carrier[X]$ by
                \auto{uniform_ball_subset_carrier}.
            Hence $D\in\pow{\carrier[X]}$ by
                \obv{powerset_intro}.
            Thus $D\in B$.
            Show that $D\subseteq A\inter C$.
            \begin{subproof}
                It suffices to show that every element of $D$ is an
                    element of $A\inter C$ by \obv{subseteq}.
                Fix $y\in D$.
                $(x,y)\in W$ by \obv{uniform_ball_iff}.
                Hence $(x,y)\in U$ and $(x,y)\in V$ by \obv{inter}.
                Hence $y\in A$ and $y\in C$ by
                    \auto{uniform_ball_iff}.
                Follows by \obv{inter_intro}.
            \end{subproof}
            Follows by \auto.
        \end{subproof}
        Follows by \obv{filter_basis}.
    \end{subproof}

    Show that for every $A\in B$ we have $A$ is inhabited.
    \begin{subproof}
        Fix $A\in B$.
        Take $U\in\uniformity[X]$ such that
            $A=\uniformBall{X}{U}{x}$.
        $x\in\uniformBall{X}{U}{x}$ by
            \obv{uniform_ball_contains_centre}.
        Hence $x\in A$.
    \end{subproof}
    Hence $B$ is a proper filter basis on $\carrier[X]$ by
        \obv{proper_filter_basis}.
    $\generatedFilter{\carrier[X]}{B}
        =\{N\in\pow{\carrier[X]}\mid
            \text{there exists $A\in B$ such that $A\subseteq N$}\}$ by
        \obv{filter_basis_generated_filter_eq_supersets}.

    Show that every element of
        $\neighbourhoods{x}{\uniformTopology{X}}$ is an element of
        $\generatedFilter{\carrier[X]}{B}$.
    \begin{subproof}
        Fix $N\in\neighbourhoods{x}{\uniformTopology{X}}$.
        $N\in\neighbourhoods{x}{\uniformTopology{X}}$ iff
            $N\subseteq\carrier[X]$ and there exists $U\in\uniformity[X]$
            such that $\uniformBall{X}{U}{x}\subseteq N$ by
            \obv{uniform_neighbourhood_iff}.
        Hence $N\subseteq\carrier[X]$ and there exists
            $U\in\uniformity[X]$ such that
            $\uniformBall{X}{U}{x}\subseteq N$.
        Take $U\in\uniformity[X]$ such that
            $\uniformBall{X}{U}{x}\subseteq N$.
        Let $A=\uniformBall{X}{U}{x}$.
        $A\subseteq\carrier[X]$ by
            \obv{uniform_ball_subset_carrier}.
        Hence $A\in\pow{\carrier[X]}$ by \obv{powerset_intro}.
        Thus $A\in B$.
        $N\in\pow{\carrier[X]}$ by \obv{powerset_intro}.
        Follows by \auto.
    \end{subproof}
    Hence $\neighbourhoods{x}{\uniformTopology{X}}
        \subseteq\generatedFilter{\carrier[X]}{B}$ by
        \obv{subseteq}.

    Show that every element of $\generatedFilter{\carrier[X]}{B}$ is
        an element of $\neighbourhoods{x}{\uniformTopology{X}}$.
    \begin{subproof}
        Fix $N\in\generatedFilter{\carrier[X]}{B}$.
        Hence $N\in\pow{\carrier[X]}$ and there exists $A\in B$ such
            that $A\subseteq N$ by \auto.
        Hence $N\subseteq\carrier[X]$ by \obv{powerset_elim}.
        Take $A\in B$ such that $A\subseteq N$.
        Take $U\in\uniformity[X]$ such that
            $A=\uniformBall{X}{U}{x}$.
        Hence $\uniformBall{X}{U}{x}\subseteq N$.
        Follows by \auto{uniform_neighbourhood_iff}.
    \end{subproof}
    Hence $\generatedFilter{\carrier[X]}{B}
        \subseteq\neighbourhoods{x}{\uniformTopology{X}}$ by
        \obv{subseteq}.
    Hence $\neighbourhoods{x}{\uniformTopology{X}}
        =\generatedFilter{\carrier[X]}{B}$ by
        \obv{subseteq_antisymmetric}.
\end{proof}
