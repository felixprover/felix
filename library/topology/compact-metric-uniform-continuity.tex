\section{Uniform continuity on compact metric subspaces}

\import{set/finite-choice.tex}
\import{surreal/finite.tex}
\import{surreal/real-dyadic-estimates.tex}
\import{topology/compactness.tex}

The familiar finite-cover proof works for pseudometrics.  Around each point
we first choose a ball on which the image stays within half the requested
tolerance. Compactness leaves finitely many half-radius balls, and the least
of their radii is a uniform modulus.

\begin{theorem}[Heine--Cantor]\label{compact_metric_continuous_map_is_uniformly_continuous}
    Let $M,N$ be pseudometric spaces.
    Let $X,Y$ be topological spaces.
    Suppose $X=\metricTopology{M}$ and $Y=\metricTopology{N}$.
    Suppose $K$ is compact in $X$.
    Suppose $f$ is continuous from
        $\subspaceTopology{X}{K}$ to $Y$.
    Then for every $e\in\reals$ we have if
        $\surrealLt{\surrealZero}{e}$, then there exists $d\in\reals$
        such that $\surrealLt{\surrealZero}{d}$ and for all $x,y\in K$
        we have if $\surrealLt{\metricDistance[M](x,y)}{d}$, then
        $\surrealLt{\metricDistance[N](f(x),f(y))}{e}$.
\end{theorem}
\begin{proof}
    Let $S=\subspaceTopology{X}{K}$.
    $K\subseteq\carrier[X]$ and $\subspaceTopology{X}{K}$ is compact by
        \obv{compact_in}.
    Hence $K\subseteq\carrier[M]$ by
        \auto{metric_topology_carrier}.
    Hence $S$ is a topological space and $S$ is compact.
    $\carrier[\subspaceTopology{X}{K}]=K$ by
        \obv{subspace_topology_carrier_of_subset}.
    Hence $\carrier[S]=K$.
    $f\in\funs{K}{\carrier[N]}$ by
        \auto{continuous_map,metric_topology_carrier}.
    $f$ is continuous from $S$ to $\metricTopology{N}$.
    Hence $f$ is a function and $\dom{f}=K$ by
        \obv{funs_is_function,funs}.

    Show that for every $e\in\reals$ we have if
        $\surrealLt{\surrealZero}{e}$, then there exists $d\in\reals$
        such that $\surrealLt{\surrealZero}{d}$ and for all $x,y\in K$
        we have if $\surrealLt{\metricDistance[M](x,y)}{d}$, then
        $\surrealLt{\metricDistance[N](f(x),f(y))}{e}$.
    \begin{subproof}
        Fix $e\in\reals$.
        Assume $\surrealLt{\surrealZero}{e}$.
        \begin{byCase}
        \caseOf{$K=\emptyset$.}
            $\surrealOne\in\reals$ and
                $\surrealLt{\surrealZero}{\surrealOne}$ by
                \obv{surreal_one_is_real,surreal_zero_lt_one}.
            For all $x,y\in K$ we have if
                $\surrealLt{\metricDistance[M](x,y)}{\surrealOne}$,
                then $\surrealLt{\metricDistance[N](f(x),f(y))}{e}$
                by \auto{emptyset}.
            Follows by \auto.
        \caseOf{$K\neq\emptyset$.}
            $K$ is inhabited by \obv{nonempty_inhabited}.
            Take $h\in\reals$ such that
                $\surrealLt{\surrealZero}{h}$ and
                $\surrealLt{\surrealAdd{h}{h}}{e}$ by
                \obv{positive_real_has_twofold_dyadic_budget}.

            Let $C=\{W\in\opens[S]\mid
                \text{there exists $a\in K$ such that there exists
                    $d\in\reals$ such that
                    $\surrealLt{\surrealZero}{d}$ and
                    $W=\metricBall{M}{a}{d}\inter K$ and
                    for every $z\in K$ we have if
                        $\surrealLt{\metricDistance[M](a,z)}
                            {\surrealAdd{d}{d}}$, then
                        $\surrealLt{\metricDistance[N](f(a),f(z))}{h}$}
                \}$.
            $C\subseteq\opens[S]$ by \obv{subseteq}.
            Show that $C$ covers $\carrier[S]$.
            \begin{subproof}
                Show that for every $a\in\carrier[S]$ there exists
                    $W\in C$ such that $a\in W$.
                \begin{subproof}
                    Fix $a\in\carrier[S]$.
                    Hence $a\in K$.
                    Hence $a\in\carrier[M]$ by \obv{subseteq}.
                    Take $U\in\opens[S]$ such that $a\in U$ and
                        for every $z\in U$ we have
                            $f(z)\in\metricBall{N}{f(a)}{h}$ by
                        \auto{metric_continuous_iff_local_balls}.
                    Take $V\in\opens[X]$ such that
                        $U=V\inter\carrier[S]$ by
                        \auto{subspace_open_iff}.
                    Hence $a\in V$ by \auto{inter}.
                    $\opens[X]=\metricOpens{M}$ by
                        \auto{metric_topology_opens}.
                    Hence $V\in\metricOpens{M}$.
                    Thus $\metricOpen{M}{V}$ by
                        \obv{metric_opens_iff}.
                    Take $r\in\reals$ such that
                        $\surrealLt{\surrealZero}{r}$ and
                        $\metricBall{M}{a}{r}\subseteq V$ by
                        \obv{metric_open}.
                    Let $d=\metricHalfRadius{r}$.
                    $\metricHalfRadius{r}\in\reals$ and
                        $\surrealLt{\surrealZero}{\metricHalfRadius{r}}$
                        by \obv{metric_half_radius_positive_real}.
                    Hence $d\in\reals$ and
                        $\surrealLt{\surrealZero}{d}$.
                    $\surrealAdd{\metricHalfRadius{r}}
                        {\metricHalfRadius{r}}=r$ by
                        \obv{metric_half_radius_double}.
                    Hence $\surrealAdd{d}{d}=r$.
                    Let $W=\metricBall{M}{a}{d}\inter K$.
                    $\metricBall{M}{a}{d}\in\metricOpens{M}$ by
                        \obv{metric_ball_is_open}.
                    Hence $\metricBall{M}{a}{d}\in\opens[X]$.
                    $W=\metricBall{M}{a}{d}\inter\carrier[S]$.
                    Hence $W\in\opens[S]$ by
                        \auto{subspace_open_iff}.
                    Show that for every $z\in K$ we have if
                        $\surrealLt{\metricDistance[M](a,z)}
                            {\surrealAdd{d}{d}}$, then
                        $\surrealLt{\metricDistance[N](f(a),f(z))}{h}$.
                    \begin{subproof}
                        Fix $z\in K$.
                        Assume $\surrealLt{\metricDistance[M](a,z)}
                            {\surrealAdd{d}{d}}$.
                        Hence $\surrealLt{\metricDistance[M](a,z)}{r}$.
                        Thus $z\in\metricBall{M}{a}{r}$ by
                            \obv{metric_ball,subseteq}.
                        Hence $z\in V$ by \obv{subseteq}.
                        Thus $z\in U$ by \auto{inter}.
                        Hence $f(z)\in\metricBall{N}{f(a)}{h}$.
                        $f(z)\in\carrier[N]$ by
                            \obv{funs_type_apply}.
                        Follows by \auto{metric_ball_iff}.
                    \end{subproof}
                    There exists $b\in K$ such that there exists
                        $c\in\reals$ such that
                        $\surrealLt{\surrealZero}{c}$ and
                        $W=\metricBall{M}{b}{c}\inter K$ and
                        for every $z\in K$ we have if
                            $\surrealLt{\metricDistance[M](b,z)}
                                {\surrealAdd{c}{c}}$, then
                            $\surrealLt{\metricDistance[N](f(b),f(z))}{h}$
                        by \auto.
                    Hence $W\in C$ by \auto.
                    $a\in\metricBall{M}{a}{d}$ by
                        \obv{metric_ball_contains_centre}.
                    Hence $a\in W$ by \obv{inter}.
                \end{subproof}
            \end{subproof}
            Hence $C$ is an open cover of $S$ by \obv{open_cover}.
            Take $D$ such that $D$ is finite and $D\subseteq C$ and
                $D$ covers $\carrier[S]$ by \auto{is_compact}.
            Take $x_0$ such that $x_0\in K$.
            Hence $x_0\in\carrier[S]$.
            Take $W_0\in D$ such that $x_0\in W_0$.
            Hence $D$ is inhabited by \auto.

            Let $R=\{p\in D\times\reals\mid
                \text{$\surrealLt{\surrealZero}{\snd{p}}$ and there exists
                    $a\in K$ such that
                    $\fst{p}=\metricBall{M}{a}{\snd{p}}\inter K$ and
                    for every $z\in K$ we have if
                        $\surrealLt{\metricDistance[M](a,z)}
                            {\surrealAdd{\snd{p}}{\snd{p}}}$, then
                        $\surrealLt{\metricDistance[N](f(a),f(z))}{h}$}
                \}$.
            $R$ is a relation by \auto{relation,times_elem_is_tuple}.
            Show that for every $W\in D$ there exists $d$ such that
                $W\mathrel{R}d$.
            \begin{subproof}
                Fix $W\in D$.
                Hence $W\in C$ by \obv{subseteq}.
                Take $a\in K$ such that there exists $d\in\reals$ such that
                    $\surrealLt{\surrealZero}{d}$ and
                    $W=\metricBall{M}{a}{d}\inter K$ and
                    for every $z\in K$ we have if
                        $\surrealLt{\metricDistance[M](a,z)}
                            {\surrealAdd{d}{d}}$, then
                        $\surrealLt{\metricDistance[N](f(a),f(z))}{h}$.
                Take $d\in\reals$ such that
                    $\surrealLt{\surrealZero}{d}$ and
                    $W=\metricBall{M}{a}{d}\inter K$ and
                    for every $z\in K$ we have if
                        $\surrealLt{\metricDistance[M](a,z)}
                            {\surrealAdd{d}{d}}$, then
                        $\surrealLt{\metricDistance[N](f(a),f(z))}{h}$.
                $(W,d)\in D\times\reals$ by \obv{times_tuple_intro}.
                $\fst{(W,d)}=W$ and $\snd{(W,d)}=d$ by
                    \obv{fst_eq,snd_eq}.
                Hence $(W,d)\in R$ by \auto.
            \end{subproof}
            Take $g$ such that $g$ is a function on $D$ and
                for every $W\in D$ we have $W\mathrel{R}g(W)$ by
                \obv{finite_choice}.
            Hence $g$ is a function and $\dom{g}=D$.
            Let $E=\ran{g}$.
            $\img{g}{D}$ is finite by
                \obv{finite_function_image,subseteq_refl}.
            $\img{g}{\dom{g}}=\ran{g}$ by \obv{img_dom}.
            Hence $E$ is finite.
            $g(W_0)\in\ran{g}$ by \auto{fun_ran_iff}.
            Hence $E$ is inhabited by \auto.
            Show that for every $d\in E$ we have
                $d\in\reals$ and $\surrealLt{\surrealZero}{d}$.
            \begin{subproof}
                Fix $d\in E$.
                Take $W\in\dom{g}$ such that $g(W)=d$ by
                    \obv{fun_ran_iff}.
                Hence $W\in D$.
                $(W,g(W))\in R$.
                Hence $(W,g(W))\in D\times\reals$ and
                    $\surrealLt{\surrealZero}{g(W)}$ by \auto.
                Hence $g(W)\in\reals$ by \obv{times_tuple_elim}.
            \end{subproof}
            Hence for every $d\in E$ we have $\surreal{d}$ by
                \obv{surreal_real_is_surreal}.
            Take $d\in E$ such that $\surreal{d}$ and for every
                $c\in E$ we have $\surrealLe{d}{c}$ by
                \obv{finite_surreal_minimum}.
            Hence $d\in\reals$ and $\surrealLt{\surrealZero}{d}$.

            Show that for all $x,y\in K$ we have if
                $\surrealLt{\metricDistance[M](x,y)}{d}$, then
                $\surrealLt{\metricDistance[N](f(x),f(y))}{e}$.
            \begin{subproof}
                Fix $x,y\in K$.
                Assume $\surrealLt{\metricDistance[M](x,y)}{d}$.
                Hence $x\in\carrier[S]$.
                Take $W\in D$ such that $x\in W$.
                $(W,g(W))\in R$.
                Hence $(W,g(W))\in D\times\reals$ and
                    $\surrealLt{\surrealZero}{g(W)}$ and there exists
                    $a\in K$ such that
                    $W=\metricBall{M}{a}{g(W)}\inter K$ and
                    for every $z\in K$ we have if
                        $\surrealLt{\metricDistance[M](a,z)}
                            {\surrealAdd{g(W)}{g(W)}}$, then
                        $\surrealLt{\metricDistance[N](f(a),f(z))}{h}$
                    by \auto.
                Take $a\in K$ such that
                    $W=\metricBall{M}{a}{g(W)}\inter K$ and
                    for every $z\in K$ we have if
                        $\surrealLt{\metricDistance[M](a,z)}
                            {\surrealAdd{g(W)}{g(W)}}$, then
                        $\surrealLt{\metricDistance[N](f(a),f(z))}{h}$.
                Hence $g(W)\in\reals$ by \obv{times_tuple_elim}.
                $a,x,y\in\carrier[M]$ by \obv{subseteq}.
                $\metricDistance[M](x,y)\in\reals$ by
                    \obv{pseudometric_distance_type}.
                Hence $\surreal{\metricDistance[M](x,y)}$ by
                    \obv{surreal_real_is_surreal}.
                $\surreal{d}$ and $\surreal{g(W)}$ by
                    \obv{surreal_real_is_surreal}.
                $g(W)\in E$ by \auto{fun_ran_iff}.
                Hence $\surrealLe{d}{g(W)}$.
                $x\in\metricBall{M}{a}{g(W)}$ by \auto{inter}.
                Hence $\surrealLt{\metricDistance[M](a,x)}{g(W)}$ by
                    \auto{metric_ball_iff}.
                $\surrealLt{\metricDistance[M](x,y)}{g(W)}$ by
                    \obv{surreal_lt_le_transitive}.
                Hence $\surrealLt{
                    \surrealAdd{\metricDistance[M](a,x)}
                        {\metricDistance[M](x,y)}}
                    {\surrealAdd{g(W)}{g(W)}}$ by
                    \obv{surreal_add_lt_both,
                        pseudometric_distance_type,surreal_real_is_surreal}.
                $\surrealLe{\metricDistance[M](a,y)}{
                    \surrealAdd{\metricDistance[M](a,x)}
                        {\metricDistance[M](x,y)}}$ by
                    \obv{pseudometric_triangle}.
                $\metricDistance[M](a,x)\in\reals$ and
                    $\metricDistance[M](a,y)\in\reals$ by
                    \obv{pseudometric_distance_type}.
                $\surrealAdd{\metricDistance[M](a,x)}
                    {\metricDistance[M](x,y)}\in\reals$ by
                    \obv{surreal_real_addition_closed}.
                Hence $\surreal{\metricDistance[M](a,y)}$ by
                    \obv{surreal_real_is_surreal}.
                $\surreal{\surrealAdd{\metricDistance[M](a,x)}
                    {\metricDistance[M](x,y)}}$ by
                    \obv{surreal_real_is_surreal}.
                $\surrealAdd{g(W)}{g(W)}\in\reals$ by
                    \obv{surreal_real_addition_closed}.
                Hence $\surreal{\surrealAdd{g(W)}{g(W)}}$ by
                    \obv{surreal_real_is_surreal}.
                Hence $\surrealLt{\metricDistance[M](a,y)}
                    {\surrealAdd{g(W)}{g(W)}}$ by
                    \obv{surreal_le_lt_transitive}.
                $\surrealLt{g(W)}{\surrealAdd{g(W)}{g(W)}}$ by
                    \auto{surreal_add_lt_right,surreal_add_zero_right,
                        empty_surreal,surreal_real_is_surreal}.
                Hence $\surrealLt{\metricDistance[M](a,x)}
                    {\surrealAdd{g(W)}{g(W)}}$ by
                    \auto{surreal_lt_transitive,
                        pseudometric_distance_type,surreal_real_is_surreal}.
                Hence
                    $\surrealLt{\metricDistance[N](f(a),f(x))}{h}$ and
                    $\surrealLt{\metricDistance[N](f(a),f(y))}{h}$.
                $f(a)\in\carrier[N]$ by \obv{funs_type_apply}.
                $f(x)\in\carrier[N]$ by \obv{funs_type_apply}.
                $f(y)\in\carrier[N]$ by \obv{funs_type_apply}.
                $\metricDistance[N](f(x),f(a))
                    =\metricDistance[N](f(a),f(x))$ by
                    \obv{pseudometric_distance_symmetric}.
                Hence $\surrealLt{\metricDistance[N](f(x),f(a))}{h}$.
                Hence $\surrealLt{
                    \surrealAdd{\metricDistance[N](f(x),f(a))}
                        {\metricDistance[N](f(a),f(y))}}
                    {\surrealAdd{h}{h}}$ by
                    \obv{surreal_add_lt_both,
                        pseudometric_distance_type,surreal_real_is_surreal}.
                $\surrealLe{\metricDistance[N](f(x),f(y))}{
                    \surrealAdd{\metricDistance[N](f(x),f(a))}
                        {\metricDistance[N](f(a),f(y))}}$ by
                    \obv{pseudometric_triangle}.
                $\metricDistance[N](f(x),f(y))\in\reals$ by
                    \obv{pseudometric_distance_type}.
                $\metricDistance[N](f(x),f(a))\in\reals$ by
                    \obv{pseudometric_distance_type}.
                $\metricDistance[N](f(a),f(y))\in\reals$ by
                    \obv{pseudometric_distance_type}.
                $\surrealAdd{\metricDistance[N](f(x),f(a))}
                    {\metricDistance[N](f(a),f(y))}\in\reals$ by
                    \obv{surreal_real_addition_closed}.
                $\surrealAdd{h}{h}\in\reals$ by
                    \obv{surreal_real_addition_closed}.
                $\surreal{\metricDistance[N](f(x),f(y))}$ by
                    \obv{surreal_real_is_surreal}.
                $\surreal{\surrealAdd{\metricDistance[N](f(x),f(a))}
                    {\metricDistance[N](f(a),f(y))}}$ by
                    \obv{surreal_real_is_surreal}.
                $\surreal{\surrealAdd{h}{h}}$ by
                    \obv{surreal_real_is_surreal}.
                $\surreal{e}$ by
                    \obv{surreal_real_is_surreal}.
                Hence $\surrealLt{\metricDistance[N](f(x),f(y))}
                    {\surrealAdd{h}{h}}$ by
                    \obv{surreal_le_lt_transitive}.
                Follows by \obv{surreal_lt_transitive}.
            \end{subproof}
            Follows by \auto.
        \end{byCase}
    \end{subproof}
\end{proof}
