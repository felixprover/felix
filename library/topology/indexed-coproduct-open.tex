\section{Open sets and injections of indexed coproducts}

\import{topology/indexed-coproduct.tex}
\import{topology/open-map.tex}
\import{topology/embedding.tex}

An open set in a coproduct is exactly a subset whose trace in every tagged
summand is open.  In particular, each summand is an open subspace of the
coproduct.

\begin{theorem}\label{indexed_coproduct_open_iff}
    Suppose $X$ is a function on $I$.
    Suppose for every $i\in I$ we have $X(i)$ is a topological space.
    Then $U\in\opens[\IndexedCoproductTopology{I}{X}]$ iff
        $U\subseteq\indexedSum{I}{
            \indexedTopologicalCarrierFamily{I}{X}}$ and
        for every $i\in I$ we have
            $\preimg{
                \indexedSumInjection{I}{
                    \indexedTopologicalCarrierFamily{I}{X}}{i}}{U}
            \in\opens[X(i)]$.
\end{theorem}
\begin{proof}
    Let $S=\indexedSum{I}{
        \indexedTopologicalCarrierFamily{I}{X}}$.
    Let $f=\indexedCoproductInjectionFamily{I}{X}$.
    For every $i\in I$ we have
        $(\indexedCoproductInjectionFamily{I}{X})(i)
            =\indexedSumInjection{I}{
                \indexedTopologicalCarrierFamily{I}{X}}{i}$ by
        \obv{indexed_coproduct_injection_family_apply}.
    $\IndexedCoproductTopology{I}{X}=\FinalTopology{S}{I}{X}{f}$ by
        \obv{indexed_coproduct_topology}.

    Show that if $U\in\opens[\IndexedCoproductTopology{I}{X}]$, then
        $U\subseteq S$ and for every $i\in I$ we have
            $\preimg{
                \indexedSumInjection{I}{
                    \indexedTopologicalCarrierFamily{I}{X}}{i}}{U}
            \in\opens[X(i)]$.
    \begin{subproof}
        Assume $U\in\opens[\IndexedCoproductTopology{I}{X}]$.
        Hence $U\in\opens[\FinalTopology{S}{I}{X}{f}]$.
        Hence $U\in\finalOpens{S}{I}{X}{f}$ by
            \obv{final_topology_opens}.
        Hence $U\in\pow{S}$ and for every $i\in I$ we have
            $\preimg{f(i)}{U}\in\opens[X(i)]$ by \obv{final_opens}.
        $U\subseteq S$ by \obv{powerset_elim}.
        For every $i\in I$ we have
            $f(i)=\indexedSumInjection{I}{
                \indexedTopologicalCarrierFamily{I}{X}}{i}$ by
            \auto.
        Follows by \auto.
    \end{subproof}

    Show that if $U\subseteq S$ and for every $i\in I$ we have
        $\preimg{
            \indexedSumInjection{I}{
                \indexedTopologicalCarrierFamily{I}{X}}{i}}{U}
        \in\opens[X(i)]$, then
        $U\in\opens[\IndexedCoproductTopology{I}{X}]$.
    \begin{subproof}
        Assume $U\subseteq S$ and for every $i\in I$ we have
            $\preimg{
                \indexedSumInjection{I}{
                    \indexedTopologicalCarrierFamily{I}{X}}{i}}{U}
            \in\opens[X(i)]$.
        $U\in\pow{S}$ by \obv{powerset_intro}.
        For every $i\in I$ we have
            $f(i)=\indexedSumInjection{I}{
                \indexedTopologicalCarrierFamily{I}{X}}{i}$ by
            \auto.
        Hence for every $i\in I$ we have
            $\preimg{f(i)}{U}\in\opens[X(i)]$ by \auto.
        Hence $U\in\finalOpens{S}{I}{X}{f}$ by \obv{final_opens}.
        Hence $U\in\opens[\FinalTopology{S}{I}{X}{f}]$ by
            \obv{final_topology_opens}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{theorem}\label{indexed_coproduct_injection_is_open_map}
    Suppose $X$ is a function on $I$.
    Suppose for every $j\in I$ we have $X(j)$ is a topological space.
    Let $i\in I$.
    Then $\indexedSumInjection{I}{
        \indexedTopologicalCarrierFamily{I}{X}}{i}$ is an open map from
        $X(i)$ to $\IndexedCoproductTopology{I}{X}$.
\end{theorem}
\begin{proof}
    Let $A=\indexedTopologicalCarrierFamily{I}{X}$.
    $\indexedTopologicalCarrierFamily{I}{X}$ is a function on $I$ by
        \obv{indexed_topological_carrier_family_type}.
    Hence $A$ is a function on $I$.
    $\IndexedCoproductTopology{I}{X}$ is a topological space by
        \obv{indexed_coproduct_topology_is_topological_space}.
    $\indexedSumInjection{I}{
        \indexedTopologicalCarrierFamily{I}{X}}{i}
        \in\inj{\carrier[X(i)]}{
            \carrier[\IndexedCoproductTopology{I}{X}]}$ by
        \obv{indexed_coproduct_injection_injective}.
    Hence $\indexedSumInjection{I}{
        \indexedTopologicalCarrierFamily{I}{X}}{i}
        \in\funs{\carrier[X(i)]}{
            \carrier[\IndexedCoproductTopology{I}{X}]}$ by \obv{inj}.
    Hence $\indexedSumInjection{I}{A}{i}
        \in\inj{\carrier[X(i)]}{
            \carrier[\IndexedCoproductTopology{I}{X}]}$.
    Hence $\indexedSumInjection{I}{A}{i}
        \in\funs{\carrier[X(i)]}{
            \carrier[\IndexedCoproductTopology{I}{X}]}$ by \obv{inj}.

    Show that for every $V\in\opens[X(i)]$ we have
        $\img{\indexedSumInjection{I}{A}{i}}{V}
            \in\opens[\IndexedCoproductTopology{I}{X}]$.
    \begin{subproof}
        Fix $V\in\opens[X(i)]$.
        $V\subseteq\carrier[X(i)]$ by \obv{opens_type}.
        $(\indexedTopologicalCarrierFamily{I}{X})(i)
            =\carrier[X(i)]$ by
            \obv{indexed_topological_carrier_family_apply}.
        Hence $A(i)=\carrier[X(i)]$.
        Hence $V\subseteq A(i)$.
        $\img{\indexedSumInjection{I}{A}{i}}{V}=\{i\}\times V$ by
            \obv{indexed_sum_injection_image}.

        Show that $\{i\}\times V\subseteq\indexedSum{I}{A}$.
        \begin{subproof}
            It suffices to show that every element of $\{i\}\times V$
                is an element of $\indexedSum{I}{A}$ by \obv{subseteq}.
            Fix $p\in\{i\}\times V$.
            Take $j,x$ such that $p=(j,x)$ by
                \auto{times_elem_is_tuple}.
            Hence $(j,x)\in\{i\}\times V$.
            Hence $j\in\{i\}$ and $x\in V$ by
                \obv{times_tuple_elim}.
            Hence $j=i$ by \obv{singleton_iff}.
            Hence $x\in A(i)$ by \obv{subseteq}.
            Thus $(i,x)\in\indexedSum{I}{A}$ by
                \auto{indexed_sum_iff}.
        \end{subproof}

        Show that for every $j\in I$ we have
            $\preimg{\indexedSumInjection{I}{A}{j}}{\{i\}\times V}
                \in\opens[X(j)]$.
        \begin{subproof}
            Fix $j\in I$.
            \begin{byCase}
            \caseOf{$j=i$.}
                Show that
                    $\preimg{\indexedSumInjection{I}{A}{j}}{
                        \{i\}\times V}=V$.
                \begin{subproof}
                    Show that every element of
                        $\preimg{\indexedSumInjection{I}{A}{j}}{
                            \{i\}\times V}$ is an element of $V$.
                    \begin{subproof}
                        Fix $x\in
                            \preimg{\indexedSumInjection{I}{A}{j}}{
                                \{i\}\times V}$.
                        Hence $(j,x)\in\{i\}\times V$ by
                            \auto{indexed_sum_injection_preimage_iff}.
                        Hence $x\in V$ by \obv{times_tuple_elim}.
                    \end{subproof}
                    Hence
                        $\preimg{\indexedSumInjection{I}{A}{j}}{
                            \{i\}\times V}\subseteq V$ by
                        \obv{subseteq}.
                    Show that every element of $V$ is an element of
                        $\preimg{\indexedSumInjection{I}{A}{j}}{
                            \{i\}\times V}$.
                    \begin{subproof}
                        Fix $x\in V$.
                        Hence $x\in A(i)$ by \obv{subseteq}.
                        Hence $x\in A(j)$.
                        $(j,x)\in\{i\}\times V$ by
                            \obv{times_tuple_intro,singleton_iff}.
                        Follows by
                            \auto{indexed_sum_injection_preimage_iff}.
                    \end{subproof}
                    Hence $V\subseteq
                        \preimg{\indexedSumInjection{I}{A}{j}}{
                            \{i\}\times V}$ by \obv{subseteq}.
                    Follows by \obv{subseteq_antisymmetric}.
                \end{subproof}
                Hence
                    $\preimg{\indexedSumInjection{I}{A}{j}}{
                        \{i\}\times V}\in\opens[X(i)]$.
            \caseOf{$j\neq i$.}
                Show that
                    $\preimg{\indexedSumInjection{I}{A}{j}}{
                        \{i\}\times V}=\emptyset$.
                \begin{subproof}
                    Show that every element of
                        $\preimg{\indexedSumInjection{I}{A}{j}}{
                            \{i\}\times V}$ is an element of $\emptyset$.
                    \begin{subproof}
                        Fix $x\in
                            \preimg{\indexedSumInjection{I}{A}{j}}{
                                \{i\}\times V}$.
                        Hence $(j,x)\in\{i\}\times V$ by
                            \auto{indexed_sum_injection_preimage_iff}.
                        Hence $j\in\{i\}$ by \obv{times_tuple_elim}.
                        Hence $j=i$ by \obv{singleton_iff}.
                        Contradiction.
                    \end{subproof}
                    Hence
                        $\preimg{\indexedSumInjection{I}{A}{j}}{
                            \{i\}\times V}\subseteq\emptyset$ by
                        \obv{subseteq}.
                    Follows by \obv{subseteq_emptyset_iff}.
                \end{subproof}
                $\emptyset\in\opens[X(j)]$ by
                    \auto{emptyset_open,is_open}.
            \end{byCase}
        \end{subproof}
        Hence $\{i\}\times V\subseteq
            \indexedSum{I}{
                \indexedTopologicalCarrierFamily{I}{X}}$.
        Hence for every $j\in I$ we have
            $\preimg{\indexedSumInjection{I}{
                \indexedTopologicalCarrierFamily{I}{X}}{j}}{
                \{i\}\times V}\in\opens[X(j)]$ by
            \auto.
        Hence $\{i\}\times V
            \in\opens[\IndexedCoproductTopology{I}{X}]$ by
            \auto{indexed_coproduct_open_iff}.
    \end{subproof}
    For every $V\in\opens[X(i)]$ we have
        $\img{\indexedSumInjection{I}{
            \indexedTopologicalCarrierFamily{I}{X}}{i}}{V}
        \in\opens[\IndexedCoproductTopology{I}{X}]$ by
        \auto.
    Follows by \obv{open_map}.
\end{proof}

\begin{theorem}\label{indexed_coproduct_injection_is_continuous_open_injection}
    Suppose $X$ is a function on $I$.
    Suppose for every $j\in I$ we have $X(j)$ is a topological space.
    Let $i\in I$.
    Then $\indexedSumInjection{I}{
        \indexedTopologicalCarrierFamily{I}{X}}{i}$ is continuous from
        $X(i)$ to $\IndexedCoproductTopology{I}{X}$ and
        $\indexedSumInjection{I}{
            \indexedTopologicalCarrierFamily{I}{X}}{i}$ is an open map
        from $X(i)$ to $\IndexedCoproductTopology{I}{X}$ and
        $\indexedSumInjection{I}{
            \indexedTopologicalCarrierFamily{I}{X}}{i}
        \in\inj{\carrier[X(i)]}{
            \carrier[\IndexedCoproductTopology{I}{X}]}$.
\end{theorem}
\begin{proof}
    Follows by
        \obv{indexed_coproduct_injection_continuous,
            indexed_coproduct_injection_is_open_map,
            indexed_coproduct_injection_injective}.
\end{proof}

\begin{theorem}\label{indexed_coproduct_injection_is_open_embedding}
    Suppose $X$ is a function on $I$.
    Suppose for every $j\in I$ we have $X(j)$ is a topological space.
    Let $i\in I$.
    Then $\indexedSumInjection{I}{\indexedTopologicalCarrierFamily{I}{X}}{i}$
        is an open embedding from $X(i)$ to $\IndexedCoproductTopology{I}{X}$.
\end{theorem}
\begin{proof}
    Let $f=\indexedSumInjection{I}{\indexedTopologicalCarrierFamily{I}{X}}{i}$.
    $\IndexedCoproductTopology{I}{X}$ is a topological space by
        \obv{indexed_coproduct_topology_is_topological_space}.
    $f$ is continuous from $X(i)$ to $\IndexedCoproductTopology{I}{X}$ and
        $f$ is an open map from $X(i)$ to $\IndexedCoproductTopology{I}{X}$ and
        $f\in\inj{\carrier[X(i)]}{\carrier[\IndexedCoproductTopology{I}{X}]}$ by
        \auto{indexed_coproduct_injection_is_continuous_open_injection}.
    $f$ is injective by \auto{inj,funs,funs_is_function,injective_function}.
    Follows by \auto{continuous_injective_open_map_is_open_embedding}.
\end{proof}
