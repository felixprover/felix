\section{Continuous maps and subspaces}

\import{topology/subspace.tex}
\import{topology/continuity.tex}

\begin{theorem}\label{subspace_inclusion_continuous}
    Let $X$ be a topological space.
    Let $S=\subspaceTopology{X}{A}$.
    Then $\identity{\carrier[S]}$ is continuous from $S$ to $X$.
\end{theorem}
\begin{proof}
    $S$ is a topological space
        by \obv{subspace_topology_is_topological_space}.
    $\carrier[S]=A\inter\carrier[X]$
        by \auto{subspace_topology_carrier}.
    Hence $\carrier[S]\subseteq\carrier[X]$
        by \auto{inter_lower_right}.
    $\identity{\carrier[S]}\in
        \funs{\carrier[S]}{\carrier[S]}$
        by \obv{id_elem_funs}.
    Hence $\identity{\carrier[S]}\in
        \funs{\carrier[S]}{\carrier[X]}$
        by \obv{funs_weaken_codom}.
    Show that for every $x\in\carrier[S]$ we have
        for every $V\in\opens[X]$ we have if
            $(\identity{\carrier[S]})(x)\in V$, then there exists
            $U\in\opens[S]$ such that $x\in U$ and
            for every $y\in U$ we have
                $(\identity{\carrier[S]})(y)\in V$.
    \begin{subproof}
        Fix $x\in\carrier[S]$.
        Fix $V\in\opens[X]$.
        Assume $(\identity{\carrier[S]})(x)\in V$.
        Let $U=V\inter\carrier[S]$.
        $U\in\opens[S]$ by \auto{subspace_open_iff}.
        $(\identity{\carrier[S]})(x)=x$ by \obv{id_apply}.
        Hence $x\in V$.
        Thus $x\in U$ by \obv{inter}.
        Show that for every $y\in U$ we have
            $(\identity{\carrier[S]})(y)\in V$.
        \begin{subproof}
            Fix $y\in U$.
            Then $y\in V$ and $y\in\carrier[S]$ by \obv{inter}.
            $(\identity{\carrier[S]})(y)=y$ by \obv{id_apply}.
        \end{subproof}
    \end{subproof}
    Follows by \obv{continuous_iff_open_local}.
\end{proof}

\begin{theorem}\label{subspace_structure_map_continuous}
    Let $X$ be a topological space.
    Then $\subspaceInclusion{X}{A}$ is continuous from
        $\subspaceTopology{X}{A}$ to $X$.
\end{theorem}
\begin{proof}
    Let $S=\subspaceTopology{X}{A}$.
    $\identity{\carrier[S]}$ is continuous from $S$ to $X$ by
        \obv{subspace_inclusion_continuous}.
    $\carrier[\subspaceTopology{X}{A}]=A\inter\carrier[X]$ by
        \obv{subspace_topology_carrier}.
    Hence $\carrier[S]=A\inter\carrier[X]$.
    Hence $\identity{A\inter\carrier[X]}$ is continuous from
        $S$ to $X$.
    $\subspaceInclusion{X}{A}=\identity{A\inter\carrier[X]}$ by
        \obv{subspace_inclusion}.
\end{proof}

\begin{theorem}\label{subspace_universal_property}
    Let $X,Z$ be topological spaces.
    Let $g\in\funs{\carrier[Z]}
        {\carrier[\subspaceTopology{X}{A}]}$.
    Then $g$ is continuous from $Z$ to $\subspaceTopology{X}{A}$ iff
        $\subspaceInclusion{X}{A}\circ g$ is continuous from $Z$ to $X$.
\end{theorem}
\begin{proof}
    $\carrier[\subspaceTopology{X}{A}]=A\inter\carrier[X]$ by
        \obv{subspace_topology_carrier}.
    Hence $g\in\funs{\carrier[Z]}{A\inter\carrier[X]}$.
    $\subspaceInclusion{X}{A}\in
        \funs{A\inter\carrier[X]}{\carrier[X]}$ by
        \obv{subspace_inclusion_type}.
    Hence $g$ is continuous from $Z$ to
        $\SingleInitialTopology{A\inter\carrier[X]}{X}
            {\subspaceInclusion{X}{A}}$ iff
        $\subspaceInclusion{X}{A}\circ g$ is continuous from $Z$ to $X$ by
        \obv{single_initial_topology_universal_property}.
    $\subspaceTopology{X}{A}
        =\SingleInitialTopology{A\inter\carrier[X]}{X}
            {\subspaceInclusion{X}{A}}$ by
        \obv{subspace_topology_is_single_initial_topology}.
\end{proof}

\begin{theorem}\label{continuous_domain_restriction}
    Suppose $f$ is continuous from $X$ to $Y$.
    Let $S=\subspaceTopology{X}{A}$.
    Then $\restrl{f}{\carrier[S]}$ is continuous from $S$ to $Y$.
\end{theorem}
\begin{proof}
    Let $D=\carrier[S]$.
    Let $g=\restrl{f}{D}$.
    $X$ is a topological space and $Y$ is a topological space
        by \obv{continuous_map}.
    $f\in\funs{\carrier[X]}{\carrier[Y]}$
        by \obv{continuous_map}.
    Hence $f$ is a function and $\dom{f}=\carrier[X]$
        by \obv{funs_is_function,funs}.
    $S$ is a topological space
        by \obv{subspace_topology_is_topological_space}.
    $D=A\inter\carrier[X]$ by \auto{subspace_topology_carrier}.
    Hence $D\subseteq\carrier[X]$ by \auto{inter_lower_right}.
    $g\in\funs{D}{\carrier[Y]}$ by \auto{funs_restrl}.
    Show that for every $x\in D$ we have
        for every $V\in\opens[Y]$ we have if $g(x)\in V$, then
        there exists $U\in\opens[S]$ such that $x\in U$ and
            for every $y\in U$ we have $g(y)\in V$.
    \begin{subproof}
        Fix $x\in D$.
        Fix $V\in\opens[Y]$.
        Assume $g(x)\in V$.
        $g(x)=f(x)$ by \auto{restrl_of_function_apply,funs_is_function}.
        Hence $f(x)\in V$.
        $x\in\carrier[X]$ by \obv{subseteq}.
        Take $O\in\opens[X]$ such that $x\in O$ and
            for every $z\in O$ we have $f(z)\in V$
            by \auto{continuous_iff_open_local}.
        Let $U=O\inter D$.
        $U\in\opens[S]$ by \auto{subspace_open_iff}.
        $x\in U$ by \obv{inter}.
        Show that for every $y\in U$ we have $g(y)\in V$.
        \begin{subproof}
            Fix $y\in U$.
            Then $y\in O$ and $y\in D$ by \obv{inter}.
            $f(y)\in V$.
            $g(y)=f(y)$
                by \auto{restrl_of_function_apply,funs_is_function}.
        \end{subproof}
    \end{subproof}
    Follows by \auto{continuous_iff_open_local}.
\end{proof}

\begin{theorem}\label{continuous_domain_restriction_on_subset}
    Let $X,Y$ be topological spaces.
    Suppose $A\subseteq\carrier[X]$.
    Suppose $f$ is continuous from $X$ to $Y$.
    Then $\restrl{f}{A}$ is continuous from
        $\subspaceTopology{X}{A}$ to $Y$.
\end{theorem}
\begin{proof}
    Let $S=\subspaceTopology{X}{A}$.
    $\carrier[S]=A$ by \auto{subspace_topology_carrier_of_subset}.
    $\restrl{f}{\carrier[S]}$ is continuous from $S$ to $Y$
        by \obv{continuous_domain_restriction}.
\end{proof}

\begin{theorem}\label{continuous_codomain_corestriction}
    Suppose $f$ is continuous from $X$ to $Y$.
    Let $S=\subspaceTopology{Y}{B}$.
    Suppose $\ran{f}\subseteq\carrier[S]$.
    Then $f$ is continuous from $X$ to $S$.
\end{theorem}
\begin{proof}
    Let $D=\carrier[S]$.
    $X$ is a topological space and $Y$ is a topological space
        by \obv{continuous_map}.
    $f\in\funs{\carrier[X]}{\carrier[Y]}$
        by \obv{continuous_map}.
    $S$ is a topological space
        by \obv{subspace_topology_is_topological_space}.
    $f$ is a function and $\dom{f}=\carrier[X]$
        by \obv{funs_is_function,funs}.
    Show that for every $x\in\dom{f}$ we have $f(x)\in D$.
    \begin{subproof}
        Fix $x\in\dom{f}$.
        $(x,f(x))\in f$ by \obv{function_apply_elim}.
        Hence $f(x)\in\ran{f}$ by \auto{ran_iff}.
        Follows by \obv{subseteq}.
    \end{subproof}
    Hence $f$ is a function to $D$.
    Thus $f\in\funs{\carrier[X]}{D}$ by \obv{funs_intro}.
    Show that for every $x\in\carrier[X]$ we have
        for every $V\in\opens[S]$ we have if $f(x)\in V$, then
        there exists $U\in\opens[X]$ such that $x\in U$ and
            for every $y\in U$ we have $f(y)\in V$.
    \begin{subproof}
        Fix $x\in\carrier[X]$.
        Fix $V\in\opens[S]$.
        Assume $f(x)\in V$.
        Take $W\in\opens[Y]$ such that $V=W\inter D$
            by \auto{subspace_open_iff}.
        Hence $f(x)\in W$ by \auto{inter}.
        Take $U\in\opens[X]$ such that $x\in U$ and
            for every $y\in U$ we have $f(y)\in W$
            by \auto{continuous_iff_open_local}.
        Show that for every $y\in U$ we have $f(y)\in V$.
        \begin{subproof}
            Fix $y\in U$.
            $U\subseteq\carrier[X]$ by \obv{opens_type,subseteq}.
            Hence $y\in\dom{f}$ by \auto{subseteq}.
            Thus $(y,f(y))\in f$ by \obv{function_apply_elim}.
            Hence $f(y)\in\ran{f}$ by \auto{ran_iff}.
            Thus $f(y)\in D$ by \obv{subseteq}.
            $f(y)\in W$.
            Follows by \obv{inter}.
        \end{subproof}
        Follows by \auto.
    \end{subproof}
    Follows by \obv{continuous_iff_open_local}.
\end{proof}

\begin{theorem}\label{continuous_subspace_codomain_inclusion}
    Let $X,Y$ be topological spaces.
    Suppose $f$ is continuous from $X$ to
        $\subspaceTopology{Y}{B}$.
    Then $f$ is continuous from $X$ to $Y$.
\end{theorem}
\begin{proof}
    Let $S=\subspaceTopology{Y}{B}$.
    $\identity{\carrier[S]}$ is continuous from $S$ to $Y$
        by \obv{subspace_inclusion_continuous}.
    Hence $\identity{\carrier[S]}\circ f$ is continuous from $X$ to $Y$
        by \obv{continuous_composition}.
    $f\in\funs{\carrier[X]}{\carrier[S]}$
        by \obv{continuous_map}.
    Thus $\identity{\carrier[S]}\circ f=f$
        by \obv{funs_identity_circ_left}.
\end{proof}

\begin{theorem}\label{continuous_to_subspace_iff}
    Let $X,Y$ be topological spaces.
    Then $f$ is continuous from $X$ to $\subspaceTopology{Y}{B}$ iff
        $f$ is continuous from $X$ to $Y$ and
        $\ran{f}\subseteq\carrier[\subspaceTopology{Y}{B}]$.
\end{theorem}
\begin{proof}
    Show that if $f$ is continuous from $X$ to
        $\subspaceTopology{Y}{B}$, then
        $f$ is continuous from $X$ to $Y$ and
        $\ran{f}\subseteq\carrier[\subspaceTopology{Y}{B}]$.
    \begin{subproof}
        Assume $f$ is continuous from $X$ to
            $\subspaceTopology{Y}{B}$.
        Hence $f$ is continuous from $X$ to $Y$ by
            \obv{continuous_subspace_codomain_inclusion}.
        $f\in\funs{\carrier[X]}
            {\carrier[\subspaceTopology{Y}{B}]}$ by
            \obv{continuous_map}.
        Hence $\ran{f}\subseteq\carrier[\subspaceTopology{Y}{B}]$ by
            \obv{funs_ran}.
    \end{subproof}
    Show that if $f$ is continuous from $X$ to $Y$ and
        $\ran{f}\subseteq\carrier[\subspaceTopology{Y}{B}]$, then
        $f$ is continuous from $X$ to $\subspaceTopology{Y}{B}$.
    \begin{subproof}
        Assume $f$ is continuous from $X$ to $Y$ and
            $\ran{f}\subseteq\carrier[\subspaceTopology{Y}{B}]$.
        Let $S=\subspaceTopology{Y}{B}$.
        Hence $\ran{f}\subseteq\carrier[S]$.
        Thus $f$ is continuous from $X$ to $S$ by
            \obv{continuous_codomain_corestriction}.
        Hence $f$ is continuous from $X$ to $\subspaceTopology{Y}{B}$.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{theorem}\label{nested_subspace_inclusion_continuous}
    Let $X$ be a topological space.
    Suppose $A\subseteq B$.
    Let $S=\subspaceTopology{X}{A}$.
    Let $T=\subspaceTopology{X}{B}$.
    Then $\identity{\carrier[S]}$ is continuous from $S$ to $T$.
\end{theorem}
\begin{proof}
    $\carrier[S]=A\inter\carrier[X]$ and
        $\carrier[T]=B\inter\carrier[X]$
        by \auto{subspace_topology_carrier}.
    Hence $\carrier[S]\subseteq\carrier[T]$
        by \auto{inter,subseteq}.
    $\identity{\carrier[S]}$ is continuous from $S$ to $X$
        by \obv{subspace_inclusion_continuous}.
    $\ran{\identity{\carrier[S]}}=\carrier[S]$
        by \obv{id_ran}.
    Hence
        $\ran{\identity{\carrier[S]}}\subseteq\carrier[T]$.
    Follows by \obv{continuous_codomain_corestriction}.
\end{proof}

\begin{theorem}\label{continuous_codomain_weaken_to_nested_subspace}
    Let $X,Y$ be topological spaces.
    Suppose $A\subseteq B$.
    Suppose $f$ is continuous from $X$ to
        $\subspaceTopology{Y}{A}$.
    Then $f$ is continuous from $X$ to
        $\subspaceTopology{Y}{B}$.
\end{theorem}
\begin{proof}
    Let $S=\subspaceTopology{Y}{A}$.
    Let $T=\subspaceTopology{Y}{B}$.
    $\identity{\carrier[S]}$ is continuous from $S$ to $T$
        by \obv{nested_subspace_inclusion_continuous}.
    Hence $\identity{\carrier[S]}\circ f$ is continuous from $X$ to $T$
        by \obv{continuous_composition}.
    $f\in\funs{\carrier[X]}{\carrier[S]}$
        by \obv{continuous_map}.
    Thus $\identity{\carrier[S]}\circ f=f$
        by \obv{funs_identity_circ_left}.
\end{proof}
