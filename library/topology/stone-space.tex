\begin{theorem}\label{stone_field_is_topological_basis}
    Let $B$ be a Boolean algebra.
    Then $\stoneField{B}$ is a topological basis for
        $\booleanUltrafilters{B}$.
\end{theorem}
\begin{proof}
    $\stoneField{B}$ is a field of sets on
        $\booleanUltrafilters{B}$ by
        \obv{stone_field_is_field_of_sets}.
    Follows by \obv{field_of_sets_is_topological_basis}.
\end{proof}

\begin{theorem}\label{stone_field_separates_points}
    Let $B$ be a Boolean algebra.
    Then $\fieldSeparatesPoints
        {\stoneField{B}}{\booleanUltrafilters{B}}$.
\end{theorem}
\begin{proof}
    Show that for all $U,V\in\booleanUltrafilters{B}$ such that
        $U\neq V$ there exists $A\in\stoneField{B}$ such that
        $(U\in A\land V\notin A)\lor
            (V\in A\land U\notin A)$.
    \begin{subproof}
        Fix $U,V\in\booleanUltrafilters{B}$.
        Assume $U\neq V$.
        Take $a$ such that
            ($a\in U$ and $a\notin V$) or
            ($a\notin U$ and $a\in V$) by \auto{neq_witness}.
        $U,V\subseteq\carrier[B]$ by
            \obv{boolean_ultrafilters_member_subseteq_carrier}.
        $a\in U$ or $a\in V$.
        Hence $a\in\carrier[B]$ by \obv{subseteq}.
        $\stoneSet{B}{a}\in\stoneField{B}$ by
            \auto{stone_field_iff}.
        \begin{byCase}
        \caseOf{$a\in U$ and $a\notin V$.}
            $U\in\stoneSet{B}{a}$ and
                $V\notin\stoneSet{B}{a}$ by
                \obv{stone_set_iff}.
        \caseOf{$a\notin U$ or $a\in V$.}
            Hence $a\notin U$ and $a\in V$.
            $V\in\stoneSet{B}{a}$ and
                $U\notin\stoneSet{B}{a}$ by
                \obv{stone_set_iff}.
        \end{byCase}
    \end{subproof}
    Follows by \obv{field_separates_points_iff}.
\end{proof}

\subsection{The canonical ultrafilter limit point}

\begin{definition}\label{stone_limit_point}
    $\stoneLimitPoint{B}{U}=\preimg{\stoneMap{B}}{U}$.
\end{definition}

\begin{proposition}\label{stone_limit_point_iff}
    Let $B$ be a Boolean algebra.
    Let $a\in\carrier[B]$.
    Then $a\in\stoneLimitPoint{B}{U}$ iff
        $\stoneSet{B}{a}\in U$.
\end{proposition}
\begin{proof}
    $\booleanHom
        {\stoneMap{B}}
        {B}
        {\powersetBooleanAlgebra{\booleanUltrafilters{B}}}$ by
        \obv{stone_map_is_boolean_hom}.
    $a\in\preimg{\stoneMap{B}}{U}$ iff
        $a\in\carrier[B]$ and
        $\apply{\stoneMap{B}}{a}\in U$ by
        \obv{boolean_hom_preimage_iff}.
    $\apply{\stoneMap{B}}{a}=\stoneSet{B}{a}$ by
        \obv{stone_map_apply}.
    Follows by \auto{stone_limit_point}.
\end{proof}

\begin{theorem}\label{stone_limit_point_is_boolean_ultrafilter}
    Let $B$ be a Boolean algebra.
    Suppose $U$ is an ultrafilter on
        $\booleanUltrafilters{B}$.
    Then $\booleanUltrafilter{\stoneLimitPoint{B}{U}}{B}$.
\end{theorem}
\begin{proof}
    $\booleanUltrafilter
        {U}
        {\powersetBooleanAlgebra{\booleanUltrafilters{B}}}$ by
        \obv{ultrafilter_iff_boolean_ultrafilter_on_powerset_boolean_algebra}.
    $\booleanHom
        {\stoneMap{B}}
        {B}
        {\powersetBooleanAlgebra{\booleanUltrafilters{B}}}$ by
        \obv{stone_map_is_boolean_hom}.
    Hence $\booleanUltrafilter
        {\preimg{\stoneMap{B}}{U}}{B}$ by
        \obv{boolean_hom_preimage_boolean_ultrafilter}.
    Follows by \obv{stone_limit_point}.
\end{proof}

\begin{proposition}\label{stone_limit_point_is_stone_space_point}
    Let $B$ be a Boolean algebra.
    Suppose $U$ is an ultrafilter on
        $\booleanUltrafilters{B}$.
    Then $\stoneLimitPoint{B}{U}\in\carrier[\stoneSpace{B}]$.
\end{proposition}
\begin{proof}
    $\booleanUltrafilter{\stoneLimitPoint{B}{U}}{B}$ by
        \obv{stone_limit_point_is_boolean_ultrafilter}.
    Hence $\stoneLimitPoint{B}{U}\in
        \booleanUltrafilters{B}$ by
        \auto{boolean_ultrafilters_iff}.
    Follows by \obv{stone_space_carrier}.
\end{proof}

\begin{theorem}\label{stone_limit_point_convergence}
    Let $B$ be a Boolean algebra.
    Suppose $U$ is an ultrafilter on
        $\booleanUltrafilters{B}$.
    Then $\filterConverges
        {U}{\stoneLimitPoint{B}{U}}{\stoneSpace{B}}$.
\end{theorem}
\begin{proof}
    Let $p=\stoneLimitPoint{B}{U}$.
    Let $T=\stoneSpace{B}$.
    $T$ is a topological space by
        \obv{stone_space_is_topological_space}.
    $\carrier[T]=\booleanUltrafilters{B}$ by
        \auto{stone_space_carrier}.
    $U$ is a proper filter on $\booleanUltrafilters{B}$ by
        \auto{ultrafilter,maximalfilter}.
    Hence $U$ is a filter on $\booleanUltrafilters{B}$ by
        \obv{proper_filter_is_filter}.
    Hence $U$ is a filter on $\carrier[T]$.
    $\stoneField{B}$ is an open basis of $T$ by
        \obv{stone_field_is_open_basis}.
    $p\in\carrier[T]$ by
        \auto{stone_limit_point_is_stone_space_point}.
    Show that for every $V\in\stoneField{B}$ such that $p\in V$
        we have $V\in U$.
    \begin{subproof}
        Fix $V\in\stoneField{B}$.
        Assume $p\in V$.
        Take $a\in\carrier[B]$ such that
            $V=\stoneSet{B}{a}$ by \obv{stone_field_iff}.
        Hence $p\in\stoneSet{B}{a}$.
        Thus $a\in p$ by \auto{stone_set_iff}.
        Hence $\stoneSet{B}{a}\in U$ by
            \obv{stone_limit_point_iff}.
    \end{subproof}
    Follows by
        \auto{filter_converges_iff_open_basis_members}.
\end{proof}

\begin{theorem}\label{stone_space_is_compact}
    Let $B$ be a Boolean algebra.
    Then $\stoneSpace{B}$ is compact.
\end{theorem}
\begin{proof}
    Let $T=\stoneSpace{B}$.
    $T$ is a topological space by
        \obv{stone_space_is_topological_space}.
    $\carrier[T]=\booleanUltrafilters{B}$ by
        \auto{stone_space_carrier}.
    Show that for every $U$ such that $U$ is an ultrafilter on
        $\carrier[T]$ there exists $p\in\carrier[T]$ such that
            $\filterConverges{U}{p}{T}$.
    \begin{subproof}
        Fix $U$.
        Assume $U$ is an ultrafilter on $\carrier[T]$.
        Hence $U$ is an ultrafilter on $\booleanUltrafilters{B}$.
        Let $p=\stoneLimitPoint{B}{U}$.
        $p\in\carrier[T]$ by
            \auto{stone_limit_point_is_stone_space_point}.
        $\filterConverges{U}{p}{T}$ by
            \auto{stone_limit_point_convergence}.
    \end{subproof}
    Follows by \obv{compact_iff_every_ultrafilter_converges}.
\end{proof}

\begin{theorem}\label{stone_space_clopens_eq_stone_field}

    Then $\clopens{\stoneSpace{B}}=\stoneField{B}$.
\end{theorem}
\begin{proof}

        $\booleanUltrafilters{B}$ by
        \obv{stone_field_is_field_of_sets}.
    $\stoneSpace{B}$ is compact by \obv{stone_space_is_compact}.
    Follows by

\end{proof}

\begin{theorem}\label{stone_space_clopen_iff_stone_set}
    Let $B$ be a Boolean algebra.
    Then $C$ is clopen in $\stoneSpace{B}$ iff
        there exists $a\in\carrier[B]$ such that
            $C=\stoneSet{B}{a}$.
\end{theorem}
\begin{proof}
    $\stoneSpace{B}$ is a topological space by
        \obv{stone_space_is_topological_space}.
    $C$ is clopen in $\stoneSpace{B}$ iff
        $C\in\clopens{\stoneSpace{B}}$ by \obv{clopens_iff}.
    $\clopens{\stoneSpace{B}}=\stoneField{B}$ by
        \obv{stone_space_clopens_eq_stone_field}.
    $C\in\stoneField{B}$ iff
        there exists $a\in\carrier[B]$ such that
            $C=\stoneSet{B}{a}$ by \obv{stone_field_iff}.
\end{proof}
