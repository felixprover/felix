\section{Local path connectedness of products}

\import{topology/local-path.tex}
\import{topology/path-product.tex}
\import{topology/product-subspace.tex}

Inside an open rectangle, choose a path-connected open neighbourhood in each
factor. Their product is again an open path-connected neighbourhood.

\begin{theorem}\label{binary_product_of_locally_path_connected_spaces}
    Let $X,Y$ be locally path-connected topological spaces.
    Then $\BinaryProductTopology{X}{Y}$ is locally path-connected.
\end{theorem}
\begin{proof}
    Let $P=\BinaryProductTopology{X}{Y}$.
    $P$ is a topological space by \obv{binary_product_topology_is_topological_space}.
    $\productRectangles{X}{Y}$ is an open basis of $P$ by
        \obv{product_rectangles_form_open_basis}.
    Show that for all $W,z$ such that $W\in\opens[P]$ and $z\in W$ there exists
        $N\in\opens[P]$ such that $z\in N\subseteq W$ and
        $\subspaceTopology{P}{N}$ is path-connected.
    \begin{subproof}
        Fix $W,z$.
        Assume $W\in\opens[P]$ and $z\in W$.
        Take $R\in\productRectangles{X}{Y}$ such that $z\in R\subseteq W$ by \obv{open_basis}.
        Take $U,V$ such that $U\in\opens[X]$ and $V\in\opens[Y]$ and $R=U\times V$ by
            \auto{product_rectangles}.
        Take $x,y$ such that $x\in U$ and $y\in V$ and $z=(x,y)$ by \auto{times}.
        Take $K\in\opens[X]$ such that $x\in K\subseteq U$ and
            $\subspaceTopology{X}{K}$ is path-connected by \obv{locally_path_connected}.
        Take $L\in\opens[Y]$ such that $y\in L\subseteq V$ and
            $\subspaceTopology{Y}{L}$ is path-connected by \obv{locally_path_connected}.
        $K\times L\in\opens[P]$ by \auto{product_of_open_sets_is_open}.
        $z\in K\times L$ by \auto{times_tuple_intro}.
        $K\times L\subseteq W$ by \auto{times,subseteq}.
        $\subspaceTopology{X}{K}$ is a topological space and
            $\subspaceTopology{Y}{L}$ is a topological space by
            \obv{subspace_topology_is_topological_space}.
        $\BinaryProductTopology{\subspaceTopology{X}{K}}{\subspaceTopology{Y}{L}}$
            is path-connected by \obv{binary_product_of_path_connected_spaces}.
        Hence $\subspaceTopology{P}{K\times L}$ is path-connected by \auto{product_of_subspaces}.
    \end{subproof}
    Follows by \auto{locally_path_connected}.
\end{proof}
