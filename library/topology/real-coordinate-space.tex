\section{Real coordinate spaces}

\import{topology/power-coordinate.tex}
\import{topology/finite-power-split.tex}
\import{topology/real-line.tex}

Real coordinates are functions on an explicit index set. For a natural number
n, the von Neumann set n is already the index set of its n coordinates. Thus
finite-dimensional spaces use this same construction, including the empty
function in dimension zero. No norm or vector-space structure is chosen here.

\begin{definition}\label{real_coordinate_space}
    $\RealCoordinateSpace{I}=\TopologicalPower{\RealTopology}{I}$.
\end{definition}

\begin{classification}\label{real_coordinate_space_is_topological_space}
    $\RealCoordinateSpace{I}$ is a topological space.
\end{classification}
\begin{proof}
    $\RealTopology$ is a topological space by \obv{real_topology_is_topological_space}.
    Follows by \auto{real_coordinate_space,topological_power_is_topological_space}.
\end{proof}

\begin{theorem}\label{real_coordinate_space_is_exact}
    $\RealCoordinateSpace{I}$ is an exact topological space.
\end{theorem}
\begin{proof}
    $\RealTopology$ is a topological space by \obv{real_topology_is_topological_space}.
    Follows by \auto{real_coordinate_space,topological_power_is_exact}.
\end{proof}

\begin{proposition}\label{real_coordinate_space_carrier}
    $\carrier[\RealCoordinateSpace{I}]=\funs{I}{\reals}$.
\end{proposition}
\begin{proof}
    $\RealTopology$ is a topological space by \obv{real_topology_is_topological_space}.
    Follows by \auto{real_coordinate_space,topological_power_carrier,real_topology_carrier}.
\end{proof}

\begin{theorem}\label{real_coordinate_evaluation_continuous}
    Let $i\in I$.
    Then $\functionEvaluation{I}{\reals}{i}$ is continuous from
        $\RealCoordinateSpace{I}$ to $\RealTopology$.
\end{theorem}
\begin{proof}
    $\RealTopology$ is a topological space by \obv{real_topology_is_topological_space}.
    Follows by \auto{real_coordinate_space,real_topology_carrier,topological_power_evaluation}.
\end{proof}

\begin{theorem}\label{real_coordinate_reindex_continuous}

\begin{theorem}\label{real_coordinate_reindex_homeomorphism}

\begin{proof}

\subsection{Deleting and inserting coordinates}

\begin{theorem}\label{real_coordinate_restriction_continuous}
    Suppose $J\subseteq I$.
    Then $\functionRestriction{I}{J}{\reals}$ is continuous from
        $\RealCoordinateSpace{I}$ to $\RealCoordinateSpace{J}$.
\end{theorem}
\begin{proof}
    $\RealTopology$ is a topological space by \obv{real_topology_is_topological_space}.
    Follows by \auto{real_coordinate_space,real_topology_carrier,
        topological_power_restriction_continuous}.
\end{proof}

\begin{theorem}\label{real_coordinate_insertion_continuous}
    Let $i\in I$ and $a\in\reals$.
    Then $\functionInsertion{I}{\reals}{i}{a}$ is continuous from
        $\RealCoordinateSpace{I\setminus\{i\}}$ to $\RealCoordinateSpace{I}$.
\end{theorem}
\begin{proof}
    $\RealTopology$ is a topological space by \obv{real_topology_is_topological_space}.
    Follows by \auto{real_coordinate_space,real_topology_carrier,
        topological_power_insertion_continuous}.
\end{proof}

\begin{theorem}\label{real_coordinate_insertion_homeomorphism}
    Let $i\in I$ and $a\in\reals$.
    Then $\functionInsertion{I}{\reals}{i}{a}$ is a homeomorphism from
        $\RealCoordinateSpace{I\setminus\{i\}}$ to
        $\subspaceTopology{\RealCoordinateSpace{I}}{
            \fibre{\functionEvaluation{I}{\reals}{i}}{a}}$.
\end{theorem}
\begin{proof}
    $\RealTopology$ is a topological space by \obv{real_topology_is_topological_space}.
    Follows by \auto{real_coordinate_space,real_topology_carrier,
        topological_power_insertion_homeomorphism}.
\end{proof}

\begin{theorem}\label{real_zero_coordinate_insertion}
    $\functionInsertion{\{i\}}{\reals}{i}{\surrealZero}$ is a homeomorphism from
        $\RealCoordinateSpace{\emptyset}$ to
        $\subspaceTopology{\RealCoordinateSpace{\{i\}}}{\{\{i\}\times\{\surrealZero\}\}}$.
\end{theorem}
\begin{proof}
    $i\in\{i\}$ by \obv{singleton_intro}.
    $\surrealZero\in\reals$ by \obv{surreal_zero_is_real}.
    $\{i\}\setminus\{i\}=\emptyset$ by \obv{setminus_self}.
    $\fibre{\functionEvaluation{\{i\}}{\reals}{i}}{\surrealZero}
        =\{\{i\}\times\{\surrealZero\}\}$ by \obv{function_evaluation_singleton_fibre}.
    Follows by \auto{real_coordinate_insertion_homeomorphism}.
\end{proof}

\subsection{Splitting finite coordinate blocks}

\begin{theorem}\label{real_finite_coordinate_split_homeomorphism}
    Let $m,n$ be natural numbers.
    Then $\finiteFunctionSplit{m}{n}{\reals}$ is a homeomorphism from
        $\RealCoordinateSpace{m+n}$ to
        $\BinaryProductTopology{\RealCoordinateSpace{m}}{\RealCoordinateSpace{n}}$.
\end{theorem}
\begin{proof}
    $\RealTopology$ is a topological space by \obv{real_topology_is_topological_space}.
    Follows by \auto{real_coordinate_space,real_topology_carrier,
        finite_power_split_homeomorphism}.
\end{proof}

\begin{theorem}\label{real_finite_coordinate_join_homeomorphism}
    Let $m,n$ be natural numbers.
    Then $\finiteFunctionJoin{m}{n}{\reals}$ is a homeomorphism from
        $\BinaryProductTopology{\RealCoordinateSpace{m}}{\RealCoordinateSpace{n}}$ to
        $\RealCoordinateSpace{m+n}$.
\end{theorem}
\begin{proof}
    $\RealTopology$ is a topological space by \obv{real_topology_is_topological_space}.
    Follows by \auto{real_coordinate_space,real_topology_carrier,
        finite_power_join_homeomorphism}.
\end{proof}

\begin{corollary}\label{real_finite_coordinate_left_unit}
    Let $n$ be a natural number.
    Then $\finiteFunctionJoin{\emptyset}{n}{\reals}$ is a homeomorphism from
        $\BinaryProductTopology{\RealCoordinateSpace{\emptyset}}{\RealCoordinateSpace{n}}$ to
        $\RealCoordinateSpace{n}$.
\end{corollary}
\begin{proof}
    $\emptyset$ is a natural number by \obv{omega_empty}.
    $\emptyset+n=n$ by \obv{num_addition_zero_left}.
    Follows by \auto{real_finite_coordinate_join_homeomorphism}.
\end{proof}

\begin{corollary}\label{real_finite_coordinate_right_unit}
    Let $n$ be a natural number.
    Then $\finiteFunctionJoin{n}{\emptyset}{\reals}$ is a homeomorphism from
        $\BinaryProductTopology{\RealCoordinateSpace{n}}{\RealCoordinateSpace{\emptyset}}$ to
        $\RealCoordinateSpace{n}$.
\end{corollary}
\begin{proof}
    $\emptyset$ is a natural number by \obv{omega_empty}.
    $n+\emptyset=n$ by \obv{num_addition_zero_right}.
    Follows by \auto{real_finite_coordinate_join_homeomorphism}.
\end{proof}
