Product charts first land in a product of Euclidean models. The explicit
finite-coordinate join homeomorphism then identifies that product with the
model of dimension $m+n$, including zero-dimensional factors.

\begin{theorem}\label{product_of_locally_euclidean_spaces}
    Let $m,n$ be natural numbers and $X,Y$ be topological spaces.
    Suppose $X$ is locally Euclidean of dimension $m$ and
        $Y$ is locally Euclidean of dimension $n$.
    Then $\BinaryProductTopology{X}{Y}$ is locally Euclidean of dimension $m+n$.
\end{theorem}
\begin{proof}
    Let $P=\BinaryProductTopology{X}{Y}$.
    Let $E=\RealCoordinateSpace{m}$ and $F=\RealCoordinateSpace{n}$.
    Let $M=\BinaryProductTopology{E}{F}$ and $R=\RealCoordinateSpace{m+n}$.
    Let $j=\finiteFunctionJoin{m}{n}{\reals}$.
    $P$ is a topological space and $M$ is a topological space by
        \obv{binary_product_topology_is_topological_space}.
    $E$ is a topological space and $F$ is a topological space and $R$ is a topological space by
        \obv{real_coordinate_space_is_topological_space}.
    $j$ is a homeomorphism from $M$ to $R$ by \auto{real_finite_coordinate_join_homeomorphism}.
    $j$ is an open map from $M$ to $R$ by \obv{homeomorphism_is_open_map}.
    Show that for every $z\in\carrier[P]$ there exists $W\in\opens[P]$ such that
        $z\in W$ and there exists $C\in\opens[R]$ such that there exists $k$ such that
        $k$ is a homeomorphism from $\subspaceTopology{P}{W}$ to $\subspaceTopology{R}{C}$.
    \begin{subproof}
        Fix $z\in\carrier[P]$.
        Take $x,y$ such that $x\in\carrier[X]$ and $y\in\carrier[Y]$ and $z=(x,y)$ by
            \auto{binary_product_topology_carrier,times}.
        Take $U,A,f$ such that $U\in\opens[X]$ and $x\in U$ and $A\in\opens[E]$ and
            $f$ is a homeomorphism from $\subspaceTopology{X}{U}$ to $\subspaceTopology{E}{A}$ by
            \auto{locally_euclidean}.
        Take $V,B,g$ such that $V\in\opens[Y]$ and $y\in V$ and $B\in\opens[F]$ and
            $g$ is a homeomorphism from $\subspaceTopology{Y}{V}$ to $\subspaceTopology{F}{B}$ by
            \auto{locally_euclidean}.
        $U\times V\in\opens[P]$ and $A\times B\in\opens[M]$ by
            \auto{product_of_open_sets_is_open}.
        $z\in U\times V$ by \auto{times_tuple_intro}.
        Let $h=\productMap{\carrier[\subspaceTopology{X}{U}]}{\carrier[\subspaceTopology{Y}{V}]}{f}{g}$.
        $h$ is a homeomorphism from
            $\BinaryProductTopology{\subspaceTopology{X}{U}}{\subspaceTopology{Y}{V}}$ to
            $\BinaryProductTopology{\subspaceTopology{E}{A}}{\subspaceTopology{F}{B}}$ by
            \auto{product_map_homeomorphism}.
        Hence $h$ is a homeomorphism from $\subspaceTopology{P}{U\times V}$ to
            $\subspaceTopology{M}{A\times B}$ by \auto{product_of_subspaces}.
        $\img{j}{A\times B}\in\opens[R]$ by \obv{open_map}.
        $A\times B\subseteq\carrier[M]$ by \auto{opens_type,subseteq}.
        $\restrl{j}{A\times B}$ is a homeomorphism from $\subspaceTopology{M}{A\times B}$ to
            $\subspaceTopology{R}{\img{j}{A\times B}}$ by \obv{homeomorphism_restriction_to_image}.
        Hence $\restrl{j}{A\times B}\circ h$ is a homeomorphism from
            $\subspaceTopology{P}{U\times V}$ to $\subspaceTopology{R}{\img{j}{A\times B}}$ by
            \obv{homeomorphism_composition}.
        Follows by \auto{homeomorphism}.
    \end{subproof}
    Follows by \auto{locally_euclidean,num_addition_on_naturals}.
\end{proof}

\begin{theorem}\label{product_of_topological_manifolds}
    Let $m,n$ be natural numbers and $X,Y$ be topological spaces.
    Suppose $X$ is a topological manifold of dimension $m$ and
        $Y$ is a topological manifold of dimension $n$.
    Then $\BinaryProductTopology{X}{Y}$ is a topological manifold of dimension $m+n$.
\end{theorem}
\begin{proof}
    $X$ is locally Euclidean of dimension $m$ and $Y$ is locally Euclidean of dimension $n$ by
        \obv{topological_manifold}.
    $\BinaryProductTopology{X}{Y}$ is locally Euclidean of dimension $m+n$ by
        \obv{product_of_locally_euclidean_spaces}.
    $X$ is Hausdorff and $Y$ is Hausdorff by \obv{topological_manifold}.
    Hence $\BinaryProductTopology{X}{Y}$ is Hausdorff by \obv{binary_product_of_hausdorff_spaces}.
    $X$ is second-countable and $Y$ is second-countable by \obv{topological_manifold}.
    Hence $\BinaryProductTopology{X}{Y}$ is second-countable by
        \obv{binary_product_of_second_countable_spaces}.
    Follows by \auto{topological_manifold,num_addition_on_naturals,
        binary_product_topology_is_topological_space}.
\end{proof}
