\section{The topology induced by one final map}

\import{topology/final.tex}

For one map, the final topology is the direct topology of subsets with open
inverse image.  This presentation exposes the computation rule without
packaging the map into a singleton-indexed family.

\begin{definition}\label{single_final_topology}
    $\SingleFinalTopology{T}{X}{f}
        =\TopologicalSpaceOn{T}{\preimageOpenFamily{X}{T}{f}}$.
\end{definition}

\begin{theorem}\label{single_final_topology_is_topological_space}
    Let $X$ be a topological space.
    Let $f\in\funs{\carrier[X]}{T}$.
    Then $\SingleFinalTopology{T}{X}{f}$ is a topological space.
\end{theorem}
\begin{proof}
    $\preimageOpenFamily{X}{T}{f}$ is a topology on $T$ by
        \obv{preimage_open_family_is_topology}.
    Follows by
        \auto{single_final_topology,
            topological_space_on_is_topological_space}.
\end{proof}

\begin{theorem}\label{single_final_topology_is_exact}
    Let $X$ be a topological space.
    Let $f\in\funs{\carrier[X]}{T}$.
    Then $\SingleFinalTopology{T}{X}{f}$ is an exact topological space.
\end{theorem}
\begin{proof}
    $\preimageOpenFamily{X}{T}{f}$ is a topology on $T$ by
        \obv{preimage_open_family_is_topology}.
    Follows by
        \auto{single_final_topology,topological_space_on_is_exact}.
\end{proof}

\begin{proposition}\label{single_final_topology_carrier}
    $\carrier[\SingleFinalTopology{T}{X}{f}]=T$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{single_final_topology,topological_space_on_carrier}.
\end{proof}

\begin{proposition}\label{single_final_topology_opens}
    $\opens[\SingleFinalTopology{T}{X}{f}]
        =\preimageOpenFamily{X}{T}{f}$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{single_final_topology,topological_space_on_opens}.
\end{proof}

\begin{proposition}\label{single_final_topology_open_iff}
    $U\in\opens[\SingleFinalTopology{T}{X}{f}]$ iff
        $U\subseteq T$ and $\preimg{f}{U}\in\opens[X]$.
\end{proposition}
\begin{proof}
    $U\in\opens[\SingleFinalTopology{T}{X}{f}]$ iff
        $U\in\preimageOpenFamily{X}{T}{f}$ by
        \obv{single_final_topology_opens}.
    $U\in\preimageOpenFamily{X}{T}{f}$ iff
        $U\in\pow{T}$ and $\preimg{f}{U}\in\opens[X]$ by
        \obv{preimage_open_family}.
    $U\in\pow{T}$ iff $U\subseteq T$ by \obv{pow_iff}.
    Follows by \auto.
\end{proof}

\begin{theorem}\label{single_final_topology_structure_map_continuous}
    Let $X$ be a topological space.
    Let $f\in\funs{\carrier[X]}{T}$.
    Then $f$ is continuous from $X$ to
        $\SingleFinalTopology{T}{X}{f}$.
\end{theorem}
\begin{proof}
    $\SingleFinalTopology{T}{X}{f}$ is a topological space by
        \obv{single_final_topology_is_topological_space}.
    $\carrier[\SingleFinalTopology{T}{X}{f}]=T$ by
        \obv{single_final_topology_carrier}.
    Hence $f\in\funs{\carrier[X]}{
        \carrier[\SingleFinalTopology{T}{X}{f}]}$.
    Show that for every
        $U\in\opens[\SingleFinalTopology{T}{X}{f}]$ we have
        $\preimg{f}{U}\in\opens[X]$.
    \begin{subproof}
        Fix $U\in\opens[\SingleFinalTopology{T}{X}{f}]$.
        Follows by \auto{single_final_topology_open_iff}.
    \end{subproof}
    Follows by \obv{continuous_map}.
\end{proof}

\begin{theorem}\label{single_final_topology_universal_property}
    Let $X,Y$ be topological spaces.
    Let $f\in\funs{\carrier[X]}{T}$.
    Let $g\in\funs{T}{\carrier[Y]}$.
    Then $g$ is continuous from $\SingleFinalTopology{T}{X}{f}$ to $Y$
        iff $g\circ f$ is continuous from $X$ to $Y$.
\end{theorem}
\begin{proof}
    Show that if $g$ is continuous from
        $\SingleFinalTopology{T}{X}{f}$ to $Y$, then
        $g\circ f$ is continuous from $X$ to $Y$.
    \begin{subproof}
        Assume $g$ is continuous from $\SingleFinalTopology{T}{X}{f}$
            to $Y$.
        $f$ is continuous from $X$ to
            $\SingleFinalTopology{T}{X}{f}$ by
            \obv{single_final_topology_structure_map_continuous}.
        Follows by \obv{continuous_composition}.
    \end{subproof}

    Show that if $g\circ f$ is continuous from $X$ to $Y$, then
        $g$ is continuous from $\SingleFinalTopology{T}{X}{f}$ to $Y$.
    \begin{subproof}
        Assume $g\circ f$ is continuous from $X$ to $Y$.
        $\SingleFinalTopology{T}{X}{f}$ is a topological space by
            \obv{single_final_topology_is_topological_space}.
        $\carrier[\SingleFinalTopology{T}{X}{f}]=T$ by
            \obv{single_final_topology_carrier}.
        Hence $g\in\funs{
            \carrier[\SingleFinalTopology{T}{X}{f}]}{\carrier[Y]}$.
        Show that for every $V\in\opens[Y]$ we have
            $\preimg{g}{V}\in
                \opens[\SingleFinalTopology{T}{X}{f}]$.
        \begin{subproof}
            Fix $V\in\opens[Y]$.
            $\preimg{g\circ f}{V}\in\opens[X]$ by
                \obv{continuous_map}.
            $\preimg{g\circ f}{V}
                =\preimg{f}{\preimg{g}{V}}$ by
                \obv{preimage_composition}.
            Hence $\preimg{f}{\preimg{g}{V}}\in\opens[X]$.
            $\preimg{g}{V}\subseteq T$ by
                \auto{preimg_subseteq_dom,funs,subseteq}.
            Follows by \auto{single_final_topology_open_iff}.
        \end{subproof}
        Follows by \obv{continuous_map}.
    \end{subproof}
    Follows by \auto.
\end{proof}
