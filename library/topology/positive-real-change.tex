\section{Locally increasing real coordinate changes}

\import{topology/increasing-real-map.tex}

A positive real coordinate change is an open partial homeomorphism that is
increasing near each domain point. Distinct components need not retain their
global order.

\begin{definition}\label{positive_real_change}
    $h$ is a positive real coordinate change iff
        $h$ is an open partial homeomorphism from $\RealTopology$ to $\RealTopology$ and
        for every $x\in\dom{h}$ there exists $U\in\opens[\RealTopology]$ such that
            $x\in U$ and $\restrl{h}{U}$ is an increasing real map.
\end{definition}

\begin{theorem}\label{increasing_real_homeomorphism_is_positive}
    Suppose $h$ is an open partial homeomorphism from $\RealTopology$ to $\RealTopology$.
    Suppose $h$ is an increasing real map.
    Then $h$ is a positive real coordinate change.
\end{theorem}
\begin{proof}
    $\dom{h}\in\opens[\RealTopology]$ by \obv{open_partial_homeomorphism}.
    $\restrl{h}{\dom{h}}=h$ by \obv{restrl_by_dom}.
    For every $x\in\dom{h}$ there exists $U\in\opens[\RealTopology]$ such that
        $x\in U$ and $\restrl{h}{U}$ is an increasing real map by \auto.
    Follows by \obv{positive_real_change}.
\end{proof}

\begin{theorem}\label{open_identity_positive_real_change}
    Let $U\in\opens[\RealTopology]$.
    Then $\identity{U}$ is a positive real coordinate change.
\end{theorem}
\begin{proof}
    $\RealTopology$ is a topological space by \obv{real_topology_is_topological_space}.
    $\carrier[\RealTopology]=\reals$ by \obv{real_topology_carrier}.
    $U\subseteq\reals$ by \obv{opens_type}.
    $\identity{U}$ is an increasing real map by \obv{identity_increasing_real_map}.
    $\identity{U}$ is an open partial homeomorphism from $\RealTopology$ to $\RealTopology$
        by \obv{identity_open_partial_homeomorphism}.
    Follows by \obv{increasing_real_homeomorphism_is_positive}.
\end{proof}

\begin{theorem}\label{positive_real_change_restriction}
    Suppose $h$ is a positive real coordinate change.
    Let $U\in\opens[\RealTopology]$.
    Then $\restrl{h}{U}$ is a positive real coordinate change.
\end{theorem}
\begin{proof}
    $h$ is an open partial homeomorphism from $\RealTopology$ to $\RealTopology$
        by \obv{positive_real_change}.
    $h\in\bijections{\dom{h}}{\ran{h}}$ by
        \obv{open_partial_homeomorphism_bijection}.
    $h$ is a function by \obv{bijection_is_function}.
    Let $k=\restrl{h}{U}$.
    $k$ is an open partial homeomorphism from $\RealTopology$ to $\RealTopology$ by
        \auto{open_partial_homeomorphism_restriction}.
    $\dom{k}=\dom{h}\inter U$ by \auto{dom_of_restrl_eq_inter}.
    Show that for every $x\in\dom{k}$ there exists $V\in\opens[\RealTopology]$
        such that $x\in V$ and $\restrl{k}{V}$ is an increasing real map.
    \begin{subproof}
        Fix $x\in\dom{k}$.
        $x\in\dom{h}$ by \auto{inter}.
        Take $V\in\opens[\RealTopology]$ such that $x\in V$ and
            $\restrl{h}{V}$ is an increasing real map by \obv{positive_real_change}.
        $\restrl{\restrl{h}{V}}{U}$ is an increasing real map by
            \obv{increasing_real_map_restriction}.
        $\restrl{k}{V}=\restrl{\restrl{h}{V}}{U}$ by
            \auto{restriction_intersection,inter_comm}.
        Follows by \auto.
    \end{subproof}
    Follows by \auto{positive_real_change}.
\end{proof}

\begin{theorem}\label{positive_real_change_converse}
    Suppose $h$ is a positive real coordinate change.
    Then $\converse{h}$ is a positive real coordinate change.
\end{theorem}
\begin{proof}
    $h$ is an open partial homeomorphism from $\RealTopology$ to $\RealTopology$
        by \obv{positive_real_change}.
    $h\in\bijections{\dom{h}}{\ran{h}}$ by
        \obv{open_partial_homeomorphism_bijection}.
    $h$ is an injection by \auto{bijections,bijection_is_function}.
    Let $k=\converse{h}$.
    $k$ is an open partial homeomorphism from $\RealTopology$ to $\RealTopology$ by
        \auto{open_partial_homeomorphism_converse}.
    $\dom{k}=\ran{h}$ by \auto{dom_converse}.
    Show that for every $y\in\dom{k}$ there exists $V\in\opens[\RealTopology]$
        such that $y\in V$ and $\restrl{k}{V}$ is an increasing real map.
    \begin{subproof}
        Fix $y\in\dom{k}$.
        Take $x\in\dom{h}$ such that $h(x)=y$ by \auto{fun_ran_iff}.
        Take $U\in\opens[\RealTopology]$ such that $x\in U$ and
            $\restrl{h}{U}$ is an increasing real map by \obv{positive_real_change}.
        Let $V=\img{h}{U}$.
        $V\in\opens[\RealTopology]$ by \auto{open_partial_homeomorphism_image_open}.
        $(x,y)\in h$ by \auto{function_apply_elim}.
        $y\in V$ by \auto{img_iff}.
        $\converse{\restrl{h}{U}}$ is an increasing real map by
            \obv{increasing_real_map_converse}.
        $\converse{\restrl{h}{U}}=\restrl{k}{V}$ by \auto{restriction_converse}.
        Follows by \auto.
    \end{subproof}
    Follows by \auto{positive_real_change}.
\end{proof}

\begin{theorem}\label{positive_real_change_locality}
    Suppose $h$ is an open partial homeomorphism from $\RealTopology$ to $\RealTopology$.
    Suppose $B\subseteq\opens[\RealTopology]$.
    Suppose for every $x\in\dom{h}$ there exists $U\in B$ such that $x\in U$.
    Suppose for every $U\in B$ we have $\restrl{h}{U}$ is a positive real coordinate change.
    Then $h$ is a positive real coordinate change.
\end{theorem}
\begin{proof}
    $h\in\bijections{\dom{h}}{\ran{h}}$ by
        \obv{open_partial_homeomorphism_bijection}.
    $h$ is a function by \obv{bijection_is_function}.
    Show that for every $x\in\dom{h}$ there exists $W\in\opens[\RealTopology]$
        such that $x\in W$ and $\restrl{h}{W}$ is an increasing real map.
    \begin{subproof}
        Fix $x\in\dom{h}$.
        Take $U\in B$ such that $x\in U$.
        $U\in\opens[\RealTopology]$ by \obv{subseteq}.
        $\restrl{h}{U}$ is a positive real coordinate change.
        $x\in\dom{\restrl{h}{U}}$ by \auto{dom_of_restrl_eq_inter,inter}.
        Take $V\in\opens[\RealTopology]$ such that $x\in V$ and
            $\restrl{\restrl{h}{U}}{V}$ is an increasing real map by
            \obv{positive_real_change}.
        Let $W=U\inter V$.
        $W\in\opens[\RealTopology]$ by \auto{opens_inter}.
        $x\in W$ by \obv{inter}.
        $\restrl{\restrl{h}{U}}{V}=\restrl{h}{W}$ by
            \auto{restriction_intersection}.
        Follows by \auto.
    \end{subproof}
    Follows by \obv{positive_real_change}.
\end{proof}

\begin{theorem}\label{positive_real_change_composition}
    Suppose $f$ and $g$ are positive real coordinate changes.
    Then $g\circ f$ is a positive real coordinate change.
\end{theorem}
\begin{proof}
    $f$ is an open partial homeomorphism from $\RealTopology$ to $\RealTopology$ and
        $g$ is an open partial homeomorphism from $\RealTopology$ to $\RealTopology$ by
        \obv{positive_real_change}.
    $f$ and $g$ are functions by
        \auto{open_partial_homeomorphism_bijection,bijection_is_function}.
    $g\circ f$ is an open partial homeomorphism from $\RealTopology$ to $\RealTopology$
        by \obv{open_partial_homeomorphism_composition}.
    Show that for every $x\in\dom{g\circ f}$ there exists $W\in\opens[\RealTopology]$
        such that $x\in W$ and $\restrl{g\circ f}{W}$ is an increasing real map.
    \begin{subproof}
        Fix $x\in\dom{g\circ f}$.
        $x\in\dom{f}$ and $f(x)\in\dom{g}$ by \auto{partial_composition_domain_iff}.
        Take $U\in\opens[\RealTopology]$ such that $x\in U$ and
            $\restrl{f}{U}$ is an increasing real map by \obv{positive_real_change}.
        Take $V\in\opens[\RealTopology]$ such that $f(x)\in V$ and
            $\restrl{g}{V}$ is an increasing real map by \obv{positive_real_change}.
        $\preimg{f}{V}\in\opens[\RealTopology]$ by
            \obv{open_partial_homeomorphism_preimage_open}.
        $(x,f(x))\in f$ by \obv{function_apply_elim}.
        $x\in\preimg{f}{V}$ by \auto{preimg_iff}.
        Let $W=U\inter\preimg{f}{V}$.
        $W\in\opens[\RealTopology]$ by \auto{opens_inter}.
        $x\in W$ by \obv{inter}.
        $\restrl{\restrl{f}{U}}{\preimg{f}{V}}$ is an increasing real map by
            \obv{increasing_real_map_restriction}.
        $\restrl{\restrl{f}{U}}{\preimg{f}{V}}=\restrl{f}{W}$ by
            \auto{restriction_intersection}.
        $\restrl{f}{W}$ is an increasing real map.
        $(\restrl{g}{V})\circ(\restrl{f}{W})$ is an increasing real map by
            \obv{increasing_real_map_composition}.
        $W\subseteq\preimg{f}{V}$ by \auto{inter_lower_right}.
        $\restrl{g\circ f}{W}=(\restrl{g}{V})\circ(\restrl{f}{W})$ by
            \obv{partial_composition_local_restriction}.
        Follows by \auto.
    \end{subproof}
    Follows by \obv{positive_real_change}.
\end{proof}
