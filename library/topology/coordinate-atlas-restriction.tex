\section{Restricting coordinate atlases}

\import{topology/coordinate-atlas.tex}

Restricting each chart gives a presentation on an open subspace. For a
maximal atlas this is simply the family of charts whose domains lie in that
subspace; no extension of their graphs is needed.

\begin{definition}\label{restricted_atlas}
    $\RestrictedAtlas{A}{U}=\{\restrl{c}{U}\mid c\in A\}$.
\end{definition}

\begin{theorem}\label{restricted_atlas_iff}
    Then $b\in\RestrictedAtlas{A}{U}$ iff there exists $c\in A$ such that $b=\restrl{c}{U}$.
\end{theorem}
\begin{proof}
    Follows by \auto{restricted_atlas,logical_replacement_iff}.
\end{proof}

\begin{classification}\label{restricted_atlas_is_atlas}
    Let $G$ be a coordinate pseudogroup and $X$ be a topological space.
    Let $A$ be a coordinate atlas on $X$ under $G$.
    Let $U\in\opens[X]$.
    Then $\RestrictedAtlas{A}{U}$ is a coordinate atlas on $\subspaceTopology{X}{U}$ under $G$.
\end{classification}
\begin{proof}
    Let $B=\RestrictedAtlas{A}{U}$ and $Y=\subspaceTopology{X}{U}$.
    $Y$ is a topological space by \obv{subspace_topology_is_topological_space}.
    $\model[G]$ is a topological space by \obv{pseudogroup_model}.
    $U\subseteq\carrier[X]$ by \obv{opens_type}.
    $\carrier[Y]=U$ by \auto{subspace_topology_carrier_of_subset}.
    Show that every element of $B$ is an open partial homeomorphism from $Y$ to $\model[G]$.
    \begin{subproof}
        Fix $b\in B$.
        Take $c\in A$ such that $b=\restrl{c}{U}$ by \obv{restricted_atlas_iff}.
        $c$ is an open partial homeomorphism from $X$ to $\model[G]$ by \obv{coordinate_atlas}.
        $b$ is an open partial homeomorphism from $X$ to $\model[G]$ by
            \auto{open_partial_homeomorphism_restriction}.
        $\dom{b}\subseteq U$ by \auto{restrl_dom}.
        Follows by \auto{open_partial_homeomorphism_open_subspace}.
    \end{subproof}
    Show that for all $b,d\in B$ we have $b$ is compatible with $d$ on $Y$ under $G$.
    \begin{subproof}
        Fix $b,d\in B$.
        Take $c\in A$ such that $b=\restrl{c}{U}$ by \obv{restricted_atlas_iff}.
        Take $e\in A$ such that $d=\restrl{e}{U}$ by \obv{restricted_atlas_iff}.
        $c$ is compatible with $e$ on $X$ under $G$ by \obv{coordinate_atlas}.
        $b$ is compatible with $e$ on $X$ under $G$ by \auto{coordinate_compatibility_restrict_left}.
        $b$ is compatible with $d$ on $X$ under $G$ by \auto{coordinate_compatibility_restrict_right}.
        $\dom{b}\subseteq U$ and $\dom{d}\subseteq U$ by
            \auto{open_partial_homeomorphism_open_subspace}.
        Follows by \auto{coordinate_compatibility_open_subspace}.
    \end{subproof}
    Show that for every $p\in\carrier[Y]$ there exists $b\in B$ such that $p\in\dom{b}$.
    \begin{subproof}
        Fix $p\in\carrier[Y]$.
        $p\in\carrier[X]$ by \obv{subseteq}.
        Take $c\in A$ such that $p\in\dom{c}$ by \obv{coordinate_atlas}.
        $c$ is a function by
            \auto{coordinate_atlas,open_partial_homeomorphism_bijection,bijection_is_function}.
        $\restrl{c}{U}\in B$ by \obv{restricted_atlas_iff}.
        $p\in\dom{\restrl{c}{U}}$ by \auto{dom_of_restrl_eq_inter,inter}.
    \end{subproof}
    Follows by \auto{coordinate_atlas}.
\end{proof}

\begin{theorem}\label{restricted_atlas_composition}
    Suppose every element of $A$ is a relation.
    Then $\RestrictedAtlas{\RestrictedAtlas{A}{U}}{V}=\RestrictedAtlas{A}{U\inter V}$.
\end{theorem}
\begin{proof}
    For every $c\in A$ we have $\restrl{\restrl{c}{U}}{V}=\restrl{c}{U\inter V}$ by
        \obv{restriction_intersection}.
    For every $b$ we have $b\in\RestrictedAtlas{\RestrictedAtlas{A}{U}}{V}$ iff
        $b\in\RestrictedAtlas{A}{U\inter V}$ by \auto{restricted_atlas_iff}.
    Follows by \obv{setext}.
\end{proof}

\subsection{Restriction of maximal atlases}

\begin{theorem}\label{maximal_atlas_restriction_member}
    Let $G$ be a coordinate pseudogroup and $X$ be a topological space.
    Let $A$ be a maximal coordinate atlas on $X$ under $G$.
    Let $c\in A$ and $U\in\opens[X]$.
    Then $\restrl{c}{U}\in A$.
\end{theorem}
\begin{proof}
    $A$ is a coordinate atlas on $X$ under $G$ and $\compatibleHull{G}{X}{A}\subseteq A$ by
        \obv{maximal_coordinate_atlas}.
    $c$ is an open partial homeomorphism from $X$ to $\model[G]$ by \obv{coordinate_atlas}.
    $\restrl{c}{U}$ is an open partial homeomorphism from $X$ to $\model[G]$ by
        \obv{open_partial_homeomorphism_restriction}.
    For every $d\in A$ we have $c$ is compatible with $d$ on $X$ under $G$ by
        \obv{coordinate_atlas}.
    For every $d\in A$ we have $\restrl{c}{U}$ is compatible with $d$ on $X$ under $G$ by
        \obv{coordinate_compatibility_restrict_left}.
    $\restrl{c}{U}\in\compatibleHull{G}{X}{A}$ by \auto{compatible_hull_iff}.
    Follows by \obv{subseteq}.
\end{proof}

\begin{theorem}\label{restricted_maximal_atlas_iff}
    Let $G$ be a coordinate pseudogroup and $X$ be a topological space.
    Let $A$ be a maximal coordinate atlas on $X$ under $G$.
    Let $U\in\opens[X]$.
    Then $b\in\RestrictedAtlas{A}{U}$ iff $b\in A$ and $\dom{b}\subseteq U$.
\end{theorem}
\begin{proof}
    If $b\in\RestrictedAtlas{A}{U}$, then $b\in A$ and $\dom{b}\subseteq U$ by
        \auto{restricted_atlas_iff,maximal_atlas_restriction_member,restrl_dom}.
    Show that if $b\in A$ and $\dom{b}\subseteq U$, then $b\in\RestrictedAtlas{A}{U}$.
    \begin{subproof}
        Assume $b\in A$ and $\dom{b}\subseteq U$.
        $A$ is a coordinate atlas on $X$ under $G$ by \obv{maximal_coordinate_atlas}.
        $b$ is a function by
            \auto{coordinate_atlas,open_partial_homeomorphism_bijection,bijection_is_function}.
        $\restrl{b}{U}=b$ by
            \auto{restrl_by_dom,restriction_intersection,inter_absorb_supseteq_right}.
        Follows by \obv{restricted_atlas_iff}.
    \end{subproof}
\end{proof}

\begin{classification}\label{restricted_maximal_atlas_is_maximal}
    Let $G$ be a coordinate pseudogroup and $X$ be a topological space.
    Let $A$ be a maximal coordinate atlas on $X$ under $G$.
    Let $U\in\opens[X]$.
    Then $\RestrictedAtlas{A}{U}$ is a maximal coordinate atlas on $\subspaceTopology{X}{U}$ under $G$.
\end{classification}
\begin{proof}
    Let $B=\RestrictedAtlas{A}{U}$ and $Y=\subspaceTopology{X}{U}$.
    $Y$ is a topological space by \obv{subspace_topology_is_topological_space}.
    $\model[G]$ is a topological space by \obv{pseudogroup_model}.
    $A$ is a coordinate atlas on $X$ under $G$ and $\compatibleHull{G}{X}{A}\subseteq A$ by
        \obv{maximal_coordinate_atlas}.
    $B$ is a coordinate atlas on $Y$ under $G$ by \auto{restricted_atlas_is_atlas}.
    Show that every element of $\compatibleHull{G}{Y}{B}$ is an element of $B$.
    \begin{subproof}
        Fix $c\in\compatibleHull{G}{Y}{B}$.
        $c$ is an open partial homeomorphism from $Y$ to $\model[G]$ by \auto{compatible_hull_iff}.
        $c$ is an open partial homeomorphism from $X$ to $\model[G]$ and $\dom{c}\subseteq U$ by
            \auto{open_partial_homeomorphism_open_subspace}.
        $c$ is a function by \auto{open_partial_homeomorphism_bijection,bijection_is_function}.
        Show that for every $d\in A$ we have $c$ is compatible with $d$ on $X$ under $G$.
        \begin{subproof}
            Fix $d\in A$.
            $d$ is an open partial homeomorphism from $X$ to $\model[G]$ by \obv{coordinate_atlas}.
            $d$ is a function by \auto{open_partial_homeomorphism_bijection,bijection_is_function}.
            $\restrl{d}{U}\in B$ by \obv{restricted_atlas_iff}.
            $c$ is compatible with $\restrl{d}{U}$ on $Y$ under $G$ by \auto{compatible_hull_iff}.
            $\chartTransition{\restrl{d}{U}}{c}\in\changes[G]$ by \obv{coordinate_chart_compatibility}.
            $\chartTransition{\restrl{d}{U}}{c}=\chartTransition{d}{c}$ by
                \obv{chart_transition_target_restriction}.
            Follows by \obv{coordinate_chart_compatibility}.
        \end{subproof}
        $c\in\compatibleHull{G}{X}{A}$ by \auto{compatible_hull_iff}.
        $c\in A$ by \obv{subseteq}.
        Follows by \auto{restricted_maximal_atlas_iff}.
    \end{subproof}
    $\compatibleHull{G}{Y}{B}\subseteq B$ by \obv{subseteq}.
    Follows by \obv{maximal_coordinate_atlas}.
\end{proof}

\begin{theorem}\label{restricted_atlas_maximalization}
    Let $G$ be a coordinate pseudogroup and $X$ be a topological space.
    Let $A$ be a coordinate atlas on $X$ under $G$.
    Let $U\in\opens[X]$.
    Then $\MaxAtlas{G}{\subspaceTopology{X}{U}}{\RestrictedAtlas{A}{U}}
        =\RestrictedAtlas{\MaxAtlas{G}{X}{A}}{U}$.
\end{theorem}
\begin{proof}
    Let $Y=\subspaceTopology{X}{U}$ and $B=\RestrictedAtlas{A}{U}$.
    Let $D=\RestrictedAtlas{\MaxAtlas{G}{X}{A}}{U}$.
    $Y$ is a topological space by \obv{subspace_topology_is_topological_space}.
    $B$ is a coordinate atlas on $Y$ under $G$ by \auto{restricted_atlas_is_atlas}.
    $\MaxAtlas{G}{X}{A}$ is a maximal coordinate atlas on $X$ under $G$ by
        \obv{max_atlas_is_maximal}.
    $D$ is a maximal coordinate atlas on $Y$ under $G$ by
        \auto{restricted_maximal_atlas_is_maximal}.
    $D$ is a coordinate atlas on $Y$ under $G$ by \obv{maximal_coordinate_atlas}.
    $A\subseteq\MaxAtlas{G}{X}{A}$ by \obv{atlas_subset_max_atlas}.
    $B\subseteq D$ by \auto{restricted_atlas_iff,subseteq}.
    $\MaxAtlas{G}{Y}{B}=\MaxAtlas{G}{Y}{D}$ by \obv{atlas_extension_same_max_atlas}.
    $D=\MaxAtlas{G}{Y}{D}$ by \obv{maximal_coordinate_atlas_fixed_point}.
\end{proof}
