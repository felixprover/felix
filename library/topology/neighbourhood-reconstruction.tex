\section{Recovering a topology from its neighbourhood filters}

\import{topology/neighbourhood-within.tex}

An open set is precisely a carrier subset which is a neighbourhood of each
of its points.  Consequently, the pointwise neighbourhood filters determine
the topology, though not any additional fields of a richer structure.

\begin{theorem}\label{open_iff_neighbourhood_of_each_point}
    Let $X$ be a topological space.
    Then $U\in\opens[X]$ iff
        $U\subseteq\carrier[X]$ and
        for every $x\in U$ we have $U\in\neighbourhoods{x}{X}$.
\end{theorem}
\begin{proof}
    Show that if $U\in\opens[X]$, then
        $U\subseteq\carrier[X]$ and
        for every $x\in U$ we have $U\in\neighbourhoods{x}{X}$.
    \begin{subproof}
        Assume $U\in\opens[X]$.
        Hence $U\subseteq\carrier[X]$ by \obv{opens_type}.
        Show that for every $x\in U$ we have
            $U\in\neighbourhoods{x}{X}$.
        \begin{subproof}
            Fix $x\in U$.
            Follows by \obv{open_set_containing_point_is_neighbourhood}.
        \end{subproof}
    \end{subproof}

    Show that if $U\subseteq\carrier[X]$ and
        for every $x\in U$ we have $U\in\neighbourhoods{x}{X}$,
        then $U\in\opens[X]$.
    \begin{subproof}
        Assume $U\subseteq\carrier[X]$ and
            for every $x\in U$ we have $U\in\neighbourhoods{x}{X}$.
        Show that $U\subseteq\interior{U}{X}$.
        \begin{subproof}
            It suffices to show that every element of $U$ is an element of
                $\interior{U}{X}$ by \obv{subseteq}.
            Fix $x\in U$.
            Hence $x\in\carrier[X]$ by \obv{subseteq}.
            $U\in\neighbourhoods{x}{X}$.
            Follows by \obv{interior_iff_neighbourhood}.
        \end{subproof}
        $\interior{U}{X}\subseteq U$ by \obv{interior_subseteq}.
        Hence $\interior{U}{X}=U$ by \obv{subseteq_antisymmetric}.
        Follows by \obv{interior_eq_self_implies_open,is_open}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{theorem}\label{equal_neighbourhood_filters_determine_opens}
    Let $X,Y$ be topological spaces.
    Suppose $\carrier[X]=\carrier[Y]$ and
        for every $x\in\carrier[X]$ we have
        $\neighbourhoods{x}{X}=\neighbourhoods{x}{Y}$.
    Then $\opens[X]=\opens[Y]$.
\end{theorem}
\begin{proof}
    Show that $\opens[X]\subseteq\opens[Y]$.
    \begin{subproof}
        It suffices to show that every element of $\opens[X]$ is an element
            of $\opens[Y]$ by \obv{subseteq}.
        Fix $U\in\opens[X]$.
        $U\subseteq\carrier[X]$ by \obv{opens_type}.
        For every $x\in U$ we have $U\in\neighbourhoods{x}{X}$ by
            \obv{open_set_containing_point_is_neighbourhood}.
        Hence $U\subseteq\carrier[Y]$.
        Show that for every $x\in U$ we have
            $U\in\neighbourhoods{x}{Y}$.
        \begin{subproof}
            Fix $x\in U$.
            Hence $x\in\carrier[X]$ by \obv{subseteq}.
            $\neighbourhoods{x}{X}=\neighbourhoods{x}{Y}$.
            $U\in\neighbourhoods{x}{X}$.
        \end{subproof}
        $U\in\opens[Y]$ iff
            $U\subseteq\carrier[Y]$ and
            for every $x\in U$ we have $U\in\neighbourhoods{x}{Y}$ by
            \obv{open_iff_neighbourhood_of_each_point}.
    \end{subproof}

    Show that $\opens[Y]\subseteq\opens[X]$.
    \begin{subproof}
        It suffices to show that every element of $\opens[Y]$ is an element
            of $\opens[X]$ by \obv{subseteq}.
        Fix $U\in\opens[Y]$.
        $U\subseteq\carrier[Y]$ by \obv{opens_type}.
        For every $x\in U$ we have $U\in\neighbourhoods{x}{Y}$ by
            \obv{open_set_containing_point_is_neighbourhood}.
        Hence $U\subseteq\carrier[X]$.
        Show that for every $x\in U$ we have
            $U\in\neighbourhoods{x}{X}$.
        \begin{subproof}
            Fix $x\in U$.
            Hence $x\in\carrier[X]$ by \obv{subseteq}.
            $\neighbourhoods{x}{X}=\neighbourhoods{x}{Y}$.
            $U\in\neighbourhoods{x}{Y}$.
        \end{subproof}
        $U\in\opens[X]$ iff
            $U\subseteq\carrier[X]$ and
            for every $x\in U$ we have $U\in\neighbourhoods{x}{X}$ by
            \obv{open_iff_neighbourhood_of_each_point}.
    \end{subproof}
    Follows by \obv{subseteq_antisymmetric}.
\end{proof}

\begin{corollary}\label{equal_neighbourhood_filters_determine_topological_reduct}
    Let $X,Y$ be topological spaces.
    Suppose $\carrier[X]=\carrier[Y]$ and
        for every $x\in\carrier[X]$ we have
        $\neighbourhoods{x}{X}=\neighbourhoods{x}{Y}$.
    Then $\TopologicalReduct{X}=\TopologicalReduct{Y}$.
\end{corollary}
\begin{proof}
    $\opens[X]=\opens[Y]$ by
        \obv{equal_neighbourhood_filters_determine_opens}.
    $\TopologicalReduct{X}$ is an exact topological space by
        \obv{topological_reduct_is_exact}.
    $\TopologicalReduct{Y}$ is an exact topological space by
        \obv{topological_reduct_is_exact}.
    $\carrier[\TopologicalReduct{X}]=\carrier[X]$ and
        $\opens[\TopologicalReduct{X}]=\opens[X]$ by
        \obv{topological_reduct_carrier,topological_reduct_opens}.
    $\carrier[\TopologicalReduct{Y}]=\carrier[Y]$ and
        $\opens[\TopologicalReduct{Y}]=\opens[Y]$ by
        \obv{topological_reduct_carrier,topological_reduct_opens}.
    Hence
        $\carrier[\TopologicalReduct{X}]
            =\carrier[\TopologicalReduct{Y}]$ and
        $\opens[\TopologicalReduct{X}]
            =\opens[\TopologicalReduct{Y}]$.
    Follows by structure extensionality.
\end{proof}

\begin{corollary}\label{equal_neighbourhood_filters_determine_exact_spaces}
    Let $X,Y$ be topological spaces.
    Suppose $X$ is an exact topological space and
        $Y$ is an exact topological space.
    Suppose $\carrier[X]=\carrier[Y]$ and
        for every $x\in\carrier[X]$ we have
        $\neighbourhoods{x}{X}=\neighbourhoods{x}{Y}$.
    Then $X=Y$.
\end{corollary}
\begin{proof}
    $\TopologicalReduct{X}=\TopologicalReduct{Y}$ by
        \obv{equal_neighbourhood_filters_determine_topological_reduct}.
    $\TopologicalReduct{X}=X$ and $\TopologicalReduct{Y}=Y$ by
        \obv{topological_reduct_of_exact}.
\end{proof}
