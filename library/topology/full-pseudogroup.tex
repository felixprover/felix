\section{The full coordinate pseudogroup}

\import{topology/coordinate-pseudogroup.tex}

For a fixed model, the possible coordinate changes form a set of graphs.
Taking all of them gives the largest coordinate pseudogroup on that model.

\begin{definition}\label{open_partial_homeomorphisms}
    $\openPartialHomeomorphisms{X}{Y}=
        \{h\in\pow{\carrier[X]\times\carrier[Y]}\mid
            \text{$h$ is an open partial homeomorphism from $X$ to $Y$}\}$.
\end{definition}

\begin{theorem}\label{open_partial_homeomorphisms_iff}
    Then $h\in\openPartialHomeomorphisms{X}{Y}$ iff
        $h$ is an open partial homeomorphism from $X$ to $Y$.
\end{theorem}
\begin{proof}
    If $h\in\openPartialHomeomorphisms{X}{Y}$, then
        $h$ is an open partial homeomorphism from $X$ to $Y$ by
        \obv{open_partial_homeomorphisms}.
    Show that if $h$ is an open partial homeomorphism from $X$ to $Y$,
        then $h\in\openPartialHomeomorphisms{X}{Y}$.
    \begin{subproof}
        Assume $h$ is an open partial homeomorphism from $X$ to $Y$.
        $h$ is a continuous partial map from $X$ to $Y$ by
            \obv{open_partial_homeomorphism_is_continuous_partial_map}.
        $h\subseteq\carrier[X]\times\carrier[Y]$ by
            \obv{continuous_partial_map_graph_bound}.
        $h\in\pow{\carrier[X]\times\carrier[Y]}$ by \obv{pow_iff}.
        Follows by \obv{open_partial_homeomorphisms}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{construction}\label{full_pseudogroup}
    Let $E$ be a topological space.
    The coordinate pseudogroup $\FullPseudogroup{E}=
        \structure{\model=E,\changes=\openPartialHomeomorphisms{E}{E}}$.
\end{construction}
\begin{proof}
    Let $G=\FullPseudogroup{E}$.
    $\model[G]=E$ and $\changes[G]=\openPartialHomeomorphisms{E}{E}$ by
        \auto{coordinate_pseudogroupaggregateprojection_model,
            coordinate_pseudogroupaggregateprojection_changes}.
    $G$ is a record by \obv{coordinate_pseudogroupaggregateexact}.
    Every element of $\changes[G]$ is an open partial homeomorphism from
        $\model[G]$ to $\model[G]$ by \auto{open_partial_homeomorphisms_iff}.
    $\carrier[E]\in\opens[E]$ by \obv{carrier_open}.
    $\identity{\carrier[E]}$ is an open partial homeomorphism from $E$ to $E$ by
        \obv{identity_open_partial_homeomorphism}.
    $\identity{\carrier[\model[G]]}\in\changes[G]$ by
        \auto{open_partial_homeomorphisms_iff}.
    Show that for every $h\in\changes[G]$ we have for every $U\in\opens[\model[G]]$
        we have $\restrl{h}{U}\in\changes[G]$.
    \begin{subproof}
        Fix $h\in\changes[G]$ and $U\in\opens[\model[G]]$.
        $h$ is an open partial homeomorphism from $E$ to $E$ by
            \auto{open_partial_homeomorphisms_iff}.
        $\restrl{h}{U}$ is an open partial homeomorphism from $E$ to $E$ by
            \obv{open_partial_homeomorphism_restriction}.
        Follows by \auto{open_partial_homeomorphisms_iff}.
    \end{subproof}
    Show that for every $h\in\changes[G]$ we have $\converse{h}\in\changes[G]$.
    \begin{subproof}
        Fix $h\in\changes[G]$.
        $h$ is an open partial homeomorphism from $E$ to $E$ by
            \auto{open_partial_homeomorphisms_iff}.
        $\converse{h}$ is an open partial homeomorphism from $E$ to $E$ by
            \obv{open_partial_homeomorphism_converse}.
        Follows by \auto{open_partial_homeomorphisms_iff}.
    \end{subproof}
    Show that for all $f,g\in\changes[G]$ we have $g\circ f\in\changes[G]$.
    \begin{subproof}
        Fix $f,g\in\changes[G]$.
        $f$ is an open partial homeomorphism from $E$ to $E$ and
            $g$ is an open partial homeomorphism from $E$ to $E$ by
            \auto{open_partial_homeomorphisms_iff}.
        $g\circ f$ is an open partial homeomorphism from $E$ to $E$ by
            \obv{open_partial_homeomorphism_composition}.
        Follows by \auto{open_partial_homeomorphisms_iff}.
    \end{subproof}
    For every $h$ such that $h$ is an open partial homeomorphism from
        $\model[G]$ to $\model[G]$ we have $h\in\changes[G]$ by
        \auto{open_partial_homeomorphisms_iff}.
    For every $h$ such that $h$ is an open partial homeomorphism
        from $\model[G]$ to $\model[G]$ we have
        for every $B\subseteq\opens[\model[G]]$ such that
            (for every $x\in\dom{h}$ there exists $U\in B$ such that $x\in U$) and
            (for every $U\in B$ we have $\restrl{h}{U}\in\changes[G]$)
        we have $h\in\changes[G]$ by \auto.
    $G$ is a coordinate pseudogroup by \auto{coordinate_pseudogroup}.
    Follows by \auto{coordinate_pseudogroup}.
\end{proof}

\begin{theorem}\label{full_pseudogroup_model}
    Let $E$ be a topological space.
    Then $\model[\FullPseudogroup{E}]=E$.
\end{theorem}
\begin{proof}
    Follows by \obv{full_pseudogroupprojection_model}.
\end{proof}

\begin{classification}\label{full_pseudogroup_on}
    Let $E$ be a topological space.
    Then $\FullPseudogroup{E}$ is a pseudogroup on $E$.
\end{classification}
\begin{proof}
    $\FullPseudogroup{E}$ is a coordinate pseudogroup by \obv{full_pseudogroup}.
    $\model[\FullPseudogroup{E}]=E$ by \obv{full_pseudogroup_model}.
    Follows by \obv{pseudogroup_on}.
\end{proof}

\begin{theorem}\label{full_pseudogroup_changes}
    Let $E$ be a topological space.
    Then $h\in\changes[\FullPseudogroup{E}]$ iff
        $h$ is an open partial homeomorphism from $E$ to $E$.
\end{theorem}
\begin{proof}
    $\changes[\FullPseudogroup{E}]=\openPartialHomeomorphisms{E}{E}$ by
        \obv{full_pseudogroupprojection_changes}.
    Follows by \auto{open_partial_homeomorphisms_iff}.
\end{proof}

\begin{theorem}\label{full_pseudogroup_is_exact}
    Let $E$ be a topological space.
    Then $\FullPseudogroup{E}$ is an exact coordinate pseudogroup.
\end{theorem}
\begin{proof}
    Follows by \auto{full_pseudogroup,full_pseudogroupshape}.
\end{proof}

\begin{theorem}\label{full_pseudogroup_maximal}
    Let $G$ be a coordinate pseudogroup.
    Then $G$ is a subpseudogroup of $\FullPseudogroup{\model[G]}$.
\end{theorem}
\begin{proof}
    $\model[G]$ is a topological space by \obv{pseudogroup_model}.
    $\FullPseudogroup{\model[G]}$ is a coordinate pseudogroup by
        \obv{full_pseudogroup}.
    $\model[\FullPseudogroup{\model[G]}]=\model[G]$ by
        \obv{full_pseudogroup_model}.
    Every element of $\changes[G]$ is an open partial homeomorphism from
        $\model[G]$ to $\model[G]$ by \obv{pseudogroup_changes}.
    $\changes[G]\subseteq\changes[\FullPseudogroup{\model[G]}]$ by
        \auto{full_pseudogroup_changes,subseteq}.
    Follows by \obv{subpseudogroup}.
\end{proof}
