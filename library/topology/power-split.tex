\section{Splitting topological powers}

\import{topology/power-coordinate.tex}
\import{topology/product.tex}
\import{set/function-split.tex}

Splitting a coordinate set into disjoint blocks turns its power into a
binary product of powers. The identification is a homeomorphism of explicit
function-space carriers, not an equality of those carriers.

\begin{theorem}\label{topological_power_split_continuous}
    Let $X$ be a topological space.
    Then $\functionSplit{I}{J}{\carrier[X]}$ is continuous from
        $\TopologicalPower{X}{I\union J}$ to
        $\BinaryProductTopology{\TopologicalPower{X}{I}}{\TopologicalPower{X}{J}}$.
\end{theorem}
\begin{proof}
    $I\subseteq I\union J$ and $J\subseteq I\union J$ by \auto{subseteq,union_iff}.
    $\functionRestriction{I\union J}{I}{\carrier[X]}$ is continuous from
        $\TopologicalPower{X}{I\union J}$ to $\TopologicalPower{X}{I}$ by
        \obv{topological_power_restriction_continuous}.
    $\functionRestriction{I\union J}{J}{\carrier[X]}$ is continuous from
        $\TopologicalPower{X}{I\union J}$ to $\TopologicalPower{X}{J}$ by
        \obv{topological_power_restriction_continuous}.
    Follows by \auto{function_split,topological_power_carrier,continuous_product_pairing}.
\end{proof}

\begin{theorem}\label{topological_power_join_continuous}
    Let $X$ be a topological space.
    Suppose $I$ is disjoint from $J$.
    Then $\functionJoin{I}{J}{\carrier[X]}$ is continuous from
        $\BinaryProductTopology{\TopologicalPower{X}{I}}{\TopologicalPower{X}{J}}$ to
        $\TopologicalPower{X}{I\union J}$.
\end{theorem}
\begin{proof}
    Let $P=\TopologicalPower{X}{I}$ and $Q=\TopologicalPower{X}{J}$.
    Let $Z=\BinaryProductTopology{P}{Q}$ and $f=\functionJoin{I}{J}{\carrier[X]}$.
    $P$ is a topological space and $Q$ is a topological space by
        \auto{topological_power_is_topological_space}.
    Hence $Z$ is a topological space by \auto{binary_product_topology_is_topological_space}.
    $\carrier[P]=\funs{I}{\carrier[X]}$ and $\carrier[Q]=\funs{J}{\carrier[X]}$ by
        \auto{topological_power_carrier}.
    $\carrier[Z]=\carrier[P]\times\carrier[Q]$ by \auto{binary_product_topology_carrier}.
    $f\in\funs{\carrier[Z]}{\funs{I\union J}{\carrier[X]}}$ by \auto{function_join_type}.
    Show that for every $k\in I\union J$ we have
        $\functionEvaluation{I\union J}{\carrier[X]}{k}\circ f$ is continuous from $Z$ to $X$.
    \begin{subproof}
        Fix $k\in I\union J$.
        $k\in I$ or $k\in J$ by \obv{union_iff}.
        \begin{byCase}
            \caseOf{$k\in I$.}
                $\firstProductProjection{\carrier[P]}{\carrier[Q]}$ is continuous
                    from $Z$ to $P$ by \auto{first_product_projection_continuous}.
                $\functionEvaluation{I}{\carrier[X]}{k}$ is continuous from $P$ to $X$ by
                    \auto{topological_power_evaluation}.
                Hence $\functionEvaluation{I}{\carrier[X]}{k}\circ
                    \firstProductProjection{\carrier[P]}{\carrier[Q]}$ is continuous from $Z$ to $X$
                    by \obv{continuous_composition}.
                Follows by \auto{function_join_left_coordinate}.
            \caseOf{$k\in J$.}
                $\secondProductProjection{\carrier[P]}{\carrier[Q]}$ is continuous
                    from $Z$ to $Q$ by \auto{second_product_projection_continuous}.
                $\functionEvaluation{J}{\carrier[X]}{k}$ is continuous from $Q$ to $X$ by
                    \auto{topological_power_evaluation}.
                Hence $\functionEvaluation{J}{\carrier[X]}{k}\circ
                    \secondProductProjection{\carrier[P]}{\carrier[Q]}$ is continuous from $Z$ to $X$
                    by \obv{continuous_composition}.
                Follows by \auto{function_join_right_coordinate}.
        \end{byCase}
    \end{subproof}
    Follows by \auto{topological_power_continuity_iff}.
\end{proof}

\begin{theorem}\label{topological_power_split_homeomorphism}
    Let $X$ be a topological space.
    Suppose $I$ is disjoint from $J$.
    Then $\functionSplit{I}{J}{\carrier[X]}$ is a homeomorphism from
        $\TopologicalPower{X}{I\union J}$ to
        $\BinaryProductTopology{\TopologicalPower{X}{I}}{\TopologicalPower{X}{J}}$.
\end{theorem}
\begin{proof}
    $\functionSplit{I}{J}{\carrier[X]}$ is continuous from
        $\TopologicalPower{X}{I\union J}$ to
        $\BinaryProductTopology{\TopologicalPower{X}{I}}{\TopologicalPower{X}{J}}$ by
        \obv{topological_power_split_continuous}.
    $\functionJoin{I}{J}{\carrier[X]}$ is continuous from
        $\BinaryProductTopology{\TopologicalPower{X}{I}}{\TopologicalPower{X}{J}}$ to
        $\TopologicalPower{X}{I\union J}$ by \obv{topological_power_join_continuous}.
    $\converse{\functionSplit{I}{J}{\carrier[X]}}=\functionJoin{I}{J}{\carrier[X]}$ by
        \obv{function_split_converse}.
    $\functionSplit{I}{J}{\carrier[X]}\in
        \bijections{\carrier[\TopologicalPower{X}{I\union J}]}{
            \carrier[\BinaryProductTopology{\TopologicalPower{X}{I}}{\TopologicalPower{X}{J}}]}$ by
        \auto{function_split_bijection,topological_power_carrier,binary_product_topology_carrier}.
    Follows by \obv{homeomorphism}.
\end{proof}

\begin{corollary}\label{topological_power_join_homeomorphism}
    Let $X$ be a topological space.
    Suppose $I$ is disjoint from $J$.
    Then $\functionJoin{I}{J}{\carrier[X]}$ is a homeomorphism from
        $\BinaryProductTopology{\TopologicalPower{X}{I}}{\TopologicalPower{X}{J}}$ to
        $\TopologicalPower{X}{I\union J}$.
\end{corollary}
\begin{proof}
    Follows by \auto{topological_power_split_homeomorphism,homeomorphism_inverse,
        function_split_converse}.
\end{proof}
