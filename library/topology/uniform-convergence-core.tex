\section{Uniform convergence and completeness}

Uniform convergence is convergence in the topology induced by the
uniformity.  Thus the public convergence predicates assert their ambient
structure and typing conditions, while the familiar entourage formulas are
available as characterizations.  A Cauchy filter is proper and its
self-product refines the uniformity.  A Cauchy sequence is one whose tail
filter is Cauchy.

\begin{definition}\label{uniform_filter_convergence}
    $\uniformFilterConverges{X}{F}{x}$ iff
        $X$ is a uniform space and
        $F$ is a filter on $\carrier[X]$ and
        $\filterConverges{F}{x}{\uniformTopology{X}}$.
\end{definition}

\begin{definition}\label{uniform_sequence_convergence}
    $\uniformSequenceConverges{X}{s}{x}$ iff
        $X$ is a uniform space and
        $\topologicalSequenceConverges{\uniformTopology{X}}{s}{x}$.
\end{definition}

\begin{definition}\label{uniform_cauchy_filter}
    $\uniformCauchyFilter{X}{F}$ iff
        $X$ is a uniform space and
        $F$ is a proper filter on $\carrier[X]$ and
        $\productFilter{\carrier[X]}{\carrier[X]}{F}{F}$ refines
            $\uniformity[X]$ on
            $\carrier[X]\times\carrier[X]$.
\end{definition}

\begin{proposition}\label{uniform_filter_convergence_space}
    Suppose $\uniformFilterConverges{X}{F}{x}$.
    Then $X$ is a uniform space.
\end{proposition}
\begin{proof}
    Follows by \obv{uniform_filter_convergence}.
\end{proof}

\begin{proposition}\label{uniform_filter_convergence_filter}
    Suppose $\uniformFilterConverges{X}{F}{x}$.
    Then $F$ is a filter on $\carrier[X]$.
\end{proposition}
\begin{proof}
    Follows by \obv{uniform_filter_convergence}.
\end{proof}

\begin{proposition}\label{uniform_filter_convergence_point}
    Suppose $\uniformFilterConverges{X}{F}{x}$.
    Then $x\in\carrier[X]$.
\end{proposition}
\begin{proof}
    $\filterConverges{F}{x}{\uniformTopology{X}}$
        by \obv{uniform_filter_convergence}.
    Hence $x\in\carrier[\uniformTopology{X}]$
        by \obv{filter_converges}.
    $\carrier[\uniformTopology{X}]=\carrier[X]$
        by \obv{uniform_topology_carrier}.
\end{proof}

\begin{proposition}\label{uniform_sequence_convergence_space}
    Suppose $\uniformSequenceConverges{X}{s}{x}$.
    Then $X$ is a uniform space.
\end{proposition}
\begin{proof}
    Follows by \obv{uniform_sequence_convergence}.
\end{proof}

\begin{proposition}\label{uniform_sequence_convergence_sequence}
    Suppose $\uniformSequenceConverges{X}{s}{x}$.
    Then $s\in\funs{\naturals}{\carrier[X]}$.
\end{proposition}
\begin{proof}
    $\topologicalSequenceConverges{\uniformTopology{X}}{s}{x}$
        by \obv{uniform_sequence_convergence}.
    Hence $s\in\funs{\naturals}{\carrier[\uniformTopology{X}]}$
        by \obv{topological_sequence_convergence_sequence}.
    $\carrier[\uniformTopology{X}]=\carrier[X]$
        by \obv{uniform_topology_carrier}.
\end{proof}

\begin{proposition}\label{uniform_sequence_convergence_point}
    Suppose $\uniformSequenceConverges{X}{s}{x}$.
    Then $x\in\carrier[X]$.
\end{proposition}
\begin{proof}
    $\topologicalSequenceConverges{\uniformTopology{X}}{s}{x}$
        by \obv{uniform_sequence_convergence}.
    Hence $x\in\carrier[\uniformTopology{X}]$
        by \obv{topological_sequence_convergence_point}.
    $\carrier[\uniformTopology{X}]=\carrier[X]$
        by \obv{uniform_topology_carrier}.
\end{proof}

\begin{proposition}\label{uniform_cauchy_filter_space}
    Suppose $\uniformCauchyFilter{X}{F}$.
    Then $X$ is a uniform space.
\end{proposition}
\begin{proof}
    Follows by \obv{uniform_cauchy_filter}.
\end{proof}

\begin{theorem}\label{uniform_cauchy_filter_iff_entourage_squares}
    Let $X$ be a uniform space.
    Then $\uniformCauchyFilter{X}{F}$ iff
        $F$ is a proper filter on $\carrier[X]$ and
        for every $U\in\uniformity[X]$ there exists $A\in F$ such that
            $A\times A\subseteq U$.
\end{theorem}
\begin{proof}
    Show that if $\uniformCauchyFilter{X}{F}$, then
        $F$ is a proper filter on $\carrier[X]$ and
        for every $U\in\uniformity[X]$ there exists $A\in F$ such that
            $A\times A\subseteq U$.
    \begin{subproof}
        Assume $\uniformCauchyFilter{X}{F}$.
        $F$ is a proper filter on $\carrier[X]$ and
            $\productFilter{\carrier[X]}{\carrier[X]}{F}{F}$ refines
                $\uniformity[X]$ on
                $\carrier[X]\times\carrier[X]$ by
            \obv{uniform_cauchy_filter}.
        Hence $F$ is a filter on $\carrier[X]$ by
            \obv{proper_filter_is_filter}.
        $\uniformity[X]$ is a filter on
            $\carrier[X]\times\carrier[X]$ by \obv{uniformity_filter}.
        For every $U\in\uniformity[X]$ there exists $A\in F$ such that
            $A\times A\subseteq U$ by
            \obv{self_product_filter_refines_iff_squares}.
    \end{subproof}

    Show that if $F$ is a proper filter on $\carrier[X]$ and
        for every $U\in\uniformity[X]$ there exists $A\in F$ such that
            $A\times A\subseteq U$, then $\uniformCauchyFilter{X}{F}$.
    \begin{subproof}
        Assume $F$ is a proper filter on $\carrier[X]$ and
            for every $U\in\uniformity[X]$ there exists $A\in F$ such
            that $A\times A\subseteq U$.
        Hence $F$ is a filter on $\carrier[X]$ by
            \obv{proper_filter_is_filter}.
        $\uniformity[X]$ is a filter on
            $\carrier[X]\times\carrier[X]$ by \obv{uniformity_filter}.
        Hence $\productFilter{\carrier[X]}{\carrier[X]}{F}{F}$ refines
            $\uniformity[X]$ on
            $\carrier[X]\times\carrier[X]$ by
            \obv{self_product_filter_refines_iff_squares}.
        Follows by \obv{uniform_cauchy_filter}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{proposition}\label{uniform_cauchy_filter_is_proper}
    Suppose $\uniformCauchyFilter{X}{F}$.
    Then $F$ is a proper filter on $\carrier[X]$.
\end{proposition}
\begin{proof}
    Follows by \obv{uniform_cauchy_filter}.
\end{proof}

\begin{proposition}\label{uniform_cauchy_filter_entourage_square}
    Let $X$ be a uniform space.
    Suppose $\uniformCauchyFilter{X}{F}$ and $U\in\uniformity[X]$.
    Then there exists $A\in F$ such that $A\times A\subseteq U$.
\end{proposition}
\begin{proof}
    $\uniformCauchyFilter{X}{F}$ iff
        $F$ is a proper filter on $\carrier[X]$ and
        for every $V\in\uniformity[X]$ there exists $A\in F$ such that
            $A\times A\subseteq V$ by
        \obv{uniform_cauchy_filter_iff_entourage_squares}.
    For every $V\in\uniformity[X]$ there exists $A\in F$ such that
        $A\times A\subseteq V$ by \auto.
\end{proof}

\begin{definition}\label{uniform_cauchy_sequence}
    $\uniformCauchySequence{X}{s}$ iff
        $s\in\funs{\naturals}{\carrier[X]}$ and
        $\uniformCauchyFilter{X}{\sequenceTailFilter{\carrier[X]}{s}}$.
\end{definition}

\begin{proposition}\label{uniform_cauchy_sequence_space}
    Suppose $\uniformCauchySequence{X}{s}$.
    Then $X$ is a uniform space.
\end{proposition}
\begin{proof}
    $\uniformCauchyFilter{X}{\sequenceTailFilter{\carrier[X]}{s}}$
        by \obv{uniform_cauchy_sequence}.
    Follows by \obv{uniform_cauchy_filter_space}.
\end{proof}

\begin{proposition}\label{uniform_cauchy_sequence_sequence}
    Suppose $\uniformCauchySequence{X}{s}$.
    Then $s\in\funs{\naturals}{\carrier[X]}$.
\end{proposition}
\begin{proof}
    Follows by \obv{uniform_cauchy_sequence}.
\end{proof}

\begin{theorem}\label{uniform_cauchy_sequence_iff_entourages}
    Let $X$ be a uniform space.
    Then $\uniformCauchySequence{X}{s}$ iff
        $s\in\funs{\naturals}{\carrier[X]}$ and for every
            $U\in\uniformity[X]$ there exists $N\in\naturals$ such that for
            all $m,n\in\naturals$ such that $N\subseteq m$ and
            $N\subseteq n$ we have $(s(m),s(n))\in U$.
\end{theorem}
\begin{proof}
    Show that if $\uniformCauchySequence{X}{s}$, then
        $s\in\funs{\naturals}{\carrier[X]}$ and for every
            $U\in\uniformity[X]$ there exists $N\in\naturals$ such that for
            all $m,n\in\naturals$ such that $N\subseteq m$ and
            $N\subseteq n$ we have $(s(m),s(n))\in U$.
    \begin{subproof}
        Assume $\uniformCauchySequence{X}{s}$.
        $s\in\funs{\naturals}{\carrier[X]}$ by
            \obv{uniform_cauchy_sequence_sequence}.
        $\uniformCauchyFilter{X}
            {\sequenceTailFilter{\carrier[X]}{s}}$
            by \obv{uniform_cauchy_sequence}.
        Show that for every $U\in\uniformity[X]$ there exists
            $N\in\naturals$ such that for all $m,n\in\naturals$ such that
            $N\subseteq m$ and $N\subseteq n$ we have $(s(m),s(n))\in U$.
        \begin{subproof}
            Fix $U\in\uniformity[X]$.
            Take $A\in\sequenceTailFilter{\carrier[X]}{s}$ such that
                $A\times A\subseteq U$
                by \obv{uniform_cauchy_filter_entourage_square}.
            Take $N\in\naturals$ such that $\sequenceTail{s}{N}\subseteq A$
                by \auto{sequence_tail_filter_iff}.
            Show that for all $m,n\in\naturals$ such that $N\subseteq m$ and
                $N\subseteq n$ we have $(s(m),s(n))\in U$.
            \begin{subproof}
                Fix $m,n\in\naturals$.
                Assume $N\subseteq m$ and $N\subseteq n$.
                $s(m),s(n)\in\sequenceTail{s}{N}$
                    by \obv{sequence_tail_contains_term}.
                Hence $s(m),s(n)\in A$ by \obv{subseteq}.
                Hence $(s(m),s(n))\in A\times A$
                    by \obv{times_tuple_intro}.
                Follows by \obv{subseteq}.
            \end{subproof}
            Follows by \auto.
        \end{subproof}
    \end{subproof}

    Show that if $s\in\funs{\naturals}{\carrier[X]}$ and for every
        $U\in\uniformity[X]$ there exists $N\in\naturals$ such that for all
        $m,n\in\naturals$ such that $N\subseteq m$ and $N\subseteq n$ we
        have $(s(m),s(n))\in U$, then $\uniformCauchySequence{X}{s}$.
    \begin{subproof}
        Assume $s\in\funs{\naturals}{\carrier[X]}$ and for every
            $U\in\uniformity[X]$ there exists $N\in\naturals$ such that for
            all $m,n\in\naturals$ such that $N\subseteq m$ and
            $N\subseteq n$ we have $(s(m),s(n))\in U$.
        $\sequenceTailFilter{\carrier[X]}{s}$ is a proper filter on
            $\carrier[X]$ by \obv{sequence_tail_filter_is_proper_filter}.
        Show that for every $U\in\uniformity[X]$ there exists
            $A\in\sequenceTailFilter{\carrier[X]}{s}$ such that
            $A\times A\subseteq U$.
        \begin{subproof}
            Fix $U\in\uniformity[X]$.
            Take $N\in\naturals$ such that for all
                $m,n\in\naturals$ such that $N\subseteq m$ and
                $N\subseteq n$ we have $(s(m),s(n))\in U$.
            Let $A=\sequenceTail{s}{N}$.
            $A\subseteq\carrier[X]$ by \auto{sequence_tail_type}.
            Hence $A\in\sequenceTailFilter{\carrier[X]}{s}$
                by \auto{sequence_tail_filter_iff,subseteq_refl}.
            Show that $A\times A\subseteq U$.
            \begin{subproof}
                It suffices to show that every element of $A\times A$ is
                    an element of $U$ by \obv{subseteq}.
                Fix $p\in A\times A$.
                Take $a,b\in A$ such that $p=(a,b)$
                    by \obv{times_elem_is_tuple,times_tuple_elim}.
                Take $m\in\naturalTail{N}$ such that $(m,a)\in s$
                    by \auto{sequence_tail,img_iff}.
                Take $n\in\naturalTail{N}$ such that $(n,b)\in s$
                    by \auto{sequence_tail,img_iff}.
                $m,n\in\naturals$ and $N\subseteq m$ and $N\subseteq n$
                    by \obv{natural_tail}.
                $a=s(m)$ and $b=s(n)$
                    by \obv{funs_is_function,function_apply_intro}.
                Hence $(s(m),s(n))\in U$.
            \end{subproof}
        \end{subproof}
        $\uniformCauchyFilter{X}
                {\sequenceTailFilter{\carrier[X]}{s}}$ iff
            $\sequenceTailFilter{\carrier[X]}{s}$ is a proper filter on
                $\carrier[X]$ and
            for every $V\in\uniformity[X]$ there exists
                $A\in\sequenceTailFilter{\carrier[X]}{s}$ such that
                $A\times A\subseteq V$ by
            \obv{uniform_cauchy_filter_iff_entourage_squares}.
        Hence $\uniformCauchyFilter{X}
            {\sequenceTailFilter{\carrier[X]}{s}}$.
        Follows by \obv{uniform_cauchy_sequence}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{proposition}\label{uniform_cauchy_sequence_entourage}
    Suppose $\uniformCauchySequence{X}{s}$ and $U\in\uniformity[X]$.
    Then there exists $N\in\naturals$ such that for all
        $m,n\in\naturals$ such that $N\subseteq m$ and $N\subseteq n$
        we have $(s(m),s(n))\in U$.
\end{proposition}
\begin{proof}
    $X$ is a uniform space by \obv{uniform_cauchy_sequence_space}.
    $s\in\funs{\naturals}{\carrier[X]}$ by
        \obv{uniform_cauchy_sequence_sequence}.
    $\uniformCauchySequence{X}{s}$ iff
        $s\in\funs{\naturals}{\carrier[X]}$ and for every
            $V\in\uniformity[X]$ there exists $N\in\naturals$ such that for
            all $m,n\in\naturals$ such that $N\subseteq m$ and
            $N\subseteq n$ we have $(s(m),s(n))\in V$ by
        \obv{uniform_cauchy_sequence_iff_entourages}.
    For every $V\in\uniformity[X]$ there exists $N\in\naturals$ such that
        for all $m,n\in\naturals$ such that $N\subseteq m$ and
        $N\subseteq n$ we have $(s(m),s(n))\in V$ by \auto.
\end{proof}

\begin{definition}\label{complete_uniform_space}
    $X$ is complete uniform iff
        $X$ is a uniform space and
        for every $F$ such that $\uniformCauchyFilter{X}{F}$ there exists
            $x\in\carrier[X]$ such that
                $\uniformFilterConverges{X}{F}{x}$.
\end{definition}

\begin{definition}\label{sequentially_complete_uniform_space}
    $X$ is sequentially complete uniform iff
        $X$ is a uniform space and
        for every $s$ such that $\uniformCauchySequence{X}{s}$ there exists
            $x\in\carrier[X]$ such that
                $\uniformSequenceConverges{X}{s}{x}$.
\end{definition}

\subsection{Comparison with the induced topology}

\begin{theorem}\label{uniform_filter_convergence_iff_topological}
    Let $X$ be a uniform space and $F$ be a filter on $\carrier[X]$.
    Then $\uniformFilterConverges{X}{F}{x}$ iff
        $\filterConverges{F}{x}{\uniformTopology{X}}$.
\end{theorem}
\begin{proof}
    Show that if $\uniformFilterConverges{X}{F}{x}$, then
        $\filterConverges{F}{x}{\uniformTopology{X}}$.
    \begin{subproof}
        Assume $\uniformFilterConverges{X}{F}{x}$.
        Follows by \obv{uniform_filter_convergence}.
    \end{subproof}
    Show that if $\filterConverges{F}{x}{\uniformTopology{X}}$, then
        $\uniformFilterConverges{X}{F}{x}$.
    \begin{subproof}
        Assume $\filterConverges{F}{x}{\uniformTopology{X}}$.
        Follows by \obv{uniform_filter_convergence}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{theorem}\label{uniform_filter_convergence_iff_entourages}
    Let $X$ be a uniform space.
    Then $\uniformFilterConverges{X}{F}{x}$ iff
        $F$ is a filter on $\carrier[X]$ and $x\in\carrier[X]$ and for every
            $U\in\uniformity[X]$ we have $\uniformBall{X}{U}{x}\in F$.
\end{theorem}
\begin{proof}
    $\carrier[\uniformTopology{X}]=\carrier[X]$
        by \obv{uniform_topology_carrier}.
    $\opens[\uniformTopology{X}]=\uniformOpens{X}$
        by \obv{uniform_topology_opens}.

    Show that if $\uniformFilterConverges{X}{F}{x}$, then
        $F$ is a filter on $\carrier[X]$ and $x\in\carrier[X]$ and for every
            $U\in\uniformity[X]$ we have $\uniformBall{X}{U}{x}\in F$.
    \begin{subproof}
        Assume $\uniformFilterConverges{X}{F}{x}$.
        $F$ is a filter on $\carrier[X]$ by
            \obv{uniform_filter_convergence_filter}.
        $x\in\carrier[X]$ by \obv{uniform_filter_convergence_point}.
        Hence $x\in\carrier[\uniformTopology{X}]$.
        $\filterConverges{F}{x}{\uniformTopology{X}}$
            by \obv{uniform_filter_convergence}.
        $\neighbourhoods{x}{\uniformTopology{X}}\subseteq F$
            by \obv{filter_converges}.
        Show that for every $U\in\uniformity[X]$ we have
            $\uniformBall{X}{U}{x}\in F$.
        \begin{subproof}
            Fix $U\in\uniformity[X]$.
            Take $A\in\opens[\uniformTopology{X}]$ such that
                $x\in A\subseteq\uniformBall{X}{U}{x}$
                by \obv{uniform_ball_contains_open_neighbourhood}.
            $\uniformBall{X}{U}{x}\subseteq\carrier[X]$
                by \obv{uniform_ball_subset_carrier}.
            Hence $\uniformBall{X}{U}{x}
                \subseteq\carrier[\uniformTopology{X}]$.
            Hence $\uniformBall{X}{U}{x}
                \in\pow{\carrier[\uniformTopology{X}]}$
                by \obv{powerset_intro}.
            Hence $\uniformBall{X}{U}{x}
                \in\neighbourhoods{x}{\uniformTopology{X}}$
                by \auto{neighbourhoods}.
            Follows by \obv{subseteq}.
        \end{subproof}
    \end{subproof}

    Show that if $F$ is a filter on $\carrier[X]$ and
        $x\in\carrier[X]$ and for every $U\in\uniformity[X]$ we have
        $\uniformBall{X}{U}{x}\in F$, then
            $\uniformFilterConverges{X}{F}{x}$.
    \begin{subproof}
        Assume $F$ is a filter on $\carrier[X]$ and
            $x\in\carrier[X]$ and for every $U\in\uniformity[X]$ we have
            $\uniformBall{X}{U}{x}\in F$.
        Hence $x\in\carrier[\uniformTopology{X}]$.
        Show that $\neighbourhoods{x}{\uniformTopology{X}}\subseteq F$.
        \begin{subproof}
            It suffices to show that every element of
                $\neighbourhoods{x}{\uniformTopology{X}}$ is an element of
                $F$ by \obv{subseteq}.
            Fix $N\in\neighbourhoods{x}{\uniformTopology{X}}$.
            Take $A\in\opens[\uniformTopology{X}]$ such that
                $x\in A\subseteq N$ by \obv{neighbourhoods}.
            Hence $A\in\uniformOpens{X}$.
            Hence $\uniformOpen{X}{A}$ by \obv{uniform_opens_iff}.
            Take $U\in\uniformity[X]$ such that
                $\uniformBall{X}{U}{x}\subseteq A$
                by \obv{uniform_open}.
            $\uniformBall{X}{U}{x}\in F$.
            $\uniformBall{X}{U}{x}\subseteq\carrier[X]$
                by \obv{uniform_ball_subset_carrier}.
            $N\subseteq\carrier[\uniformTopology{X}]$
                by \auto{neighbourhoods,pow_iff,subseteq}.
            Hence $N\subseteq\carrier[X]$.
            $\uniformBall{X}{U}{x}\subseteq N$
                by \obv{subseteq_transitive}.
            Follows by \obv{filter}.
        \end{subproof}
        Hence $\filterConverges{F}{x}{\uniformTopology{X}}$
            by \obv{filter_converges}.
        Follows by \obv{uniform_filter_convergence}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{proposition}\label{uniform_filter_convergence_entourage}
    Suppose $\uniformFilterConverges{X}{F}{x}$ and $U\in\uniformity[X]$.
    Then $\uniformBall{X}{U}{x}\in F$.
\end{proposition}
\begin{proof}
    $F$ is a filter on $\carrier[X]$ by
        \obv{uniform_filter_convergence_filter}.
    $x\in\carrier[X]$ by \obv{uniform_filter_convergence_point}.
    $\uniformFilterConverges{X}{F}{x}$ iff
        $F$ is a filter on $\carrier[X]$ and $x\in\carrier[X]$ and for every
            $V\in\uniformity[X]$ we have $\uniformBall{X}{V}{x}\in F$ by
        \obv{uniform_filter_convergence_iff_entourages}.
    For every $V\in\uniformity[X]$ we have
        $\uniformBall{X}{V}{x}\in F$ by \auto.
\end{proof}

\subsection{Convergence and Cauchy criteria for filter bases}

\begin{theorem}\label{uniform_generated_filter_convergence_iff}
    Let $X$ be a uniform space and $B$ be a proper filter basis on
        $\carrier[X]$.
    Let $x\in\carrier[X]$.
    Then
        $\uniformFilterConverges{X}
            {\generatedFilter{\carrier[X]}{B}}{x}$ iff
        for every $U\in\uniformity[X]$ there exists $A\in B$ such that
            $A\subseteq\uniformBall{X}{U}{x}$.
\end{theorem}
\begin{proof}
    $B$ is a filter basis on $\carrier[X]$ by
        \obv{proper_filter_basis_is_filter_basis}.
    $\generatedFilter{\carrier[X]}{B}$ is a filter on $\carrier[X]$
        by \obv{generated_filter_is_filter}.
    Show that if
        $\uniformFilterConverges{X}
            {\generatedFilter{\carrier[X]}{B}}{x}$, then
        for every $U\in\uniformity[X]$ there exists $A\in B$ such that
            $A\subseteq\uniformBall{X}{U}{x}$.
    \begin{subproof}
        Assume
            $\uniformFilterConverges{X}
                {\generatedFilter{\carrier[X]}{B}}{x}$.
        Fix $U\in\uniformity[X]$.
        $\uniformBall{X}{U}{x}
            \in\generatedFilter{\carrier[X]}{B}$
            by \auto{uniform_filter_convergence_iff_entourages}.
        Follows by \auto{generated_filter_iff}.
    \end{subproof}

    Show that if for every $U\in\uniformity[X]$ there exists $A\in B$
        such that $A\subseteq\uniformBall{X}{U}{x}$, then
        $\uniformFilterConverges{X}
            {\generatedFilter{\carrier[X]}{B}}{x}$.
    \begin{subproof}
        Assume for every $U\in\uniformity[X]$ there exists $A\in B$
            such that $A\subseteq\uniformBall{X}{U}{x}$.
        Show that for every $U\in\uniformity[X]$ we have
            $\uniformBall{X}{U}{x}
                \in\generatedFilter{\carrier[X]}{B}$.
        \begin{subproof}
            Fix $U\in\uniformity[X]$.
            Take $A\in B$ such that
                $A\subseteq\uniformBall{X}{U}{x}$.
            $\uniformBall{X}{U}{x}\subseteq\carrier[X]$
                by \obv{uniform_ball_subset_carrier}.
            Follows by \auto{generated_filter_iff}.
        \end{subproof}
        Follows by \auto{uniform_filter_convergence_iff_entourages}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{theorem}\label{uniform_generated_filter_cauchy_iff}
    Let $X$ be a uniform space and $B$ be a proper filter basis on
        $\carrier[X]$.
    Then
        $\uniformCauchyFilter{X}
            {\generatedFilter{\carrier[X]}{B}}$ iff
        for every $U\in\uniformity[X]$ there exists $A\in B$ such that
            $A\times A\subseteq U$.
\end{theorem}
\begin{proof}
    $B$ is a filter basis on $\carrier[X]$ by
        \obv{proper_filter_basis_is_filter_basis}.
    $\generatedFilter{\carrier[X]}{B}$ is a proper filter on $\carrier[X]$
        by \obv{generated_filter_is_proper_filter}.
    Show that if
        $\uniformCauchyFilter{X}
            {\generatedFilter{\carrier[X]}{B}}$, then
        for every $U\in\uniformity[X]$ there exists $A\in B$ such that
            $A\times A\subseteq U$.
    \begin{subproof}
        Assume
            $\uniformCauchyFilter{X}
                {\generatedFilter{\carrier[X]}{B}}$.
        Fix $U\in\uniformity[X]$.
        Take $C\in\generatedFilter{\carrier[X]}{B}$ such that
            $C\times C\subseteq U$ by
            \obv{uniform_cauchy_filter_entourage_square}.
        Take $A\in B$ such that $A\subseteq C$
            by \auto{generated_filter_iff}.
        $A\times A\subseteq C\times A$ by \obv{times_subseteq_left}.
        $C\times A\subseteq C\times C$ by \obv{times_subseteq_right}.
        Hence $A\times A\subseteq C\times C$
            by \obv{subseteq_transitive}.
        Hence $A\times A\subseteq U$ by \obv{subseteq_transitive}.
    \end{subproof}

    Show that if for every $U\in\uniformity[X]$ there exists $A\in B$
        such that $A\times A\subseteq U$, then
        $\uniformCauchyFilter{X}
            {\generatedFilter{\carrier[X]}{B}}$.
    \begin{subproof}
        Assume for every $U\in\uniformity[X]$ there exists $A\in B$
            such that $A\times A\subseteq U$.
        $B\subseteq\generatedFilter{\carrier[X]}{B}$
            by \obv{proper_filter_basis_in_generated_filter}.
        Show that for every $U\in\uniformity[X]$ there exists
            $A\in\generatedFilter{\carrier[X]}{B}$ such that
            $A\times A\subseteq U$.
        \begin{subproof}
            Fix $U\in\uniformity[X]$.
            Take $A\in B$ such that $A\times A\subseteq U$.
            Hence $A\in\generatedFilter{\carrier[X]}{B}$
                by \obv{subseteq}.
        \end{subproof}
        $\uniformCauchyFilter{X}{
                \generatedFilter{\carrier[X]}{B}}$ iff
            $\generatedFilter{\carrier[X]}{B}$ is a proper filter on
                $\carrier[X]$ and
            for every $V\in\uniformity[X]$ there exists
                $C\in\generatedFilter{\carrier[X]}{B}$ such that
                $C\times C\subseteq V$ by
            \obv{uniform_cauchy_filter_iff_entourage_squares}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{theorem}\label{complete_uniform_iff_filter_basis_convergence}
    Let $X$ be a uniform space.
    Then $X$ is complete uniform iff for every $B$ such that
        $B$ is a proper filter basis on $\carrier[X]$, if
        for every $U\in\uniformity[X]$ there exists $A\in B$ such that
            $A\times A\subseteq U$, then there exists $x\in\carrier[X]$
        such that for every $U\in\uniformity[X]$ there exists $A\in B$
        such that $A\subseteq\uniformBall{X}{U}{x}$.
\end{theorem}
\begin{proof}
    Show that if $X$ is complete uniform, then for every $B$ such that
        $B$ is a proper filter basis on $\carrier[X]$, if
        for every $U\in\uniformity[X]$ there exists $A\in B$ such that
            $A\times A\subseteq U$, then there exists $x\in\carrier[X]$
        such that for every $U\in\uniformity[X]$ there exists $A\in B$
        such that $A\subseteq\uniformBall{X}{U}{x}$.
    \begin{subproof}
        Assume $X$ is complete uniform.
        Fix $B$.
        Assume $B$ is a proper filter basis on $\carrier[X]$.
        Assume for every $U\in\uniformity[X]$ there exists $A\in B$
            such that $A\times A\subseteq U$.
        $\uniformCauchyFilter{X}
            {\generatedFilter{\carrier[X]}{B}}$
            by \obv{uniform_generated_filter_cauchy_iff}.
        Take $x\in\carrier[X]$ such that
            $\uniformFilterConverges{X}
                {\generatedFilter{\carrier[X]}{B}}{x}$
            by \obv{complete_uniform_space}.
        Follows by \auto{uniform_generated_filter_convergence_iff}.
    \end{subproof}

    Show that if for every $B$ such that
        $B$ is a proper filter basis on $\carrier[X]$, if
        for every $U\in\uniformity[X]$ there exists $A\in B$ such that
            $A\times A\subseteq U$, then there exists $x\in\carrier[X]$
        such that for every $U\in\uniformity[X]$ there exists $A\in B$
        such that $A\subseteq\uniformBall{X}{U}{x}$, then
        $X$ is complete uniform.
    \begin{subproof}
        Assume for every $B$ such that
            $B$ is a proper filter basis on $\carrier[X]$, if
            for every $U\in\uniformity[X]$ there exists $A\in B$ such that
                $A\times A\subseteq U$, then there exists
            $x\in\carrier[X]$ such that for every $U\in\uniformity[X]$
            there exists $A\in B$ such that
                $A\subseteq\uniformBall{X}{U}{x}$.
        Show that for every $F$ such that $\uniformCauchyFilter{X}{F}$
            there exists $x\in\carrier[X]$ such that
                $\uniformFilterConverges{X}{F}{x}$.
        \begin{subproof}
            Fix $F$.
            Assume $\uniformCauchyFilter{X}{F}$.
            $F$ is a proper filter on $\carrier[X]$
                by \obv{uniform_cauchy_filter_is_proper}.
            Hence $F$ is a proper filter basis on $\carrier[X]$
                by \obv{filter_is_proper_filter_basis}.
            For every $U\in\uniformity[X]$ there exists $A\in F$
                such that $A\times A\subseteq U$
                by \obv{uniform_cauchy_filter_entourage_square}.
            Take $x\in\carrier[X]$ such that for every
                $U\in\uniformity[X]$ there exists $A\in F$ such that
                $A\subseteq\uniformBall{X}{U}{x}$.
            $\uniformFilterConverges{X}
                {\generatedFilter{\carrier[X]}{F}}{x}$
                by \obv{uniform_generated_filter_convergence_iff}.
            $\generatedFilter{\carrier[X]}{F}=F$
                by \obv{filter_generates_itself}.
            Hence $\uniformFilterConverges{X}{F}{x}$.
        \end{subproof}
        Follows by \obv{complete_uniform_space}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\subsection{Comparison for sequences}

\begin{theorem}\label{uniform_sequence_convergence_iff_topological}
    Let $X$ be a uniform space.
    Then $\uniformSequenceConverges{X}{s}{x}$ iff
        $\topologicalSequenceConverges{\uniformTopology{X}}{s}{x}$.
\end{theorem}
\begin{proof}
    Show that if $\uniformSequenceConverges{X}{s}{x}$, then
        $\topologicalSequenceConverges{\uniformTopology{X}}{s}{x}$.
    \begin{subproof}
        Assume $\uniformSequenceConverges{X}{s}{x}$.
        Follows by \obv{uniform_sequence_convergence}.
    \end{subproof}
    Show that if
        $\topologicalSequenceConverges{\uniformTopology{X}}{s}{x}$, then
        $\uniformSequenceConverges{X}{s}{x}$.
    \begin{subproof}
        Assume $\topologicalSequenceConverges{\uniformTopology{X}}{s}{x}$.
        Follows by \obv{uniform_sequence_convergence}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{theorem}\label{uniform_sequence_convergence_iff_entourages}
    Let $X$ be a uniform space.
    Then $\uniformSequenceConverges{X}{s}{x}$ iff
        $s\in\funs{\naturals}{\carrier[X]}$ and $x\in\carrier[X]$ and for
            every $U\in\uniformity[X]$ there exists $K\in\naturals$ such
            that for every $n\in\naturals$ such that $K\subseteq n$ we have
            $(x,s(n))\in U$.
\end{theorem}
\begin{proof}
    $\uniformTopology{X}$ is a topological space
        by \obv{uniform_topology_is_topological_space}.
    $\carrier[\uniformTopology{X}]=\carrier[X]$
        by \obv{uniform_topology_carrier}.
    $\opens[\uniformTopology{X}]=\uniformOpens{X}$
        by \obv{uniform_topology_opens}.

    Show that if $\uniformSequenceConverges{X}{s}{x}$, then
        $s\in\funs{\naturals}{\carrier[X]}$ and $x\in\carrier[X]$ and for
            every $U\in\uniformity[X]$ there exists $K\in\naturals$ such
            that for every $n\in\naturals$ such that $K\subseteq n$ we have
            $(x,s(n))\in U$.
    \begin{subproof}
        Assume $\uniformSequenceConverges{X}{s}{x}$.
        $s\in\funs{\naturals}{\carrier[X]}$ by
            \obv{uniform_sequence_convergence_sequence}.
        $x\in\carrier[X]$ by \obv{uniform_sequence_convergence_point}.
        Hence $s\in\funs{\naturals}{\carrier[\uniformTopology{X}]}$ and
            $x\in\carrier[\uniformTopology{X}]$.
        $\topologicalSequenceConverges{\uniformTopology{X}}{s}{x}$
            by \obv{uniform_sequence_convergence}.
        Show that for every $U\in\uniformity[X]$ there exists
            $K\in\naturals$ such that for every $n\in\naturals$
            such that $K\subseteq n$ we have $(x,s(n))\in U$.
        \begin{subproof}
            Fix $U\in\uniformity[X]$.
            Take $A\in\opens[\uniformTopology{X}]$ such that
                $x\in A\subseteq\uniformBall{X}{U}{x}$
                by \obv{uniform_ball_contains_open_neighbourhood}.
            $A\subseteq\carrier[\uniformTopology{X}]$
                by \obv{opens_type}.
            Hence $A\in\pow{\carrier[\uniformTopology{X}]}$
                by \obv{powerset_intro}.
            Hence $A\in\neighbourhoods{x}{\uniformTopology{X}}$
                by \obv{neighbourhoods,subseteq_refl}.
            Take $K\in\naturals$ such that for every $n\in\naturals$
                such that $K\subseteq n$ we have $s(n)\in A$
                by
                \auto{topological_sequence_converges_iff_eventually_in_neighbourhoods}.
            Show that for every $n\in\naturals$ such that $K\subseteq n$
                we have $(x,s(n))\in U$.
            \begin{subproof}
                Fix $n\in\naturals$.
                Assume $K\subseteq n$.
                Hence $s(n)\in A$.
                Hence $s(n)\in\uniformBall{X}{U}{x}$ by \obv{subseteq}.
                Follows by \obv{uniform_ball_iff}.
            \end{subproof}
            Follows by \auto.
        \end{subproof}
    \end{subproof}

    Show that if $s\in\funs{\naturals}{\carrier[X]}$ and
        $x\in\carrier[X]$ and for every $U\in\uniformity[X]$ there exists
        $K\in\naturals$ such that for every $n\in\naturals$ such that
        $K\subseteq n$ we have $(x,s(n))\in U$, then
            $\uniformSequenceConverges{X}{s}{x}$.
    \begin{subproof}
        Assume $s\in\funs{\naturals}{\carrier[X]}$ and
            $x\in\carrier[X]$ and for every $U\in\uniformity[X]$ there
            exists $K\in\naturals$ such that for every $n\in\naturals$
            such that $K\subseteq n$ we have $(x,s(n))\in U$.
        Hence $s\in\funs{\naturals}{\carrier[\uniformTopology{X}]}$ and
            $x\in\carrier[\uniformTopology{X}]$.
        Show that for every
            $N\in\neighbourhoods{x}{\uniformTopology{X}}$ there exists
            $K\in\naturals$ such that for every $n\in\naturals$
            such that $K\subseteq n$ we have $s(n)\in N$.
        \begin{subproof}
            Fix $N\in\neighbourhoods{x}{\uniformTopology{X}}$.
            Take $A\in\opens[\uniformTopology{X}]$ such that
                $x\in A\subseteq N$ by \obv{neighbourhoods}.
            Hence $A\in\uniformOpens{X}$.
            Hence $\uniformOpen{X}{A}$ by \obv{uniform_opens_iff}.
            Take $U\in\uniformity[X]$ such that
                $\uniformBall{X}{U}{x}\subseteq A$
                by \obv{uniform_open}.
            Take $K\in\naturals$ such that for every $n\in\naturals$
                such that $K\subseteq n$ we have $(x,s(n))\in U$.
            Show that for every $n\in\naturals$ such that $K\subseteq n$
                we have $s(n)\in N$.
            \begin{subproof}
                Fix $n\in\naturals$.
                Assume $K\subseteq n$.
                Hence $(x,s(n))\in U$.
                Hence $s(n)\in\uniformBall{X}{U}{x}$
                    by \obv{uniform_ball_iff}.
                Hence $s(n)\in A$ by \obv{subseteq}.
                Follows by \obv{subseteq}.
            \end{subproof}
            Follows by \auto.
        \end{subproof}
        Hence $\topologicalSequenceConverges{\uniformTopology{X}}{s}{x}$ by
            \obv{topological_sequence_converges_iff_eventually_in_neighbourhoods}.
        Follows by \obv{uniform_sequence_convergence}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{proposition}\label{uniform_sequence_convergence_entourage}
    Suppose $\uniformSequenceConverges{X}{s}{x}$ and
        $U\in\uniformity[X]$.
    Then there exists $N\in\naturals$ such that for every
        $n\in\naturals$ such that $N\subseteq n$ we have $(x,s(n))\in U$.
\end{proposition}
\begin{proof}
    $s\in\funs{\naturals}{\carrier[X]}$ by
        \obv{uniform_sequence_convergence_sequence}.
    $x\in\carrier[X]$ by \obv{uniform_sequence_convergence_point}.
    $\uniformSequenceConverges{X}{s}{x}$ iff
        $s\in\funs{\naturals}{\carrier[X]}$ and $x\in\carrier[X]$ and for
            every $V\in\uniformity[X]$ there exists $N\in\naturals$ such
            that for every $n\in\naturals$ such that $N\subseteq n$ we have
            $(x,s(n))\in V$ by
        \obv{uniform_sequence_convergence_iff_entourages}.
    For every $V\in\uniformity[X]$ there exists $N\in\naturals$ such that
        for every $n\in\naturals$ such that $N\subseteq n$ we have
        $(x,s(n))\in V$ by \auto.
\end{proof}
