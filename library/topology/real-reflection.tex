\section{Reflection of the real line}

\import{topology/real-map-linear.tex}
\import{topology/partial-homeomorphism.tex}
\import{surreal/real-structure.tex}

Reflection uses the existing real negation graph. It is its own inverse and
is a homeomorphism of the real line.

\begin{theorem}\label{real_reflection_converse}
    Then $\converse{\surrealRealNegationGraph}=\surrealRealNegationGraph$.
\end{theorem}
\begin{proof}
    Let $n=\surrealRealNegationGraph$.
    $n\in\funs{\reals}{\reals}$ by \obv{surreal_real_negation_graph_type}.
    $n$ is a function and $\dom{n}=\reals$ and $\ran{n}\subseteq\reals$ by
        \obv{funs_is_function,funs,funs_ran}.
    Show that for all $x,y$ such that $(x,y)\in n$ we have $(y,x)\in n$.
    \begin{subproof}
        Fix $x,y$ such that $(x,y)\in n$.
        $x\in\dom{n}$ and $y\in\ran{n}$ by \auto{dom_iff,ran_iff}.
        $x,y\in\reals$ by \auto{subseteq}.
        $n(x)=y$ by \obv{function_apply_intro}.
        $n(x)=\surrealNeg{x}$ and $n(y)=\surrealNeg{y}$ by
            \auto{surreal_real_negation_graph_application}.
        $\surreal{x}$ by \obv{surreal_real_is_surreal}.
        $\surrealNeg{\surrealNeg{x}}=x$ by \obv{surreal_negation_involutive}.
        $n(y)=x$.
        Follows by \auto{function_apply_elim}.
    \end{subproof}
    $\converse{n}$ is a relation by \obv{converse_is_relation}.
For all $x,y$ we have $(x,y)\in\converse{n}$ iff $(x,y)\in n$ by
        \auto{converse_iff}.
    Follows by \auto{relext}.
\end{proof}

\begin{theorem}\label{real_reflection_bijection}
    Then $\surrealRealNegationGraph\in\bijections{\reals}{\reals}$.
\end{theorem}
\begin{proof}
    $\surrealRealNegationGraph\in\funs{\reals}{\reals}$ by
        \obv{surreal_real_negation_graph_type}.
    $\surrealRealNegationGraph$ is a function and
        $\dom{\surrealRealNegationGraph}=\reals$ by \obv{funs_is_function,funs}.
    $\converse{\surrealRealNegationGraph}=\surrealRealNegationGraph$ by
        \obv{real_reflection_converse}.
    $\converse{\surrealRealNegationGraph}$ is injective by
        \obv{converse_of_function_is_injective}.
    $\surrealRealNegationGraph$ is an injection.
    $\ran{\surrealRealNegationGraph}=\reals$ by \auto{dom_converse}.
    Follows by \auto{injective_function_bijection_to_range}.
\end{proof}

\begin{theorem}\label{real_reflection_continuous}
    Then $\surrealRealNegationGraph$ is continuous from $\RealTopology$ to $\RealTopology$.
\end{theorem}
\begin{proof}
    $\RealTopology$ is a topological space by \obv{real_topology_is_topological_space}.
    $\carrier[\RealTopology]=\reals$ by \obv{real_topology_carrier}.
    Let $f=\identity{\reals}$.
    $f\in\funs{\reals}{\reals}$ by \auto{id_elem_funs}.
    $f$ is continuous from $\RealTopology$ to $\RealTopology$ by
        \auto{identity_continuous}.
    Let $g=\realMapNeg{\reals}{f}$.
    $g\in\funs{\reals}{\reals}$ by \auto{real_map_neg_type}.
    $g$ is continuous from $\RealTopology$ to $\RealTopology$ by
        \auto{real_map_neg_continuous}.
    $\surrealRealNegationGraph\in\funs{\reals}{\reals}$ by
        \obv{surreal_real_negation_graph_type}.
    For every $x\in\reals$ we have $g(x)=\surrealNeg{x}$ by
        \auto{real_map_neg_apply,id_apply}.
    For every $x\in\reals$ we have $g(x)=\apply{\surrealRealNegationGraph}{x}$ by
        \auto{surreal_real_negation_graph_application}.
    $g=\surrealRealNegationGraph$ by \obv{typed_functions_equal_from_agreement}.
\end{proof}

\begin{classification}\label{real_reflection_homeomorphism}
    Then $\surrealRealNegationGraph$ is a homeomorphism from $\RealTopology$ to $\RealTopology$.
\end{classification}
\begin{proof}
    $\surrealRealNegationGraph$ is continuous from $\RealTopology$ to $\RealTopology$ by
        \obv{real_reflection_continuous}.
    $\surrealRealNegationGraph\in\bijections{\reals}{\reals}$ by
        \obv{real_reflection_bijection}.
    $\carrier[\RealTopology]=\reals$ by \obv{real_topology_carrier}.
    $\converse{\surrealRealNegationGraph}=\surrealRealNegationGraph$ by
        \obv{real_reflection_converse}.
    Follows by \obv{homeomorphism}.
\end{proof}

\begin{theorem}\label{real_reflection_partial_homeomorphism}
    Then $\surrealRealNegationGraph$ is an open partial homeomorphism
        from $\RealTopology$ to $\RealTopology$.
\end{theorem}
\begin{proof}
    $\surrealRealNegationGraph$ is a homeomorphism from $\RealTopology$ to $\RealTopology$
        by \obv{real_reflection_homeomorphism}.
    Follows by \obv{homeomorphism_is_open_partial_homeomorphism}.
\end{proof}
