\section{Connected subspaces and continuous images}

\import{topology/disconnection.tex}
\import{topology/subspace-continuity.tex}

A subset is connected in an ambient space when its subspace topology is
connected.  Continuous images are handled at the level of spaces first; the
usual statement for the range of a map is then only a corestriction.

\begin{definition}\label{connected_in}
    Let $X$ be a topological space.
    $A$ is connected in $X$ iff
        $A\subseteq\carrier[X]$ and
        $\subspaceTopology{X}{A}$ is connected.
\end{definition}

\begin{proposition}\label{connected_in_subseteq_carrier}
    Let $X$ be a topological space.
    Suppose $A$ is connected in $X$.
    Then $A\subseteq\carrier[X]$.
\end{proposition}
\begin{proof}
    Follows by \obv{connected_in}.
\end{proof}

\begin{theorem}\label{continuous_surjective_image_is_connected}
    Let $X,Y$ be topological spaces.
    Suppose $X$ is connected.
    Suppose $f$ is continuous from $X$ to $Y$ and
        $f\in\surj{\carrier[X]}{\carrier[Y]}$.
    Then $Y$ is connected.
\end{theorem}
\begin{proof}
    $f\in\funs{\carrier[X]}{\carrier[Y]}$ by
        \obv{continuous_map}.
    Hence $f$ is a function from $\carrier[X]$ to $\carrier[Y]$
        by \obv{funs}.
    Hence $f$ is a function by \obv{funs_is_function}.
    Suppose not.
    Hence $Y$ is disconnected by \auto{connected}.
    Take $U,V\in\opens[Y]$ such that
        $\carrier[Y]$ is partitioned by $U$ and $V$ by
        \obv{disconnected}.
    Let $P=\preimg{f}{U}$ and $Q=\preimg{f}{V}$.
    $\preimg{f}{U},\preimg{f}{V}\in\opens[X]$ by
        \obv{continuous_map}.
    Hence $P,Q\in\opens[X]$.
    $U$ is inhabited by \obv{nonempty_inhabited}.
    $V$ is inhabited by \obv{nonempty_inhabited}.

    Show that $P$ is inhabited.
    \begin{subproof}
        Take $u$ such that $u\in U$.
        $U\subseteq\carrier[Y]$ by \obv{opens_type,subseteq}.
        Hence $u\in\carrier[Y]$ by \obv{subseteq}.
        Take $x\in\carrier[X]$ such that $f(x)=u$ by
            \obv{surj}.
        $x\in\dom{f}$ by \auto{funs}.
        Hence $(x,f(x))\in f$ by \obv{function_apply_elim}.
        Thus $x\in\preimg{f}{U}$ by \obv{preimg_iff}.
        Hence $x\in P$.
    \end{subproof}
    Show that $Q$ is inhabited.
    \begin{subproof}
        Take $v$ such that $v\in V$.
        $V\subseteq\carrier[Y]$ by \obv{opens_type,subseteq}.
        Hence $v\in\carrier[Y]$ by \obv{subseteq}.
        Take $x\in\carrier[X]$ such that $f(x)=v$ by
            \obv{surj}.
        $x\in\dom{f}$ by \auto{funs}.
        Hence $(x,f(x))\in f$ by \obv{function_apply_elim}.
        Thus $x\in\preimg{f}{V}$ by \obv{preimg_iff}.
        Hence $x\in Q$.
    \end{subproof}

    $U$ is disjoint from $V$.
    Hence $P$ is disjoint from $Q$ by
        \auto{disjoint,preimg_iff,rightunique}.
    $U\union V=\carrier[Y]$.
    $\preimg{f}{U\union V}
        =\preimg{f}{U}\union\preimg{f}{V}$ by
        \obv{preimg_union}.
    $\preimg{f}{\carrier[Y]}=\carrier[X]$ by
        \obv{preimg_codom_of_function}.
    Hence $P\union Q=\carrier[X]$.
    Thus $\carrier[X]$ is partitioned by $P$ and $Q$ by
        \auto.
    Hence $X$ is disconnected by \auto{disconnected}.
    Contradiction by \obv{connected}.
\end{proof}

\begin{theorem}\label{continuous_range_subspace_is_connected}
    Let $X,Y$ be topological spaces.
    Suppose $X$ is connected and $f$ is continuous from $X$ to $Y$.
    Then $\subspaceTopology{Y}{\ran{f}}$ is connected.
\end{theorem}
\begin{proof}
    Let $Z=\subspaceTopology{Y}{\ran{f}}$.
    $f\in\funs{\carrier[X]}{\carrier[Y]}$ by
        \obv{continuous_map}.
    Hence $\ran{f}\subseteq\carrier[Y]$ by \obv{funs_ran}.
    $\carrier[\subspaceTopology{Y}{\ran{f}}]=\ran{f}$ by
        \obv{subspace_topology_carrier_of_subset}.
    Hence $\carrier[Z]=\ran{f}$.
    Thus $\ran{f}\subseteq\carrier[Z]$ by \obv{subseteq_refl}.
    Hence $f$ is continuous from $X$ to $Z$ by
        \obv{continuous_codomain_corestriction}.
    Hence $f\in\funs{\carrier[X]}{\carrier[Z]}$ by
        \obv{continuous_map}.
    Thus $f\in\surj{\carrier[X]}{\carrier[Z]}$ by
        \auto{funs_surj_iff}.
    $Z$ is connected by \obv{continuous_surjective_image_is_connected}.
    Hence $\subspaceTopology{Y}{\ran{f}}$ is connected.
\end{proof}

\begin{corollary}\label{continuous_image_is_connected}
    Let $X,Y$ be topological spaces.
    Suppose $X$ is connected and $f$ is continuous from $X$ to $Y$.
    Then $\ran{f}$ is connected in $Y$.
\end{corollary}
\begin{proof}
    $f\in\funs{\carrier[X]}{\carrier[Y]}$ by
        \obv{continuous_map}.
    Hence $\ran{f}\subseteq\carrier[Y]$ by \obv{funs_ran}.
    $\subspaceTopology{Y}{\ran{f}}$ is connected by
        \obv{continuous_range_subspace_is_connected}.
    Follows by \obv{connected_in}.
\end{proof}
