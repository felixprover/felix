\section{Complete regularity}

\import{topology/clopen-basis.tex}
\import{topology/continuity.tex}
\import{topology/real-line.tex}
\import{topology/separation.tex}
\import{set/function-space.tex}
\import{surreal/integer.tex}

\begin{definition}\label{is_completely_regular}
    Let $X$ be a topological space.
    $X$ is completely regular iff
        for every $C\in\closeds{X}$ we have for every
            $x\in\carrier[X]\setminus C$
        there exists $f\in\funs{\carrier[X]}{\RealUnitInterval}$ such that
            $f$ is continuous from $X$ to $\RealUnitIntervalTopology$ and
            $f(x)=\surrealZero$ and
            $f$ is constantly $\surrealOne$ on $C$.
\end{definition}

\begin{definition}\label{is_tychonoff}
    Let $X$ be a topological space.
    $X$ is Tychonoff iff
        $X$ is completely regular and $X$ is Hausdorff.
\end{definition}

\begin{theorem}\label{clopen_basis_implies_completely_regular}
    Let $X$ be a topological space.
    Suppose $X$ has a clopen basis.
    Then $X$ is completely regular.
\end{theorem}
\begin{proof}
    Show that for every $C\in\closeds{X}$ we have for every
        $x\in\carrier[X]\setminus C$
        there exists $f\in\funs{\carrier[X]}{\RealUnitInterval}$ such that
            $f$ is continuous from $X$ to $\RealUnitIntervalTopology$ and
            $f(x)=\surrealZero$ and
            $f$ is constantly $\surrealOne$ on $C$.
    \begin{subproof}
        Fix $C\in\closeds{X}$.
        Fix $x\in\carrier[X]\setminus C$.
        Let $U=\carrier[X]\setminus C$.
        $C\subseteq\carrier[X]$ by \obv{closeds,pow_iff,subseteq}.
        $C$ is closed in $X$ by \obv{is_closed_in}.
        Hence $U$ is open in $X$ by \obv{complement_of_closed_is_open}.
        Hence $U\in\opens[X]$ by \obv{is_open}.
        Take $V$ such that $V$ is clopen in $X$ and $x\in V\subseteq U$
            by \obv{has_clopen_basis}.
        $V$ is open in $X$ and $V$ is closed in $X$
            by \obv{is_clopen_in}.
        Hence $V\in\opens[X]$ by \obv{is_open}.
        Hence $V\subseteq\carrier[X]$ by \obv{opens_type}.
        $\surrealOne\neq\surrealZero$
            by \obv{surreal_integer_zero_neq_one}.
        Let $f=\characteristicGraph{\carrier[X]}{V}
            {\surrealOne}{\surrealZero}$.
        $f\in\funs{\carrier[X]}{\{\surrealOne,\surrealZero\}}$
            by \auto{characteristic_graph_type}.
        $\surrealZero,\surrealOne\in\RealUnitInterval$ by
            \obv{surreal_zero_in_real_unit_interval,
                surreal_one_in_real_unit_interval}.
        Hence $\{\surrealOne,\surrealZero\}\subseteq\RealUnitInterval$
            by \auto{cons_iff,singleton_iff,subseteq}.
        Hence $f\in\funs{\carrier[X]}{\RealUnitInterval}$
            by \obv{funs_weaken_codom}.
        Show that $f$ is locally constant from $X$ to
            $\RealUnitIntervalTopology$.
        \begin{subproof}
            $\RealUnitIntervalTopology$ is a topological space
                by \obv{real_unit_interval_topology_is_topological_space}.
            $\carrier[\RealUnitIntervalTopology]=\RealUnitInterval$
                by \obv{real_unit_interval_topology_carrier}.
            Hence $f\in\funs{\carrier[X]}
                {\carrier[\RealUnitIntervalTopology]}$.
            Show that for every $a\in\carrier[X]$ there exists
                $W\in\opens[X]$ such that $a\in W$ and
                for every $b\in W$ we have $f(b)=f(a)$.
            \begin{subproof}
                Fix $a\in\carrier[X]$.
                \begin{byCase}
                    \caseOf{$a\in V$.}
                        $f(a)=\surrealZero$
                            by \auto{characteristic_graph_one}.
                        For every $b\in V$ we have $f(b)=\surrealZero$
                            by \auto{characteristic_graph_one}.
                        Hence for every $b\in V$ we have $f(b)=f(a)$.
                    \caseOf{$a\notin V$.}
                        Let $W=\carrier[X]\setminus V$.
                        $W$ is open in $X$
                            by \auto{complement_of_closed_is_open}.
                        Hence $W\in\opens[X]$ by \obv{is_open}.
                        $a\in W$ by \obv{setminus}.
                        $f(a)=\surrealOne$
                            by \auto{characteristic_graph_zero}.
                        For every $b\in W$ we have $f(b)=\surrealOne$
                            by \auto{characteristic_graph_zero}.
                        Hence for every $b\in W$ we have $f(b)=f(a)$.
                \end{byCase}
            \end{subproof}
            Follows by \obv{locally_constant_map}.
        \end{subproof}
        Hence $f$ is continuous from $X$ to $\RealUnitIntervalTopology$
            by \obv{locally_constant_map_is_continuous}.
        $x\in V$.
        Hence $f(x)=\surrealZero$ by \auto{characteristic_graph_one}.
        Show that $f$ is constantly $\surrealOne$ on $C$.
        \begin{subproof}
            Fix $y\in C$.
            $y\in\carrier[X]$ by \obv{subseteq}.
            $y\notin U$ by \auto{setminus}.
            Hence $y\notin V$ by \obv{subseteq}.
            Hence $y\in\carrier[X]\setminus V$ by \obv{setminus}.
            Follows by \auto{characteristic_graph_zero}.
        \end{subproof}
    \end{subproof}
    Follows by \obv{is_completely_regular}.
\end{proof}
