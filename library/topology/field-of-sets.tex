\begin{theorem}\label{field_of_sets_is_topological_basis}
    Suppose $F$ is a field of sets on $X$.
    Then $F$ is a topological basis for $X$.
\end{theorem}
\begin{proof}
    $F\subseteq\pow{X}$ by \obv{field_of_sets_subseteq_powerset}.
    Hence $F$ is a family of subsets of $X$ by
        \auto{pow_iff,subseteq}.
    $X\in F$ by \obv{field_of_sets_whole}.
    Hence $F$ covers $X$ by \obv{subseteq_refl}.
    Show that for all $A,C,x$ such that $A,C\in F$ and
        $x\in A,C$ there exists $V\in F$ such that
            $x\in V\subseteq A,C$.
    \begin{subproof}
        Fix $A,C,x$.
        Assume $A,C\in F$ and $x\in A,C$.
        $A\inter C\in F$ by \obv{field_of_sets_inter}.
        $x\in A\inter C$ by \obv{inter_intro}.
        $A\inter C\subseteq A,C$ by
            \obv{inter_lower_left,inter_lower_right}.
    \end{subproof}
    Follows by \obv{topological_basis}.
\end{proof}

\begin{classification}\label{field_generated_topology_is_topological_space}
    Suppose $F$ is a field of sets on $X$.
    Then $\topologyGeneratedByBasis{F}{X}$ is a topological space.
\end{classification}
\begin{proof}
    $F$ is a topological basis for $X$ by
        \obv{field_of_sets_is_topological_basis}.
    Follows by
        \obv{topology_generated_by_basis_is_topological_space}.
\end{proof}

\begin{theorem}\label{field_of_sets_is_open_basis_of_generated_topology}
    Suppose $F$ is a field of sets on $X$.
    Then $F$ is an open basis of $\topologyGeneratedByBasis{F}{X}$.
\end{theorem}
\begin{proof}
    $F$ is a topological basis for $X$ by
        \obv{field_of_sets_is_topological_basis}.
    Follows by \obv{topology_generated_by_basis_has_open_basis}.
\end{proof}

\begin{proposition}\label{field_of_sets_member_open_in_generated_topology}
    Suppose $F$ is a field of sets on $X$.
    Let $A\in F$.
    Then $A$ is open in $\topologyGeneratedByBasis{F}{X}$.
\end{proposition}
\begin{proof}
    Let $T=\topologyGeneratedByBasis{F}{X}$.
    $F$ is an open basis of $T$ by
        \obv{field_of_sets_is_open_basis_of_generated_topology}.
    Hence $F\subseteq\opens[T]$ by \obv{open_basis}.
    Thus $A\in\opens[T]$ by \obv{subseteq}.
    Follows by \obv{is_open}.
\end{proof}

\begin{proposition}\label{field_of_sets_member_clopen_in_generated_topology}
    Suppose $F$ is a field of sets on $X$.
    Let $A\in F$.
    Then $A$ is clopen in $\topologyGeneratedByBasis{F}{X}$.
\end{proposition}
\begin{proof}
    Let $T=\topologyGeneratedByBasis{F}{X}$.
    $\carrier[T]=X$ by
        \auto{topology_generated_by_basis_carrier}.
    $A$ is open in $T$ by
        \obv{field_of_sets_member_open_in_generated_topology}.
    $X\setminus A\in F$ by \obv{field_of_sets_complement}.
    Hence $X\setminus A$ is open in $T$ by
        \obv{field_of_sets_member_open_in_generated_topology}.
    $A\subseteq X$ by
        \auto{field_of_sets_subseteq_powerset,subseteq,pow_iff}.
    Hence $A\in\pow{\carrier[T]}$ by
        \obv{pow_iff,subseteq}.
    Thus $A\in\closeds{T}$ by
        \obv{closeds,is_open}.
    Hence $A$ is closed in $T$ by \obv{is_closed_in}.
    Follows by \obv{is_clopen_in}.
\end{proof}

\begin{theorem}\label{field_of_sets_is_clopen_open_basis}
    Suppose $F$ is a field of sets on $X$.
    Then $F$ is an open basis of $\topologyGeneratedByBasis{F}{X}$ and
        for every $A\in F$ we have
            $A$ is clopen in $\topologyGeneratedByBasis{F}{X}$.
\end{theorem}
\begin{proof}
    Follows by
        \obv{field_of_sets_is_open_basis_of_generated_topology,
            field_of_sets_member_clopen_in_generated_topology}.
\end{proof}

\begin{theorem}\label{field_generated_topology_has_clopen_basis}
    Suppose $F$ is a field of sets on $X$.
    Then $\topologyGeneratedByBasis{F}{X}$ has a clopen basis.
\end{theorem}
\begin{proof}
    Let $T=\topologyGeneratedByBasis{F}{X}$.
    $T$ is a topological space by
        \obv{field_generated_topology_is_topological_space}.
    $F$ is an open basis of $T$ and for every $A\in F$ we have
        $A$ is clopen in $T$ by
        \auto{field_of_sets_is_clopen_open_basis}.
    Follows by \auto{clopen_open_basis_implies_has_clopen_basis}.
\end{proof}

\begin{definition}\label{field_separates_points}
    $\fieldSeparatesPoints{F}{X}$ iff
        for all $x,y\in X$ such that $x\neq y$ there exists $A\in F$
        such that $(x\in A\land y\notin A)\lor
            (y\in A\land x\notin A)$.
\end{definition}

\begin{proposition}\label{field_separates_points_iff}
    $\fieldSeparatesPoints{F}{X}$ iff
        for all $x,y\in X$ such that $x\neq y$ there exists $A\in F$
        such that $(x\in A\land y\notin A)\lor
            (y\in A\land x\notin A)$.
\end{proposition}
\begin{proof}
    Follows by \obv{field_separates_points}.
\end{proof}

\begin{theorem}\label{point_separating_field_generated_topology_is_hausdorff}
    Suppose $F$ is a field of sets on $X$ and
        $\fieldSeparatesPoints{F}{X}$.
    Then $\topologyGeneratedByBasis{F}{X}$ is Hausdorff.
\end{theorem}
\begin{proof}
    Let $T=\topologyGeneratedByBasis{F}{X}$.
    $T$ is a topological space by
        \obv{field_generated_topology_is_topological_space}.
    $\carrier[T]=X$ by
        \auto{topology_generated_by_basis_carrier}.
    Show that for all $x,y\in\carrier[T]$ such that $x\neq y$
        there exist $U,V\in\opens[T]$ such that $x\in U$ and
            $y\in V$ and $U$ is disjoint from $V$.
    \begin{subproof}
        Fix $x,y\in\carrier[T]$.
        Assume $x\neq y$.
        Hence $x,y\in X$.
        Take $A\in F$ such that
            $(x\in A\land y\notin A)\lor
            (y\in A\land x\notin A)$ by
            \obv{field_separates_points_iff}.
        $X\setminus A\in F$ by \obv{field_of_sets_complement}.
        $A$ is open in $T$ and $X\setminus A$ is open in $T$ by
            \obv{field_of_sets_member_open_in_generated_topology}.
        Hence $A,X\setminus A\in\opens[T]$ by \obv{is_open}.
        $A$ is disjoint from $X\setminus A$ by
            \obv{disjoint,setminus}.
        \begin{byCase}
        \caseOf{$x\in A$ and $y\notin A$.}
            Hence $y\in X\setminus A$ by \obv{setminus}.
            Follows by \auto.
        \caseOf{$x\notin A$ or $y\in A$.}
            Hence $y\in A$ and $x\notin A$.
            Thus $x\in X\setminus A$ by \obv{setminus}.
            $X\setminus A$ is disjoint from $A$ by
                \obv{disjoint_symmetric}.
            Follows by \auto.
        \end{byCase}
    \end{subproof}
    Follows by \auto{is_hausdorff}.
\end{proof}

\begin{theorem}\label{compact_field_generated_topology_clopens}
    Suppose $F$ is a field of sets on $X$.
    Let $T=\topologyGeneratedByBasis{F}{X}$.
    Suppose $T$ is compact.
    Then $\clopens{T}=F$.
\end{theorem}
\begin{proof}
    $T$ is a topological space by
        \obv{field_generated_topology_is_topological_space}.
    $\carrier[T]=X$ by
        \auto{topology_generated_by_basis_carrier}.
    Show that $F\subseteq\clopens{T}$.
    \begin{subproof}
        It suffices to show that every element of $F$ is an element of
            $\clopens{T}$ by \obv{subseteq}.
        Fix $A\in F$.
        $A$ is clopen in $T$ by
            \obv{field_of_sets_member_clopen_in_generated_topology}.
        Follows by \obv{clopens_iff}.
    \end{subproof}

    Show that $\clopens{T}\subseteq F$.
    \begin{subproof}
        It suffices to show that every element of $\clopens{T}$ is an
            element of $F$ by \obv{subseteq}.
        Fix $C\in\clopens{T}$.
        Hence $C$ is clopen in $T$ by \obv{clopens_iff}.
        Thus $C$ is open in $T$ and $C$ is closed in $T$ by
            \obv{is_clopen_in}.
        Hence $C\in\opens[T]$ by \obv{is_open}.
        $C\subseteq\carrier[T]$ by
            \obv{closed_in_subseteq_carrier}.
        Hence $C\subseteq X$.

        Let $K=\{V\in F\mid V\subseteq C\}$.
        $K\subseteq F$ by \obv{subseteq}.
        Show that $K$ covers $C$.
        \begin{subproof}
            Show that for every $x\in C$ there exists $V\in K$ such
                that $x\in V$.
            \begin{subproof}
                Fix $x\in C$.
                $F$ is an open basis of $T$ by
                    \obv{field_of_sets_is_open_basis_of_generated_topology}.
                Take $V\in F$ such that $x\in V\subseteq C$ by
                    \obv{open_basis}.
                Hence $V\in K$.
            \end{subproof}
        \end{subproof}

        Let $O=X\setminus C$.
        $O=\carrier[T]\setminus C$.
        Hence $O$ is open in $T$ by
            \auto{complement_of_closed_is_open}.
        Thus $O\in\opens[T]$ by \obv{is_open}.
        Let $D=\cons{O}{K}$.
        Show that $D\subseteq\opens[T]$.
        \begin{subproof}
            It suffices to show that every element of $D$ is an element
                of $\opens[T]$ by \obv{subseteq}.
            Fix $V\in D$.
            $V=O$ or $V\in K$ by \obv{cons_iff}.
            If $V=O$, then $V\in\opens[T]$.
            If $V\in K$, then $V\in F$ by \obv{subseteq}.
            If $V\in F$, then $V$ is open in $T$ by
                \obv{field_of_sets_member_open_in_generated_topology}.
            If $V\in F$, then $V\in\opens[T]$ by \auto{is_open}.
        \end{subproof}
        Show that $D$ covers $\carrier[T]$.
        \begin{subproof}
            Show that for every $x\in\carrier[T]$ there exists $V\in D$
                such that $x\in V$.
            \begin{subproof}
                Fix $x\in\carrier[T]$.
                Hence $x\in X$.
                \begin{byCase}
                \caseOf{$x\in C$.}
                    Take $V\in K$ such that $x\in V$.
                    $V\in D$ by \auto{cons_iff}.
                \caseOf{$x\notin C$.}
                    Hence $x\in O$ by \obv{setminus}.
                    $O\in D$ by \obv{cons_left}.
                \end{byCase}
            \end{subproof}
        \end{subproof}
        Hence $D$ is an open cover of $T$ by \obv{open_cover}.

        Take $G$ such that $G$ is finite and $G\subseteq D$ and
            $G$ covers $\carrier[T]$ by \auto{is_compact}.
        Let $H=G\inter K$.
        $H$ is finite by \obv{finite_intersection}.
        $H\subseteq K$ by \auto{inter_lower_right}.
        Hence $H\subseteq F$ by \obv{subseteq_transitive}.

        Show that $\unions{H}\subseteq C$.
        \begin{subproof}
            It suffices to show that every element of $\unions{H}$ is an
                element of $C$ by \obv{subseteq}.
            Fix $x\in\unions{H}$.
            Take $V\in H$ such that $x\in V$ by \obv{unions_iff}.
            Hence $V\in K$ by \obv{subseteq}.
            Thus $V\subseteq C$.
            Follows by \obv{subseteq}.
        \end{subproof}
        Show that $C\subseteq\unions{H}$.
        \begin{subproof}
            It suffices to show that every element of $C$ is an element
                of $\unions{H}$ by \obv{subseteq}.
            Fix $x\in C$.
            Hence $x\in\carrier[T]$ by \obv{subseteq}.
            Take $V\in G$ such that $x\in V$.
            Hence $V\in D$ by \obv{subseteq}.
            $V=O$ or $V\in K$ by \obv{cons_iff}.
            $x\notin O$ by \auto{setminus}.
            Hence $V\in K$.
            Thus $V\in H$ by \obv{inter_intro}.
            Follows by \auto{unions_iff}.
        \end{subproof}
        Hence $C=\unions{H}$ by \obv{subseteq_antisymmetric}.
        $\emptyset\in F$ by \obv{field_of_sets_emptyset}.
        For all $A,V\in F$ we have $A\union V\in F$ by
            \obv{field_of_sets_union}.
        $\unions{H}\in F$ by
            \obv{finite_family_union_in_binary_union_closed_family}.
    \end{subproof}
    Follows by \obv{subseteq_antisymmetric}.
\end{proof}
