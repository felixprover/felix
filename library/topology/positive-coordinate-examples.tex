\section{Positive and full coordinates on the real line}

\import{topology/positive-real-pseudogroup.tex}

Reflection reverses order in every neighbourhood. It is an allowed change in
the full pseudogroup but not in the positive one.

\subsection{Reflection reverses local order}

\begin{theorem}\label{real_reflection_not_positive}
    Then $\surrealRealNegationGraph$ is not a positive real coordinate change.
\end{theorem}
\begin{proof}
    Suppose $\surrealRealNegationGraph$ is a positive real coordinate change.
    Let $n=\surrealRealNegationGraph$.
    $n\in\funs{\reals}{\reals}$ by \obv{surreal_real_negation_graph_type}.
    $n$ is a function and $\dom{n}=\reals$ by \obv{funs_is_function,funs}.
    $\surrealZero,\surrealOne\in\reals$ by
        \obv{surreal_zero_is_real,surreal_one_is_real}.
    Take $U\in\opens[\RealTopology]$ such that $\surrealZero\in U$ and
        $\restrl{n}{U}$ is an increasing real map by \auto{positive_real_change}.
    $\opens[\RealTopology]=\metricOpens{\RealMetric}$ by
        \auto{real_topology,metric_topology_opens}.
    $\metricOpen{\RealMetric}{U}$ by \auto{metric_opens_iff}.
    Take $r\in\reals$ such that $\surrealLt{\surrealZero}{r}$ and
        $\metricBall{\RealMetric}{\surrealZero}{r}\subseteq U$ by \obv{metric_open}.
    $\surrealLt{\surrealZero}{\surrealOne}$ by \obv{surreal_zero_lt_one}.
    Take $t\in\reals$ such that $\surrealLt{\surrealZero}{t}$ and
        $\surrealLt{t}{\surrealOne}$ and
        $t\in\metricBall{\RealMetric}{\surrealZero}{r}$ by
        \obv{real_ball_meets_right_open_interval}.
    $t\in U$ by \obv{subseteq}.
    Let $k=\restrl{n}{U}$.
    $\dom{k}=\dom{n}\inter U$ by \auto{dom_of_restrl_eq_inter}.
    $\surrealZero,t\in\dom{k}$ by \auto{inter}.
    $k(\surrealZero)=n(\surrealZero)$ and $k(t)=n(t)$ by
        \auto{restriction_apply_on_domain}.
    $n(\surrealZero)=\surrealNeg{\surrealZero}$ and $n(t)=\surrealNeg{t}$ by
        \auto{surreal_real_negation_graph_application}.
    $\surrealLt{k(\surrealZero)}{k(t)}$ by \auto{increasing_real_map}.
    $\surreal{\surrealZero}$ and $\surreal{t}$ by \obv{surreal_real_is_surreal}.
    $\surrealLt{t}{\surrealZero}$ by \auto{surreal_negation_lt_iff}.
    Contradiction by \obv{surreal_lt_asymmetric}.
\end{proof}

\begin{theorem}\label{real_reflection_distinguishes_pseudogroups}
    Then $\surrealRealNegationGraph\in\changes[\FullPseudogroup{\RealTopology}]$ and
        $\surrealRealNegationGraph\notin\changes[\PositiveRealPseudogroup]$.
\end{theorem}
\begin{proof}
    $\RealTopology$ is a topological space by \obv{real_topology_is_topological_space}.
    $\surrealRealNegationGraph$ is an open partial homeomorphism from $\RealTopology$
        to $\RealTopology$ by \obv{real_reflection_partial_homeomorphism}.
    $\surrealRealNegationGraph\in\changes[\FullPseudogroup{\RealTopology}]$ by
        \auto{full_pseudogroup_changes}.
    $\surrealRealNegationGraph$ is not a positive real coordinate change by
        \obv{real_reflection_not_positive}.
    Follows by \auto{positive_real_pseudogroup_changes}.
\end{proof}

\subsection{Two positive singleton atlases}

A singleton chart is compatible with itself because its self-transition is
identity. Thus reflection defines a positive atlas even though reflection
itself is not a positive coordinate change.

\begin{definition}\label{real_reflection_chart}
    $\RealReflectionChart=\{\surrealRealNegationGraph\}$.
\end{definition}

\begin{classification}\label{real_identity_positive_atlas}
    Then $\RealOneChart$ is a coordinate atlas on $\RealTopology$ under
        $\PositiveRealPseudogroup$.
\end{classification}
\begin{proof}
    $\RealTopology$ is a topological space by \obv{real_topology_is_topological_space}.
    $\PositiveRealPseudogroup$ is a coordinate pseudogroup by \obv{positive_real_pseudogroup}.
    $\model[\PositiveRealPseudogroup]=\RealTopology$ by
        \obv{positive_real_pseudogroup_model}.
    $\carrier[\RealTopology]=\reals$ by \obv{real_topology_carrier}.
    $\reals\in\opens[\RealTopology]$ by \obv{carrier_open}.
    $\identity{\reals}$ is an open partial homeomorphism from $\RealTopology$ to
        $\RealTopology$ by \obv{identity_open_partial_homeomorphism}.
    $\dom{\identity{\reals}}=\reals$ by \obv{id_dom}.
    Follows by \auto{real_one_chart,single_chart_atlas}.
\end{proof}

\begin{classification}\label{real_reflection_positive_atlas}
    Then $\RealReflectionChart$ is a coordinate atlas on $\RealTopology$ under
        $\PositiveRealPseudogroup$.
\end{classification}
\begin{proof}
    $\RealTopology$ is a topological space by \obv{real_topology_is_topological_space}.
    $\PositiveRealPseudogroup$ is a coordinate pseudogroup by \obv{positive_real_pseudogroup}.
    $\model[\PositiveRealPseudogroup]=\RealTopology$ by
        \obv{positive_real_pseudogroup_model}.
    $\carrier[\RealTopology]=\reals$ by \obv{real_topology_carrier}.
    $\surrealRealNegationGraph$ is an open partial homeomorphism from $\RealTopology$
        to $\RealTopology$ by \obv{real_reflection_partial_homeomorphism}.
    $\surrealRealNegationGraph\in\funs{\reals}{\reals}$ by
        \obv{surreal_real_negation_graph_type}.
    $\dom{\surrealRealNegationGraph}=\reals$ by \obv{funs}.
    Follows by \auto{real_reflection_chart,single_chart_atlas}.
\end{proof}

\begin{theorem}\label{real_identity_reflection_transitions}
    Then $\chartTransition{\surrealRealNegationGraph}{\identity{\reals}}
        =\surrealRealNegationGraph$ and
        $\chartTransition{\identity{\reals}}{\surrealRealNegationGraph}
        =\surrealRealNegationGraph$.
\end{theorem}
\begin{proof}
    $\surrealRealNegationGraph\in\funs{\reals}{\reals}$ by
        \obv{surreal_real_negation_graph_type}.
    $\converse{\identity{\reals}}=\identity{\reals}$ by \obv{converse_identity}.
    $\converse{\surrealRealNegationGraph}=\surrealRealNegationGraph$ by
        \obv{real_reflection_converse}.
    $\surrealRealNegationGraph\circ\identity{\reals}=\surrealRealNegationGraph$ and
        $\identity{\reals}\circ\surrealRealNegationGraph=\surrealRealNegationGraph$ by
        \obv{funs_circ_identity_right,funs_identity_circ_left}.
    Follows by \obv{chart_transition}.
\end{proof}

\subsection{Distinct positive structures}

Reflection transports the identity presentation to the reflection presentation.
The resulting positive structure will be distinct, although the two structures
are related by this homeomorphism.

\begin{theorem}\label{real_reflection_transported_atlas}
    Then $\TransportedAtlas{\RealOneChart}{\surrealRealNegationGraph}=\RealReflectionChart$.
\end{theorem}
\begin{proof}
    $\identity{\reals}\circ\converse{\surrealRealNegationGraph}=\surrealRealNegationGraph$ by
        \obv{chart_transition,real_identity_reflection_transitions}.
    For every $b$ we have $b\in\TransportedAtlas{\RealOneChart}{\surrealRealNegationGraph}$ iff
        $b\in\RealReflectionChart$ by
        \auto{transported_atlas_iff,real_one_chart,real_reflection_chart,singleton_iff}.
    Follows by set extensionality.
\end{proof}

\begin{theorem}\label{real_identity_reflection_incompatible}
    Then $\identity{\reals}$ is not compatible with $\surrealRealNegationGraph$
        on $\RealTopology$ under $\PositiveRealPseudogroup$.
\end{theorem}
\begin{proof}
    $\chartTransition{\surrealRealNegationGraph}{\identity{\reals}}
        =\surrealRealNegationGraph$ by \obv{real_identity_reflection_transitions}.
    $\surrealRealNegationGraph\notin\changes[\PositiveRealPseudogroup]$ by
        \obv{real_reflection_distinguishes_pseudogroups}.
    Follows by \obv{coordinate_chart_compatibility}.
\end{proof}

\begin{theorem}\label{real_positive_max_atlases_differ}
    Then $\MaxAtlas{\PositiveRealPseudogroup}{\RealTopology}{\RealOneChart}
        \neq\MaxAtlas{\PositiveRealPseudogroup}{\RealTopology}{\RealReflectionChart}$.
\end{theorem}
\begin{proof}
    $\RealTopology$ is a topological space by \obv{real_topology_is_topological_space}.
    $\PositiveRealPseudogroup$ is a coordinate pseudogroup by \obv{positive_real_pseudogroup}.
    $\RealOneChart$ is a coordinate atlas on $\RealTopology$ under $\PositiveRealPseudogroup$
        by \obv{real_identity_positive_atlas}.
    $\RealReflectionChart$ is a coordinate atlas on $\RealTopology$ under $\PositiveRealPseudogroup$
        by \obv{real_reflection_positive_atlas}.
    $\identity{\reals}\in\RealOneChart$ and $\surrealRealNegationGraph\in\RealReflectionChart$
        by \auto{real_one_chart,real_reflection_chart,singleton_iff}.
    $\identity{\reals}$ is not compatible with $\surrealRealNegationGraph$ on
        $\RealTopology$ under $\PositiveRealPseudogroup$ by
        \obv{real_identity_reflection_incompatible}.
    Follows by \auto{max_atlas_equality_iff}.
\end{proof}

\begin{theorem}\label{real_positive_coordinate_structures_differ}
    Then $\GeneratedCoordinates{\PositiveRealPseudogroup}{\RealTopology}{\RealOneChart}
        \neq\GeneratedCoordinates{\PositiveRealPseudogroup}{\RealTopology}{\RealReflectionChart}$.
\end{theorem}
\begin{proof}
    Let $G=\PositiveRealPseudogroup$ and $X=\RealTopology$.
    $\RealOneChart$ is a coordinate atlas on $X$ under $G$ by
        \obv{real_identity_positive_atlas}.
    $\RealReflectionChart$ is a coordinate atlas on $X$ under $G$ by
        \obv{real_reflection_positive_atlas}.
    $\atlas[\GeneratedCoordinates{G}{X}{\RealOneChart}]=\MaxAtlas{G}{X}{\RealOneChart}$ and
        $\atlas[\GeneratedCoordinates{G}{X}{\RealReflectionChart}]=\MaxAtlas{G}{X}{\RealReflectionChart}$
        by \obv{generated_coordinatesprojection_atlas}.
    $\MaxAtlas{G}{X}{\RealOneChart}\neq\MaxAtlas{G}{X}{\RealReflectionChart}$ by
        \obv{real_positive_max_atlases_differ}.
\end{proof}

\begin{theorem}\label{real_reflection_transports_positive_coordinates}
    Then $\TransportedCoordinates{\PositiveRealPseudogroup}{\RealTopology}{\RealTopology}
            {\GeneratedCoordinates{\PositiveRealPseudogroup}{\RealTopology}{\RealOneChart}}{\surrealRealNegationGraph}
        =\GeneratedCoordinates{\PositiveRealPseudogroup}{\RealTopology}{\RealReflectionChart}$.
\end{theorem}
\begin{proof}
    Let $G=\PositiveRealPseudogroup$ and $X=\RealTopology$.
    $G$ is a coordinate pseudogroup by \obv{positive_real_pseudogroup}.
    $X$ is a topological space by \obv{real_topology_is_topological_space}.
    $\RealOneChart$ is a coordinate atlas on $X$ under $G$ by \obv{real_identity_positive_atlas}.
    Let $S=\GeneratedCoordinates{G}{X}{\RealOneChart}$.
    $\RealOneChart$ presents $S$ on $X$ under $G$ by \auto{generated_coordinates_presentation}.
    $\surrealRealNegationGraph$ is a homeomorphism from $X$ to $X$ by \obv{real_reflection_homeomorphism}.
    $\TransportedCoordinates{G}{X}{X}{S}{\surrealRealNegationGraph}
        =\GeneratedCoordinates{G}{X}{\TransportedAtlas{\RealOneChart}{\surrealRealNegationGraph}}$ by
        \obv{transported_coordinates_from_presentation}.
    Follows by \obv{real_reflection_transported_atlas}.
\end{proof}

\begin{theorem}\label{real_reflection_transported_admissible}
    Let $C$ be a local-map calculus.
    Suppose $\PositiveRealPseudogroup$ is compatible with the calculus $C$.
    Then $\surrealRealNegationGraph$ is coordinate-admissible
        from $\Equip{\RealTopology}{\GeneratedCoordinates{\PositiveRealPseudogroup}{\RealTopology}{\RealOneChart}}$
        under $\PositiveRealPseudogroup$
        to $\Equip{\RealTopology}{\GeneratedCoordinates{\PositiveRealPseudogroup}{\RealTopology}{\RealReflectionChart}}$
        under $\PositiveRealPseudogroup$ in $C$ and
        $\surrealRealNegationGraph$ is coordinate-admissible
        from $\Equip{\RealTopology}{\GeneratedCoordinates{\PositiveRealPseudogroup}{\RealTopology}{\RealReflectionChart}}$
        under $\PositiveRealPseudogroup$
        to $\Equip{\RealTopology}{\GeneratedCoordinates{\PositiveRealPseudogroup}{\RealTopology}{\RealOneChart}}$
        under $\PositiveRealPseudogroup$ in $C$.
\end{theorem}
\begin{proof}
    Let $G=\PositiveRealPseudogroup$ and $X=\RealTopology$.
    $G$ is a coordinate pseudogroup by \obv{positive_real_pseudogroup}.
    $X$ is a topological space by \obv{real_topology_is_topological_space}.
    $\RealOneChart$ is a coordinate atlas on $X$ under $G$ by \obv{real_identity_positive_atlas}.
    Let $S=\GeneratedCoordinates{G}{X}{\RealOneChart}$.
    $S$ is a relative coordinate structure on $X$ under $G$ by \auto{generated_coordinates}.
    $\surrealRealNegationGraph$ is a homeomorphism from $X$ to $X$ by \obv{real_reflection_homeomorphism}.
    Let $T=\TransportedCoordinates{G}{X}{X}{S}{\surrealRealNegationGraph}$.
    $\surrealRealNegationGraph$ is coordinate-admissible from $\Equip{X}{S}$ under $G$
        to $\Equip{X}{T}$ under $G$ in $C$ by \auto{transported_coordinates_admissible}.
    $\converse{\surrealRealNegationGraph}$ is coordinate-admissible from $\Equip{X}{T}$ under $G$
        to $\Equip{X}{S}$ under $G$ in $C$ by \auto{transported_coordinates_inverse_admissible}.
    $\converse{\surrealRealNegationGraph}=\surrealRealNegationGraph$ by \obv{real_reflection_converse}.
    $T=\GeneratedCoordinates{G}{X}{\RealReflectionChart}$ by
        \obv{real_reflection_transports_positive_coordinates}.
\end{proof}

\subsection{The full pseudogroup forgets the distinction}

We now generate under the larger pseudogroup, starting from the same two
singleton presentations. This enlarges the maximal atlases; it does not merely
give a positive maximal atlas a new label.

\begin{classification}\label{real_identity_full_atlas}
    Then $\RealOneChart$ is a coordinate atlas on $\RealTopology$ under
        $\FullPseudogroup{\RealTopology}$.
\end{classification}
\begin{proof}
    $\RealTopology$ is a topological space by \obv{real_topology_is_topological_space}.
    $\PositiveRealPseudogroup$ is a coordinate pseudogroup by \obv{positive_real_pseudogroup}.
    $\PositiveRealPseudogroup$ is a subpseudogroup of $\FullPseudogroup{\RealTopology}$ by
        \obv{positive_real_pseudogroup_in_full}.
    $\RealOneChart$ is a coordinate atlas on $\RealTopology$ under $\PositiveRealPseudogroup$
        by \obv{real_identity_positive_atlas}.
    Follows by \obv{subpseudogroup_atlas}.
\end{proof}

\begin{classification}\label{real_reflection_full_atlas}
    Then $\RealReflectionChart$ is a coordinate atlas on $\RealTopology$ under
        $\FullPseudogroup{\RealTopology}$.
\end{classification}
\begin{proof}
    $\RealTopology$ is a topological space by \obv{real_topology_is_topological_space}.
    $\PositiveRealPseudogroup$ is a coordinate pseudogroup by \obv{positive_real_pseudogroup}.
    $\PositiveRealPseudogroup$ is a subpseudogroup of $\FullPseudogroup{\RealTopology}$ by
        \obv{positive_real_pseudogroup_in_full}.
    $\RealReflectionChart$ is a coordinate atlas on $\RealTopology$ under $\PositiveRealPseudogroup$
        by \obv{real_reflection_positive_atlas}.
    Follows by \obv{subpseudogroup_atlas}.
\end{proof}

\begin{theorem}\label{real_full_presentations_compatible}
    Then for every $a\in\RealOneChart$ we have for every $b\in\RealReflectionChart$
        we have $a$ is compatible with $b$ on $\RealTopology$ under
        $\FullPseudogroup{\RealTopology}$.
\end{theorem}
\begin{proof}
    $\RealOneChart$ is a coordinate atlas on $\RealTopology$ under $\FullPseudogroup{\RealTopology}$
        by \obv{real_identity_full_atlas}.
    $\RealReflectionChart$ is a coordinate atlas on $\RealTopology$ under $\FullPseudogroup{\RealTopology}$
        by \obv{real_reflection_full_atlas}.
    Fix $a\in\RealOneChart$ and $b\in\RealReflectionChart$.
    $a$ and $b$ are open partial homeomorphisms from $\RealTopology$ to
        $\model[\FullPseudogroup{\RealTopology}]$ by \obv{coordinate_atlas}.
    $a=\identity{\reals}$ and $b=\surrealRealNegationGraph$ by
        \auto{real_one_chart,real_reflection_chart,singleton_iff}.
    $\chartTransition{b}{a}=\surrealRealNegationGraph$ by
        \obv{real_identity_reflection_transitions}.
    $\surrealRealNegationGraph\in\changes[\FullPseudogroup{\RealTopology}]$ by
        \obv{real_reflection_distinguishes_pseudogroups}.
    Follows by \obv{coordinate_chart_compatibility}.
\end{proof}

\begin{theorem}\label{real_full_coordinate_presentations_equal}
    Then $\GeneratedCoordinates{\FullPseudogroup{\RealTopology}}{\RealTopology}{\RealOneChart}
        =\GeneratedCoordinates{\FullPseudogroup{\RealTopology}}{\RealTopology}{\RealReflectionChart}$ and
        $\Equip{\RealTopology}{\GeneratedCoordinates{\FullPseudogroup{\RealTopology}}{\RealTopology}{\RealOneChart}}
        =\Equip{\RealTopology}{\GeneratedCoordinates{\FullPseudogroup{\RealTopology}}{\RealTopology}{\RealReflectionChart}}$.
\end{theorem}
\begin{proof}
    $\RealOneChart$ is a coordinate atlas on $\RealTopology$ under $\FullPseudogroup{\RealTopology}$
        by \obv{real_identity_full_atlas}.
    $\RealReflectionChart$ is a coordinate atlas on $\RealTopology$ under $\FullPseudogroup{\RealTopology}$
        by \obv{real_reflection_full_atlas}.
    For every $a\in\RealOneChart$ we have for every $b\in\RealReflectionChart$ we have
        $a$ is compatible with $b$ on $\RealTopology$ under $\FullPseudogroup{\RealTopology}$
        by \obv{real_full_presentations_compatible}.
    Follows by \obv{generated_coordinates_equal,equipped_presentations_equal}.
\end{proof}

\subsection{Weakening the positive structures}

Forgetting the positive-change restriction generates the full maximal atlas.
The two distinct positive structures therefore have the same weakening.

\begin{theorem}\label{real_positive_coordinates_weaken_to_full}
    Then $\WeakenedCoordinates{\PositiveRealPseudogroup}{\FullPseudogroup{\RealTopology}}{\RealTopology}
            {\GeneratedCoordinates{\PositiveRealPseudogroup}{\RealTopology}{\RealOneChart}}
        =\GeneratedCoordinates{\FullPseudogroup{\RealTopology}}{\RealTopology}{\RealOneChart}$ and
        $\WeakenedCoordinates{\PositiveRealPseudogroup}{\FullPseudogroup{\RealTopology}}{\RealTopology}
            {\GeneratedCoordinates{\PositiveRealPseudogroup}{\RealTopology}{\RealReflectionChart}}
        =\GeneratedCoordinates{\FullPseudogroup{\RealTopology}}{\RealTopology}{\RealReflectionChart}$.
\end{theorem}
\begin{proof}
    $\PositiveRealPseudogroup$ is a subpseudogroup of $\FullPseudogroup{\RealTopology}$ by
        \obv{positive_real_pseudogroup_in_full}.
    $\RealOneChart$ is a coordinate atlas on $\RealTopology$ under $\PositiveRealPseudogroup$ and
        $\RealReflectionChart$ is a coordinate atlas on $\RealTopology$ under $\PositiveRealPseudogroup$ by
        \obv{real_identity_positive_atlas,real_reflection_positive_atlas}.
    $\RealOneChart$ presents $\GeneratedCoordinates{\PositiveRealPseudogroup}{\RealTopology}{\RealOneChart}$
        on $\RealTopology$ under $\PositiveRealPseudogroup$ by \obv{generated_coordinates_presentation}.
    $\RealReflectionChart$ presents $\GeneratedCoordinates{\PositiveRealPseudogroup}{\RealTopology}{\RealReflectionChart}$
        on $\RealTopology$ under $\PositiveRealPseudogroup$ by \obv{generated_coordinates_presentation}.
    $\GeneratedCoordinates{\PositiveRealPseudogroup}{\RealTopology}{\RealOneChart}$ is a coordinate datum and
        $\GeneratedCoordinates{\PositiveRealPseudogroup}{\RealTopology}{\RealReflectionChart}$ is a coordinate datum by
        \auto{generated_coordinates,relative_coordinate_structure}.
    Follows by \obv{weakened_coordinates_from_presentation}.
\end{proof}

\begin{theorem}\label{real_positive_weakening_agrees}
    Then $\WeakenedCoordinates{\PositiveRealPseudogroup}{\FullPseudogroup{\RealTopology}}{\RealTopology}
            {\GeneratedCoordinates{\PositiveRealPseudogroup}{\RealTopology}{\RealOneChart}}
        =\WeakenedCoordinates{\PositiveRealPseudogroup}{\FullPseudogroup{\RealTopology}}{\RealTopology}
            {\GeneratedCoordinates{\PositiveRealPseudogroup}{\RealTopology}{\RealReflectionChart}}$ and
        $\Equip{\RealTopology}{\WeakenedCoordinates{\PositiveRealPseudogroup}{\FullPseudogroup{\RealTopology}}{\RealTopology}
            {\GeneratedCoordinates{\PositiveRealPseudogroup}{\RealTopology}{\RealOneChart}}}
        =\Equip{\RealTopology}{\WeakenedCoordinates{\PositiveRealPseudogroup}{\FullPseudogroup{\RealTopology}}{\RealTopology}
            {\GeneratedCoordinates{\PositiveRealPseudogroup}{\RealTopology}{\RealReflectionChart}}}$.
\end{theorem}
\begin{proof}
    Follows by \obv{real_positive_coordinates_weaken_to_full,real_full_coordinate_presentations_equal}.
\end{proof}
