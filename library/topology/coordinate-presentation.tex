\section{From coordinate presentations to structures}

\import{topology/coordinate-topology.tex}
\import{topology/coordinate-atlas-presentation.tex}

A raw presentation first supplies its source topology. Its charts then form
an atlas on that space, so maximal-atlas generation and equipping apply without
choosing additional charts.

\begin{theorem}\label{full_pseudogroup_chart_compatibility}
    Let $X,E$ be topological spaces.
    Suppose $c$ is an open partial homeomorphism from $X$ to $E$ and
        $d$ is an open partial homeomorphism from $X$ to $E$.
    Then $c$ is compatible with $d$ on $X$ under $\FullPseudogroup{E}$.
\end{theorem}
\begin{proof}
    $\FullPseudogroup{E}$ is a coordinate pseudogroup by \obv{full_pseudogroup}.
    $\model[\FullPseudogroup{E}]=E$ by \obv{full_pseudogroup_model}.
    $\converse{c}$ is an open partial homeomorphism from $E$ to $X$ by
        \obv{open_partial_homeomorphism_converse}.
    $d\circ\converse{c}$ is an open partial homeomorphism from $E$ to $E$ by
        \obv{open_partial_homeomorphism_composition}.
    $\chartTransition{d}{c}=d\circ\converse{c}$ by \obv{chart_transition}.
    $\chartTransition{d}{c}\in\changes[\FullPseudogroup{E}]$ by
        \auto{full_pseudogroup_changes}.
    Follows by \obv{coordinate_chart_compatibility}.
\end{proof}

\begin{theorem}\label{full_pseudogroup_atlas_iff}
    Let $X,E$ be topological spaces.
    Then $A$ is a coordinate atlas on $X$ under $\FullPseudogroup{E}$ iff
        (every element of $A$ is an open partial homeomorphism from $X$ to $E$) and
        (for every $x\in\carrier[X]$ there exists $c\in A$ such that $x\in\dom{c}$).
\end{theorem}
\begin{proof}
    $\FullPseudogroup{E}$ is a coordinate pseudogroup by \obv{full_pseudogroup}.
    $\model[\FullPseudogroup{E}]=E$ by \obv{full_pseudogroup_model}.
    Follows by \auto{coordinate_atlas,full_pseudogroup_chart_compatibility}.
\end{proof}

\begin{classification}\label{coordinate_atlas_raw_presentation}
    Let $G$ be a coordinate pseudogroup.
    Let $X$ be a topological space.
    Let $A$ be a coordinate atlas on $X$ under $G$.
    Then $A$ is a topological coordinate presentation on $\carrier[X]$
        modelled on $\model[G]$.
\end{classification}
\begin{proof}
    Every element of $A$ is an open partial homeomorphism from $X$ to $\model[G]$ by
        \obv{coordinate_atlas}.
    For every $x\in\carrier[X]$ there exists $c\in A$ such that $x\in\dom{c}$ by
        \obv{coordinate_atlas}.
    Follows by \obv{open_chart_cover_is_coordinate_presentation}.
\end{proof}

\begin{classification}\label{coordinate_presentation_is_atlas}
    Let $E$ be a topological space.
    Suppose $A$ is a topological coordinate presentation on $X$ modelled on $E$.
    Then $A$ is a coordinate atlas on $\CoordinateTopology{X}{E}{A}$
        under $\FullPseudogroup{E}$.
\end{classification}
\begin{proof}
    $\CoordinateTopology{X}{E}{A}$ is a topological space by
        \obv{coordinate_presentation_topology}.
    Every element of $A$ is an open partial homeomorphism
        from $\CoordinateTopology{X}{E}{A}$ to $E$ by
        \obv{coordinate_chart_is_partial_homeomorphism}.
    $\carrier[\CoordinateTopology{X}{E}{A}]=X$ by \obv{coordinate_topology_carrier}.
    For every $x\in X$ there exists $c\in A$ such that $x\in\dom{c}$ by
        \obv{topological_coordinate_presentation}.
    Follows by \auto{full_pseudogroup_atlas_iff}.
\end{proof}

\subsection{The generated equipped space}

\begin{construction}\label{space_from_charts}
    Let $X$ be a set.
    Let $E$ be a topological space.
    Let $A$ be a topological coordinate presentation on $X$ modelled on $E$.
    The equipped coordinate space $\SpaceFromCharts{X}{E}{A}=
        equipped coordinate space based on \CoordinateTopology{X}{E}{A} with
        \atlas=\atlas[\GeneratedCoordinates{\FullPseudogroup{E}}
            {\CoordinateTopology{X}{E}{A}}{A}]$.
\end{construction}
\begin{proof}
    $\CoordinateTopology{X}{E}{A}$ is a topological space by
        \obv{coordinate_presentation_topology}.
    $\opens[\CoordinateTopology{X}{E}{A}]$ is a topology on
        $\carrier[\CoordinateTopology{X}{E}{A}]$ by
        \obv{topological_space_opens_form_topology}.
    Follows by \auto{topology_family,
        equipped_coordinate_spaceaggregateprojection_carrier,
        equipped_coordinate_spaceaggregateprojection_opens}.
\end{proof}

\begin{theorem}\label{space_from_charts_is_exact}
    Let $E$ be a topological space.
    Suppose $A$ is a topological coordinate presentation on $X$ modelled on $E$.
    Then $\SpaceFromCharts{X}{E}{A}$ is an exact equipped coordinate space.
\end{theorem}
\begin{proof}
    Follows by \auto{space_from_charts,space_from_chartsshape}.
\end{proof}

\begin{theorem}\label{space_from_charts_carrier}
    Let $E$ be a topological space.
    Suppose $A$ is a topological coordinate presentation on $X$ modelled on $E$.
    Then $\carrier[\SpaceFromCharts{X}{E}{A}]=X$.
\end{theorem}
\begin{proof}
    Follows by \auto{space_from_chartsprojection_carrier,coordinate_topology_carrier}.
\end{proof}

\begin{theorem}\label{space_from_charts_equipment}
    Let $E$ be a topological space.
    Suppose $A$ is a topological coordinate presentation on $X$ modelled on $E$.
    Then $\SpaceFromCharts{X}{E}{A}
        =\Equip{\CoordinateTopology{X}{E}{A}}
            {\GeneratedCoordinates{\FullPseudogroup{E}}{\CoordinateTopology{X}{E}{A}}{A}}$.
\end{theorem}
\begin{proof}
    Let $M=\SpaceFromCharts{X}{E}{A}$.
    Let $N=\Equip{\CoordinateTopology{X}{E}{A}}
        {\GeneratedCoordinates{\FullPseudogroup{E}}{\CoordinateTopology{X}{E}{A}}{A}}$.
    $\CoordinateTopology{X}{E}{A}$ is a topological space by
        \obv{coordinate_presentation_topology}.
    $A$ is a coordinate atlas on $\CoordinateTopology{X}{E}{A}$ under
        $\FullPseudogroup{E}$ by \obv{coordinate_presentation_is_atlas}.
    $\GeneratedCoordinates{\FullPseudogroup{E}}{\CoordinateTopology{X}{E}{A}}{A}$
        is a coordinate datum by \auto{generated_coordinates,relative_coordinate_structure}.
    $M$ is an exact equipped coordinate space by \auto{space_from_charts_is_exact}.
    $N$ is an exact equipped coordinate space by \auto{equip_is_exact}.
    $\carrier[M]=\carrier[N]$ by
        \auto{space_from_chartsprojection_carrier,equipprojection_carrier}.
    $\opens[M]=\opens[N]$ by
        \auto{space_from_chartsprojection_opens,equipprojection_opens}.
    $\atlas[M]=\atlas[N]$ by
        \auto{space_from_chartsprojection_atlas,equipprojection_atlas}.
    Follows by structure extensionality.
\end{proof}

\begin{theorem}\label{space_from_charts_open_iff}
    Let $E$ be a topological space.
    Suppose $A$ is a topological coordinate presentation on $X$ modelled on $E$.
    Then $U$ is open in $\SpaceFromCharts{X}{E}{A}$ iff
        $U\subseteq X$ and for every $c\in A$ we have $\img{c}{U}\in\opens[E]$.
\end{theorem}
\begin{proof}
    $\SpaceFromCharts{X}{E}{A}$ is a topological space by
        \obv{space_from_charts,equipped_coordinate_space}.
    $\opens[\SpaceFromCharts{X}{E}{A}]=\opens[\CoordinateTopology{X}{E}{A}]$ by
        \obv{space_from_chartsprojection_opens}.
    Follows by \auto{is_open,coordinate_topology_open_iff}.
\end{proof}

\begin{theorem}\label{space_from_charts_topological_reduct}
    Let $E$ be a topological space.
    Suppose $A$ is a topological coordinate presentation on $X$ modelled on $E$.
    Then $\TopologicalReduct{\SpaceFromCharts{X}{E}{A}}=\CoordinateTopology{X}{E}{A}$.
\end{theorem}
\begin{proof}
    $\CoordinateTopology{X}{E}{A}$ is an exact topological space by
        \auto{topological_coordinate_presentation,coordinate_topology_is_exact}.
    $\carrier[\SpaceFromCharts{X}{E}{A}]=\carrier[\CoordinateTopology{X}{E}{A}]$ and
        $\opens[\SpaceFromCharts{X}{E}{A}]=\opens[\CoordinateTopology{X}{E}{A}]$ by
        \obv{space_from_chartsprojection_carrier,space_from_chartsprojection_opens}.
    $\TopologicalReduct{\SpaceFromCharts{X}{E}{A}}=\TopologicalReduct{\CoordinateTopology{X}{E}{A}}$ by
        \obv{topological_reduct}.
    $\TopologicalReduct{\CoordinateTopology{X}{E}{A}}=\CoordinateTopology{X}{E}{A}$ by
        \obv{topological_reduct_of_exact}.
\end{proof}

\begin{theorem}\label{space_from_charts_coordinate_structure}
    Let $E$ be a topological space.
    Suppose $A$ is a topological coordinate presentation on $X$ modelled on $E$.
    Then $\CoordinateStructure{\SpaceFromCharts{X}{E}{A}}
        =\GeneratedCoordinates{\FullPseudogroup{E}}{\CoordinateTopology{X}{E}{A}}{A}$.
\end{theorem}
\begin{proof}
    $\CoordinateTopology{X}{E}{A}$ is a topological space by \obv{coordinate_presentation_topology}.
    $\FullPseudogroup{E}$ is a coordinate pseudogroup by \obv{full_pseudogroup}.
    $A$ is a coordinate atlas on $\CoordinateTopology{X}{E}{A}$ under $\FullPseudogroup{E}$ by
        \obv{coordinate_presentation_is_atlas}.
    $\GeneratedCoordinates{\FullPseudogroup{E}}{\CoordinateTopology{X}{E}{A}}{A}$ is an exact coordinate datum by
        \auto{generated_coordinates_is_exact,generated_coordinatesshape,relative_coordinate_structure}.
    Follows by \auto{space_from_charts_equipment,coordinate_structure_of_equip}.
\end{proof}

\begin{theorem}\label{space_from_charts_coordinates}
    Let $E$ be a topological space.
    Suppose $A$ is a topological coordinate presentation on $X$ modelled on $E$.
    Then $\SpaceFromCharts{X}{E}{A}$ has coordinates under $\FullPseudogroup{E}$.
\end{theorem}
\begin{proof}

        {\GeneratedCoordinates{\FullPseudogroup{E}}{\CoordinateTopology{X}{E}{A}}{A}}$

\begin{theorem}\label{full_max_atlas}
    Let $X,E$ be topological spaces.
    Suppose $A$ is a coordinate atlas on $X$ under $\FullPseudogroup{E}$.
    Then $\MaxAtlas{\FullPseudogroup{E}}{X}{A}=\openPartialHomeomorphisms{X}{E}$.
\end{theorem}
\begin{proof}
    $\FullPseudogroup{E}$ is a coordinate pseudogroup by \obv{full_pseudogroup}.
    $\model[\FullPseudogroup{E}]=E$ by \obv{full_pseudogroup_model}.
    Every element of $A$ is an open partial homeomorphism from $X$ to $E$ by
        \auto{full_pseudogroup_atlas_iff}.
    For every $c$ such that $c$ is an open partial homeomorphism from $X$ to $E$
        we have for every $a\in A$ we have
            $c$ is compatible with $a$ on $X$ under $\FullPseudogroup{E}$ by
        \obv{full_pseudogroup_chart_compatibility}.
    For every $c$ we have $c\in\MaxAtlas{\FullPseudogroup{E}}{X}{A}$ iff
        $c\in\openPartialHomeomorphisms{X}{E}$ by
        \auto{max_atlas_member_iff,open_partial_homeomorphisms_iff}.
    Follows by set extensionality.
\end{proof}

\begin{theorem}\label{space_from_charts_atlas}
    Let $E$ be a topological space.
    Suppose $A$ is a topological coordinate presentation on $X$ modelled on $E$.
    Then $\atlas[\SpaceFromCharts{X}{E}{A}]
        =\openPartialHomeomorphisms{\CoordinateTopology{X}{E}{A}}{E}$.
\end{theorem}
\begin{proof}
    $A$ is a coordinate atlas on $\CoordinateTopology{X}{E}{A}$ under
        $\FullPseudogroup{E}$ by \obv{coordinate_presentation_is_atlas}.
    $\atlas[\GeneratedCoordinates{\FullPseudogroup{E}}{\CoordinateTopology{X}{E}{A}}{A}]
        =\MaxAtlas{\FullPseudogroup{E}}{\CoordinateTopology{X}{E}{A}}{A}$ by
        \obv{generated_coordinatesprojection_atlas}.
    $\MaxAtlas{\FullPseudogroup{E}}{\CoordinateTopology{X}{E}{A}}{A}
        =\openPartialHomeomorphisms{\CoordinateTopology{X}{E}{A}}{E}$ by
        \obv{full_max_atlas}.
    Follows by \auto{space_from_chartsprojection_atlas}.
\end{proof}

\begin{theorem}\label{space_from_charts_chart_iff}
    Let $E$ be a topological space.
    Suppose $A$ is a topological coordinate presentation on $X$ modelled on $E$.
    Then $c\in\atlas[\SpaceFromCharts{X}{E}{A}]$ iff
        $c$ is an open partial homeomorphism from $\SpaceFromCharts{X}{E}{A}$ to $E$.
\end{theorem}
\begin{proof}
    $\SpaceFromCharts{X}{E}{A}$ is a topological space by
        \obv{space_from_charts,equipped_coordinate_space}.
    $\CoordinateTopology{X}{E}{A}$ is a topological space by \obv{coordinate_presentation_topology}.
    $\carrier[\SpaceFromCharts{X}{E}{A}]=\carrier[\CoordinateTopology{X}{E}{A}]$ and
        $\opens[\SpaceFromCharts{X}{E}{A}]=\opens[\CoordinateTopology{X}{E}{A}]$ by
        \obv{space_from_chartsprojection_carrier,space_from_chartsprojection_opens}.
    $c$ is an open partial homeomorphism from $\CoordinateTopology{X}{E}{A}$ to $E$ iff
        $c$ is an open partial homeomorphism from $\SpaceFromCharts{X}{E}{A}$ to $E$ by
        \obv{open_partial_homeomorphism_same_source_fields}.
    Follows by \auto{space_from_charts_atlas,open_partial_homeomorphisms_iff}.
\end{proof}

\subsection{Recovery and presentation independence}

\begin{theorem}\label{space_from_presented_coordinates}
    Let $E$ be a topological space.
    Suppose $M$ is an exact equipped coordinate space.
    Suppose $A$ presents $M$ on $M$ under $\FullPseudogroup{E}$.
    Then $\SpaceFromCharts{\carrier[M]}{E}{A}=M$.
\end{theorem}
\begin{proof}
    $M$ is a topological space and $M$ is a coordinate datum by \obv{equipped_coordinate_space}.
    $\FullPseudogroup{E}$ is a coordinate pseudogroup by \obv{full_pseudogroup}.
    $\model[\FullPseudogroup{E}]=E$ by \obv{full_pseudogroup_model}.
    $A$ is a coordinate atlas on $M$ under $\FullPseudogroup{E}$ by \obv{presentation_is_atlas}.
    $A$ is a topological coordinate presentation on $\carrier[M]$ modelled on $E$ by
        \auto{coordinate_atlas_raw_presentation}.
    $\CoordinateTopology{\carrier[M]}{E}{A}=\TopologicalReduct{M}$ by
        \auto{full_pseudogroup_atlas_iff,coordinate_topology_recovers_reduct}.
    $\TopologicalReduct{M}$ is a topological space by \obv{topological_reduct_is_topological_space}.
    $\CoordinateStructure{M}$ is an exact coordinate datum by \obv{coordinate_structure_is_exact}.
    $\carrier[\TopologicalReduct{M}]=\carrier[M]$ and
        $\opens[\TopologicalReduct{M}]=\opens[M]$ and
        $\atlas[\CoordinateStructure{M}]=\atlas[M]$ by
        \obv{topological_reduct_carrier,topological_reduct_opens,coordinate_structureprojection_atlas}.
    $A$ presents $\CoordinateStructure{M}$ on $\TopologicalReduct{M}$ under $\FullPseudogroup{E}$ by
        \obv{coordinate_presentation_same_fields}.
    $\GeneratedCoordinates{\FullPseudogroup{E}}{\TopologicalReduct{M}}{A}=\CoordinateStructure{M}$ by
        \obv{exact_presented_coordinates}.
    Follows by \auto{space_from_charts_equipment,equip_coordinate_structure}.
\end{proof}

\begin{theorem}\label{space_from_charts_same_topology}
    Let $E$ be a topological space.
    Suppose $A$ is a topological coordinate presentation on $X$ modelled on $E$ and
        $B$ is a topological coordinate presentation on $X$ modelled on $E$.
    Suppose $\CoordinateTopology{X}{E}{A}=\CoordinateTopology{X}{E}{B}$.
    Then $\SpaceFromCharts{X}{E}{A}=\SpaceFromCharts{X}{E}{B}$.
\end{theorem}
\begin{proof}
    $\SpaceFromCharts{X}{E}{A}$ is an exact equipped coordinate space by
        \obv{space_from_charts_is_exact}.
    $\SpaceFromCharts{X}{E}{B}$ is an exact equipped coordinate space by
        \obv{space_from_charts_is_exact}.
    $\carrier[\SpaceFromCharts{X}{E}{A}]=\carrier[\SpaceFromCharts{X}{E}{B}]$ by
        \auto{space_from_charts_carrier}.
    $\opens[\SpaceFromCharts{X}{E}{A}]=\opens[\SpaceFromCharts{X}{E}{B}]$ by
        \auto{space_from_chartsprojection_opens}.
    $\atlas[\SpaceFromCharts{X}{E}{A}]=\atlas[\SpaceFromCharts{X}{E}{B}]$ by
        \auto{space_from_charts_atlas}.
    Follows by structure extensionality.
\end{proof}

\begin{theorem}\label{space_from_charts_equal_iff}
    Let $E$ be a topological space.
    Suppose $A$ is a topological coordinate presentation on $X$ modelled on $E$ and
        $B$ is a topological coordinate presentation on $X$ modelled on $E$.
    Then $\SpaceFromCharts{X}{E}{A}=\SpaceFromCharts{X}{E}{B}$ iff
        $\CoordinateTopology{X}{E}{A}=\CoordinateTopology{X}{E}{B}$.
\end{theorem}
\begin{proof}
    Follows by \auto{space_from_charts_same_topology,space_from_charts_topological_reduct}.
\end{proof}

\begin{theorem}\label{space_from_cross_compatible_charts}
    Let $E$ be a topological space.
    Suppose $A$ is a topological coordinate presentation on $X$ modelled on $E$ and
        $B$ is a topological coordinate presentation on $X$ modelled on $E$.
    Suppose for every $a\in A$ we have for every $b\in B$ we have
        $a$ is topologically compatible with $b$ on $X$ modelled on $E$.
    Then $\SpaceFromCharts{X}{E}{A}=\SpaceFromCharts{X}{E}{B}$.
\end{theorem}
\begin{proof}
    $\CoordinateTopology{X}{E}{A}=\CoordinateTopology{X}{E}{B}$ by
        \obv{cross_compatible_presentations_same_topology}.
    Follows by \obv{space_from_charts_same_topology}.
\end{proof}
