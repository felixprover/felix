\section{Separation through neighbourhood filters}

The open-set definitions remain primary.  Neighbourhood filters package the
same separation data in the form used by convergence arguments.

\begin{theorem}\label{hausdorff_iff_neighbourhood_filters_do_not_mesh}
    Let $X$ be a topological space.
    Then $X$ is Hausdorff iff for all $x,y\in\carrier[X]$ such that
        $x\neq y$ we have $\neighbourhoods{x}{X}$ does not mesh with
        $\neighbourhoods{y}{X}$ on $\carrier[X]$.
\end{theorem}
\begin{proof}
    Show that if $X$ is Hausdorff, then for all
        $x,y\in\carrier[X]$ such that $x\neq y$ we have
        $\neighbourhoods{x}{X}$ does not mesh with
        $\neighbourhoods{y}{X}$ on $\carrier[X]$.
    \begin{subproof}
        Assume $X$ is Hausdorff.
        Fix $x,y\in\carrier[X]$.
        Assume $x\neq y$.
        Take $U,V\in\opens[X]$ such that $x\in U$ and $y\in V$ and
            $U$ is disjoint from $V$ by \obv{is_hausdorff}.
        $U\in\neighbourhoods{x}{X}$ and
            $V\in\neighbourhoods{y}{X}$ by
            \obv{open_set_containing_point_is_neighbourhood}.
        Show that $\neighbourhoods{x}{X}$ does not mesh with
            $\neighbourhoods{y}{X}$ on $\carrier[X]$.
        \begin{subproof}
            Suppose not.
            Hence $\neighbourhoods{x}{X}$ meshes with
                $\neighbourhoods{y}{X}$ on $\carrier[X]$.
            Hence $U\inter V$ is inhabited by \obv{filters_mesh}.
            Take $z$ such that $z\in U\inter V$.
            Hence $z\in U,V$ by \obv{inter}.
            Hence $U$ is not disjoint from $V$ by \auto{disjoint}.
            Contradiction.
        \end{subproof}
    \end{subproof}

    Show that if for all $x,y\in\carrier[X]$ such that $x\neq y$ we have
        $\neighbourhoods{x}{X}$ does not mesh with
        $\neighbourhoods{y}{X}$ on $\carrier[X]$, then
        $X$ is Hausdorff.
    \begin{subproof}
        Assume for all $x,y\in\carrier[X]$ such that $x\neq y$ we have
            $\neighbourhoods{x}{X}$ does not mesh with
            $\neighbourhoods{y}{X}$ on $\carrier[X]$.
        Show that for all $x,y\in\carrier[X]$ such that $x\neq y$ there
            exist $U,V\in\opens[X]$ such that $x\in U$ and $y\in V$ and
            $U$ is disjoint from $V$.
        \begin{subproof}
            Fix $x,y\in\carrier[X]$.
            Assume $x\neq y$.
            $\neighbourhoods{x}{X}$ and $\neighbourhoods{y}{X}$ are
                filters on $\carrier[X]$ by
                \obv{neighbourhoods_is_filter}.
            $\neighbourhoods{x}{X}$ does not mesh with
                $\neighbourhoods{y}{X}$ on $\carrier[X]$.
            Take $N,M$ such that $N\in\neighbourhoods{x}{X}$ and
                $M\in\neighbourhoods{y}{X}$ and
                $N\inter M$ is not inhabited by \auto{filters_mesh}.
            Take $U\in\opens[X]$ such that $x\in U\subseteq N$ by
                \obv{neighbourhoods}.
            Take $V\in\opens[X]$ such that $y\in V\subseteq M$ by
                \obv{neighbourhoods}.
            Show that $U$ is disjoint from $V$.
            \begin{subproof}
                Suppose not.
                Take $z$ such that $z\in U,V$ by \auto{disjoint}.
                Hence $z\in N,M$ by \obv{subseteq}.
                Hence $z\in N\inter M$ by \obv{inter_intro}.
                Hence $N\inter M$ is inhabited.
                Contradiction.
            \end{subproof}
            Follows by \auto.
        \end{subproof}
        Follows by \obv{is_hausdorff}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{theorem}\label{neighbourhood_filters_do_not_mesh_iff_common_refinement_improper}
    Let $X$ be a topological space and $x,y\in\carrier[X]$.
    Then $\neighbourhoods{x}{X}$ does not mesh with
        $\neighbourhoods{y}{X}$ on $\carrier[X]$ iff
        $\binaryCommonFilterRefinement{\carrier[X]}{
            \neighbourhoods{x}{X}}{\neighbourhoods{y}{X}}$
        is an improper filter on $\carrier[X]$.
\end{theorem}
\begin{proof}
    $\neighbourhoods{x}{X}$ and $\neighbourhoods{y}{X}$ are filters on
        $\carrier[X]$ by \obv{neighbourhoods_is_filter}.
    $\binaryCommonFilterRefinement{\carrier[X]}{
        \neighbourhoods{x}{X}}{\neighbourhoods{y}{X}}$ is a filter on
        $\carrier[X]$ by \obv{binary_common_filter_refinement_is_filter}.
    $\neighbourhoods{x}{X}$ meshes with $\neighbourhoods{y}{X}$ on
        $\carrier[X]$ iff
        $\binaryCommonFilterRefinement{\carrier[X]}{
            \neighbourhoods{x}{X}}{\neighbourhoods{y}{X}}$
        is a proper filter on $\carrier[X]$ by
        \obv{filters_mesh_iff_common_refinement_proper}.
    Follows by \auto{proper_filter,improper_filter}.
\end{proof}

\begin{theorem}\label{hausdorff_proper_filter_limit_unique}
    Let $X$ be a Hausdorff space and
        $F$ be a proper filter on $\carrier[X]$.
    Suppose $\filterConverges{F}{x}{X}$ and
        $\filterConverges{F}{y}{X}$.
    Then $x=y$.
\end{theorem}
\begin{proof}
    $F$ is a filter on $\carrier[X]$ by \obv{proper_filter_is_filter}.
    $X$ is Hausdorff iff for all $u,v\in\carrier[X]$ such that
        $u\neq v$ we have $\neighbourhoods{u}{X}$ does not mesh with
        $\neighbourhoods{v}{X}$ on $\carrier[X]$ by
        \obv{hausdorff_iff_neighbourhood_filters_do_not_mesh}.
    Hence for all $u,v\in\carrier[X]$ such that $u\neq v$ we have
        $\neighbourhoods{u}{X}$ does not mesh with
        $\neighbourhoods{v}{X}$ on $\carrier[X]$.
    $x,y\in\carrier[X]$ and
        $\neighbourhoods{x}{X},\neighbourhoods{y}{X}\subseteq F$ by
        \obv{filter_converges}.
    $\neighbourhoods{x}{X}$ and $\neighbourhoods{y}{X}$ are filters on
        $\carrier[X]$ by \obv{neighbourhoods_is_filter}.
    Show that $\neighbourhoods{x}{X}$ meshes with
        $\neighbourhoods{y}{X}$ on $\carrier[X]$.
    \begin{subproof}
        Show that for all $A,B$ such that
            $A\in\neighbourhoods{x}{X}$ and
            $B\in\neighbourhoods{y}{X}$ we have
            $A\inter B$ is inhabited.
        \begin{subproof}
            Fix $A,B$.
            Assume $A\in\neighbourhoods{x}{X}$ and
                $B\in\neighbourhoods{y}{X}$.
            Hence $A,B\in F$ by \obv{subseteq}.
            Hence $A\inter B\in F$ by \obv{filter}.
            $\emptyset\notin F$ by \obv{proper_filter_excludes_emptyset}.
            Thus $A\inter B\neq\emptyset$.
            Follows by \obv{nonempty_inhabited}.
        \end{subproof}
        Follows by \obv{filters_mesh}.
    \end{subproof}
    Show that $x=y$.
    \begin{subproof}
        Suppose not.
        Hence $\neighbourhoods{x}{X}$ does not mesh with
            $\neighbourhoods{y}{X}$ on $\carrier[X]$.
        Contradiction.
    \end{subproof}
\end{proof}

\subsection{Singleton closures in \teeone-spaces}

\begin{theorem}\label{teeone_iff_singleton_closures}
    Let $X$ be a topological space.
    Then $X$ is \teeone\ iff for every $x\in\carrier[X]$ we have
        $\closure{\{x\}}{X}=\{x\}$.
\end{theorem}
\begin{proof}
    Show that if $X$ is \teeone, then for every $x\in\carrier[X]$ we have
        $\closure{\{x\}}{X}=\{x\}$.
    \begin{subproof}
        Assume $X$ is \teeone.
        Fix $x\in\carrier[X]$.
        $\{x\}\subseteq\carrier[X]$ by notion.
        $\{x\}$ is closed in $X$ by
            \obv{teeone_implies_singletons_closed}.
        Hence $\{x\}\in\closeds{X}$ by \obv{is_closed_in}.
        $\closure{\{x\}}{X}\subseteq\{x\}$ by
            \obv{closure_is_minimal_closed_set}.
        $\{x\}\subseteq\closure{\{x\}}{X}$ by
            \obv{set_is_subseteq_to_closure_of_the_set}.
        Follows by \obv{subseteq_antisymmetric}.
    \end{subproof}

    Show that if for every $x\in\carrier[X]$ we have
        $\closure{\{x\}}{X}=\{x\}$, then $X$ is \teeone.
    \begin{subproof}
        Assume for every $x\in\carrier[X]$ we have
            $\closure{\{x\}}{X}=\{x\}$.
        Show that for every $x\in\carrier[X]$ we have
            $\{x\}$ is closed in $X$.
        \begin{subproof}
            Fix $x\in\carrier[X]$.
            $\closure{\{x\}}{X}$ is closed in $X$ by
                \obv{closure_is_closed}.
            $\closure{\{x\}}{X}=\{x\}$.
        \end{subproof}
        Follows by \obv{teeone_iff_singletons_closed}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{theorem}\label{teeone_iff_singleton_closure_points}
    Let $X$ be a topological space.
    Then $X$ is \teeone\ iff for all $x,y\in\carrier[X]$ such that
        $y\in\closure{\{x\}}{X}$ we have $y=x$.
\end{theorem}
\begin{proof}
    $X$ is \teeone\ iff for every $z\in\carrier[X]$ we have
        $\closure{\{z\}}{X}=\{z\}$ by
        \obv{teeone_iff_singleton_closures}.
    Show that if $X$ is \teeone, then for all $x,y\in\carrier[X]$ such
        that $y\in\closure{\{x\}}{X}$ we have $y=x$.
    \begin{subproof}
        Assume $X$ is \teeone.
        Hence for every $z\in\carrier[X]$ we have
            $\closure{\{z\}}{X}=\{z\}$.
        Fix $x,y\in\carrier[X]$.
        Assume $y\in\closure{\{x\}}{X}$.
        Hence $\closure{\{x\}}{X}=\{x\}$.
        Hence $y\in\{x\}$.
        $y\in\{x\}$ iff $y=x$ by \obv{singleton_iff}.
    \end{subproof}

    Show that if for all $x,y\in\carrier[X]$ such that
        $y\in\closure{\{x\}}{X}$ we have $y=x$, then
        $X$ is \teeone.
    \begin{subproof}
        Assume for all $x,y\in\carrier[X]$ such that
            $y\in\closure{\{x\}}{X}$ we have $y=x$.
        Show that for every $x\in\carrier[X]$ we have
            $\closure{\{x\}}{X}=\{x\}$.
        \begin{subproof}
            Fix $x\in\carrier[X]$.
            $\{x\}\subseteq\carrier[X]$ by notion.
            $\carrier[X]\in\closeds{X}$ by
                \auto{carrier_is_closed,is_closed_in}.
            Hence $\closure{\{x\}}{X}\subseteq\carrier[X]$ by
                \obv{closure_is_minimal_closed_set}.
            Show that $\closure{\{x\}}{X}\subseteq\{x\}$.
            \begin{subproof}
                It suffices to show that every element of
                    $\closure{\{x\}}{X}$ is an element of $\{x\}$ by
                    \obv{subseteq}.
                Fix $y\in\closure{\{x\}}{X}$.
                Hence $y\in\carrier[X]$ by \obv{subseteq}.
                Hence $y=x$.
                $y\in\{x\}$ iff $y=x$ by \obv{singleton_iff}.
            \end{subproof}
            $\{x\}\subseteq\closure{\{x\}}{X}$ by
                \obv{set_is_subseteq_to_closure_of_the_set}.
            For every $z$ we have $z\in\closure{\{x\}}{X}$ iff
                $z\in\{x\}$ by \auto{subseteq}.
            Follows by set extensionality.
        \end{subproof}
    \end{subproof}
    Follows by \auto.
\end{proof}
