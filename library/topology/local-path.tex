\section{Local path connectedness}

\import{topology/path-subspace.tex}
\import{topology/basis.tex}

Local path connectedness means that every open neighbourhood contains an
open path-connected neighbourhood of the same point. The path-connected
neighbourhood carries its subspace topology.

\begin{definition}\label{locally_path_connected}
    Let $X$ be a topological space.
    $X$ is locally path-connected iff for all $U,x$ such that
        $U\in\opens[X]$ and $x\in U$ there exists $V\in\opens[X]$ such that
        $x\in V\subseteq U$ and $\subspaceTopology{X}{V}$ is path-connected.
\end{definition}

\begin{theorem}\label{path_connected_open_basis_implies_locally_path_connected}
    Let $X$ be a topological space.
    Suppose $B$ is an open basis of $X$ and for every $V\in B$ we have
        $\subspaceTopology{X}{V}$ is path-connected.
    Then $X$ is locally path-connected.
\end{theorem}
\begin{proof}
    Show that for all $U,x$ such that $U\in\opens[X]$ and $x\in U$ there exists
        $V\in\opens[X]$ such that $x\in V\subseteq U$ and
        $\subspaceTopology{X}{V}$ is path-connected.
    \begin{subproof}
        Fix $U,x$.
        Assume $U\in\opens[X]$ and $x\in U$.
        Take $V\in B$ such that $x\in V\subseteq U$ by \obv{open_basis}.
        $V\in\opens[X]$ by \auto{open_basis,subseteq}.
        $\subspaceTopology{X}{V}$ is path-connected.
    \end{subproof}
    Follows by \obv{locally_path_connected}.
\end{proof}

\begin{theorem}\label{subsingleton_space_is_locally_path_connected}
    Let $X$ be a topological space.
    Suppose $\carrier[X]$ is a subsingleton.
    Then $X$ is locally path-connected.
\end{theorem}
\begin{proof}
    Show that for all $U,x$ such that $U\in\opens[X]$ and $x\in U$ there exists
        $V\in\opens[X]$ such that $x\in V\subseteq U$ and
        $\subspaceTopology{X}{V}$ is path-connected.
    \begin{subproof}
        Fix $U,x$.
        Assume $U\in\opens[X]$ and $x\in U$.
        $U\subseteq\carrier[X]$ by \auto{opens_type,subseteq}.
        $U=\{x\}$ by \auto{setext,singleton_iff,subseteq}.
        $\carrier[\subspaceTopology{X}{U}]=\{x\}$ by
            \auto{subspace_topology_carrier_of_subset}.
        $\subspaceTopology{X}{U}$ is path-connected by
            \auto{singleton_space_is_path_connected,subspace_topology_is_topological_space}.
        Follows by \auto{subseteq_refl}.
    \end{subproof}
    Follows by \obv{locally_path_connected}.
\end{proof}
