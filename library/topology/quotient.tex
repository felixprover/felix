\section{Quotient topologies and quotient maps}

\import{topology/single-final.tex}

A quotient topology is the final topology of one map.  The constructor is
total, while its structural theorems require the displayed source and map to
be well typed.  Surjectivity enters in the notion of quotient map.

\begin{definition}\label{quotient_topology}
    $\QuotientTopology{X}{Q}{q}=\SingleFinalTopology{Q}{X}{q}$.
\end{definition}

\begin{theorem}\label{quotient_topology_is_topological_space}
    Let $X$ be a topological space.
    Let $q\in\funs{\carrier[X]}{Q}$.
    Then $\QuotientTopology{X}{Q}{q}$ is a topological space.
\end{theorem}
\begin{proof}
    $\SingleFinalTopology{Q}{X}{q}$ is a topological space by
        \obv{single_final_topology_is_topological_space}.
    Follows by \obv{quotient_topology}.
\end{proof}

\begin{theorem}\label{quotient_topology_is_exact}
    Let $X$ be a topological space.
    Let $q\in\funs{\carrier[X]}{Q}$.
    Then $\QuotientTopology{X}{Q}{q}$ is an exact topological space.
\end{theorem}
\begin{proof}
    $\SingleFinalTopology{Q}{X}{q}$ is an exact topological space by
        \obv{single_final_topology_is_exact}.
    Follows by \auto{quotient_topology}.
\end{proof}

\begin{proposition}\label{quotient_topology_carrier}
    $\carrier[\QuotientTopology{X}{Q}{q}]=Q$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{quotient_topology,single_final_topology_carrier}.
\end{proof}

\begin{proposition}\label{quotient_topology_opens}
    $\opens[\QuotientTopology{X}{Q}{q}]
        =\preimageOpenFamily{X}{Q}{q}$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{quotient_topology,single_final_topology_opens}.
\end{proof}

\begin{proposition}\label{quotient_topology_open_iff}
    $U\in\opens[\QuotientTopology{X}{Q}{q}]$ iff
        $U\subseteq Q$ and $\preimg{q}{U}\in\opens[X]$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{quotient_topology,single_final_topology_open_iff}.
\end{proof}

\begin{theorem}\label{quotient_topology_structure_map_continuous}
    Let $X$ be a topological space.
    Let $q\in\funs{\carrier[X]}{Q}$.
    Then $q$ is continuous from $X$ to
        $\QuotientTopology{X}{Q}{q}$.
\end{theorem}
\begin{proof}
    $q$ is continuous from $X$ to
        $\SingleFinalTopology{Q}{X}{q}$ by
        \obv{single_final_topology_structure_map_continuous}.
    Follows by \obv{quotient_topology}.
\end{proof}

\begin{theorem}\label{quotient_topology_universal_property}
    Let $X,Y$ be topological spaces.
    Let $q\in\funs{\carrier[X]}{Q}$.
    Let $g\in\funs{Q}{\carrier[Y]}$.
    Then $g$ is continuous from $\QuotientTopology{X}{Q}{q}$ to $Y$
        iff $g\circ q$ is continuous from $X$ to $Y$.
\end{theorem}
\begin{proof}
    Follows by
        \auto{quotient_topology,
            single_final_topology_universal_property}.
\end{proof}

\begin{definition}\label{quotient_map}
    $q$ is a quotient map from $X$ to $Y$ iff
        $X$ is a topological space and
        $Y$ is a topological space and
        $q\in\surj{\carrier[X]}{\carrier[Y]}$ and
        $\opens[Y]
            =\preimageOpenFamily{X}{\carrier[Y]}{q}$.
\end{definition}

\begin{theorem}\label{quotient_map_is_continuous}
    Suppose $q$ is a quotient map from $X$ to $Y$.
    Then $q$ is continuous from $X$ to $Y$.
\end{theorem}
\begin{proof}
    $X$ is a topological space and $Y$ is a topological space and
        $q\in\surj{\carrier[X]}{\carrier[Y]}$ and
        $\opens[Y]
            =\preimageOpenFamily{X}{\carrier[Y]}{q}$ by
        \obv{quotient_map}.
    Hence $q\in\funs{\carrier[X]}{\carrier[Y]}$ by \obv{surj}.
    Show that for every $U\in\opens[Y]$ we have
        $\preimg{q}{U}\in\opens[X]$.
    \begin{subproof}
        Fix $U\in\opens[Y]$.
        Hence $U\in\preimageOpenFamily{X}{\carrier[Y]}{q}$.
        Follows by \obv{preimage_open_family}.
    \end{subproof}
    Follows by \obv{continuous_map}.
\end{proof}

\begin{theorem}\label{quotient_topology_structure_map_is_quotient_map}
    Let $X$ be a topological space.
    Let $q\in\surj{\carrier[X]}{Q}$.
    Then $q$ is a quotient map from $X$ to
        $\QuotientTopology{X}{Q}{q}$.
\end{theorem}
\begin{proof}
    $q\in\funs{\carrier[X]}{Q}$ by \obv{surj}.
    Hence $\QuotientTopology{X}{Q}{q}$ is a topological space by
        \obv{quotient_topology_is_topological_space}.
    $\carrier[\QuotientTopology{X}{Q}{q}]=Q$ by
        \obv{quotient_topology_carrier}.
    Hence $q\in\surj{\carrier[X]}{
        \carrier[\QuotientTopology{X}{Q}{q}]}$.
    $\opens[\QuotientTopology{X}{Q}{q}]
        =\preimageOpenFamily{X}{Q}{q}$ by
        \obv{quotient_topology_opens}.
    Hence $\opens[\QuotientTopology{X}{Q}{q}]
        =\preimageOpenFamily{X}{
            \carrier[\QuotientTopology{X}{Q}{q}]}{q}$.
    Follows by \obv{quotient_map}.
\end{proof}

\begin{proposition}\label{quotient_map_open_iff}
    Suppose $q$ is a quotient map from $X$ to $Y$.
    Then $U\in\opens[Y]$ iff
        $U\subseteq\carrier[Y]$ and
        $\preimg{q}{U}\in\opens[X]$.
\end{proposition}
\begin{proof}
    $\opens[Y]=\preimageOpenFamily{X}{\carrier[Y]}{q}$ by
        \obv{quotient_map}.
    Hence $U\in\opens[Y]$ iff
        $U\in\preimageOpenFamily{X}{\carrier[Y]}{q}$.
    $U\in\preimageOpenFamily{X}{\carrier[Y]}{q}$ iff
        $U\in\pow{\carrier[Y]}$ and
        $\preimg{q}{U}\in\opens[X]$ by
        \obv{preimage_open_family}.
    $U\in\pow{\carrier[Y]}$ iff
        $U\subseteq\carrier[Y]$ by \obv{pow_iff}.
    Follows by \auto.
\end{proof}

\begin{theorem}\label{quotient_map_continuity_criterion}
    Suppose $q$ is a quotient map from $X$ to $Y$.
    Let $Z$ be a topological space.
    Let $g\in\funs{\carrier[Y]}{\carrier[Z]}$.
    Then $g$ is continuous from $Y$ to $Z$ iff
        $g\circ q$ is continuous from $X$ to $Z$.
\end{theorem}
\begin{proof}
    Show that if $g$ is continuous from $Y$ to $Z$, then
        $g\circ q$ is continuous from $X$ to $Z$.
    \begin{subproof}
        Assume $g$ is continuous from $Y$ to $Z$.
        $q$ is continuous from $X$ to $Y$ by
            \obv{quotient_map_is_continuous}.
        Follows by \obv{continuous_composition}.
    \end{subproof}

    Show that if $g\circ q$ is continuous from $X$ to $Z$, then
        $g$ is continuous from $Y$ to $Z$.
    \begin{subproof}
        Assume $g\circ q$ is continuous from $X$ to $Z$.
        Show that for every $V\in\opens[Z]$ we have
            $\preimg{g}{V}\in\opens[Y]$.
        \begin{subproof}
            Fix $V\in\opens[Z]$.
            $\preimg{g\circ q}{V}\in\opens[X]$ by
                \obv{continuous_map}.
            $q\in\funs{\carrier[X]}{\carrier[Y]}$ by
                \auto{quotient_map,surj}.
            $\preimg{g\circ q}{V}
                =\preimg{q}{\preimg{g}{V}}$ by
                \obv{preimage_composition}.
            Hence $\preimg{q}{\preimg{g}{V}}\in\opens[X]$.
            $\preimg{g}{V}\subseteq\carrier[Y]$ by
                \auto{preimg_subseteq_dom,funs,subseteq}.
            Follows by \auto{quotient_map_open_iff}.
        \end{subproof}
        Follows by \obv{continuous_map}.
    \end{subproof}
    Follows by \auto.
\end{proof}
