\section{Final topologies}

\import{topology/generated-continuity.tex}
\import{topology/discrete.tex}

Final topologies are dual to initial topologies at the level of their
universal property.  Their open sets, however, can be defined directly: a
subset is open precisely when all its inverse images are open.

\begin{definition}\label{final_topology_data}
    $f$ is a final mapping family from $X$ to $T$ indexed by $I$ iff
        $X$ is a function on $I$ and
        $f$ is a function on $I$ and
        for every $i\in I$ we have $X(i)$ is a topological space and
            $f(i)\in\funs{\carrier[X(i)]}{T}$.
\end{definition}

\begin{proposition}\label{final_topology_data_at}
    Suppose $f$ is a final mapping family from $X$ to $T$ indexed by $I$.
    Let $i\in I$.
    Then $X(i)$ is a topological space and
        $f(i)\in\funs{\carrier[X(i)]}{T}$.
\end{proposition}
\begin{proof}
    Follows by \obv{final_topology_data}.
\end{proof}

\begin{definition}\label{final_opens}
    $\finalOpens{T}{I}{X}{f}
        =\{U\in\pow{T}\mid
            \text{for every $i\in I$ we have
                $\preimg{f(i)}{U}\in\opens[X(i)]$}\}$.
\end{definition}

\begin{proposition}\label{final_opens_type}
    $\finalOpens{T}{I}{X}{f}$ is a family of subsets of $T$.
\end{proposition}
\begin{proof}
    $\finalOpens{T}{I}{X}{f}\subseteq\pow{T}$ by
        \obv{final_opens,subseteq}.
    Follows by \obv{pow_iff,subseteq}.
\end{proof}

\begin{theorem}\label{final_opens_is_topology}
    Suppose $f$ is a final mapping family from $X$ to $T$ indexed by $I$.
    Then $\finalOpens{T}{I}{X}{f}$ is a topology on $T$.
\end{theorem}
\begin{proof}
    $\finalOpens{T}{I}{X}{f}\subseteq\pow{T}$ by
        \obv{final_opens,subseteq}.
    For every $i\in I$ we have $X(i)$ is a topological space and
        $f(i)\in\funs{\carrier[X(i)]}{T}$ by
        \obv{final_topology_data}.

    Show that $T\in\finalOpens{T}{I}{X}{f}$.
    \begin{subproof}
        $T\in\pow{T}$ by \obv{powerset_top}.
        Show that for every $i\in I$ we have
            $\preimg{f(i)}{T}\in\opens[X(i)]$.
        \begin{subproof}
            Fix $i\in I$.
            $X(i)$ is a topological space and
                $f(i)\in\funs{\carrier[X(i)]}{T}$ by
                \obv{final_topology_data_at}.
            $\preimg{f(i)}{T}=\carrier[X(i)]$ by
                \obv{preimg_codom_of_function}.
            $\carrier[X(i)]\in\opens[X(i)]$ by \obv{carrier_open}.
        \end{subproof}
        Follows by \obv{final_opens}.
    \end{subproof}

    Show that for all $A,B\in\finalOpens{T}{I}{X}{f}$ we have
        $A\inter B\in\finalOpens{T}{I}{X}{f}$.
    \begin{subproof}
        Fix $A,B\in\finalOpens{T}{I}{X}{f}$.
        $A,B\subseteq T$ by \obv{final_opens,powerset_elim}.
        Hence $A\inter B\subseteq T$ by \obv{inter_subseteq}.
        Thus $A\inter B\in\pow{T}$ by \obv{powerset_intro}.
        Show that for every $i\in I$ we have
            $\preimg{f(i)}{A\inter B}\in\opens[X(i)]$.
        \begin{subproof}
            Fix $i\in I$.
            $X(i)$ is a topological space and
                $f(i)\in\funs{\carrier[X(i)]}{T}$ by
                \obv{final_topology_data_at}.
            $A,B\in\preimageOpenFamily{X(i)}{T}{f(i)}$ by
                \obv{preimage_open_family,final_opens}.
            $\preimageOpenFamily{X(i)}{T}{f(i)}$ is a topology on $T$ by
                \obv{preimage_open_family_is_topology}.
            Hence $A\inter B\in\preimageOpenFamily{X(i)}{T}{f(i)}$ by
                \obv{topology_family}.
            Follows by \obv{preimage_open_family}.
        \end{subproof}
        Follows by \obv{final_opens}.
    \end{subproof}

    Show that for every $F\subseteq\finalOpens{T}{I}{X}{f}$ we have
        $\unions{F}\in\finalOpens{T}{I}{X}{f}$.
    \begin{subproof}
        Fix $F\subseteq\finalOpens{T}{I}{X}{f}$.
        For every $A\in F$ we have $A\subseteq T$ by
            \obv{final_opens,powerset_elim,subseteq}.
        Hence $\unions{F}\subseteq T$ by \obv{unions_family}.
        Thus $\unions{F}\in\pow{T}$ by \obv{powerset_intro}.
        Show that for every $i\in I$ we have
            $\preimg{f(i)}{\unions{F}}\in\opens[X(i)]$.
        \begin{subproof}
            Fix $i\in I$.
            $X(i)$ is a topological space and
                $f(i)\in\funs{\carrier[X(i)]}{T}$ by
                \obv{final_topology_data_at}.
            Show that $F\subseteq\preimageOpenFamily{X(i)}{T}{f(i)}$.
            \begin{subproof}
                It suffices to show that every element of $F$ is an element
                    of $\preimageOpenFamily{X(i)}{T}{f(i)}$ by
                    \obv{subseteq}.
                Fix $A\in F$.
                Hence $A\in\finalOpens{T}{I}{X}{f}$ by \obv{subseteq}.
                Hence $A\in\pow{T}$ and
                    $\preimg{f(i)}{A}\in\opens[X(i)]$ by
                    \obv{final_opens}.
                Follows by \obv{preimage_open_family}.
            \end{subproof}
            $\preimageOpenFamily{X(i)}{T}{f(i)}$ is a topology on $T$ by
                \obv{preimage_open_family_is_topology}.
            Hence $\unions{F}\in\preimageOpenFamily{X(i)}{T}{f(i)}$ by
                \obv{topology_family}.
            Follows by \obv{preimage_open_family}.
        \end{subproof}
        Follows by \obv{final_opens}.
    \end{subproof}
    Follows by \obv{topology_family}.
\end{proof}

\begin{definition}\label{final_topology}
    $\FinalTopology{T}{I}{X}{f}
        =\TopologicalSpaceOn{T}{\finalOpens{T}{I}{X}{f}}$.
\end{definition}

\begin{classification}\label{final_topology_is_topological_space}
    Suppose $f$ is a final mapping family from $X$ to $T$ indexed by $I$.
    Then $\FinalTopology{T}{I}{X}{f}$ is a topological space.
\end{classification}
\begin{proof}
    $\finalOpens{T}{I}{X}{f}$ is a topology on $T$ by
        \obv{final_opens_is_topology}.
    Follows by
        \auto{final_topology,topological_space_on_is_topological_space}.
\end{proof}

\begin{theorem}\label{final_topology_is_exact}
    Suppose $f$ is a final mapping family from $X$ to $T$ indexed by $I$.
    Then $\FinalTopology{T}{I}{X}{f}$ is an exact topological space.
\end{theorem}
\begin{proof}
    $\finalOpens{T}{I}{X}{f}$ is a topology on $T$ by
        \obv{final_opens_is_topology}.
    Follows by
        \auto{final_topology,topological_space_on_is_exact}.
\end{proof}

\begin{proposition}\label{final_topology_carrier}
    $\carrier[\FinalTopology{T}{I}{X}{f}]=T$.
\end{proposition}
\begin{proof}
    Follows by \auto{final_topology,topological_space_on_carrier}.
\end{proof}

\begin{proposition}\label{final_topology_opens}
    $\opens[\FinalTopology{T}{I}{X}{f}]=\finalOpens{T}{I}{X}{f}$.
\end{proposition}
\begin{proof}
    Follows by \auto{final_topology,topological_space_on_opens}.
\end{proof}

\begin{theorem}\label{final_topology_structure_map_continuous}
    Suppose $f$ is a final mapping family from $X$ to $T$ indexed by $I$.
    Let $i\in I$.
    Then $f(i)$ is continuous from $X(i)$ to
        $\FinalTopology{T}{I}{X}{f}$.
\end{theorem}
\begin{proof}
    $X(i)$ is a topological space and
        $f(i)\in\funs{\carrier[X(i)]}{T}$ by
        \obv{final_topology_data}.
    $\FinalTopology{T}{I}{X}{f}$ is a topological space by
        \obv{final_topology_is_topological_space}.
    $\carrier[\FinalTopology{T}{I}{X}{f}]=T$ by
        \obv{final_topology_carrier}.
    Hence $f(i)\in
        \funs{\carrier[X(i)]}{\carrier[\FinalTopology{T}{I}{X}{f}]}$.
    Show that for every $U\in\opens[\FinalTopology{T}{I}{X}{f}]$ we have
        $\preimg{f(i)}{U}\in\opens[X(i)]$.
    \begin{subproof}
        Fix $U\in\opens[\FinalTopology{T}{I}{X}{f}]$.
        Hence $U\in\finalOpens{T}{I}{X}{f}$ by
            \obv{final_topology_opens}.
        Follows by \obv{final_opens}.
    \end{subproof}
    Follows by \obv{continuous_map}.
\end{proof}

\begin{theorem}\label{final_topology_universal_property}
    Suppose $f$ is a final mapping family from $X$ to $T$ indexed by $I$.
    Let $Y$ be a topological space.
    Let $g\in\funs{T}{\carrier[Y]}$.
    Then $g$ is continuous from $\FinalTopology{T}{I}{X}{f}$ to $Y$ iff
        for every $i\in I$ we have
            $g\circ f(i)$ is continuous from $X(i)$ to $Y$.
\end{theorem}
\begin{proof}
    For every $i\in I$ we have $X(i)$ is a topological space and
        $f(i)\in\funs{\carrier[X(i)]}{T}$ by
        \obv{final_topology_data}.
    Show that if $g$ is continuous from $\FinalTopology{T}{I}{X}{f}$
        to $Y$, then for every $i\in I$ we have
        $g\circ f(i)$ is continuous from $X(i)$ to $Y$.
    \begin{subproof}
        Assume $g$ is continuous from $\FinalTopology{T}{I}{X}{f}$ to $Y$.
        Fix $i\in I$.
        $f(i)$ is continuous from $X(i)$ to
            $\FinalTopology{T}{I}{X}{f}$ by
            \obv{final_topology_structure_map_continuous}.
        Follows by \obv{continuous_composition}.
    \end{subproof}

    Show that if for every $i\in I$ we have
        $g\circ f(i)$ is continuous from $X(i)$ to $Y$, then
        $g$ is continuous from $\FinalTopology{T}{I}{X}{f}$ to $Y$.
    \begin{subproof}
        Assume for every $i\in I$ we have
            $g\circ f(i)$ is continuous from $X(i)$ to $Y$.
        $\FinalTopology{T}{I}{X}{f}$ is a topological space by
            \obv{final_topology_is_topological_space}.
        $\carrier[\FinalTopology{T}{I}{X}{f}]=T$ by
            \obv{final_topology_carrier}.
        Hence $g\in\funs{\carrier[\FinalTopology{T}{I}{X}{f}]}
            {\carrier[Y]}$.
        Show that for every $U\in\opens[Y]$ we have
            $\preimg{g}{U}\in\opens[\FinalTopology{T}{I}{X}{f}]$.
        \begin{subproof}
            Fix $U\in\opens[Y]$.
            $\preimg{g}{U}\subseteq T$ by
                \auto{preimg_subseteq_dom,funs,subseteq}.
            Hence $\preimg{g}{U}\in\pow{T}$ by \obv{powerset_intro}.
            Show that for every $i\in I$ we have
                $\preimg{f(i)}{\preimg{g}{U}}\in\opens[X(i)]$.
            \begin{subproof}
                Fix $i\in I$.
                $f(i)\in\funs{\carrier[X(i)]}{T}$ by
                    \obv{final_topology_data_at}.
                $g\circ f(i)$ is continuous from $X(i)$ to $Y$.
                $\preimg{g\circ f(i)}{U}\in\opens[X(i)]$ by
                    \obv{continuous_map}.
                $\preimg{g\circ f(i)}{U}
                    =\preimg{f(i)}{\preimg{g}{U}}$ by
                    \obv{preimage_composition}.
            \end{subproof}
            Hence $\preimg{g}{U}\in\finalOpens{T}{I}{X}{f}$ by
                \obv{final_opens}.
            Follows by \obv{final_topology_opens}.
        \end{subproof}
        Follows by \obv{continuous_map}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{theorem}\label{final_topology_is_finest}
    Suppose $f$ is a final mapping family from $X$ to $T$ indexed by $I$.
    Suppose $P$ is a topology on $T$.
    Suppose for every $i\in I$ we have
        $f(i)$ is continuous from $X(i)$ to $\TopologicalSpaceOn{T}{P}$.
    Then $P\subseteq\opens[\FinalTopology{T}{I}{X}{f}]$.
\end{theorem}
\begin{proof}
    Show that $P\subseteq\finalOpens{T}{I}{X}{f}$.
    \begin{subproof}
        It suffices to show that every element of $P$ is an element of
            $\finalOpens{T}{I}{X}{f}$ by \obv{subseteq}.
        Fix $U\in P$.
        $P\subseteq\pow{T}$ by \obv{topology_family}.
        Hence $U\in\pow{T}$ by \obv{subseteq}.
        Show that for every $i\in I$ we have
            $\preimg{f(i)}{U}\in\opens[X(i)]$.
        \begin{subproof}
            Fix $i\in I$.
            $f(i)$ is continuous from $X(i)$ to
                $\TopologicalSpaceOn{T}{P}$.
            $\opens[\TopologicalSpaceOn{T}{P}]=P$ by
                \obv{topological_space_on_opens}.
            Hence $U\in\opens[\TopologicalSpaceOn{T}{P}]$.
            Follows by \obv{continuous_map}.
        \end{subproof}
        Follows by \obv{final_opens}.
    \end{subproof}
    Follows by \obv{final_topology_opens}.
\end{proof}

\begin{proposition}\label{empty_final_opens}
    $\finalOpens{T}{\emptyset}{X}{f}=\pow{T}$.
\end{proposition}
\begin{proof}
    $\finalOpens{T}{\emptyset}{X}{f}\subseteq\pow{T}$ by
        \obv{final_opens,subseteq}.
    Show that $\pow{T}\subseteq\finalOpens{T}{\emptyset}{X}{f}$.
    \begin{subproof}
        It suffices to show that every element of $\pow{T}$ is an element of
            $\finalOpens{T}{\emptyset}{X}{f}$ by \obv{subseteq}.
        Fix $U\in\pow{T}$.
        Show that for every $i\in\emptyset$ we have
            $\preimg{f(i)}{U}\in\opens[X(i)]$.
        \begin{subproof}
            Fix $i\in\emptyset$.
            Thus $i\in\emptyset$.
            Contradiction by \obv{emptyset}.
        \end{subproof}
        Follows by \obv{final_opens}.
    \end{subproof}
    Follows by \obv{subseteq_antisymmetric}.
\end{proof}

\begin{theorem}\label{empty_final_topology_is_discrete}
    $\FinalTopology{T}{\emptyset}{X}{f}=\discreteTopology{T}$.
\end{theorem}
\begin{proof}
    Follows by
        \auto{final_topology,empty_final_opens,discrete_topology,
            discrete_opens}.
\end{proof}
