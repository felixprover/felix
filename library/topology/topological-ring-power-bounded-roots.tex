\import{nat/divisibility.tex}

\section{Roots of power-bounded elements}

\begin{theorem}\label{power_bounded_root}
    Let $R$ be a commutative topological ring and $z\in\carrier[R]$.
    Let $m$ be a natural number such that $m\neq\emptyset$.
    Suppose $\ringPower{R}{z}{m}$ is power bounded in $R$.
    Then $z$ is power bounded in $R$.
\end{theorem}
\begin{proof}
    $R$ is a commutative ring by
        \obv{commutative_topological_ring_is_commutative_ring}.
    Hence $R$ is a ring by \obv{commutative_ring_is_ring}.
    Let $M=\MultiplicativeMonoid{R}$.
    Let $y=\ringPower{R}{z}{m}$.
    Let $s=\Apply{\graphOnOp}{
        \naturals,\carrier[R],
        \fun{n:\Set}{\ringPower{R}{z}{n}}}$.
    Let $E=\pointwiseSubsetPowers{M}{\{y\}}$.
    Let $F=\img{s}{m}$.
    Let $B=\pointwiseSubsetPowers{M}{\{z\}}$.
    $M$ is a monoid by \obv{multiplicative_monoid_is_exact_monoid}.
    $\carrier[M]=\carrier[R]$ and $\mul[M]=\ringMul[R]$ by
        \auto{multiplicative_monoid_fields}.
    Hence $z\in\carrier[M]$.
    $\ringPower{R}{z}{m}\in\carrier[R]$ by \obv{ring_power_type}.
    Hence $y\in\carrier[R]$.
    Hence $y\in\carrier[M]$.
    $\PowerBounded{R}{\{\ringPower{R}{z}{m}\}}$ by
        \obv{power_bounded_element}.
    Hence $\PowerBounded{R}{\{y\}}$.
    Hence $\Bounded{R}{E}$ by
        \auto{power_bounded_set_powers_bounded}.

    $s\in\funs{\naturals}{\carrier[R]}$ by
        \auto{logical_graph_on_function,ring_power_type}.
    Hence $s$ is a function and $\dom{s}=\naturals$ and
        $\ran{s}\subseteq\carrier[R]$ by \auto{funs_elim,funs_ran}.
    $m$ is finite by \obv{natural_number_is_finite}.
    $m\subseteq\naturals$ by
        \auto{naturals_is_transitive_set,transitiveset_iff_subseteq}.
    Hence $m\subseteq\dom{s}$.
    Hence $F$ is finite by \obv{finite_function_image}.
    $\img{s}{m}\subseteq\ran{s}$ by \obv{img_subseteq_ran}.
    Hence $F\subseteq\ran{s}$.
    Hence $F\subseteq\carrier[R]$ by \obv{subseteq_transitive}.
    Hence $\Bounded{R}{F}$ by \obv{finite_subset_bounded}.
    Hence $\Bounded{R}{\pointwiseProduct{
        \MultiplicativeMonoid{R}}{E}{F}}$ by
        \obv{bounded_pointwise_product}.
    Hence $\Bounded{R}{\pointwiseProduct{M}{E}{F}}$.

    Show that every element of $B$ is an element of
        $\pointwiseProduct{M}{E}{F}$.
    \begin{subproof}
        Fix $a\in B$.
        Take a natural number $n$ such that
            $a\in\pointwiseSubsetPower{M}{\{z\}}{n}$ by
            \obv{pointwise_subset_powers_iff}.
        $\pointwiseSubsetPower{M}{\{z\}}{n}
            =\{\monoidPower{M}{z}{n}\}$ by
            \obv{pointwise_subset_power_singleton}.
        Hence $a=\monoidPower{M}{z}{n}$ by \auto{singleton_iff}.
        $\ringPower{R}{z}{n}=\monoidPower{M}{z}{n}$ by
            \obv{ring_power}.
        Hence $a=\ringPower{R}{z}{n}$.

        Take $q,r\in\naturals$ such that
            $n=\naturalMul{m}{q}+r$ and $r\in m$ by
            \obv{natural_division_algorithm}.
        $s(r)=\ringPower{R}{z}{r}$ by
            \auto{logical_graph_on_apply}.
        Hence $\ringPower{R}{z}{r}\in F$ by
            \auto{img_iff,function_apply_iff}.

        $\pointwiseSubsetPower{M}{\{y\}}{q}
            =\{\monoidPower{M}{y}{q}\}$ by
            \obv{pointwise_subset_power_singleton}.
        $\ringPower{R}{y}{q}=\monoidPower{M}{y}{q}$ by
            \obv{ring_power}.
        Hence $\ringPower{R}{y}{q}
            \in\pointwiseSubsetPower{M}{\{y\}}{q}$ by
            \auto{singleton_iff}.
        Hence $\ringPower{R}{y}{q}\in E$ by
            \auto{pointwise_subset_powers_iff}.

        $\ringPower{R}{z}{\naturalMul{m}{q}}
            =\ringPower{R}{\ringPower{R}{z}{m}}{q}$ by
            \obv{ring_power_multiplication}.
        Hence $\ringPower{R}{z}{\naturalMul{m}{q}}
            =\ringPower{R}{y}{q}$.
        $\naturalMul{m}{q}$ is a natural number by
            \obv{natural_multiplication_closed}.
        $\ringPower{R}{z}{\naturalMul{m}{q}+r}
            =\ringMul[R](
                \ringPower{R}{z}{\naturalMul{m}{q}},
                \ringPower{R}{z}{r})$ by
            \obv{ring_power_addition}.
        Hence $a=\mul[M](
            \ringPower{R}{y}{q},\ringPower{R}{z}{r})$.
        Follows by \auto{pointwise_product_iff}.
    \end{subproof}
    Hence $B\subseteq\pointwiseProduct{M}{E}{F}$ by \obv{subseteq}.
    Hence $\Bounded{R}{B}$ by \obv{bounded_downward}.
    $\{z\}\subseteq\carrier[R]$ by \auto{subseteq,singleton_iff}.
    Hence $\PowerBounded{R}{\{z\}}$ by \auto{power_bounded_set}.
    Follows by \obv{power_bounded_element}.
\end{proof}
