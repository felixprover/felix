\section{Weakening coordinate structures}

\import{topology/coordinate-homeomorphism.tex}

Enlarging the pseudogroup preserves the original charts and allows new ones.
The resulting structure stores the maximal atlas generated under the larger
pseudogroup; the original maximal atlas need not already be maximal there.
The ambient topology is unchanged.

\begin{construction}\label{weakened_coordinates}
    Let $G,H$ be coordinate pseudogroups and $X$ be a topological space.
    Let $S$ be a relative coordinate structure on $X$ under $G$.
    Suppose $G$ is a subpseudogroup of $H$.
    The relative coordinate structure $\WeakenedCoordinates{G}{H}{X}{S}=
        \structure{\atlas=\MaxAtlas{H}{X}{\atlas[S]}}$ on $X$ under $H$.
\end{construction}
\begin{proof}
    $\atlas[S]$ is a coordinate atlas on $X$ under $G$ by
        \auto{relative_atlas_maximal,maximal_coordinate_atlas}.
    $\atlas[S]$ is a coordinate atlas on $X$ under $H$ by \obv{subpseudogroup_atlas}.
    $\MaxAtlas{H}{X}{\atlas[S]}$ is a maximal coordinate atlas on $X$ under $H$ by
        \obv{max_atlas_is_maximal}.
    Follows by \auto{relative_coordinate_structureaggregateprojection_atlas}.
\end{proof}

\begin{theorem}\label{weakened_coordinates_is_exact}
    Let $G,H$ be coordinate pseudogroups and $X$ be a topological space.
    Let $S$ be a relative coordinate structure on $X$ under $G$.
    Suppose $G$ is a subpseudogroup of $H$.
    Then $\WeakenedCoordinates{G}{H}{X}{S}$ is an exact relative coordinate structure
        on $X$ under $H$.
\end{theorem}
\begin{proof}
    Follows by \auto{weakened_coordinates,weakened_coordinatesshape}.
\end{proof}

\begin{theorem}\label{weakened_coordinates_chart_inclusion}
    Let $G,H$ be coordinate pseudogroups and $X$ be a topological space.
    Let $S$ be a relative coordinate structure on $X$ under $G$.
    Suppose $G$ is a subpseudogroup of $H$.
    Then $\atlas[S]\subseteq\atlas[\WeakenedCoordinates{G}{H}{X}{S}]$.
\end{theorem}
\begin{proof}
    $\atlas[S]$ is a coordinate atlas on $X$ under $G$ by
        \auto{relative_atlas_maximal,maximal_coordinate_atlas}.
    $\atlas[S]$ is a coordinate atlas on $X$ under $H$ by \obv{subpseudogroup_atlas}.
    Follows by \auto{weakened_coordinatesprojection_atlas,atlas_subset_max_atlas}.
\end{proof}

\subsection{Computation from a presentation}

Any original presentation also presents the weakened structure under the
larger pseudogroup. In particular, weakening is independent of the choice
of a presenting atlas.

\begin{theorem}\label{weakened_coordinates_presentation}
    Let $G,H$ be coordinate pseudogroups and $X$ be a topological space.
    Let $S$ be a coordinate datum.
    Suppose $A$ presents $S$ on $X$ under $G$ and $G$ is a subpseudogroup of $H$.
    Then $A$ presents $\WeakenedCoordinates{G}{H}{X}{S}$ on $X$ under $H$.
\end{theorem}
\begin{proof}
    $S$ is a relative coordinate structure on $X$ under $G$ by \obv{presented_coordinates}.
    $A$ is a coordinate atlas on $X$ under $G$ and $\atlas[S]=\MaxAtlas{G}{X}{A}$ by
        \obv{coordinate_presentation}.
    $A$ is a coordinate atlas on $X$ under $H$ by \obv{subpseudogroup_atlas}.
    $\WeakenedCoordinates{G}{H}{X}{S}$ is a coordinate datum by
        \auto{weakened_coordinates,relative_coordinate_structure}.
    $\atlas[\WeakenedCoordinates{G}{H}{X}{S}]=\MaxAtlas{H}{X}{A}$ by
        \auto{weakened_coordinatesprojection_atlas,subpseudogroup_max_atlas_regeneration}.
    Follows by \obv{coordinate_presentation}.
\end{proof}

\begin{theorem}\label{weakened_coordinates_from_presentation}
    Let $G,H$ be coordinate pseudogroups and $X$ be a topological space.
    Let $S$ be a coordinate datum.
    Suppose $A$ presents $S$ on $X$ under $G$ and $G$ is a subpseudogroup of $H$.
    Then $\WeakenedCoordinates{G}{H}{X}{S}=\GeneratedCoordinates{H}{X}{A}$.
\end{theorem}
\begin{proof}
    $S$ is a relative coordinate structure on $X$ under $G$ by \obv{presented_coordinates}.
    $\WeakenedCoordinates{G}{H}{X}{S}$ is an exact coordinate datum by
        \auto{weakened_coordinates_is_exact,weakened_coordinatesshape,relative_coordinate_structure}.
    $A$ presents $\WeakenedCoordinates{G}{H}{X}{S}$ on $X$ under $H$ by
        \obv{weakened_coordinates_presentation}.
    Follows by \obv{exact_presented_coordinates}.
\end{proof}

\subsection{Identity and successive weakening}

Weakening retains only the atlas. Consequently, identity weakening recovers
an exact relative structure, not any additional fields of a richer record.

\begin{theorem}\label{weakened_coordinates_identity}
    Let $G$ be a coordinate pseudogroup and $X$ be a topological space.
    Suppose $S$ is an exact relative coordinate structure on $X$ under $G$.
    Then $\WeakenedCoordinates{G}{G}{X}{S}=S$.
\end{theorem}
\begin{proof}
    $G$ is a subpseudogroup of $G$ by \obv{subpseudogroup_reflexive}.
    $\WeakenedCoordinates{G}{G}{X}{S}$ is an exact relative coordinate structure
        on $X$ under $G$ by \obv{weakened_coordinates_is_exact}.
    $\atlas[S]$ is a maximal coordinate atlas on $X$ under $G$ by \obv{relative_atlas_maximal}.
    $\atlas[S]=\MaxAtlas{G}{X}{\atlas[S]}$ by
        \auto{maximal_coordinate_atlas_fixed_point,maximal_coordinate_atlas}.
    $\atlas[\WeakenedCoordinates{G}{G}{X}{S}]=\atlas[S]$ by
        \auto{weakened_coordinatesprojection_atlas}.
    Follows by structure extensionality.
\end{proof}

\begin{theorem}\label{weakened_coordinates_transitive}
    Let $G,H,K$ be coordinate pseudogroups and $X$ be a topological space.
    Let $S$ be a relative coordinate structure on $X$ under $G$.
    Suppose $G$ is a subpseudogroup of $H$ and $H$ is a subpseudogroup of $K$.
    Then $\WeakenedCoordinates{H}{K}{X}{\WeakenedCoordinates{G}{H}{X}{S}}
        =\WeakenedCoordinates{G}{K}{X}{S}$.
\end{theorem}
\begin{proof}
    $G$ is a subpseudogroup of $K$ by \obv{subpseudogroup_transitive}.
    $\atlas[S]$ presents $S$ on $X$ under $G$ by \obv{relative_atlas_presentation}.
    $\atlas[S]$ presents $\WeakenedCoordinates{G}{H}{X}{S}$ on $X$ under $H$ by
        \obv{weakened_coordinates_presentation}.
    $\WeakenedCoordinates{H}{K}{X}{\WeakenedCoordinates{G}{H}{X}{S}}
        =\GeneratedCoordinates{K}{X}{\atlas[S]}$ by \obv{weakened_coordinates_from_presentation}.
    $\WeakenedCoordinates{G}{K}{X}{S}=\GeneratedCoordinates{K}{X}{\atlas[S]}$ by
        \obv{weakened_coordinates_from_presentation}.
\end{proof}

\begin{theorem}\label{weakened_coordinates_same_fields}
    Let $G,H$ be coordinate pseudogroups and $X,Y$ be topological spaces.
    Let $S$ be a relative coordinate structure on $X$ under $G$.
    Let $T$ be a relative coordinate structure on $Y$ under $G$.
    Suppose $G$ is a subpseudogroup of $H$.
    Suppose $\carrier[X]=\carrier[Y]$ and $\opens[X]=\opens[Y]$ and $\atlas[S]=\atlas[T]$.
    Then $\WeakenedCoordinates{G}{H}{X}{S}=\WeakenedCoordinates{G}{H}{Y}{T}$.
\end{theorem}
\begin{proof}
    $\WeakenedCoordinates{G}{H}{X}{S}$ is an exact relative coordinate structure
        on $X$ under $H$ by \obv{weakened_coordinates_is_exact}.
    $\WeakenedCoordinates{G}{H}{Y}{T}$ is an exact relative coordinate structure
        on $Y$ under $H$ by \obv{weakened_coordinates_is_exact}.
    $\compatibleHull{H}{X}{\atlas[S]}=\compatibleHull{H}{Y}{\atlas[T]}$ by
        \auto{compatible_hull_same_fields}.
    $\atlas[\WeakenedCoordinates{G}{H}{X}{S}]=\atlas[\WeakenedCoordinates{G}{H}{Y}{T}]$ by
        \auto{weakened_coordinatesprojection_atlas,max_atlas}.
    Follows by structure extensionality.
\end{proof}

\subsection{Equipping and extraction}

\begin{theorem}\label{equipped_weakened_coordinates}
    Let $G,H$ be coordinate pseudogroups and $X$ be a topological space.
    Let $S$ be a relative coordinate structure on $X$ under $G$.
    Suppose $G$ is a subpseudogroup of $H$.
    Then $\Equip{X}{\WeakenedCoordinates{G}{H}{X}{S}}$ has coordinates under $H$ and
        $\TopologicalReduct{\Equip{X}{\WeakenedCoordinates{G}{H}{X}{S}}}=\TopologicalReduct{X}$ and
        $\CoordinateStructure{\Equip{X}{\WeakenedCoordinates{G}{H}{X}{S}}}
            =\WeakenedCoordinates{G}{H}{X}{S}$.
\end{theorem}
\begin{proof}
    $\WeakenedCoordinates{G}{H}{X}{S}$ is a relative coordinate structure on $X$ under $H$ by
        \obv{weakened_coordinates}.
    $\WeakenedCoordinates{G}{H}{X}{S}$ is an exact coordinate datum by
        \auto{weakened_coordinates_is_exact,weakened_coordinatesshape,relative_coordinate_structure}.
    Follows by \obv{equip_preserves_coordinates,equip_topological_reduct,coordinate_structure_of_equip}.
\end{proof}

\begin{theorem}\label{weakened_coordinates_of_equip}
    Let $G,H$ be coordinate pseudogroups and $X$ be a topological space.
    Let $S$ be a relative coordinate structure on $X$ under $G$.
    Suppose $G$ is a subpseudogroup of $H$.
    Then $\WeakenedCoordinates{G}{H}{\TopologicalReduct{\Equip{X}{S}}}{\CoordinateStructure{\Equip{X}{S}}}
        =\WeakenedCoordinates{G}{H}{X}{S}$.
\end{theorem}
\begin{proof}
    $S$ is a coordinate datum by \obv{relative_coordinate_structure}.
    Let $M=\Equip{X}{S}$ and $T=\CoordinateStructure{M}$.
    $M$ is an equipped coordinate space and $M$ has coordinates under $G$ by
        \auto{equip,equip_preserves_coordinates}.
    $\TopologicalReduct{M}$ is a topological space by \obv{topological_reduct_is_topological_space}.
    $T$ is a relative coordinate structure on $\TopologicalReduct{M}$ under $G$ by
        \auto{coordinate_structure_relative}.
    $\carrier[\TopologicalReduct{M}]=\carrier[X]$ and
        $\opens[\TopologicalReduct{M}]=\opens[X]$ by
        \auto{topological_reduct_carrier,topological_reduct_opens,equipprojection_carrier,equipprojection_opens}.
    $\atlas[T]=\atlas[S]$ by \auto{coordinate_structureprojection_atlas,equipprojection_atlas}.
    Follows by \auto{weakened_coordinates_same_fields}.
\end{proof}

\subsection{Open restriction and transport}

Both orders have the same presentation under the larger pseudogroup.
The equalities below compare exact relative structures on the same space.

\begin{theorem}\label{weakened_coordinates_restriction}
    Let $G,H$ be coordinate pseudogroups and $X$ be a topological space.
    Let $S$ be a relative coordinate structure on $X$ under $G$.
    Suppose $G$ is a subpseudogroup of $H$ and $U$ is open in $X$.
    Then $\WeakenedCoordinates{G}{H}{\subspaceTopology{X}{U}}{\RestrictedCoordinates{G}{X}{S}{U}}
        =\RestrictedCoordinates{H}{X}{\WeakenedCoordinates{G}{H}{X}{S}}{U}$.
\end{theorem}
\begin{proof}
    $\atlas[S]$ presents $S$ on $X$ under $G$ by \obv{relative_atlas_presentation}.
    $\RestrictedAtlas{\atlas[S]}{U}$ presents $\RestrictedCoordinates{G}{X}{S}{U}$
        on $\subspaceTopology{X}{U}$ under $G$ by \obv{restricted_coordinates_presentation}.
    $\WeakenedCoordinates{G}{H}{\subspaceTopology{X}{U}}{\RestrictedCoordinates{G}{X}{S}{U}}
        =\GeneratedCoordinates{H}{\subspaceTopology{X}{U}}{\RestrictedAtlas{\atlas[S]}{U}}$ by
        \obv{weakened_coordinates_from_presentation}.
    $\atlas[S]$ presents $\WeakenedCoordinates{G}{H}{X}{S}$ on $X$ under $H$ by
        \obv{weakened_coordinates_presentation}.
    $\RestrictedCoordinates{H}{X}{\WeakenedCoordinates{G}{H}{X}{S}}{U}
        =\GeneratedCoordinates{H}{\subspaceTopology{X}{U}}{\RestrictedAtlas{\atlas[S]}{U}}$ by
        \obv{restricted_coordinates_from_presentation}.
\end{proof}

\begin{theorem}\label{weakened_coordinates_transport}
    Let $G,H$ be coordinate pseudogroups and $X,Y$ be topological spaces.
    Let $S$ be a relative coordinate structure on $X$ under $G$.
    Suppose $G$ is a subpseudogroup of $H$ and $h$ is a homeomorphism from $X$ to $Y$.
    Then $\WeakenedCoordinates{G}{H}{Y}{\TransportedCoordinates{G}{X}{Y}{S}{h}}
        =\TransportedCoordinates{H}{X}{Y}{\WeakenedCoordinates{G}{H}{X}{S}}{h}$.
\end{theorem}
\begin{proof}
    $\atlas[S]$ presents $S$ on $X$ under $G$ by \obv{relative_atlas_presentation}.
    $\TransportedAtlas{\atlas[S]}{h}$ presents $\TransportedCoordinates{G}{X}{Y}{S}{h}$
        on $Y$ under $G$ by \obv{transported_coordinates_presentation}.
    $\WeakenedCoordinates{G}{H}{Y}{\TransportedCoordinates{G}{X}{Y}{S}{h}}
        =\GeneratedCoordinates{H}{Y}{\TransportedAtlas{\atlas[S]}{h}}$ by
        \obv{weakened_coordinates_from_presentation}.
    $\atlas[S]$ presents $\WeakenedCoordinates{G}{H}{X}{S}$ on $X$ under $H$ by
        \obv{weakened_coordinates_presentation}.
    $\TransportedCoordinates{H}{X}{Y}{\WeakenedCoordinates{G}{H}{X}{S}}{h}
        =\GeneratedCoordinates{H}{Y}{\TransportedAtlas{\atlas[S]}{h}}$ by
        \obv{transported_coordinates_from_presentation}.
\end{proof}

\subsection{Admissible maps survive weakening}

The old charts remain available, so their coordinate expressions still check
the map. Compatibility of the larger pseudogroups with the calculus is a
separate hypothesis; inclusion alone does not provide it.

\begin{theorem}\label{weakened_coordinates_admissible}
    Let $C$ be a local-map calculus and $G,H,K,L$ be coordinate pseudogroups.
    Let $X,Y$ be topological spaces.
    Let $S$ be a relative coordinate structure on $X$ under $G$.
    Let $T$ be a relative coordinate structure on $Y$ under $H$.
    Suppose $G$ is a subpseudogroup of $K$ and $H$ is a subpseudogroup of $L$.
    Suppose $K$ is compatible with the calculus $C$ and $L$ is compatible with the calculus $C$.
    Suppose $f$ is coordinate-admissible from $\Equip{X}{S}$ under $G$ to $\Equip{Y}{T}$ under $H$ in $C$.
    Then $f$ is coordinate-admissible
        from $\Equip{X}{\WeakenedCoordinates{G}{K}{X}{S}}$ under $K$
        to $\Equip{Y}{\WeakenedCoordinates{H}{L}{Y}{T}}$ under $L$ in $C$.
\end{theorem}
\begin{proof}
    Let $P=\WeakenedCoordinates{G}{K}{X}{S}$ and $Q=\WeakenedCoordinates{H}{L}{Y}{T}$.
    $P$ is a relative coordinate structure on $X$ under $K$ and
        $Q$ is a relative coordinate structure on $Y$ under $L$ by \auto{weakened_coordinates}.
    $S$ is a coordinate datum and $T$ is a coordinate datum and
        $P$ is a coordinate datum and $Q$ is a coordinate datum by \obv{relative_coordinate_structure}.
    $G$ is compatible with the calculus $C$ and $H$ is compatible with the calculus $C$ by
        \obv{coordinate_admissible_map}.
    $\dom{f}\in\opens[X]$ and $f\in\funs{\dom{f}}{\carrier[Y]}$ by
        \auto{coordinate_admissible_equip_iff}.
    $\atlas[S]\subseteq\atlas[P]$ and $\atlas[T]\subseteq\atlas[Q]$ by
        \auto{weakened_coordinates_chart_inclusion}.
    $\model[G]=\model[K]$ and $\model[H]=\model[L]$ by \obv{subpseudogroup}.
    Show that for every $p\in\dom{f}$ there exists $c\in\ChartsAt{P}{p}$ such that
        there exists $d\in\atlas[Q]$ such that there exists $U\in\opens[X]$ such that
            $p\in U$ and $U\subseteq\dom{f}$ and $U\subseteq\dom{c}$ and
            $\img{f}{U}\subseteq\dom{d}$ and
            $\CoordExpr{d}{f}{c}{U}$ is admissible from $\model[K]$ to $\model[L]$ in $C$.
    \begin{subproof}
        Fix $p\in\dom{f}$.
        Take $c,d,U$ such that $c\in\ChartsAt{S}{p}$ and $d\in\atlas[T]$ and
            $U\in\opens[X]$ and $p\in U$ and $U\subseteq\dom{f}$ and $U\subseteq\dom{c}$ and
            $\img{f}{U}\subseteq\dom{d}$ and
            $\CoordExpr{d}{f}{c}{U}$ is admissible from $\model[G]$ to $\model[H]$ in $C$ by
            \auto{coordinate_admissible_equip_iff}.
        $c\in\ChartsAt{P}{p}$ and $d\in\atlas[Q]$ by \auto{charts_at_iff,subseteq}.
        Follows by \auto.
    \end{subproof}
    Follows by \auto{coordinate_admissible_equip_iff}.
\end{proof}
