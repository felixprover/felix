\section{Increasing partial maps of the real line}

\import{topology/partial-homeomorphism.tex}
\import{topology/real-line.tex}

Increasing maps preserve strict order on their domains. For disconnected
domains this is a global condition; the coordinate pseudogroup will instead
require it only near each domain point.

\begin{definition}\label{increasing_real_map}
    $f$ is an increasing real map iff
        $f$ is a function and $\dom{f}\subseteq\reals$ and $\ran{f}\subseteq\reals$ and
        for all $x,y\in\dom{f}$ such that $\surrealLt{x}{y}$ we have
            $\surrealLt{f(x)}{f(y)}$.
\end{definition}

\begin{theorem}\label{increasing_real_map_restriction}
    Suppose $f$ is an increasing real map.
    Then $\restrl{f}{A}$ is an increasing real map.
\end{theorem}
\begin{proof}
    $f$ is a function and $\dom{f}\subseteq\reals$ and $\ran{f}\subseteq\reals$ by
        \obv{increasing_real_map}.
    Let $g=\restrl{f}{A}$.
    $g$ is a function by \obv{restrl_of_function_is_function}.
    $\dom{g}=\dom{f}\inter A$ by \auto{dom_of_restrl_eq_inter}.
    $\dom{g}\subseteq\dom{f}$ by \auto{inter_lower_left}.
    $\dom{g}\subseteq\reals$ by \obv{subseteq_transitive}.
    $\ran{g}=\img{f}{A}$ by \auto{restriction_range}.
    $\img{f}{A}\subseteq\ran{f}$ by \obv{img_subseteq_ran}.
    $\ran{g}\subseteq\reals$ by \auto{subseteq_transitive}.
    Show that for all $x,y\in\dom{g}$ such that $\surrealLt{x}{y}$ we have
        $\surrealLt{g(x)}{g(y)}$.
    \begin{subproof}
        Fix $x,y\in\dom{g}$ such that $\surrealLt{x}{y}$.
        $x,y\in\dom{f}$ and $x,y\in A$ by \auto{inter}.
        $g(x)=f(x)$ and $g(y)=f(y)$ by \auto{restriction_apply_on_domain}.
        Follows by \auto{increasing_real_map}.
    \end{subproof}
    Follows by \auto{increasing_real_map}.
\end{proof}

\begin{theorem}\label{identity_increasing_real_map}
    Suppose $A\subseteq\reals$.
    Then $\identity{A}$ is an increasing real map.
\end{theorem}
\begin{proof}
    $\identity{A}$ is a function by \obv{id_is_function}.
    $\dom{\identity{A}}=A$ and $\ran{\identity{A}}=A$ by \obv{id_dom,id_ran}.
    For all $x,y\in A$ such that $\surrealLt{x}{y}$ we have
        $\surrealLt{(\identity{A})(x)}{(\identity{A})(y)}$ by \auto{id_apply}.
    Follows by \obv{increasing_real_map}.
\end{proof}

\begin{theorem}\label{increasing_real_map_injective}
    Suppose $f$ is an increasing real map.
    Then $f$ is injective.
\end{theorem}
\begin{proof}
    $f$ is a function and $\dom{f}\subseteq\reals$ and $\ran{f}\subseteq\reals$ by
        \obv{increasing_real_map}.
    Show that for all $x,y,z$ such that $(x,z)\in f$ and $(y,z)\in f$ we have $x=y$.
    \begin{subproof}
        Fix $x,y,z$ such that $(x,z)\in f$ and $(y,z)\in f$.
        $x,y\in\dom{f}$ and $z\in\ran{f}$ by \auto{dom_iff,ran_iff}.
        $x,y,z\in\reals$ by \obv{subseteq}.
        $\surreal{x}$ and $\surreal{y}$ and $\surreal{z}$ by
            \obv{surreal_real_is_surreal}.
        $f(x)=z$ and $f(y)=z$ by \obv{function_apply_intro}.
        $\surrealLt{x}{y}$ or $x=y$ or $\surrealLt{y}{x}$ by
            \obv{surreal_lt_trichotomy}.
        Not $\surrealLt{z}{z}$ by \obv{surreal_lt_irreflexive}.
        Follows by \auto{increasing_real_map}.
    \end{subproof}
    Follows by \obv{injective}.
\end{proof}

\begin{theorem}\label{increasing_real_map_order_iff}
    Suppose $f$ is an increasing real map.
    Let $x,y\in\dom{f}$.
    Then $\surrealLt{x}{y}$ iff $\surrealLt{f(x)}{f(y)}$.
\end{theorem}
\begin{proof}
    $f$ is a function and $\dom{f}\subseteq\reals$ and $\ran{f}\subseteq\reals$ by
        \obv{increasing_real_map}.
    $x,y\in\reals$ by \obv{subseteq}.
    $f(x),f(y)\in\ran{f}$ by \auto{fun_ran_iff}.
    $f(x),f(y)\in\reals$ by \obv{subseteq}.
    $\surreal{x}$ and $\surreal{y}$ and $\surreal{f(x)}$ and $\surreal{f(y)}$ by
        \obv{surreal_real_is_surreal}.
    $\surrealLt{x}{y}$ or $x=y$ or $\surrealLt{y}{x}$ by
        \obv{surreal_lt_trichotomy}.
    Follows by \auto{increasing_real_map,surreal_lt_asymmetric,surreal_lt_irreflexive}.
\end{proof}

\begin{theorem}\label{increasing_real_map_converse}
    Suppose $f$ is an increasing real map.
    Then $\converse{f}$ is an increasing real map.
\end{theorem}
\begin{proof}
    $f$ is a function and $\dom{f}\subseteq\reals$ and $\ran{f}\subseteq\reals$ by
        \obv{increasing_real_map}.
    $f$ is injective by \obv{increasing_real_map_injective}.
    Let $g=\converse{f}$.
    $g$ is a function by \obv{injective_converse_is_function}.
    $\dom{g}=\ran{f}$ and $\ran{g}=\dom{f}$ by \auto{dom_converse,ran_converse}.
    Show that for all $u,v\in\dom{g}$ such that $\surrealLt{u}{v}$ we have
        $\surrealLt{g(u)}{g(v)}$.
    \begin{subproof}
        Fix $u,v\in\dom{g}$ such that $\surrealLt{u}{v}$.
        $(u,g(u)),(v,g(v))\in g$ by \obv{function_apply_elim}.
        $(g(u),u),(g(v),v)\in f$ by \auto{converse_iff}.
        $g(u),g(v)\in\dom{f}$ by \auto{dom_iff}.
        $f(g(u))=u$ and $f(g(v))=v$ by \obv{function_apply_intro}.
        Follows by \auto{increasing_real_map_order_iff}.
    \end{subproof}
    Follows by \auto{increasing_real_map}.
\end{proof}

\begin{theorem}\label{increasing_real_map_composition}
    Suppose $f$ and $g$ are increasing real maps.
    Then $g\circ f$ is an increasing real map.
\end{theorem}
\begin{proof}
    $f$ and $g$ are functions and $\dom{f}\subseteq\reals$ and $\ran{g}\subseteq\reals$
        by \obv{increasing_real_map}.
    $g\circ f$ is a function by \obv{function_circ}.
    $\dom{g\circ f}=\preimg{f}{\dom{g}}$ by \obv{relation_composition_domain}.
    $\preimg{f}{\dom{g}}\subseteq\dom{f}$ by \obv{preimg_subseteq_dom}.
    $\dom{g\circ f}\subseteq\reals$ by \auto{subseteq_transitive}.
    $\ran{g\circ f}=\img{g}{\ran{f}}$ by \obv{relation_composition_range}.
    $\img{g}{\ran{f}}\subseteq\ran{g}$ by \obv{img_subseteq_ran}.
    $\ran{g\circ f}\subseteq\reals$ by \auto{subseteq_transitive}.
    Show that for all $x,y\in\dom{g\circ f}$ such that $\surrealLt{x}{y}$ we have
        $\surrealLt{(g\circ f)(x)}{(g\circ f)(y)}$.
    \begin{subproof}
        Fix $x,y\in\dom{g\circ f}$ such that $\surrealLt{x}{y}$.
        $x,y\in\dom{f}$ and $f(x),f(y)\in\dom{g}$ by
            \auto{partial_composition_domain_iff}.
        $\surrealLt{f(x)}{f(y)}$ by \obv{increasing_real_map}.
        $\surrealLt{g(f(x))}{g(f(y))}$ by \obv{increasing_real_map}.
        $(g\circ f)(x)=g(f(x))$ and $(g\circ f)(y)=g(f(y))$ by
            \obv{partial_composition_apply}.
    \end{subproof}
    Follows by \obv{increasing_real_map}.
\end{proof}
