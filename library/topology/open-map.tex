\section{Open maps}

\import{topology/continuity.tex}

An open map sends every open subset of its displayed source to an open subset
of its displayed target.  As with continuity, the endpoint and function
typing are part of the predicate.

\begin{definition}\label{open_map}
    $f$ is an open map from $X$ to $Y$ iff
        $X$ is a topological space and
        $Y$ is a topological space and
        $f\in\funs{\carrier[X]}{\carrier[Y]}$ and
        for every $U\in\opens[X]$ we have $\img{f}{U}\in\opens[Y]$.
\end{definition}

\begin{classification}\label{open_map_image_open}
    Suppose $f$ is an open map from $X$ to $Y$.
    Let $U\in\opens[X]$.
    Then $\img{f}{U}\in\opens[Y]$.
\end{classification}
\begin{proof}
    Follows by \obv{open_map}.
\end{proof}

\begin{theorem}\label{identity_is_open_map}
    Let $X$ be a topological space.
    Then $\identity{\carrier[X]}$ is an open map from $X$ to $X$.
\end{theorem}
\begin{proof}
    $\identity{\carrier[X]}\in
        \funs{\carrier[X]}{\carrier[X]}$ by
        \obv{id_elem_funs}.
    Show that for every $U\in\opens[X]$ we have
        $\img{\identity{\carrier[X]}}{U}\in\opens[X]$.
    \begin{subproof}
        Fix $U\in\opens[X]$.
        $U\subseteq\carrier[X]$ by \obv{opens_type}.
        Hence $\carrier[X]\inter U=U$ by
            \obv{inter_absorb_supseteq_left}.
        $\img{\identity{\carrier[X]}}{U}
            =\carrier[X]\inter U$ by \obv{id_img}.
    \end{subproof}
    Follows by \obv{open_map}.
\end{proof}

\begin{theorem}\label{open_map_composition}
    Suppose $f$ is an open map from $X$ to $Y$ and
        $g$ is an open map from $Y$ to $Z$.
    Then $g\circ f$ is an open map from $X$ to $Z$.
\end{theorem}
\begin{proof}
    $f\in\funs{\carrier[X]}{\carrier[Y]}$ and
        $g\in\funs{\carrier[Y]}{\carrier[Z]}$ by
        \obv{open_map}.
    Hence $g\circ f\in\funs{\carrier[X]}{\carrier[Z]}$ by
        \obv{funs_circ}.
    Show that for every $U\in\opens[X]$ we have
        $\img{g\circ f}{U}\in\opens[Z]$.
    \begin{subproof}
        Fix $U\in\opens[X]$.
        $\img{f}{U}\in\opens[Y]$ by \obv{open_map_image_open}.
        Hence $\img{g}{\img{f}{U}}\in\opens[Z]$ by
            \obv{open_map_image_open}.
        $f$ and $g$ are functions by \obv{funs_is_function}.
        $\ran{f}\subseteq\carrier[Y]$ by \obv{funs_ran}.
        $\dom{g}=\carrier[Y]$ by \obv{funs}.
        Hence $\ran{f}\subseteq\dom{g}$.
        $\img{g\circ f}{U}=\img{g}{\img{f}{U}}$ by
            \obv{img_of_circ}.
    \end{subproof}
    Follows by \obv{open_map}.
\end{proof}

\begin{classification}\label{homeomorphism_is_open_map}
    Suppose $f$ is a homeomorphism from $X$ to $Y$.
    Then $f$ is an open map from $X$ to $Y$.
\end{classification}
\begin{proof}
    $f$ is continuous from $X$ to $Y$ by \obv{homeomorphism}.
    $Y$ is a topological space and
        $f\in\funs{\carrier[X]}{\carrier[Y]}$ by
        \obv{continuous_map}.
    $f\in\bijections{\carrier[X]}{\carrier[Y]}$ and
        $\converse{f}$ is continuous from $Y$ to $X$ by
        \obv{homeomorphism}.
    Show that for every $U\in\opens[X]$ we have
        $\img{f}{U}\in\opens[Y]$.
    \begin{subproof}
        Fix $U\in\opens[X]$.
        $\preimg{\converse{f}}{U}\in\opens[Y]$ by
            \obv{continuous_map}.
        $\preimg{\converse{f}}{U}
            =\img{\converse{\converse{f}}}{U}$ by
            \obv{preim_eq_img_of_converse}.
        $f$ is a function by \obv{funs_is_function}.
        Hence $\converse{\converse{f}}=f$ by
            \obv{converse_converse_eq}.
    \end{subproof}
    Follows by \obv{open_map}.
\end{proof}

\begin{theorem}\label{continuous_bijection_open_map_is_homeomorphism}
    Suppose $f$ is continuous from $X$ to $Y$.
    Suppose $f\in\bijections{\carrier[X]}{\carrier[Y]}$.
    Suppose $f$ is an open map from $X$ to $Y$.
    Then $f$ is a homeomorphism from $X$ to $Y$.
\end{theorem}
\begin{proof}
    $X$ is a topological space and $Y$ is a topological space and
        $f\in\funs{\carrier[X]}{\carrier[Y]}$ by
        \obv{continuous_map}.
    $\converse{f}\in\funs{\carrier[Y]}{\carrier[X]}$ by
        \obv{bijective_converse_are_funs}.
    Show that for every $U\in\opens[X]$ we have
        $\preimg{\converse{f}}{U}\in\opens[Y]$.
    \begin{subproof}
        Fix $U\in\opens[X]$.
        $\img{f}{U}\in\opens[Y]$ by \obv{open_map_image_open}.
        $\preimg{\converse{f}}{U}
            =\img{\converse{\converse{f}}}{U}$ by
            \obv{preim_eq_img_of_converse}.
        $f$ is a function by \obv{funs_is_function}.
        Hence $\converse{\converse{f}}=f$ by
            \obv{converse_converse_eq}.
    \end{subproof}
    Hence $\converse{f}$ is continuous from $Y$ to $X$ by
        \obv{continuous_map}.
    Follows by \obv{homeomorphism}.
\end{proof}

\begin{theorem}\label{homeomorphism_iff_continuous_bijection_open_map}
    Let $X,Y$ be topological spaces.
    Let $f\in\bijections{\carrier[X]}{\carrier[Y]}$.
    Then $f$ is a homeomorphism from $X$ to $Y$ iff
        $f$ is continuous from $X$ to $Y$ and
        $f$ is an open map from $X$ to $Y$.
\end{theorem}
\begin{proof}
    Show that if $f$ is a homeomorphism from $X$ to $Y$, then
        $f$ is continuous from $X$ to $Y$ and
        $f$ is an open map from $X$ to $Y$.
    \begin{subproof}
        Assume $f$ is a homeomorphism from $X$ to $Y$.
        $f$ is continuous from $X$ to $Y$ by \obv{homeomorphism}.
        $f$ is an open map from $X$ to $Y$ by
            \obv{homeomorphism_is_open_map}.
    \end{subproof}
    Show that if $f$ is continuous from $X$ to $Y$ and
        $f$ is an open map from $X$ to $Y$, then
        $f$ is a homeomorphism from $X$ to $Y$.
    \begin{subproof}
        Assume $f$ is continuous from $X$ to $Y$ and
            $f$ is an open map from $X$ to $Y$.
        Follows by
            \obv{continuous_bijection_open_map_is_homeomorphism}.
    \end{subproof}
    Follows by \auto.
\end{proof}
