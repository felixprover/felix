\section{Topological powers}

\import{topology/indexed-product-empty.tex}
\import{set/function-power.tex}

The power of X indexed by I is the indexed product of the constant family
with value X. Its points are functions on I, not iterated ordered pairs.
The empty power contains the empty function and is a one-point space.

\subsection{The power topology}

\begin{proposition}\label{constant_topological_carrier_family}
    Let $X$ be a topological space.
    Then $\indexedTopologicalCarrierFamily{I}{I\times\{X\}}
        =I\times\{\carrier[X]\}$.
\end{proposition}
\begin{proof}
    Let $C=\indexedTopologicalCarrierFamily{I}{I\times\{X\}}$ and
        $D=I\times\{\carrier[X]\}$.
    $C$ is a function on $I$ by \auto{indexed_topological_carrier_family_type}.
    $D$ is a function on $I$ by
        \auto{constant_graph_type,funs_is_function,funs}.
    For every $i\in I$ we have $C(i)=D(i)$ by
        \auto{indexed_topological_carrier_family_apply,constant_graph_application}.
    Follows by \auto{funext_on_domain}.
\end{proof}

\begin{definition}\label{topological_power}
    Let $X$ be a topological space.
    $\TopologicalPower{X}{I}=\IndexedProductTopology{I}{I\times\{X\}}$.
\end{definition}

\begin{classification}\label{topological_power_is_topological_space}
    Let $X$ be a topological space.
    Then $\TopologicalPower{X}{I}$ is a topological space.
\end{classification}
\begin{proof}
    Follows by \auto{topological_power,indexed_product_topology_is_topological_space}.
\end{proof}

\begin{theorem}\label{topological_power_is_exact}
    Let $X$ be a topological space.
    Then $\TopologicalPower{X}{I}$ is an exact topological space.
\end{theorem}
\begin{proof}
    Follows by \auto{topological_power,indexed_product_topology_is_exact}.
\end{proof}

\begin{proposition}\label{topological_power_carrier}
    Let $X$ be a topological space.
    Then $\carrier[\TopologicalPower{X}{I}]=\funs{I}{\carrier[X]}$.
\end{proposition}
\begin{proof}
    $\carrier[\TopologicalPower{X}{I}]
        =\dependentProduct{I}{\indexedTopologicalCarrierFamily{I}{I\times\{X\}}}$ by
        \auto{topological_power,indexed_product_topology_carrier}.
    $\indexedTopologicalCarrierFamily{I}{I\times\{X\}}
        =I\times\{\carrier[X]\}$ by \obv{constant_topological_carrier_family}.
    Follows by \auto{constant_family_dependent_product}.
\end{proof}

\begin{theorem}\label{empty_topological_power}
    Let $X$ be a topological space.
    Then $\TopologicalPower{X}{\emptyset}=\discreteTopology{\{\emptyset\}}$.
\end{theorem}
\begin{proof}
    Follows by \auto{topological_power,empty_indexed_product_is_discrete}.
\end{proof}

\subsection{Coordinatewise continuity}

\begin{theorem}\label{topological_power_evaluation}
    Let $X$ be a topological space and $i\in I$.
    Then $\functionEvaluation{I}{\carrier[X]}{i}$ is continuous from
        $\TopologicalPower{X}{I}$ to $X$.
\end{theorem}
\begin{proof}
    $I\times\{X\}$ is a function on $I$ by
        \auto{constant_graph_type,funs_is_function,funs}.
    For every $j\in I$ we have $(I\times\{X\})(j)=X$ by
        \obv{constant_graph_application}.
    Hence for every $j\in I$ we have $(I\times\{X\})(j)$ is a topological space.
    $\dependentProjection{I}{\indexedTopologicalCarrierFamily{I}{I\times\{X\}}}{i}$
        is continuous from $\IndexedProductTopology{I}{I\times\{X\}}$
        to $(I\times\{X\})(i)$ by \obv{indexed_product_coordinate_continuous}.
    $\indexedTopologicalCarrierFamily{I}{I\times\{X\}}=I\times\{\carrier[X]\}$ by
        \obv{constant_topological_carrier_family}.
    Follows by \auto{topological_power,function_evaluation}.
\end{proof}

\begin{theorem}\label{topological_power_continuity_iff}
    Let $X,Z$ be topological spaces.
    Let $g\in\funs{\carrier[Z]}{\funs{I}{\carrier[X]}}$.
    Then $g$ is continuous from $Z$ to $\TopologicalPower{X}{I}$ iff
        for every $i\in I$ we have
            $\functionEvaluation{I}{\carrier[X]}{i}\circ g$ is continuous from $Z$ to $X$.
\end{theorem}
\begin{proof}
    $I\times\{X\}$ is a function on $I$ by
        \auto{constant_graph_type,funs_is_function,funs}.
    For every $i\in I$ we have $(I\times\{X\})(i)=X$ by
        \obv{constant_graph_application}.
    Hence for every $i\in I$ we have $(I\times\{X\})(i)$ is a topological space.
    $\indexedTopologicalCarrierFamily{I}{I\times\{X\}}=I\times\{\carrier[X]\}$ by
        \obv{constant_topological_carrier_family}.
    Hence $g\in\funs{\carrier[Z]}{
        \dependentProduct{I}{\indexedTopologicalCarrierFamily{I}{I\times\{X\}}}}$ by
        \auto{constant_family_dependent_product}.
    Hence $g$ is continuous from $Z$ to $\IndexedProductTopology{I}{I\times\{X\}}$ iff
        for every $i\in I$ we have
            $\dependentProjection{I}{\indexedTopologicalCarrierFamily{I}{I\times\{X\}}}{i}\circ g$
                is continuous from $Z$ to $(I\times\{X\})(i)$ by
        \obv{indexed_product_topology_universal_property}.
    Follows by \auto{topological_power,function_evaluation}.
\end{proof}

\subsection{Changing the index set}

\begin{theorem}\label{topological_power_reindex_continuous}
    Let $X$ be a topological space and $e\in\funs{J}{I}$.
    Then $\functionReindex{I}{J}{\carrier[X]}{e}$ is continuous from
        $\TopologicalPower{X}{I}$ to $\TopologicalPower{X}{J}$.
\end{theorem}
\begin{proof}
    $\TopologicalPower{X}{I}$ is a topological space by
        \obv{topological_power_is_topological_space}.
    $\functionReindex{I}{J}{\carrier[X]}{e}\in
        \funs{\carrier[\TopologicalPower{X}{I}]}{\funs{J}{\carrier[X]}}$ by
        \auto{function_reindex_type,topological_power_carrier}.
    Show that for every $j\in J$ we have
        $\functionEvaluation{J}{\carrier[X]}{j}\circ\functionReindex{I}{J}{\carrier[X]}{e}$
            is continuous from $\TopologicalPower{X}{I}$ to $X$.
    \begin{subproof}
        Fix $j\in J$.
        $e(j)\in I$ by \obv{funs_type_apply}.
        $\functionEvaluation{I}{\carrier[X]}{e(j)}$ is continuous from
            $\TopologicalPower{X}{I}$ to $X$ by \obv{topological_power_evaluation}.
        Follows by \auto{function_reindex_coordinate}.
    \end{subproof}
    Follows by \obv{topological_power_continuity_iff}.
\end{proof}

\begin{theorem}\label{topological_power_reindex_homeomorphism}
    Let $X$ be a topological space and $e\in\bijections{J}{I}$.
    Then $\functionReindex{I}{J}{\carrier[X]}{e}$ is a homeomorphism from
        $\TopologicalPower{X}{I}$ to $\TopologicalPower{X}{J}$.
\end{theorem}
\begin{proof}
    $e\in\funs{J}{I}$ and $\converse{e}\in\funs{I}{J}$ by
        \obv{bijection_converse_is_bijection,bijections_to_funs}.
    $\functionReindex{I}{J}{\carrier[X]}{e}$ is continuous from
        $\TopologicalPower{X}{I}$ to $\TopologicalPower{X}{J}$ by
        \obv{topological_power_reindex_continuous}.
    $\functionReindex{J}{I}{\carrier[X]}{\converse{e}}$ is continuous from
        $\TopologicalPower{X}{J}$ to $\TopologicalPower{X}{I}$ by
        \obv{topological_power_reindex_continuous}.
    $\converse{\functionReindex{I}{J}{\carrier[X]}{e}}
        =\functionReindex{J}{I}{\carrier[X]}{\converse{e}}$ by
        \obv{function_reindex_converse}.
    $\functionReindex{I}{J}{\carrier[X]}{e}\in
        \bijections{\carrier[\TopologicalPower{X}{I}]}{\carrier[\TopologicalPower{X}{J}]}$ by
        \auto{function_reindex_bijection,topological_power_carrier}.
    Follows by \obv{homeomorphism}.
\end{proof}

\subsection{The diagonal and a single coordinate}

\begin{theorem}\label{topological_power_diagonal_continuous}
    Let $X$ be a topological space.
    Then $\functionDiagonal{I}{\carrier[X]}$ is continuous from $X$ to
        $\TopologicalPower{X}{I}$.
\end{theorem}
\begin{proof}
    $\functionDiagonal{I}{\carrier[X]}\in\funs{\carrier[X]}{\funs{I}{\carrier[X]}}$
        by \obv{function_diagonal_type}.
    $\identity{\carrier[X]}$ is continuous from $X$ to $X$ by
        \obv{identity_continuous}.
    For every $i\in I$ we have
        $\functionEvaluation{I}{\carrier[X]}{i}\circ\functionDiagonal{I}{\carrier[X]}
            =\identity{\carrier[X]}$ by \obv{function_evaluation_diagonal}.
    Hence for every $i\in I$ we have
        $\functionEvaluation{I}{\carrier[X]}{i}\circ\functionDiagonal{I}{\carrier[X]}$
            is continuous from $X$ to $X$.
    Follows by \obv{topological_power_continuity_iff}.
\end{proof}

\begin{theorem}\label{singleton_power_evaluation_homeomorphism}
    Let $X$ be a topological space.
    Then $\functionEvaluation{\{i\}}{\carrier[X]}{i}$ is a homeomorphism from
        $\TopologicalPower{X}{\{i\}}$ to $X$.
\end{theorem}
\begin{proof}
    $i\in\{i\}$ by \obv{singleton_intro}.
    $\functionEvaluation{\{i\}}{\carrier[X]}{i}$ is continuous from
        $\TopologicalPower{X}{\{i\}}$ to $X$ by \obv{topological_power_evaluation}.
    $\functionDiagonal{\{i\}}{\carrier[X]}$ is continuous from $X$ to
        $\TopologicalPower{X}{\{i\}}$ by \obv{topological_power_diagonal_continuous}.
    $\converse{\functionEvaluation{\{i\}}{\carrier[X]}{i}}
        =\functionDiagonal{\{i\}}{\carrier[X]}$ by \obv{singleton_evaluation_converse}.
    $\functionEvaluation{\{i\}}{\carrier[X]}{i}\in
        \bijections{\carrier[\TopologicalPower{X}{\{i\}}]}{\carrier[X]}$ by
        \auto{singleton_evaluation_bijection,topological_power_carrier}.
    Follows by \obv{homeomorphism}.
\end{proof}
