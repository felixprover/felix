\section{Continuous partial maps}

\import{topology/subspace-continuity.tex}
\import{set/partial-map.tex}

A continuous partial map has an open domain in its displayed source. Its
image need not be open, and it need not be injective. As for partial
homeomorphisms, restriction and composition use ordinary function graphs.

\begin{definition}\label{continuous_partial_map}
    $f$ is a continuous partial map from $X$ to $Y$ iff
        $X$ is a topological space and $Y$ is a topological space and
        $\dom{f}\in\opens[X]$ and
        $f$ is continuous from $\subspaceTopology{X}{\dom{f}}$ to $Y$.
\end{definition}

\begin{proposition}\label{continuous_partial_map_type}
    Suppose $f$ is a continuous partial map from $X$ to $Y$.
    Then $f\in\funs{\dom{f}}{\carrier[Y]}$.
\end{proposition}
\begin{proof}
    $X$ is a topological space and $\dom{f}\in\opens[X]$ and
        $f$ is continuous from $\subspaceTopology{X}{\dom{f}}$ to $Y$ by
        \obv{continuous_partial_map}.
    $\dom{f}\subseteq\carrier[X]$ by \obv{opens_type}.
    $\carrier[\subspaceTopology{X}{\dom{f}}]=\dom{f}$ by
        \obv{subspace_topology_carrier_of_subset}.
    Follows by \obv{continuous_map}.
\end{proof}

\begin{proposition}\label{continuous_partial_map_graph_bound}
    Suppose $f$ is a continuous partial map from $X$ to $Y$.
    Then $f\subseteq\carrier[X]\times\carrier[Y]$.
\end{proposition}
\begin{proof}
    $X$ is a topological space and $\dom{f}\in\opens[X]$ by
        \obv{continuous_partial_map}.
    $\dom{f}\subseteq\carrier[X]$ by \obv{opens_type}.
    $f\in\funs{\dom{f}}{\carrier[Y]}$ by \obv{continuous_partial_map_type}.
    $f\in\rels{\dom{f}}{\carrier[Y]}$ by \obv{funs}.
    $f\in\rels{\carrier[X]}{\carrier[Y]}$ by \obv{rels_weaken_dom}.
    Follows by \obv{rels_elim}.
\end{proof}

\begin{theorem}\label{continuous_partial_map_preimage_open}
    Suppose $f$ is a continuous partial map from $X$ to $Y$.
    Suppose $V\in\opens[Y]$.
    Then $\preimg{f}{V}\in\opens[X]$.
\end{theorem}
\begin{proof}
    $X$ is a topological space and $\dom{f}\in\opens[X]$ and
        $f$ is continuous from $\subspaceTopology{X}{\dom{f}}$ to $Y$ by
        \obv{continuous_partial_map}.
    $\preimg{f}{V}\in\opens[\subspaceTopology{X}{\dom{f}}]$ by
        \obv{continuous_map}.
    Follows by \auto{open_subspace_open_iff}.
\end{proof}

\subsection{Restriction and composition}

\begin{theorem}\label{identity_continuous_partial_map}
    Let $X$ be a topological space.
    Suppose $U\in\opens[X]$.
    Then $\identity{U}$ is a continuous partial map from $X$ to $X$.
\end{theorem}
\begin{proof}
    $U\subseteq\carrier[X]$ by \obv{opens_type}.
    $\carrier[\subspaceTopology{X}{U}]=U$ by
        \obv{subspace_topology_carrier_of_subset}.
    $\identity{U}$ is continuous from $\subspaceTopology{X}{U}$ to $X$ by
        \auto{subspace_inclusion_continuous}.
    $\dom{\identity{U}}=U$ by \obv{id_dom}.
    Follows by \obv{continuous_partial_map}.
\end{proof}

\begin{theorem}\label{continuous_partial_map_restriction}
    Suppose $f$ is a continuous partial map from $X$ to $Y$.
    Suppose $U\in\opens[X]$.
    Then $\restrl{f}{U}$ is a continuous partial map from $X$ to $Y$.
\end{theorem}
\begin{proof}
    $X$ is a topological space and $Y$ is a topological space and
        $\dom{f}\in\opens[X]$ and
        $f$ is continuous from $\subspaceTopology{X}{\dom{f}}$ to $Y$ by
        \obv{continuous_partial_map}.
    $f\in\funs{\dom{f}}{\carrier[Y]}$ by \obv{continuous_partial_map_type}.
    $f$ is a function by \obv{funs_is_function}.
    Let $D=\dom{f}\inter U$.
    $D\in\opens[X]$ by \auto{opens_inter}.
    $D\subseteq\dom{f}$ by \auto{inter_lower_left}.
    $\dom{f}\subseteq\carrier[X]$ by \obv{opens_type}.
    $\carrier[\subspaceTopology{X}{\dom{f}}]=\dom{f}$ by
        \obv{subspace_topology_carrier_of_subset}.
    $\restrl{f}{D}$ is continuous from
        $\subspaceTopology{\subspaceTopology{X}{\dom{f}}}{D}$ to $Y$ by
        \obv{continuous_domain_restriction_on_subset}.
    $\subspaceTopology{\subspaceTopology{X}{\dom{f}}}{D}
        =\subspaceTopology{X}{D}$ by \obv{nested_subspace_topology}.
    $\restrl{f}{\dom{f}}=f$ by \obv{restrl_by_dom}.
    $\restrl{\restrl{f}{\dom{f}}}{U}=\restrl{f}{D}$ by
        \auto{restriction_intersection}.
    $\dom{\restrl{f}{U}}=D$ by \auto{dom_of_restrl_eq_inter}.
    Follows by \obv{continuous_partial_map}.
\end{proof}

\begin{theorem}\label{continuous_partial_map_iff_open_preimages}
    Let $X,Y$ be topological spaces.
    Suppose $f\in\funs{\dom{f}}{\carrier[Y]}$ and $\dom{f}\in\opens[X]$.
    Then $f$ is a continuous partial map from $X$ to $Y$ iff
        for every $V\in\opens[Y]$ we have $\preimg{f}{V}\in\opens[X]$.
\end{theorem}
\begin{proof}
    If $f$ is a continuous partial map from $X$ to $Y$, then
        for every $V\in\opens[Y]$ we have $\preimg{f}{V}\in\opens[X]$ by
        \auto{continuous_partial_map_preimage_open}.
    Show that if for every $V\in\opens[Y]$ we have $\preimg{f}{V}\in\opens[X]$,
        then $f$ is a continuous partial map from $X$ to $Y$.
    \begin{subproof}
        Assume for every $V\in\opens[Y]$ we have $\preimg{f}{V}\in\opens[X]$.
        Let $S=\subspaceTopology{X}{\dom{f}}$.
        $S$ is a topological space by \auto{subspace_topology_is_topological_space}.
        $\dom{f}\subseteq\carrier[X]$ by \obv{opens_type}.
        $\carrier[S]=\dom{f}$ by \auto{subspace_topology_carrier_of_subset}.
        Show that for every $V\in\opens[Y]$ we have $\preimg{f}{V}\in\opens[S]$.
        \begin{subproof}
            Fix $V\in\opens[Y]$.
            $\preimg{f}{V}\in\opens[X]$.
            $\preimg{f}{V}\subseteq\dom{f}$ by \obv{preimg_subseteq_dom}.
            Follows by \auto{open_subspace_open_iff}.
        \end{subproof}
        $f$ is continuous from $S$ to $Y$ by \obv{continuous_map}.
        Follows by \obv{continuous_partial_map}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{theorem}\label{continuous_partial_map_composition}
    Suppose $f$ is a continuous partial map from $X$ to $Y$ and
        $g$ is a continuous partial map from $Y$ to $Z$.
    Then $g\circ f$ is a continuous partial map from $X$ to $Z$.
\end{theorem}
\begin{proof}
    $X$ is a topological space and $Z$ is a topological space and
        $\dom{g}\in\opens[Y]$ by \obv{continuous_partial_map}.
    $f\in\funs{\dom{f}}{\carrier[Y]}$ and
        $g\in\funs{\dom{g}}{\carrier[Z]}$ by
        \obv{continuous_partial_map_type}.
    $f$ and $g$ are functions by \obv{funs_is_function}.
    $g\circ f$ is a function by \obv{function_circ}.
    $\dom{g\circ f}=\preimg{f}{\dom{g}}$ by \obv{relation_composition_domain}.
    $\preimg{f}{\dom{g}}\in\opens[X]$ by
        \obv{continuous_partial_map_preimage_open}.
    $\ran{g\circ f}=\img{g}{\ran{f}}$ by \obv{relation_composition_range}.
    $\img{g}{\ran{f}}\subseteq\ran{g}$ by \obv{img_subseteq_ran}.
    $\ran{g}\subseteq\carrier[Z]$ by \obv{funs_ran}.
    Hence $\ran{g\circ f}\subseteq\carrier[Z]$ by \auto{subseteq_transitive}.
    $g\circ f\in\funs{\dom{g\circ f}}{\ran{g\circ f}}$ by
        \obv{function_exact_type_to_range}.
    $g\circ f\in\funs{\dom{g\circ f}}{\carrier[Z]}$ by \obv{funs_weaken_codom}.
    Show that for every $V\in\opens[Z]$ we have $\preimg{g\circ f}{V}\in\opens[X]$.
    \begin{subproof}
        Fix $V\in\opens[Z]$.
        $\preimg{g}{V}\in\opens[Y]$ by \obv{continuous_partial_map_preimage_open}.
        $\preimg{f}{\preimg{g}{V}}\in\opens[X]$ by
            \obv{continuous_partial_map_preimage_open}.
        $\preimg{g\circ f}{V}=\preimg{f}{\preimg{g}{V}}$ by \obv{preimg_circ}.
    \end{subproof}
    Follows by \auto{continuous_partial_map_iff_open_preimages}.
\end{proof}

\subsection{Locality}

The restrictions in the next theorem already have open effective domains.
The covering sets themselves therefore need no additional openness hypothesis.

\begin{theorem}\label{continuous_partial_map_locality}
    Let $X,Y$ be topological spaces.
    Suppose $f\in\funs{\dom{f}}{\carrier[Y]}$ and $\dom{f}\in\opens[X]$.
    Suppose for every $x\in\dom{f}$ there exists $U\in F$ such that $x\in U$.
    Suppose for every $U\in F$ we have
        $\restrl{f}{U}$ is a continuous partial map from $X$ to $Y$.
    Then $f$ is a continuous partial map from $X$ to $Y$.
\end{theorem}
\begin{proof}
    $f$ is a function by \obv{funs_is_function}.
    Show that for every $V\in\opens[Y]$ we have $\preimg{f}{V}\in\opens[X]$.
    \begin{subproof}
        Fix $V\in\opens[Y]$.
        Let $P=\preimg{f}{V}$.
        Show that for every $x\in P$ there exists $W\in\opens[X]$ such that
            $x\in W$ and $W\subseteq P$.
        \begin{subproof}
            Fix $x\in P$.
            $P\subseteq\dom{f}$ by \auto{preimg_subseteq_dom}.
            $x\in\dom{f}$ by \obv{subseteq}.
            Take $U\in F$ such that $x\in U$.
            $\restrl{f}{U}$ is a continuous partial map from $X$ to $Y$.
            Let $W=\preimg{\restrl{f}{U}}{V}$.
            $W\in\opens[X]$ by \auto{continuous_partial_map_preimage_open}.
            $W=P\inter U$ by \auto{restriction_preimage}.
            $x\in W$ and $W\subseteq P$ by \auto{inter,inter_lower_left}.
        \end{subproof}
        $P\in\opens[X]$ by \auto{open_iff_local_open_subsets}.
    \end{subproof}
    Follows by \auto{continuous_partial_map_iff_open_preimages}.
\end{proof}

\subsection{Total and empty maps}

\begin{theorem}\label{continuous_map_is_partial}
    Suppose $f$ is continuous from $X$ to $Y$.
    Then $f$ is a continuous partial map from $X$ to $Y$.
\end{theorem}
\begin{proof}
    $X$ is a topological space and $Y$ is a topological space and
        $f\in\funs{\carrier[X]}{\carrier[Y]}$ by \obv{continuous_map}.
    $\dom{f}=\carrier[X]$ by \obv{funs}.
    $\dom{f}\in\opens[X]$ by \obv{carrier_open}.
    For every $V\in\opens[Y]$ we have $\preimg{f}{V}\in\opens[X]$ by
        \obv{continuous_map}.
    Follows by \auto{continuous_partial_map_iff_open_preimages}.
\end{proof}

\begin{theorem}\label{total_continuous_partial_map}
    Suppose $f$ is a continuous partial map from $X$ to $Y$.
    Suppose $\dom{f}=\carrier[X]$.
    Then $f$ is continuous from $X$ to $Y$.
\end{theorem}
\begin{proof}
    $f\in\funs{\dom{f}}{\carrier[Y]}$ by \obv{continuous_partial_map_type}.
    For every $V\in\opens[Y]$ we have $\preimg{f}{V}\in\opens[X]$ by
        \auto{continuous_partial_map_preimage_open}.
    Follows by \obv{continuous_map}.
\end{proof}

\begin{theorem}\label{empty_continuous_partial_map}
    Let $X,Y$ be topological spaces.
    Then $\emptyset$ is a continuous partial map from $X$ to $Y$.
\end{theorem}
\begin{proof}
    Let $S=\subspaceTopology{X}{\emptyset}$.
    $S$ is a topological space by \auto{subspace_topology_is_topological_space}.
    $\carrier[S]=\emptyset$ by
        \auto{subspace_topology_carrier_of_subset,emptyset_subseteq}.
    $\emptyset$ is continuous from $S$ to $Y$ by \obv{empty_function_continuous}.
    $\dom{\emptyset}=\emptyset$ by \obv{dom_emptyset}.
    $\emptyset\in\opens[X]$ by \auto{emptyset_open,is_open}.
    Follows by \obv{continuous_partial_map}.
\end{proof}
