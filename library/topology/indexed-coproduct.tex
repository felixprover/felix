\section{Indexed coproduct topologies}

\import{topology/final.tex}
\import{topology/indexed-family.tex}

The carrier of an indexed coproduct is the tagged indexed sum of the
carriers.  Its topology is final with respect to the canonical injections of
the summands.

\subsection{The injection family}

\begin{definition}\label{indexed_coproduct_injection_op}
    $\indexedCoproductInjectionOp:\Set\to\Set\to\Set\to\Set
        =\fun{I,X,i:\Set}{
            \indexedSumInjection{I}{
                \indexedTopologicalCarrierFamily{I}{X}}{i}}$.
\end{definition}

\begin{definition}\label{indexed_coproduct_injection_family}
    $\indexedCoproductInjectionFamily{I}{X}
        =\Apply{\graphOnDomainOp}{
            I,\Apply{\indexedCoproductInjectionOp}{I,X}}$.
\end{definition}

\begin{proposition}\label{indexed_coproduct_injection_family_type}
    $\indexedCoproductInjectionFamily{I}{X}$ is a function on $I$.
\end{proposition}
\begin{proof}
    $\indexedCoproductInjectionFamily{I}{X}\in
        \funs{I}{\ran{\indexedCoproductInjectionFamily{I}{X}}}$ by
        \auto{indexed_coproduct_injection_family,
            logical_graph_on_domain_function}.
    Follows by \obv{funs_is_function,funs}.
\end{proof}

\begin{proposition}\label{indexed_coproduct_injection_family_apply}
    Let $i\in I$.
    Then $(\indexedCoproductInjectionFamily{I}{X})(i)
        =\indexedSumInjection{I}{
            \indexedTopologicalCarrierFamily{I}{X}}{i}$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{indexed_coproduct_injection_family,
            indexed_coproduct_injection_op,
            logical_graph_on_domain_apply}.
\end{proof}

\subsection{The coproduct topology}

\begin{definition}\label{indexed_coproduct_topology}
    $\IndexedCoproductTopology{I}{X}
        =\FinalTopology{
            \indexedSum{I}{
                \indexedTopologicalCarrierFamily{I}{X}}}{I}{X}{
            \indexedCoproductInjectionFamily{I}{X}}$.
\end{definition}

\begin{proposition}\label{indexed_coproduct_final_data}
    Suppose $X$ is a function on $I$.
    Suppose for every $i\in I$ we have $X(i)$ is a topological space.
    Then $\indexedCoproductInjectionFamily{I}{X}$ is a final mapping
        family from $X$ to
        $\indexedSum{I}{\indexedTopologicalCarrierFamily{I}{X}}$
        indexed by $I$.
\end{proposition}
\begin{proof}
    $\indexedCoproductInjectionFamily{I}{X}$ is a function on $I$ by
        \obv{indexed_coproduct_injection_family_type}.
    $\indexedTopologicalCarrierFamily{I}{X}$ is a function on $I$ by
        \obv{indexed_topological_carrier_family_type}.
    Show that for every $i\in I$ we have
        $X(i)$ is a topological space and
        $(\indexedCoproductInjectionFamily{I}{X})(i)
            \in\funs{\carrier[X(i)]}{
                \indexedSum{I}{
                    \indexedTopologicalCarrierFamily{I}{X}}}$.
    \begin{subproof}
        Fix $i\in I$.
        $\indexedSumInjection{I}{
            \indexedTopologicalCarrierFamily{I}{X}}{i}
            \in\inj{
                (\indexedTopologicalCarrierFamily{I}{X})(i)}{
                \indexedSum{I}{
                    \indexedTopologicalCarrierFamily{I}{X}}}$ by
            \obv{indexed_sum_injection_type}.
        Hence $\indexedSumInjection{I}{
            \indexedTopologicalCarrierFamily{I}{X}}{i}
            \in\funs{
                (\indexedTopologicalCarrierFamily{I}{X})(i)}{
                \indexedSum{I}{
                    \indexedTopologicalCarrierFamily{I}{X}}}$ by
            \obv{inj}.
        $(\indexedTopologicalCarrierFamily{I}{X})(i)
            =\carrier[X(i)]$ by
            \obv{indexed_topological_carrier_family_apply}.
        $(\indexedCoproductInjectionFamily{I}{X})(i)
            =\indexedSumInjection{I}{
                \indexedTopologicalCarrierFamily{I}{X}}{i}$ by
            \obv{indexed_coproduct_injection_family_apply}.
    \end{subproof}
    Follows by \obv{final_topology_data}.
\end{proof}

\begin{classification}\label{indexed_coproduct_topology_is_topological_space}
    Suppose $X$ is a function on $I$.
    Suppose for every $i\in I$ we have $X(i)$ is a topological space.
    Then $\IndexedCoproductTopology{I}{X}$ is a topological space.
\end{classification}
\begin{proof}
    $\indexedCoproductInjectionFamily{I}{X}$ is a final mapping family
        from $X$ to
        $\indexedSum{I}{\indexedTopologicalCarrierFamily{I}{X}}$
        indexed by $I$ by \obv{indexed_coproduct_final_data}.
    Follows by
        \auto{indexed_coproduct_topology,
            final_topology_is_topological_space}.
\end{proof}

\begin{theorem}\label{indexed_coproduct_topology_is_exact}
    Suppose $X$ is a function on $I$.
    Suppose for every $i\in I$ we have $X(i)$ is a topological space.
    Then $\IndexedCoproductTopology{I}{X}$ is an exact topological space.
\end{theorem}
\begin{proof}
    $\indexedCoproductInjectionFamily{I}{X}$ is a final mapping family
        from $X$ to
        $\indexedSum{I}{\indexedTopologicalCarrierFamily{I}{X}}$
        indexed by $I$ by \obv{indexed_coproduct_final_data}.
    Follows by
        \auto{indexed_coproduct_topology,final_topology_is_exact}.
\end{proof}

\begin{proposition}\label{indexed_coproduct_topology_carrier}
    $\carrier[\IndexedCoproductTopology{I}{X}]
        =\indexedSum{I}{\indexedTopologicalCarrierFamily{I}{X}}$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{indexed_coproduct_topology,final_topology_carrier}.
\end{proof}

\begin{proposition}\label{indexed_coproduct_injection_injective}
    Suppose $X$ is a function on $I$.
    Let $i\in I$.
    Then $\indexedSumInjection{I}{
        \indexedTopologicalCarrierFamily{I}{X}}{i}
        \in\inj{\carrier[X(i)]}{
            \carrier[\IndexedCoproductTopology{I}{X}]}$.
\end{proposition}
\begin{proof}
    $\indexedTopologicalCarrierFamily{I}{X}$ is a function on $I$ by
        \obv{indexed_topological_carrier_family_type}.
    Hence $\indexedSumInjection{I}{
        \indexedTopologicalCarrierFamily{I}{X}}{i}
        \in\inj{
            (\indexedTopologicalCarrierFamily{I}{X})(i)}{
            \indexedSum{I}{
                \indexedTopologicalCarrierFamily{I}{X}}}$ by
        \obv{indexed_sum_injection_type}.
    $(\indexedTopologicalCarrierFamily{I}{X})(i)
        =\carrier[X(i)]$ by
        \obv{indexed_topological_carrier_family_apply}.
    $\carrier[\IndexedCoproductTopology{I}{X}]
        =\indexedSum{I}{\indexedTopologicalCarrierFamily{I}{X}}$ by
        \obv{indexed_coproduct_topology_carrier}.
\end{proof}

\begin{theorem}\label{indexed_coproduct_injection_continuous}
    Suppose $X$ is a function on $I$.
    Suppose for every $j\in I$ we have $X(j)$ is a topological space.
    Let $i\in I$.
    Then $\indexedSumInjection{I}{
        \indexedTopologicalCarrierFamily{I}{X}}{i}$ is continuous from
        $X(i)$ to $\IndexedCoproductTopology{I}{X}$.
\end{theorem}
\begin{proof}
    $\indexedCoproductInjectionFamily{I}{X}$ is a final mapping family
        from $X$ to
        $\indexedSum{I}{\indexedTopologicalCarrierFamily{I}{X}}$
        indexed by $I$ by \obv{indexed_coproduct_final_data}.
    Hence $(\indexedCoproductInjectionFamily{I}{X})(i)$ is continuous
        from $X(i)$ to
        $\FinalTopology{
            \indexedSum{I}{
                \indexedTopologicalCarrierFamily{I}{X}}}{I}{X}{
            \indexedCoproductInjectionFamily{I}{X}}$ by
        \obv{final_topology_structure_map_continuous}.
    $(\indexedCoproductInjectionFamily{I}{X})(i)
        =\indexedSumInjection{I}{
            \indexedTopologicalCarrierFamily{I}{X}}{i}$ by
        \obv{indexed_coproduct_injection_family_apply}.
    $\IndexedCoproductTopology{I}{X}
        =\FinalTopology{
            \indexedSum{I}{
                \indexedTopologicalCarrierFamily{I}{X}}}{I}{X}{
            \indexedCoproductInjectionFamily{I}{X}}$ by
        \obv{indexed_coproduct_topology}.
\end{proof}

\begin{theorem}\label{indexed_coproduct_topology_universal_property}
    Suppose $X$ is a function on $I$.
    Suppose for every $i\in I$ we have $X(i)$ is a topological space.
    Let $Y$ be a topological space.
    Let $g\in\funs{
        \indexedSum{I}{\indexedTopologicalCarrierFamily{I}{X}}}{
        \carrier[Y]}$.
    Then $g$ is continuous from $\IndexedCoproductTopology{I}{X}$ to $Y$
        iff for every $i\in I$ we have
            $g\circ\indexedSumInjection{I}{
                \indexedTopologicalCarrierFamily{I}{X}}{i}$
            is continuous from $X(i)$ to $Y$.
\end{theorem}
\begin{proof}
    $\indexedCoproductInjectionFamily{I}{X}$ is a final mapping family
        from $X$ to
        $\indexedSum{I}{\indexedTopologicalCarrierFamily{I}{X}}$
        indexed by $I$ by \obv{indexed_coproduct_final_data}.
    Hence $g$ is continuous from
        $\FinalTopology{
            \indexedSum{I}{
                \indexedTopologicalCarrierFamily{I}{X}}}{I}{X}{
            \indexedCoproductInjectionFamily{I}{X}}$ to $Y$ iff
        for every $i\in I$ we have
            $g\circ(\indexedCoproductInjectionFamily{I}{X})(i)$
            is continuous from $X(i)$ to $Y$ by
        \obv{final_topology_universal_property}.
    For every $i\in I$ we have
        $(\indexedCoproductInjectionFamily{I}{X})(i)
            =\indexedSumInjection{I}{
                \indexedTopologicalCarrierFamily{I}{X}}{i}$ by
        \obv{indexed_coproduct_injection_family_apply}.
    $\IndexedCoproductTopology{I}{X}
        =\FinalTopology{
            \indexedSum{I}{
                \indexedTopologicalCarrierFamily{I}{X}}}{I}{X}{
            \indexedCoproductInjectionFamily{I}{X}}$ by
        \obv{indexed_coproduct_topology}.
    Follows by \auto.
\end{proof}
