\section{Locality of local path connectedness}

\import{topology/local-path.tex}

Open subspaces inherit local path connectedness. Conversely, it suffices to
check the property on one open neighbourhood of each point.

\begin{theorem}\label{open_subspace_is_locally_path_connected}
    Let $X$ be a locally path-connected topological space.
    Suppose $A\in\opens[X]$.
    Then $\subspaceTopology{X}{A}$ is locally path-connected.
\end{theorem}
\begin{proof}
    Let $S=\subspaceTopology{X}{A}$.
    $S$ is a topological space by \obv{subspace_topology_is_topological_space}.
    $A\subseteq\carrier[X]$ by \auto{opens_type,subseteq}.
    Show that for all $U,x$ such that $U\in\opens[S]$ and $x\in U$ there exists
        $V\in\opens[S]$ such that $x\in V\subseteq U$ and
        $\subspaceTopology{S}{V}$ is path-connected.
    \begin{subproof}
        Fix $U,x$.
        Assume $U\in\opens[S]$ and $x\in U$.
        $U\in\opens[X]$ and $U\subseteq A$ by \auto{open_subspace_open_iff}.
        Take $V\in\opens[X]$ such that $x\in V\subseteq U$ and
            $\subspaceTopology{X}{V}$ is path-connected by \obv{locally_path_connected}.
        $V\subseteq A$ by \obv{subseteq_transitive}.
        $V\in\opens[S]$ by \auto{open_subspace_open_iff}.
        $\subspaceTopology{S}{V}=\subspaceTopology{X}{V}$ by \auto{nested_subspace_topology}.
        Follows by \auto{path_connected}.
    \end{subproof}
    Follows by \auto{locally_path_connected}.
\end{proof}

\begin{theorem}\label{locally_path_connected_from_open_neighbourhoods}
    Let $X$ be a topological space.
    Suppose for every $x\in\carrier[X]$ there exists $A\in\opens[X]$ such that
        $x\in A$ and $\subspaceTopology{X}{A}$ is locally path-connected.
    Then $X$ is locally path-connected.
\end{theorem}
\begin{proof}
    Show that for all $U,x$ such that $U\in\opens[X]$ and $x\in U$ there exists
        $V\in\opens[X]$ such that $x\in V\subseteq U$ and
        $\subspaceTopology{X}{V}$ is path-connected.
    \begin{subproof}
        Fix $U,x$.
        Assume $U\in\opens[X]$ and $x\in U$.
        $x\in\carrier[X]$ by \auto{opens_type,subseteq}.
        Take $A\in\opens[X]$ such that $x\in A$ and
            $\subspaceTopology{X}{A}$ is locally path-connected.
        Let $S=\subspaceTopology{X}{A}$.
        $A\subseteq\carrier[X]$ by \auto{opens_type,subseteq}.
        $U\inter A\in\opens[S]$ by \auto{open_subspace_open_iff,opens_inter,inter,subseteq}.
        $x\in U\inter A$ by \obv{inter}.
        Take $V\in\opens[S]$ such that $x\in V\subseteq U\inter A$ and
            $\subspaceTopology{S}{V}$ is path-connected by \auto{locally_path_connected}.
        $V\in\opens[X]$ and $V\subseteq A$ by \auto{open_subspace_open_iff}.
        $V\subseteq U$ by \auto{inter,subseteq}.
        $\subspaceTopology{S}{V}=\subspaceTopology{X}{V}$ by \auto{nested_subspace_topology}.
        Follows by \auto{path_connected}.
    \end{subproof}
    Follows by \obv{locally_path_connected}.
\end{proof}
