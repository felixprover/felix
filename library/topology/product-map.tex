\section{Maps between binary products}

\import{topology/product.tex}

Products act functorially on continuous maps.  The same pairing construction
also gives the canonical symmetry of a binary product.

\subsection{Products of maps}

\begin{theorem}\label{continuous_product_map}
    Suppose $f$ is continuous from $X$ to $X'$ and
        $g$ is continuous from $Y$ to $Y'$.
    Then $\productMap{\carrier[X]}{\carrier[Y]}{f}{g}$ is continuous
        from $\BinaryProductTopology{X}{Y}$ to
        $\BinaryProductTopology{X'}{Y'}$.
\end{theorem}
\begin{proof}
    $X'$ is a topological space by \obv{continuous_map}.
    $Y'$ is a topological space by \obv{continuous_map}.
    $\BinaryProductTopology{X}{Y}$ is a topological space by
        \obv{binary_product_topology_is_topological_space}.
    $f\in\funs{\carrier[X]}{\carrier[X']}$ and
        $g\in\funs{\carrier[Y]}{\carrier[Y']}$ by
        \obv{continuous_map}.
    Hence $\productMap{\carrier[X]}{\carrier[Y]}{f}{g}\in
        \funs{
            \carrier[X]\times\carrier[Y]}{
            \carrier[X']\times\carrier[Y']}$ by
        \obv{product_map_type}.
    $\carrier[\BinaryProductTopology{X}{Y}]
        =\carrier[X]\times\carrier[Y]$ and
        $\carrier[\BinaryProductTopology{X'}{Y'}]
        =\carrier[X']\times\carrier[Y']$ by
        \obv{binary_product_topology_carrier}.
    Hence $\productMap{\carrier[X]}{\carrier[Y]}{f}{g}\in
        \funs{
            \carrier[\BinaryProductTopology{X}{Y}]}{
            \carrier[\BinaryProductTopology{X'}{Y'}]}$.
    $\firstProductProjection{\carrier[X]}{\carrier[Y]}$ is continuous
        from $\BinaryProductTopology{X}{Y}$ to $X$ by
        \obv{first_product_projection_continuous}.
    Hence $f\circ\firstProductProjection{
        \carrier[X]}{\carrier[Y]}$ is continuous from
        $\BinaryProductTopology{X}{Y}$ to $X'$ by
        \obv{continuous_composition}.
    $\secondProductProjection{\carrier[X]}{\carrier[Y]}$ is continuous
        from $\BinaryProductTopology{X}{Y}$ to $Y$ by
        \obv{second_product_projection_continuous}.
    Hence $g\circ\secondProductProjection{
        \carrier[X]}{\carrier[Y]}$ is continuous from
        $\BinaryProductTopology{X}{Y}$ to $Y'$ by
        \obv{continuous_composition}.
    $\firstProductProjection{\carrier[X']}{\carrier[Y']}\circ
        \productMap{\carrier[X]}{\carrier[Y]}{f}{g}
        =f\circ\firstProductProjection{\carrier[X]}{\carrier[Y]}$ by
        \obv{first_projection_of_product_map}.
    $\secondProductProjection{\carrier[X']}{\carrier[Y']}\circ
        \productMap{\carrier[X]}{\carrier[Y]}{f}{g}
        =g\circ\secondProductProjection{\carrier[X]}{\carrier[Y]}$ by
        \obv{second_projection_of_product_map}.
    Follows by \auto{binary_product_topology_universal_property}.
\end{proof}

\begin{theorem}\label{product_map_homeomorphism}
    Suppose $f$ is a homeomorphism from $X$ to $X'$ and
        $g$ is a homeomorphism from $Y$ to $Y'$.
    Then $\productMap{\carrier[X]}{\carrier[Y]}{f}{g}$ is a homeomorphism from
        $\BinaryProductTopology{X}{Y}$ to $\BinaryProductTopology{X'}{Y'}$.
\end{theorem}
\begin{proof}
    $f$ is continuous from $X$ to $X'$ and $g$ is continuous from $Y$ to $Y'$ by
        \obv{homeomorphism}.
    $\converse{f}$ is continuous from $X'$ to $X$ and
        $\converse{g}$ is continuous from $Y'$ to $Y$ by \obv{homeomorphism}.
    $\productMap{\carrier[X]}{\carrier[Y]}{f}{g}$ is continuous from
        $\BinaryProductTopology{X}{Y}$ to $\BinaryProductTopology{X'}{Y'}$ by
        \obv{continuous_product_map}.
    $\productMap{\carrier[X']}{\carrier[Y']}{\converse{f}}{\converse{g}}$ is continuous from
        $\BinaryProductTopology{X'}{Y'}$ to $\BinaryProductTopology{X}{Y}$ by
        \obv{continuous_product_map}.
    $f\in\bijections{\carrier[X]}{\carrier[X']}$ and
        $g\in\bijections{\carrier[Y]}{\carrier[Y']}$ by \obv{homeomorphism}.
    $\converse{\productMap{\carrier[X]}{\carrier[Y]}{f}{g}}
        =\productMap{\carrier[X']}{\carrier[Y']}{\converse{f}}{\converse{g}}$ by
        \obv{product_map_converse}.
    $\productMap{\carrier[X]}{\carrier[Y]}{f}{g}\in
        \bijections{\carrier[\BinaryProductTopology{X}{Y}]}{
            \carrier[\BinaryProductTopology{X'}{Y'}]}$ by
        \auto{product_map_bijection,binary_product_topology_carrier}.
    Follows by \obv{homeomorphism}.
\end{proof}

\subsection{Symmetry}

\begin{definition}\label{product_swap}
    $\productSwap{S}{T}
        =\productPairing{S\times T}{
            \secondProductProjection{S}{T}}{
            \firstProductProjection{S}{T}}$.
\end{definition}

\begin{proposition}\label{product_swap_type}
    $\productSwap{S}{T}\in\funs{S\times T}{T\times S}$.
\end{proposition}
\begin{proof}
    $\secondProductProjection{S}{T}\in\funs{S\times T}{T}$ and
        $\firstProductProjection{S}{T}\in\funs{S\times T}{S}$ by
        \obv{second_product_projection_type,
            first_product_projection_type}.
    Follows by \auto{product_swap,product_pairing_type}.
\end{proof}

\begin{proposition}\label{product_swap_apply}
    Let $p\in S\times T$.
    Then $(\productSwap{S}{T})(p)=(\snd{p},\fst{p})$.
\end{proposition}
\begin{proof}
    $(\secondProductProjection{S}{T})(p)=\snd{p}$ and
        $(\firstProductProjection{S}{T})(p)=\fst{p}$ by
        \obv{second_product_projection_apply,
            first_product_projection_apply}.
    Follows by \auto{product_swap,product_pairing_apply}.
\end{proof}

\begin{proposition}\label{product_swap_involution}
    Let $p\in S\times T$.
    Then $(\productSwap{T}{S})((\productSwap{S}{T})(p))=p$.
\end{proposition}
\begin{proof}
    $(\productSwap{S}{T})(p)=(\snd{p},\fst{p})$ by
        \obv{product_swap_apply}.
    $\snd{p}\in T$ and $\fst{p}\in S$ by
        \obv{snd_type,fst_type}.
    Hence $(\snd{p},\fst{p})\in T\times S$ by
        \obv{times_tuple_intro}.
    Hence $(\productSwap{T}{S})((\snd{p},\fst{p}))
        =(\fst{p},\snd{p})$ by
        \auto{product_swap_apply,fst_eq,snd_eq}.
    $p=(\fst{p},\snd{p})$ by
        \auto{times_elem_is_tuple,pair_eq_pair_of_proj}.
\end{proof}

\begin{theorem}\label{product_swap_bijection}
    $\productSwap{S}{T}$ is a bijection from $S\times T$ to $T\times S$.
\end{theorem}
\begin{proof}
    $\productSwap{S}{T}\in\funs{S\times T}{T\times S}$ and
        $\productSwap{T}{S}\in\funs{T\times S}{S\times T}$ by
        \obv{product_swap_type}.
    For every $p\in S\times T$ we have
        $(\productSwap{T}{S})((\productSwap{S}{T})(p))=p$ by
        \obv{product_swap_involution}.
    Hence $\productSwap{T}{S}$ is a left inverse of
        $\productSwap{S}{T}$ by \obv{leftinverse_from_to_iff}.
    For every $q\in T\times S$ we have
        $(\productSwap{S}{T})((\productSwap{T}{S})(q))=q$ by
        \obv{product_swap_involution}.
    Hence $\productSwap{T}{S}$ is a right inverse of
        $\productSwap{S}{T}$ by \obv{rightinverse_from_to_iff}.
    Hence $\productSwap{S}{T}$ is an injection by
        \obv{leftinverse_implies_injection}.
    Hence $\productSwap{S}{T}$ is a surjection from $S\times T$ to
        $T\times S$ by \obv{rightinverse_implies_surjection}.
    Follows by \obv{bijections}.
\end{proof}

\begin{theorem}\label{product_swap_converse}
    $\converse{\productSwap{S}{T}}=\productSwap{T}{S}$.
\end{theorem}
\begin{proof}
    $\productSwap{S}{T}\in\bijections{S\times T}{T\times S}$ by
        \obv{product_swap_bijection}.
    $\productSwap{S}{T}\in\funs{S\times T}{T\times S}$ by
        \obv{product_swap_type}.
    Hence $\converse{\productSwap{S}{T}}
        \in\funs{T\times S}{S\times T}$ by
        \obv{bijective_converse_are_funs}.
    $\productSwap{T}{S}\in\funs{T\times S}{S\times T}$ by
        \obv{product_swap_type}.
    Show that for every $q\in T\times S$ we have
        $(\converse{\productSwap{S}{T}})(q)
        =(\productSwap{T}{S})(q)$.
    \begin{subproof}
        Fix $q\in T\times S$.
        $(\productSwap{T}{S})(q)\in S\times T$ by
            \obv{funs_type_apply}.
        $(\productSwap{S}{T})((\productSwap{T}{S})(q))=q$ by
            \obv{product_swap_involution}.
        $((\productSwap{T}{S})(q),
            (\productSwap{S}{T})((\productSwap{T}{S})(q)))
            \in\productSwap{S}{T}$ by
            \obv{function_apply_elim,funs_is_function,funs}.
        Hence $((\productSwap{T}{S})(q),q)
            \in\productSwap{S}{T}$.
        Thus $(q,(\productSwap{T}{S})(q))
            \in\converse{\productSwap{S}{T}}$ by
            \obv{converse_iff}.
        $\converse{\productSwap{S}{T}}$ is a function by
            \obv{funs_is_function}.
        Follows by \obv{function_apply_intro}.
    \end{subproof}
    Follows by \obv{typed_functions_equal_from_agreement}.
\end{proof}

\begin{theorem}\label{product_swap_continuous}
    Let $X,Y$ be topological spaces.
    Then $\productSwap{\carrier[X]}{\carrier[Y]}$ is continuous from
        $\BinaryProductTopology{X}{Y}$ to
        $\BinaryProductTopology{Y}{X}$.
\end{theorem}
\begin{proof}
    $\secondProductProjection{\carrier[X]}{\carrier[Y]}$ is continuous
        from $\BinaryProductTopology{X}{Y}$ to $Y$ by
        \obv{second_product_projection_continuous}.
    $\firstProductProjection{\carrier[X]}{\carrier[Y]}$ is continuous
        from $\BinaryProductTopology{X}{Y}$ to $X$ by
        \obv{first_product_projection_continuous}.
    Hence $\productPairing{
        \carrier[\BinaryProductTopology{X}{Y}]}{
        \secondProductProjection{\carrier[X]}{\carrier[Y]}}{
        \firstProductProjection{\carrier[X]}{\carrier[Y]}}$ is continuous
        from $\BinaryProductTopology{X}{Y}$ to
        $\BinaryProductTopology{Y}{X}$ by
        \obv{continuous_product_pairing}.
    $\carrier[\BinaryProductTopology{X}{Y}]
        =\carrier[X]\times\carrier[Y]$ by
        \obv{binary_product_topology_carrier}.
    Hence $\productPairing{
        \carrier[X]\times\carrier[Y]}{
        \secondProductProjection{\carrier[X]}{\carrier[Y]}}{
        \firstProductProjection{\carrier[X]}{\carrier[Y]}}$ is continuous
        from $\BinaryProductTopology{X}{Y}$ to
        $\BinaryProductTopology{Y}{X}$.
    Follows by \obv{product_swap}.
\end{proof}

\begin{theorem}\label{product_swap_homeomorphism}
    Let $X,Y$ be topological spaces.
    Then $\productSwap{\carrier[X]}{\carrier[Y]}$ is a homeomorphism from
        $\BinaryProductTopology{X}{Y}$ to
        $\BinaryProductTopology{Y}{X}$.
\end{theorem}
\begin{proof}
    $\productSwap{\carrier[X]}{\carrier[Y]}$ is continuous from
        $\BinaryProductTopology{X}{Y}$ to
        $\BinaryProductTopology{Y}{X}$ by
        \obv{product_swap_continuous}.
    $\productSwap{\carrier[X]}{\carrier[Y]}$ is a bijection from
        $\carrier[X]\times\carrier[Y]$ to
        $\carrier[Y]\times\carrier[X]$ by
        \obv{product_swap_bijection}.
    $\carrier[\BinaryProductTopology{X}{Y}]
        =\carrier[X]\times\carrier[Y]$ and
        $\carrier[\BinaryProductTopology{Y}{X}]
        =\carrier[Y]\times\carrier[X]$ by
        \obv{binary_product_topology_carrier}.
    Hence $\productSwap{\carrier[X]}{\carrier[Y]}$ is a bijection from
        $\carrier[\BinaryProductTopology{X}{Y}]$ to
        $\carrier[\BinaryProductTopology{Y}{X}]$.
    $\converse{\productSwap{\carrier[X]}{\carrier[Y]}}
        =\productSwap{\carrier[Y]}{\carrier[X]}$ by
        \obv{product_swap_converse}.
    $\productSwap{\carrier[Y]}{\carrier[X]}$ is continuous from
        $\BinaryProductTopology{Y}{X}$ to
        $\BinaryProductTopology{X}{Y}$ by
        \obv{product_swap_continuous}.
    Hence $\converse{\productSwap{\carrier[X]}{\carrier[Y]}}$ is
        continuous from $\BinaryProductTopology{Y}{X}$ to
        $\BinaryProductTopology{X}{Y}$.
    Follows by \obv{homeomorphism}.
\end{proof}
