\section{Indexed families of topological carriers}

\import{topology/topological-space.tex}
\import{set/dependent-operation.tex}

For a set-coded family of topological spaces, the corresponding carrier
family is itself a set-coded function on the same index set.  This common
interface is shared by indexed products and coproducts.

\begin{definition}\label{indexed_topological_carrier_op}
    $\indexedTopologicalCarrierOp:\Set\to\Set\to\Set
        =\fun{X,i:\Set}{\carrier[X(i)]}$.
\end{definition}

\begin{definition}\label{indexed_topological_carrier_family}
    $\indexedTopologicalCarrierFamily{I}{X}
        =\Apply{\graphOnDomainOp}{
            I,\Apply{\indexedTopologicalCarrierOp}{X}}$.
\end{definition}

\begin{proposition}\label{indexed_topological_carrier_family_type}
    $\indexedTopologicalCarrierFamily{I}{X}$ is a function on $I$.
\end{proposition}
\begin{proof}
    $\indexedTopologicalCarrierFamily{I}{X}\in
        \funs{I}{\ran{\indexedTopologicalCarrierFamily{I}{X}}}$ by
        \auto{indexed_topological_carrier_family,
            logical_graph_on_domain_function}.
    Follows by \obv{funs_is_function,funs}.
\end{proof}

\begin{proposition}\label{indexed_topological_carrier_family_apply}
    Let $i\in I$.
    Then $(\indexedTopologicalCarrierFamily{I}{X})(i)
        =\carrier[X(i)]$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{indexed_topological_carrier_family,
            indexed_topological_carrier_op,
            logical_graph_on_domain_apply}.
\end{proof}
