\section{Open projections from binary products}

\import{topology/product-basis.tex}
\import{topology/open-map.tex}

The image of a rectangle under a projection is either its corresponding
factor or the empty set.  This elementary observation, combined with the
rectangle basis, proves that both product projections are open maps.

\begin{proposition}\label{first_projection_image_rectangle}
    Suppose $U\subseteq S$ and $V\subseteq T$ and $V$ is inhabited.
    Then
        $\img{\firstProductProjection{S}{T}}{U\times V}=U$.
\end{proposition}
\begin{proof}
    Show that every element of
        $\img{\firstProductProjection{S}{T}}{U\times V}$ is an element
        of $U$.
    \begin{subproof}
        Fix $x\in\img{\firstProductProjection{S}{T}}{U\times V}$.
        Take $p\in U\times V$ such that
            $(p,x)\in\firstProductProjection{S}{T}$ by \obv{img_iff}.
        Take $u,v$ such that $u\in U$ and $v\in V$ and $p=(u,v)$ by
            \obv{times_elem_is_tuple}.
        $u\in S$ and $v\in T$ by \obv{subseteq}.
        Hence $p\in S\times T$ by \auto{times_tuple_intro}.
        Hence $x=\fst{p}$ by \auto{first_product_projection_iff}.
        $\fst{p}=u$ by \auto{fst_eq}.
    \end{subproof}
    Hence $\img{\firstProductProjection{S}{T}}{U\times V}
        \subseteq U$ by \obv{subseteq}.
    Take $v$ such that $v\in V$.
    Show that every element of $U$ is an element of
        $\img{\firstProductProjection{S}{T}}{U\times V}$.
    \begin{subproof}
        Fix $u\in U$.
        $u\in S$ and $v\in T$ by \obv{subseteq}.
        Hence $(u,v)\in S\times T$ by \obv{times_tuple_intro}.
        $\fst{(u,v)}=u$ by \obv{fst_eq}.
        Hence $((u,v),u)\in\firstProductProjection{S}{T}$ by
            \obv{first_product_projection_iff}.
        $(u,v)\in U\times V$ by \obv{times_tuple_intro}.
        Follows by \auto{img_iff}.
    \end{subproof}
    Hence $U\subseteq
        \img{\firstProductProjection{S}{T}}{U\times V}$ by
        \obv{subseteq}.
    Follows by \obv{subseteq_antisymmetric}.
\end{proof}

\begin{proposition}\label{second_projection_image_rectangle}
    Suppose $U\subseteq S$ and $V\subseteq T$ and $U$ is inhabited.
    Then
        $\img{\secondProductProjection{S}{T}}{U\times V}=V$.
\end{proposition}
\begin{proof}
    Show that every element of
        $\img{\secondProductProjection{S}{T}}{U\times V}$ is an element
        of $V$.
    \begin{subproof}
        Fix $y\in\img{\secondProductProjection{S}{T}}{U\times V}$.
        Take $p\in U\times V$ such that
            $(p,y)\in\secondProductProjection{S}{T}$ by \obv{img_iff}.
        Take $u,v$ such that $u\in U$ and $v\in V$ and $p=(u,v)$ by
            \obv{times_elem_is_tuple}.
        $u\in S$ and $v\in T$ by \obv{subseteq}.
        Hence $p\in S\times T$ by \auto{times_tuple_intro}.
        Hence $y=\snd{p}$ by \auto{second_product_projection_iff}.
        $\snd{p}=v$ by \auto{snd_eq}.
    \end{subproof}
    Hence $\img{\secondProductProjection{S}{T}}{U\times V}
        \subseteq V$ by \obv{subseteq}.
    Take $u$ such that $u\in U$.
    Show that every element of $V$ is an element of
        $\img{\secondProductProjection{S}{T}}{U\times V}$.
    \begin{subproof}
        Fix $v\in V$.
        $u\in S$ and $v\in T$ by \obv{subseteq}.
        Hence $(u,v)\in S\times T$ by \obv{times_tuple_intro}.
        $\snd{(u,v)}=v$ by \obv{snd_eq}.
        Hence $((u,v),v)\in\secondProductProjection{S}{T}$ by
            \obv{second_product_projection_iff}.
        $(u,v)\in U\times V$ by \obv{times_tuple_intro}.
        Follows by \auto{img_iff}.
    \end{subproof}
    Hence $V\subseteq
        \img{\secondProductProjection{S}{T}}{U\times V}$ by
        \obv{subseteq}.
    Follows by \obv{subseteq_antisymmetric}.
\end{proof}

\begin{proposition}\label{first_projection_image_product_rectangle_open}
    Let $X,Y$ be topological spaces.
    Let $R\in\productRectangles{X}{Y}$.
    Then
        $\img{\firstProductProjection{
            \carrier[X]}{\carrier[Y]}}{R}\in\opens[X]$.
\end{proposition}
\begin{proof}
    Take $U\in\opens[X]$ such that there exists $V\in\opens[Y]$
        such that $R=U\times V$ by \obv{product_rectangles}.
    Take $V\in\opens[Y]$ such that $R=U\times V$.
    $U\subseteq\carrier[X]$ and $V\subseteq\carrier[Y]$ by
        \obv{opens_type}.
    \begin{byCase}
        \caseOf{$V$ is inhabited.}
            $\img{\firstProductProjection{
                \carrier[X]}{\carrier[Y]}}{U\times V}=U$ by
                \obv{first_projection_image_rectangle}.
        \caseOf{$V$ is not inhabited.}
            Hence $V=\emptyset$ by \obv{nonempty_inhabited}.
            $U\times\emptyset=\emptyset$ by \obv{times_empty_right}.
            Hence $U\times V=\emptyset$.
            $\img{\firstProductProjection{
                \carrier[X]}{\carrier[Y]}}{\emptyset}=\emptyset$ by
                \obv{img_emptyset}.
            Hence $\img{\firstProductProjection{
                \carrier[X]}{\carrier[Y]}}{U\times V}=\emptyset$.
            $\emptyset\in\opens[X]$ by \auto{emptyset_open,is_open}.
    \end{byCase}
\end{proof}

\begin{proposition}\label{second_projection_image_product_rectangle_open}
    Let $X,Y$ be topological spaces.
    Let $R\in\productRectangles{X}{Y}$.
    Then
        $\img{\secondProductProjection{
            \carrier[X]}{\carrier[Y]}}{R}\in\opens[Y]$.
\end{proposition}
\begin{proof}
    Take $U\in\opens[X]$ such that there exists $V\in\opens[Y]$
        such that $R=U\times V$ by \obv{product_rectangles}.
    Take $V\in\opens[Y]$ such that $R=U\times V$.
    $U\subseteq\carrier[X]$ and $V\subseteq\carrier[Y]$ by
        \obv{opens_type}.

        \caseOf{$U$ is inhabited.}
            $\img{\secondProductProjection{
                \carrier[X]}{\carrier[Y]}}{U\times V}=V$ by

\begin{theorem}\label{first_product_projection_is_open_map}
    Let $X,Y$ be topological spaces.
    Then $\firstProductProjection{\carrier[X]}{\carrier[Y]}$ is an open
        map from $\BinaryProductTopology{X}{Y}$ to $X$.
\end{theorem}
\begin{proof}
    Let $p_1=\firstProductProjection{\carrier[X]}{\carrier[Y]}$.
    $\firstProductProjection{\carrier[X]}{\carrier[Y]}\in
        \funs{\carrier[X]\times\carrier[Y]}{\carrier[X]}$ by
        \obv{first_product_projection_type}.
    Hence $p_1\in\funs{
        \carrier[X]\times\carrier[Y]}{\carrier[X]}$.
    $\carrier[\BinaryProductTopology{X}{Y}]
        =\carrier[X]\times\carrier[Y]$ by
        \obv{binary_product_topology_carrier}.
    Hence $\firstProductProjection{\carrier[X]}{\carrier[Y]}\in
        \funs{\carrier[\BinaryProductTopology{X}{Y}]}{\carrier[X]}$.
    Hence $p_1\in\funs{
        \carrier[\BinaryProductTopology{X}{Y}]}{\carrier[X]}$.
    $\productRectangles{X}{Y}$ is an open basis of
        $\BinaryProductTopology{X}{Y}$ by
        \obv{product_rectangles_form_open_basis}.

    Show that for every $O\in\opens[\BinaryProductTopology{X}{Y}]$
        we have $\img{p_1}{O}\in\opens[X]$.
    \begin{subproof}
        Fix $O\in\opens[\BinaryProductTopology{X}{Y}]$.
        Let $G=\{A\in\opens[X]\mid
            \text{there exists $R\in\productRectangles{X}{Y}$ such that
                $R\subseteq O$ and $A=\img{p_1}{R}$}\}$.
        $G\subseteq\opens[X]$ by \obv{subseteq}.
        Hence $\unions{G}\in\opens[X]$ by \obv{opens_unions}.
        Show that $\img{p_1}{O}=\unions{G}$.
        \begin{subproof}
            Show that every element of $\img{p_1}{O}$ is an element of
                $\unions{G}$.
            \begin{subproof}
                Fix $x\in\img{p_1}{O}$.
                Take $p\in O$ such that $(p,x)\in p_1$ by \obv{img_iff}.
                Take $R\in\productRectangles{X}{Y}$ such that
                    $p\in R\subseteq O$ by \obv{open_basis}.
                $\img{\firstProductProjection{
                    \carrier[X]}{\carrier[Y]}}{R}\in\opens[X]$ by
                    \obv{first_projection_image_product_rectangle_open}.
                Hence $\img{p_1}{R}\in\opens[X]$.
                Hence $\img{p_1}{R}\in G$.
                $x\in\img{p_1}{R}$ by \auto{img_iff}.
                Follows by \auto{unions_iff}.
            \end{subproof}
            Show that every element of $\unions{G}$ is an element of
                $\img{p_1}{O}$.
            \begin{subproof}
                Fix $x\in\unions{G}$.
                Take $A\in G$ such that $x\in A$ by \obv{unions_iff}.
                Take $R\in\productRectangles{X}{Y}$ such that
                    $R\subseteq O$ and $A=\img{p_1}{R}$.
                Hence $x\in\img{p_1}{R}$.
                $\img{p_1}{R}\subseteq\img{p_1}{O}$ by
                    \obv{img_subseteq}.
                Follows by \obv{subseteq}.
            \end{subproof}
            Follows by set extensionality.
        \end{subproof}
    \end{subproof}
    For every $O\in\opens[\BinaryProductTopology{X}{Y}]$ we have
        $\img{\firstProductProjection{
            \carrier[X]}{\carrier[Y]}}{O}\in\opens[X]$ by
        \auto.
    Follows by \obv{open_map}.
\end{proof}
