\section{Hausdorff subspaces}

\import{topology/separation-continuity.tex}
\import{topology/subspace-continuity.tex}

\begin{theorem}\label{subspace_of_hausdorff_space}
    Suppose $X$ is Hausdorff.
    Then $\subspaceTopology{X}{A}$ is Hausdorff.
\end{theorem}
\begin{proof}
    Let $S=\subspaceTopology{X}{A}$.
    $\identity{\carrier[S]}$ is continuous from $S$ to $X$ by \obv{subspace_inclusion_continuous}.
    $\identity{\carrier[S]}\in\bijections{\carrier[S]}{\carrier[S]}$ by \obv{id_elem_bijections}.
    Hence $\identity{\carrier[S]}$ is injective by \obv{bijections}.
    Follows by \auto{continuous_injection_into_hausdorff}.
\end{proof}
