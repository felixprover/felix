\section{Continuity into metric topologies}

\import{topology/continuity.tex}
\import{topology/induced-topology.tex}

\begin{theorem}\label{metric_continuous_iff_local_balls}
    Let $X$ be a topological space.
    Let $M$ be a pseudometric space.
    Let $g\in\funs{\carrier[X]}{\carrier[M]}$.
    Then $g$ is continuous from $X$ to $\metricTopology{M}$ iff
        for every $x\in\carrier[X]$ we have
        for every $e\in\reals$ we have if
            $\surrealLt{\surrealZero}{e}$, then there exists
            $U\in\opens[X]$ such that $x\in U$ and
            for every $y\in U$ we have
                $g(y)\in\metricBall{M}{g(x)}{e}$.
\end{theorem}
\begin{proof}
    Let $T=\metricTopology{M}$.
    $T$ is a topological space
        by \obv{metric_topology_is_topological_space}.
    $\carrier[T]=\carrier[M]$ and $\opens[T]=\metricOpens{M}$
        by \auto{metric_topology_carrier,metric_topology_opens}.
    Show that if $g$ is continuous from $X$ to $T$, then
        for every $x\in\carrier[X]$ we have
        for every $e\in\reals$ we have if
            $\surrealLt{\surrealZero}{e}$, then there exists
            $U\in\opens[X]$ such that $x\in U$ and
            for every $y\in U$ we have
                $g(y)\in\metricBall{M}{g(x)}{e}$.
    \begin{subproof}
        Assume $g$ is continuous from $X$ to $T$.
        Fix $x\in\carrier[X]$.
        Fix $e\in\reals$.
        Assume $\surrealLt{\surrealZero}{e}$.
        $g(x)\in\carrier[M]$ by \obv{funs_type_apply}.
        $\metricBall{M}{g(x)}{e}\in\metricOpens{M}$
            by \obv{metric_ball_is_open}.
        Hence $\metricBall{M}{g(x)}{e}\in\opens[T]$.
        $g(x)\in\metricBall{M}{g(x)}{e}$
            by \obv{metric_ball_contains_centre}.
        Take $U\in\opens[X]$ such that $x\in U$ and
            for every $y\in U$ we have
                $g(y)\in\metricBall{M}{g(x)}{e}$
            by \auto{continuous_iff_open_local}.
        Follows by \auto.
    \end{subproof}
    Show that if for every $x\in\carrier[X]$ we have
        for every $e\in\reals$ we have if
            $\surrealLt{\surrealZero}{e}$, then there exists
            $U\in\opens[X]$ such that $x\in U$ and
            for every $y\in U$ we have
                $g(y)\in\metricBall{M}{g(x)}{e}$, then
        $g$ is continuous from $X$ to $T$.
    \begin{subproof}
        Assume for every $x\in\carrier[X]$ we have
            for every $e\in\reals$ we have if
                $\surrealLt{\surrealZero}{e}$, then there exists
                $U\in\opens[X]$ such that $x\in U$ and
                for every $y\in U$ we have
                    $g(y)\in\metricBall{M}{g(x)}{e}$.
        Show that for every $x\in\carrier[X]$ we have
            for every $V\in\opens[T]$ we have if $g(x)\in V$, then
            there exists $U\in\opens[X]$ such that $x\in U$ and
                for every $y\in U$ we have $g(y)\in V$.
        \begin{subproof}
            Fix $x\in\carrier[X]$.
            Fix $V\in\opens[T]$.
            Assume $g(x)\in V$.
            $V\in\metricOpens{M}$.
            Hence $\metricOpen{M}{V}$ by \obv{metric_opens_iff}.
            Take $e\in\reals$ such that
                $\surrealLt{\surrealZero}{e}$ and
                $\metricBall{M}{g(x)}{e}\subseteq V$
                by \obv{metric_open}.
            Take $U\in\opens[X]$ such that $x\in U$ and
                for every $y\in U$ we have
                    $g(y)\in\metricBall{M}{g(x)}{e}$.
            For every $y\in U$ we have $g(y)\in V$
                by \obv{subseteq}.
            Follows by \auto.
        \end{subproof}
        Follows by \auto{continuous_iff_open_local}.
    \end{subproof}
    Follows by \auto.
\end{proof}
