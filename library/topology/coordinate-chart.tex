\section{Coordinate charts on a bare set}

\import{topology/partial-homeomorphism.tex}

A raw chart gives coordinates without assuming a topology on its source.
Compatibility will supply that topology. Every chart is just its bijective
graph; its source domain and coordinate image are recovered from the graph.

\begin{definition}\label{raw_coordinate_chart}
    $c$ is a raw coordinate chart on $X$ modelled on $E$ iff
        $E$ is a topological space and $c$ is an injection and
        $\dom{c}\subseteq X$ and $\ran{c}\in\opens[E]$.
\end{definition}

\begin{definition}\label{chart_transition}
    $\chartTransition{d}{c}=d\circ\converse{c}$.
\end{definition}

Composition already restricts to the common source domain. The following
identity makes that overlap explicit without adding another graph operation.

\begin{theorem}\label{chart_transition_overlap}
    Suppose $c$ and $d$ are raw coordinate charts on $X$ modelled on $E$.
    Then $\chartTransition{d}{c}
        =(\restrl{d}{\dom{c}\inter\dom{d}})
            \circ\converse{\restrl{c}{\dom{c}\inter\dom{d}}}$.
\end{theorem}
\begin{proof}
    $c$ and $d$ are injections by \obv{raw_coordinate_chart}.
    Follows by \auto{chart_transition,transition_restrict_overlap}.
\end{proof}

\begin{theorem}\label{chart_transition_domain}
    Suppose $c$ and $d$ are raw coordinate charts on $X$ modelled on $E$.
    Then $\dom{\chartTransition{d}{c}}
        =\img{c}{\dom{c}\inter\dom{d}}$.
\end{theorem}
\begin{proof}
    $c$ and $d$ are injections by \obv{raw_coordinate_chart}.
    Follows by \auto{chart_transition,partial_transition_domain}.
\end{proof}

\begin{theorem}\label{chart_transition_range}
    Suppose $c$ and $d$ are raw coordinate charts on $X$ modelled on $E$.
    Then $\ran{\chartTransition{d}{c}}
        =\img{d}{\dom{c}\inter\dom{d}}$.
\end{theorem}
\begin{proof}
    $c$ and $d$ are injections by \obv{raw_coordinate_chart}.
    $\converse{c}$ is a relation by \obv{converse_is_relation}.
    $\ran{d\circ\converse{c}}=\img{d}{\ran{\converse{c}}}$ by
        \obv{relation_composition_range}.
    $\ran{\converse{c}}=\dom{c}$ by \obv{ran_converse}.
    $\img{d}{\dom{d}\inter\dom{c}}=\img{d}{\dom{c}}$ by
        \obv{image_domain_trace}.
    $\dom{d}\inter\dom{c}=\dom{c}\inter\dom{d}$ by \obv{inter_comm}.
    Follows by \auto{chart_transition}.
\end{proof}

\begin{theorem}\label{chart_transition_converse}
    Suppose $c$ and $d$ are raw coordinate charts on $X$ modelled on $E$.
    Then $\converse{\chartTransition{d}{c}}=\chartTransition{c}{d}$.
\end{theorem}
\begin{proof}
    $c$ and $d$ are injections by \obv{raw_coordinate_chart}.
    $\converse{d\circ\converse{c}}
        =\converse{\converse{c}}\circ\converse{d}$ by \obv{circ_converse}.
    $\converse{\converse{c}}=c$ by \obv{converse_converse_eq}.
    Follows by \auto{chart_transition}.
\end{proof}

\begin{theorem}\label{chart_transition_self}
    Suppose $c$ is a raw coordinate chart on $X$ modelled on $E$.
    Then $\chartTransition{c}{c}=\identity{\ran{c}}$.
\end{theorem}
\begin{proof}
    $c$ is an injection by \obv{raw_coordinate_chart}.
    $c\in\bijections{\dom{c}}{\ran{c}}$ by
        \obv{injective_function_bijection_to_range}.
    $c\circ\converse{c}=\identity{\ran{c}}$ by \obv{bijection_circ_converse}.
    Follows by \obv{chart_transition}.
\end{proof}

\begin{theorem}\label{chart_transition_change_of_source}
    Let $c,d$ be relations and $h\in\bijections{A}{B}$.
    Suppose $\dom{c}\subseteq A$.
    Then $\chartTransition{d\circ\converse{h}}{c\circ\converse{h}}
        =\chartTransition{d}{c}$.
\end{theorem}
\begin{proof}
    $h$ is a function by \obv{bijection_is_function}.
    $\converse{\converse{h}}=h$ by \obv{converse_converse_eq}.
    $\converse{c\circ\converse{h}}=h\circ\converse{c}$ by \auto{circ_converse}.
    $\converse{h}\circ h=\identity{A}$ by \obv{bijection_converse_circ}.
    $\converse{c}$ is a relation and $\ran{\converse{c}}=\dom{c}$ by
        \obv{converse_is_relation,ran_converse}.
    $\identity{A}\circ\converse{c}=\converse{c}$ by \obv{relation_identity_circ}.
    Follows by \auto{chart_transition,circ_assoc}.
\end{proof}

\begin{theorem}\label{chart_transition_source_restriction}
    Let $c$ be an injection.
    Let $d$ be a relation.
    Then $\chartTransition{d}{\restrl{c}{U}}
        =\restrl{\chartTransition{d}{c}}{\img{c}{U}}$.
\end{theorem}
\begin{proof}
    $\converse{\restrl{c}{U}}=\restrl{\converse{c}}{\img{c}{U}}$ by \obv{restriction_converse}.
    Follows by \auto{chart_transition,restrl_circ}.
\end{proof}

\begin{theorem}\label{chart_transition_target_restriction}
    Let $c,d$ be relations.
    Suppose $\dom{c}\subseteq U$.
    Then $\chartTransition{\restrl{d}{U}}{c}=\chartTransition{d}{c}$.
\end{theorem}
\begin{proof}
    $\converse{c}$ is a relation and $\ran{\converse{c}}=\dom{c}$ by
        \obv{converse_is_relation,ran_converse}.
    $\restrl{d}{U}$ is a relation by \auto{relation,restrl_subseteq,subseteq}.
    $\restrl{\restrl{d}{U}}{\dom{c}}=\restrl{d}{\dom{c}}$ by \obv{restrl_restrl}.
    Follows by \auto{chart_transition,composition_restrict_intermediate_range}.
\end{proof}

\begin{theorem}\label{chart_transition_restriction}
    Let $c$ be an injection.
    Let $d$ be a relation.
    Suppose $U\subseteq\dom{c}$.
    Then $\chartTransition{\restrl{d}{U}}{\restrl{c}{U}}
        =\restrl{\chartTransition{d}{c}}{\img{c}{U}}$.
\end{theorem}
\begin{proof}
    $\dom{\restrl{c}{U}}=\dom{c}\inter U$ by \obv{dom_of_restrl_eq_inter}.
    $\dom{c}\inter U=U$ by \obv{inter_absorb_supseteq_left}.
    $\ran{\converse{\restrl{c}{U}}}=U$ by \auto{ran_converse}.
    $\converse{\restrl{c}{U}}$ is a relation by \obv{converse_is_relation}.
    $(\restrl{d}{U})\circ\converse{\restrl{c}{U}}
        =d\circ\converse{\restrl{c}{U}}$ by \auto{composition_restrict_intermediate_range}.
    $\converse{\restrl{c}{U}}=\restrl{\converse{c}}{\img{c}{U}}$ by \obv{restriction_converse}.
    Follows by \auto{chart_transition,restrl_circ}.
\end{proof}

\begin{theorem}\label{raw_coordinate_chart_restriction}
    Suppose $c$ is a raw coordinate chart on $X$ modelled on $E$.
    Suppose $\img{c}{A}\in\opens[E]$.
    Then $\restrl{c}{A}$ is a raw coordinate chart on $X$ modelled on $E$.
\end{theorem}
\begin{proof}
    $E$ is a topological space and $c$ is an injection and
        $\dom{c}\subseteq X$ by \obv{raw_coordinate_chart}.
    $\restrl{c}{A}$ is a function and $\restrl{c}{A}$ is injective by
        \obv{restrl_of_function_is_function,restrl_injective}.
    $\dom{\restrl{c}{A}}=\dom{c}\inter A$ by
        \obv{dom_of_restrl_eq_inter}.
    Hence $\dom{\restrl{c}{A}}\subseteq X$ by
        \auto{inter_lower_left,subseteq_transitive}.
    $\ran{\restrl{c}{A}}=\img{c}{A}$ by \obv{restriction_range}.
    Follows by \obv{raw_coordinate_chart}.
\end{proof}

\subsection{Topological compatibility}

\begin{definition}\label{raw_chart_compatibility}
    $c$ is topologically compatible with $d$ on $X$ modelled on $E$ iff
        $c$ and $d$ are raw coordinate charts on $X$ modelled on $E$ and
        $\chartTransition{d}{c}$ is an open partial homeomorphism from $E$ to $E$.
\end{definition}

\begin{theorem}\label{raw_chart_compatibility_reflexive}
    Suppose $c$ is a raw coordinate chart on $X$ modelled on $E$.
    Then $c$ is topologically compatible with $c$ on $X$ modelled on $E$.
\end{theorem}
\begin{proof}
    $E$ is a topological space and $\ran{c}\in\opens[E]$ by
        \obv{raw_coordinate_chart}.
    $\identity{\ran{c}}$ is an open partial homeomorphism from $E$ to $E$ by
        \obv{identity_open_partial_homeomorphism}.
    $\chartTransition{c}{c}=\identity{\ran{c}}$ by \obv{chart_transition_self}.
    Follows by \obv{raw_chart_compatibility}.
\end{proof}

\begin{theorem}\label{raw_chart_compatibility_symmetric}
    Suppose $c$ is topologically compatible with $d$ on $X$ modelled on $E$.
    Then $d$ is topologically compatible with $c$ on $X$ modelled on $E$.
\end{theorem}
\begin{proof}
    $c$ and $d$ are raw coordinate charts on $X$ modelled on $E$ and
        $\chartTransition{d}{c}$ is an open partial homeomorphism from $E$ to $E$
        by \obv{raw_chart_compatibility}.
    $\converse{\chartTransition{d}{c}}$ is an open partial homeomorphism
        from $E$ to $E$ by \obv{open_partial_homeomorphism_converse}.
    $\converse{\chartTransition{d}{c}}=\chartTransition{c}{d}$ by
        \obv{chart_transition_converse}.
    Follows by \obv{raw_chart_compatibility}.
\end{proof}

\subsection{Charts on an existing space}

\begin{theorem}\label{open_partial_homeomorphism_is_raw_chart}
    Suppose $c$ is an open partial homeomorphism from $X$ to $E$.
    Then $c$ is a raw coordinate chart on $\carrier[X]$ modelled on $E$.
\end{theorem}
\begin{proof}
    $X$ is a topological space and $E$ is a topological space and
        $\dom{c}\in\opens[X]$ and $\ran{c}\in\opens[E]$ by
        \obv{open_partial_homeomorphism}.
    $\dom{c}\subseteq\carrier[X]$ by \obv{opens_type}.
    $c\in\bijections{\dom{c}}{\ran{c}}$ by
        \obv{open_partial_homeomorphism_bijection}.
    $c$ is a function and $c$ is injective by
        \obv{bijection_is_function,bijections}.
    Follows by \obv{raw_coordinate_chart}.
\end{proof}

\begin{theorem}\label{open_partial_homeomorphisms_are_compatible}
    Suppose $c$ and $d$ are open partial homeomorphisms from $X$ to $E$.
    Then $c$ is topologically compatible with $d$ on $\carrier[X]$
        modelled on $E$.
\end{theorem}
\begin{proof}
    $c$ and $d$ are raw coordinate charts on $\carrier[X]$ modelled on $E$
        by \obv{open_partial_homeomorphism_is_raw_chart}.
    $\converse{c}$ is an open partial homeomorphism from $E$ to $X$ by
        \obv{open_partial_homeomorphism_converse}.
    $d\circ\converse{c}$ is an open partial homeomorphism from $E$ to $E$ by
        \obv{open_partial_homeomorphism_composition}.
    $\chartTransition{d}{c}=d\circ\converse{c}$ by \obv{chart_transition}.
    Follows by \obv{raw_chart_compatibility}.
\end{proof}
