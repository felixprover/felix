\section{Neighbourhoods within sets}

\import{topology/neighbourhood-filter.tex}
\import{set/filter-within.tex}

To speak about an arbitrary ambient set, we first intersect it with the
carrier.  The resulting filter stays on the ambient carrier; it is allowed to
be improper when the set is locally absent.

\begin{definition}\label{neighbourhoods_within}
    $\neighbourhoodsWithin{x}{A}{X}
        =\filterWithin{\carrier[X]}{\neighbourhoods{x}{X}}{
            A\inter\carrier[X]}$.
\end{definition}

\begin{classification}\label{neighbourhoods_within_is_filter}
    Let $X$ be a topological space and $x\in\carrier[X]$.
    Then $\neighbourhoodsWithin{x}{A}{X}$ is a filter on $\carrier[X]$.
\end{classification}
\begin{proof}
    $\neighbourhoods{x}{X}$ is a filter on $\carrier[X]$ by
        \obv{neighbourhoods_is_filter}.
    $A\inter\carrier[X]\subseteq\carrier[X]$ by
        \obv{inter_lower_right}.
    Follows by \auto{neighbourhoods_within,filter_within_is_filter}.
\end{proof}

\begin{theorem}\label{neighbourhoods_within_iff}
    Let $X$ be a topological space and $x\in\carrier[X]$.
    Then $B\in\neighbourhoodsWithin{x}{A}{X}$ iff
        $B\subseteq\carrier[X]$ and there exists
        $N\in\neighbourhoods{x}{X}$ such that
        $N\inter(A\inter\carrier[X])\subseteq B$.
\end{theorem}
\begin{proof}
    $\neighbourhoods{x}{X}$ is a filter on $\carrier[X]$ by
        \obv{neighbourhoods_is_filter}.
    $A\inter\carrier[X]\subseteq\carrier[X]$ by
        \obv{inter_lower_right}.
    Follows by \auto{neighbourhoods_within,filter_within_iff}.
\end{proof}

\begin{proposition}\label{neighbourhoods_within_refines_neighbourhoods}
    Let $X$ be a topological space and $x\in\carrier[X]$.
    Then $\neighbourhoodsWithin{x}{A}{X}$ refines
        $\neighbourhoods{x}{X}$ on $\carrier[X]$.
\end{proposition}
\begin{proof}
    $\neighbourhoodsWithin{x}{A}{X}$ is a filter on $\carrier[X]$ by
        \obv{neighbourhoods_within_is_filter}.
    $\neighbourhoods{x}{X}$ is a filter on $\carrier[X]$ by
        \obv{neighbourhoods_is_filter}.
    Show that $\neighbourhoods{x}{X}
        \subseteq\neighbourhoodsWithin{x}{A}{X}$.
    \begin{subproof}
        It suffices to show that every element of $\neighbourhoods{x}{X}$
            is an element of $\neighbourhoodsWithin{x}{A}{X}$ by
            \obv{subseteq}.
        Fix $N\in\neighbourhoods{x}{X}$.
        $N\subseteq\carrier[X]$ by \obv{filter}.
        $N\inter(A\inter\carrier[X])\subseteq N$ by
            \obv{inter_lower_left}.
        Follows by \auto{neighbourhoods_within_iff}.
    \end{subproof}
    Follows by \obv{filter_refines}.
\end{proof}

\begin{proposition}\label{neighbourhoods_within_carrier}
    Let $X$ be a topological space and $x\in\carrier[X]$.
    Then $\neighbourhoodsWithin{x}{\carrier[X]}{X}
        =\neighbourhoods{x}{X}$.
\end{proposition}
\begin{proof}
    $\neighbourhoods{x}{X}$ is a filter on $\carrier[X]$ by
        \obv{neighbourhoods_is_filter}.
    $\carrier[X]\inter\carrier[X]=\carrier[X]$ by
        \obv{inter_idempotent}.
    $\filterWithin{\carrier[X]}{\neighbourhoods{x}{X}}{\carrier[X]}
        =\neighbourhoods{x}{X}$ by \obv{filter_within_carrier}.
    Follows by \obv{neighbourhoods_within}.
\end{proof}

\begin{proposition}\label{neighbourhoods_within_emptyset}
    Let $X$ be a topological space and $x\in\carrier[X]$.
    Then $\neighbourhoodsWithin{x}{\emptyset}{X}
        =\pow{\carrier[X]}$.
\end{proposition}
\begin{proof}
    $\neighbourhoods{x}{X}$ is a filter on $\carrier[X]$ by
        \obv{neighbourhoods_is_filter}.
    $\emptyset\inter\carrier[X]=\emptyset$ by
        \auto{inter_emptyset,inter_comm}.
    $\filterWithin{\carrier[X]}{\neighbourhoods{x}{X}}{\emptyset}
        =\pow{\carrier[X]}$ by \obv{filter_within_emptyset}.
    Follows by \obv{neighbourhoods_within}.
\end{proof}

\subsection{Interior and closure}

\begin{theorem}\label{interior_iff_neighbourhood}
    Let $X$ be a topological space and $x\in\carrier[X]$.
    Suppose $A\subseteq\carrier[X]$.
    Then $x\in\interior{A}{X}$ iff $A\in\neighbourhoods{x}{X}$.
\end{theorem}
\begin{proof}
    Show that if $x\in\interior{A}{X}$, then
        $A\in\neighbourhoods{x}{X}$.
    \begin{subproof}
        Assume $x\in\interior{A}{X}$.
        Take $U\in\opens[X]$ such that $x\in U\subseteq A$ by
            \obv{interior_elem_elim}.
        $A\in\pow{\carrier[X]}$ by \obv{powerset_intro}.
        Follows by \auto{neighbourhoods}.
    \end{subproof}
    Show that if $A\in\neighbourhoods{x}{X}$, then
        $x\in\interior{A}{X}$.
    \begin{subproof}
        Assume $A\in\neighbourhoods{x}{X}$.
        Take $U\in\opens[X]$ such that $x\in U\subseteq A$ by
            \obv{neighbourhoods}.
        Follows by \obv{interior_elem_intro}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{theorem}\label{closure_iff_neighbourhood_frequent}
    Let $X$ be a topological space and $x\in\carrier[X]$.
    Then $x\in\closure{A}{X}$ iff
        $A\inter\carrier[X]$ is frequent along
        $\neighbourhoods{x}{X}$ on $\carrier[X]$.
\end{theorem}
\begin{proof}
    $\neighbourhoods{x}{X}$ is a filter on $\carrier[X]$ by
        \obv{neighbourhoods_is_filter}.
    $A\inter\carrier[X]\subseteq\carrier[X]$ by
        \obv{inter_lower_right}.
    Hence $\carrier[X]\setminus(A\inter\carrier[X])
        \subseteq\carrier[X]$ by \obv{setminus_subseteq}.
    $\carrier[X]\setminus
        \interior{(\carrier[X]\setminus(A\inter\carrier[X]))}{X}
        =\closure{(\carrier[X]\setminus
            (\carrier[X]\setminus(A\inter\carrier[X])))}{X}$ by
        \obv{complement_interior_eq_closure_complement}.
    $\carrier[X]\setminus
        (\carrier[X]\setminus(A\inter\carrier[X]))
        =A\inter\carrier[X]$ by
        \obv{double_relative_complement}.
    $\closure{A}{X}=\closure{(A\inter\carrier[X])}{X}$ by
        \obv{closure_trace}.
    Hence
        $\carrier[X]\setminus
            \interior{(\carrier[X]\setminus
                (A\inter\carrier[X]))}{X}
            =\closure{A}{X}$.

    Show that if $x\in\closure{A}{X}$, then
        $A\inter\carrier[X]$ is frequent along
            $\neighbourhoods{x}{X}$ on $\carrier[X]$.
    \begin{subproof}
        Assume $x\in\closure{A}{X}$.
        Hence $x\in\carrier[X]\setminus
            \interior{(\carrier[X]\setminus
                (A\inter\carrier[X]))}{X}$.
        Thus $x\notin\interior{(\carrier[X]\setminus
            (A\inter\carrier[X]))}{X}$ by \obv{setminus}.
        Hence $\carrier[X]\setminus(A\inter\carrier[X])
            \notin\neighbourhoods{x}{X}$ by
            \obv{interior_iff_neighbourhood}.
        Follows by \obv{filter_frequent}.
    \end{subproof}

    Show that if $A\inter\carrier[X]$ is frequent along
        $\neighbourhoods{x}{X}$ on $\carrier[X]$,
        then $x\in\closure{A}{X}$.
    \begin{subproof}
        Assume $A\inter\carrier[X]$ is frequent along
            $\neighbourhoods{x}{X}$ on $\carrier[X]$.
        Hence $\carrier[X]\setminus(A\inter\carrier[X])
            \notin\neighbourhoods{x}{X}$ by
            \obv{filter_frequent}.
        Thus $x\notin\interior{(\carrier[X]\setminus
            (A\inter\carrier[X]))}{X}$ by
            \obv{interior_iff_neighbourhood}.
        Hence $x\in\carrier[X]\setminus
            \interior{(\carrier[X]\setminus
                (A\inter\carrier[X]))}{X}$ by
            \obv{setminus_intro}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{theorem}\label{closure_iff_neighbourhoods_within_proper}
    Let $X$ be a topological space and $x\in\carrier[X]$.
    Then $x\in\closure{A}{X}$ iff
        $\neighbourhoodsWithin{x}{A}{X}$ is a proper filter on
        $\carrier[X]$.
\end{theorem}
\begin{proof}
    $\neighbourhoods{x}{X}$ is a filter on $\carrier[X]$ by
        \obv{neighbourhoods_is_filter}.
    $A\inter\carrier[X]\subseteq\carrier[X]$ by
        \obv{inter_lower_right}.
    $\filterWithin{\carrier[X]}{\neighbourhoods{x}{X}}{
        A\inter\carrier[X]}$ is a proper filter on $\carrier[X]$ iff
        $A\inter\carrier[X]$ is frequent along
            $\neighbourhoods{x}{X}$ on $\carrier[X]$ by
        \obv{filter_within_is_proper_iff_frequent}.
    Follows by
        \auto{neighbourhoods_within,closure_iff_neighbourhood_frequent}.
\end{proof}

\begin{theorem}\label{closure_iff_neighbourhood_meshes_principal}
    Let $X$ be a topological space and $x\in\carrier[X]$.
    Then $x\in\closure{A}{X}$ iff
        $\neighbourhoods{x}{X}$ meshes with
        $\principalfilter{\carrier[X]}{A\inter\carrier[X]}$ on
        $\carrier[X]$.
\end{theorem}
\begin{proof}
    $\neighbourhoods{x}{X}$ is a filter on $\carrier[X]$ by
        \obv{neighbourhoods_is_filter}.
    $A\inter\carrier[X]\subseteq\carrier[X]$ by
        \obv{inter_lower_right}.
    $A\inter\carrier[X]$ is frequent along
        $\neighbourhoods{x}{X}$ on $\carrier[X]$ iff
        $\neighbourhoods{x}{X}$ meshes with
        $\principalfilter{\carrier[X]}{A\inter\carrier[X]}$ on
        $\carrier[X]$ by \obv{filter_frequent_iff_meshes_principal}.
    Follows by \auto{closure_iff_neighbourhood_frequent}.
\end{proof}
