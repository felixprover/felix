\section{Local compactness of products}

\import{topology/local-compactness.tex}
\import{topology/compactness-product.tex}
\import{topology/product-subspace.tex}

Products of compact subsets are compact in the product topology. Taking
products of the open neighbourhoods they contain then proves local compactness.

\begin{theorem}\label{product_of_compact_subsets}
    Let $X,Y$ be topological spaces.
    Suppose $K$ is compact in $X$ and $L$ is compact in $Y$.
    Then $K\times L$ is compact in $\BinaryProductTopology{X}{Y}$.
\end{theorem}
\begin{proof}
    $\BinaryProductTopology{X}{Y}$ is a topological space by
        \obv{binary_product_topology_is_topological_space}.
    $K\times L\subseteq\carrier[\BinaryProductTopology{X}{Y}]$ by
        \auto{compact_in,binary_product_topology_carrier,times,subseteq}.
    $\subspaceTopology{X}{K}$ is a compact topological space and
        $\subspaceTopology{Y}{L}$ is a compact topological space by
        \auto{compact_in,subspace_topology_is_topological_space}.
    Hence $\BinaryProductTopology{\subspaceTopology{X}{K}}{\subspaceTopology{Y}{L}}$
        is compact by \obv{binary_product_of_compact_spaces_is_compact}.
    Hence $\subspaceTopology{\BinaryProductTopology{X}{Y}}{K\times L}$ is compact by
        \obv{product_of_subspaces}.
    Follows by \obv{compact_in}.
\end{proof}

\begin{theorem}\label{binary_product_of_locally_compact_spaces}
    Let $X,Y$ be locally compact topological spaces.
    Then $\BinaryProductTopology{X}{Y}$ is locally compact.
\end{theorem}
\begin{proof}
    Let $P=\BinaryProductTopology{X}{Y}$.
    $P$ is a topological space by \obv{binary_product_topology_is_topological_space}.
    $\carrier[P]=\carrier[X]\times\carrier[Y]$ by \auto{binary_product_topology_carrier}.
    Show that for every $p\in\carrier[P]$ there exist $W,M$ such that
        $W\in\opens[P]$ and $p\in W\subseteq M$ and $M$ is compact in $P$.
    \begin{subproof}
        Fix $p\in\carrier[P]$.
        Take $x,y$ such that $x\in\carrier[X]$ and $y\in\carrier[Y]$ and $p=(x,y)$ by
            \auto{times_elem_is_tuple}.
        Take $U,K$ such that $U\in\opens[X]$ and $x\in U\subseteq K$ and
            $K$ is compact in $X$ by \obv{is_locally_compact}.
        Take $V,L$ such that $V\in\opens[Y]$ and $y\in V\subseteq L$ and
            $L$ is compact in $Y$ by \obv{is_locally_compact}.
        $U\times V\in\opens[P]$ by \auto{product_of_open_sets_is_open}.
        $p\in U\times V$ by \auto{times_tuple_intro}.
        $K\times L$ is compact in $P$ by \obv{product_of_compact_subsets}.
        Follows by \auto{times,subseteq}.
    \end{subproof}
    Follows by \auto{is_locally_compact}.
\end{proof}
