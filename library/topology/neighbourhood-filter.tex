\section{Neighbourhood filters}

\import{topology/topological-space.tex}
\import{set/filter-order.tex}

The open sets which contain a point form the canonical basis of its
neighbourhood filter.  We retain the direct neighbourhood definition for
membership proofs and identify it here with the filter generated by that
basis.

\begin{definition}\label{open_neighbourhood_basis}
    $\openNeighbourhoodBasis{x}{X}
        =\{U\in\opens[X]\mid x\in U\}$.
\end{definition}

\begin{proposition}\label{open_neighbourhood_basis_type}
    Let $X$ be a topological space.
    Then $\openNeighbourhoodBasis{x}{X}$ is a family of subsets of
        $\carrier[X]$.
\end{proposition}
\begin{proof}
    $\openNeighbourhoodBasis{x}{X}\subseteq\opens[X]$ by
        \obv{open_neighbourhood_basis,subseteq}.
    Follows by \obv{opens_type,pow_iff,subseteq}.
\end{proof}

\begin{theorem}\label{open_neighbourhood_basis_is_filter_basis}
    Let $X$ be a topological space and $x\in\carrier[X]$.
    Then $\openNeighbourhoodBasis{x}{X}$ is a filter basis on
        $\carrier[X]$.
\end{theorem}
\begin{proof}
    $\openNeighbourhoodBasis{x}{X}$ is a family of subsets of
        $\carrier[X]$ by \obv{open_neighbourhood_basis_type}.
    $\carrier[X]\in\opens[X]$ by \obv{carrier_open}.
    Hence $\carrier[X]\in\openNeighbourhoodBasis{x}{X}$ by
        \obv{open_neighbourhood_basis}.
    Thus $\openNeighbourhoodBasis{x}{X}$ is inhabited.

    Show that for all $U,V$ such that
        $U\in\openNeighbourhoodBasis{x}{X}$ and
        $V\in\openNeighbourhoodBasis{x}{X}$
        there exists $W\in\openNeighbourhoodBasis{x}{X}$ such that
        $W\subseteq U\inter V$.
    \begin{subproof}
        Fix $U,V$.
        Assume $U\in\openNeighbourhoodBasis{x}{X}$ and
            $V\in\openNeighbourhoodBasis{x}{X}$.
        $U,V\in\opens[X]$ and $x\in U\inter V$ by
            \obv{open_neighbourhood_basis,inter}.
        Hence $U\inter V\in\opens[X]$ by \obv{opens_inter}.
        Thus $U\inter V\in\openNeighbourhoodBasis{x}{X}$ by
            \obv{open_neighbourhood_basis}.
        Follows by \obv{subseteq_refl}.
    \end{subproof}
    Follows by \obv{filter_basis}.
\end{proof}

\begin{classification}\label{neighbourhoods_is_proper_filter}
    Let $X$ be a topological space.
    Let $x\in\carrier[X]$.
    Then $\neighbourhoods{x}{X}$ is a proper filter on $\carrier[X]$.
\end{classification}
\begin{proof}
    $\neighbourhoods{x}{X}$ is a family of subsets of $\carrier[X]$
        by \auto{neighbourhoods,pow_iff,subseteq}.
    $\carrier[X]\in\opens[X]$ by \obv{carrier_open}.
    Hence $\carrier[X]\in\neighbourhoods{x}{X}$
        by \auto{neighbourhoods,pow_iff,subseteq_refl}.

    Show that $\emptyset\notin\neighbourhoods{x}{X}$.
    \begin{subproof}
        Suppose not.
        Take $U\in\opens[X]$ such that $x\in U\subseteq\emptyset$
            by \auto{neighbourhoods}.
        Hence $x\in\emptyset$ by \obv{subseteq}.
        Contradiction by \obv{emptyset}.
    \end{subproof}

    Show that $\neighbourhoods{x}{X}$ is closed under binary intersections.
    \begin{subproof}
        Fix $A,B\in\neighbourhoods{x}{X}$.
        Take $U\in\opens[X]$ such that $x\in U\subseteq A$
            by \obv{neighbourhoods}.
        Take $V\in\opens[X]$ such that $x\in V\subseteq B$
            by \obv{neighbourhoods}.
        $U\inter V\in\opens[X]$ by \obv{opens_inter}.
        $x\in U\inter V$ by \obv{inter}.
        $U\inter V\subseteq A\inter B$
            by \obv{inter,subseteq}.
        $A,B\subseteq\carrier[X]$
            by \obv{neighbourhoods,pow_iff,subseteq}.
        Hence $A\inter B\subseteq\carrier[X]$
            by \obv{inter_subseteq}.
        Hence $A\inter B\in\pow{\carrier[X]}$
            by \obv{powerset_intro}.
        Follows by \obv{neighbourhoods}.
    \end{subproof}

    Show that $\neighbourhoods{x}{X}$ is upward-closed in $\carrier[X]$.
    \begin{subproof}
        Fix $A,B$.
        Assume $A\subseteq B\subseteq\carrier[X]$ and
            $A\in\neighbourhoods{x}{X}$.
        Take $U\in\opens[X]$ such that $x\in U\subseteq A$
            by \obv{neighbourhoods}.
        Hence $U\subseteq B$ by \obv{subseteq_transitive}.
        $B\in\pow{\carrier[X]}$ by \obv{powerset_intro}.
        Follows by \obv{neighbourhoods}.
    \end{subproof}
    Hence $\neighbourhoods{x}{X}$ is a filter on $\carrier[X]$
        by \obv{filter}.
    Follows by \obv{proper_filter}.
\end{proof}

\begin{classification}\label{neighbourhoods_is_filter}
    Let $X$ be a topological space.
    Let $x\in\carrier[X]$.
    Then $\neighbourhoods{x}{X}$ is a filter on $\carrier[X]$.
\end{classification}
\begin{proof}
    $\neighbourhoods{x}{X}$ is a proper filter on $\carrier[X]$
        by \obv{neighbourhoods_is_proper_filter}.
    Follows by \obv{proper_filter_is_filter}.
\end{proof}

\begin{proposition}\label{open_set_containing_point_is_neighbourhood}
    Let $X$ be a topological space and $x\in\carrier[X]$.
    Suppose $U\in\opens[X]$ and $x\in U$.
    Then $U\in\neighbourhoods{x}{X}$.
\end{proposition}
\begin{proof}
    $U\subseteq\carrier[X]$ by \obv{opens_type,powerset_elim}.
    Hence $U\in\pow{\carrier[X]}$ by \obv{powerset_intro}.
    There exists $V\in\opens[X]$ such that $x\in V\subseteq U$ by
        \obv{subseteq_refl}.
    Follows by \obv{neighbourhoods}.
\end{proof}

\begin{theorem}\label{neighbourhood_filter_is_generated_by_open_neighbourhoods}
    Let $X$ be a topological space and $x\in\carrier[X]$.
    Then
        $\neighbourhoods{x}{X}
            =\generatedFilter{\carrier[X]}{\openNeighbourhoodBasis{x}{X}}$.
\end{theorem}
\begin{proof}
    $\openNeighbourhoodBasis{x}{X}$ is a filter basis on $\carrier[X]$
        by \obv{open_neighbourhood_basis_is_filter_basis}.
    Let $G=\{N\in\pow{\carrier[X]}\mid
        \text{there exists $U\in\openNeighbourhoodBasis{x}{X}$
            such that $U\subseteq N$}\}$.
    Hence
        $\generatedFilter{\carrier[X]}{\openNeighbourhoodBasis{x}{X}}
            =G$ by
        \obv{filter_basis_generated_filter_eq_supersets}.

    Show that $\neighbourhoods{x}{X}\subseteq G$.
    \begin{subproof}
        It suffices to show that every element of $\neighbourhoods{x}{X}$
            is an element of $G$ by \obv{subseteq}.
        Fix $N\in\neighbourhoods{x}{X}$.
        Follows by \auto{neighbourhoods,open_neighbourhood_basis}.
    \end{subproof}
    Show that $G\subseteq\neighbourhoods{x}{X}$.
    \begin{subproof}
        It suffices to show that every element of $G$ is an element of
            $\neighbourhoods{x}{X}$ by \obv{subseteq}.
        Fix $N\in G$.
        Follows by \auto{neighbourhoods,open_neighbourhood_basis}.
    \end{subproof}
    Hence $\neighbourhoods{x}{X}=G$ by \obv{subseteq_antisymmetric}.
\end{proof}

\begin{corollary}\label{open_neighbourhoods_are_neighbourhood_filter_basis}
    Let $X$ be a topological space and $x\in\carrier[X]$.
    Then $\openNeighbourhoodBasis{x}{X}$ is a basis of
        $\neighbourhoods{x}{X}$ on $\carrier[X]$.
\end{corollary}
\begin{proof}
    $\neighbourhoods{x}{X}$ is a filter on $\carrier[X]$ by
        \obv{neighbourhoods_is_filter}.
    Show that $\openNeighbourhoodBasis{x}{X}
        \subseteq\neighbourhoods{x}{X}$.
    \begin{subproof}
        It suffices to show that every element of
            $\openNeighbourhoodBasis{x}{X}$ is an element of
            $\neighbourhoods{x}{X}$ by \obv{subseteq}.
        Fix $U\in\openNeighbourhoodBasis{x}{X}$.
        $U\in\opens[X]$ and $x\in U$ by
            \obv{open_neighbourhood_basis}.
        Follows by \obv{open_set_containing_point_is_neighbourhood}.
    \end{subproof}
    Show that $\openNeighbourhoodBasis{x}{X}$ refines
        $\neighbourhoods{x}{X}$ as filter families on $\carrier[X]$.
    \begin{subproof}
        $\openNeighbourhoodBasis{x}{X}$ is a family of subsets of
            $\carrier[X]$ by \obv{open_neighbourhood_basis_type}.
        $\neighbourhoods{x}{X}$ is a family of subsets of $\carrier[X]$
            by \obv{filter}.
        Show that for every $N\in\neighbourhoods{x}{X}$ there exists
            $U\in\openNeighbourhoodBasis{x}{X}$ such that $U\subseteq N$.
        \begin{subproof}
            Fix $N\in\neighbourhoods{x}{X}$.
            Take $U\in\opens[X]$ such that $x\in U\subseteq N$ by
                \obv{neighbourhoods}.
            Hence $U\in\openNeighbourhoodBasis{x}{X}$ by
                \obv{open_neighbourhood_basis}.
        \end{subproof}
        Follows by \obv{filter_family_refinement}.
    \end{subproof}
    Follows by \obv{filter_basis_of}.
\end{proof}

\begin{proposition}\label{pure_filter_refines_neighbourhood_filter}
    Let $X$ be a topological space and $x\in\carrier[X]$.
    Then $\pureFilter{\carrier[X]}{x}$ refines
        $\neighbourhoods{x}{X}$ on $\carrier[X]$.
\end{proposition}
\begin{proof}
    $\pureFilter{\carrier[X]}{x}$ is a filter on $\carrier[X]$ by
        \auto{pure_filter,principalfilter_singleton_is_filter}.
    $\neighbourhoods{x}{X}$ is a filter on $\carrier[X]$ by
        \obv{neighbourhoods_is_filter}.
    Show that $\neighbourhoods{x}{X}
        \subseteq\pureFilter{\carrier[X]}{x}$.
    \begin{subproof}
        It suffices to show that every element of $\neighbourhoods{x}{X}$
            is an element of $\pureFilter{\carrier[X]}{x}$ by
            \obv{subseteq}.
        Fix $N\in\neighbourhoods{x}{X}$.
        Take $U\in\opens[X]$ such that $x\in U\subseteq N$ by
            \obv{neighbourhoods}.
        Hence $x\in N$ by \obv{subseteq}.
        $N\subseteq\carrier[X]$ by
            \obv{neighbourhoods,pow_iff,subseteq}.
        Follows by \obv{pure_filter_iff}.
    \end{subproof}
    Follows by \obv{filter_refines}.
\end{proof}
