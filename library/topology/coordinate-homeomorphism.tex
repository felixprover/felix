\section{Transporting coordinate structures}

\import{topology/coordinate-restriction.tex}

A homeomorphism transports a relative coordinate structure by composing its
charts with the inverse. The transported maximal atlas is already maximal.
Its model and pseudogroup are unchanged; no regularity of the homeomorphism
in a finer calculus is assumed.

\begin{construction}\label{transported_coordinates}
    Let $G$ be a coordinate pseudogroup and $X,Y$ be topological spaces.
    Let $S$ be a relative coordinate structure on $X$ under $G$.
    Suppose $h$ is a homeomorphism from $X$ to $Y$.
    The relative coordinate structure $\TransportedCoordinates{G}{X}{Y}{S}{h}=
        \structure{\atlas=\TransportedAtlas{\atlas[S]}{h}}$ on $Y$ under $G$.
\end{construction}
\begin{proof}
    $\atlas[S]$ is a maximal coordinate atlas on $X$ under $G$ by \obv{relative_atlas_maximal}.
    $\TransportedAtlas{\atlas[S]}{h}$ is a maximal coordinate atlas on $Y$ under $G$ by
        \obv{transported_maximal_atlas_is_maximal}.
    Follows by \auto{relative_coordinate_structureaggregateprojection_atlas}.
\end{proof}

\begin{theorem}\label{transported_coordinates_is_exact}
    Let $G$ be a coordinate pseudogroup and $X,Y$ be topological spaces.
    Let $S$ be a relative coordinate structure on $X$ under $G$.
    Suppose $h$ is a homeomorphism from $X$ to $Y$.
    Then $\TransportedCoordinates{G}{X}{Y}{S}{h}$ is an exact relative coordinate structure
        on $Y$ under $G$.
\end{theorem}
\begin{proof}
    Follows by \auto{transported_coordinates,transported_coordinatesshape}.
\end{proof}

\begin{theorem}\label{transported_coordinates_chart_iff}
    Let $G$ be a coordinate pseudogroup and $X,Y$ be topological spaces.
    Let $S$ be a relative coordinate structure on $X$ under $G$.
    Suppose $h$ is a homeomorphism from $X$ to $Y$.
    Then $b\in\atlas[\TransportedCoordinates{G}{X}{Y}{S}{h}]$ iff
        there exists $c\in\atlas[S]$ such that $b=c\circ\converse{h}$.
\end{theorem}
\begin{proof}
    $\atlas[\TransportedCoordinates{G}{X}{Y}{S}{h}]=\TransportedAtlas{\atlas[S]}{h}$ by
        \obv{transported_coordinatesprojection_atlas}.
    Follows by \auto{transported_atlas_iff}.
\end{proof}

\subsection{Transport through presentations}

\begin{theorem}\label{transported_coordinates_presentation}
    Let $G$ be a coordinate pseudogroup and $X,Y$ be topological spaces.
    Let $S$ be a coordinate datum.
    Suppose $A$ presents $S$ on $X$ under $G$.
    Suppose $h$ is a homeomorphism from $X$ to $Y$.
    Then $\TransportedAtlas{A}{h}$ presents $\TransportedCoordinates{G}{X}{Y}{S}{h}$
        on $Y$ under $G$.
\end{theorem}
\begin{proof}
    $S$ is a relative coordinate structure on $X$ under $G$ by \obv{presented_coordinates}.
    $A$ is a coordinate atlas on $X$ under $G$ and $\atlas[S]=\MaxAtlas{G}{X}{A}$ by
        \obv{coordinate_presentation}.
    $\TransportedAtlas{A}{h}$ is a coordinate atlas on $Y$ under $G$ by
        \obv{transported_atlas_is_atlas}.
    $\TransportedCoordinates{G}{X}{Y}{S}{h}$ is a coordinate datum by
        \auto{transported_coordinates,relative_coordinate_structure}.
    $\atlas[\TransportedCoordinates{G}{X}{Y}{S}{h}]=\TransportedAtlas{\atlas[S]}{h}$ by
        \obv{transported_coordinatesprojection_atlas}.
    $\MaxAtlas{G}{Y}{\TransportedAtlas{A}{h}}=\TransportedAtlas{\atlas[S]}{h}$ by
        \auto{transported_atlas_maximalization}.
    Follows by \obv{coordinate_presentation}.
\end{proof}

\begin{theorem}\label{transported_coordinates_from_presentation}
    Let $G$ be a coordinate pseudogroup and $X,Y$ be topological spaces.
    Let $S$ be a coordinate datum.
    Suppose $A$ presents $S$ on $X$ under $G$.
    Suppose $h$ is a homeomorphism from $X$ to $Y$.
    Then $\TransportedCoordinates{G}{X}{Y}{S}{h}
        =\GeneratedCoordinates{G}{Y}{\TransportedAtlas{A}{h}}$.
\end{theorem}
\begin{proof}
    $S$ is a relative coordinate structure on $X$ under $G$ by \obv{presented_coordinates}.
    $\TransportedCoordinates{G}{X}{Y}{S}{h}$ is an exact relative coordinate structure
        on $Y$ under $G$ by \obv{transported_coordinates_is_exact}.
    $\TransportedCoordinates{G}{X}{Y}{S}{h}$ is an exact coordinate datum by
        \auto{transported_coordinatesshape,relative_coordinate_structure}.
    $\TransportedAtlas{A}{h}$ presents $\TransportedCoordinates{G}{X}{Y}{S}{h}$
        on $Y$ under $G$ by \obv{transported_coordinates_presentation}.
    Follows by \obv{exact_presented_coordinates}.
\end{proof}

\subsection{Transport laws}

Identity and inverse transport recover an exact relative datum. If the input
record has additional fields, transport retains its atlas, not those fields.

\begin{theorem}\label{transported_coordinates_identity}
    Let $G$ be a coordinate pseudogroup and $X$ be a topological space.
    Suppose $S$ is an exact relative coordinate structure on $X$ under $G$.
    Then $\TransportedCoordinates{G}{X}{X}{S}{\identity{\carrier[X]}}=S$.
\end{theorem}
\begin{proof}
    $\identity{\carrier[X]}$ is a homeomorphism from $X$ to $X$ by \obv{identity_homeomorphism}.
    $\TransportedCoordinates{G}{X}{X}{S}{\identity{\carrier[X]}}$ is an exact relative coordinate structure
        on $X$ under $G$ by \obv{transported_coordinates_is_exact}.
    $\atlas[S]$ is a coordinate atlas on $X$ under $G$ by
        \auto{relative_atlas_maximal,maximal_coordinate_atlas}.
    Every element of $\atlas[S]$ is a relation by
        \auto{coordinate_atlas,open_partial_homeomorphism_bijection,bijection_is_function}.
    For every $c\in\atlas[S]$ we have $\dom{c}\subseteq\carrier[X]$ by
        \auto{coordinate_atlas,open_partial_homeomorphism,opens_type}.
    $\TransportedAtlas{\atlas[S]}{\identity{\carrier[X]}}=\atlas[S]$ by
        \obv{transported_atlas_identity}.
    $\atlas[\TransportedCoordinates{G}{X}{X}{S}{\identity{\carrier[X]}}]=\atlas[S]$ by
        \auto{transported_coordinatesprojection_atlas}.
    Follows by structure extensionality.
\end{proof}

\begin{theorem}\label{transported_coordinates_composition}
    Let $G$ be a coordinate pseudogroup and $X,Y,Z$ be topological spaces.
    Let $S$ be a relative coordinate structure on $X$ under $G$.
    Suppose $h$ is a homeomorphism from $X$ to $Y$ and $k$ is a homeomorphism from $Y$ to $Z$.
    Then $\TransportedCoordinates{G}{Y}{Z}{\TransportedCoordinates{G}{X}{Y}{S}{h}}{k}
        =\TransportedCoordinates{G}{X}{Z}{S}{k\circ h}$.
\end{theorem}
\begin{proof}
    Let $T=\TransportedCoordinates{G}{X}{Y}{S}{h}$.
    $T$ is a relative coordinate structure on $Y$ under $G$ by \auto{transported_coordinates}.
    $k\circ h$ is a homeomorphism from $X$ to $Z$ by \obv{homeomorphism_composition}.
    $\TransportedCoordinates{G}{Y}{Z}{T}{k}$ is an exact relative coordinate structure
        on $Z$ under $G$ by \obv{transported_coordinates_is_exact}.
    $\TransportedCoordinates{G}{X}{Z}{S}{k\circ h}$ is an exact relative coordinate structure
        on $Z$ under $G$ by \obv{transported_coordinates_is_exact}.
    $\atlas[T]=\TransportedAtlas{\atlas[S]}{h}$ by \auto{transported_coordinatesprojection_atlas}.
    $\atlas[\TransportedCoordinates{G}{Y}{Z}{T}{k}]=\TransportedAtlas{\atlas[T]}{k}$ and
        $\atlas[\TransportedCoordinates{G}{X}{Z}{S}{k\circ h}]=\TransportedAtlas{\atlas[S]}{k\circ h}$ by
        \obv{transported_coordinatesprojection_atlas}.
    $\TransportedAtlas{\atlas[T]}{k}=\TransportedAtlas{\atlas[S]}{k\circ h}$ by
        \auto{transported_atlas_composition}.
    $\atlas[\TransportedCoordinates{G}{Y}{Z}{T}{k}]
        =\atlas[\TransportedCoordinates{G}{X}{Z}{S}{k\circ h}]$.
    Follows by structure extensionality.
\end{proof}

\begin{theorem}\label{transported_coordinates_roundtrip}
    Let $G$ be a coordinate pseudogroup and $X,Y$ be topological spaces.
    Suppose $S$ is an exact relative coordinate structure on $X$ under $G$.
    Suppose $h$ is a homeomorphism from $X$ to $Y$.
    Then $\TransportedCoordinates{G}{Y}{X}{\TransportedCoordinates{G}{X}{Y}{S}{h}}{\converse{h}}=S$.
\end{theorem}
\begin{proof}
    $\converse{h}$ is a homeomorphism from $Y$ to $X$ by \obv{homeomorphism_inverse}.
    $h\in\bijections{\carrier[X]}{\carrier[Y]}$ by \obv{homeomorphism}.
    $\converse{h}\circ h=\identity{\carrier[X]}$ by \obv{bijection_converse_circ}.
    $\TransportedCoordinates{G}{Y}{X}{\TransportedCoordinates{G}{X}{Y}{S}{h}}{\converse{h}}
        =\TransportedCoordinates{G}{X}{X}{S}{\converse{h}\circ h}$ by
        \obv{transported_coordinates_composition}.
    $\TransportedCoordinates{G}{X}{X}{S}{\identity{\carrier[X]}}=S$ by
        \obv{transported_coordinates_identity}.
\end{proof}

\subsection{Open restriction commutes with transport}

The homeomorphism on an open subspace is its restricted graph, with target
the image subspace. Restricting coordinates before or after transport gives
the same exact relative record on that target.

\begin{theorem}\label{transported_coordinates_restriction}
    Let $G$ be a coordinate pseudogroup and $X,Y$ be topological spaces.
    Let $S$ be a relative coordinate structure on $X$ under $G$.
    Suppose $h$ is a homeomorphism from $X$ to $Y$ and $U$ is open in $X$.
    Then $\TransportedCoordinates{G}{\subspaceTopology{X}{U}}{\subspaceTopology{Y}{\img{h}{U}}}
            {\RestrictedCoordinates{G}{X}{S}{U}}{\restrl{h}{U}}
        =\RestrictedCoordinates{G}{Y}{\TransportedCoordinates{G}{X}{Y}{S}{h}}{\img{h}{U}}$.
\end{theorem}
\begin{proof}
    Let $V=\img{h}{U}$.
    Let $P=\subspaceTopology{X}{U}$ and $Q=\subspaceTopology{Y}{V}$.
    $P$ is a topological space and $Q$ is a topological space by \obv{subspace_topology_is_topological_space}.
    $U\in\opens[X]$ and $U\subseteq\carrier[X]$ by \auto{is_open,opens_type}.
    $h$ is an open partial homeomorphism from $X$ to $Y$ by
        \obv{homeomorphism_is_open_partial_homeomorphism}.
    $V\in\opens[Y]$ by \auto{open_partial_homeomorphism_image_open}.
    $V$ is open in $Y$ by \obv{is_open}.
    $h$ is an injection by \auto{homeomorphism,bijections,bijection_is_function}.
    $\restrl{h}{U}$ is a homeomorphism from $P$ to $Q$ by \auto{homeomorphism_restriction_to_image}.
    Let $R=\RestrictedCoordinates{G}{X}{S}{U}$ and $T=\TransportedCoordinates{G}{X}{Y}{S}{h}$.
    $R$ is a relative coordinate structure on $P$ under $G$ by \auto{restricted_coordinates}.
    $T$ is a relative coordinate structure on $Y$ under $G$ by \auto{transported_coordinates}.
    $\TransportedCoordinates{G}{P}{Q}{R}{\restrl{h}{U}}$ is an exact relative coordinate structure
        on $Q$ under $G$ by \obv{transported_coordinates_is_exact}.
    $\RestrictedCoordinates{G}{Y}{T}{V}$ is an exact relative coordinate structure
        on $Q$ under $G$ by \auto{restricted_coordinates_is_exact}.
    $\atlas[S]$ is a coordinate atlas on $X$ under $G$ by
        \auto{relative_atlas_maximal,maximal_coordinate_atlas}.
    Every element of $\atlas[S]$ is a function by
        \auto{coordinate_atlas,open_partial_homeomorphism_bijection,bijection_is_function}.
    $\atlas[R]=\RestrictedAtlas{\atlas[S]}{U}$ and $\atlas[T]=\TransportedAtlas{\atlas[S]}{h}$ by
        \auto{restricted_coordinatesprojection_atlas,transported_coordinatesprojection_atlas}.
    $\atlas[\TransportedCoordinates{G}{P}{Q}{R}{\restrl{h}{U}}]=\TransportedAtlas{\atlas[R]}{\restrl{h}{U}}$ and
        $\atlas[\RestrictedCoordinates{G}{Y}{T}{V}]=\RestrictedAtlas{\atlas[T]}{V}$ by
        \obv{transported_coordinatesprojection_atlas,restricted_coordinatesprojection_atlas}.
    $\TransportedAtlas{\atlas[R]}{\restrl{h}{U}}=\RestrictedAtlas{\atlas[T]}{V}$ by
        \auto{transported_atlas_restriction}.
    $\atlas[\TransportedCoordinates{G}{P}{Q}{R}{\restrl{h}{U}}]=\atlas[\RestrictedCoordinates{G}{Y}{T}{V}]$.
    Follows by structure extensionality.
\end{proof}

\subsection{Equipping and admissibility}

\begin{theorem}\label{equipped_transported_coordinates}
    Let $G$ be a coordinate pseudogroup and $X,Y$ be topological spaces.
    Let $S$ be a relative coordinate structure on $X$ under $G$.
    Suppose $h$ is a homeomorphism from $X$ to $Y$.
    Then $\Equip{Y}{\TransportedCoordinates{G}{X}{Y}{S}{h}}$ has coordinates under $G$ and
        $\TopologicalReduct{\Equip{Y}{\TransportedCoordinates{G}{X}{Y}{S}{h}}}=\TopologicalReduct{Y}$ and
        $\CoordinateStructure{\Equip{Y}{\TransportedCoordinates{G}{X}{Y}{S}{h}}}
            =\TransportedCoordinates{G}{X}{Y}{S}{h}$.
\end{theorem}
\begin{proof}
    Let $T=\TransportedCoordinates{G}{X}{Y}{S}{h}$.
    $T$ is a relative coordinate structure on $Y$ under $G$ by \auto{transported_coordinates}.
    $T$ is an exact coordinate datum by
        \auto{transported_coordinatesshape,relative_coordinate_structure}.
    $\Equip{Y}{T}$ has coordinates under $G$ by \obv{equip_preserves_coordinates}.
    $\TopologicalReduct{\Equip{Y}{T}}=\TopologicalReduct{Y}$ by \obv{equip_topological_reduct}.
    $\CoordinateStructure{\Equip{Y}{T}}=T$ by \obv{coordinate_structure_of_equip}.
\end{proof}

\begin{theorem}\label{equipped_transported_coordinates_presentation}
    Let $G$ be a coordinate pseudogroup and $X,Y$ be topological spaces.
    Let $S$ be a coordinate datum.
    Suppose $A$ presents $S$ on $X$ under $G$.
    Suppose $h$ is a homeomorphism from $X$ to $Y$.
    Then $\TransportedAtlas{A}{h}$ presents $\Equip{Y}{\TransportedCoordinates{G}{X}{Y}{S}{h}}$
        on $\Equip{Y}{\TransportedCoordinates{G}{X}{Y}{S}{h}}$ under $G$.
\end{theorem}
\begin{proof}
    $S$ is a relative coordinate structure on $X$ under $G$ by \obv{presented_coordinates}.
    $\TransportedCoordinates{G}{X}{Y}{S}{h}$ is a coordinate datum by
        \auto{transported_coordinates,relative_coordinate_structure}.
    $\TransportedAtlas{A}{h}$ presents $\TransportedCoordinates{G}{X}{Y}{S}{h}$
        on $Y$ under $G$ by \obv{transported_coordinates_presentation}.
    Follows by \obv{equip_presentation_iff}.
\end{proof}

In a source chart and its transported chart, the homeomorphism is represented
by the identity. Thus compatibility of the pseudogroup with a calculus is
enough to make the transport map admissible in that calculus.

\begin{theorem}\label{transported_coordinates_admissible}
    Let $C$ be a local-map calculus and $G$ be a coordinate pseudogroup.
    Let $X,Y$ be topological spaces.
    Let $S$ be a relative coordinate structure on $X$ under $G$.
    Suppose $G$ is compatible with the calculus $C$.
    Suppose $h$ is a homeomorphism from $X$ to $Y$.
    Then $h$ is coordinate-admissible from $\Equip{X}{S}$ under $G$
        to $\Equip{Y}{\TransportedCoordinates{G}{X}{Y}{S}{h}}$ under $G$ in $C$.
\end{theorem}
\begin{proof}
    Let $T=\TransportedCoordinates{G}{X}{Y}{S}{h}$.
    $T$ is a relative coordinate structure on $Y$ under $G$ by \auto{transported_coordinates}.
    $\atlas[S]$ is a coordinate atlas on $X$ under $G$ by
        \auto{relative_atlas_maximal,maximal_coordinate_atlas}.
    $\model[G]\in\models[C]$ by \auto{compatible_pseudogroup_model,calculus_model}.
    $h\in\bijections{\carrier[X]}{\carrier[Y]}$ by \obv{homeomorphism}.
    $h$ is a function and $\dom{h}=\carrier[X]$ by \obv{bijection_is_function,bijections_dom}.
    $h\in\funs{\dom{h}}{\carrier[Y]}$ by \auto{homeomorphism,continuous_map}.
    $\dom{h}\in\opens[X]$ by \auto{carrier_open}.
    $\converse{h}$ is a relation and $\converse{\converse{h}}=h$ by
        \obv{converse_is_relation,converse_converse_eq}.
    Show that for every $p\in\dom{h}$ there exists $c\in\ChartsAt{S}{p}$ such that
        there exists $d\in\atlas[T]$ such that there exists $U\in\opens[X]$ such that
            $p\in U$ and $U\subseteq\dom{h}$ and $U\subseteq\dom{c}$ and
            $\img{h}{U}\subseteq\dom{d}$ and
            $\CoordExpr{d}{h}{c}{U}$ is admissible from $\model[G]$ to $\model[G]$ in $C$.
    \begin{subproof}
        Fix $p\in\dom{h}$.
        Take $c\in\atlas[S]$ such that $p\in\dom{c}$ by \obv{coordinate_atlas}.
        $c\in\ChartsAt{S}{p}$ by \obv{charts_at_iff}.
        $c$ is an open partial homeomorphism from $X$ to $\model[G]$ by \obv{coordinate_atlas}.
        $c$ is a function by \auto{open_partial_homeomorphism_bijection,bijection_is_function}.
        $\dom{c}\in\opens[X]$ and $\dom{c}\subseteq\carrier[X]$ by
            \auto{open_partial_homeomorphism,opens_type}.
        Let $d=c\circ\converse{h}$.
        $d\in\atlas[T]$ by \auto{transported_coordinates_chart_iff}.
        $d$ is a relation by \auto{circ_is_relation}.
        $\dom{d}=\preimg{\converse{h}}{\dom{c}}$ by \auto{relation_composition_domain}.
        $\preimg{\converse{h}}{\dom{c}}=\img{h}{\dom{c}}$ by \auto{preim_eq_img_of_converse}.
        $d\circ h=c$ by \auto{composition_bijection_roundtrip}.
        $\CoordExpr{d}{h}{c}{\dom{c}}=\identity{\img{c}{\dom{c}}}$ by
            \obv{coordinate_expression_factorization}.
        $\img{c}{\dom{c}}\in\opens[\model[G]]$ by \obv{open_partial_homeomorphism_image_open}.
        $\identity{\img{c}{\dom{c}}}$ is admissible from $\model[G]$ to $\model[G]$ in $C$ by
            \obv{admissible_open_identity}.
        Follows by \auto{subseteq_refl}.
    \end{subproof}
    Follows by \auto{coordinate_admissible_equip_iff}.
\end{proof}

\begin{theorem}\label{transported_coordinates_inverse_admissible}
    Let $C$ be a local-map calculus and $G$ be a coordinate pseudogroup.
    Let $X,Y$ be topological spaces.
    Let $S$ be a relative coordinate structure on $X$ under $G$.
    Suppose $G$ is compatible with the calculus $C$.
    Suppose $h$ is a homeomorphism from $X$ to $Y$.
    Then $\converse{h}$ is coordinate-admissible from
        $\Equip{Y}{\TransportedCoordinates{G}{X}{Y}{S}{h}}$ under $G$
        to $\Equip{X}{S}$ under $G$ in $C$.
\end{theorem}
\begin{proof}
    Let $T=\TransportedCoordinates{G}{X}{Y}{S}{h}$.
    $T$ is a relative coordinate structure on $Y$ under $G$ by \auto{transported_coordinates}.
    $\converse{h}$ is a homeomorphism from $Y$ to $X$ by \obv{homeomorphism_inverse}.
    Let $U=\TransportedCoordinates{G}{Y}{X}{T}{\converse{h}}$.
    $U$ is a relative coordinate structure on $X$ under $G$ by \auto{transported_coordinates}.
    $\converse{h}$ is coordinate-admissible from $\Equip{Y}{T}$ under $G$
        to $\Equip{X}{U}$ under $G$ in $C$ by \auto{transported_coordinates_admissible}.
    $h\in\bijections{\carrier[X]}{\carrier[Y]}$ by \obv{homeomorphism}.
    $\atlas[S]$ is a coordinate atlas on $X$ under $G$ by
        \auto{relative_atlas_maximal,maximal_coordinate_atlas}.
    Every element of $\atlas[S]$ is a relation by
        \auto{coordinate_atlas,open_partial_homeomorphism_bijection,bijection_is_function}.
    For every $c\in\atlas[S]$ we have $\dom{c}\subseteq\carrier[X]$ by
        \auto{coordinate_atlas,open_partial_homeomorphism,opens_type}.
    $\atlas[U]=\TransportedAtlas{\TransportedAtlas{\atlas[S]}{h}}{\converse{h}}$ by
        \auto{transported_coordinatesprojection_atlas}.
    $\atlas[U]=\atlas[S]$ by \auto{transported_atlas_roundtrip}.
    $\Equip{X}{U}=\Equip{X}{S}$ by \obv{equip_same_fields}.
\end{proof}

\subsection{Transport to a bare set}

A bijection onto a bare set first supplies its topology: the initial topology
of the inverse map. Its opens are exactly the images of the original opens.
The preceding construction then transports coordinates without any further
choice of topology or charts.

\begin{theorem}\label{bijection_transported_coordinates_presentation}
    Let $G$ be a coordinate pseudogroup and $X$ be a topological space.
    Let $S$ be a coordinate datum.
    Suppose $A$ presents $S$ on $X$ under $G$.
    Let $h\in\bijections{\carrier[X]}{B}$.
    Then $\TransportedAtlas{A}{h}$ presents
        $\TransportedCoordinates{G}{X}{\SingleInitialTopology{B}{X}{\converse{h}}}{S}{h}$
        on $\SingleInitialTopology{B}{X}{\converse{h}}$ under $G$.
\end{theorem}
\begin{proof}
    $\SingleInitialTopology{B}{X}{\converse{h}}$ is a topological space by
        \obv{single_initial_topology_is_topological_space}.
    $h$ is a homeomorphism from $X$ to $\SingleInitialTopology{B}{X}{\converse{h}}$ by
        \obv{bijection_transported_topology}.
    Follows by \obv{transported_coordinates_presentation}.
\end{proof}

\begin{theorem}\label{bijection_transported_coordinates_admissible}
    Let $C$ be a local-map calculus and $G$ be a coordinate pseudogroup.
    Let $X$ be a topological space.
    Let $S$ be a relative coordinate structure on $X$ under $G$.
    Suppose $G$ is compatible with the calculus $C$.
    Let $h\in\bijections{\carrier[X]}{B}$.
    Then $h$ is coordinate-admissible from $\Equip{X}{S}$ under $G$ to
        $\Equip{\SingleInitialTopology{B}{X}{\converse{h}}}
            {\TransportedCoordinates{G}{X}{\SingleInitialTopology{B}{X}{\converse{h}}}{S}{h}}$
        under $G$ in $C$.
\end{theorem}
\begin{proof}
    $\SingleInitialTopology{B}{X}{\converse{h}}$ is a topological space by
        \obv{single_initial_topology_is_topological_space}.
    $h$ is a homeomorphism from $X$ to $\SingleInitialTopology{B}{X}{\converse{h}}$ by
        \obv{bijection_transported_topology}.
    Follows by \obv{transported_coordinates_admissible}.
\end{proof}
