\section{Associativity of binary products}

\import{topology/product-map.tex}
\import{set/product-associator.tex}

Binary products with different parenthesizations have different literal
carriers.  Their canonical identification is assembled from projections and
pairing maps.  Thus associativity will be a homeomorphism, not an equality of
topological-space records.


\begin{theorem}\label{product_associator_continuous}
    Let $X,Y,Z$ be topological spaces.
    Then $\productAssociator{\carrier[X]}{\carrier[Y]}{\carrier[Z]}$
        is continuous from
        $\BinaryProductTopology{\BinaryProductTopology{X}{Y}}{Z}$ to
        $\BinaryProductTopology{X}{\BinaryProductTopology{Y}{Z}}$.
\end{theorem}
\begin{proof}
    $\BinaryProductTopology{X}{Y}$ is a topological space by
        \obv{binary_product_topology_is_topological_space}.
    $\BinaryProductTopology{Y}{Z}$ is a topological space by
        \obv{binary_product_topology_is_topological_space}.
    $\BinaryProductTopology{\BinaryProductTopology{X}{Y}}{Z}$ is a
        topological space by
        \obv{binary_product_topology_is_topological_space}.
    $\firstProductProjection{
        \carrier[\BinaryProductTopology{X}{Y}]}{\carrier[Z]}$ is
        continuous from
        $\BinaryProductTopology{\BinaryProductTopology{X}{Y}}{Z}$ to
        $\BinaryProductTopology{X}{Y}$ by
        \obv{first_product_projection_continuous}.
    $\firstProductProjection{\carrier[X]}{\carrier[Y]}$ is continuous
        from $\BinaryProductTopology{X}{Y}$ to $X$ and
        $\secondProductProjection{\carrier[X]}{\carrier[Y]}$ is
        continuous from $\BinaryProductTopology{X}{Y}$ to $Y$ by
        \obv{first_product_projection_continuous,
            second_product_projection_continuous}.
    Hence $\firstProductProjection{\carrier[X]}{\carrier[Y]}\circ
        \firstProductProjection{
            \carrier[\BinaryProductTopology{X}{Y}]}{\carrier[Z]}$ is
        continuous from
        $\BinaryProductTopology{\BinaryProductTopology{X}{Y}}{Z}$ to
        $X$ by \obv{continuous_composition}.
    Hence $\secondProductProjection{\carrier[X]}{\carrier[Y]}\circ
        \firstProductProjection{
            \carrier[\BinaryProductTopology{X}{Y}]}{\carrier[Z]}$ is
        continuous from
        $\BinaryProductTopology{\BinaryProductTopology{X}{Y}}{Z}$ to
        $Y$ by \obv{continuous_composition}.
    $\secondProductProjection{
        \carrier[\BinaryProductTopology{X}{Y}]}{\carrier[Z]}$ is
        continuous from
        $\BinaryProductTopology{\BinaryProductTopology{X}{Y}}{Z}$ to
        $Z$ by \obv{second_product_projection_continuous}.
    $\carrier[\BinaryProductTopology{X}{Y}]
        =\carrier[X]\times\carrier[Y]$ by
        \obv{binary_product_topology_carrier}.
    $\carrier[\BinaryProductTopology{
        \BinaryProductTopology{X}{Y}}{Z}]
        =\carrier[\BinaryProductTopology{X}{Y}]\times\carrier[Z]$ by
        \obv{binary_product_topology_carrier}.
    Hence $\carrier[\BinaryProductTopology{
        \BinaryProductTopology{X}{Y}}{Z}]
        =(\carrier[X]\times\carrier[Y])\times\carrier[Z]$.
    Hence $\firstProductProjection{\carrier[X]}{\carrier[Y]}\circ
        \firstProductProjection{
            \carrier[X]\times\carrier[Y]}{\carrier[Z]}$ is continuous
        from $\BinaryProductTopology{
            \BinaryProductTopology{X}{Y}}{Z}$ to $X$.
    Hence $\secondProductProjection{\carrier[X]}{\carrier[Y]}\circ
        \firstProductProjection{
            \carrier[X]\times\carrier[Y]}{\carrier[Z]}$ is continuous
        from $\BinaryProductTopology{
            \BinaryProductTopology{X}{Y}}{Z}$ to $Y$.
    Hence $\secondProductProjection{
        \carrier[X]\times\carrier[Y]}{\carrier[Z]}$ is continuous from
        $\BinaryProductTopology{\BinaryProductTopology{X}{Y}}{Z}$ to
        $Z$.
    Hence $\productPairing{
        \carrier[\BinaryProductTopology{
            \BinaryProductTopology{X}{Y}}{Z}]}{
        \secondProductProjection{\carrier[X]}{\carrier[Y]}\circ
            \firstProductProjection{
                \carrier[X]\times\carrier[Y]}{\carrier[Z]}}{
        \secondProductProjection{
            \carrier[X]\times\carrier[Y]}{\carrier[Z]}}$ is continuous
        from
        $\BinaryProductTopology{\BinaryProductTopology{X}{Y}}{Z}$ to
        $\BinaryProductTopology{Y}{Z}$ by
        \obv{continuous_product_pairing}.
    Hence $\productPairing{
        (\carrier[X]\times\carrier[Y])\times\carrier[Z]}{
        \secondProductProjection{\carrier[X]}{\carrier[Y]}\circ
            \firstProductProjection{
                \carrier[X]\times\carrier[Y]}{\carrier[Z]}}{
        \secondProductProjection{
            \carrier[X]\times\carrier[Y]}{\carrier[Z]}}$ is continuous
        from
        $\BinaryProductTopology{\BinaryProductTopology{X}{Y}}{Z}$ to
        $\BinaryProductTopology{Y}{Z}$.
    $\carrier[\BinaryProductTopology{Y}{Z}]
        =\carrier[Y]\times\carrier[Z]$ by
        \obv{binary_product_topology_carrier}.
    $\firstProductProjection{\carrier[X]}{
        \carrier[Y]\times\carrier[Z]}\circ
        \productAssociator{\carrier[X]}{\carrier[Y]}{\carrier[Z]}
        =\firstProductProjection{\carrier[X]}{\carrier[Y]}\circ
            \firstProductProjection{
                \carrier[X]\times\carrier[Y]}{\carrier[Z]}$ by
        \obv{first_projection_of_product_associator}.
    Hence $\firstProductProjection{\carrier[X]}{
        \carrier[\BinaryProductTopology{Y}{Z}]}\circ
        \productAssociator{\carrier[X]}{\carrier[Y]}{\carrier[Z]}
        =\firstProductProjection{\carrier[X]}{\carrier[Y]}\circ
            \firstProductProjection{
                \carrier[X]\times\carrier[Y]}{\carrier[Z]}$.
    Hence $\firstProductProjection{\carrier[X]}{
        \carrier[\BinaryProductTopology{Y}{Z}]}\circ
        \productAssociator{\carrier[X]}{\carrier[Y]}{\carrier[Z]}$
        is continuous from
        $\BinaryProductTopology{\BinaryProductTopology{X}{Y}}{Z}$ to
        $X$.
    $\secondProductProjection{\carrier[X]}{
        \carrier[Y]\times\carrier[Z]}\circ
        \productAssociator{\carrier[X]}{\carrier[Y]}{\carrier[Z]}
        =\productPairing{
            (\carrier[X]\times\carrier[Y])\times\carrier[Z]}{
            \secondProductProjection{\carrier[X]}{\carrier[Y]}\circ
                \firstProductProjection{
                    \carrier[X]\times\carrier[Y]}{\carrier[Z]}}{
            \secondProductProjection{
                \carrier[X]\times\carrier[Y]}{\carrier[Z]}}$ by
        \obv{second_projection_of_product_associator}.
    Hence $\secondProductProjection{\carrier[X]}{
        \carrier[\BinaryProductTopology{Y}{Z}]}\circ
        \productAssociator{\carrier[X]}{\carrier[Y]}{\carrier[Z]}
        =\productPairing{
            (\carrier[X]\times\carrier[Y])\times\carrier[Z]}{
            \secondProductProjection{\carrier[X]}{\carrier[Y]}\circ
                \firstProductProjection{
                    \carrier[X]\times\carrier[Y]}{\carrier[Z]}}{
            \secondProductProjection{
                \carrier[X]\times\carrier[Y]}{\carrier[Z]}}$.
    Hence $\secondProductProjection{\carrier[X]}{
        \carrier[\BinaryProductTopology{Y}{Z}]}\circ
        \productAssociator{\carrier[X]}{\carrier[Y]}{\carrier[Z]}$
        is continuous from
        $\BinaryProductTopology{\BinaryProductTopology{X}{Y}}{Z}$ to
        $\BinaryProductTopology{Y}{Z}$.
    $\productAssociator{\carrier[X]}{\carrier[Y]}{\carrier[Z]}\in
        \funs{(\carrier[X]\times\carrier[Y])\times\carrier[Z]}{
            \carrier[X]\times(\carrier[Y]\times\carrier[Z])}$ by
        \obv{product_associator_type}.
    Hence $\productAssociator{
        \carrier[X]}{\carrier[Y]}{\carrier[Z]}\in
        \funs{
            \carrier[\BinaryProductTopology{
                \BinaryProductTopology{X}{Y}}{Z}]}{
            \carrier[X]\times
                \carrier[\BinaryProductTopology{Y}{Z}]}$.
    Follows by \auto{binary_product_topology_universal_property}.
\end{proof}

\begin{theorem}\label{inverse_product_associator_continuous}
    Let $X,Y,Z$ be topological spaces.
    Then
        $\inverseProductAssociator{
            \carrier[X]}{\carrier[Y]}{\carrier[Z]}$ is continuous from
        $\BinaryProductTopology{X}{\BinaryProductTopology{Y}{Z}}$ to
        $\BinaryProductTopology{\BinaryProductTopology{X}{Y}}{Z}$.
\end{theorem}
\begin{proof}
    $\BinaryProductTopology{Y}{Z}$ is a topological space by
        \obv{binary_product_topology_is_topological_space}.
    $\BinaryProductTopology{X}{\BinaryProductTopology{Y}{Z}}$ is a
        topological space by
        \obv{binary_product_topology_is_topological_space}.
    $\firstProductProjection{\carrier[X]}{
        \carrier[\BinaryProductTopology{Y}{Z}]}$ is continuous from
        $\BinaryProductTopology{X}{\BinaryProductTopology{Y}{Z}}$ to
        $X$ by \obv{first_product_projection_continuous}.
    $\secondProductProjection{\carrier[X]}{
        \carrier[\BinaryProductTopology{Y}{Z}]}$ is continuous from
        $\BinaryProductTopology{X}{\BinaryProductTopology{Y}{Z}}$ to
        $\BinaryProductTopology{Y}{Z}$ by
        \obv{second_product_projection_continuous}.
    $\firstProductProjection{\carrier[Y]}{\carrier[Z]}$ is continuous
        from $\BinaryProductTopology{Y}{Z}$ to $Y$ and
        $\secondProductProjection{\carrier[Y]}{\carrier[Z]}$ is
        continuous from $\BinaryProductTopology{Y}{Z}$ to $Z$ by
        \obv{first_product_projection_continuous,
            second_product_projection_continuous}.
    Hence $\firstProductProjection{\carrier[Y]}{\carrier[Z]}\circ
        \secondProductProjection{\carrier[X]}{
            \carrier[\BinaryProductTopology{Y}{Z}]}$ is continuous from
        $\BinaryProductTopology{X}{\BinaryProductTopology{Y}{Z}}$ to
        $Y$ by \obv{continuous_composition}.
    Hence $\secondProductProjection{\carrier[Y]}{\carrier[Z]}\circ
        \secondProductProjection{\carrier[X]}{
            \carrier[\BinaryProductTopology{Y}{Z}]}$ is continuous from
        $\BinaryProductTopology{X}{\BinaryProductTopology{Y}{Z}}$ to
        $Z$ by \obv{continuous_composition}.
    Hence $\productPairing{
        \carrier[\BinaryProductTopology{X}{
            \BinaryProductTopology{Y}{Z}}]}{
        \firstProductProjection{\carrier[X]}{
            \carrier[\BinaryProductTopology{Y}{Z}]}}{
        \firstProductProjection{\carrier[Y]}{\carrier[Z]}\circ
            \secondProductProjection{\carrier[X]}{
                \carrier[\BinaryProductTopology{Y}{Z}]}}$ is continuous
        from $\BinaryProductTopology{X}{\BinaryProductTopology{Y}{Z}}$
        to $\BinaryProductTopology{X}{Y}$ by
        \obv{continuous_product_pairing}.
    $\carrier[\BinaryProductTopology{Y}{Z}]
        =\carrier[Y]\times\carrier[Z]$ by
        \obv{binary_product_topology_carrier}.
    $\carrier[\BinaryProductTopology{X}{
        \BinaryProductTopology{Y}{Z}}]
        =\carrier[X]\times
            \carrier[\BinaryProductTopology{Y}{Z}]$ by
        \obv{binary_product_topology_carrier}.
    Hence $\carrier[\BinaryProductTopology{X}{
        \BinaryProductTopology{Y}{Z}}]
        =\carrier[X]\times(\carrier[Y]\times\carrier[Z])$.
    Hence $\productPairing{
        \carrier[X]\times(\carrier[Y]\times\carrier[Z])}{
        \firstProductProjection{\carrier[X]}{
            \carrier[Y]\times\carrier[Z]}}{
        \firstProductProjection{\carrier[Y]}{\carrier[Z]}\circ
            \secondProductProjection{\carrier[X]}{
                \carrier[Y]\times\carrier[Z]}}$ is continuous from
        $\BinaryProductTopology{X}{\BinaryProductTopology{Y}{Z}}$ to
        $\BinaryProductTopology{X}{Y}$.
    Hence $\productPairing{
        \carrier[\BinaryProductTopology{X}{
            \BinaryProductTopology{Y}{Z}}]}{
        \productPairing{
            \carrier[X]\times(\carrier[Y]\times\carrier[Z])}{
            \firstProductProjection{\carrier[X]}{
                \carrier[Y]\times\carrier[Z]}}{
            \firstProductProjection{\carrier[Y]}{\carrier[Z]}\circ
                \secondProductProjection{\carrier[X]}{
                    \carrier[Y]\times\carrier[Z]}}}{
        \secondProductProjection{\carrier[Y]}{\carrier[Z]}\circ
            \secondProductProjection{\carrier[X]}{
                \carrier[\BinaryProductTopology{Y}{Z}]}}$ is continuous
        from $\BinaryProductTopology{X}{\BinaryProductTopology{Y}{Z}}$
        to $\BinaryProductTopology{\BinaryProductTopology{X}{Y}}{Z}$ by
        \obv{continuous_product_pairing}.
    Hence $\productPairing{
        \carrier[X]\times(\carrier[Y]\times\carrier[Z])}{
        \productPairing{
            \carrier[X]\times(\carrier[Y]\times\carrier[Z])}{
            \firstProductProjection{\carrier[X]}{
                \carrier[Y]\times\carrier[Z]}}{
            \firstProductProjection{\carrier[Y]}{\carrier[Z]}\circ
                \secondProductProjection{\carrier[X]}{
                    \carrier[Y]\times\carrier[Z]}}}{
        \secondProductProjection{\carrier[Y]}{\carrier[Z]}\circ
            \secondProductProjection{\carrier[X]}{
                \carrier[Y]\times\carrier[Z]}}$ is continuous from
        $\BinaryProductTopology{X}{\BinaryProductTopology{Y}{Z}}$ to
        $\BinaryProductTopology{\BinaryProductTopology{X}{Y}}{Z}$.
    Follows by \obv{inverse_product_associator}.
\end{proof}

\begin{theorem}\label{product_associator_homeomorphism}
    Let $X,Y,Z$ be topological spaces.
    Then $\productAssociator{\carrier[X]}{\carrier[Y]}{\carrier[Z]}$
        is a homeomorphism from
        $\BinaryProductTopology{\BinaryProductTopology{X}{Y}}{Z}$ to
        $\BinaryProductTopology{X}{\BinaryProductTopology{Y}{Z}}$.
\end{theorem}
\begin{proof}
    $\productAssociator{\carrier[X]}{\carrier[Y]}{\carrier[Z]}$ is
        continuous from
        $\BinaryProductTopology{\BinaryProductTopology{X}{Y}}{Z}$ to
        $\BinaryProductTopology{X}{\BinaryProductTopology{Y}{Z}}$ by
        \obv{product_associator_continuous}.
    $\productAssociator{\carrier[X]}{\carrier[Y]}{\carrier[Z]}$ is a
        bijection from
        $(\carrier[X]\times\carrier[Y])\times\carrier[Z]$ to
        $\carrier[X]\times(\carrier[Y]\times\carrier[Z])$ by
        \obv{product_associator_bijection}.
    $\carrier[\BinaryProductTopology{X}{Y}]
        =\carrier[X]\times\carrier[Y]$ by
        \obv{binary_product_topology_carrier}.
    $\carrier[\BinaryProductTopology{\BinaryProductTopology{X}{Y}}{Z}]
        =\carrier[\BinaryProductTopology{X}{Y}]\times\carrier[Z]$ by
        \obv{binary_product_topology_carrier}.
    Hence $\carrier[\BinaryProductTopology{
        \BinaryProductTopology{X}{Y}}{Z}]
        =(\carrier[X]\times\carrier[Y])\times\carrier[Z]$.
    $\carrier[\BinaryProductTopology{Y}{Z}]
        =\carrier[Y]\times\carrier[Z]$ by
        \obv{binary_product_topology_carrier}.
    $\carrier[\BinaryProductTopology{X}{
        \BinaryProductTopology{Y}{Z}}]
        =\carrier[X]\times
            \carrier[\BinaryProductTopology{Y}{Z}]$ by
        \obv{binary_product_topology_carrier}.
    Hence $\carrier[\BinaryProductTopology{X}{
        \BinaryProductTopology{Y}{Z}}]
        =\carrier[X]\times(\carrier[Y]\times\carrier[Z])$.
    Hence $\productAssociator{
        \carrier[X]}{\carrier[Y]}{\carrier[Z]}$ is a bijection from
        $\carrier[\BinaryProductTopology{
            \BinaryProductTopology{X}{Y}}{Z}]$ to
        $\carrier[\BinaryProductTopology{X}{
            \BinaryProductTopology{Y}{Z}}]$.
    $\converse{\productAssociator{
        \carrier[X]}{\carrier[Y]}{\carrier[Z]}}
        =\inverseProductAssociator{
            \carrier[X]}{\carrier[Y]}{\carrier[Z]}$ by
        \obv{product_associator_converse}.
    $\inverseProductAssociator{
        \carrier[X]}{\carrier[Y]}{\carrier[Z]}$ is continuous from
        $\BinaryProductTopology{X}{\BinaryProductTopology{Y}{Z}}$ to
        $\BinaryProductTopology{\BinaryProductTopology{X}{Y}}{Z}$ by
        \obv{inverse_product_associator_continuous}.
    Hence $\converse{\productAssociator{
        \carrier[X]}{\carrier[Y]}{\carrier[Z]}}$ is continuous from
        $\BinaryProductTopology{X}{\BinaryProductTopology{Y}{Z}}$ to
        $\BinaryProductTopology{\BinaryProductTopology{X}{Y}}{Z}$.
    Follows by \obv{homeomorphism}.
\end{proof}
