Intersecting a chart domain with an open subset restricts the chart to an
open part of the same Euclidean model. This construction includes the empty
open subset and does not change dimension.

\begin{theorem}\label{open_subspace_is_locally_euclidean}
    Let $n$ be a natural number and $X$ be a topological space.
    Suppose $X$ is locally Euclidean of dimension $n$ and $A\in\opens[X]$.
    Then $\subspaceTopology{X}{A}$ is locally Euclidean of dimension $n$.
\end{theorem}
\begin{proof}
    Let $S=\subspaceTopology{X}{A}$ and $E=\RealCoordinateSpace{n}$.
    $A\subseteq\carrier[X]$ by \auto{opens_type,subseteq}.
    $S$ is a topological space and $\carrier[S]=A$ by
        \auto{subspace_topology_is_topological_space,subspace_topology_carrier_of_subset}.
    $E$ is a topological space by \obv{real_coordinate_space_is_topological_space}.
    Show that for every $x\in\carrier[S]$ there exists $W\in\opens[S]$ such that
        $x\in W$ and there exists $B\in\opens[E]$ such that there exists $k$ such that
        $k$ is a homeomorphism from $\subspaceTopology{S}{W}$ to $\subspaceTopology{E}{B}$.
    \begin{subproof}
        Fix $x\in\carrier[S]$.
        $x\in\carrier[X]$ by \obv{subseteq}.
        Take $U,V,h$ such that $U\in\opens[X]$ and $x\in U$ and $V\in\opens[E]$ and
            $h$ is a homeomorphism from $\subspaceTopology{X}{U}$ to $\subspaceTopology{E}{V}$ by
            \auto{locally_euclidean}.
        Let $W=U\inter A$.
        $x\in W$ by \obv{inter}.
        $W\in\opens[X]$ by \auto{opens_inter}.
        $W\subseteq U$ and $W\subseteq A$ by \auto{inter,subseteq}.
        $W\in\opens[S]$ and $W\in\opens[\subspaceTopology{X}{U}]$ by
            \auto{open_subspace_open_iff}.
        $U\subseteq\carrier[X]$ and $V\subseteq\carrier[E]$ by \auto{opens_type,subseteq}.
        $\carrier[\subspaceTopology{X}{U}]=U$ by \obv{subspace_topology_carrier_of_subset}.
        $h$ is an open map from $\subspaceTopology{X}{U}$ to $\subspaceTopology{E}{V}$ by
            \obv{homeomorphism_is_open_map}.
        $\img{h}{W}\in\opens[\subspaceTopology{E}{V}]$ by \obv{open_map}.
        Hence $\img{h}{W}\in\opens[E]$ and $\img{h}{W}\subseteq V$ by
            \auto{open_subspace_open_iff}.
        $\restrl{h}{W}$ is a homeomorphism from
            $\subspaceTopology{\subspaceTopology{X}{U}}{W}$ to
            $\subspaceTopology{\subspaceTopology{E}{V}}{\img{h}{W}}$ by
            \obv{homeomorphism_restriction_to_image}.
        $\subspaceTopology{\subspaceTopology{X}{U}}{W}=\subspaceTopology{X}{W}$ and
            $\subspaceTopology{S}{W}=\subspaceTopology{X}{W}$ by
            \auto{nested_subspace_topology}.
        $\subspaceTopology{\subspaceTopology{E}{V}}{\img{h}{W}}
            =\subspaceTopology{E}{\img{h}{W}}$ by \obv{nested_subspace_topology}.
        Follows by \auto{homeomorphism}.
    \end{subproof}
    Follows by \auto{locally_euclidean}.
\end{proof}

\begin{theorem}\label{locally_euclidean_from_open_neighbourhoods}
    Let $n$ be a natural number and $X$ be a topological space.
    Suppose for every $x\in\carrier[X]$ there exists $A\in\opens[X]$ such that
        $x\in A$ and $\subspaceTopology{X}{A}$ is locally Euclidean of dimension $n$.
    Then $X$ is locally Euclidean of dimension $n$.
\end{theorem}
\begin{proof}
    Show that for every $x\in\carrier[X]$ there exists $U\in\opens[X]$ such that
        $x\in U$ and there exists $V\in\opens[\RealCoordinateSpace{n}]$ such that
        there exists $h$ such that $h$ is a homeomorphism from $\subspaceTopology{X}{U}$ to
            $\subspaceTopology{\RealCoordinateSpace{n}}{V}$.
    \begin{subproof}
        Fix $x\in\carrier[X]$.
        Take $A\in\opens[X]$ such that $x\in A$ and
            $\subspaceTopology{X}{A}$ is locally Euclidean of dimension $n$.
        Let $S=\subspaceTopology{X}{A}$.
        $A\subseteq\carrier[X]$ by \auto{opens_type,subseteq}.
        $\carrier[S]=A$ by \auto{subspace_topology_carrier_of_subset}.
        Take $U,V,h$ such that $U\in\opens[S]$ and $x\in U$ and
            $V\in\opens[\RealCoordinateSpace{n}]$ and $h$ is a homeomorphism from
            $\subspaceTopology{S}{U}$ to $\subspaceTopology{\RealCoordinateSpace{n}}{V}$ by
            \auto{locally_euclidean}.
        $U\in\opens[X]$ and $U\subseteq A$ by \auto{open_subspace_open_iff}.
        $\subspaceTopology{S}{U}=\subspaceTopology{X}{U}$ by \auto{nested_subspace_topology}.
        Follows by \auto.
    \end{subproof}
    Follows by \obv{locally_euclidean}.
\end{proof}

\begin{theorem}\label{open_subspace_is_topological_manifold}
    Let $n$ be a natural number and $X$ be a topological space.
    Suppose $X$ is a topological manifold of dimension $n$ and $A\in\opens[X]$.
    Then $\subspaceTopology{X}{A}$ is a topological manifold of dimension $n$.
\end{theorem}
\begin{proof}
    $X$ is locally Euclidean of dimension $n$ and $X$ is Hausdorff and $X$ is second-countable by
        \obv{topological_manifold}.
    $\subspaceTopology{X}{A}$ is locally Euclidean of dimension $n$ by
        \obv{open_subspace_is_locally_euclidean}.
    $\subspaceTopology{X}{A}$ is Hausdorff by \obv{subspace_of_hausdorff_space}.
    $\subspaceTopology{X}{A}$ is second-countable by \obv{subspace_of_second_countable_space}.
    Follows by \obv{topological_manifold,subspace_topology_is_topological_space}.
\end{proof}
