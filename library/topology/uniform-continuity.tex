\section{Uniformly continuous maps}

\import{topology/uniform.tex}
\import{topology/product-map.tex}
\import{set/filter-tendsto.tex}

A map of uniform spaces is uniformly continuous when its product with itself
carries every target entourage back to a source entourage.  At the
foundational level this is simply tendency between the two uniformity
filters.

\begin{definition}\label{uniformly_continuous_map}
    $f$ is uniformly continuous from $X$ to $Y$ iff
        $X$ is a uniform space and
        $Y$ is a uniform space and
        $f\in\funs{\carrier[X]}{\carrier[Y]}$ and
        $\productMap{\carrier[X]}{\carrier[X]}{f}{f}$ tends from
            $\uniformity[X]$ on
            $\carrier[X]\times\carrier[X]$ to
            $\uniformity[Y]$ on
            $\carrier[Y]\times\carrier[Y]$.
\end{definition}

\begin{theorem}\label{uniformly_continuous_iff_entourages}
    Let $X,Y$ be uniform spaces.
    Let $f\in\funs{\carrier[X]}{\carrier[Y]}$.
    Then $f$ is uniformly continuous from $X$ to $Y$ iff
        for every $V\in\uniformity[Y]$ we have
        $\preimg{
            \productMap{\carrier[X]}{\carrier[X]}{f}{f}}{V}
            \in\uniformity[X]$.
\end{theorem}
\begin{proof}
    $\productMap{\carrier[X]}{\carrier[X]}{f}{f}\in\funs{
        \carrier[X]\times\carrier[X]}{
        \carrier[Y]\times\carrier[Y]}$ by \obv{product_map_type}.
    $\uniformity[X]$ is a filter on
        $\carrier[X]\times\carrier[X]$ and
        $\uniformity[Y]$ is a filter on
        $\carrier[Y]\times\carrier[Y]$ by \obv{uniformity_filter}.
    Hence $\productMap{\carrier[X]}{\carrier[X]}{f}{f}$ tends from
            $\uniformity[X]$ on
            $\carrier[X]\times\carrier[X]$ to
            $\uniformity[Y]$ on
            $\carrier[Y]\times\carrier[Y]$ iff
        for every $V\in\uniformity[Y]$ we have
            $\preimg{
                \productMap{\carrier[X]}{\carrier[X]}{f}{f}}{V}
                \in\uniformity[X]$ by
        \obv{filter_tendsto_iff_preimage}.
    Follows by \auto{uniformly_continuous_map}.
\end{proof}

\begin{theorem}\label{identity_is_uniformly_continuous}
    Let $X$ be a uniform space.
    Then $\identity{\carrier[X]}$ is uniformly continuous from $X$ to $X$.
\end{theorem}
\begin{proof}
    $\identity{\carrier[X]}\in
        \funs{\carrier[X]}{\carrier[X]}$ by \obv{id_elem_funs}.
    $\uniformity[X]$ is a filter on
        $\carrier[X]\times\carrier[X]$ by \obv{uniformity_filter}.
    $\identity{\carrier[X]\times\carrier[X]}$ tends from
        $\uniformity[X]$ on $\carrier[X]\times\carrier[X]$ to
        $\uniformity[X]$ on $\carrier[X]\times\carrier[X]$ by
        \obv{identity_filter_tendsto}.
    $\productMap{\carrier[X]}{\carrier[X]}{
        \identity{\carrier[X]}}{\identity{\carrier[X]}}
        =\identity{\carrier[X]\times\carrier[X]}$ by
        \obv{identity_product_map}.
    Follows by \obv{uniformly_continuous_map}.
\end{proof}

\begin{theorem}\label{uniformly_continuous_composition}
    Suppose $f$ is uniformly continuous from $X$ to $Y$ and
        $g$ is uniformly continuous from $Y$ to $Z$.
    Then $g\circ f$ is uniformly continuous from $X$ to $Z$.
\end{theorem}
\begin{proof}
    $X$ is a uniform space and $Y$ is a uniform space and
        $Z$ is a uniform space and
        $f\in\funs{\carrier[X]}{\carrier[Y]}$ and
        $g\in\funs{\carrier[Y]}{\carrier[Z]}$ by
        \obv{uniformly_continuous_map}.
    Hence $g\circ f\in\funs{\carrier[X]}{\carrier[Z]}$ by
        \obv{funs_circ}.
    $\productMap{\carrier[X]}{\carrier[X]}{f}{f}\in\funs{
        \carrier[X]\times\carrier[X]}{
        \carrier[Y]\times\carrier[Y]}$ by \obv{product_map_type}.
    $\productMap{\carrier[Y]}{\carrier[Y]}{g}{g}\in\funs{
        \carrier[Y]\times\carrier[Y]}{
        \carrier[Z]\times\carrier[Z]}$ by \obv{product_map_type}.
    $\uniformity[X]$ is a filter on
        $\carrier[X]\times\carrier[X]$ and
        $\uniformity[Y]$ is a filter on
        $\carrier[Y]\times\carrier[Y]$ and
        $\uniformity[Z]$ is a filter on
        $\carrier[Z]\times\carrier[Z]$ by \obv{uniformity_filter}.
    $\productMap{\carrier[X]}{\carrier[X]}{f}{f}$ tends from
        $\uniformity[X]$ on
        $\carrier[X]\times\carrier[X]$ to
        $\uniformity[Y]$ on
        $\carrier[Y]\times\carrier[Y]$ by
        \obv{uniformly_continuous_map}.
    $\productMap{\carrier[Y]}{\carrier[Y]}{g}{g}$ tends from
        $\uniformity[Y]$ on
        $\carrier[Y]\times\carrier[Y]$ to
        $\uniformity[Z]$ on
        $\carrier[Z]\times\carrier[Z]$ by
        \obv{uniformly_continuous_map}.
    Hence $\productMap{\carrier[Y]}{\carrier[Y]}{g}{g}\circ
        \productMap{\carrier[X]}{\carrier[X]}{f}{f}$ tends from
        $\uniformity[X]$ on
        $\carrier[X]\times\carrier[X]$ to
        $\uniformity[Z]$ on
        $\carrier[Z]\times\carrier[Z]$ by
        \obv{composition_filter_tendsto}.
    $\productMap{\carrier[Y]}{\carrier[Y]}{g}{g}\circ
        \productMap{\carrier[X]}{\carrier[X]}{f}{f}=
        \productMap{\carrier[X]}{\carrier[X]}{g\circ f}{g\circ f}$
        by \obv{composition_product_map}.
    Hence $\productMap{\carrier[X]}{\carrier[X]}{g\circ f}{g\circ f}$
        tends from $\uniformity[X]$ on
        $\carrier[X]\times\carrier[X]$ to
        $\uniformity[Z]$ on
        $\carrier[Z]\times\carrier[Z]$.
    Follows by \obv{uniformly_continuous_map}.
\end{proof}
