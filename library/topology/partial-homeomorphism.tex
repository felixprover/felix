\section{Open partial homeomorphisms}

\import{topology/embedding.tex}
\import{topology/partial-continuity.tex}

An open partial homeomorphism identifies an open part of its source with an
open part of its target. It is an ordinary function graph, so converse and
composition have their usual set-theoretic meanings and exact natural domains.

\begin{definition}\label{open_partial_homeomorphism}
    $h$ is an open partial homeomorphism from $X$ to $Y$ iff
        $X$ is a topological space and $Y$ is a topological space and
        $\dom{h}\in\opens[X]$ and $\ran{h}\in\opens[Y]$ and
        $h$ is a homeomorphism from $\subspaceTopology{X}{\dom{h}}$
            to $\subspaceTopology{Y}{\ran{h}}$.
\end{definition}

\begin{proposition}\label{open_partial_homeomorphism_bijection}
    Suppose $h$ is an open partial homeomorphism from $X$ to $Y$.
    Then $h\in\bijections{\dom{h}}{\ran{h}}$.
\end{proposition}
\begin{proof}
    $X$ is a topological space and $Y$ is a topological space and
        $\dom{h}\in\opens[X]$ and $\ran{h}\in\opens[Y]$ by
        \obv{open_partial_homeomorphism}.
    $\dom{h}\subseteq\carrier[X]$ and $\ran{h}\subseteq\carrier[Y]$ by
        \obv{opens_type}.
    $\carrier[\subspaceTopology{X}{\dom{h}}]=\dom{h}$ and
        $\carrier[\subspaceTopology{Y}{\ran{h}}]=\ran{h}$ by
        \obv{subspace_topology_carrier_of_subset}.
    $h$ is a homeomorphism from $\subspaceTopology{X}{\dom{h}}$
        to $\subspaceTopology{Y}{\ran{h}}$ by
        \obv{open_partial_homeomorphism}.
    Follows by \obv{homeomorphism}.
\end{proof}

\begin{theorem}\label{open_partial_homeomorphism_is_continuous_partial_map}
    Suppose $h$ is an open partial homeomorphism from $X$ to $Y$.
    Then $h$ is a continuous partial map from $X$ to $Y$.
\end{theorem}
\begin{proof}
    $X$ is a topological space and $Y$ is a topological space and
        $\dom{h}\in\opens[X]$ and
        $h$ is a homeomorphism from $\subspaceTopology{X}{\dom{h}}$ to
            $\subspaceTopology{Y}{\ran{h}}$ by \obv{open_partial_homeomorphism}.
    $h$ is continuous from $\subspaceTopology{X}{\dom{h}}$ to
        $\subspaceTopology{Y}{\ran{h}}$ by \obv{homeomorphism}.
    $h$ is continuous from $\subspaceTopology{X}{\dom{h}}$ to $Y$ by
        \obv{continuous_subspace_codomain_inclusion}.
    Follows by \obv{continuous_partial_map}.
\end{proof}

\subsection{Converse and open sets}

\begin{theorem}\label{open_partial_homeomorphism_converse}
    Suppose $h$ is an open partial homeomorphism from $X$ to $Y$.
    Then $\converse{h}$ is an open partial homeomorphism from $Y$ to $X$.
\end{theorem}
\begin{proof}
    $X$ is a topological space and $Y$ is a topological space and
        $\dom{h}\in\opens[X]$ and $\ran{h}\in\opens[Y]$ and
        $h$ is a homeomorphism from $\subspaceTopology{X}{\dom{h}}$
            to $\subspaceTopology{Y}{\ran{h}}$ by
        \obv{open_partial_homeomorphism}.
    $\converse{h}$ is a homeomorphism from $\subspaceTopology{Y}{\ran{h}}$
        to $\subspaceTopology{X}{\dom{h}}$ by \obv{homeomorphism_inverse}.
    $\dom{\converse{h}}=\ran{h}$ and $\ran{\converse{h}}=\dom{h}$ by
        \obv{dom_converse,ran_converse}.
    Follows by \obv{open_partial_homeomorphism}.
\end{proof}

\begin{theorem}\label{open_partial_homeomorphism_preimage_open}
    Suppose $h$ is an open partial homeomorphism from $X$ to $Y$.
    Suppose $V\in\opens[Y]$.
    Then $\preimg{h}{V}\in\opens[X]$.
\end{theorem}
\begin{proof}
    $h$ is a continuous partial map from $X$ to $Y$ by
        \obv{open_partial_homeomorphism_is_continuous_partial_map}.
    Follows by \obv{continuous_partial_map_preimage_open}.
\end{proof}

\begin{theorem}\label{open_partial_homeomorphism_image_open}
    Suppose $h$ is an open partial homeomorphism from $X$ to $Y$.
    Suppose $U\in\opens[X]$.
    Then $\img{h}{U}\in\opens[Y]$.
\end{theorem}
\begin{proof}
    $\converse{h}$ is an open partial homeomorphism from $Y$ to $X$ by
        \obv{open_partial_homeomorphism_converse}.
    $\preimg{\converse{h}}{U}\in\opens[Y]$ by
        \obv{open_partial_homeomorphism_preimage_open}.
    $h\in\bijections{\dom{h}}{\ran{h}}$ by
        \obv{open_partial_homeomorphism_bijection}.
    $h$ is a function by \obv{bijection_is_function}.
    $\converse{\converse{h}}=h$ by \obv{converse_converse_eq}.
    $\preimg{\converse{h}}{U}=\img{\converse{\converse{h}}}{U}$ by
        \obv{preim_eq_img_of_converse}.
\end{proof}

\subsection{Restriction and composition}

\begin{theorem}\label{open_partial_homeomorphism_restriction_on_subset}
    Suppose $h$ is an open partial homeomorphism from $X$ to $Y$.
    Suppose $U\in\opens[X]$ and $U\subseteq\dom{h}$.
    Then $\restrl{h}{U}$ is an open partial homeomorphism from $X$ to $Y$.
\end{theorem}
\begin{proof}
    Let $k=\restrl{h}{U}$.
    Let $V=\img{h}{U}$.
    $X$ is a topological space and $Y$ is a topological space and
        $\dom{h}\in\opens[X]$ and $\ran{h}\in\opens[Y]$ and
        $h$ is a homeomorphism from $\subspaceTopology{X}{\dom{h}}$
            to $\subspaceTopology{Y}{\ran{h}}$ by
        \obv{open_partial_homeomorphism}.
    $\dom{h}\subseteq\carrier[X]$ and $\ran{h}\subseteq\carrier[Y]$ by
        \obv{opens_type}.
    $\carrier[\subspaceTopology{X}{\dom{h}}]=\dom{h}$ by
        \obv{subspace_topology_carrier_of_subset}.
    $h\in\bijections{\dom{h}}{\ran{h}}$ by
        \obv{open_partial_homeomorphism_bijection}.
    $h$ is a function by \obv{bijection_is_function}.
    $k\in\bijections{U}{V}$ by \auto{bijection_restriction_to_image}.
    $\dom{k}=U$ and $\ran{k}=V$ by
        \auto{bijections_dom,restriction_range}.
    $V\in\opens[Y]$ by \auto{open_partial_homeomorphism_image_open}.
    $V\subseteq\ran{h}$ by \auto{img_subseteq_ran}.
    $k$ is a homeomorphism from
        $\subspaceTopology{\subspaceTopology{X}{\dom{h}}}{U}$ to
        $\subspaceTopology{\subspaceTopology{Y}{\ran{h}}}{V}$ by
        \auto{homeomorphism_restriction_to_image}.
    $\subspaceTopology{\subspaceTopology{X}{\dom{h}}}{U}
        =\subspaceTopology{X}{U}$ and
        $\subspaceTopology{\subspaceTopology{Y}{\ran{h}}}{V}
        =\subspaceTopology{Y}{V}$ by \obv{nested_subspace_topology}.
    Follows by \obv{open_partial_homeomorphism}.
\end{proof}

\begin{theorem}\label{open_partial_homeomorphism_restriction}
    Suppose $h$ is an open partial homeomorphism from $X$ to $Y$.
    Suppose $U\in\opens[X]$.
    Then $\restrl{h}{U}$ is an open partial homeomorphism from $X$ to $Y$.
\end{theorem}
\begin{proof}
    $X$ is a topological space and $\dom{h}\in\opens[X]$ by
        \obv{open_partial_homeomorphism}.
    $\dom{h}\inter U\in\opens[X]$ by \obv{opens_inter}.
    $\dom{h}\inter U\subseteq\dom{h}$ by \obv{inter_lower_left}.
    Hence $\restrl{h}{\dom{h}\inter U}$ is an open partial homeomorphism
        from $X$ to $Y$ by
        \obv{open_partial_homeomorphism_restriction_on_subset}.
    $h\in\bijections{\dom{h}}{\ran{h}}$ by
        \obv{open_partial_homeomorphism_bijection}.
    $h$ is a function by \obv{bijection_is_function}.
    $\restrl{h}{\dom{h}}=h$ by \obv{restrl_by_dom}.
    $\restrl{\restrl{h}{\dom{h}}}{U}
        =\restrl{h}{\dom{h}\inter U}$ by \obv{restriction_intersection}.
\end{proof}

\begin{theorem}\label{open_partial_homeomorphism_composition}
    Suppose $f$ is an open partial homeomorphism from $X$ to $Y$ and
        $g$ is an open partial homeomorphism from $Y$ to $Z$.
    Then $g\circ f$ is an open partial homeomorphism from $X$ to $Z$.
\end{theorem}
\begin{proof}
    $X$ is a topological space and $Z$ is a topological space and
        $\dom{g}\in\opens[Y]$ by \obv{open_partial_homeomorphism}.
    $f\in\bijections{\dom{f}}{\ran{f}}$ and
        $g\in\bijections{\dom{g}}{\ran{g}}$ by
        \obv{open_partial_homeomorphism_bijection}.
    $f$ and $g$ are functions by \obv{bijection_is_function}.
    $g\circ f$ is a function by \obv{function_circ}.
    Let $D=\preimg{f}{\dom{g}}$.
    Let $k=\restrl{f}{D}$.
    $D\in\opens[X]$ by \auto{open_partial_homeomorphism_preimage_open}.
    $D\subseteq\dom{f}$ by \auto{preimg_subseteq_dom}.
    $k$ is an open partial homeomorphism from $X$ to $Y$ by
        \auto{open_partial_homeomorphism_restriction}.
    $\dom{k}=D$ by
        \auto{dom_of_restrl_eq_inter,inter_absorb_supseteq_left}.
    $\ran{k}=\img{f}{D}$ by \auto{restriction_range}.
    $\img{f}{D}=\ran{f}\inter\dom{g}$ by
        \auto{image_preimage_range_trace}.
    Hence $\ran{k}\subseteq\dom{g}$ by \auto{inter_lower_right}.

    Let $l=\restrl{g}{\ran{k}}$.
    $\ran{k}\in\opens[Y]$ by \obv{open_partial_homeomorphism}.
    $l$ is an open partial homeomorphism from $Y$ to $Z$ by
        \auto{open_partial_homeomorphism_restriction}.
    $\dom{l}=\ran{k}$ by
        \auto{dom_of_restrl_eq_inter,inter_absorb_supseteq_left}.
    $k$ is a homeomorphism from $\subspaceTopology{X}{D}$ to
        $\subspaceTopology{Y}{\ran{k}}$ and
        $l$ is a homeomorphism from $\subspaceTopology{Y}{\ran{k}}$ to
        $\subspaceTopology{Z}{\ran{l}}$ by
        \auto{open_partial_homeomorphism}.
    Hence $l\circ k$ is a homeomorphism from $\subspaceTopology{X}{D}$
        to $\subspaceTopology{Z}{\ran{l}}$ by
        \obv{homeomorphism_composition}.

    $\dom{g\circ f}=D$ by \auto{relation_composition_domain}.
    $\restrl{g\circ f}{D}=g\circ f$ by \auto{restrl_by_dom}.
    $\restrl{g\circ f}{D}=g\circ k$ by \auto{restrl_circ}.
    $k$ is a function by \auto{restrl_of_function_is_function}.
    $l\circ k=g\circ k$ by \auto{composition_restrict_intermediate_range}.
    $\ran{l\circ k}=\img{l}{\ran{k}}$ by \obv{relation_composition_range}.
    $\img{l}{\ran{k}}=\ran{l}$ by \auto{img_dom}.
    $\ran{l}\in\opens[Z]$ by \obv{open_partial_homeomorphism}.
    Follows by \obv{open_partial_homeomorphism}.
\end{proof}

\subsection{Identities and empty domains}

\begin{theorem}\label{identity_open_partial_homeomorphism}
    Let $X$ be a topological space.
    Suppose $U\in\opens[X]$.
    Then $\identity{U}$ is an open partial homeomorphism from $X$ to $X$.
\end{theorem}
\begin{proof}
    Let $S=\subspaceTopology{X}{U}$.
    $S$ is a topological space by
        \auto{subspace_topology_is_topological_space}.
    $U\subseteq\carrier[X]$ by \obv{opens_type}.
    $\carrier[S]=U$ by \auto{subspace_topology_carrier_of_subset}.
    $\identity{U}$ is continuous from $S$ to $S$ by
        \auto{identity_continuous}.
    $\identity{U}\in\bijections{U}{U}$ by \obv{id_elem_bijections}.
    $\converse{\identity{U}}=\identity{U}$ by \obv{converse_identity}.
    $\identity{U}$ is a homeomorphism from $S$ to $S$ by \obv{homeomorphism}.
    $\dom{\identity{U}}=U$ and $\ran{\identity{U}}=U$ by \obv{id_dom,id_ran}.
    Follows by \obv{open_partial_homeomorphism}.
\end{proof}

\begin{theorem}\label{empty_open_partial_homeomorphism}
    Let $X,Y$ be topological spaces.
    Then $\emptyset$ is an open partial homeomorphism from $X$ to $Y$.
\end{theorem}
\begin{proof}
    Let $S=\subspaceTopology{X}{\emptyset}$.
    Let $T=\subspaceTopology{Y}{\emptyset}$.
    $S$ is a topological space and $T$ is a topological space by
        \auto{subspace_topology_is_topological_space}.
    $\carrier[S]=\emptyset$ and $\carrier[T]=\emptyset$ by
        \auto{subspace_topology_carrier_of_subset,emptyset_subseteq}.
    $\emptyset$ is continuous from $S$ to $T$ and
        $\emptyset$ is continuous from $T$ to $S$ by
        \obv{empty_function_continuous}.
    $\emptyset$ is a function and $\emptyset$ is injective by
        \obv{emptyset_is_function,emptyset_is_injective}.
    $\dom{\emptyset}=\emptyset$ and $\ran{\emptyset}=\emptyset$ by
        \obv{dom_emptyset,ran_emptyset}.
    $\emptyset\in\bijections{\carrier[S]}{\carrier[T]}$ by
        \auto{injective_function_bijection_to_range}.
    $\converse{\emptyset}=\emptyset$ by \obv{converse_emptyset}.
    $\emptyset$ is a homeomorphism from $S$ to $T$ by \obv{homeomorphism}.
    $\emptyset\in\opens[X]$ and $\emptyset\in\opens[Y]$ by
        \auto{emptyset_open,is_open}.
    Follows by \obv{open_partial_homeomorphism}.
\end{proof}

\begin{theorem}\label{open_partial_homeomorphism_empty_domain}
    Suppose $h$ is an open partial homeomorphism from $X$ to $Y$.
    Suppose $\dom{h}=\emptyset$.
    Then $h=\emptyset$.
\end{theorem}
\begin{proof}
    $h\in\bijections{\dom{h}}{\ran{h}}$ by
        \obv{open_partial_homeomorphism_bijection}.
    $h$ is a function by \obv{bijection_is_function}.
    Follows by \auto{setext,relation,dom_iff,emptyset}.
\end{proof}

\begin{corollary}\label{disjoint_open_identity_composition}
    Let $X$ be a topological space.
    Suppose $U,V\in\opens[X]$ and $U\inter V=\emptyset$.
    Then $\identity{V}\circ\identity{U}=\emptyset$.
\end{corollary}
\begin{proof}
    $\identity{U}$ is an open partial homeomorphism from $X$ to $X$ and
        $\identity{V}$ is an open partial homeomorphism from $X$ to $X$ by
        \obv{identity_open_partial_homeomorphism}.
    Hence $\identity{V}\circ\identity{U}$ is an open partial homeomorphism
        from $X$ to $X$ by \obv{open_partial_homeomorphism_composition}.
    $\dom{\identity{V}\circ\identity{U}}=\emptyset$ by
        \obv{disjoint_identity_composition_domain}.
    Follows by \obv{open_partial_homeomorphism_empty_domain}.
\end{proof}

\begin{theorem}\label{homeomorphism_is_open_partial_homeomorphism}
    Suppose $f$ is a homeomorphism from $X$ to $Y$.
    Then $f$ is an open partial homeomorphism from $X$ to $Y$.
\end{theorem}
\begin{proof}
    $X$ is a topological space and $Y$ is a topological space and
        $f\in\bijections{\carrier[X]}{\carrier[Y]}$ by \auto{homeomorphism,continuous_map}.
    $f$ is a function by \obv{bijection_is_function}.
    $\dom{f}=\carrier[X]$ by \obv{bijections_dom}.
    $f\in\surj{\carrier[X]}{\carrier[Y]}$ by \obv{bijections}.
    $\ran{f}=\carrier[Y]$ by \obv{ran_of_surj}.
    $\dom{f}\in\opens[X]$ and $\ran{f}\in\opens[Y]$ by \obv{carrier_open}.
    $\carrier[X]\subseteq\carrier[X]$ by \obv{subseteq_refl}.
    $\restrl{f}{\carrier[X]}$ is a homeomorphism from
        $\subspaceTopology{X}{\carrier[X]}$ to
        $\subspaceTopology{Y}{\img{f}{\carrier[X]}}$ by
        \obv{homeomorphism_restriction_to_image}.
    $\restrl{f}{\carrier[X]}=f$ by \auto{restrl_by_dom}.
    $\img{f}{\carrier[X]}=\ran{f}$ by \auto{img_dom}.
    Follows by \auto{open_partial_homeomorphism}.
\end{proof}

\begin{theorem}\label{open_partial_homeomorphism_same_source_fields}
    Let $X,Y,E$ be topological spaces.
    Suppose $\carrier[X]=\carrier[Y]$ and $\opens[X]=\opens[Y]$.
    Then $h$ is an open partial homeomorphism from $X$ to $E$ iff
        $h$ is an open partial homeomorphism from $Y$ to $E$.
\end{theorem}
\begin{proof}
    $\subspaceTopology{X}{\dom{h}}=\subspaceTopology{Y}{\dom{h}}$ by
        \obv{subspace_topology_same_fields}.
    Follows by \auto{open_partial_homeomorphism}.
\end{proof}

\begin{theorem}\label{open_partial_homeomorphism_open_subspace}
    Let $X,E$ be topological spaces.
    Let $U\in\opens[X]$.
    Then $h$ is an open partial homeomorphism from $\subspaceTopology{X}{U}$ to $E$ iff
        $h$ is an open partial homeomorphism from $X$ to $E$ and $\dom{h}\subseteq U$.
\end{theorem}
\begin{proof}
    $\subspaceTopology{X}{U}$ is a topological space by \obv{subspace_topology_is_topological_space}.
    $U\subseteq\carrier[X]$ by \obv{opens_type}.
    $\dom{h}\in\opens[\subspaceTopology{X}{U}]$ iff
        $\dom{h}\in\opens[X]$ and $\dom{h}\subseteq U$ by \obv{open_subspace_open_iff}.
    If $\dom{h}\subseteq U$, then
        $\subspaceTopology{\subspaceTopology{X}{U}}{\dom{h}}=\subspaceTopology{X}{\dom{h}}$ by
        \obv{nested_subspace_topology}.
    Follows by \auto{open_partial_homeomorphism}.
\end{proof}
