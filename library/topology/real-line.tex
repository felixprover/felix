\section{The real line and its unit interval}

\import{topology/subspace.tex}
\import{topology/induced-topology.tex}
\import{surreal/real-metric.tex}

\begin{definition}\label{real_topology}
    $\RealTopology=\metricTopology{\RealMetric}$.
\end{definition}

\begin{classification}\label{real_topology_is_topological_space}
    $\RealTopology$ is a topological space.
\end{classification}
\begin{proof}
    Follows by
        \obv{real_topology,metric_topology_is_topological_space,surreal_real_pseudometric_space}.
\end{proof}

\begin{proposition}\label{real_topology_carrier}
    $\carrier[\RealTopology]=\reals$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{real_topology,metric_topology_carrier,surreal_real_metric_carrier}.
\end{proof}

\begin{definition}\label{real_unit_interval}
    $\RealUnitInterval=
        \{x\in\reals\mid
            \surrealLe{\surrealZero}{x}\land
            \surrealLe{x}{\surrealOne}\}$.
\end{definition}

\begin{proposition}\label{real_unit_interval_iff}
    $x\in\RealUnitInterval$ iff
        $x\in\reals$ and
        $\surrealLe{\surrealZero}{x}$ and
        $\surrealLe{x}{\surrealOne}$.
\end{proposition}
\begin{proof}
    Follows by \auto{real_unit_interval}.
\end{proof}

\begin{proposition}\label{real_unit_interval_subset_reals}
    $\RealUnitInterval\subseteq\reals$.
\end{proposition}
\begin{proof}
    Follows by \obv{real_unit_interval_iff,subseteq}.
\end{proof}

\begin{proposition}\label{surreal_zero_in_real_unit_interval}
    $\surrealZero\in\RealUnitInterval$.
\end{proposition}
\begin{proof}
    $\surrealZero\in\reals$ by \obv{surreal_zero_is_real}.
    $\surrealLe{\surrealZero}{\surrealZero}$
        by \obv{surreal_le_reflexive,empty_surreal}.
    $\surrealLe{\surrealZero}{\surrealOne}$
        by \obv{surreal_zero_lt_one,surreal_le}.
    Follows by \obv{real_unit_interval_iff}.
\end{proof}

\begin{proposition}\label{surreal_one_in_real_unit_interval}
    $\surrealOne\in\RealUnitInterval$.
\end{proposition}
\begin{proof}
    $\surrealOne\in\reals$ by \obv{surreal_one_is_real}.
    $\surrealLe{\surrealZero}{\surrealOne}$
        by \obv{surreal_zero_lt_one,surreal_le}.
    $\surrealLe{\surrealOne}{\surrealOne}$
        by \obv{surreal_le_reflexive,surreal_one_is_surreal}.
    Follows by \obv{real_unit_interval_iff}.
\end{proof}

\begin{definition}\label{real_unit_interval_topology}
    $\RealUnitIntervalTopology=
        \subspaceTopology{\RealTopology}{\RealUnitInterval}$.
\end{definition}

\begin{classification}\label{real_unit_interval_topology_is_topological_space}
    $\RealUnitIntervalTopology$ is a topological space.
\end{classification}
\begin{proof}
    Follows by
        \auto{real_unit_interval_topology,subspace_topology_is_topological_space,real_topology_is_topological_space}.
\end{proof}

\begin{proposition}\label{real_unit_interval_topology_carrier}
    $\carrier[\RealUnitIntervalTopology]=\RealUnitInterval$.
\end{proposition}
\begin{proof}
    $\RealUnitInterval\subseteq\carrier[\RealTopology]$
        by
        \obv{real_unit_interval_subset_reals,real_topology_carrier}.
    Follows by
        \auto{real_unit_interval_topology,subspace_topology_carrier_of_subset}.
\end{proof}

\begin{theorem}\label{real_unit_ball_is_open_neighbourhood}
    Let $a\in\reals$.
    Then $\metricBall{\RealMetric}{a}{\surrealOne}
        \in\opens[\RealTopology]$ and
        $a\in\metricBall{\RealMetric}{a}{\surrealOne}$.
\end{theorem}
\begin{proof}
    $\RealMetric$ is a pseudometric space by
        \obv{surreal_real_pseudometric_space}.
    $a\in\carrier[\RealMetric]$ by
        \obv{surreal_real_metric_carrier}.
    $\surrealOne\in\reals$ by \obv{surreal_one_is_real}.
    $\surrealLt{\surrealZero}{\surrealOne}$ by
        \obv{surreal_zero_lt_one}.
    Hence $\metricBall{\RealMetric}{a}{\surrealOne}
        \in\metricOpens{\RealMetric}$ by \obv{metric_ball_is_open}.
    $\opens[\metricTopology{\RealMetric}]=\metricOpens{\RealMetric}$
        by \obv{metric_topology_opens}.
    $\RealTopology=\metricTopology{\RealMetric}$ by \obv{real_topology}.
    Hence $\metricBall{\RealMetric}{a}{\surrealOne}
        \in\opens[\RealTopology]$.
    $a\in\metricBall{\RealMetric}{a}{\surrealOne}$ by
        \obv{metric_ball_contains_centre}.
\end{proof}
