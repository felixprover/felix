\section{Components of topological manifolds}

\import{topology/connected-component.tex}
\import{topology/manifold-properties.tex}
\import{topology/manifold-open.tex}

Local connectedness makes each component open. Its manifold structure is
therefore just its subspace topology, with the original dimension. No atlas
or additional record is needed. Outside the carrier the component is empty.

\begin{theorem}\label{locally_euclidean_components_are_open}
    Let $n$ be a natural number and $X$ be a topological space.
    Suppose $X$ is locally Euclidean of dimension $n$.
    Then $\connectedComponent{X}{p}\in\opens[X]$.
\end{theorem}
\begin{proof}
    $X$ is locally connected by \obv{locally_euclidean_is_locally_connected}.
    Follows by \obv{locally_connected_components_are_open}.
\end{proof}

\begin{theorem}\label{component_of_topological_manifold}
    Let $n$ be a natural number and $X$ be a topological space.
    Suppose $X$ is a topological manifold of dimension $n$.
    Then $\subspaceTopology{X}{\connectedComponent{X}{p}}$ is a topological manifold of dimension $n$.
\end{theorem}
\begin{proof}
    $X$ is locally Euclidean of dimension $n$ by \obv{topological_manifold}.
    $\connectedComponent{X}{p}\in\opens[X]$ by \obv{locally_euclidean_components_are_open}.
    Follows by \obv{open_subspace_is_topological_manifold}.
\end{proof}
