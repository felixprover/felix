\section{Unions of connected subsets}

\import{topology/connectedness.tex}

A connected subset cannot cross a separation of its ambient space. This
observation proves that a family of connected subsets with a common point
has connected union.

\begin{theorem}\label{connected_subset_in_separation}
    Let $X$ be a topological space.
    Suppose $A$ is connected in $X$ and $U,V\in\opens[X]$.
    Suppose $\carrier[X]=U\union V$ and $U\inter V=\emptyset$.
    Then $A\subseteq U$ or $A\subseteq V$.
\end{theorem}
\begin{proof}
    Let $S=\subspaceTopology{X}{A}$.
    $A\subseteq\carrier[X]$ and $S$ is connected by \obv{connected_in}.
    $S$ is a topological space by \obv{subspace_topology_is_topological_space}.
    $\carrier[S]=A$ by \auto{subspace_topology_carrier_of_subset}.
    $U\inter A\in\opens[S]$ and $V\inter A\in\opens[S]$ by
        \auto{subspace_open_iff}.
    Suppose not.
    $U\inter A$ and $V\inter A$ are inhabited by \auto{subseteq,union,inter}.
    $U\inter A$ is disjoint from $V\inter A$ by \auto{disjoint,inter,emptyset}.
    $(U\inter A)\union(V\inter A)=(U\union V)\inter A$ by
        \auto{inter_distrib_union,inter_comm}.
    Hence $(U\inter A)\union(V\inter A)=\carrier[S]$ by
        \auto{inter_absorb_supseteq_left}.
    $\carrier[S]$ is partitioned by $U\inter A$ and $V\inter A$ by
        \auto{nonempty_inhabited}.
    $S$ is disconnected by \auto{disconnected}.
    Contradiction by \obv{connected}.
\end{proof}

\begin{theorem}\label{connected_in_subspace_iff}
    Let $X$ be a topological space.
    Suppose $A\subseteq B$ and $B\subseteq\carrier[X]$.
    Then $A$ is connected in $\subspaceTopology{X}{B}$ iff $A$ is connected in $X$.
\end{theorem}
\begin{proof}
    $\carrier[\subspaceTopology{X}{B}]=B$ by \obv{subspace_topology_carrier_of_subset}.
    $A\subseteq\carrier[X]$ by \obv{subseteq_transitive}.
    $\subspaceTopology{\subspaceTopology{X}{B}}{A}=\subspaceTopology{X}{A}$ by
        \obv{nested_subspace_topology}.
    Follows by \auto{connected_in,subspace_topology_is_topological_space}.
\end{proof}

\begin{theorem}\label{union_of_connected_subsets_with_common_point}
    Let $X$ be a topological space.
    Suppose for every $A\in F$ we have $A$ is connected in $X$ and $p\in A$.
    Then $\unions{F}$ is connected in $X$.
\end{theorem}
\begin{proof}
    Let $B=\unions{F}$ and $S=\subspaceTopology{X}{B}$.
    $B\subseteq\carrier[X]$ by \auto{connected_in,unions_iff,subseteq}.
    $S$ is a topological space by \obv{subspace_topology_is_topological_space}.
    $\carrier[S]=B$ by \auto{subspace_topology_carrier_of_subset}.
    Show that $S$ is connected.
    \begin{subproof}
        Suppose not.
        $S$ is disconnected by \auto{connected}.
        Take $U,V\in\opens[S]$ such that $\carrier[S]$ is partitioned by $U$ and $V$ by
            \obv{disconnected}.
        $U\inter V=\emptyset$ by \auto{disjoint,inter,emptyset,setext}.
        Show that for every $A\in F$ we have $A\subseteq U$ or $A\subseteq V$.
        \begin{subproof}
            Fix $A\in F$.
            $A$ is connected in $X$.
            $A\subseteq B$ by \auto{unions_intro,subseteq}.
            $A$ is connected in $S$ by \auto{connected_in_subspace_iff}.
            Follows by \obv{connected_subset_in_separation}.
        \end{subproof}
        $U$ and $V$ are inhabited by \obv{nonempty_inhabited}.
        Take $u$ such that $u\in U$.
        Take $v$ such that $v\in V$.
        $u\in B$ and $v\in B$ by \auto{union}.
        Take $A\in F$ such that $u\in A$ by \auto{unions_iff}.
        Take $D\in F$ such that $v\in D$ by \auto{unions_iff}.
        $p\in A$ and $p\in D$.
        $A\subseteq U$ or $A\subseteq V$.
        Hence $p\in U$ by \auto{disjoint,subseteq}.
        $D\subseteq U$ or $D\subseteq V$.
        Contradiction by \auto{disjoint,subseteq}.
    \end{subproof}
    Follows by \obv{connected_in}.
\end{proof}

\begin{corollary}\label{union_of_intersecting_connected_subsets}
    Let $X$ be a topological space.
    Suppose $A$ and $B$ are connected in $X$ and $p\in A\inter B$.
    Then $A\union B$ is connected in $X$.
\end{corollary}
\begin{proof}
    For every $D\in\{A,B\}$ we have $D$ is connected in $X$ and $p\in D$ by
        \auto{upair_iff,inter}.
    $\unions{\{A,B\}}$ is connected in $X$ by
        \obv{union_of_connected_subsets_with_common_point}.
    Follows by \obv{union_as_unions}.
\end{proof}
