\section{Perfectoid Tate rings}

The implications in the following definitions are formation guards for
guarded ring and completeness expressions.  Their antecedents follow from
the preceding prime, Tate, and pseudouniformizer clauses; the propositions
below expose the usual unguarded mathematical consequences.

\begin{definition}\label{perfectoid_witness}
    $\pi$ is a perfectoid witness for $R$ at $p$ iff
        $p$ is a prime natural number and
        $R$ is a Tate ring and
        $\pi$ is a pseudouniformizer of $R$ and
        (if $R$ is a commutative topological ring and
            $R$ is a nonarchimedean topological ring and
            $p$ is a prime natural number and
            $\pi\in\carrier[\PowerBoundedRing{R}]$, then
            $\ringPower{\PowerBoundedRing{R}}{\pi}{p}$ divides
                $\ringNatural{\PowerBoundedRing{R}}{p}$ in
                $\PowerBoundedRing{R}$ and
            $\principalQuotientFrobenius{\PowerBoundedRing{R}}{p}{\pi}$
                is a ring isomorphism from
                $\QuotientRing{\PowerBoundedRing{R}}{
                    \principalIdeal{\PowerBoundedRing{R}}{\pi}}$ to
                $\QuotientRing{\PowerBoundedRing{R}}{
                    \principalIdeal{\PowerBoundedRing{R}}{
                        \ringPower{\PowerBoundedRing{R}}{\pi}{p}}}$).
\end{definition}

\begin{definition}\label{perfectoid_tate_ring}
    $R$ is a perfectoid Tate ring at $p$ iff
        $p$ is a prime natural number and
        $R$ is a Tate ring and
        (if $R$ is a topological ring, then
            $R$ is additively complete) and
        $R$ is a uniform topological ring and
        there exists $\pi$ such that
            $\pi$ is a perfectoid witness for $R$ at $p$.
\end{definition}

\subsection{Basic consequences}

\begin{proposition}\label{perfectoid_witness_prime}
    Suppose $\pi$ is a perfectoid witness for $R$ at $p$.
    Then $p$ is a prime natural number.
\end{proposition}
\begin{proof}
    Follows by \obv{perfectoid_witness}.
\end{proof}

\begin{proposition}\label{perfectoid_witness_tate_ring}
    Suppose $\pi$ is a perfectoid witness for $R$ at $p$.
    Then $R$ is a Tate ring.
\end{proposition}
\begin{proof}
    Follows by \obv{perfectoid_witness}.
\end{proof}

\begin{proposition}\label{perfectoid_witness_pseudouniformizer}
    Suppose $\pi$ is a perfectoid witness for $R$ at $p$.
    Then $\pi$ is a pseudouniformizer of $R$.
\end{proposition}
\begin{proof}
    Follows by \obv{perfectoid_witness}.
\end{proof}

\begin{proposition}\label{perfectoid_witness_in_power_bounded_ring}
    Suppose $\pi$ is a perfectoid witness for $R$ at $p$.
    Then $\pi\in\carrier[\PowerBoundedRing{R}]$.
\end{proposition}
\begin{proof}
    $R$ is a Tate ring by \obv{perfectoid_witness_tate_ring}.
    Hence $R$ is a commutative topological ring by
        \obv{tate_ring_is_commutative_topological_ring}.
    $R$ is a nonarchimedean topological ring by
        \obv{tate_ring_is_nonarchimedean}.
    $\pi$ is a pseudouniformizer of $R$ by
        \obv{perfectoid_witness_pseudouniformizer}.
    Follows by \obv{pseudouniformizer_in_power_bounded_ring}.
\end{proof}

\begin{proposition}\label{perfectoid_witness_divisibility}
    Let $R$ be a Tate ring and $p$ be a prime natural number.
    Let $\pi$ be a pseudouniformizer of $R$.
    Suppose $\pi$ is a perfectoid witness for $R$ at $p$.
    Then $\ringPower{\PowerBoundedRing{R}}{\pi}{p}$ divides
        $\ringNatural{\PowerBoundedRing{R}}{p}$ in
        $\PowerBoundedRing{R}$.
\end{proposition}
\begin{proof}
    $R$ is a commutative topological ring by
        \obv{tate_ring_is_commutative_topological_ring}.
    $R$ is a nonarchimedean topological ring by
        \obv{tate_ring_is_nonarchimedean}.
    $\pi\in\carrier[\PowerBoundedRing{R}]$ by
        \obv{perfectoid_witness_in_power_bounded_ring}.
    Follows by \auto{perfectoid_witness}.
\end{proof}

\subsection{The perfectoid Frobenius}

\begin{proposition}\label{perfectoid_witness_frobenius_context}
    Let $R$ be a Tate ring and $p$ be a prime natural number.
    Let $\pi$ be a pseudouniformizer of $R$.
    Suppose $\pi$ is a perfectoid witness for $R$ at $p$.
    Then $R$ is a commutative topological ring and
        $R$ is a nonarchimedean topological ring and
        $\pi\in\carrier[\PowerBoundedRing{R}]$ and
        $\ringPower{\PowerBoundedRing{R}}{\pi}{p}$ divides
            $\ringNatural{\PowerBoundedRing{R}}{p}$ in
            $\PowerBoundedRing{R}$.
\end{proposition}
\begin{proof}
    $R$ is a commutative topological ring by
        \obv{tate_ring_is_commutative_topological_ring}.
    $R$ is a nonarchimedean topological ring by
        \obv{tate_ring_is_nonarchimedean}.
    $\pi\in\carrier[\PowerBoundedRing{R}]$ by
        \obv{perfectoid_witness_in_power_bounded_ring}.
    $\ringPower{\PowerBoundedRing{R}}{\pi}{p}$ divides
        $\ringNatural{\PowerBoundedRing{R}}{p}$ in
        $\PowerBoundedRing{R}$ by
        \obv{perfectoid_witness_divisibility}.
\end{proof}

\begin{proposition}\label{perfectoid_witness_frobenius_type}
    Let $R$ be a Tate ring and $p$ be a prime natural number.
    Let $\pi$ be a pseudouniformizer of $R$.
    Suppose $\pi$ is a perfectoid witness for $R$ at $p$.
    Then $\principalQuotientFrobenius{\PowerBoundedRing{R}}{p}{\pi}
        \in\funs{
            \carrier[\QuotientRing{\PowerBoundedRing{R}}{
                \principalIdeal{\PowerBoundedRing{R}}{\pi}}]}{
            \carrier[\QuotientRing{\PowerBoundedRing{R}}{
                \principalIdeal{\PowerBoundedRing{R}}{
                    \ringPower{\PowerBoundedRing{R}}{\pi}{p}}}]}$.
\end{proposition}
\begin{proof}
    $R$ is a commutative topological ring and
        $R$ is a nonarchimedean topological ring and
        $\ringPower{\PowerBoundedRing{R}}{\pi}{p}$ divides
            $\ringNatural{\PowerBoundedRing{R}}{p}$ in
            $\PowerBoundedRing{R}$ by
        \auto{perfectoid_witness_frobenius_context}.
    Follows by \obv{power_bounded_principal_frobenius_type}.
\end{proof}

\begin{proposition}\label{perfectoid_witness_frobenius_apply}
    Let $R$ be a Tate ring and $p$ be a prime natural number.
    Let $\pi$ be a pseudouniformizer of $R$.
    Suppose $\pi$ is a perfectoid witness for $R$ at $p$.
    Let $x\in\carrier[\PowerBoundedRing{R}]$.
    Let
        $q_a=\groupQuotientProjection{
            \AdditiveGroup{\PowerBoundedRing{R}}}{
            \principalIdeal{\PowerBoundedRing{R}}{\pi}}$ and
        $q_p=\groupQuotientProjection{
            \AdditiveGroup{\PowerBoundedRing{R}}}{
            \principalIdeal{\PowerBoundedRing{R}}{
                \ringPower{\PowerBoundedRing{R}}{\pi}{p}}}$.
    Then $\apply{
            \principalQuotientFrobenius{\PowerBoundedRing{R}}{p}{\pi}}{
            q_a(x)}
        =q_p(\ringPower{\PowerBoundedRing{R}}{x}{p})$.
\end{proposition}
\begin{proof}
    $R$ is a commutative topological ring and
        $R$ is a nonarchimedean topological ring and
        $\ringPower{\PowerBoundedRing{R}}{\pi}{p}$ divides
            $\ringNatural{\PowerBoundedRing{R}}{p}$ in
            $\PowerBoundedRing{R}$ by
        \auto{perfectoid_witness_frobenius_context}.
    Follows by \obv{power_bounded_principal_frobenius_apply}.
\end{proof}

\begin{theorem}\label{perfectoid_witness_frobenius_injective}
    Let $R$ be a Tate ring and $p$ be a prime natural number.
    Let $\pi$ be a pseudouniformizer of $R$.
    Suppose $\pi$ is a perfectoid witness for $R$ at $p$.
    Then $\principalQuotientFrobenius{\PowerBoundedRing{R}}{p}{\pi}$
        is injective.
\end{theorem}
\begin{proof}
    $R$ is a commutative topological ring and
        $R$ is a nonarchimedean topological ring and
        $\ringPower{\PowerBoundedRing{R}}{\pi}{p}$ divides
            $\ringNatural{\PowerBoundedRing{R}}{p}$ in
            $\PowerBoundedRing{R}$ by
        \auto{perfectoid_witness_frobenius_context}.
    Follows by \obv{power_bounded_principal_frobenius_injective}.
\end{proof}

\begin{theorem}\label{perfectoid_witness_frobenius_isomorphism_iff_surjective}
    Let $R$ be a Tate ring and $p$ be a prime natural number.
    Let $\pi$ be a pseudouniformizer of $R$.
    Suppose $\pi$ is a perfectoid witness for $R$ at $p$.

            \principalIdeal{\PowerBoundedRing{R}}{\pi}}$ to
        $\QuotientRing{\PowerBoundedRing{R}}{
            \principalIdeal{\PowerBoundedRing{R}}{
                \ringPower{\PowerBoundedRing{R}}{\pi}{p}}}$ iff
        $\principalQuotientFrobenius{\PowerBoundedRing{R}}{p}{\pi}
            \in\surj{
                \carrier[\QuotientRing{\PowerBoundedRing{R}}{
                    \principalIdeal{\PowerBoundedRing{R}}{\pi}}]}{
                \carrier[\QuotientRing{\PowerBoundedRing{R}}{
                    \principalIdeal{\PowerBoundedRing{R}}{
                        \ringPower{\PowerBoundedRing{R}}{\pi}{p}}}]}$.
\end{theorem}
\begin{proof}
    $R$ is a commutative topological ring and
        $R$ is a nonarchimedean topological ring and
        $\ringPower{\PowerBoundedRing{R}}{\pi}{p}$ divides
            $\ringNatural{\PowerBoundedRing{R}}{p}$ in
            $\PowerBoundedRing{R}$ by
        \auto{perfectoid_witness_frobenius_context}.
    Follows by
        \obv{power_bounded_principal_frobenius_isomorphism_iff_surjective}.
\end{proof}

\begin{theorem}\label{perfectoid_witness_frobenius_injective_iff}
    Let $R$ be a Tate ring and $p$ be a prime natural number.
    Let $\pi$ be a pseudouniformizer of $R$.
    Suppose $\pi$ is a perfectoid witness for $R$ at $p$.
    Then $\principalQuotientFrobenius{\PowerBoundedRing{R}}{p}{\pi}$
        is injective iff for every
        $x\in\carrier[\PowerBoundedRing{R}]$ such that
            $\ringPower{\PowerBoundedRing{R}}{x}{p}
                \in\principalIdeal{\PowerBoundedRing{R}}{
                    \ringPower{\PowerBoundedRing{R}}{\pi}{p}}$
        we have $x\in\principalIdeal{\PowerBoundedRing{R}}{\pi}$.
\end{theorem}
\begin{proof}
    $R$ is a commutative topological ring and
        $R$ is a nonarchimedean topological ring and
        $\pi\in\carrier[\PowerBoundedRing{R}]$ and
        $\ringPower{\PowerBoundedRing{R}}{\pi}{p}$ divides
            $\ringNatural{\PowerBoundedRing{R}}{p}$ in
            $\PowerBoundedRing{R}$ by
        \obv{perfectoid_witness_frobenius_context}.
    $\PowerBoundedRing{R}$ is a commutative ring by
        \obv{power_bounded_ring_is_commutative_ring}.
    Follows by \obv{principal_quotient_frobenius_injective_iff}.
\end{proof}

\begin{theorem}\label{perfectoid_witness_frobenius_kernel}
    Let $R$ be a Tate ring and $p$ be a prime natural number.
    Let $\pi$ be a pseudouniformizer of $R$.
    Suppose $\pi$ is a perfectoid witness for $R$ at $p$.
    Then for every $x\in\carrier[\PowerBoundedRing{R}]$ such that
        $\ringPower{\PowerBoundedRing{R}}{x}{p}
            \in\principalIdeal{\PowerBoundedRing{R}}{
                \ringPower{\PowerBoundedRing{R}}{\pi}{p}}$
        we have $x\in\principalIdeal{\PowerBoundedRing{R}}{\pi}$.
\end{theorem}
\begin{proof}
    $\principalQuotientFrobenius{\PowerBoundedRing{R}}{p}{\pi}$
        is injective by
        \obv{perfectoid_witness_frobenius_injective}.
    Follows by \obv{perfectoid_witness_frobenius_injective_iff}.
\end{proof}

\begin{theorem}\label{perfectoid_witness_frobenius_surjective_iff}
    Let $R$ be a Tate ring and $p$ be a prime natural number.
    Let $\pi$ be a pseudouniformizer of $R$.
    Suppose $\pi$ is a perfectoid witness for $R$ at $p$.
    Then $\principalQuotientFrobenius{\PowerBoundedRing{R}}{p}{\pi}
        \in\surj{
            \carrier[\QuotientRing{\PowerBoundedRing{R}}{
                \principalIdeal{\PowerBoundedRing{R}}{\pi}}]}{
            \carrier[\QuotientRing{\PowerBoundedRing{R}}{
                \principalIdeal{\PowerBoundedRing{R}}{
                    \ringPower{\PowerBoundedRing{R}}{\pi}{p}}}]}$ iff
        for every $y\in\carrier[\PowerBoundedRing{R}]$ there exists
            $x\in\carrier[\PowerBoundedRing{R}]$ such that
            $y\mathrel{\idealCongruence{\PowerBoundedRing{R}}{
                \principalIdeal{\PowerBoundedRing{R}}{
                    \ringPower{\PowerBoundedRing{R}}{\pi}{p}}}}
                \ringPower{\PowerBoundedRing{R}}{x}{p}$.
\end{theorem}
\begin{proof}
    $R$ is a commutative topological ring and
        $R$ is a nonarchimedean topological ring and
        $\pi\in\carrier[\PowerBoundedRing{R}]$ and
        $\ringPower{\PowerBoundedRing{R}}{\pi}{p}$ divides
            $\ringNatural{\PowerBoundedRing{R}}{p}$ in
            $\PowerBoundedRing{R}$ by
        \obv{perfectoid_witness_frobenius_context}.
    $\PowerBoundedRing{R}$ is a commutative ring by
        \obv{power_bounded_ring_is_commutative_ring}.
    Follows by \obv{principal_quotient_frobenius_surjective_iff}.
\end{proof}

                    \ringPower{\PowerBoundedRing{R}}{\pi}{p}}}}

\end{theorem}
\begin{proof}
    Let $F=\principalQuotientFrobenius{\PowerBoundedRing{R}}{p}{\pi}$.
    $F$ is a ring isomorphism from
        $\QuotientRing{\PowerBoundedRing{R}}{
            \principalIdeal{\PowerBoundedRing{R}}{\pi}}$ to

            \principalIdeal{\PowerBoundedRing{R}}{

            \carrier[\QuotientRing{\PowerBoundedRing{R}}{
                \principalIdeal{\PowerBoundedRing{R}}{\pi}}]}{
            \carrier[\QuotientRing{\PowerBoundedRing{R}}{
                \principalIdeal{\PowerBoundedRing{R}}{
                    \ringPower{\PowerBoundedRing{R}}{\pi}{p}}}]}$ by
        \auto{perfectoid_witness_frobenius_isomorphism_iff_surjective}.
    $F\in\surj{
            \carrier[\QuotientRing{\PowerBoundedRing{R}}{
                \principalIdeal{\PowerBoundedRing{R}}{\pi}}]}{
            \carrier[\QuotientRing{\PowerBoundedRing{R}}{
                \principalIdeal{\PowerBoundedRing{R}}{
                    \ringPower{\PowerBoundedRing{R}}{\pi}{p}}}]}$ iff

                \ringPower{\PowerBoundedRing{R}}{x}{p}$ by
        \auto{perfectoid_witness_frobenius_surjective_iff}.
    For every $x\in\carrier[\PowerBoundedRing{R}]$ such that

            \in\principalIdeal{\PowerBoundedRing{R}}{
                \ringPower{\PowerBoundedRing{R}}{\pi}{p}}$

\subsection{The defining criterion}

\begin{theorem}\label{perfectoid_tate_ring_iff}
    Let $R$ be a Tate ring and $p$ be a prime natural number.
    Then $R$ is a perfectoid Tate ring at $p$ iff
        $p$ is a prime natural number and
        $R$ is a Tate ring and
        $R$ is additively complete and
        $\Bounded{R}{\PowerBoundedCarrier{R}}$ and
        there exists $\pi$ such that
            $\pi$ is a pseudouniformizer of $R$ and
            $\ringPower{\PowerBoundedRing{R}}{\pi}{p}$ divides
                $\ringNatural{\PowerBoundedRing{R}}{p}$ in
                $\PowerBoundedRing{R}$ and
            $\principalQuotientFrobenius{
                    \PowerBoundedRing{R}}{p}{\pi}$
                is a ring isomorphism from
                $\QuotientRing{\PowerBoundedRing{R}}{
                    \principalIdeal{\PowerBoundedRing{R}}{\pi}}$ to
                $\QuotientRing{\PowerBoundedRing{R}}{
                    \principalIdeal{\PowerBoundedRing{R}}{
                        \ringPower{\PowerBoundedRing{R}}{\pi}{p}}}$.
\end{theorem}
\begin{proof}
    $R$ is a commutative topological ring by
        \obv{tate_ring_is_commutative_topological_ring}.
    $R$ is a nonarchimedean topological ring by
        \obv{tate_ring_is_nonarchimedean}.
    Hence $R$ is a topological ring by
        \obv{commutative_topological_ring_is_topological_ring}.

    Show that if $R$ is a perfectoid Tate ring at $p$, then
        $p$ is a prime natural number and
        $R$ is a Tate ring and
        $R$ is additively complete and
        $\Bounded{R}{\PowerBoundedCarrier{R}}$ and
        there exists $\pi$ such that
            $\pi$ is a pseudouniformizer of $R$ and
            $\ringPower{\PowerBoundedRing{R}}{\pi}{p}$ divides
                $\ringNatural{\PowerBoundedRing{R}}{p}$ in
                $\PowerBoundedRing{R}$ and
            $\principalQuotientFrobenius{
                    \PowerBoundedRing{R}}{p}{\pi}$
                is a ring isomorphism from
                $\QuotientRing{\PowerBoundedRing{R}}{
                    \principalIdeal{\PowerBoundedRing{R}}{\pi}}$ to
                $\QuotientRing{\PowerBoundedRing{R}}{
                    \principalIdeal{\PowerBoundedRing{R}}{
                        \ringPower{\PowerBoundedRing{R}}{\pi}{p}}}$.

        Assume $R$ is a perfectoid Tate ring at $p$.
        $R$ is additively complete by
            \obv{perfectoid_tate_ring}.
        $R$ is a uniform topological ring by
            \obv{perfectoid_tate_ring}.
        Hence $\Bounded{R}{\PowerBoundedCarrier{R}}$ by
            \obv{uniform_topological_ring}.
        Take $\pi$ such that $\pi$ is a perfectoid witness for $R$ at $p$
            by \obv{perfectoid_tate_ring}.
        $\pi$ is a pseudouniformizer of $R$ by
            \obv{perfectoid_witness_pseudouniformizer}.
        $\ringPower{\PowerBoundedRing{R}}{\pi}{p}$ divides
            $\ringNatural{\PowerBoundedRing{R}}{p}$ in
            $\PowerBoundedRing{R}$ by
            \obv{perfectoid_witness_divisibility}.

                \PowerBoundedRing{R}}{p}{\pi}$

            $\QuotientRing{\PowerBoundedRing{R}}{
                \principalIdeal{\PowerBoundedRing{R}}{\pi}}$ to
            $\QuotientRing{\PowerBoundedRing{R}}{
                \principalIdeal{\PowerBoundedRing{R}}{
                    \ringPower{\PowerBoundedRing{R}}{\pi}{p}}}$ by
            \obv{perfectoid_witness_frobenius_isomorphism}.
    \end{subproof}

    Show that if $p$ is a prime natural number and
        $R$ is a Tate ring and
        $R$ is additively complete and
        $\Bounded{R}{\PowerBoundedCarrier{R}}$ and
        there exists $\pi$ such that
            $\pi$ is a pseudouniformizer of $R$ and
            $\ringPower{\PowerBoundedRing{R}}{\pi}{p}$ divides
                $\ringNatural{\PowerBoundedRing{R}}{p}$ in
                $\PowerBoundedRing{R}$ and
            $\principalQuotientFrobenius{
                    \PowerBoundedRing{R}}{p}{\pi}$
                is a ring isomorphism from
                $\QuotientRing{\PowerBoundedRing{R}}{
                    \principalIdeal{\PowerBoundedRing{R}}{\pi}}$ to
                $\QuotientRing{\PowerBoundedRing{R}}{
                    \principalIdeal{\PowerBoundedRing{R}}{
                        \ringPower{\PowerBoundedRing{R}}{\pi}{p}}}$,
        then $R$ is a perfectoid Tate ring at $p$.
    \begin{subproof}
        Assume $p$ is a prime natural number and
            $R$ is a Tate ring and
            $R$ is additively complete and
            $\Bounded{R}{\PowerBoundedCarrier{R}}$ and
            there exists $\pi$ such that
                $\pi$ is a pseudouniformizer of $R$ and
                $\ringPower{\PowerBoundedRing{R}}{\pi}{p}$ divides
                    $\ringNatural{\PowerBoundedRing{R}}{p}$ in
                    $\PowerBoundedRing{R}$ and
                $\principalQuotientFrobenius{
                        \PowerBoundedRing{R}}{p}{\pi}$
                    is a ring isomorphism from
                    $\QuotientRing{\PowerBoundedRing{R}}{
                        \principalIdeal{\PowerBoundedRing{R}}{\pi}}$ to
                    $\QuotientRing{\PowerBoundedRing{R}}{
                        \principalIdeal{\PowerBoundedRing{R}}{
                            \ringPower{\PowerBoundedRing{R}}{\pi}{p}}}$.
        Take $\pi$ such that
            $\pi$ is a pseudouniformizer of $R$ and
            $\ringPower{\PowerBoundedRing{R}}{\pi}{p}$ divides
                $\ringNatural{\PowerBoundedRing{R}}{p}$ in
                $\PowerBoundedRing{R}$ and
            $\principalQuotientFrobenius{
                    \PowerBoundedRing{R}}{p}{\pi}$
                is a ring isomorphism from
                $\QuotientRing{\PowerBoundedRing{R}}{
                    \principalIdeal{\PowerBoundedRing{R}}{\pi}}$ to

                    \principalIdeal{\PowerBoundedRing{R}}{
                        \ringPower{\PowerBoundedRing{R}}{\pi}{p}}}$.
        $\pi\in\carrier[\PowerBoundedRing{R}]$ by
            \obv{pseudouniformizer_in_power_bounded_ring}.
        Hence $\pi$ is a perfectoid witness for $R$ at $p$ by
            \obv{perfectoid_witness}.
        $R$ is a uniform topological ring by
            \obv{uniform_topological_ring}.
        Follows by \obv{perfectoid_tate_ring}.
    \end{subproof}
    Follows by \auto.
\end{proof}
