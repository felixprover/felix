\section{Local compactness of finite powers}

Finite-set induction adds one coordinate at a time. This proof uses the
actual index set, without choosing an enumeration or a nested-product encoding.

\begin{theorem}\label{singleton_topological_power_is_locally_compact}
    Let $X$ be a locally compact topological space.
    Then $\TopologicalPower{X}{\{i\}}$ is locally compact.
\end{theorem}
\begin{proof}
    $\TopologicalPower{X}{\{i\}}$ is a topological space by
        \obv{topological_power_is_topological_space}.
    $\functionEvaluation{\{i\}}{\carrier[X]}{i}$ is a homeomorphism from
        $\TopologicalPower{X}{\{i\}}$ to $X$ by \obv{singleton_power_evaluation_homeomorphism}.
    Hence $\converse{\functionEvaluation{\{i\}}{\carrier[X]}{i}}$ is a homeomorphism
        from $X$ to $\TopologicalPower{X}{\{i\}}$ by \obv{homeomorphism_inverse}.
    Follows by \obv{homeomorphism_preserves_local_compactness}.
\end{proof}

\begin{theorem}\label{finite_topological_power_is_locally_compact}
    Let $X$ be a locally compact topological space.
    Suppose $I$ is finite.
    Then $\TopologicalPower{X}{I}$ is locally compact.
\end{theorem}
\begin{proof}
    Let $C=\{J\in\pow{I}\mid\text{$\TopologicalPower{X}{J}$ is locally compact}\}$.
    $\TopologicalPower{X}{\emptyset}$ is a compact topological space by
        \auto{topological_power_is_topological_space,empty_topological_power_is_compact}.
    Hence $\TopologicalPower{X}{\emptyset}$ is locally compact by
        \obv{compact_space_is_locally_compact}.
    Hence $\emptyset\in C$ by \obv{powerset_bottom}.
    Show that for all $J,i$ such that $J\in C$ and $i\in I$ we have $\cons{i}{J}\in C$.
    \begin{subproof}
        Fix $J,i$.
        Assume $J\in C$ and $i\in I$.
        $J\subseteq I$ by \obv{pow_iff}.
        Hence $\cons{i}{J}\in\pow{I}$ by \auto{cons_subseteq_intro,powerset_intro}.
        \begin{byCase}
            \caseOf{$i\in J$.}
                $\cons{i}{J}=J$ by \obv{cons_absorb}.
            \caseOf{$i\notin J$.}
                $J$ is disjoint from $\{i\}$ by \auto{disjoint,singleton_iff}.
                $J\union\{i\}=\cons{i}{J}$ by \auto{union_eq_cons,union_comm}.
                $\TopologicalPower{X}{J}$ is a locally compact topological space by
                    \auto{topological_power_is_topological_space}.
                $\TopologicalPower{X}{\{i\}}$ is a locally compact topological space by
                    \auto{topological_power_is_topological_space,singleton_topological_power_is_locally_compact}.
                Hence $\BinaryProductTopology{\TopologicalPower{X}{J}}{\TopologicalPower{X}{\{i\}}}$
                    is locally compact by \obv{binary_product_of_locally_compact_spaces}.
                $\functionJoin{J}{\{i\}}{\carrier[X]}$ is a homeomorphism from
                    $\BinaryProductTopology{\TopologicalPower{X}{J}}{\TopologicalPower{X}{\{i\}}}$
                    to $\TopologicalPower{X}{J\union\{i\}}$ by \obv{topological_power_join_homeomorphism}.
                $\TopologicalPower{X}{J\union\{i\}}$ is locally compact by
                    \auto{homeomorphism_preserves_local_compactness,
                        topological_power_is_topological_space,binary_product_topology_is_topological_space}.
        \end{byCase}
    \end{subproof}
    $I\in C$ by \obv{finite_induction}.
\end{proof}
