\section{Bases under open continuous maps}

\import{topology/basis.tex}
\import{topology/open-map.tex}

\begin{theorem}\label{open_continuous_surjection_maps_basis}
    Let $X,Y$ be topological spaces.
    Let $B$ be an open basis of $X$.
    Suppose $f$ is continuous from $X$ to $Y$ and $f$ is an open map from $X$ to $Y$.
    Suppose $f\in\surj{\carrier[X]}{\carrier[Y]}$.
    Then $\{\img{f}{U}\mid U\in B\}$ is an open basis of $Y$.
\end{theorem}
\begin{proof}
    Let $C=\{\img{f}{U}\mid U\in B\}$.
    $f$ is a function and $\dom{f}=\carrier[X]$ by
        \auto{continuous_map,funs_is_function,funs}.
    $B\subseteq\opens[X]$ by \obv{open_basis}.
    Hence $C\subseteq\opens[Y]$ by \auto{open_map_image_open,subseteq}.
    Show that for all $V,y$ such that $V\in\opens[Y]$ and $y\in V$
        there exists $W\in C$ such that $y\in W\subseteq V$.
    \begin{subproof}
        Fix $V,y$.
        Assume $V\in\opens[Y]$ and $y\in V$.
        $y\in\carrier[Y]$ by \auto{opens_type,subseteq}.
        Take $x\in\carrier[X]$ such that $f(x)=y$ by \obv{surj}.
        $\preimg{f}{V}\in\opens[X]$ by \obv{continuous_map}.
        $x\in\preimg{f}{V}$ by \auto{preimg_iff,function_apply_elim}.
        Take $U\in B$ such that $x\in U\subseteq\preimg{f}{V}$ by \obv{open_basis}.
        $\img{f}{U}\in C$.
        $y\in\img{f}{U}$ by \auto{img_of_function_intro,inter_intro}.
        $\img{f}{U}\subseteq V$ by
            \auto{subseteq,img_iff,preimg_iff,function_apply_iff}.
    \end{subproof}
    Follows by \auto{open_basis}.
\end{proof}
