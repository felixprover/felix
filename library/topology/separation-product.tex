\section{Hausdorff products}

\import{topology/separation-continuity.tex}
\import{topology/indexed-product.tex}
\import{topology/product.tex}

Two distinct points of a product differ in some coordinate. Separation in
that factor pulls back along its continuous projection. The argument needs
neither a finite index set nor inhabited factors.

\begin{theorem}\label{indexed_product_of_hausdorff_spaces}
    Suppose $X$ is a function on $I$.
    Suppose for every $i\in I$ we have $X(i)$ is Hausdorff.
    Then $\IndexedProductTopology{I}{X}$ is Hausdorff.
\end{theorem}
\begin{proof}
    Let $A=\indexedTopologicalCarrierFamily{I}{X}$ and $P=\IndexedProductTopology{I}{X}$.
    $P$ is a topological space by \auto{indexed_product_topology_is_topological_space}.
    $\carrier[P]=\dependentProduct{I}{A}$ by \auto{indexed_product_topology_carrier}.
    $A$ is a function on $I$ by \auto{indexed_topological_carrier_family_type}.
    For every $i\in I$ we have $X(i)$ is a topological space by \auto{hausdorff_is_topological}.
    Show that for all $x,y\in\carrier[P]$ such that $x\neq y$ there exist
        $U,V\in\opens[P]$ such that $x\in U$ and $y\in V$ and $U$ is disjoint from $V$.
    \begin{subproof}
        Fix $x,y\in\carrier[P]$.
        Assume $x\neq y$.
        $x,y\in\dependentProduct{I}{A}$.
        Hence $x,y\in\funs{I}{\unions{\ran{A}}}$ by \obv{dependent_product_iff}.
        Take $i\in I$ such that $x(i)\neq y(i)$ by
            \auto{typed_functions_equal_from_agreement}.
        Let $p=\dependentProjection{I}{A}{i}$.
        $p$ is continuous from $P$ to $X(i)$ by \auto{indexed_product_coordinate_continuous}.
        $p(x)=x(i)$ and $p(y)=y(i)$ by \auto{dependent_projection_apply}.
        Follows by \auto{continuous_map_separates_points}.
    \end{subproof}
    Follows by \auto{is_hausdorff}.
\end{proof}

\begin{theorem}\label{binary_product_of_hausdorff_spaces}
    Suppose $X$ is Hausdorff and $Y$ is Hausdorff.
    Then $\BinaryProductTopology{X}{Y}$ is Hausdorff.
\end{theorem}
\begin{proof}
    Let $P=\BinaryProductTopology{X}{Y}$.
    $X$ is a topological space and $Y$ is a topological space by \obv{is_hausdorff}.
    $P$ is a topological space by \auto{binary_product_topology_is_topological_space}.
    $\carrier[P]=\carrier[X]\times\carrier[Y]$ by \auto{binary_product_topology_carrier}.
    Show that for all $x,y\in\carrier[P]$ such that $x\neq y$ there exist
        $U,V\in\opens[P]$ such that $x\in U$ and $y\in V$ and $U$ is disjoint from $V$.
    \begin{subproof}
        Fix $x,y\in\carrier[P]$.
        Assume $x\neq y$.
        $x,y\in\carrier[X]\times\carrier[Y]$.
        $x=(\fst{x},\snd{x})$ and $y=(\fst{y},\snd{y})$ by
            \auto{times_elem_is_tuple,pair_eq_pair_of_proj}.
        \begin{byCase}
            \caseOf{$\fst{x}\neq\fst{y}$.}
                Let $p=\firstProductProjection{\carrier[X]}{\carrier[Y]}$.
                $p$ is continuous from $P$ to $X$ by \auto{first_product_projection_continuous}.
                $p(x)=\fst{x}$ and $p(y)=\fst{y}$ by \auto{first_product_projection_apply}.
                Follows by \auto{continuous_map_separates_points}.
            \caseOf{$\fst{x}=\fst{y}$.}
                Hence $\snd{x}\neq\snd{y}$.
                Let $p=\secondProductProjection{\carrier[X]}{\carrier[Y]}$.
                $p$ is continuous from $P$ to $Y$ by \auto{second_product_projection_continuous}.
                $p(x)=\snd{x}$ and $p(y)=\snd{y}$ by \auto{second_product_projection_apply}.
                Follows by \auto{continuous_map_separates_points}.
        \end{byCase}
    \end{subproof}
    Follows by \auto{is_hausdorff}.
\end{proof}
