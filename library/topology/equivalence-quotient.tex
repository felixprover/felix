\section{Topological quotients by equivalence relations}

\import{topology/quotient.tex}
\import{relation/quotient-lift.tex}

An equivalence relation on the carrier of a space supplies both the quotient
carrier and its canonical surjection.  The corresponding topological
quotient therefore needs no new map construction.

\begin{definition}\label{equivalence_quotient_topology}
    $\EquivalenceQuotientTopology{X}{E}
        =\QuotientTopology{X}{\quotient{\carrier[X]}{E}}{
            \quotientProjection{\carrier[X]}{E}}$.
\end{definition}

\begin{theorem}\label{equivalence_quotient_topology_is_topological_space}
    Let $X$ be a topological space.
    Suppose $E$ is an equivalence on $\carrier[X]$.
    Then $\EquivalenceQuotientTopology{X}{E}$ is a topological space.
\end{theorem}
\begin{proof}
    $\quotientProjection{\carrier[X]}{E}\in
        \funs{\carrier[X]}{\quotient{\carrier[X]}{E}}$ by
        \obv{quotient_projection_type}.
    Hence $\QuotientTopology{X}{\quotient{\carrier[X]}{E}}{
        \quotientProjection{\carrier[X]}{E}}$ is a topological space by
        \obv{quotient_topology_is_topological_space}.
    Follows by \obv{equivalence_quotient_topology}.
\end{proof}

\begin{theorem}\label{equivalence_quotient_topology_is_exact}
    Let $X$ be a topological space.
    Suppose $E$ is an equivalence on $\carrier[X]$.
    Then $\EquivalenceQuotientTopology{X}{E}$ is an exact topological
        space.
\end{theorem}
\begin{proof}
    $\quotientProjection{\carrier[X]}{E}\in
        \funs{\carrier[X]}{\quotient{\carrier[X]}{E}}$ by
        \obv{quotient_projection_type}.
    Hence $\QuotientTopology{X}{\quotient{\carrier[X]}{E}}{
        \quotientProjection{\carrier[X]}{E}}$
        is an exact topological space by
        \obv{quotient_topology_is_exact}.
    Follows by \auto{equivalence_quotient_topology}.
\end{proof}

\begin{proposition}\label{equivalence_quotient_topology_carrier}
    $\carrier[\EquivalenceQuotientTopology{X}{E}]
        =\quotient{\carrier[X]}{E}$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{equivalence_quotient_topology,quotient_topology_carrier}.
\end{proof}

\begin{proposition}\label{equivalence_quotient_topology_open_iff}
    $U\in\opens[\EquivalenceQuotientTopology{X}{E}]$ iff
        $U\subseteq\quotient{\carrier[X]}{E}$ and
        $\preimg{\quotientProjection{\carrier[X]}{E}}{U}
            \in\opens[X]$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{equivalence_quotient_topology,quotient_topology_open_iff}.
\end{proof}

\begin{theorem}\label{equivalence_quotient_projection_is_quotient_map}
    Let $X$ be a topological space.
    Suppose $E$ is an equivalence on $\carrier[X]$.
    Then $\quotientProjection{\carrier[X]}{E}$ is a quotient map from
        $X$ to $\EquivalenceQuotientTopology{X}{E}$.
\end{theorem}
\begin{proof}
    $\quotientProjection{\carrier[X]}{E}\in
        \surj{\carrier[X]}{\quotient{\carrier[X]}{E}}$ by
        \obv{quotient_projection_surjective}.
    Hence $\quotientProjection{\carrier[X]}{E}$ is a quotient map from
        $X$ to
        $\QuotientTopology{X}{\quotient{\carrier[X]}{E}}{
            \quotientProjection{\carrier[X]}{E}}$ by
        \obv{quotient_topology_structure_map_is_quotient_map}.
    Follows by \obv{equivalence_quotient_topology}.
\end{proof}

\begin{corollary}\label{equivalence_quotient_projection_continuous}
    Let $X$ be a topological space.
    Suppose $E$ is an equivalence on $\carrier[X]$.
    Then $\quotientProjection{\carrier[X]}{E}$ is continuous from $X$
        to $\EquivalenceQuotientTopology{X}{E}$.
\end{corollary}
\begin{proof}
    $\quotientProjection{\carrier[X]}{E}$ is a quotient map from $X$
        to $\EquivalenceQuotientTopology{X}{E}$ by
        \obv{equivalence_quotient_projection_is_quotient_map}.
    Follows by \obv{quotient_map_is_continuous}.
\end{proof}

\begin{theorem}\label{continuous_equivalence_quotient_descent}
    Let $X,Y$ be topological spaces.
    Suppose $E$ is an equivalence on $\carrier[X]$.
    Suppose $f$ is continuous from $X$ to $Y$.
    Suppose $\quotientCompatible{f}{\carrier[X]}{E}$.
    Then $\quotientDescent{f}{\carrier[X]}{E}$ is continuous from
        $\EquivalenceQuotientTopology{X}{E}$ to $Y$ and
        $\quotientDescent{f}{\carrier[X]}{E}\circ
            \quotientProjection{\carrier[X]}{E}=f$.
\end{theorem}
\begin{proof}
    $f\in\funs{\carrier[X]}{\carrier[Y]}$ by
        \obv{continuous_map}.
    Hence $\quotientDescent{f}{\carrier[X]}{E}\in
        \funs{\quotient{\carrier[X]}{E}}{\carrier[Y]}$ by
        \obv{quotient_descent_type}.
    $\quotientProjection{\carrier[X]}{E}\in
        \funs{\carrier[X]}{\quotient{\carrier[X]}{E}}$ by
        \obv{quotient_projection_type}.
    $\quotientDescent{f}{\carrier[X]}{E}\circ
        \quotientProjection{\carrier[X]}{E}=f$ by
        \obv{quotient_descent_factorization}.
    Hence $\quotientDescent{f}{\carrier[X]}{E}\circ
        \quotientProjection{\carrier[X]}{E}$ is continuous from $X$
        to $Y$.
    Hence $\quotientDescent{f}{\carrier[X]}{E}$ is continuous from
        $\QuotientTopology{X}{\quotient{\carrier[X]}{E}}{
            \quotientProjection{\carrier[X]}{E}}$ to $Y$ by
        \obv{quotient_topology_universal_property}.
    Hence $\quotientDescent{f}{\carrier[X]}{E}$ is continuous from
        $\EquivalenceQuotientTopology{X}{E}$ to $Y$ by
        \obv{equivalence_quotient_topology}.
\end{proof}
