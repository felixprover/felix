\section{Topological groups}

The multiplication axiom below is the usual continuity condition expressed
locally.  It avoids requiring a packaged product topology while retaining
joint, rather than merely separate, continuity.

\begin{definition}\label{left_translation_preimage}
    $\LeftTranslationPreimage{G}{a}{U}=
        \{x\in\carrier[G]\mid\mul[G](a,x)\in U\}$.
\end{definition}

\begin{definition}\label{right_translation_preimage}
    $\RightTranslationPreimage{G}{a}{U}=
        \{x\in\carrier[G]\mid\mul[G](x,a)\in U\}$.
\end{definition}

\begin{definition}\label{inverse_preimage}
    $\InversePreimage{G}{U}=
        \{x\in\carrier[G]\mid\groupInv[G](x)\in U\}$.
\end{definition}

\begin{struct}\label{topological_group}
    A topological group $G$ is a group and a topological space such that
    \begin{enumerate}
        \item\label{topological_group_multiplication}
            for all $x,y,U$ such that $x,y\in\carrier[G]$ and
                $U\in\opens[G]$ and $\mul[G](x,y)\in U$ there exist
                $V,W\in\opens[G]$ such that $x\in V$ and $y\in W$ and
                for all $v,w$ such that $v\in V$ and $w\in W$ we have
                    $\mul[G](v,w)\in U$.
        \item\label{topological_group_inverse}
            for every $U\in\opens[G]$ we have
                $\InversePreimage{G}{U}\in\opens[G]$.
    \end{enumerate}
\end{struct}

\begin{corollary}\label{topological_group_is_group}
    Let $G$ be a topological group.
    Then $G$ is a group.
\end{corollary}
\begin{proof}
    Follows by \obv{topological_group}.
\end{proof}

\begin{corollary}\label{topological_group_is_own_group_substructure}
    Let $G$ be a topological group.
    Then $\groupSubstructure{G}{G}$.
\end{corollary}
\begin{proof}
    Follows by
        \obv{topological_group_is_group,group_substructure_reflexive}.
\end{proof}

\begin{corollary}\label{topological_group_is_topological_space}
    Let $G$ be a topological group.
    Then $G$ is a topological space.
\end{corollary}
\begin{proof}
    Follows by \obv{topological_group}.
\end{proof}

\begin{theorem}\label{left_translation_preimage_open}
    Let $G$ be a topological group.
    Let $a\in\carrier[G]$.
    Let $U\in\opens[G]$.
    Then $\LeftTranslationPreimage{G}{a}{U}\in\opens[G]$.
\end{theorem}
\begin{proof}
    Let $P=\LeftTranslationPreimage{G}{a}{U}$.
    $P\subseteq\carrier[G]$ by \auto{left_translation_preimage_type}.
    Show that $P\subseteq\interior{P}{G}$.
    \begin{subproof}
        Show that for every $x\in P$ we have $x\in\interior{P}{G}$.
        \begin{subproof}
            Fix $x\in P$.
            $x\in\carrier[G]$ and $\mul[G](a,x)\in U$
                by \obv{left_translation_preimage}.
            Take $V,W\in\opens[G]$ such that $a\in V$ and $x\in W$
                and for all $v,w$ such that $v\in V$ and $w\in W$
                we have $\mul[G](v,w)\in U$
                by \obv{topological_group_multiplication}.
            Show that $W\subseteq P$.
            \begin{subproof}
                Show that for every $w\in W$ we have $w\in P$.
                \begin{subproof}
                    Fix $w\in W$.
                    $w\in\carrier[G]$
                        by \auto{topological_group_is_topological_space,opens_type,pow_iff,subseteq}.
                    $\mul[G](a,w)\in U$.
                    Follows by \obv{left_translation_preimage}.
                \end{subproof}
                Follows by \obv{subseteq}.
            \end{subproof}
            Follows by \obv{interior_elem_intro}.
        \end{subproof}
        Follows by \obv{subseteq}.
    \end{subproof}
    $\interior{P}{G}\subseteq P$ by \obv{interior_subseteq}.
    Hence $\interior{P}{G}=P$ by \obv{subseteq_antisymmetric}.
    Follows by
        \auto{topological_group_is_topological_space,interior_eq_self_implies_open}.
\end{proof}

\begin{lemma}\label{left_translation_set_preimage}
    Let $G$ be a group.
    Let $a\in\carrier[G]$.
    Then $\preimg{\LeftTranslation{G}{a}}{U}
        =\LeftTranslationPreimage{G}{a}{U}$.
\end{lemma}
\begin{proof}
    Show that for every $x\in\preimg{\LeftTranslation{G}{a}}{U}$ we have
        $x\in\LeftTranslationPreimage{G}{a}{U}$.
    \begin{subproof}
        Fix $x\in\preimg{\LeftTranslation{G}{a}}{U}$.
        Take $y\in U$ such that $(x,y)\in\LeftTranslation{G}{a}$
            by \obv{preimg_iff}.
        $x\in\carrier[G]$ and $y=\mul[G](a,x)$
            by \auto{left_translation_iff}.
        Follows by \obv{left_translation_preimage}.
    \end{subproof}
    Show that for every $x\in\LeftTranslationPreimage{G}{a}{U}$ we have
        $x\in\preimg{\LeftTranslation{G}{a}}{U}$.
    \begin{subproof}
        Fix $x\in\LeftTranslationPreimage{G}{a}{U}$.
        $x\in\carrier[G]$ and $\mul[G](a,x)\in U$
            by \obv{left_translation_preimage}.
        $(x,\mul[G](a,x))\in\LeftTranslation{G}{a}$
            by \obv{left_translation_iff}.
        Follows by \auto{preimg_iff}.
    \end{subproof}
    Follows by set extensionality.
\end{proof}

\begin{theorem}\label{left_translation_continuous}
    Let $G$ be a topological group.
    Let $a\in\carrier[G]$.
    Then $\LeftTranslation{G}{a}$ is continuous from $G$ to $G$.
\end{theorem}
\begin{proof}
    $\LeftTranslation{G}{a}\in
        \funs{\carrier[G]}{\carrier[G]}$
        by \obv{left_translation_function}.
    Show that for every $U\in\opens[G]$ we have
        $\preimg{\LeftTranslation{G}{a}}{U}\in\opens[G]$.
    \begin{subproof}
        Fix $U\in\opens[G]$.
        $\preimg{\LeftTranslation{G}{a}}{U}
            =\LeftTranslationPreimage{G}{a}{U}$
            by \obv{left_translation_set_preimage}.
        Follows by \auto{left_translation_preimage_open}.
    \end{subproof}
    Follows by \obv{continuous_map}.
\end{proof}

\begin{theorem}\label{left_translation_homeomorphism}
    Let $G$ be a topological group.
    Let $a\in\carrier[G]$.
    Then $\LeftTranslation{G}{a}$ is a homeomorphism from $G$ to $G$.
\end{theorem}
\begin{proof}
    $\LeftTranslation{G}{a}$ is continuous from $G$ to $G$
        by \obv{left_translation_continuous}.
    $\LeftTranslation{G}{a}\in
        \bijections{\carrier[G]}{\carrier[G]}$
        by \obv{left_translation_bijection}.
    $\groupInv[G](a)\in\carrier[G]$ by \obv{group_inverse_type}.
    Hence $\LeftTranslation{G}{\groupInv[G](a)}$ is continuous
        from $G$ to $G$ by \obv{left_translation_continuous}.
    $\converse{\LeftTranslation{G}{a}}
        =\LeftTranslation{G}{\groupInv[G](a)}$
        by \obv{left_translation_converse}.
    Follows by \obv{homeomorphism}.
\end{proof}

\subsection{Open subgroups}

\begin{theorem}\label{subgroup_open_iff_neighbourhood_of_neutral}
    Let $G$ be a topological group.
    Suppose $\subgroup{H}{G}$.
    Then $H$ is open in $G$ iff
        $H\in\neighbourhoods{\neutral[G]}{G}$.
\end{theorem}
\begin{proof}
    $G$ is a group by \obv{topological_group_is_group}.
    Hence $G$ is a monoid by \obv{group_is_monoid}.
    Hence $G$ is a unital magma by \obv{monoid_is_unital_magma}.
    $H\subseteq\carrier[G]$ and $\neutral[G]\in H$
        by \obv{subgroup_subseteq,subgroup_neutral}.

    Show that if $H$ is open in $G$, then
        $H\in\neighbourhoods{\neutral[G]}{G}$.
    \begin{subproof}
        Assume $H$ is open in $G$.
        Hence $H\in\opens[G]$ by \obv{is_open}.
        $H\in\pow{\carrier[G]}$ by \obv{powerset_intro}.
        Follows by \obv{neighbourhoods,subseteq_refl}.
    \end{subproof}

    Show that if $H\in\neighbourhoods{\neutral[G]}{G}$, then
        $H$ is open in $G$.
    \begin{subproof}
        Assume $H\in\neighbourhoods{\neutral[G]}{G}$.
        Take $U\in\opens[G]$ such that
            $\neutral[G]\in U\subseteq H$ by \obv{neighbourhoods}.
        Show that $H\subseteq\interior{H}{G}$.
        \begin{subproof}
            Show that for every $h\in H$ we have
                $h\in\interior{H}{G}$.
            \begin{subproof}
                Fix $h\in H$.
                Hence $h\in\carrier[G]$ by \obv{subseteq}.
                $\groupInv[G](h)\in\carrier[G]$
                    by \obv{group_inverse_type}.
                Let $P=\LeftTranslationPreimage{G}{\groupInv[G](h)}{U}$.
                $P\in\opens[G]$ by \auto{left_translation_preimage_open}.
                $\mul[G](\groupInv[G](h),h)=\neutral[G]$
                    by \obv{group_inverse_left}.
                Hence $h\in P$ by \obv{left_translation_preimage}.
                Show that $P\subseteq H$.
                \begin{subproof}
                    Show that for every $x\in P$ we have $x\in H$.
                    \begin{subproof}
                        Fix $x\in P$.
                        Hence $x\in\carrier[G]$ and
                            $\mul[G](\groupInv[G](h),x)\in U$
                            by \obv{left_translation_preimage}.
                        Hence $\mul[G](\groupInv[G](h),x)\in H$
                            by \obv{subseteq}.
                        Hence
                            $\mul[G](h,\mul[G](\groupInv[G](h),x))\in H$
                            by \obv{subgroup_mul}.
                        $\mul[G](h,\mul[G](\groupInv[G](h),x))=x$ by
                            \auto{monoid_assoc,group_inverse_right,
                                unitalmagma_left}.
                    \end{subproof}
                    Follows by \obv{subseteq}.
                \end{subproof}
                Follows by \obv{interior_elem_intro}.
            \end{subproof}
            Follows by \obv{subseteq}.
        \end{subproof}
        $\interior{H}{G}\subseteq H$ by \obv{interior_subseteq}.
        Hence $\interior{H}{G}=H$ by \obv{subseteq_antisymmetric}.
        Follows by \obv{interior_eq_self_implies_open}.
    \end{subproof}
    Follows by \auto.
\end{proof}
