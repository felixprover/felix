\section{Limits of functions}

\import{topology/continuity-within.tex}
\import{topology/net-convergence.tex}

The primitive limit interface is tendency from an explicitly displayed
source filter to a neighbourhood filter.  Limits at a point, within a set,
at a punctured point, and at the top of an order are specializations of this
single notion.

\begin{definition}\label{function_has_limit_along}
    $f$ has limit $y$ along $F$ from $S$ to $Y$ iff
        $Y$ is a topological space and
        $f\in\funs{S}{\carrier[Y]}$ and
        $y\in\carrier[Y]$ and
        $F$ is a filter on $S$ and
        $f$ tends from $F$ on $S$ to
            $\neighbourhoods{y}{Y}$ on $\carrier[Y]$.
\end{definition}

\begin{theorem}\label{function_has_limit_along_iff_preimages}
    Let $Y$ be a topological space.
    Let $f\in\funs{S}{\carrier[Y]}$, $y\in\carrier[Y]$, and
        $F$ be a filter on $S$.
    Then $f$ has limit $y$ along $F$ from $S$ to $Y$ iff
        for every $N\in\neighbourhoods{y}{Y}$ we have
        $\preimg{f}{N}\in F$.
\end{theorem}
\begin{proof}
    $\neighbourhoods{y}{Y}$ is a filter on $\carrier[Y]$ by
        \obv{neighbourhoods_is_filter}.
    Follows by
        \auto{function_has_limit_along,filter_tendsto_iff_preimage}.
\end{proof}

\begin{theorem}\label{continuous_at_preserves_function_limits}
    Suppose $f$ has limit $y$ along $F$ from $S$ to $Y$.
    Suppose $g$ is continuous at $y$ from $Y$ to $Z$.
    Then $g\circ f$ has limit $g(y)$ along $F$ from $S$ to $Z$.
\end{theorem}
\begin{proof}
    $f\in\funs{S}{\carrier[Y]}$ and
        $g\in\funs{\carrier[Y]}{\carrier[Z]}$ by
        \obv{function_has_limit_along,continuous_at}.
    $y\in\carrier[Y]$ and $F$ is a filter on $S$ by
        \obv{function_has_limit_along}.
    $g(y)\in\carrier[Z]$ by \obv{funs_type_apply}.
    $\neighbourhoods{y}{Y}$ is a filter on $\carrier[Y]$ by
        \obv{neighbourhoods_is_filter}.
    $\neighbourhoods{g(y)}{Z}$ is a filter on $\carrier[Z]$ by
        \obv{neighbourhoods_is_filter}.
    $f$ tends from $F$ on $S$ to
        $\neighbourhoods{y}{Y}$ on $\carrier[Y]$ by
        \obv{function_has_limit_along}.
    $g$ tends from $\neighbourhoods{y}{Y}$ on $\carrier[Y]$ to
        $\neighbourhoods{g(y)}{Z}$ on $\carrier[Z]$ by
        \obv{continuous_at}.
    Hence $g\circ f$ tends from $F$ on $S$ to
        $\neighbourhoods{g(y)}{Z}$ on $\carrier[Z]$ by
        \obv{composition_filter_tendsto}.
    $g\circ f\in\funs{S}{\carrier[Z]}$ by \obv{funs_circ}.
    Follows by \obv{function_has_limit_along}.
\end{proof}

\subsection{Limits at a point}

\begin{definition}\label{function_has_limit_at}
    $f$ has limit $y$ at $x$ from $X$ to $Y$ iff
        $X$ is a topological space and
        $x\in\carrier[X]$ and
        $f$ has limit $y$ along $\neighbourhoods{x}{X}$ from
            $\carrier[X]$ to $Y$.
\end{definition}

\begin{definition}\label{function_has_limit_within}
    $f$ has limit $y$ at $x$ within $A$ from $X$ to $Y$ iff
        $X$ is a topological space and
        $x\in\carrier[X]$ and
        $f$ has limit $y$ along $\neighbourhoodsWithin{x}{A}{X}$ from
            $\carrier[X]$ to $Y$.
\end{definition}

\begin{theorem}\label{continuous_at_iff_limit_at_self}
    Let $X,Y$ be topological spaces.
    Let $f\in\funs{\carrier[X]}{\carrier[Y]}$ and
        $x\in\carrier[X]$.
    Then $f$ is continuous at $x$ from $X$ to $Y$ iff
        $f$ has limit $f(x)$ at $x$ from $X$ to $Y$.
\end{theorem}
\begin{proof}
    $f(x)\in\carrier[Y]$ by \obv{funs_type_apply}.
    $\neighbourhoods{x}{X}$ is a filter on $\carrier[X]$ by
        \obv{neighbourhoods_is_filter}.
    Follows by
        \auto{continuous_at,function_has_limit_at,
            function_has_limit_along}.
\end{proof}

\begin{theorem}\label{continuous_at_within_iff_limit_within_self}
    Let $X,Y$ be topological spaces.
    Let $f\in\funs{\carrier[X]}{\carrier[Y]}$ and
        $x\in\carrier[X]$.
    Then $f$ is continuous at $x$ within $A$ from $X$ to $Y$ iff
        $f$ has limit $f(x)$ at $x$ within $A$ from $X$ to $Y$.
\end{theorem}
\begin{proof}
    $f(x)\in\carrier[Y]$ by \obv{funs_type_apply}.
    $\neighbourhoodsWithin{x}{A}{X}$ is a filter on $\carrier[X]$ by
        \obv{neighbourhoods_within_is_filter}.
    Follows by
        \auto{continuous_at_within,function_has_limit_within,
            function_has_limit_along}.
\end{proof}

\begin{theorem}\label{function_limit_within_carrier_iff_limit_at}
    Let $X,Y$ be topological spaces.
    Let $f\in\funs{\carrier[X]}{\carrier[Y]}$.
    Let $x\in\carrier[X]$ and $y\in\carrier[Y]$.
    Then $f$ has limit $y$ at $x$ within $\carrier[X]$ from $X$ to $Y$
        iff $f$ has limit $y$ at $x$ from $X$ to $Y$.
\end{theorem}
\begin{proof}
    $\neighbourhoodsWithin{x}{\carrier[X]}{X}
        =\neighbourhoods{x}{X}$ by
        \obv{neighbourhoods_within_carrier}.
    Follows by
        \auto{function_has_limit_within,function_has_limit_at}.
\end{proof}

\begin{definition}\label{function_has_punctured_limit}
    $f$ has punctured limit $y$ at $x$ from $X$ to $Y$ iff
        $f$ has limit $y$ at $x$ within
            $\carrier[X]\setminus\{x\}$ from $X$ to $Y$.
\end{definition}

\subsection{Exact-domain limits}

For a graph defined only on the carrier trace of $A$, the source filter is
the neighbourhood filter in the subspace.  Thus the graph is total on the
carrier of its source filter, and no ambient extension is involved.

\begin{definition}\label{partial_function_has_limit_at}
    $f$ has partial limit $y$ at $x$ on $A$ from $X$ to $Y$ iff
        $X$ is a topological space and
        $f\in\funs{A\inter\carrier[X]}{\carrier[Y]}$ and
        $x\in A\inter\carrier[X]$ and
        $f$ has limit $y$ along
            $\neighbourhoods{x}{\subspaceTopology{X}{A}}$ from
            $A\inter\carrier[X]$ to $Y$.
\end{definition}

\begin{theorem}\label{function_limit_within_iff_restriction_partial_limit}
    Let $X,Y$ be topological spaces.
    Let $f\in\funs{\carrier[X]}{\carrier[Y]}$,
        $x\in A\inter\carrier[X]$, and $y\in\carrier[Y]$.
    Then $f$ has limit $y$ at $x$ within $A$ from $X$ to $Y$ iff
        $\restrl{f}{A\inter\carrier[X]}$ has partial limit $y$ at $x$
            on $A$ from $X$ to $Y$.
\end{theorem}
\begin{proof}
    $x\in\carrier[X]$ by \obv{inter}.
    $\subspaceTopology{X}{A}$ is a topological space by
        \obv{subspace_topology_is_topological_space}.
    $\carrier[\subspaceTopology{X}{A}]=A\inter\carrier[X]$ by
        \obv{subspace_topology_carrier}.
    $\restrl{f}{A\inter\carrier[X]}
        \in\funs{A\inter\carrier[X]}{\carrier[Y]}$ by
        \obv{funs_restrl}.
    $\subspaceInclusion{X}{A}\in
        \funs{A\inter\carrier[X]}{\carrier[X]}$ by
        \obv{subspace_inclusion_type}.
    $\neighbourhoods{x}{\subspaceTopology{X}{A}}$ is a filter on
        $A\inter\carrier[X]$ by \obv{neighbourhoods_is_filter}.
    $\neighbourhoodsWithin{x}{A}{X}$ is a filter on $\carrier[X]$ by
        \obv{neighbourhoods_within_is_filter}.
    $\neighbourhoods{y}{Y}$ is a filter on $\carrier[Y]$ by
        \obv{neighbourhoods_is_filter}.
    $\filterPushforward{A\inter\carrier[X]}{\carrier[X]}{
        \subspaceInclusion{X}{A}}{
        \neighbourhoods{x}{\subspaceTopology{X}{A}}}
        =\neighbourhoodsWithin{x}{A}{X}$ by
        \obv{subspace_neighbourhood_pushforward_is_within}.
    $f$ tends from
        $\filterPushforward{A\inter\carrier[X]}{\carrier[X]}{
            \subspaceInclusion{X}{A}}{
            \neighbourhoods{x}{\subspaceTopology{X}{A}}}$ on
        $\carrier[X]$ to $\neighbourhoods{y}{Y}$ on $\carrier[Y]$ iff
        $f\circ\subspaceInclusion{X}{A}$ tends from
            $\neighbourhoods{x}{\subspaceTopology{X}{A}}$ on
            $A\inter\carrier[X]$ to
            $\neighbourhoods{y}{Y}$ on $\carrier[Y]$ by
        \obv{filter_tendsto_from_pushforward_iff_composition}.
    $\restrl{f}{A\inter\carrier[X]}
        =f\circ\subspaceInclusion{X}{A}$ by
        \obv{restriction_is_composition_with_subspace_inclusion}.
    Hence $f$ tends from $\neighbourhoodsWithin{x}{A}{X}$ on
        $\carrier[X]$ to $\neighbourhoods{y}{Y}$ on $\carrier[Y]$ iff
        $\restrl{f}{A\inter\carrier[X]}$ tends from
            $\neighbourhoods{x}{\subspaceTopology{X}{A}}$ on
            $A\inter\carrier[X]$ to
            $\neighbourhoods{y}{Y}$ on $\carrier[Y]$.
    $f$ has limit $y$ at $x$ within $A$ from $X$ to $Y$ iff
        $f$ tends from $\neighbourhoodsWithin{x}{A}{X}$ on
            $\carrier[X]$ to $\neighbourhoods{y}{Y}$ on $\carrier[Y]$ by
        \auto{function_has_limit_within,function_has_limit_along}.
    $\restrl{f}{A\inter\carrier[X]}$ has partial limit $y$ at $x$ on
        $A$ from $X$ to $Y$ iff
        $\restrl{f}{A\inter\carrier[X]}$ tends from
            $\neighbourhoods{x}{\subspaceTopology{X}{A}}$ on
            $A\inter\carrier[X]$ to
            $\neighbourhoods{y}{Y}$ on $\carrier[Y]$ by
        \auto{partial_function_has_limit_at,function_has_limit_along}.
\end{proof}

\subsection{Limits at the top}

\begin{definition}\label{function_has_limit_at_top}
    $f$ has limit $y$ at the top of $I$ in $Y$ iff
        $I$ is a directed quasiordered set and
        $f$ has limit $y$ along $\atTop{I}$ from $\carrier[I]$ to $Y$.
\end{definition}

\begin{theorem}\label{net_convergence_iff_limit_at_top}
    Let $I$ be a directed quasiordered set and $X$ be a topological space.
    Let $s\in\funs{\carrier[I]}{\carrier[X]}$ and
        $x\in\carrier[X]$.
    Then $s$ converges topologically to $x$ in $X$ along $I$ iff
        $s$ has limit $x$ at the top of $I$ in $X$.
\end{theorem}
\begin{proof}
    $\atTop{I}$ is a filter on $\carrier[I]$ by
        \obv{at_top_is_filter}.
    $\neighbourhoods{x}{X}$ is a filter on $\carrier[X]$ by
        \obv{neighbourhoods_is_filter}.
    Follows by
        \auto{topological_net_convergence,function_has_limit_at_top,
            function_has_limit_along}.
\end{proof}
