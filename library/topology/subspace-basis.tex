\section{Bases of subspaces}

\import{topology/subspace.tex}
\import{topology/basis.tex}

An open basis restricts to a subspace by taking traces on the subspace
carrier.  The definition is hosted on that carrier, so it remains valid for
an arbitrary set rather than only for a subset already known to lie in the
ambient carrier.

\begin{definition}\label{subspace_basis}
    $\subspaceBasis{X}{A}{B}
        =\{V\in\pow{\carrier[\subspaceTopology{X}{A}]}\mid
            \text{there exists $U\in B$ such that
                $V=U\inter\carrier[\subspaceTopology{X}{A}]$}\}$.
\end{definition}

\begin{proposition}\label{subspace_basis_iff}
    $V\in\subspaceBasis{X}{A}{B}$ iff
        $V\subseteq\carrier[\subspaceTopology{X}{A}]$ and
        there exists $U\in B$ such that
            $V=U\inter\carrier[\subspaceTopology{X}{A}]$.
\end{proposition}
\begin{proof}
    Follows by \auto{subspace_basis,pow_iff,subseteq}.
\end{proof}

\begin{proposition}\label{subspace_basis_type}
    $\subspaceBasis{X}{A}{B}
        \subseteq\pow{\carrier[\subspaceTopology{X}{A}]}$.
\end{proposition}
\begin{proof}
    Follows by \obv{subspace_basis,subseteq}.
\end{proof}

\begin{theorem}\label{open_basis_restricts_to_subspace}
    Let $X$ be a topological space.
    Suppose $B$ is an open basis of $X$.
    Then $\subspaceBasis{X}{A}{B}$ is an open basis of
        $\subspaceTopology{X}{A}$.
\end{theorem}
\begin{proof}
    $\subspaceTopology{X}{A}$ is a topological space by
        \obv{subspace_topology_is_topological_space}.
    Let $S=\subspaceTopology{X}{A}$.
    $B\subseteq\opens[X]$ by \obv{open_basis}.
    Show that every element of $\subspaceBasis{X}{A}{B}$ is an element
        of $\opens[S]$.
    \begin{subproof}
        Fix $V\in\subspaceBasis{X}{A}{B}$.
        Take $U\in B$ such that
            $V=U\inter\carrier[\subspaceTopology{X}{A}]$ by
            \obv{subspace_basis}.
        Hence $U\in\opens[X]$ by \obv{subseteq}.
        $\carrier[S]=\carrier[\subspaceTopology{X}{A}]$.
        Hence $V=U\inter\carrier[S]$.
        Follows by \auto{subspace_open_iff}.
    \end{subproof}
    Hence $\subspaceBasis{X}{A}{B}\subseteq\opens[S]$ by
        \obv{subseteq}.

    Show that for all $V,x$ such that $V\in\opens[S]$ and $x\in V$
        there exists $W\in\subspaceBasis{X}{A}{B}$ such that
            $x\in W\subseteq V$.
    \begin{subproof}
        Fix $V,x$.
        Assume $V\in\opens[S]$ and $x\in V$.
        Hence $V\in\opens[\subspaceTopology{X}{A}]$.
        Hence $V\in\subspaceOpens{X}{A}$ by
            \obv{subspace_topology_opens}.
        Take $O\in\opens[X]$ such that
            $V=O\inter(A\inter\carrier[X])$ by
            \obv{subspace_opens_iff}.
        $\carrier[\subspaceTopology{X}{A}]=A\inter\carrier[X]$ by
            \obv{subspace_topology_carrier}.
        Hence $V=O\inter\carrier[\subspaceTopology{X}{A}]$.
        $\carrier[S]=\carrier[\subspaceTopology{X}{A}]$.
        Hence $V=O\inter\carrier[S]$.
        Hence $x\in O$ and $x\in\carrier[S]$ by \auto{inter}.
        Take $U\in B$ such that $x\in U\subseteq O$ by
            \obv{open_basis}.
        Let $W=U\inter\carrier[S]$.
        $U\inter\carrier[S]\subseteq\carrier[S]$ by
            \obv{inter_lower_right}.
        Hence $W\subseteq\carrier[S]$.
        $\carrier[S]=\carrier[\subspaceTopology{X}{A}]$.
        Hence $W\subseteq\carrier[\subspaceTopology{X}{A}]$.
        Hence $W=U\inter\carrier[\subspaceTopology{X}{A}]$.
        Hence $W\in\subspaceBasis{X}{A}{B}$ by
            \auto{subspace_basis_iff}.
        $x\in W$ by \obv{inter}.
        $W\subseteq V$ by \auto{inter,subseteq}.
    \end{subproof}
    Hence $\subspaceBasis{X}{A}{B}$ is an open basis of $S$ by
        \obv{open_basis}.
    $S=\subspaceTopology{X}{A}$.
\end{proof}
