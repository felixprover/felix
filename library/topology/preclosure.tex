\section{Preclosure spaces}

\begin{struct}\label{preclosure_space}
    A preclosure space $X$ is a onesorted structure equipped with
    \begin{enumerate}
        \item\label{preclosure_graph}
            a preclosure operation
            $\cl[X]:\pow{\carrier[X]}\to\pow{\carrier[X]}$;
    \end{enumerate}
    such that
    \begin{enumerate}
        \item\label{preclosure_empty}
            $\cl[X](\emptyset)=\emptyset$.
        \item\label{preclosure_extensive}
            for every $A\subseteq\carrier[X]$ we have
                $A\subseteq\cl[X](A)$.
        \item\label{preclosure_union}
            for all $A,B\subseteq\carrier[X]$ we have
                $\cl[X](A\union B)=\cl[X](A)\union\cl[X](B)$.
    \end{enumerate}
\end{struct}

\begin{proposition}\label{preclosure_type}
    Let $X$ be a preclosure space and
        $Y$ be a subset of $\carrier[X]$.
    Then $\cl[X](Y)\subseteq\carrier[X]$.
\end{proposition}
\begin{proof}
    $Y\in\pow{\carrier[X]}$ by \obv{subseteq,pow_iff}.
    Hence $\cl[X](Y)\in\pow{\carrier[X]}$
        by \obv{preclosure_graph,funs_type_apply}.
    Follows by \obv{subseteq,pow_iff}.
\end{proof}
