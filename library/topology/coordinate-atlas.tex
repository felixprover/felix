\section{Coordinate atlases}

\import{topology/coordinate-chart.tex}
\import{topology/full-pseudogroup.tex}

On a space already equipped with a topology, a chart is an open partial
homeomorphism into the model. Compatibility asks that its transition to
another chart be an allowed coordinate change. Compatibility is symmetric,
but it need not be transitive: passing through a third chart sees only the
part of the overlap covered by that chart.

\begin{definition}\label{coordinate_chart_compatibility}
    Let $G$ be a coordinate pseudogroup and $X$ be a topological space.
    $c$ is compatible with $d$ on $X$ under $G$ iff
        $c$ is an open partial homeomorphism from $X$ to $\model[G]$ and
        $d$ is an open partial homeomorphism from $X$ to $\model[G]$ and
        $\chartTransition{d}{c}\in\changes[G]$.
\end{definition}

\begin{theorem}\label{coordinate_chart_compatibility_reflexive}
    Let $G$ be a coordinate pseudogroup and $X$ be a topological space.
    Suppose $c$ is an open partial homeomorphism from $X$ to $\model[G]$.
    Then $c$ is compatible with $c$ on $X$ under $G$.
\end{theorem}
\begin{proof}
    $c$ is a raw coordinate chart on $\carrier[X]$ modelled on $\model[G]$ by
        \obv{open_partial_homeomorphism_is_raw_chart}.
    $\chartTransition{c}{c}=\identity{\ran{c}}$ by \obv{chart_transition_self}.
    $\ran{c}\in\opens[\model[G]]$ by \obv{open_partial_homeomorphism}.
    $\identity{\ran{c}}\in\changes[G]$ by \obv{pseudogroup_open_identity}.
    Follows by \obv{coordinate_chart_compatibility}.
\end{proof}

\begin{theorem}\label{coordinate_chart_compatibility_symmetric}
    Let $G$ be a coordinate pseudogroup and $X$ be a topological space.
    Suppose $c$ is compatible with $d$ on $X$ under $G$.
    Then $d$ is compatible with $c$ on $X$ under $G$.
\end{theorem}
\begin{proof}
    $c$ is an open partial homeomorphism from $X$ to $\model[G]$ and
        $d$ is an open partial homeomorphism from $X$ to $\model[G]$ and
        $\chartTransition{d}{c}\in\changes[G]$ by
        \obv{coordinate_chart_compatibility}.
    $c$ and $d$ are raw coordinate charts on $\carrier[X]$ modelled on
        $\model[G]$ by \obv{open_partial_homeomorphism_is_raw_chart}.
    $\converse{\chartTransition{d}{c}}=\chartTransition{c}{d}$ by
        \obv{chart_transition_converse}.
    $\converse{\chartTransition{d}{c}}\in\changes[G]$ by \obv{pseudogroup_converse}.
    Follows by \obv{coordinate_chart_compatibility}.
\end{proof}

\begin{theorem}\label{coordinate_compatibility_restrict_left}
    Let $G$ be a coordinate pseudogroup and $X$ be a topological space.
    Suppose $c$ is compatible with $d$ on $X$ under $G$.
    Let $U\in\opens[X]$.
    Then $\restrl{c}{U}$ is compatible with $d$ on $X$ under $G$.
\end{theorem}
\begin{proof}
    $c$ is an open partial homeomorphism from $X$ to $\model[G]$ and
        $d$ is an open partial homeomorphism from $X$ to $\model[G]$ and
        $\chartTransition{d}{c}\in\changes[G]$ by \obv{coordinate_chart_compatibility}.
    $\restrl{c}{U}$ is an open partial homeomorphism from $X$ to $\model[G]$ by
        \obv{open_partial_homeomorphism_restriction}.
    $\img{c}{U}\in\opens[\model[G]]$ by \obv{open_partial_homeomorphism_image_open}.
    $\restrl{\chartTransition{d}{c}}{\img{c}{U}}\in\changes[G]$ by \obv{pseudogroup_restriction}.
    $c$ is an injection and $d$ is a function by
        \auto{open_partial_homeomorphism_bijection,bijections,bijection_is_function}.
    $\chartTransition{d}{\restrl{c}{U}}=\restrl{\chartTransition{d}{c}}{\img{c}{U}}$ by
        \obv{chart_transition_source_restriction}.
    Follows by \obv{coordinate_chart_compatibility}.
\end{proof}

\begin{theorem}\label{coordinate_compatibility_restrict_right}
    Let $G$ be a coordinate pseudogroup and $X$ be a topological space.
    Suppose $c$ is compatible with $d$ on $X$ under $G$.
    Let $U\in\opens[X]$.
    Then $c$ is compatible with $\restrl{d}{U}$ on $X$ under $G$.
\end{theorem}
\begin{proof}
    $d$ is compatible with $c$ on $X$ under $G$ by \obv{coordinate_chart_compatibility_symmetric}.
    $\restrl{d}{U}$ is compatible with $c$ on $X$ under $G$ by
        \obv{coordinate_compatibility_restrict_left}.
    Follows by \obv{coordinate_chart_compatibility_symmetric}.
\end{proof}

\begin{theorem}\label{coordinate_compatibility_restriction}
    Let $G$ be a coordinate pseudogroup.
    Let $X$ be a topological space.
    Suppose $c$ is compatible with $d$ on $X$ under $G$.
    Suppose $U\in\opens[X]$ and $U\subseteq\dom{c}$.
    Then $\restrl{c}{U}$ is compatible with $\restrl{d}{U}$ on $X$ under $G$.
\end{theorem}
\begin{proof}
    $c$ is an open partial homeomorphism from $X$ to $\model[G]$ and
        $d$ is an open partial homeomorphism from $X$ to $\model[G]$ and
        $\chartTransition{d}{c}\in\changes[G]$ by \obv{coordinate_chart_compatibility}.
    $\restrl{c}{U}$ is an open partial homeomorphism from $X$ to $\model[G]$ and
        $\restrl{d}{U}$ is an open partial homeomorphism from $X$ to $\model[G]$ by
        \obv{open_partial_homeomorphism_restriction}.
    $\img{c}{U}\in\opens[\model[G]]$ by \obv{open_partial_homeomorphism_image_open}.
    $\restrl{\chartTransition{d}{c}}{\img{c}{U}}\in\changes[G]$ by \obv{pseudogroup_restriction}.
    $c$ is an injection and $d$ is a function by
        \auto{open_partial_homeomorphism_bijection,bijections,bijection_is_function}.
    $\chartTransition{\restrl{d}{U}}{\restrl{c}{U}}
        =\restrl{\chartTransition{d}{c}}{\img{c}{U}}$ by \obv{chart_transition_restriction}.
    Follows by \obv{coordinate_chart_compatibility}.
\end{proof}

\begin{theorem}\label{coordinate_compatibility_open_subspace}
    Let $G$ be a coordinate pseudogroup and $X$ be a topological space.
    Let $U\in\opens[X]$.
    Then $c$ is compatible with $d$ on $\subspaceTopology{X}{U}$ under $G$ iff
        $c$ is compatible with $d$ on $X$ under $G$ and
        $\dom{c}\subseteq U$ and $\dom{d}\subseteq U$.
\end{theorem}
\begin{proof}
    $\subspaceTopology{X}{U}$ is a topological space by \obv{subspace_topology_is_topological_space}.
    For every $h$ we have
        $h$ is an open partial homeomorphism from $\subspaceTopology{X}{U}$ to $\model[G]$ iff
        $h$ is an open partial homeomorphism from $X$ to $\model[G]$ and $\dom{h}\subseteq U$ by
        \obv{open_partial_homeomorphism_open_subspace}.
    Follows by \auto{coordinate_chart_compatibility}.
\end{proof}

\begin{definition}\label{coordinate_atlas}
    Let $G$ be a coordinate pseudogroup and $X$ be a topological space.
    $A$ is a coordinate atlas on $X$ under $G$ iff
        (every element of $A$ is an open partial homeomorphism from $X$ to $\model[G]$) and
        (for all $c,d\in A$ we have $c$ is compatible with $d$ on $X$ under $G$) and
        (for every $x\in\carrier[X]$ there exists $c\in A$ such that $x\in\dom{c}$).
\end{definition}

\begin{theorem}\label{single_chart_atlas}
    Let $G$ be a coordinate pseudogroup and $X$ be a topological space.
    Suppose $c$ is an open partial homeomorphism from $X$ to $\model[G]$.
    Suppose $\dom{c}=\carrier[X]$.
    Then $\{c\}$ is a coordinate atlas on $X$ under $G$.
\end{theorem}
\begin{proof}
    Every element of $\{c\}$ is an open partial homeomorphism from $X$ to $\model[G]$
        by \auto{singleton_iff}.
    $c$ is compatible with $c$ on $X$ under $G$ by
        \obv{coordinate_chart_compatibility_reflexive}.
    For all $a,b\in\{c\}$ we have $a$ is compatible with $b$ on $X$ under $G$ by
        \auto{singleton_iff}.
    For every $x\in\carrier[X]$ there exists $a\in\{c\}$ such that $x\in\dom{a}$ by
        \auto{singleton_iff}.
    Follows by \obv{coordinate_atlas}.
\end{proof}

\begin{theorem}\label{coordinate_compatibility_from_cover}
    Let $G$ be a coordinate pseudogroup and $X$ be a topological space.
    Suppose $c$ is an open partial homeomorphism from $X$ to $\model[G]$ and
        $d$ is an open partial homeomorphism from $X$ to $\model[G]$.
    Suppose for every $x\in\carrier[X]$ there exists $a\in A$ such that $x\in\dom{a}$.
    Suppose for every $a\in A$ we have $c$ is compatible with $a$ on $X$ under $G$
        and $a$ is compatible with $d$ on $X$ under $G$.
    Then $c$ is compatible with $d$ on $X$ under $G$.
\end{theorem}
\begin{proof}
    Let $E=\model[G]$ and $h=\chartTransition{d}{c}$.
    $c$ and $d$ are raw coordinate charts on $\carrier[X]$ modelled on $E$ by
        \obv{open_partial_homeomorphism_is_raw_chart}.
    $c$ and $d$ are injections and $\dom{c}\subseteq\carrier[X]$ by
        \obv{raw_coordinate_chart}.
    $\converse{c}$ is an open partial homeomorphism from $E$ to $X$ by
        \obv{open_partial_homeomorphism_converse}.
    $d\circ\converse{c}$ is an open partial homeomorphism from $E$ to $E$ by
        \obv{open_partial_homeomorphism_composition}.
    $h=d\circ\converse{c}$ by \obv{chart_transition}.
    $\dom{h}=\img{c}{\dom{c}\inter\dom{d}}$ by \auto{chart_transition_domain}.
    Show that for every $y\in\dom{h}$ there exists $U\in\opens[E]$ such that
        $y\in U$ and $\restrl{h}{U}\in\changes[G]$.
    \begin{subproof}
        Fix $y\in\dom{h}$.
        Take $x\in\dom{c}\inter\dom{d}$ such that $(x,y)\in c$ by \auto{img_iff}.
        $x\in\carrier[X]$ by \auto{inter,subseteq}.
        Take $a\in A$ such that $x\in\dom{a}$.
        $c$ is compatible with $a$ on $X$ under $G$ and
            $a$ is compatible with $d$ on $X$ under $G$.
        $a$ is an open partial homeomorphism from $X$ to $E$ and
            $\chartTransition{a}{c}\in\changes[G]$ and
            $\chartTransition{d}{a}\in\changes[G]$ by
            \obv{coordinate_chart_compatibility}.
        $\dom{a}\in\opens[X]$ by \obv{open_partial_homeomorphism}.
        Let $U=\img{c}{\dom{a}}$.
        $U\in\opens[E]$ by \auto{open_partial_homeomorphism_image_open}.
        $y\in U$ by \auto{img_iff}.
        $a\in\bijections{\dom{a}}{\ran{a}}$ by
            \obv{open_partial_homeomorphism_bijection}.
        $a$ is an injection by \auto{bijections,bijection_is_function}.
        $(d\circ\converse{a})\circ(a\circ\converse{c})
            =\restrl{d\circ\converse{c}}{U}$ by \auto{transition_factorization}.
        $\chartTransition{d}{a}\circ\chartTransition{a}{c}
            =\restrl{h}{U}$ by \auto{chart_transition}.
        $\chartTransition{d}{a}\circ\chartTransition{a}{c}\in\changes[G]$ by
            \obv{pseudogroup_composition}.
        Follows by \auto.
    \end{subproof}
    $h\in\changes[G]$ by \auto{pseudogroup_change_iff_locally}.
    Follows by \obv{coordinate_chart_compatibility}.
\end{proof}

\begin{theorem}\label{subpseudogroup_compatibility}
    Let $G,H$ be coordinate pseudogroups and $X$ be a topological space.
    Suppose $G$ is a subpseudogroup of $H$.
    Suppose $c$ is compatible with $d$ on $X$ under $G$.
    Then $c$ is compatible with $d$ on $X$ under $H$.
\end{theorem}
\begin{proof}
    $\model[G]=\model[H]$ and $\changes[G]\subseteq\changes[H]$ by
        \obv{subpseudogroup}.
    $c$ is an open partial homeomorphism from $X$ to $\model[G]$ and
        $d$ is an open partial homeomorphism from $X$ to $\model[G]$ and
        $\chartTransition{d}{c}\in\changes[G]$ by \obv{coordinate_chart_compatibility}.
    $\chartTransition{d}{c}\in\changes[H]$ by \obv{subseteq}.
    Follows by \obv{coordinate_chart_compatibility}.
\end{proof}

\begin{classification}\label{subpseudogroup_atlas}
    Let $G,H$ be coordinate pseudogroups and $X$ be a topological space.
    Suppose $G$ is a subpseudogroup of $H$.
    Suppose $A$ is a coordinate atlas on $X$ under $G$.
    Then $A$ is a coordinate atlas on $X$ under $H$.
\end{classification}
\begin{proof}
    $\model[G]=\model[H]$ by \obv{subpseudogroup}.
    Every element of $A$ is an open partial homeomorphism from $X$ to $\model[H]$ by
        \auto{coordinate_atlas}.
    For all $c,d\in A$ we have $c$ is compatible with $d$ on $X$ under $G$ by
        \obv{coordinate_atlas}.
    For all $c,d\in A$ we have $c$ is compatible with $d$ on $X$ under $H$ by
        \obv{subpseudogroup_compatibility}.
    For every $x\in\carrier[X]$ there exists $c\in A$ such that $x\in\dom{c}$ by
        \obv{coordinate_atlas}.
    Follows by \obv{coordinate_atlas}.
\end{proof}

\subsection{The compatible hull}

The compatible hull of an arbitrary family is antitone: more compatibility
tests admit fewer charts. For a covering compatible family, however, locality
makes its hull an atlas and hence its unique maximal extension.

\begin{definition}\label{compatible_hull}
    Let $G$ be a coordinate pseudogroup and $X$ be a topological space.
    $\compatibleHull{G}{X}{A}=\{c\in\openPartialHomeomorphisms{X}{\model[G]}\mid
        \text{for every $a\in A$ we have $c$ is compatible with $a$ on $X$ under $G$}\}$.
\end{definition}

\begin{theorem}\label{compatible_hull_iff}
    Let $G$ be a coordinate pseudogroup and $X$ be a topological space.
    Then $c\in\compatibleHull{G}{X}{A}$ iff
        $c$ is an open partial homeomorphism from $X$ to $\model[G]$ and
        for every $a\in A$ we have $c$ is compatible with $a$ on $X$ under $G$.
\end{theorem}
\begin{proof}
    Follows by \auto{compatible_hull,open_partial_homeomorphisms_iff}.
\end{proof}

\begin{theorem}\label{compatible_hull_antitone}
    Let $G$ be a coordinate pseudogroup and $X$ be a topological space.
    Suppose $A\subseteq B$.
    Then $\compatibleHull{G}{X}{B}\subseteq\compatibleHull{G}{X}{A}$.
\end{theorem}
\begin{proof}
    Follows by \auto{compatible_hull_iff,subseteq}.
\end{proof}

\begin{theorem}\label{atlas_subset_compatible_hull}
    Let $G$ be a coordinate pseudogroup and $X$ be a topological space.
    Suppose $A$ is a coordinate atlas on $X$ under $G$.
    Then $A\subseteq\compatibleHull{G}{X}{A}$.
\end{theorem}
\begin{proof}
    Follows by \auto{coordinate_atlas,compatible_hull_iff,subseteq}.
\end{proof}

\begin{theorem}\label{compatible_hull_pairwise}
    Let $G$ be a coordinate pseudogroup and $X$ be a topological space.
    Suppose $A$ is a coordinate atlas on $X$ under $G$.
    Let $c,d\in\compatibleHull{G}{X}{A}$.
    Then $c$ is compatible with $d$ on $X$ under $G$.
\end{theorem}
\begin{proof}
    $c$ is an open partial homeomorphism from $X$ to $\model[G]$ and
        $d$ is an open partial homeomorphism from $X$ to $\model[G]$ by
        \auto{compatible_hull_iff}.
    For every $x\in\carrier[X]$ there exists $a\in A$ such that $x\in\dom{a}$ by
        \obv{coordinate_atlas}.
    For every $a\in A$ we have $c$ is compatible with $a$ on $X$ under $G$ and
        $d$ is compatible with $a$ on $X$ under $G$ by \auto{compatible_hull_iff}.
    For every $a\in A$ we have $c$ is compatible with $a$ on $X$ under $G$ and
        $a$ is compatible with $d$ on $X$ under $G$ by
        \auto{coordinate_chart_compatibility_symmetric}.
    Follows by \obv{coordinate_compatibility_from_cover}.
\end{proof}

\begin{theorem}\label{compatible_hull_is_atlas}
    Let $G$ be a coordinate pseudogroup and $X$ be a topological space.
    Suppose $A$ is a coordinate atlas on $X$ under $G$.
    Then $\compatibleHull{G}{X}{A}$ is a coordinate atlas on $X$ under $G$.
\end{theorem}
\begin{proof}
    Let $B=\compatibleHull{G}{X}{A}$.
    Every element of $B$ is an open partial homeomorphism from $X$ to $\model[G]$
        by \auto{compatible_hull_iff}.
    For all $c,d\in B$ we have $c$ is compatible with $d$ on $X$ under $G$ by
        \auto{compatible_hull_pairwise}.
    $A\subseteq B$ by \obv{atlas_subset_compatible_hull}.
    For every $x\in\carrier[X]$ there exists $c\in A$ such that $x\in\dom{c}$ by
        \obv{coordinate_atlas}.
    For every $x\in\carrier[X]$ there exists $c\in B$ such that $x\in\dom{c}$ by
        \auto{subseteq}.
    Follows by \auto{coordinate_atlas}.
\end{proof}

\begin{definition}\label{max_atlas}
    Let $G$ be a coordinate pseudogroup and $X$ be a topological space.
    Let $A$ be a coordinate atlas on $X$ under $G$.
    $\MaxAtlas{G}{X}{A}=\compatibleHull{G}{X}{A}$.
\end{definition}

\begin{definition}\label{maximal_coordinate_atlas}
    Let $G$ be a coordinate pseudogroup and $X$ be a topological space.
    $A$ is a maximal coordinate atlas on $X$ under $G$ iff
        $A$ is a coordinate atlas on $X$ under $G$ and
        $\compatibleHull{G}{X}{A}\subseteq A$.
\end{definition}

\begin{theorem}\label{compatible_hull_idempotent}
    Let $G$ be a coordinate pseudogroup and $X$ be a topological space.
    Suppose $A$ is a coordinate atlas on $X$ under $G$.
    Then $\compatibleHull{G}{X}{\compatibleHull{G}{X}{A}}=\compatibleHull{G}{X}{A}$.
\end{theorem}
\begin{proof}
    Let $B=\compatibleHull{G}{X}{A}$.
    $A\subseteq B$ by \obv{atlas_subset_compatible_hull}.
    $\compatibleHull{G}{X}{B}\subseteq\compatibleHull{G}{X}{A}$ by
        \obv{compatible_hull_antitone}.
    $B$ is a coordinate atlas on $X$ under $G$ by \auto{compatible_hull_is_atlas}.
    $B\subseteq\compatibleHull{G}{X}{B}$ by \obv{atlas_subset_compatible_hull}.
    Follows by \auto{subseteq_antisymmetric}.
\end{proof}

\begin{classification}\label{max_atlas_is_maximal}
    Let $G$ be a coordinate pseudogroup and $X$ be a topological space.
    Suppose $A$ is a coordinate atlas on $X$ under $G$.
    Then $\MaxAtlas{G}{X}{A}$ is a maximal coordinate atlas on $X$ under $G$.
\end{classification}
\begin{proof}
    $\compatibleHull{G}{X}{A}$ is a coordinate atlas on $X$ under $G$ by
        \obv{compatible_hull_is_atlas}.
    $\compatibleHull{G}{X}{\compatibleHull{G}{X}{A}}=\compatibleHull{G}{X}{A}$ by
        \obv{compatible_hull_idempotent}.
    $\compatibleHull{G}{X}{\compatibleHull{G}{X}{A}}\subseteq\compatibleHull{G}{X}{A}$
        by \obv{subseteq_refl}.
    Follows by \auto{max_atlas,maximal_coordinate_atlas}.
\end{proof}

\begin{theorem}\label{max_atlas_member_iff}
    Let $G$ be a coordinate pseudogroup and $X$ be a topological space.
    Suppose $A$ is a coordinate atlas on $X$ under $G$.
    Then $c\in\MaxAtlas{G}{X}{A}$ iff
        $c$ is an open partial homeomorphism from $X$ to $\model[G]$ and
        for every $a\in A$ we have $c$ is compatible with $a$ on $X$ under $G$.
\end{theorem}
\begin{proof}
    Follows by \auto{max_atlas,compatible_hull_iff}.
\end{proof}

\begin{theorem}\label{atlas_subset_max_atlas}
    Let $G$ be a coordinate pseudogroup and $X$ be a topological space.
    Suppose $A$ is a coordinate atlas on $X$ under $G$.
    Then $A\subseteq\MaxAtlas{G}{X}{A}$.
\end{theorem}
\begin{proof}
    Follows by \auto{max_atlas,atlas_subset_compatible_hull}.
\end{proof}

\begin{theorem}\label{maximal_coordinate_atlas_fixed_point}
    Let $G$ be a coordinate pseudogroup and $X$ be a topological space.
    Suppose $A$ is a coordinate atlas on $X$ under $G$.
    Then $A$ is a maximal coordinate atlas on $X$ under $G$ iff
        $A=\MaxAtlas{G}{X}{A}$.
\end{theorem}
\begin{proof}
    $A\subseteq\compatibleHull{G}{X}{A}$ by \obv{atlas_subset_compatible_hull}.
    Follows by \auto{maximal_coordinate_atlas,max_atlas,subseteq_antisymmetric,
        subseteq_refl}.
\end{proof}

\begin{theorem}\label{atlas_extension_subset_max_atlas}
    Let $G$ be a coordinate pseudogroup and $X$ be a topological space.
    Suppose $A$ is a coordinate atlas on $X$ under $G$ and
        $B$ is a coordinate atlas on $X$ under $G$.
    Suppose $A\subseteq B$.
    Then $B\subseteq\MaxAtlas{G}{X}{A}$.
\end{theorem}
\begin{proof}
    For every $b\in B$ we have $b$ is an open partial homeomorphism from
        $X$ to $\model[G]$ by \obv{coordinate_atlas}.
    For every $b\in B$ we have for every $a\in A$ we have
        $b$ is compatible with $a$ on $X$ under $G$ by
        \auto{coordinate_atlas,subseteq}.
    Follows by \auto{max_atlas_member_iff,subseteq}.
\end{proof}

\begin{theorem}\label{max_atlas_equality_iff}
    Let $G$ be a coordinate pseudogroup and $X$ be a topological space.
    Suppose $A$ is a coordinate atlas on $X$ under $G$ and
        $B$ is a coordinate atlas on $X$ under $G$.
    Then $\MaxAtlas{G}{X}{A}=\MaxAtlas{G}{X}{B}$ iff
        for every $a\in A$ we have for every $b\in B$ we have
            $a$ is compatible with $b$ on $X$ under $G$.
\end{theorem}
\begin{proof}
    Let $H=\compatibleHull{G}{X}{A}$ and $K=\compatibleHull{G}{X}{B}$.
    $A\subseteq H$ and $B\subseteq K$ by \auto{atlas_subset_compatible_hull}.
    $\MaxAtlas{G}{X}{A}=H$ and $\MaxAtlas{G}{X}{B}=K$ by \auto{max_atlas}.
    Show that if $\MaxAtlas{G}{X}{A}=\MaxAtlas{G}{X}{B}$, then
        for every $a\in A$ we have for every $b\in B$ we have
            $a$ is compatible with $b$ on $X$ under $G$.
    \begin{subproof}
        Assume $\MaxAtlas{G}{X}{A}=\MaxAtlas{G}{X}{B}$.
        $H=K$.
        For every $a\in A$ we have for every $b\in B$ we have $a,b\in H$ by
            \auto{subseteq}.
        Follows by \auto{compatible_hull_pairwise}.
    \end{subproof}
    Show that if for every $a\in A$ we have for every $b\in B$ we have
        $a$ is compatible with $b$ on $X$ under $G$,
        then $\MaxAtlas{G}{X}{A}=\MaxAtlas{G}{X}{B}$.
    \begin{subproof}
        Assume for every $a\in A$ we have for every $b\in B$ we have
            $a$ is compatible with $b$ on $X$ under $G$.
        $A\subseteq K$ by \auto{compatible_hull_iff,coordinate_atlas,subseteq}.
        For every $b\in B$ we have for every $a\in A$ we have
            $b$ is compatible with $a$ on $X$ under $G$ by
            \auto{coordinate_chart_compatibility_symmetric}.
        $B\subseteq H$ by \auto{compatible_hull_iff,coordinate_atlas,subseteq}.
        $\compatibleHull{G}{X}{K}\subseteq H$ and
            $\compatibleHull{G}{X}{H}\subseteq K$ by
            \auto{compatible_hull_antitone}.
        $\compatibleHull{G}{X}{H}=H$ and $\compatibleHull{G}{X}{K}=K$ by
            \auto{compatible_hull_idempotent}.
        $H=K$ by \auto{subseteq_antisymmetric}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{corollary}\label{atlas_extension_same_max_atlas}
    Let $G$ be a coordinate pseudogroup and $X$ be a topological space.
    Suppose $A$ is a coordinate atlas on $X$ under $G$ and
        $B$ is a coordinate atlas on $X$ under $G$.
    Suppose $A\subseteq B$.
    Then $\MaxAtlas{G}{X}{A}=\MaxAtlas{G}{X}{B}$.
\end{corollary}
\begin{proof}
    For every $a\in A$ we have for every $b\in B$ we have
        $a$ is compatible with $b$ on $X$ under $G$ by
        \auto{coordinate_atlas,subseteq}.
    Follows by \auto{max_atlas_equality_iff}.
\end{proof}

\subsection{Enlarging the pseudogroup}

Allowing more changes enlarges the maximal atlas. Maximalizing first under
the smaller pseudogroup does not affect the final atlas under the larger one.

\begin{theorem}\label{subpseudogroup_max_atlas_inclusion}
    Let $G,H$ be coordinate pseudogroups and $X$ be a topological space.
    Suppose $G$ is a subpseudogroup of $H$.
    Let $A$ be a coordinate atlas on $X$ under $G$.
    Then $\MaxAtlas{G}{X}{A}\subseteq\MaxAtlas{H}{X}{A}$.
\end{theorem}
\begin{proof}
    $A$ is a coordinate atlas on $X$ under $H$ by \obv{subpseudogroup_atlas}.
    $\MaxAtlas{G}{X}{A}$ is a coordinate atlas on $X$ under $G$ by
        \auto{max_atlas_is_maximal,maximal_coordinate_atlas}.
    $\MaxAtlas{G}{X}{A}$ is a coordinate atlas on $X$ under $H$ by \obv{subpseudogroup_atlas}.
    $A\subseteq\MaxAtlas{G}{X}{A}$ by \obv{atlas_subset_max_atlas}.
    Follows by \obv{atlas_extension_subset_max_atlas}.
\end{proof}

\begin{theorem}\label{subpseudogroup_max_atlas_regeneration}
    Let $G,H$ be coordinate pseudogroups and $X$ be a topological space.
    Suppose $G$ is a subpseudogroup of $H$.
    Let $A$ be a coordinate atlas on $X$ under $G$.
    Then $\MaxAtlas{H}{X}{\MaxAtlas{G}{X}{A}}=\MaxAtlas{H}{X}{A}$.
\end{theorem}
\begin{proof}
    $A$ is a coordinate atlas on $X$ under $H$ by \obv{subpseudogroup_atlas}.
    $\MaxAtlas{G}{X}{A}$ is a coordinate atlas on $X$ under $G$ by
        \auto{max_atlas_is_maximal,maximal_coordinate_atlas}.
    $\MaxAtlas{G}{X}{A}$ is a coordinate atlas on $X$ under $H$ by \obv{subpseudogroup_atlas}.
    $A\subseteq\MaxAtlas{G}{X}{A}$ by \obv{atlas_subset_max_atlas}.
    Follows by \obv{atlas_extension_same_max_atlas}.
\end{proof}

\subsection{The empty space}

The empty family covers only the empty space. Its maximal atlas is not the
empty family: it contains the unique empty chart.

\begin{theorem}\label{empty_coordinate_atlas_iff}
    Let $G$ be a coordinate pseudogroup and $X$ be a topological space.
    Then $\emptyset$ is a coordinate atlas on $X$ under $G$ iff
        $\carrier[X]=\emptyset$.
\end{theorem}
\begin{proof}
    Follows by \auto{coordinate_atlas,setext}.
\end{proof}

\begin{theorem}\label{empty_source_chart}
    Let $X,E$ be topological spaces.
    Suppose $\carrier[X]=\emptyset$.
    Suppose $c$ is an open partial homeomorphism from $X$ to $E$.
    Then $c=\emptyset$.
\end{theorem}
\begin{proof}
    $c$ is a raw coordinate chart on $\carrier[X]$ modelled on $E$ by
        \obv{open_partial_homeomorphism_is_raw_chart}.
    $\dom{c}\subseteq\carrier[X]$ by \obv{raw_coordinate_chart}.
    $\dom{c}=\emptyset$ by \auto{subseteq_emptyset_iff}.
    Follows by \obv{open_partial_homeomorphism_empty_domain}.
\end{proof}

\begin{theorem}\label{empty_max_atlas}
    Let $G$ be a coordinate pseudogroup and $X$ be a topological space.
    Suppose $\emptyset$ is a coordinate atlas on $X$ under $G$.
    Then $\MaxAtlas{G}{X}{\emptyset}=\{\emptyset\}$.
\end{theorem}
\begin{proof}
    $\carrier[X]=\emptyset$ by \auto{empty_coordinate_atlas_iff}.
    $\model[G]$ is a topological space by \obv{pseudogroup_model}.
    $\emptyset$ is an open partial homeomorphism from $X$ to $\model[G]$ by
        \obv{empty_open_partial_homeomorphism}.
    For every $a\in\emptyset$ we have $\emptyset$ is compatible with $a$ on $X$
        under $G$ by \auto.
    $\emptyset\in\MaxAtlas{G}{X}{\emptyset}$ by \auto{max_atlas_member_iff}.
    Every element of $\MaxAtlas{G}{X}{\emptyset}$ is an open partial homeomorphism
        from $X$ to $\model[G]$ by \auto{max_atlas_member_iff}.
    Every element of $\MaxAtlas{G}{X}{\emptyset}$ is equal to $\emptyset$ by
        \auto{empty_source_chart}.
    Follows by \auto{setext,singleton_iff}.
\end{proof}
