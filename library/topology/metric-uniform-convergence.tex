\section{Metric and uniform convergence}

\begin{theorem}\label{metric_uniform_cauchy_filter_iff}
    Let $M$ be a pseudometric space.
    Let $F$ be a proper filter on $\carrier[M]$.
    Then $\uniformCauchyFilter{\metricUniformSpace{M}}{F}$ iff
        for every $e\in\reals$ such that
            $\surrealLt{\surrealZero}{e}$ there exists $A\in F$ such that
            for all $x,y\in A$ we have
                $\surrealLt{\metricDistance[M](x,y)}{e}$.
\end{theorem}
\begin{proof}
    $F$ is a filter on $\carrier[M]$ by
        \obv{proper_filter_is_filter}.
    $\metricUniformSpace{M}$ is a uniform space by
        \obv{metric_uniform_space_is_uniform_space}.
    $\carrier[\metricUniformSpace{M}]=\carrier[M]$ and
        $\uniformity[\metricUniformSpace{M}]=\metricUniformity{M}$ by
        \obv{metric_uniform_space_carrier,
            metric_uniform_space_uniformity}.

    Show that if $\uniformCauchyFilter{\metricUniformSpace{M}}{F}$,
        then for every $e\in\reals$ such that
        $\surrealLt{\surrealZero}{e}$ there exists $A\in F$ such that
        for all $x,y\in A$ we have
            $\surrealLt{\metricDistance[M](x,y)}{e}$.
    \begin{subproof}
        Assume $\uniformCauchyFilter{\metricUniformSpace{M}}{F}$.
        Fix $e\in\reals$.
        Assume $\surrealLt{\surrealZero}{e}$.
        $\metricEntourage{M}{e}\in\metricUniformity{M}$ by
            \obv{metric_entourage_is_entourage}.
        Hence $\metricEntourage{M}{e}\in
            \uniformity[\metricUniformSpace{M}]$.
        Take $A\in F$ such that
            $A\times A\subseteq\metricEntourage{M}{e}$ by
            \obv{uniform_cauchy_filter_entourage_square}.
        $A\subseteq\carrier[M]$ by \obv{filter}.
        Show that for all $x,y\in A$ we have
            $\surrealLt{\metricDistance[M](x,y)}{e}$.
        \begin{subproof}
            Fix $x,y\in A$.
            Hence $x,y\in\carrier[M]$ by \obv{subseteq}.
            $(x,y)\in A\times A$ by \obv{times_tuple_intro}.
            Hence $(x,y)\in\metricEntourage{M}{e}$ by \obv{subseteq}.
            $(x,y)\in\metricEntourage{M}{e}$ iff
                $x,y\in\carrier[M]$ and
                $\surrealLt{\metricDistance[M](x,y)}{e}$ by
                \obv{metric_entourage_iff}.
            Follows by \auto.
        \end{subproof}
        Follows by \auto.
    \end{subproof}

    Show that if for every $e\in\reals$ such that
        $\surrealLt{\surrealZero}{e}$ there exists $A\in F$ such that
        for all $x,y\in A$ we have
            $\surrealLt{\metricDistance[M](x,y)}{e}$, then
        $\uniformCauchyFilter{\metricUniformSpace{M}}{F}$.
    \begin{subproof}
        Assume for every $e\in\reals$ such that
            $\surrealLt{\surrealZero}{e}$ there exists $A\in F$ such
            that for all $x,y\in A$ we have
                $\surrealLt{\metricDistance[M](x,y)}{e}$.
        Show that for every
            $U\in\uniformity[\metricUniformSpace{M}]$ there exists
            $A\in F$ such that $A\times A\subseteq U$.
        \begin{subproof}
            Fix $U\in\uniformity[\metricUniformSpace{M}]$.
            Hence $U\in\metricUniformity{M}$.
            Take $e\in\reals$ such that
                $\surrealLt{\surrealZero}{e}$ and
                $\metricEntourage{M}{e}\subseteq U$ by
                \obv{metric_uniformity}.
            Take $A\in F$ such that for all $x,y\in A$ we have
                $\surrealLt{\metricDistance[M](x,y)}{e}$.
            $A\subseteq\carrier[M]$ by \obv{filter}.
            Show that $A\times A\subseteq U$.
            \begin{subproof}
                It suffices to show that every element of $A\times A$ is an
                    element of $U$ by \obv{subseteq}.
                Fix $p\in A\times A$.
                Take $x,y\in A$ such that $p=(x,y)$ by
                    \obv{times_elem_is_tuple,times_tuple_elim}.
                Hence $x,y\in\carrier[M]$ by \obv{subseteq}.
                $\surrealLt{\metricDistance[M](x,y)}{e}$.
                $(x,y)\in\metricEntourage{M}{e}$ iff
                    $x,y\in\carrier[M]$ and
                    $\surrealLt{\metricDistance[M](x,y)}{e}$ by
                    \obv{metric_entourage_iff}.
                Hence $(x,y)\in\metricEntourage{M}{e}$.
                Hence $(x,y)\in U$ by \obv{subseteq}.
            \end{subproof}
        \end{subproof}
        $\uniformCauchyFilter{\metricUniformSpace{M}}{F}$ iff
            $F$ is a proper filter on
                $\carrier[\metricUniformSpace{M}]$ and
            for every $V\in\uniformity[\metricUniformSpace{M}]$
                there exists $A\in F$ such that $A\times A\subseteq V$ by
            \obv{uniform_cauchy_filter_iff_entourage_squares}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{theorem}\label{metric_uniform_sequence_convergence_iff}
    Let $M$ be a metric space and
        $s\in\funs{\naturals}{\carrier[M]}$.
    Let $x\in\carrier[M]$.
    Then
        $\metricConverges{M}{s}{x}$ iff
        $\uniformSequenceConverges{\metricUniformSpace{M}}{s}{x}$.
\end{theorem}
\begin{proof}
    $\metricUniformSpace{M}$ is a uniform space by
        \obv{metric_uniform_space_is_uniform_space}.
    $\carrier[\metricUniformSpace{M}]=\carrier[M]$
        by \obv{metric_uniform_space_carrier}.
    Hence $s\in\funs{\naturals}
        {\carrier[\metricUniformSpace{M}]}$ and
        $x\in\carrier[\metricUniformSpace{M}]$.
    $\uniformity[\metricUniformSpace{M}]=\metricUniformity{M}$
        by \obv{metric_uniform_space_uniformity}.
    Show that if $\metricConverges{M}{s}{x}$, then
        $\uniformSequenceConverges{\metricUniformSpace{M}}{s}{x}$.
    \begin{subproof}
        Assume $\metricConverges{M}{s}{x}$.
        Show that for every
            $U\in\uniformity[\metricUniformSpace{M}]$ there exists
            $N\in\naturals$ such that for every $n\in\naturals$ such that
            $N\subseteq n$ we have $(x,s(n))\in U$.
        \begin{subproof}
            Fix $U\in\uniformity[\metricUniformSpace{M}]$.
            $U\in\metricUniformity{M}$.
            Take $e\in\reals$ such that
                $\surrealLt{\surrealZero}{e}$ and
                $\metricEntourage{M}{e}\subseteq U$
                by \obv{metric_uniformity}.
            Take $N\in\naturals$ such that for every $n\in\naturals$
                such that $N\subseteq n$ we have
                $s(n)\in\metricBall{M}{x}{e}$
                by \obv{metric_convergence}.
            Show that for every $n\in\naturals$ such that $N\subseteq n$
                we have $(x,s(n))\in U$.
            \begin{subproof}
                Fix $n$.
                Assume $n\in\naturals$ and $N\subseteq n$.
                $s(n)\in\metricBall{M}{x}{e}$.
                Hence $s(n)\in\carrier[M]$ by \obv{metric_ball}.
                $\surrealLt{\metricDistance[M](x,s(n))}{e}$
                    by \obv{metric_ball}.
                Hence $(x,s(n))\in\metricEntourage{M}{e}$
                    by \auto{metric_entourage_iff}.
                Follows by \obv{subseteq}.
            \end{subproof}
            Follows by \auto.
        \end{subproof}
        Follows by \auto{uniform_sequence_convergence_iff_entourages}.
    \end{subproof}
    Show that if
        $\uniformSequenceConverges{\metricUniformSpace{M}}{s}{x}$, then
        $\metricConverges{M}{s}{x}$.
    \begin{subproof}
        Assume
            $\uniformSequenceConverges{\metricUniformSpace{M}}{s}{x}$.
        Show that for every $e\in\reals$ such that
            $\surrealLt{\surrealZero}{e}$ there exists $N\in\naturals$
            such that for every $n\in\naturals$ such that $N\subseteq n$
            we have $s(n)\in\metricBall{M}{x}{e}$.
        \begin{subproof}
            Fix $e$.
            Assume $e\in\reals$ and
                $\surrealLt{\surrealZero}{e}$.
            $\metricEntourage{M}{e}\in\metricUniformity{M}$
                by \obv{metric_entourage_is_entourage}.
            Hence
                $\metricEntourage{M}{e}
                    \in\uniformity[\metricUniformSpace{M}]$.
            Take $N\in\naturals$ such that for every $n\in\naturals$
                such that $N\subseteq n$ we have
                $(x,s(n))\in\metricEntourage{M}{e}$
                by \obv{uniform_sequence_convergence_entourage}.
            Show that for every $n\in\naturals$ such that $N\subseteq n$
                we have $s(n)\in\metricBall{M}{x}{e}$.
            \begin{subproof}
                Fix $n$.
                Assume $n\in\naturals$ and $N\subseteq n$.
                $(x,s(n))\in\metricEntourage{M}{e}$.
                Hence $s(n)\in\carrier[M]$ and
                    $\surrealLt{\metricDistance[M](x,s(n))}{e}$
                    by \auto{metric_entourage_iff}.
                Follows by \obv{metric_ball}.
            \end{subproof}
            Follows by \auto.
        \end{subproof}
        Follows by \obv{metric_convergence}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{theorem}\label{metric_uniform_cauchy_sequence_iff}
    Let $M$ be a metric space and
        $s\in\funs{\naturals}{\carrier[M]}$.
    Then
        $\metricCauchy{M}{s}$ iff
        $\uniformCauchySequence{\metricUniformSpace{M}}{s}$.
\end{theorem}
\begin{proof}
    $\metricUniformSpace{M}$ is a uniform space by
        \obv{metric_uniform_space_is_uniform_space}.
    $\carrier[\metricUniformSpace{M}]=\carrier[M]$
        by \obv{metric_uniform_space_carrier}.
    Hence $s\in\funs{\naturals}
        {\carrier[\metricUniformSpace{M}]}$.
    $\uniformity[\metricUniformSpace{M}]=\metricUniformity{M}$
        by \obv{metric_uniform_space_uniformity}.
    Show that if $\metricCauchy{M}{s}$, then
        $\uniformCauchySequence{\metricUniformSpace{M}}{s}$.
    \begin{subproof}
        Assume $\metricCauchy{M}{s}$.
        Show that for every
            $U\in\uniformity[\metricUniformSpace{M}]$ there exists
            $N\in\naturals$ such that for all $m,n\in\naturals$ such that
            $N\subseteq m$ and $N\subseteq n$ we have $(s(m),s(n))\in U$.
        \begin{subproof}
            Fix $U\in\uniformity[\metricUniformSpace{M}]$.
            $U\in\metricUniformity{M}$.
            Take $e\in\reals$ such that
                $\surrealLt{\surrealZero}{e}$ and
                $\metricEntourage{M}{e}\subseteq U$
                by \obv{metric_uniformity}.
            Take $N\in\naturals$ such that for all $m,n\in\naturals$
                such that $N\subseteq m$ and $N\subseteq n$ we have
                $\surrealLt{\metricDistance[M](s(m),s(n))}{e}$
                by \obv{metric_cauchy_sequence}.
            Show that
                $\forall m,n:\Set.
                    (m\in\naturals\land n\in\naturals\land
                     N\subseteq m\land N\subseteq n)
                    \implies (s(m),s(n))\in U$.
            \begin{subproof}
                Fix $m,n$.
                Assume
                    $m\in\naturals\land n\in\naturals\land
                    N\subseteq m\land N\subseteq n$.
                $s(m),s(n)\in\carrier[M]$
                    by \obv{funs_type_apply}.
                $(s(m),s(n))\in\metricEntourage{M}{e}$
                    by \auto{metric_entourage_iff}.
                Follows by \obv{subseteq}.
            \end{subproof}
            Follows by \auto.
        \end{subproof}
        Follows by \auto{uniform_cauchy_sequence_iff_entourages}.
    \end{subproof}
    Show that if
        $\uniformCauchySequence{\metricUniformSpace{M}}{s}$, then
        $\metricCauchy{M}{s}$.
    \begin{subproof}
        Assume $\uniformCauchySequence{\metricUniformSpace{M}}{s}$.
        Show that for every $e\in\reals$ such that
            $\surrealLt{\surrealZero}{e}$ there exists $N\in\naturals$
            such that for all $m,n\in\naturals$ such that
                $N\subseteq m$ and $N\subseteq n$ we have
                $\surrealLt{\metricDistance[M](s(m),s(n))}{e}$.
        \begin{subproof}
            Fix $e$.
            Assume $e\in\reals$ and
                $\surrealLt{\surrealZero}{e}$.
            $\metricEntourage{M}{e}\in\metricUniformity{M}$
                by \obv{metric_entourage_is_entourage}.
            Hence
                $\metricEntourage{M}{e}
                    \in\uniformity[\metricUniformSpace{M}]$.
            Take $N\in\naturals$ such that for all $m,n\in\naturals$
                such that $N\subseteq m$ and $N\subseteq n$ we have
                $(s(m),s(n))\in\metricEntourage{M}{e}$
                by \obv{uniform_cauchy_sequence_entourage}.
            Show that
                $\forall m,n:\Set.
                    (m\in\naturals\land n\in\naturals\land
                     N\subseteq m\land N\subseteq n)
                    \implies
                    \surrealLt{\metricDistance[M](s(m),s(n))}{e}$.
            \begin{subproof}
                Fix $m,n$.
                Assume
                    $m\in\naturals\land n\in\naturals\land
                    N\subseteq m\land N\subseteq n$.
                Follows by \auto{metric_entourage_iff}.
            \end{subproof}
            Follows by \auto.
        \end{subproof}
        Follows by \obv{metric_cauchy_sequence}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{theorem}\label{metric_sequential_completeness_iff_uniform}
    Let $M$ be a metric space.
    Then $M$ is sequentially complete metric iff
        $\metricUniformSpace{M}$ is sequentially complete uniform.
\end{theorem}
\begin{proof}
    $\metricUniformSpace{M}$ is a uniform space
        by \obv{metric_uniform_space_is_uniform_space}.
    Show that if $M$ is sequentially complete metric, then
        $\metricUniformSpace{M}$ is sequentially complete uniform.
    \begin{subproof}
        Assume $M$ is sequentially complete metric.
        Show that for every $s$ such that
            $\uniformCauchySequence{\metricUniformSpace{M}}{s}$ there exists
            $x\in\carrier[\metricUniformSpace{M}]$ such that
            $\uniformSequenceConverges{\metricUniformSpace{M}}{s}{x}$.
        \begin{subproof}
            Fix $s$.
            Assume $\uniformCauchySequence{\metricUniformSpace{M}}{s}$.
            $s\in\funs{\naturals}{\carrier[\metricUniformSpace{M}]}$
                by \obv{uniform_cauchy_sequence_sequence}.
            Hence $s\in\funs{\naturals}{\carrier[M]}$
                by \obv{metric_uniform_space_carrier}.
            $\metricCauchy{M}{s}$
                by \obv{metric_uniform_cauchy_sequence_iff}.
            Take $x\in\carrier[M]$ such that
                $\metricConverges{M}{s}{x}$
                by \obv{sequentially_complete_metric_space}.
            $x\in\carrier[\metricUniformSpace{M}]$
                by \obv{metric_uniform_space_carrier}.
            $\uniformSequenceConverges{\metricUniformSpace{M}}{s}{x}$
                by \obv{metric_uniform_sequence_convergence_iff}.
        \end{subproof}
        Follows by \obv{sequentially_complete_uniform_space}.
    \end{subproof}
    Show that if
        $\metricUniformSpace{M}$ is sequentially complete uniform, then
        $M$ is sequentially complete metric.
    \begin{subproof}
        Assume $\metricUniformSpace{M}$ is sequentially complete uniform.
        Show that for every $s$ such that $\metricCauchy{M}{s}$ there exists
            $x\in\carrier[M]$ such that $\metricConverges{M}{s}{x}$.
        \begin{subproof}
            Fix $s$.
            Assume $\metricCauchy{M}{s}$.
            $s\in\funs{\naturals}{\carrier[M]}$
                by \obv{metric_cauchy_sequence}.
            Hence $\uniformCauchySequence{\metricUniformSpace{M}}{s}$
                by \obv{metric_uniform_cauchy_sequence_iff}.
            Take $x\in\carrier[\metricUniformSpace{M}]$ such that
                $\uniformSequenceConverges{\metricUniformSpace{M}}{s}{x}$
                by \obv{sequentially_complete_uniform_space}.
            $x\in\carrier[M]$ by \obv{metric_uniform_space_carrier}.
            $\metricConverges{M}{s}{x}$
                by \obv{metric_uniform_sequence_convergence_iff}.
        \end{subproof}
        Follows by \obv{sequentially_complete_metric_space}.
    \end{subproof}
    Follows by \auto.
\end{proof}
