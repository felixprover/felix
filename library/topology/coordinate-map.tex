\section{Admissible maps in coordinates}

\import{topology/coordinate-atlas-presentation.tex}

An admissible map has an open exact domain and admits an admissible coordinate
expression near each domain point. Continuity is a consequence, not a premise.
The pseudogroups remain explicit parameters, so additional restrictions on
coordinate changes do not alter the map calculus.

\begin{theorem}\label{admissible_coordinate_change}
    Let $C$ be a local-map calculus.
    Let $G,H$ be coordinate pseudogroups.
    Let $X,Y$ be topological spaces.
    Suppose $G$ is compatible with the calculus $C$ and $H$ is compatible with the calculus $C$.
    Suppose $a$ is compatible with $c$ on $X$ under $G$ and
        $d$ is compatible with $b$ on $Y$ under $H$.
    Let $f$ be a function.
    Suppose $U\in\opens[X]$ and $U\subseteq\dom{a}$ and $U\subseteq\dom{c}$ and
        $U\subseteq\dom{f}$ and $\img{f}{U}\subseteq\dom{d}$.
    Suppose $\CoordExpr{d}{f}{c}{U}$ is admissible from $\model[G]$ to $\model[H]$ in $C$.
    Then $\CoordExpr{b}{f}{a}{U}$ is admissible from $\model[G]$ to $\model[H]$ in $C$.
\end{theorem}
\begin{proof}
    $\restrl{a}{U}$ is compatible with $\restrl{c}{U}$ on $X$ under $G$ by
        \obv{coordinate_compatibility_restriction}.
    $\chartTransition{\restrl{c}{U}}{\restrl{a}{U}}\in\changes[G]$ and
        $\chartTransition{b}{d}\in\changes[H]$ by \obv{coordinate_chart_compatibility}.
    $\chartTransition{\restrl{c}{U}}{\restrl{a}{U}}$ is admissible
        from $\model[G]$ to $\model[G]$ in $C$ and
        $\chartTransition{b}{d}$ is admissible from $\model[H]$ to $\model[H]$ in $C$ by
        \auto{pseudogroup_calculus_compatible}.
    $\CoordExpr{d}{f}{c}{U}\circ\chartTransition{\restrl{c}{U}}{\restrl{a}{U}}$
        is admissible from $\model[G]$ to $\model[H]$ in $C$ by
        \obv{admissible_map_composition}.
    $\chartTransition{b}{d}\circ
        (\CoordExpr{d}{f}{c}{U}\circ\chartTransition{\restrl{c}{U}}{\restrl{a}{U}})$
        is admissible from $\model[G]$ to $\model[H]$ in $C$ by
        \obv{admissible_map_composition}.
    $a$ is a function and $b$ is a function and $c$ is an injection and $d$ is an injection by
        \auto{coordinate_chart_compatibility,open_partial_homeomorphism_bijection,
            bijections,bijection_is_function}.
    $\CoordExpr{b}{f}{a}{U}=\chartTransition{b}{d}\circ
        (\CoordExpr{d}{f}{c}{U}\circ\chartTransition{\restrl{c}{U}}{\restrl{a}{U}})$ by
        \auto{coordinate_expression_change_charts,chart_transition}.
\end{proof}

\begin{theorem}\label{continuous_coordinate_expression}
    Suppose $c$ is an open partial homeomorphism from $X$ to $E$ and
        $d$ is an open partial homeomorphism from $Y$ to $F$.
    Suppose $f$ is a continuous partial map from $X$ to $Y$ and $U\in\opens[X]$.
    Then $\CoordExpr{d}{f}{c}{U}$ is a continuous partial map from $E$ to $F$.
\end{theorem}
\begin{proof}
    $\restrl{c}{U}$ is an open partial homeomorphism from $X$ to $E$ by
        \obv{open_partial_homeomorphism_restriction}.
    $\converse{\restrl{c}{U}}$ is an open partial homeomorphism from $E$ to $X$ by
        \obv{open_partial_homeomorphism_converse}.
    $\converse{\restrl{c}{U}}$ is a continuous partial map from $E$ to $X$ and
        $d$ is a continuous partial map from $Y$ to $F$ by
        \obv{open_partial_homeomorphism_is_continuous_partial_map}.
    $\restrl{f}{U}$ is a continuous partial map from $X$ to $Y$ by
        \obv{continuous_partial_map_restriction}.
    $(\restrl{f}{U})\circ\converse{\restrl{c}{U}}$ is a continuous partial map from $E$ to $Y$ by
        \obv{continuous_partial_map_composition}.
    $d\circ((\restrl{f}{U})\circ\converse{\restrl{c}{U}})$ is a continuous partial map from $E$ to $F$ by
        \obv{continuous_partial_map_composition}.
    $c$ is a function and $d$ is a function and $f$ is a function by
        \auto{open_partial_homeomorphism_bijection,bijection_is_function,
            continuous_partial_map_type,funs_is_function}.
    Follows by \auto{coordinate_expression_normalization}.
\end{proof}

\begin{definition}\label{coordinate_admissible_at}
    Let $C$ be a local-map calculus.
    Let $G,H$ be coordinate pseudogroups.
    Let $M,N$ be equipped coordinate spaces.
    $f$ is coordinate-admissible at $p$ from $M$ under $G$ to $N$ under $H$ in $C$ iff
        there exists $c\in\ChartsAt{M}{p}$ such that there exists $d\in\atlas[N]$
        such that there exists $U\in\opens[M]$ such that
            $p\in U$ and $U\subseteq\dom{f}$ and $U\subseteq\dom{c}$ and
            $\img{f}{U}\subseteq\dom{d}$ and
            $\CoordExpr{d}{f}{c}{U}$ is admissible from $\model[G]$ to $\model[H]$ in $C$.
\end{definition}

\begin{definition}\label{coordinate_admissible_map}
    Let $C$ be a local-map calculus.
    Let $G,H$ be coordinate pseudogroups.
    Let $M,N$ be equipped coordinate spaces.
    $f$ is coordinate-admissible from $M$ under $G$ to $N$ under $H$ in $C$ iff
        $G$ is compatible with the calculus $C$ and $H$ is compatible with the calculus $C$ and
        $M$ has coordinates under $G$ and $N$ has coordinates under $H$ and
        $\dom{f}\in\opens[M]$ and $f\in\funs{\dom{f}}{\carrier[N]}$ and
        for every $p\in\dom{f}$ we have
            $f$ is coordinate-admissible at $p$ from $M$ under $G$ to $N$ under $H$ in $C$.
\end{definition}

\begin{theorem}\label{equipped_atlas_on_space}
    Let $G$ be a coordinate pseudogroup.
    Let $M$ be an equipped coordinate space.
    Suppose $M$ has coordinates under $G$.
    Then $\atlas[M]$ is a coordinate atlas on $M$ under $G$.
\end{theorem}
\begin{proof}
    $M$ is a topological space by \obv{equipped_coordinate_space}.
    $\TopologicalReduct{M}$ is a topological space by
        \obv{topological_reduct_is_topological_space}.
    $\atlas[M]$ is a maximal coordinate atlas on $\TopologicalReduct{M}$ under $G$ by
        \obv{equipped_coordinates}.
    $\atlas[M]$ is a coordinate atlas on $\TopologicalReduct{M}$ under $G$ by
        \obv{maximal_coordinate_atlas}.
    $\carrier[\TopologicalReduct{M}]=\carrier[M]$ and
        $\opens[\TopologicalReduct{M}]=\opens[M]$ by
        \obv{topological_reduct_carrier,topological_reduct_opens}.
    Follows by \auto{coordinate_atlas_same_fields}.
\end{proof}

\begin{theorem}\label{coordinate_admissible_local_continuity}
    Let $C$ be a local-map calculus.
    Let $G,H$ be coordinate pseudogroups.
    Let $M,N$ be equipped coordinate spaces.
    Suppose $M$ has coordinates under $G$ and $N$ has coordinates under $H$.
    Let $f$ be a function.
    Suppose $f$ is coordinate-admissible at $p$ from $M$ under $G$ to $N$ under $H$ in $C$.
    Then there exists $U\in\opens[M]$ such that $p\in U$ and
        $\restrl{f}{U}$ is a continuous partial map from $M$ to $N$.
\end{theorem}
\begin{proof}
    $\atlas[M]$ is a coordinate atlas on $M$ under $G$ and
        $\atlas[N]$ is a coordinate atlas on $N$ under $H$ by \obv{equipped_atlas_on_space}.
    Take $c,d,U$ such that $c\in\ChartsAt{M}{p}$ and $d\in\atlas[N]$ and
        $U\in\opens[M]$ and $p\in U$ and $U\subseteq\dom{f}$ and $U\subseteq\dom{c}$ and
        $\img{f}{U}\subseteq\dom{d}$ and
        $\CoordExpr{d}{f}{c}{U}$ is admissible from $\model[G]$ to $\model[H]$ in $C$
        by \auto{coordinate_admissible_at}.
    $c\in\atlas[M]$ by \obv{charts_at_iff}.
    $c$ is an open partial homeomorphism from $M$ to $\model[G]$ and
        $d$ is an open partial homeomorphism from $N$ to $\model[H]$ by \obv{coordinate_atlas}.
    $\CoordExpr{d}{f}{c}{U}$ is a continuous partial map from $\model[G]$ to $\model[H]$ by
        \obv{admissible_map_sound}.
    $\restrl{f}{U}$ is a continuous partial map from $M$ to $N$ by
        \obv{continuous_from_coordinate_expression}.
    Follows by \auto.
\end{proof}

\begin{theorem}\label{coordinate_admissible_is_continuous}
    Let $C$ be a local-map calculus.
    Let $G,H$ be coordinate pseudogroups.
    Let $M,N$ be equipped coordinate spaces.
    Suppose $f$ is coordinate-admissible from $M$ under $G$ to $N$ under $H$ in $C$.
    Then $f$ is a continuous partial map from $M$ to $N$.
\end{theorem}
\begin{proof}
    $M$ is a topological space and $N$ is a topological space by
        \obv{equipped_coordinate_space}.
    $M$ has coordinates under $G$ and $N$ has coordinates under $H$ and
        $\dom{f}\in\opens[M]$ and $f\in\funs{\dom{f}}{\carrier[N]}$ by
        \obv{coordinate_admissible_map}.
    $f$ is a function by \obv{funs_is_function}.
    Let $B=\{U\in\opens[M]\mid
        \text{$\restrl{f}{U}$ is a continuous partial map from $M$ to $N$}\}$.
    For every $p\in\dom{f}$ there exists $U\in B$ such that $p\in U$ by
        \auto{coordinate_admissible_map,coordinate_admissible_local_continuity}.
    For every $U\in B$ we have $\restrl{f}{U}$ is a continuous partial map from $M$ to $N$.
    Follows by \obv{continuous_partial_map_locality}.
\end{proof}

\begin{theorem}\label{coordinate_admissible_in_any_charts}
    Let $C$ be a local-map calculus.
    Let $G,H$ be coordinate pseudogroups.
    Let $M,N$ be equipped coordinate spaces.
    Suppose $f$ is coordinate-admissible from $M$ under $G$ to $N$ under $H$ in $C$.
    Let $p\in\dom{f}$.
    Let $a\in\ChartsAt{M}{p}$ and $b\in\ChartsAt{N}{f(p)}$.
    Then there exists $V\in\opens[M]$ such that $p\in V$ and
        $V\subseteq\dom{f}$ and $V\subseteq\dom{a}$ and $\img{f}{V}\subseteq\dom{b}$ and
        $\CoordExpr{b}{f}{a}{V}$ is admissible from $\model[G]$ to $\model[H]$ in $C$.
\end{theorem}
\begin{proof}
    $M$ is a coordinate datum and $N$ is a coordinate datum and
        $M$ is a topological space and $N$ is a topological space by \obv{equipped_coordinate_space}.
    $G$ is compatible with the calculus $C$ and $H$ is compatible with the calculus $C$ and
        $M$ has coordinates under $G$ and $N$ has coordinates under $H$ by
        \obv{coordinate_admissible_map}.
    $\atlas[M]$ is a coordinate atlas on $M$ under $G$ and
        $\atlas[N]$ is a coordinate atlas on $N$ under $H$ by \obv{equipped_atlas_on_space}.
    $f$ is a continuous partial map from $M$ to $N$ by \obv{coordinate_admissible_is_continuous}.
    $f$ is a function by \auto{continuous_partial_map_type,funs_is_function}.
    $a\in\atlas[M]$ and $p\in\dom{a}$ and
        $b\in\atlas[N]$ and $f(p)\in\dom{b}$ by \obv{charts_at_iff}.
    $a$ is an open partial homeomorphism from $M$ to $\model[G]$ and
        $b$ is an open partial homeomorphism from $N$ to $\model[H]$ by \obv{coordinate_atlas}.
    $\dom{a}\in\opens[M]$ and $\dom{b}\in\opens[N]$ by \obv{open_partial_homeomorphism}.
    $\preimg{f}{\dom{b}}\in\opens[M]$ by \obv{continuous_partial_map_preimage_open}.
    $p\in\preimg{f}{\dom{b}}$ by \auto{preimg_iff,function_apply_elim}.
    $f$ is coordinate-admissible at $p$ from $M$ under $G$ to $N$ under $H$ in $C$ by
        \obv{coordinate_admissible_map}.
    Take $c,d,U$ such that $c\in\ChartsAt{M}{p}$ and $d\in\atlas[N]$ and
        $U\in\opens[M]$ and $p\in U$ and $U\subseteq\dom{f}$ and $U\subseteq\dom{c}$ and
        $\img{f}{U}\subseteq\dom{d}$ and
        $\CoordExpr{d}{f}{c}{U}$ is admissible from $\model[G]$ to $\model[H]$ in $C$
        by \auto{coordinate_admissible_at}.
    $c\in\atlas[M]$ by \auto{charts_at_iff}.
    $a$ is compatible with $c$ on $M$ under $G$ and
        $d$ is compatible with $b$ on $N$ under $H$ by \obv{coordinate_atlas}.
    $c$ is an open partial homeomorphism from $M$ to $\model[G]$ by \obv{coordinate_atlas}.
    $c$ is an injection and $d$ is a function by
        \auto{coordinate_atlas,open_partial_homeomorphism_bijection,bijections,bijection_is_function}.
    Let $V=U\inter(\dom{a}\inter\preimg{f}{\dom{b}})$.
    $V\in\opens[M]$ by \auto{opens_inter}.
    $p\in V$ and $V\subseteq U$ and $V\subseteq\dom{a}$ and
        $V\subseteq\preimg{f}{\dom{b}}$ by \auto{inter,subseteq}.
    $V\subseteq\dom{f}$ and $V\subseteq\dom{c}$ by \auto{subseteq_transitive}.
    $\img{f}{V}\subseteq\dom{b}$ by \auto{subseteq,img_iff,preimg_iff,function_rightunique}.
    $\img{f}{V}\subseteq\dom{d}$ by \auto{img_subseteq,subseteq_transitive}.
    $\img{c}{V}\in\opens[\model[G]]$ by \obv{open_partial_homeomorphism_image_open}.
    $\restrl{\CoordExpr{d}{f}{c}{U}}{\img{c}{V}}$ is admissible
        from $\model[G]$ to $\model[H]$ in $C$ by \obv{admissible_map_restriction}.
    $\CoordExpr{d}{f}{c}{V}=\restrl{\CoordExpr{d}{f}{c}{U}}{\img{c}{V}}$ by
        \obv{coordinate_expression_restriction}.
    $\CoordExpr{b}{f}{a}{V}$ is admissible from $\model[G]$ to $\model[H]$ in $C$ by
        \auto{admissible_coordinate_change}.
    Follows by \auto.
\end{proof}

\subsection{Checking through presentations}

Any presenting atlases suffice for checking a map. Their charts may be chosen
separately near each point; no choice is stored in the map or its endpoints.

\begin{theorem}\label{coordinate_admissible_presentations}
    Let $C$ be a local-map calculus.
    Let $G,H$ be coordinate pseudogroups.
    Let $M,N$ be equipped coordinate spaces.
    Suppose $G$ is compatible with the calculus $C$ and $H$ is compatible with the calculus $C$.
    Suppose $M$ has coordinates under $G$ and $N$ has coordinates under $H$.
    Suppose $A$ presents $M$ on $M$ under $G$ and $B$ presents $N$ on $N$ under $H$.
    Suppose $\dom{f}\in\opens[M]$ and $f\in\funs{\dom{f}}{\carrier[N]}$.
    Then $f$ is coordinate-admissible from $M$ under $G$ to $N$ under $H$ in $C$ iff
        for every $p\in\dom{f}$ there exists $c\in A$ such that there exists $d\in B$
            such that there exists $U\in\opens[M]$ such that
                $p\in U$ and $U\subseteq\dom{f}$ and $U\subseteq\dom{c}$ and
                $\img{f}{U}\subseteq\dom{d}$ and
                $\CoordExpr{d}{f}{c}{U}$ is admissible from $\model[G]$ to $\model[H]$ in $C$.
\end{theorem}
\begin{proof}
    $M$ is a topological space and $N$ is a topological space and
        $M$ is a coordinate datum and $N$ is a coordinate datum by \obv{equipped_coordinate_space}.
    $\dom{f}\subseteq\carrier[M]$ by \obv{opens_type}.
    $A\subseteq\atlas[M]$ and $B\subseteq\atlas[N]$ by \obv{presentation_chart_inclusion}.
    Show that if $f$ is coordinate-admissible from $M$ under $G$ to $N$ under $H$ in $C$,
        then for every $p\in\dom{f}$ there exists $c\in A$ such that there exists $d\in B$
            such that there exists $U\in\opens[M]$ such that
                $p\in U$ and $U\subseteq\dom{f}$ and $U\subseteq\dom{c}$ and
                $\img{f}{U}\subseteq\dom{d}$ and
                $\CoordExpr{d}{f}{c}{U}$ is admissible from $\model[G]$ to $\model[H]$ in $C$.
    \begin{subproof}
        Assume $f$ is coordinate-admissible from $M$ under $G$ to $N$ under $H$ in $C$.
        Fix $p\in\dom{f}$.
        $p\in\carrier[M]$ by \obv{subseteq}.
        $f(p)\in\carrier[N]$ by \obv{funs_type_apply}.
        Take $c\in A$ such that $c\in\ChartsAt{M}{p}$ by \obv{presentation_chart_at_point}.
        Take $d\in B$ such that $d\in\ChartsAt{N}{f(p)}$ by \obv{presentation_chart_at_point}.
        Follows by \auto{coordinate_admissible_in_any_charts}.
    \end{subproof}
    Show that if for every $p\in\dom{f}$ there exists $c\in A$ such that there exists $d\in B$
            such that there exists $U\in\opens[M]$ such that
                $p\in U$ and $U\subseteq\dom{f}$ and $U\subseteq\dom{c}$ and
                $\img{f}{U}\subseteq\dom{d}$ and
                $\CoordExpr{d}{f}{c}{U}$ is admissible from $\model[G]$ to $\model[H]$ in $C$,
        then $f$ is coordinate-admissible from $M$ under $G$ to $N$ under $H$ in $C$.
    \begin{subproof}
        Assume for every $p\in\dom{f}$ there exists $c\in A$ such that there exists $d\in B$
            such that there exists $U\in\opens[M]$ such that
                $p\in U$ and $U\subseteq\dom{f}$ and $U\subseteq\dom{c}$ and
                $\img{f}{U}\subseteq\dom{d}$ and
                $\CoordExpr{d}{f}{c}{U}$ is admissible from $\model[G]$ to $\model[H]$ in $C$.
        For every $p\in\dom{f}$ we have
            $f$ is coordinate-admissible at $p$ from $M$ under $G$ to $N$ under $H$ in $C$ by
            \auto{coordinate_admissible_at,charts_at_iff,subseteq}.
        Follows by \obv{coordinate_admissible_map}.
    \end{subproof}
\end{proof}

The same criterion can be read on relative structures over explicitly named
spaces. Equipping copies the carrier, topology, and atlas; it does not require
the original topological records to be exact.

\begin{theorem}\label{coordinate_admissible_equip_iff}
    Let $C$ be a local-map calculus.
    Let $G,H$ be coordinate pseudogroups.
    Let $X,Y$ be topological spaces.
    Let $S$ be a relative coordinate structure on $X$ under $G$.
    Let $T$ be a relative coordinate structure on $Y$ under $H$.
    Suppose $G$ is compatible with the calculus $C$ and $H$ is compatible with the calculus $C$.
    Then $f$ is coordinate-admissible from $\Equip{X}{S}$ under $G$ to $\Equip{Y}{T}$ under $H$ in $C$ iff
        $\dom{f}\in\opens[X]$ and $f\in\funs{\dom{f}}{\carrier[Y]}$ and
        for every $p\in\dom{f}$ there exists $c\in\ChartsAt{S}{p}$ such that
            there exists $d\in\atlas[T]$ such that there exists $U\in\opens[X]$ such that
                $p\in U$ and $U\subseteq\dom{f}$ and $U\subseteq\dom{c}$ and
                $\img{f}{U}\subseteq\dom{d}$ and
                $\CoordExpr{d}{f}{c}{U}$ is admissible from $\model[G]$ to $\model[H]$ in $C$.
\end{theorem}
\begin{proof}
    $S$ is a coordinate datum and $T$ is a coordinate datum by \obv{relative_coordinate_structure}.
    Let $M=\Equip{X}{S}$ and $N=\Equip{Y}{T}$.
    $M$ is an equipped coordinate space and $N$ is an equipped coordinate space by \obv{equip}.
    $M$ has coordinates under $G$ and $N$ has coordinates under $H$ by
        \auto{equip_preserves_coordinates}.
    $\carrier[M]=\carrier[X]$ and $\opens[M]=\opens[X]$ and $\atlas[M]=\atlas[S]$ by
        \auto{equipprojection_carrier,equipprojection_opens,equipprojection_atlas}.
    $\carrier[N]=\carrier[Y]$ and $\opens[N]=\opens[Y]$ and $\atlas[N]=\atlas[T]$ by
        \auto{equipprojection_carrier,equipprojection_opens,equipprojection_atlas}.
    For every $p$ we have $\ChartsAt{M}{p}=\ChartsAt{S}{p}$ by \obv{charts_at}.
    Follows by \auto{coordinate_admissible_map,coordinate_admissible_at}.
\end{proof}

\subsection{Map laws}

\begin{theorem}\label{coordinate_admissible_identity}
    Let $C$ be a local-map calculus.
    Let $G$ be a coordinate pseudogroup.
    Let $M$ be an equipped coordinate space.
    Suppose $G$ is compatible with the calculus $C$ and $M$ has coordinates under $G$.
    Suppose $U\in\opens[M]$.
    Then $\identity{U}$ is coordinate-admissible from $M$ under $G$ to $M$ under $G$ in $C$.
\end{theorem}
\begin{proof}
    $\atlas[M]$ is a coordinate atlas on $M$ under $G$ by \obv{equipped_atlas_on_space}.
    $\model[G]\in\models[C]$ by \auto{compatible_pseudogroup_model,calculus_model}.
    $\dom{\identity{U}}=U$ by \obv{id_dom}.
    $U\subseteq\carrier[M]$ by \obv{opens_type}.
    $\identity{U}\in\funs{U}{U}$ by \obv{id_elem_funs}.
    $\identity{U}\in\funs{\dom{\identity{U}}}{\carrier[M]}$ by \auto{funs_weaken_codom}.
    Show that for every $p\in\dom{\identity{U}}$ we have
        $\identity{U}$ is coordinate-admissible at $p$ from $M$ under $G$ to $M$ under $G$ in $C$.
    \begin{subproof}
        Fix $p\in\dom{\identity{U}}$.
        $p\in\carrier[M]$ by \obv{subseteq}.
        Take $c\in\atlas[M]$ such that $p\in\dom{c}$ by \obv{coordinate_atlas}.
        $c\in\ChartsAt{M}{p}$ by \obv{charts_at_iff}.
        $c$ is an open partial homeomorphism from $M$ to $\model[G]$ by \obv{coordinate_atlas}.
        $c$ is a function by \auto{open_partial_homeomorphism_bijection,bijection_is_function}.
        $\dom{c}\in\opens[M]$ by \obv{open_partial_homeomorphism}.
        Let $V=U\inter\dom{c}$.
        $V\in\opens[M]$ by \auto{opens_inter}.
        $p\in V$ and $V\subseteq U$ and $V\subseteq\dom{c}$ by
            \auto{inter,inter_lower_left,inter_lower_right}.
        $\img{\identity{U}}{V}=V$ by \auto{id_img,inter_absorb_supseteq_left}.
        $\CoordExpr{c}{\identity{U}}{c}{V}=\identity{\img{c}{V}}$ by
            \obv{coordinate_expression_partial_identity}.
        $\img{c}{V}\in\opens[\model[G]]$ by \obv{open_partial_homeomorphism_image_open}.
        $\identity{\img{c}{V}}$ is admissible from $\model[G]$ to $\model[G]$ in $C$ by
            \obv{admissible_open_identity}.
        Follows by \auto{coordinate_admissible_at}.
    \end{subproof}
    Follows by \obv{coordinate_admissible_map}.
\end{proof}

\begin{theorem}\label{coordinate_admissible_restriction}
    Let $C$ be a local-map calculus.
    Let $G,H$ be coordinate pseudogroups.
    Let $M,N$ be equipped coordinate spaces.
    Suppose $f$ is coordinate-admissible from $M$ under $G$ to $N$ under $H$ in $C$.
    Suppose $W\in\opens[M]$.
    Then $\restrl{f}{W}$ is coordinate-admissible from $M$ under $G$ to $N$ under $H$ in $C$.
\end{theorem}
\begin{proof}
    $M$ is a topological space and $M$ is a coordinate datum by \obv{equipped_coordinate_space}.
    $G$ is compatible with the calculus $C$ and $H$ is compatible with the calculus $C$ and
        $M$ has coordinates under $G$ and $N$ has coordinates under $H$ by
        \obv{coordinate_admissible_map}.
    $\atlas[M]$ is a coordinate atlas on $M$ under $G$ by \obv{equipped_atlas_on_space}.
    $f$ is a continuous partial map from $M$ to $N$ by \obv{coordinate_admissible_is_continuous}.
    $f$ is a function by \auto{continuous_partial_map_type,funs_is_function}.
    Let $g=\restrl{f}{W}$.
    $g$ is a continuous partial map from $M$ to $N$ by \auto{continuous_partial_map_restriction}.
    $\dom{g}\in\opens[M]$ and $g\in\funs{\dom{g}}{\carrier[N]}$ by
        \obv{continuous_partial_map,continuous_partial_map_type}.
    $\dom{g}=\dom{f}\inter W$ by \auto{dom_of_restrl_eq_inter}.
    Show that for every $p\in\dom{g}$ we have
        $g$ is coordinate-admissible at $p$ from $M$ under $G$ to $N$ under $H$ in $C$.
    \begin{subproof}
        Fix $p\in\dom{g}$.
        $p\in\dom{f}$ and $p\in W$ by \auto{inter}.
        $f$ is coordinate-admissible at $p$ from $M$ under $G$ to $N$ under $H$ in $C$ by
            \obv{coordinate_admissible_map}.

\begin{theorem}\label{coordinate_admissible_composition}
    Let $C$ be a local-map calculus.
    Let $G,H,K$ be coordinate pseudogroups.
    Let $M,N,P$ be equipped coordinate spaces.
    Suppose $f$ is coordinate-admissible from $M$ under $G$ to $N$ under $H$ in $C$.
    Suppose $g$ is coordinate-admissible from $N$ under $H$ to $P$ under $K$ in $C$.
    Then $g\circ f$ is coordinate-admissible from $M$ under $G$ to $P$ under $K$ in $C$.
\end{theorem}
\begin{proof}
    $M$ is a topological space and $M$ is a coordinate datum and
        $N$ is a coordinate datum by \obv{equipped_coordinate_space}.
    $G$ is compatible with the calculus $C$ and $H$ is compatible with the calculus $C$ and
        $K$ is compatible with the calculus $C$ and
        $M$ has coordinates under $G$ and $N$ has coordinates under $H$ and
        $P$ has coordinates under $K$ by \obv{coordinate_admissible_map}.
    $\atlas[M]$ is a coordinate atlas on $M$ under $G$ and
        $\atlas[N]$ is a coordinate atlas on $N$ under $H$ by \obv{equipped_atlas_on_space}.
    $f$ is a continuous partial map from $M$ to $N$ and
        $g$ is a continuous partial map from $N$ to $P$ by \obv{coordinate_admissible_is_continuous}.
    $f$ and $g$ are functions by \auto{continuous_partial_map_type,funs_is_function}.
    Let $h=g\circ f$.
    $h$ is a continuous partial map from $M$ to $P$ by \auto{continuous_partial_map_composition}.
    $\dom{h}\in\opens[M]$ and $h\in\funs{\dom{h}}{\carrier[P]}$ by
        \obv{continuous_partial_map,continuous_partial_map_type}.
    $\dom{h}=\preimg{f}{\dom{g}}$ by \auto{relation_composition_domain}.
    $\dom{f}\in\opens[M]$ and $\dom{f}\subseteq\carrier[M]$ by
        \auto{continuous_partial_map,opens_type}.
    Show that for every $p\in\dom{h}$ we have
        $h$ is coordinate-admissible at $p$ from $M$ under $G$ to $P$ under $K$ in $C$.
    \begin{subproof}
        Fix $p\in\dom{h}$.
        $p\in\dom{f}$ and $f(p)\in\dom{g}$ by
            \auto{preimg_iff,function_apply_iff,preimg_subseteq_dom,subseteq}.
        $g$ is coordinate-admissible at $f(p)$ from $N$ under $H$ to $P$ under $K$ in $C$ by
            \obv{coordinate_admissible_map}.
        Take $d,e,V$ such that $d\in\ChartsAt{N}{f(p)}$ and $e\in\atlas[P]$ and
            $V\in\opens[N]$ and $f(p)\in V$ and $V\subseteq\dom{g}$ and $V\subseteq\dom{d}$ and
            $\img{g}{V}\subseteq\dom{e}$ and
            $\CoordExpr{e}{g}{d}{V}$ is admissible from $\model[H]$ to $\model[K]$ in $C$ by
            \auto{coordinate_admissible_at}.
        $p\in\carrier[M]$ by \obv{subseteq}.
        Take $c\in\atlas[M]$ such that $p\in\dom{c}$ by \obv{coordinate_atlas}.
        $c\in\ChartsAt{M}{p}$ by \obv{charts_at_iff}.
        Take $U\in\opens[M]$ such that $p\in U$ and $U\subseteq\dom{f}$ and
            $U\subseteq\dom{c}$ and $\img{f}{U}\subseteq\dom{d}$ and
            $\CoordExpr{d}{f}{c}{U}$ is admissible from $\model[G]$ to $\model[H]$ in $C$ by
            \obv{coordinate_admissible_in_any_charts}.
        $\preimg{f}{V}\in\opens[M]$ by \obv{continuous_partial_map_preimage_open}.
        $p\in\preimg{f}{V}$ by \auto{preimg_iff,function_apply_elim}.
        Let $W=U\inter\preimg{f}{V}$.
        $W\in\opens[M]$ by \auto{opens_inter}.
        $p\in W$ and $W\subseteq U$ and $W\subseteq\preimg{f}{V}$ by \auto{inter,subseteq}.
        $W\subseteq\dom{c}$ by \auto{subseteq_transitive}.
        $\preimg{f}{V}\subseteq\preimg{f}{\dom{g}}$ by \obv{preimg_subseteq}.
        $W\subseteq\dom{h}$ by \auto{subseteq_transitive}.
        $\img{f}{W}\subseteq V$ by \auto{img_iff,preimg_iff,function_rightunique,subseteq}.
        $\img{h}{W}\subseteq\img{g}{V}$ by \auto{img_iff,circ_iff,subseteq}.
        $\img{h}{W}\subseteq\dom{e}$ by \auto{subseteq_transitive}.
        $d\in\atlas[N]$ by \auto{charts_at_iff}.
        $c$ is an open partial homeomorphism from $M$ to $\model[G]$ and
            $d$ is an open partial homeomorphism from $N$ to $\model[H]$ by \obv{coordinate_atlas}.
        $\atlas[P]$ is a coordinate atlas on $P$ under $K$ by \obv{equipped_atlas_on_space}.
        $c$ is an injection and $d$ is an injection and $e$ is a function by
            \auto{coordinate_atlas,open_partial_homeomorphism_bijection,bijections,bijection_is_function}.
        $\img{c}{W}\in\opens[\model[G]]$ by \obv{open_partial_homeomorphism_image_open}.
        $\restrl{\CoordExpr{d}{f}{c}{U}}{\img{c}{W}}$ is admissible
            from $\model[G]$ to $\model[H]$ in $C$ by \obv{admissible_map_restriction}.
        $\CoordExpr{d}{f}{c}{W}=\restrl{\CoordExpr{d}{f}{c}{U}}{\img{c}{W}}$ by
            \obv{coordinate_expression_restriction}.
        $\CoordExpr{e}{g}{d}{V}\circ\CoordExpr{d}{f}{c}{W}$ is admissible
            from $\model[G]$ to $\model[K]$ in $C$ by \auto{admissible_map_composition}.
        $\CoordExpr{e}{h}{c}{W}=\CoordExpr{e}{g}{d}{V}\circ\CoordExpr{d}{f}{c}{W}$ by
            \auto{coordinate_expression_composition}.
        Follows by \auto{coordinate_admissible_at}.
    \end{subproof}
    Follows by \auto{coordinate_admissible_map}.
\end{proof}

\subsection{Changing the calculus}

Inclusion enlarges the admissible maps without changing either endpoint
structure or its pseudogroup.

\begin{theorem}\label{coordinate_admissible_subcalculus}
    Let $C$ be a local-map calculus.
    Let $D$ be a local-map calculus.
    Let $G,H$ be coordinate pseudogroups.
    Let $M,N$ be equipped coordinate spaces.
    Suppose $C$ is a subcalculus of $D$.
    Suppose $f$ is coordinate-admissible from $M$ under $G$ to $N$ under $H$ in $C$.
    Then $f$ is coordinate-admissible from $M$ under $G$ to $N$ under $H$ in $D$.
\end{theorem}
\begin{proof}
    $G$ is compatible with the calculus $C$ and $H$ is compatible with the calculus $C$ and
        $M$ has coordinates under $G$ and $N$ has coordinates under $H$ and
        $\dom{f}\in\opens[M]$ and $f\in\funs{\dom{f}}{\carrier[N]}$ by
        \obv{coordinate_admissible_map}.
    $G$ is compatible with the calculus $D$ and $H$ is compatible with the calculus $D$ by
        \obv{subcalculus_pseudogroup_compatible}.
    For every $p\in\dom{f}$ we have
        $f$ is coordinate-admissible at $p$ from $M$ under $G$ to $N$ under $H$ in $C$ by
        \obv{coordinate_admissible_map}.
    For every $p\in\dom{f}$ we have
        $f$ is coordinate-admissible at $p$ from $M$ under $G$ to $N$ under $H$ in $D$ by
        \auto{coordinate_admissible_at,subcalculus_admissibility}.
    Follows by \obv{coordinate_admissible_map}.
\end{proof}

\begin{theorem}\label{continuous_calculus_coordinate_maps}
    Let $P$ be a topological model family.
    Let $G,H$ be coordinate pseudogroups.
    Let $M,N$ be equipped coordinate spaces.
    Suppose $\model[G],\model[H]\in P$.
    Suppose $M$ has coordinates under $G$ and $N$ has coordinates under $H$.
    Then $f$ is coordinate-admissible from $M$ under $G$ to $N$ under $H$ in $\ContinuousCalculus{P}$ iff
        $f$ is a continuous partial map from $M$ to $N$.
\end{theorem}
\begin{proof}
    $\ContinuousCalculus{P}$ is a local-map calculus by \obv{continuous_calculus}.
    If $f$ is coordinate-admissible from $M$ under $G$ to $N$ under $H$ in $\ContinuousCalculus{P}$,
        then $f$ is a continuous partial map from $M$ to $N$ by
        \obv{coordinate_admissible_is_continuous}.
    Show that if $f$ is a continuous partial map from $M$ to $N$, then
        $f$ is coordinate-admissible from $M$ under $G$ to $N$ under $H$ in $\ContinuousCalculus{P}$.
    \begin{subproof}
        Assume $f$ is a continuous partial map from $M$ to $N$.
        $G$ is compatible with the calculus $\ContinuousCalculus{P}$ and
            $H$ is compatible with the calculus $\ContinuousCalculus{P}$ by
            \obv{pseudogroup_continuous_calculus_compatible}.
        $M$ is a topological space and $N$ is a topological space and
            $M$ is a coordinate datum by \obv{equipped_coordinate_space}.
        $\atlas[M]$ is a coordinate atlas on $M$ under $G$ and
            $\atlas[N]$ is a coordinate atlas on $N$ under $H$ by \obv{equipped_atlas_on_space}.
        $f\in\funs{\dom{f}}{\carrier[N]}$ and $\dom{f}\in\opens[M]$ by
            \obv{continuous_partial_map_type,continuous_partial_map}.
        $f$ is a function by \obv{funs_is_function}.
        $\dom{f}\subseteq\carrier[M]$ by \obv{opens_type}.
        Show that for every $p\in\dom{f}$ we have
            $f$ is coordinate-admissible at $p$ from $M$ under $G$ to $N$ under $H$ in $\ContinuousCalculus{P}$.
        \begin{subproof}

            $p\in\carrier[M]$ by \obv{subseteq}.
            $f(p)\in\carrier[N]$ by \obv{funs_type_apply}.
            Take $c\in\atlas[M]$ such that $p\in\dom{c}$ by \obv{coordinate_atlas}.

\subsection{Locality and the empty map}

\begin{theorem}\label{coordinate_admissible_locality}
    Let $C$ be a local-map calculus.
    Let $G,H$ be coordinate pseudogroups.
    Let $M,N$ be equipped coordinate spaces.
    Suppose $G$ is compatible with the calculus $C$ and $H$ is compatible with the calculus $C$.
    Suppose $M$ has coordinates under $G$ and $N$ has coordinates under $H$.
    Suppose $f\in\funs{\dom{f}}{\carrier[N]}$ and $\dom{f}\in\opens[M]$.
    Suppose for every $p\in\dom{f}$ there exists $W\in B$ such that $p\in W$.
    Suppose for every $W\in B$ we have
        $\restrl{f}{W}$ is coordinate-admissible from $M$ under $G$ to $N$ under $H$ in $C$.
    Then $f$ is coordinate-admissible from $M$ under $G$ to $N$ under $H$ in $C$.
\end{theorem}
\begin{proof}
    $f$ is a function by \obv{funs_is_function}.
    $M$ is a coordinate datum by \obv{equipped_coordinate_space}.
    $\atlas[M]$ is a coordinate atlas on $M$ under $G$ and
        $\atlas[N]$ is a coordinate atlas on $N$ under $H$ by \obv{equipped_atlas_on_space}.
    Show that for every $p\in\dom{f}$ we have
        $f$ is coordinate-admissible at $p$ from $M$ under $G$ to $N$ under $H$ in $C$.
    \begin{subproof}
        Fix $p\in\dom{f}$.
        Take $W\in B$ such that $p\in W$.
        Let $g=\restrl{f}{W}$.
        $\dom{g}=\dom{f}\inter W$ by \auto{dom_of_restrl_eq_inter}.
        $p\in\dom{g}$ by \auto{inter}.
        $g$ is coordinate-admissible from $M$ under $G$ to $N$ under $H$ in $C$ by \auto.
        $g$ is coordinate-admissible at $p$ from $M$ under $G$ to $N$ under $H$ in $C$ by
            \obv{coordinate_admissible_map}.
        Take $c,d,U$ such that $c\in\ChartsAt{M}{p}$ and $d\in\atlas[N]$ and
            $U\in\opens[M]$ and $p\in U$ and $U\subseteq\dom{g}$ and $U\subseteq\dom{c}$ and
            $\img{g}{U}\subseteq\dom{d}$ and
            $\CoordExpr{d}{g}{c}{U}$ is admissible from $\model[G]$ to $\model[H]$ in $C$
            by \auto{coordinate_admissible_at}.
        $U\subseteq W$ and $U\subseteq\dom{f}$ by \auto{subseteq,inter}.
        $\img{f}{U}=\img{g}{U}$ by \auto{setext,img_iff,restrl_iff,subseteq}.
        $c\in\atlas[M]$ by \auto{charts_at_iff}.
        $c$ is a function and $d$ is a function by
            \auto{coordinate_atlas,open_partial_homeomorphism_bijection,bijection_is_function}.
        $\CoordExpr{d}{g}{c}{U}=\CoordExpr{d}{f}{c}{U}$ by
            \auto{coordinate_expression_restrict_map}.
        Follows by \auto{coordinate_admissible_at}.
    \end{subproof}
    Follows by \obv{coordinate_admissible_map}.
\end{proof}
