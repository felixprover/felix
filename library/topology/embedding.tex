\section{Topological embeddings}

\import{topology/open-map.tex}
\import{topology/subspace-continuity.tex}
\import{set/partial-map.tex}

An embedding identifies its source with a subspace of its target. The graph
does not change when its codomain is restricted to its image.

\begin{definition}\label{topological_embedding}
    $f$ is a topological embedding from $X$ to $Y$ iff
        $X$ is a topological space and $Y$ is a topological space and
        $f\in\funs{\carrier[X]}{\carrier[Y]}$ and
        $f$ is a homeomorphism from $X$ to
            $\subspaceTopology{Y}{\ran{f}}$.
\end{definition}

\begin{definition}\label{open_embedding}
    $f$ is an open embedding from $X$ to $Y$ iff
        $f$ is a topological embedding from $X$ to $Y$ and
        $\ran{f}\in\opens[Y]$.
\end{definition}

\begin{proposition}\label{topological_embedding_continuous}
    Suppose $f$ is a topological embedding from $X$ to $Y$.
    Then $f$ is continuous from $X$ to $Y$.
\end{proposition}
\begin{proof}
    $X$ is a topological space and $Y$ is a topological space and
        $f$ is a homeomorphism from $X$ to
            $\subspaceTopology{Y}{\ran{f}}$ by
        \obv{topological_embedding}.
    $f$ is continuous from $X$ to $\subspaceTopology{Y}{\ran{f}}$
        by \obv{homeomorphism}.
    Follows by \obv{continuous_subspace_codomain_inclusion}.
\end{proof}

\subsection{Homeomorphisms onto subspaces}

\begin{classification}\label{homeomorphism_restriction_to_image}
    Suppose $f$ is a homeomorphism from $X$ to $Y$.
    Suppose $A\subseteq\carrier[X]$.
    Then $\restrl{f}{A}$ is a homeomorphism from
        $\subspaceTopology{X}{A}$ to $\subspaceTopology{Y}{\img{f}{A}}$.
\end{classification}
\begin{proof}
    Let $g=\restrl{f}{A}$.
    Let $B=\img{f}{A}$.
    Let $S=\subspaceTopology{X}{A}$.
    Let $T=\subspaceTopology{Y}{B}$.
    $f$ is continuous from $X$ to $Y$ and
        $\converse{f}$ is continuous from $Y$ to $X$ and
        $f\in\bijections{\carrier[X]}{\carrier[Y]}$ by
        \obv{homeomorphism}.
    $X$ is a topological space and $Y$ is a topological space by
        \obv{continuous_map}.
    $f$ is a function and $f$ is injective by
        \obv{bijection_is_function,bijections}.
    $f\in\funs{\carrier[X]}{\carrier[Y]}$ by \obv{continuous_map}.
    $\ran{f}\subseteq\carrier[Y]$ by \obv{funs_ran}.
    $B\subseteq\ran{f}$ by \auto{img_subseteq_ran}.
    Hence $B\subseteq\carrier[Y]$ by \obv{subseteq_transitive}.
    $\carrier[S]=A$ and $\carrier[T]=B$ by
        \auto{subspace_topology_carrier_of_subset}.
    $g\in\bijections{A}{B}$ by
        \auto{bijection_restriction_to_image}.
    $\ran{g}=B$ by \auto{restriction_range}.
    $g$ is continuous from $S$ to $Y$ by
        \auto{continuous_domain_restriction_on_subset}.
    Hence $g$ is continuous from $S$ to $T$ by
        \auto{continuous_codomain_corestriction,subseteq_refl}.
    $\converse{g}=\restrl{\converse{f}}{B}$ by
        \auto{restriction_converse}.
    Hence $\converse{g}$ is continuous from $T$ to $X$ by
        \auto{continuous_domain_restriction_on_subset}.
    $\dom{g}=A$ by \obv{bijections_dom}.
    $\ran{\converse{g}}=A$ by \obv{ran_converse}.
    Hence $\converse{g}$ is continuous from $T$ to $S$ by
        \auto{continuous_codomain_corestriction,subseteq_refl}.
    Follows by \obv{homeomorphism}.
\end{proof}

\begin{theorem}\label{homeomorphism_is_open_embedding}
    Suppose $f$ is a homeomorphism from $X$ to $Y$.
    Then $f$ is an open embedding from $X$ to $Y$.
\end{theorem}
\begin{proof}
    $f$ is continuous from $X$ to $Y$ and
        $\converse{f}$ is continuous from $Y$ to $X$ and
        $f\in\bijections{\carrier[X]}{\carrier[Y]}$ by
        \obv{homeomorphism}.
    $X$ is a topological space and $Y$ is a topological space and
        $f\in\funs{\carrier[X]}{\carrier[Y]}$ by \obv{continuous_map}.
    $\ran{f}=\carrier[Y]$ by \auto{bijections,ran_of_surj}.
    Let $T=\subspaceTopology{Y}{\ran{f}}$.
    $\carrier[T]=\ran{f}$ by
        \auto{subspace_topology_carrier_of_subset,subseteq_refl}.
    $f$ is continuous from $X$ to $T$ by
        \auto{continuous_codomain_corestriction,subseteq_refl}.
    $\converse{f}\in\funs{\carrier[Y]}{\carrier[X]}$ by
        \obv{continuous_map}.
    Hence $\converse{f}$ is a function and
        $\dom{\converse{f}}=\carrier[Y]$ by \obv{funs_is_function,funs}.
    $\restrl{\converse{f}}{\carrier[Y]}=\converse{f}$ by
        \auto{restrl_by_dom}.
    Hence $\converse{f}$ is continuous from $T$ to $X$ by
        \auto{continuous_domain_restriction_on_subset,subseteq_refl}.
    Hence $f$ is a homeomorphism from $X$ to $T$ by \obv{homeomorphism}.
    $f$ is a topological embedding from $X$ to $Y$ by
        \obv{topological_embedding}.
    $\ran{f}\in\opens[Y]$ by \obv{carrier_open}.
    Follows by \obv{open_embedding}.
\end{proof}

\begin{theorem}\label{identity_is_open_embedding}
    Let $X$ be a topological space.
    Then $\identity{\carrier[X]}$ is an open embedding from $X$ to $X$.
\end{theorem}
\begin{proof}
    $\identity{\carrier[X]}$ is continuous from $X$ to $X$ by
        \obv{identity_continuous}.
    $\identity{\carrier[X]}\in\bijections{\carrier[X]}{\carrier[X]}$
        by \obv{id_elem_bijections}.
    $\converse{\identity{\carrier[X]}}=\identity{\carrier[X]}$ by
        \obv{converse_identity}.
    Hence $\identity{\carrier[X]}$ is a homeomorphism from $X$ to $X$
        by \obv{homeomorphism}.
    Follows by \obv{homeomorphism_is_open_embedding}.
\end{proof}

\subsection{Restriction and composition}

\begin{theorem}\label{topological_embedding_restriction}
    Suppose $f$ is a topological embedding from $X$ to $Y$.
    Suppose $A\subseteq\carrier[X]$.
    Then $\restrl{f}{A}$ is a topological embedding from
        $\subspaceTopology{X}{A}$ to $Y$.
\end{theorem}
\begin{proof}
    Let $g=\restrl{f}{A}$.
    Let $S=\subspaceTopology{X}{A}$.
    $X$ is a topological space and $Y$ is a topological space and
        $f\in\funs{\carrier[X]}{\carrier[Y]}$ by
        \obv{topological_embedding}.
    $S$ is a topological space by
        \auto{subspace_topology_is_topological_space}.
    $\carrier[S]=A$ by \auto{subspace_topology_carrier_of_subset}.
    $g\in\funs{\carrier[S]}{\carrier[Y]}$ by \auto{funs_restrl}.
    $f$ is a function by \obv{funs_is_function}.
    $\ran{g}=\img{f}{A}$ by \auto{restriction_range}.
    $\ran{g}\subseteq\ran{f}$ by \auto{img_subseteq_ran}.
    $\ran{f}\subseteq\carrier[Y]$ by \obv{funs_ran}.
    $f$ is a homeomorphism from $X$ to $\subspaceTopology{Y}{\ran{f}}$
        by \obv{topological_embedding}.
    Hence $g$ is a homeomorphism from $S$ to
        $\subspaceTopology{\subspaceTopology{Y}{\ran{f}}}{\ran{g}}$
        by \auto{homeomorphism_restriction_to_image}.
    $\subspaceTopology{\subspaceTopology{Y}{\ran{f}}}{\ran{g}}
        =\subspaceTopology{Y}{\ran{g}}$ by
        \obv{nested_subspace_topology}.
    Follows by \obv{topological_embedding}.
\end{proof}

\begin{theorem}\label{topological_embedding_composition}
    Suppose $f$ is a topological embedding from $X$ to $Y$ and
        $g$ is a topological embedding from $Y$ to $Z$.
    Then $g\circ f$ is a topological embedding from $X$ to $Z$.
\end{theorem}
\begin{proof}
    $f\in\funs{\carrier[X]}{\carrier[Y]}$ and
        $g\in\funs{\carrier[Y]}{\carrier[Z]}$ by
        \obv{topological_embedding}.
    $g\circ f\in\funs{\carrier[X]}{\carrier[Z]}$ by \obv{funs_circ}.
    $f$ and $g$ are functions by \obv{funs_is_function}.
    $\ran{f}\subseteq\carrier[Y]$ and
        $\ran{g}\subseteq\carrier[Z]$ by \obv{funs_ran}.
    Let $B=\img{g}{\ran{f}}$.
    $B\subseteq\ran{g}$ by \auto{img_subseteq_ran}.
    $g$ is a homeomorphism from $Y$ to $\subspaceTopology{Z}{\ran{g}}$
        by \obv{topological_embedding}.
    Hence $\restrl{g}{\ran{f}}$ is a homeomorphism from
        $\subspaceTopology{Y}{\ran{f}}$ to
        $\subspaceTopology{\subspaceTopology{Z}{\ran{g}}}{B}$ by
        \auto{homeomorphism_restriction_to_image}.
    $\subspaceTopology{\subspaceTopology{Z}{\ran{g}}}{B}
        =\subspaceTopology{Z}{B}$ by \obv{nested_subspace_topology}.
    $f$ is a homeomorphism from $X$ to $\subspaceTopology{Y}{\ran{f}}$
        by \obv{topological_embedding}.
    Hence $(\restrl{g}{\ran{f}})\circ f$ is a homeomorphism from
        $X$ to $\subspaceTopology{Z}{B}$ by
        \obv{homeomorphism_composition}.
    $(\restrl{g}{\ran{f}})\circ f=g\circ f$ by
        \obv{composition_restrict_intermediate_range}.
    $\ran{g\circ f}=B$ by \auto{relation_composition_range}.
    Follows by \obv{topological_embedding}.
\end{proof}

\subsection{Open embeddings}

\begin{theorem}\label{open_embedding_is_open_map}
    Suppose $f$ is an open embedding from $X$ to $Y$.
    Then $f$ is an open map from $X$ to $Y$.
\end{theorem}
\begin{proof}
    $f$ is a topological embedding from $X$ to $Y$ and
        $\ran{f}\in\opens[Y]$ by \obv{open_embedding}.
    $X$ is a topological space and $Y$ is a topological space and
        $f\in\funs{\carrier[X]}{\carrier[Y]}$ and
        $f$ is a homeomorphism from $X$ to $\subspaceTopology{Y}{\ran{f}}$
        by \obv{topological_embedding}.
    Hence $f$ is an open map from $X$ to $\subspaceTopology{Y}{\ran{f}}$
        by \obv{homeomorphism_is_open_map}.
    Show that for every $U\in\opens[X]$ we have $\img{f}{U}\in\opens[Y]$.
    \begin{subproof}
        Fix $U\in\opens[X]$.
        $\img{f}{U}\in\opens[\subspaceTopology{Y}{\ran{f}}]$ by
            \obv{open_map_image_open}.
        Follows by \auto{open_subspace_open_iff}.
    \end{subproof}
    Follows by \obv{open_map}.
\end{proof}

\begin{theorem}\label{continuous_injective_open_map_is_open_embedding}
    Let $X,Y$ be topological spaces.
    Suppose $f$ is continuous from $X$ to $Y$ and $f$ is injective.
    Suppose $f$ is an open map from $X$ to $Y$.
    Then $f$ is an open embedding from $X$ to $Y$.
\end{theorem}
\begin{proof}
    $f\in\funs{\carrier[X]}{\carrier[Y]}$ by \obv{continuous_map}.
    $f$ is a function and $\dom{f}=\carrier[X]$ by
        \obv{funs_is_function,funs}.
    $\ran{f}\subseteq\carrier[Y]$ by \obv{funs_ran}.
    Let $T=\subspaceTopology{Y}{\ran{f}}$.
    $\carrier[T]=\ran{f}$ by \auto{subspace_topology_carrier_of_subset}.
    $f$ is continuous from $X$ to $T$ by
        \auto{continuous_codomain_corestriction,subseteq_refl}.
    $f\in\bijections{\carrier[X]}{\carrier[T]}$ by
        \auto{injective_function_bijection_to_range}.
    $f\in\funs{\carrier[X]}{\carrier[T]}$ by \obv{continuous_map}.
    Show that for every $U\in\opens[X]$ we have $\img{f}{U}\in\opens[T]$.
    \begin{subproof}
        Fix $U\in\opens[X]$.
        $\img{f}{U}\in\opens[Y]$ by \obv{open_map_image_open}.
        $\img{f}{U}\subseteq\ran{f}$ by \obv{img_subseteq_ran}.
        $\img{f}{U}=\img{f}{U}\inter\carrier[T]$ by
            \auto{inter_absorb_supseteq_right}.
        Follows by \auto{subspace_open_iff}.
    \end{subproof}
    Hence $f$ is an open map from $X$ to $T$ by \obv{open_map}.
    Hence $f$ is a homeomorphism from $X$ to $T$ by
        \obv{continuous_bijection_open_map_is_homeomorphism}.
    $f$ is a topological embedding from $X$ to $Y$ by
        \obv{topological_embedding}.
    $\carrier[X]\in\opens[X]$ by \obv{carrier_open}.
    Hence $\img{f}{\carrier[X]}\in\opens[Y]$ by \obv{open_map_image_open}.
    $\img{f}{\carrier[X]}=\ran{f}$ by \auto{img_dom}.
    Follows by \obv{open_embedding}.
\end{proof}

\begin{corollary}\label{topological_embedding_corestriction}
    Suppose $f$ is a topological embedding from $X$ to $Y$.
    Then $f$ is an open embedding from $X$ to $\subspaceTopology{Y}{\ran{f}}$.
\end{corollary}
\begin{proof}
    $f$ is a homeomorphism from $X$ to $\subspaceTopology{Y}{\ran{f}}$
        by \obv{topological_embedding}.
    Follows by \obv{homeomorphism_is_open_embedding}.
\end{proof}

\begin{theorem}\label{open_embedding_restriction}
    Suppose $f$ is an open embedding from $X$ to $Y$.
    Suppose $A\in\opens[X]$.
    Then $\restrl{f}{A}$ is an open embedding from
        $\subspaceTopology{X}{A}$ to $Y$.
\end{theorem}
\begin{proof}
    $f$ is a topological embedding from $X$ to $Y$ by \obv{open_embedding}.
    $A\subseteq\carrier[X]$ by \obv{opens_type}.
    $\restrl{f}{A}$ is a topological embedding from
        $\subspaceTopology{X}{A}$ to $Y$ by
        \obv{topological_embedding_restriction}.
    $f$ is an open map from $X$ to $Y$ by \obv{open_embedding_is_open_map}.
    $\img{f}{A}\in\opens[Y]$ by \obv{open_map_image_open}.
    $f\in\funs{\carrier[X]}{\carrier[Y]}$ by \obv{open_map}.
    $f$ is a function by \obv{funs_is_function}.
    $\ran{\restrl{f}{A}}=\img{f}{A}$ by \obv{restriction_range}.
    Follows by \obv{open_embedding}.
\end{proof}

\begin{theorem}\label{open_embedding_composition}
    Suppose $f$ is an open embedding from $X$ to $Y$ and
        $g$ is an open embedding from $Y$ to $Z$.
    Then $g\circ f$ is an open embedding from $X$ to $Z$.
\end{theorem}
\begin{proof}
    $f$ is a topological embedding from $X$ to $Y$ and
        $g$ is a topological embedding from $Y$ to $Z$ by
        \obv{open_embedding}.
    $g\circ f$ is a topological embedding from $X$ to $Z$ by
        \obv{topological_embedding_composition}.
    $\ran{f}\in\opens[Y]$ by \obv{open_embedding}.
    $g$ is an open map from $Y$ to $Z$ by \obv{open_embedding_is_open_map}.
    $\img{g}{\ran{f}}\in\opens[Z]$ by \obv{open_map_image_open}.
    $f\in\funs{\carrier[X]}{\carrier[Y]}$ and
        $g\in\funs{\carrier[Y]}{\carrier[Z]}$ by
        \obv{topological_embedding}.
    $f$ and $g$ are functions by \obv{funs_is_function}.
    $\ran{g\circ f}=\img{g}{\ran{f}}$ by \obv{relation_composition_range}.
    Follows by \obv{open_embedding}.
\end{proof}

\begin{theorem}\label{topological_embedding_transport}
    Suppose $f$ is a topological embedding from $X$ to $Y$.
    Suppose $a$ is a homeomorphism from $W$ to $X$ and
        $b$ is a homeomorphism from $Y$ to $Z$.
    Then $b\circ(f\circ a)$ is a topological embedding from $W$ to $Z$.
\end{theorem}
\begin{proof}
    $a$ is an open embedding from $W$ to $X$ and
        $b$ is an open embedding from $Y$ to $Z$ by
        \obv{homeomorphism_is_open_embedding}.
    $a$ is a topological embedding from $W$ to $X$ and
        $b$ is a topological embedding from $Y$ to $Z$ by
        \obv{open_embedding}.
    $f\circ a$ is a topological embedding from $W$ to $Y$ by
        \obv{topological_embedding_composition}.
    Follows by \obv{topological_embedding_composition}.
\end{proof}
