\section{Frobenius on the power-bounded ring}

\begin{classification}\label{pseudouniformizer_in_power_bounded_ring}
    Let $R$ be a commutative topological ring.
    Suppose $R$ is a nonarchimedean topological ring.
    Suppose $\pi$ is a pseudouniformizer of $R$.
    Then $\pi\in\carrier[\PowerBoundedRing{R}]$.
\end{classification}
\begin{proof}
    $\pi\in\PowerBoundedCarrier{R}$ by
        \obv{pseudouniformizer_in_power_bounded_carrier}.
    $\carrier[\PowerBoundedRing{R}]=\PowerBoundedCarrier{R}$ by
        \auto{power_bounded_ring_agreement}.
\end{proof}

\begin{theorem}\label{power_bounded_principal_power_kernel}
    Let $R$ be a commutative topological ring.
    Suppose $R$ is a nonarchimedean topological ring.
    Let $m$ be a natural number such that $m\neq\emptyset$.
    Let $x,\pi\in\carrier[\PowerBoundedRing{R}]$.
    Suppose $\pi$ is a unit of $\MultiplicativeMonoid{R}$.
    Suppose $\ringPower{\PowerBoundedRing{R}}{x}{m}
        \in\principalIdeal{\PowerBoundedRing{R}}{
            \ringPower{\PowerBoundedRing{R}}{\pi}{m}}$.
    Then $x\in\principalIdeal{\PowerBoundedRing{R}}{\pi}$.
\end{theorem}
\begin{proof}
    Let $S=\PowerBoundedRing{R}$.
    $R$ is a commutative ring by
        \obv{commutative_topological_ring_is_commutative_ring}.
    Hence $R$ is a ring by \obv{commutative_ring_is_ring}.
    $\PowerBoundedRing{R}$ is a commutative ring by
        \obv{power_bounded_ring_is_exact_commutative_ring}.
    Hence $S$ is a commutative ring.
    Hence $S$ is a ring by \obv{commutative_ring_is_ring}.
    $\carrier[S]=\PowerBoundedCarrier{R}$ by
        \auto{power_bounded_ring_agreement}.
    Hence $x,\pi\in\PowerBoundedCarrier{R}$.
    Hence $x,\pi\in\carrier[R]$ by
        \obv{power_bounded_carrier}.

    $\ringPower{S}{\pi}{m}\in\carrier[S]$ by
        \obv{ring_power_type}.
    $\principalIdeal{S}{\ringPower{S}{\pi}{m}}
        =\{\ringMul[S](\ringPower{S}{\pi}{m},c)
            \mid c\in\carrier[S]\}$ by
        \obv{principal_ideal_eq_multiple_carrier}.
    Hence $\ringPower{S}{x}{m}
        \in\{\ringMul[S](\ringPower{S}{\pi}{m},c)
            \mid c\in\carrier[S]\}$.
    Take $c\in\carrier[S]$ such that
        $\ringPower{S}{x}{m}
            =\ringMul[S](\ringPower{S}{\pi}{m},c)$ by \auto.
    Hence $c\in\PowerBoundedCarrier{R}$.
    Hence $c\in\carrier[R]$ by
        \obv{power_bounded_carrier}.

    $\ringPower{S}{x}{m}=\ringPower{R}{x}{m}$ and
        $\ringPower{S}{\pi}{m}=\ringPower{R}{\pi}{m}$ by
        \obv{power_bounded_ring_power_agreement}.
    $\ringPower{S}{\pi}{m}\in\PowerBoundedCarrier{R}$.
    Hence $\ringMul[S](\ringPower{S}{\pi}{m},c)
        =\ringMul[R](\ringPower{S}{\pi}{m},c)$ by
        \auto{power_bounded_ring_agreement}.
    Hence $\ringPower{R}{x}{m}
        =\ringMul[R](\ringPower{R}{\pi}{m},c)$.

    Let $M=\MultiplicativeMonoid{R}$.
    $M$ is a monoid by \obv{multiplicative_monoid_is_exact_monoid}.
    $\carrier[M]=\carrier[R]$ and $\mul[M]=\ringMul[R]$ and
        $\neutral[M]=\ringOne[R]$ by
        \auto{multiplicative_monoid_fields}.
    Hence $\pi$ is a unit of $M$.
    Take $v\in\carrier[M]$ such that
        $\mul[M](\pi,v)=\neutral[M]$ and
        $\mul[M](v,\pi)=\neutral[M]$ by
        \obv{monoid_unit_has_inverse}.
    Hence $v\in\carrier[R]$ and
        $\ringMul[R](\pi,v)=\ringOne[R]$.
    Let $z=\ringMul[R](x,v)$.
    Hence $z\in\carrier[R]$ by \auto{ring_mul_type}.
    $\ringPower{R}{z}{m}=c$ by
        \auto{commutative_ring_power_after_multiplying_inverse}.
    $c$ is power bounded in $R$ by
        \auto{power_bounded_carrier_iff}.
    Hence $\ringPower{R}{z}{m}$ is power bounded in $R$.
    Hence $z$ is power bounded in $R$ by \obv{power_bounded_root}.
    Hence $z\in\PowerBoundedCarrier{R}$ by
        \obv{power_bounded_carrier_iff}.
    Hence $z\in\carrier[S]$.

    $\ringMul[R](\pi,z)
        =\ringMul[R](\pi,\ringMul[R](x,v))$.
    $\ringMul[R](\pi,\ringMul[R](x,v))
        =\ringMul[R](\ringMul[R](\pi,x),v)$ by
        \obv{ring_mul_associative}.
    $\ringMul[R](\pi,x)=\ringMul[R](x,\pi)$ by
        \obv{commutative_ring_mul_commutative}.
    $\ringMul[R](\ringMul[R](x,\pi),v)
        =\ringMul[R](x,\ringMul[R](\pi,v))$ by
        \obv{ring_mul_associative}.
    Hence $\ringMul[R](\pi,z)=\ringMul[R](x,\ringOne[R])$.
    Hence $\ringMul[R](\pi,z)=x$ by \obv{ring_mul_one_right}.
    $\ringMul[S](\pi,z)=\ringMul[R](\pi,z)$ by
        \auto{power_bounded_ring_agreement}.
    Hence $x=\ringMul[S](\pi,z)$.
    $\principalIdeal{S}{\pi}
        =\{\ringMul[S](\pi,d)\mid d\in\carrier[S]\}$ by
        \obv{principal_ideal_eq_multiple_carrier}.
    Follows by \auto.
\end{proof}

\begin{proposition}\label{power_bounded_principal_frobenius_type}
    Let $R$ be a commutative topological ring.
    Suppose $R$ is a nonarchimedean topological ring.
    Let $S=\PowerBoundedRing{R}$.
    Let $p$ be a prime natural number.
    Suppose $\pi$ is a pseudouniformizer of $R$.
    Suppose $\ringPower{S}{\pi}{p}$ divides
        $\ringNatural{S}{p}$ in $S$.
    Then $\principalQuotientFrobenius{S}{p}{\pi}\in
        \funs{\carrier[\QuotientRing{S}{\principalIdeal{S}{\pi}}]}{
            \carrier[\QuotientRing{S}{
                \principalIdeal{S}{\ringPower{S}{\pi}{p}}}]}$.
\end{proposition}
\begin{proof}
    $\pi\in\carrier[\PowerBoundedRing{R}]$ by notion.
    Hence $\pi\in\carrier[S]$.
    Follows by \obv{principal_quotient_frobenius_type}.
\end{proof}

\begin{proposition}\label{power_bounded_principal_frobenius_apply}
    Let $R$ be a commutative topological ring.
    Suppose $R$ is a nonarchimedean topological ring.
    Let $S=\PowerBoundedRing{R}$.
    Let $p$ be a prime natural number.
    Suppose $\pi$ is a pseudouniformizer of $R$.
    Suppose $\ringPower{S}{\pi}{p}$ divides
        $\ringNatural{S}{p}$ in $S$.
    Let $x\in\carrier[S]$.
    Let
        $q_a=\groupQuotientProjection{\AdditiveGroup{S}}{
            \principalIdeal{S}{\pi}}$ and
        $q_p=\groupQuotientProjection{\AdditiveGroup{S}}{
            \principalIdeal{S}{\ringPower{S}{\pi}{p}}}$.
    Then $\apply{\principalQuotientFrobenius{S}{p}{\pi}}{q_a(x)}
        =q_p(\ringPower{S}{x}{p})$.
\end{proposition}
\begin{proof}
    $\pi\in\carrier[\PowerBoundedRing{R}]$ by notion.
    Hence $\pi\in\carrier[S]$.
    Follows by \obv{principal_quotient_frobenius_apply}.
\end{proof}

\begin{theorem}\label{power_bounded_principal_frobenius_hom}

    Then $\principalQuotientFrobenius{S}{p}{\pi}$ is a ring homomorphism

\end{theorem}
\begin{proof}
    $\pi\in\carrier[\PowerBoundedRing{R}]$ by notion.

\begin{theorem}\label{power_bounded_principal_frobenius_injective}
    Let $R$ be a commutative topological ring.
    Suppose $R$ is a nonarchimedean topological ring.
    Let $S=\PowerBoundedRing{R}$.
    Let $p$ be a prime natural number.
    Suppose $\pi$ is a pseudouniformizer of $R$.
    Suppose $\ringPower{S}{\pi}{p}$ divides
        $\ringNatural{S}{p}$ in $S$.
    Then $\principalQuotientFrobenius{S}{p}{\pi}$ is injective.
\end{theorem}
\begin{proof}
    Hence $S$ is a commutative ring.
    $\pi\in\carrier[\PowerBoundedRing{R}]$ by notion.
    Hence $\pi\in\carrier[S]$.
    $p$ is a natural number by \obv{prime_natural_guard}.
    $\naturalOne\in p$ by \obv{prime_natural}.
    Hence $p\neq\emptyset$ by \auto.
    $\pi$ is a unit of $\MultiplicativeMonoid{R}$ by
        \obv{pseudouniformizer_is_unit}.
    Show that for every $x\in\carrier[S]$ such that
        $\ringPower{S}{x}{p}
            \in\principalIdeal{S}{\ringPower{S}{\pi}{p}}$
        we have $x\in\principalIdeal{S}{\pi}$.
    \begin{subproof}
        Follows by \auto{power_bounded_principal_power_kernel}.
    \end{subproof}
    Follows by \obv{principal_quotient_frobenius_injective_iff}.
\end{proof}

\begin{theorem}\label{power_bounded_principal_frobenius_isomorphism_iff_surjective}
    Let $R$ be a commutative topological ring.
    Suppose $R$ is a nonarchimedean topological ring.
    Let $S=\PowerBoundedRing{R}$.
    Let $p$ be a prime natural number.
    Suppose $\pi$ is a pseudouniformizer of $R$.
    Suppose $\ringPower{S}{\pi}{p}$ divides
        $\ringNatural{S}{p}$ in $S$.
    Then $\principalQuotientFrobenius{S}{p}{\pi}$ is a ring isomorphism
        from $\QuotientRing{S}{\principalIdeal{S}{\pi}}$ to
        $\QuotientRing{S}{
            \principalIdeal{S}{\ringPower{S}{\pi}{p}}}$ iff
        $\principalQuotientFrobenius{S}{p}{\pi}\in
            \surj{\carrier[\QuotientRing{S}{\principalIdeal{S}{\pi}}]}{
                \carrier[\QuotientRing{S}{
                    \principalIdeal{S}{\ringPower{S}{\pi}{p}}}]}$.
\end{theorem}
\begin{proof}
    Hence $S$ is a commutative ring.
    $\pi\in\carrier[\PowerBoundedRing{R}]$ by notion.
    Hence $\pi\in\carrier[S]$.
    Let $A=\QuotientRing{S}{\principalIdeal{S}{\pi}}$ and
        $B=\QuotientRing{S}{
            \principalIdeal{S}{\ringPower{S}{\pi}{p}}}$.
    Let $F=\principalQuotientFrobenius{S}{p}{\pi}$.
    $F\in\funs{\carrier[A]}{\carrier[B]}$ by
        \auto{power_bounded_principal_frobenius_type}.
    $F$ is injective by
        \obv{power_bounded_principal_frobenius_injective}.
    $F\in\bijections{\carrier[A]}{\carrier[B]}$ iff
        $F$ is injective and $F\in\surj{\carrier[A]}{\carrier[B]}$ by
        \auto{bijections}.
    Hence $F\in\bijections{\carrier[A]}{\carrier[B]}$ iff
        $F\in\surj{\carrier[A]}{\carrier[B]}$ by \auto.
    $F$ is a ring isomorphism from $A$ to $B$ iff
        $F\in\bijections{\carrier[A]}{\carrier[B]}$ by
        \auto{principal_quotient_frobenius_isomorphism_iff_bijective}.
\end{proof}
