\section{Paths in products}

\import{topology/path.tex}
\import{topology/product.tex}

Pairing two paths with the same parameter interval gives a path in the
product. No nonemptiness assumption on either space is needed.

\begin{theorem}\label{product_pairing_of_paths}
    Let $X,Y$ be topological spaces.
    Suppose $p$ is a path from $a$ to $b$ in $X$ and
        $q$ is a path from $c$ to $d$ in $Y$.
    Then $\productPairing{\RealUnitInterval}{p}{q}$ is a path from
        $(a,c)$ to $(b,d)$ in $\BinaryProductTopology{X}{Y}$.
\end{theorem}
\begin{proof}
    $p$ is continuous from $\RealUnitIntervalTopology$ to $X$ and
        $q$ is continuous from $\RealUnitIntervalTopology$ to $Y$ by \obv{path_from_to}.
    $\carrier[\RealUnitIntervalTopology]=\RealUnitInterval$ by
        \obv{real_unit_interval_topology_carrier}.
    Hence $\productPairing{\RealUnitInterval}{p}{q}$ is continuous from
        $\RealUnitIntervalTopology$ to $\BinaryProductTopology{X}{Y}$ by
        \auto{continuous_product_pairing}.
    $p\in\funs{\RealUnitInterval}{\carrier[X]}$ and
        $q\in\funs{\RealUnitInterval}{\carrier[Y]}$ by \auto{continuous_map}.
    $\surrealZero,\surrealOne\in\RealUnitInterval$ by
        \obv{surreal_zero_in_real_unit_interval,surreal_one_in_real_unit_interval}.
    $(a,c),(b,d)\in\carrier[\BinaryProductTopology{X}{Y}]$ by
        \auto{path_from_to,binary_product_topology_carrier,times_tuple_intro}.
    Follows by \auto{path_from_to,product_pairing_apply,binary_product_topology_is_topological_space}.
\end{proof}

\begin{theorem}\label{binary_product_of_path_connected_spaces}
    Let $X,Y$ be path-connected topological spaces.
    Then $\BinaryProductTopology{X}{Y}$ is path-connected.
\end{theorem}
\begin{proof}
    $\BinaryProductTopology{X}{Y}$ is a topological space by
        \obv{binary_product_topology_is_topological_space}.
    Show that for all $u,v\in\carrier[\BinaryProductTopology{X}{Y}]$ there exists $r$
        such that $r$ is a path from $u$ to $v$ in $\BinaryProductTopology{X}{Y}$.
    \begin{subproof}
        Fix $u,v\in\carrier[\BinaryProductTopology{X}{Y}]$.
        $u,v\in\carrier[X]\times\carrier[Y]$ by \obv{binary_product_topology_carrier}.
        Take $a,c$ such that $a\in\carrier[X]$ and $c\in\carrier[Y]$ and $u=(a,c)$ by
            \auto{times}.
        Take $b,d$ such that $b\in\carrier[X]$ and $d\in\carrier[Y]$ and $v=(b,d)$ by
            \auto{times}.
        Take $p$ such that $p$ is a path from $a$ to $b$ in $X$ by \obv{path_connected}.
        Take $q$ such that $q$ is a path from $c$ to $d$ in $Y$ by \obv{path_connected}.
        $\productPairing{\RealUnitInterval}{p}{q}$ is a path from $(a,c)$ to $(b,d)$ in
            $\BinaryProductTopology{X}{Y}$ by \obv{product_pairing_of_paths}.
        Follows by \auto{path_from_to}.
    \end{subproof}
    Follows by \obv{path_connected}.
\end{proof}
