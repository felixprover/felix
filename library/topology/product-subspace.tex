\section{Products of subspaces}

\import{topology/product-basis.tex}
\import{topology/subspace-basis.tex}

Taking a binary product commutes exactly with taking subspaces.  No ambient
subset hypotheses are needed: both sides use the traces of the chosen sets on
the ambient carriers.

\begin{proposition}\label{product_subspace_carriers_agree}
    Let $X,Y$ be topological spaces.
    Then
        $\carrier[\BinaryProductTopology{
            \subspaceTopology{X}{A}}{\subspaceTopology{Y}{B}}]
        =\carrier[\subspaceTopology{
            \BinaryProductTopology{X}{Y}}{A\times B}]$.
\end{proposition}
\begin{proof}
    $\carrier[\subspaceTopology{X}{A}]=A\inter\carrier[X]$ and
        $\carrier[\subspaceTopology{Y}{B}]=B\inter\carrier[Y]$ by
        \obv{subspace_topology_carrier}.
    $\carrier[\BinaryProductTopology{
        \subspaceTopology{X}{A}}{\subspaceTopology{Y}{B}}]
        =\carrier[\subspaceTopology{X}{A}]\times
            \carrier[\subspaceTopology{Y}{B}]$ by
        \obv{binary_product_topology_carrier}.
    Hence $\carrier[\BinaryProductTopology{
        \subspaceTopology{X}{A}}{\subspaceTopology{Y}{B}}]
        =(A\inter\carrier[X])\times(B\inter\carrier[Y])$.
    $(A\inter\carrier[X])\times(B\inter\carrier[Y])
        =(A\times B)\inter(\carrier[X]\times\carrier[Y])$ by
        \obv{inter_times}.
    $\carrier[\BinaryProductTopology{X}{Y}]
        =\carrier[X]\times\carrier[Y]$ by
        \obv{binary_product_topology_carrier}.
    $\carrier[\subspaceTopology{
        \BinaryProductTopology{X}{Y}}{A\times B}]
        =(A\times B)\inter
            \carrier[\BinaryProductTopology{X}{Y}]$ by
        \obv{subspace_topology_carrier}.
\end{proof}

\begin{theorem}\label{product_rectangles_of_subspaces}
    Let $X,Y$ be topological spaces.
    Then
        $\productRectangles{
            \subspaceTopology{X}{A}}{\subspaceTopology{Y}{B}}
        =\subspaceBasis{\BinaryProductTopology{X}{Y}}{A\times B}{
            \productRectangles{X}{Y}}$.
\end{theorem}
\begin{proof}
    Let $X_A=\subspaceTopology{X}{A}$.
    Let $Y_B=\subspaceTopology{Y}{B}$.
    Let $P=\BinaryProductTopology{X}{Y}$.
    Let $S=\subspaceTopology{P}{A\times B}$.
    $X_A$ is a topological space by
        \obv{subspace_topology_is_topological_space}.
    $Y_B$ is a topological space by
        \obv{subspace_topology_is_topological_space}.
    $\carrier[X_A]=A\inter\carrier[X]$ and
        $\carrier[Y_B]=B\inter\carrier[Y]$ by
        \auto{subspace_topology_carrier}.
    $\carrier[S]
        =\carrier[\BinaryProductTopology{X_A}{Y_B}]$ by
        \auto{product_subspace_carriers_agree}.
    Hence $\carrier[S]=\carrier[X_A]\times\carrier[Y_B]$ by
        \auto{binary_product_topology_carrier}.

    Show that every element of
        $\productRectangles{X_A}{Y_B}$ is an element of
        $\subspaceBasis{P}{A\times B}{\productRectangles{X}{Y}}$.
    \begin{subproof}
        Fix $Q\in\productRectangles{X_A}{Y_B}$.
        Take $U\in\opens[X_A]$ such that there exists
            $V\in\opens[Y_B]$ such that $Q=U\times V$ by
            \obv{product_rectangles}.
        Take $V\in\opens[Y_B]$ such that $Q=U\times V$.
        Hence $U\in\opens[\subspaceTopology{X}{A}]$ and
            $V\in\opens[\subspaceTopology{Y}{B}]$.
        Take $O\in\opens[X]$ such that
            $U=O\inter\carrier[\subspaceTopology{X}{A}]$ by
            \obv{subspace_open_iff}.
        Take $W\in\opens[Y]$ such that
            $V=W\inter\carrier[\subspaceTopology{Y}{B}]$ by
            \obv{subspace_open_iff}.
        $O\subseteq\carrier[X]$ and $W\subseteq\carrier[Y]$ by
            \obv{opens_type}.
        Hence $O\times W\subseteq\carrier[X]\times\carrier[Y]$ by
            \auto{times,subseteq}.
        Hence $O\times W\in\pow{\carrier[X]\times\carrier[Y]}$ by
            \obv{powerset_intro}.
        Hence $O\times W\in\productRectangles{X}{Y}$ by
            \auto{product_rectangles}.
        $(O\inter\carrier[X_A])\times
            (W\inter\carrier[Y_B])
            =(O\times W)\inter
                (\carrier[X_A]\times\carrier[Y_B])$ by
            \obv{inter_times}.
        Hence $Q=(O\times W)\inter\carrier[S]$.
        $Q\subseteq\carrier[X_A]\times\carrier[Y_B]$ by
            \auto{product_rectangles,topological_basis,subseteq,
                pow_iff}.
        Hence $Q\subseteq\carrier[S]$.
        $P=\BinaryProductTopology{X}{Y}$ and
            $S=\subspaceTopology{P}{A\times B}$.
        Hence $\carrier[S]=\carrier[\subspaceTopology{
            \BinaryProductTopology{X}{Y}}{A\times B}]$.
        Thus $Q=(O\times W)\inter
            \carrier[\subspaceTopology{
                \BinaryProductTopology{X}{Y}}{A\times B}]$.
        Follows by \auto{subspace_basis_iff}.
    \end{subproof}
    Hence $\productRectangles{X_A}{Y_B}\subseteq
        \subspaceBasis{P}{A\times B}{\productRectangles{X}{Y}}$ by
        \obv{subseteq}.

    Show that every element of
        $\subspaceBasis{P}{A\times B}{\productRectangles{X}{Y}}$
        is an element of $\productRectangles{X_A}{Y_B}$.
    \begin{subproof}
        Fix $Q\in
            \subspaceBasis{P}{A\times B}{\productRectangles{X}{Y}}$.
        $P=\BinaryProductTopology{X}{Y}$ and
            $S=\subspaceTopology{P}{A\times B}$.
        Hence $Q\in\subspaceBasis{
            \BinaryProductTopology{X}{Y}}{A\times B}{
                \productRectangles{X}{Y}}$.
        Take $R\in\productRectangles{X}{Y}$ such that
            $Q=R\inter\carrier[\subspaceTopology{
                \BinaryProductTopology{X}{Y}}{A\times B}]$ by
            \obv{subspace_basis}.
        Hence $Q=R\inter\carrier[S]$.
        Take $O\in\opens[X]$ such that there exists $W\in\opens[Y]$
            such that $R=O\times W$ by \obv{product_rectangles}.
        Take $W\in\opens[Y]$ such that $R=O\times W$.
        Let $U=O\inter\carrier[X_A]$.
        Let $V=W\inter\carrier[Y_B]$.
        $\carrier[X_A]=\carrier[\subspaceTopology{X}{A}]$ and
            $\carrier[Y_B]=\carrier[\subspaceTopology{Y}{B}]$.
        Hence $U=O\inter\carrier[\subspaceTopology{X}{A}]$ and
            $V=W\inter\carrier[\subspaceTopology{Y}{B}]$.
        Hence $U\in\opens[\subspaceTopology{X}{A}]$ and
            $V\in\opens[\subspaceTopology{Y}{B}]$ by
            \auto{subspace_open_iff}.
        Hence $U\in\opens[X_A]$ and $V\in\opens[Y_B]$.
        $(O\inter\carrier[X_A])\times
            (W\inter\carrier[Y_B])
            =(O\times W)\inter
                (\carrier[X_A]\times\carrier[Y_B])$ by
            \obv{inter_times}.
        Hence $U\times V=R\inter\carrier[S]$.
        Thus $Q=U\times V$.
        $U\subseteq\carrier[X_A]$ and $V\subseteq\carrier[Y_B]$ by
            \obv{opens_type}.
        Hence $U\times V\subseteq\carrier[X_A]\times\carrier[Y_B]$ by
            \auto{times,subseteq}.
        Hence $U\times V\in\pow{\carrier[X_A]\times\carrier[Y_B]}$ by
            \obv{powerset_intro}.
        Hence $Q\in\pow{\carrier[X_A]\times\carrier[Y_B]}$.
        Follows by \auto{product_rectangles}.
    \end{subproof}
    Hence $\subspaceBasis{P}{A\times B}{\productRectangles{X}{Y}}
        \subseteq\productRectangles{X_A}{Y_B}$ by \obv{subseteq}.
    Hence $\productRectangles{X_A}{Y_B}
        =\subspaceBasis{P}{A\times B}{\productRectangles{X}{Y}}$ by
        \obv{subseteq_antisymmetric}.
    $X_A=\subspaceTopology{X}{A}$ and
        $Y_B=\subspaceTopology{Y}{B}$ and
        $P=\BinaryProductTopology{X}{Y}$.
\end{proof}

\begin{theorem}\label{product_of_subspaces}
    Let $X,Y$ be topological spaces.
    Then
        $\BinaryProductTopology{
            \subspaceTopology{X}{A}}{\subspaceTopology{Y}{B}}
        =\subspaceTopology{\BinaryProductTopology{X}{Y}}{A\times B}$.
\end{theorem}
\begin{proof}
    Let $L=\BinaryProductTopology{
        \subspaceTopology{X}{A}}{\subspaceTopology{Y}{B}}$.
    Let $P=\BinaryProductTopology{X}{Y}$.
    Let $R=\subspaceTopology{P}{A\times B}$.
    $\subspaceTopology{X}{A}$ is a topological space by
        \obv{subspace_topology_is_topological_space}.
    $\subspaceTopology{Y}{B}$ is a topological space by
        \obv{subspace_topology_is_topological_space}.
    Hence $L$ is an exact topological space by
        \auto{binary_product_topology_is_exact}.
    $\BinaryProductTopology{X}{Y}$ is a topological space by
        \obv{binary_product_topology_is_topological_space}.
    $\subspaceTopology{\BinaryProductTopology{X}{Y}}{A\times B}$ is a
        topological space by
        \obv{subspace_topology_is_topological_space}.
    $\subspaceTopology{\BinaryProductTopology{X}{Y}}{A\times B}$ is an
        exact topological space by \obv{subspace_topology_is_exact}.
    Hence $R$ is an exact topological space by \auto.
    $\carrier[L]=\carrier[R]$ by
        \auto{product_subspace_carriers_agree}.
    $\productRectangles{
        \subspaceTopology{X}{A}}{\subspaceTopology{Y}{B}}$ is an open
        basis of $L$ by \obv{product_rectangles_form_open_basis}.
    $\productRectangles{X}{Y}$ is an open basis of $P$ by
        \obv{product_rectangles_form_open_basis}.
    Hence $\subspaceBasis{P}{A\times B}{\productRectangles{X}{Y}}$ is
        an open basis of $R$ by \obv{open_basis_restricts_to_subspace}.
    $\productRectangles{
        \subspaceTopology{X}{A}}{\subspaceTopology{Y}{B}}
        =\subspaceBasis{P}{A\times B}{\productRectangles{X}{Y}}$ by
        \auto{product_rectangles_of_subspaces}.
    Hence $\productRectangles{
        \subspaceTopology{X}{A}}{\subspaceTopology{Y}{B}}$ is an open
        basis of $R$.
    $\opens[L]=\genOpens{
        \productRectangles{
            \subspaceTopology{X}{A}}{\subspaceTopology{Y}{B}}}{
        \carrier[L]}$ by \obv{open_basis_generates_opens}.
    $\opens[R]=\genOpens{
        \productRectangles{
            \subspaceTopology{X}{A}}{\subspaceTopology{Y}{B}}}{
        \carrier[R]}$ by \obv{open_basis_generates_opens}.
    Hence $\opens[L]=\opens[R]$.
    Hence $L=R$ by structure extensionality.
    $L=\BinaryProductTopology{
        \subspaceTopology{X}{A}}{\subspaceTopology{Y}{B}}$ and
        $P=\BinaryProductTopology{X}{Y}$ and
        $R=\subspaceTopology{P}{A\times B}$.
\end{proof}
