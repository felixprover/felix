\section{Straight paths in the real line}

\import{topology/path.tex}
\import{topology/path-subspace.tex}
\import{topology/real-map-linear.tex}
\import{topology/subspace-continuity.tex}

The affine map $t\mapsto a+(b-a)t$ joins any two real points. Its domain is
the existing closed unit interval, and no ordering of the endpoints is
required.

\begin{theorem}\label{real_affine_path}
    Let $a,b\in\reals$.
    Then $\realMapAffine{\RealUnitInterval}{a}{\surrealAdd{b}{\surrealNeg{a}}}
        {\identity{\RealUnitInterval}}$ is a path from $a$ to $b$ in $\RealTopology$.
\end{theorem}
\begin{proof}
    Let $I=\RealUnitInterval$ and $J=\RealUnitIntervalTopology$.
    Let $d=\surrealAdd{b}{\surrealNeg{a}}$.
    Let $p=\realMapAffine{I}{a}{d}{\identity{I}}$.
    $J$ is a topological space and $\carrier[J]=I$ by
        \obv{real_unit_interval_topology_is_topological_space,real_unit_interval_topology_carrier}.
    $\RealTopology$ is a topological space and $\carrier[\RealTopology]=\reals$ by
        \obv{real_topology_is_topological_space,real_topology_carrier}.
    $\identity{I}$ is continuous from $J$ to $\RealTopology$ by
        \auto{real_unit_interval_topology,subspace_inclusion_continuous}.
    Hence $\identity{I}\in\funs{I}{\reals}$ by \auto{continuous_map}.
    $\surrealNeg{a}\in\reals$ and $d\in\reals$ by
        \auto{surreal_real_negation_closed,surreal_real_addition_closed}.
    $p$ is continuous from $J$ to $\RealTopology$ by \auto{real_map_affine_continuous}.
    $\surrealZero,\surrealOne\in I$ by
        \obv{surreal_zero_in_real_unit_interval,surreal_one_in_real_unit_interval}.
    $\surreal{a}$ and $\surreal{b}$ and $\surreal{d}$ by \obv{surreal_real_is_surreal}.
    $p(\surrealZero)=a$ by
        \auto{real_map_affine_apply,id_apply,surreal_mul_zero_right,surreal_add_zero_right}.
    $\surrealAdd{a}{d}=b$ by
        \auto{surreal_add_associative,surreal_add_commutative,surreal_add_inverse_right,
            surreal_add_zero_left,surreal_negation_is_surreal}.
    Hence $p(\surrealOne)=b$ by \auto{real_map_affine_apply,id_apply,surreal_mul_one_right}.
    Follows by \obv{path_from_to}.
\end{proof}

\begin{theorem}\label{real_topology_is_path_connected}
    $\RealTopology$ is path-connected.
\end{theorem}
\begin{proof}
    $\RealTopology$ is a topological space and $\carrier[\RealTopology]=\reals$ by
        \obv{real_topology_is_topological_space,real_topology_carrier}.
    Show that for all $a,b\in\carrier[\RealTopology]$ there exists $p$ such that
        $p$ is a path from $a$ to $b$ in $\RealTopology$.
    \begin{subproof}
        Fix $a,b\in\carrier[\RealTopology]$.
        $a,b\in\reals$.
        $\realMapAffine{\RealUnitInterval}{a}{\surrealAdd{b}{\surrealNeg{a}}}
            {\identity{\RealUnitInterval}}$ is a path from $a$ to $b$ in $\RealTopology$ by
            \obv{real_affine_path}.
    \end{subproof}
    Follows by \obv{path_connected}.
\end{proof}

\begin{theorem}\label{real_affine_path_in_closed_interval}
    Let $a,b\in\reals$.
    Suppose $\surrealLe{a}{b}$.
    Then $\realMapAffine{\RealUnitInterval}{a}{\surrealAdd{b}{\surrealNeg{a}}}
        {\identity{\RealUnitInterval}}$ is a path from $a$ to $b$ in
        $\subspaceTopology{\RealTopology}{\realClosedInterval{a}{b}}$.
\end{theorem}
\begin{proof}
    Let $I=\RealUnitInterval$ and $J=\RealUnitIntervalTopology$.
    Let $d=\surrealAdd{b}{\surrealNeg{a}}$.
    Let $p=\realMapAffine{I}{a}{d}{\identity{I}}$.
    $p$ is a path from $a$ to $b$ in $\RealTopology$ by \auto{real_affine_path}.
    $J$ is a topological space and $\carrier[J]=I$ by
        \obv{real_unit_interval_topology_is_topological_space,real_unit_interval_topology_carrier}.
    $\RealTopology$ is a topological space and $\carrier[\RealTopology]=\reals$ by
        \obv{real_topology_is_topological_space,real_topology_carrier}.
    $\identity{I}$ is continuous from $J$ to $\RealTopology$ by
        \auto{real_unit_interval_topology,subspace_inclusion_continuous}.
    Hence $\identity{I}\in\funs{I}{\reals}$ by \auto{continuous_map}.
    For every $t\in I$ we have $\surrealLe{\surrealZero}{(\identity{I})(t)}$ and
        $\surrealLe{(\identity{I})(t)}{\surrealOne}$ by \auto{id_apply,real_unit_interval_iff}.
    For every $t\in I$ we have $p(t)\in\realClosedInterval{a}{b}$ by
        \auto{real_affine_of_unit_range_continuous}.
    $p\in\funs{I}{\reals}$ by \auto{path_from_to,continuous_map}.
    Hence $p$ is a function and $\dom{p}=I$ by \obv{funs_is_function,funs}.
    Hence $p$ is a function to $\realClosedInterval{a}{b}$.
    $p\in\funs{I}{\realClosedInterval{a}{b}}$ by \obv{funs_intro}.
    Hence $\ran{p}\subseteq\realClosedInterval{a}{b}$ by \obv{funs_ran}.
    $\realClosedInterval{a}{b}\subseteq\carrier[\RealTopology]$ by
        \obv{real_closed_interval_subset_reals}.
    Follows by \obv{path_codomain_corestriction}.
\end{proof}
