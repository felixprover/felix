\section{The Boolean algebra of clopen sets}

\import{topology/topological-space.tex}
\import{order/boolean-algebra/field-of-sets.tex}

\begin{definition}\label{clopens}
    Let $X$ be a topological space.
    $\clopens{X}=\opens[X]\inter\closeds{X}$.
\end{definition}

\begin{proposition}\label{clopens_iff}
    Let $X$ be a topological space.
    Then $A\in\clopens{X}$ iff $A$ is clopen in $X$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{clopens,is_clopen_in,is_open,is_closed_in,inter}.
\end{proof}

\begin{proposition}\label{clopens_subseteq_powerset}
    Let $X$ be a topological space.
    Then $\clopens{X}\subseteq\pow{\carrier[X]}$.
\end{proposition}
\begin{proof}
    It suffices to show that every element of $\clopens{X}$ is an
        element of $\pow{\carrier[X]}$ by \obv{subseteq}.
    Fix $A\in\clopens{X}$.
    Hence $A$ is open in $X$ by
        \auto{clopens_iff,is_clopen_in}.
    Thus $A\in\opens[X]$ by \obv{is_open}.
    Hence $A\subseteq\carrier[X]$ by \obv{opens_type}.
    Follows by \obv{powerset_intro}.
\end{proof}

\begin{proposition}\label{clopens_emptyset}
    Let $X$ be a topological space.
    Then $\emptyset\in\clopens{X}$.
\end{proposition}
\begin{proof}
    $\emptyset$ is open in $X$ and $\emptyset$ is closed in $X$ by
        \obv{emptyset_open,emptyset_is_closed}.
    Hence $\emptyset$ is clopen in $X$ by \obv{is_clopen_in}.
    Follows by \obv{clopens_iff}.
\end{proof}

\begin{proposition}\label{clopens_carrier}
    Let $X$ be a topological space.
    Then $\carrier[X]\in\clopens{X}$.
\end{proposition}
\begin{proof}
    $\carrier[X]$ is open in $X$ by \obv{is_open,carrier_open}.
    $\carrier[X]$ is closed in $X$ by \obv{carrier_is_closed}.
    Hence $\carrier[X]$ is clopen in $X$ by \obv{is_clopen_in}.
    Follows by \obv{clopens_iff}.
\end{proof}

\begin{proposition}\label{clopens_inter}
    Let $X$ be a topological space.
    Let $A,C\in\clopens{X}$.
    Then $A\inter C\in\clopens{X}$.
\end{proposition}
\begin{proof}
    $A$ is open in $X$ and $A$ is closed in $X$ and
        $C$ is open in $X$ and $C$ is closed in $X$ by
        \auto{clopens_iff,is_clopen_in}.
    Hence $A,C\in\opens[X]$ by \obv{is_open}.
    Thus $A\inter C\in\opens[X]$ by \obv{opens_inter}.
    Hence $A\inter C$ is open in $X$ by \obv{is_open}.
    $A,C\subseteq\carrier[X]$ by
        \obv{closed_in_subseteq_carrier}.
    $A\inter C$ is closed in $X$ by
        \obv{intersection_of_closed_is_closed}.
    Hence $A\inter C$ is clopen in $X$ by \obv{is_clopen_in}.
    Follows by \obv{clopens_iff}.
\end{proof}

\begin{proposition}\label{clopens_union}
    Let $X$ be a topological space.
    Let $A,C\in\clopens{X}$.
    Then $A\union C\in\clopens{X}$.
\end{proposition}
\begin{proof}
    $A$ is open in $X$ and $A$ is closed in $X$ and
        $C$ is open in $X$ and $C$ is closed in $X$ by
        \auto{clopens_iff,is_clopen_in}.
    Hence $A,C\in\opens[X]$ by \obv{is_open}.
    $A\union C$ is open in $X$ by \obv{union_open}.
    $A,C\subseteq\carrier[X]$ by
        \obv{closed_in_subseteq_carrier}.
    $A\union C$ is closed in $X$ by
        \obv{union_of_closed_is_closed}.
    Hence $A\union C$ is clopen in $X$ by \obv{is_clopen_in}.
    Follows by \obv{clopens_iff}.
\end{proof}

\begin{proposition}\label{clopens_complement}
    Let $X$ be a topological space.
    Let $A\in\clopens{X}$.
    Then $\carrier[X]\setminus A\in\clopens{X}$.
\end{proposition}
\begin{proof}
    $A$ is open in $X$ and $A$ is closed in $X$ by
        \auto{clopens_iff,is_clopen_in}.
    $A\subseteq\carrier[X]$ by \obv{closed_in_subseteq_carrier}.
    $\carrier[X]\setminus A$ is open in $X$ by
        \obv{complement_of_closed_is_open}.
    $\carrier[X]\setminus A$ is closed in $X$ by
        \obv{complement_of_open_is_closed}.
    Hence $\carrier[X]\setminus A$ is clopen in $X$ by
        \obv{is_clopen_in}.
    Follows by \obv{clopens_iff}.
\end{proof}

\begin{classification}\label{clopens_are_field_of_sets}
    Let $X$ be a topological space.
    Then $\clopens{X}$ is a field of sets on $\carrier[X]$.
\end{classification}
\begin{proof}
    $\clopens{X}$ is a bounded set-lattice family on
        $\carrier[X]$ by
        \obv{bounded_set_lattice_family,clopens_subseteq_powerset,
            clopens_emptyset,clopens_carrier,clopens_inter,
            clopens_union}.
    For every $A\in\clopens{X}$ we have
        $\carrier[X]\setminus A\in\clopens{X}$ by
        \obv{clopens_complement}.
    Follows by \obv{field_of_sets}.
\end{proof}

\begin{definition}\label{clopen_boolean_algebra}
    Let $X$ be a topological space.
    $\clopenBooleanAlgebra{X}=
        \fieldOfSetsAlgebra{\clopens{X}}{\carrier[X]}$.
\end{definition}

\begin{proposition}\label{clopen_boolean_algebra_carrier}
    Let $X$ be a topological space.
    Then $\carrier[\clopenBooleanAlgebra{X}]=\clopens{X}$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{clopen_boolean_algebra,field_of_sets_algebra_carrier}.
\end{proof}

\begin{proposition}\label{clopen_boolean_algebra_order_iff}
    Let $X$ be a topological space.
    Let $A,C\in\clopens{X}$.
    Then $A\mathrel{\leq[\clopenBooleanAlgebra{X}]}C$ iff
        $A\subseteq C$.
\end{proposition}
\begin{proof}
    $\clopens{X}$ is a field of sets on $\carrier[X]$ by
        \obv{clopens_are_field_of_sets}.
    Follows by
        \auto{clopen_boolean_algebra,
            field_of_sets_algebra_order_iff}.
\end{proof}

\begin{proposition}\label{clopen_boolean_algebra_meet_apply}
    Let $X$ be a topological space.
    Let $A,C\in\clopens{X}$.
    Then $\meet[\clopenBooleanAlgebra{X}](A,C)=A\inter C$.
\end{proposition}
\begin{proof}

\begin{proposition}\label{clopen_boolean_algebra_join_apply}

\end{theorem}
\begin{proof}

\begin{proof}
