\section{Path connectedness implies connectedness}

\import{topology/path.tex}
\import{topology/real-connectedness.tex}

A separation of a path-connected space would pull back along a joining
path to a separation of the unit interval.

\begin{theorem}\label{real_unit_interval_is_connected}
    $\RealUnitIntervalTopology$ is connected.
\end{theorem}
\begin{proof}
    $\RealTopology$ is a topological space by \obv{real_topology_is_topological_space}.
    $\RealField$ is a linearly ordered set by \obv{surreal_real_field_is_linearly_ordered_set}.
    $\surrealZero,\surrealOne\in\carrier[\RealField]$ by
        \auto{surreal_zero_is_real,surreal_one_is_real,surreal_real_field_carrier}.
    $\orderClosedInterval{\RealField}{\surrealZero}{\surrealOne}$ is connected in $\RealTopology$ by
        \obv{real_closed_interval_is_connected}.
    $\RealUnitInterval=\orderClosedInterval{\RealField}{\surrealZero}{\surrealOne}$ by
        \auto{real_unit_interval_eq_closed_interval,real_order_closed_interval_eq,
            surreal_zero_is_real,surreal_one_is_real}.
    Follows by \auto{connected_in,real_unit_interval_topology}.
\end{proof}

\begin{theorem}\label{path_connected_implies_connected}
    Let $X$ be a path-connected topological space.
    Then $X$ is connected.
\end{theorem}
\begin{proof}
    Suppose not.
    $X$ is disconnected by \auto{connected}.
    Take $U,V\in\opens[X]$ such that $\carrier[X]$ is partitioned by $U$ and $V$ by
        \obv{disconnected}.
    $U$ and $V$ are inhabited by \obv{nonempty_inhabited}.
    Take $a$ such that $a\in U$.
    Take $b$ such that $b\in V$.
    $a,b\in\carrier[X]$ by \auto{opens_type,subseteq}.
    Take $p$ such that $p$ is a path from $a$ to $b$ in $X$ by \obv{path_connected}.
    Let $J=\RealUnitIntervalTopology$.
    Let $P=\preimg{p}{U}$ and $Q=\preimg{p}{V}$.
    $p$ is continuous from $J$ to $X$ by \auto{path_from_to}.
    $P,Q\in\opens[J]$ by \auto{continuous_map}.
    $p\in\funs{\carrier[J]}{\carrier[X]}$ by \obv{continuous_map}.
    Hence $p$ is a function and $\dom{p}=\carrier[J]$ by \obv{funs_is_function,funs}.
    $\carrier[J]=\RealUnitInterval$ by \obv{real_unit_interval_topology_carrier}.
    $\surrealZero,\surrealOne\in\RealUnitInterval$ by
        \obv{surreal_zero_in_real_unit_interval,surreal_one_in_real_unit_interval}.
    $\surrealZero\in P$ and $\surrealOne\in Q$ by
        \auto{path_from_to,preimg_iff,function_apply_elim}.
    Hence $P\neq\emptyset$ and $Q\neq\emptyset$ by \auto.
    $P$ is disjoint from $Q$ by \auto{disjoint,preimg_iff,rightunique}.
    $U\union V=\carrier[X]$.
    $P\union Q=\preimg{p}{U\union V}$ by \auto{preimg_union}.
    $\preimg{p}{\carrier[X]}=\carrier[J]$ by \obv{preimg_codom_of_function}.
    Hence $P\union Q=\carrier[J]$.
    $\carrier[J]$ is partitioned by $P$ and $Q$ by \auto.
    $J$ is a topological space by \obv{real_unit_interval_topology_is_topological_space}.
    Hence $J$ is disconnected by \auto{disconnected}.
    $J$ is connected by \obv{real_unit_interval_is_connected}.
    Contradiction by \obv{connected}.
\end{proof}
