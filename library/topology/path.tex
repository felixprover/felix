\section{Paths and path connectedness}

\import{topology/real-line.tex}
\import{topology/continuity.tex}

A path is a continuous graph on the closed real unit interval, with specified
endpoints. Path connectedness does not require a nonempty carrier; in
particular the empty space is path-connected.

\begin{definition}\label{path_from_to}
    Let $X$ be a topological space.
    $p$ is a path from $a$ to $b$ in $X$ iff
        $a,b\in\carrier[X]$ and
        $p$ is continuous from $\RealUnitIntervalTopology$ to $X$ and
        $p(\surrealZero)=a$ and $p(\surrealOne)=b$.
\end{definition}

\begin{definition}\label{path_connected}
    Let $X$ be a topological space.
    $X$ is path-connected iff for all $a,b\in\carrier[X]$ there exists $p$
        such that $p$ is a path from $a$ to $b$ in $X$.
\end{definition}

\begin{theorem}\label{constant_path}
    Let $X$ be a topological space and $a\in\carrier[X]$.
    Then $\RealUnitInterval\times\{a\}$ is a path from $a$ to $a$ in $X$.
\end{theorem}
\begin{proof}
    $\RealUnitIntervalTopology$ is a topological space by
        \obv{real_unit_interval_topology_is_topological_space}.
    $\carrier[\RealUnitIntervalTopology]=\RealUnitInterval$ by
        \obv{real_unit_interval_topology_carrier}.
    $\RealUnitInterval\times\{a\}$ is continuous from
        $\RealUnitIntervalTopology$ to $X$ by \auto{constant_graph_continuous}.
    $\surrealZero,\surrealOne\in\RealUnitInterval$ by
        \obv{surreal_zero_in_real_unit_interval,surreal_one_in_real_unit_interval}.
    Follows by \auto{path_from_to,constant_graph_application}.
\end{proof}

\begin{theorem}\label{continuous_image_of_path}
    Let $X,Y$ be topological spaces.
    Suppose $p$ is a path from $a$ to $b$ in $X$ and
        $f$ is continuous from $X$ to $Y$.
    Then $f\circ p$ is a path from $f(a)$ to $f(b)$ in $Y$.
\end{theorem}
\begin{proof}
    $p$ is continuous from $\RealUnitIntervalTopology$ to $X$ by \obv{path_from_to}.
    Hence $f\circ p$ is continuous from $\RealUnitIntervalTopology$ to $Y$ by
        \obv{continuous_composition}.
    $p\in\funs{\RealUnitInterval}{\carrier[X]}$ and
        $f\in\funs{\carrier[X]}{\carrier[Y]}$ by
        \auto{continuous_map,real_unit_interval_topology_carrier}.
    $a,b\in\carrier[X]$ by \obv{path_from_to}.
    Hence $f(a),f(b)\in\carrier[Y]$ by \obv{funs_type_apply}.
    $\surrealZero,\surrealOne\in\RealUnitInterval$ by
        \obv{surreal_zero_in_real_unit_interval,surreal_one_in_real_unit_interval}.
    $f$ is composable with $p$ by \auto{funs,funs_ran}.
    Follows by \auto{path_from_to,circ_apply,funs_is_function,funs}.
\end{proof}

\begin{theorem}\label{empty_space_is_path_connected}
    Let $X$ be a topological space.
    Suppose $\carrier[X]=\emptyset$.
    Then $X$ is path-connected.
\end{theorem}
\begin{proof}
    Follows by \auto{path_connected}.
\end{proof}

\begin{theorem}\label{continuous_surjective_image_is_path_connected}
    Let $X,Y$ be topological spaces.
    Suppose $X$ is path-connected and $f$ is continuous from $X$ to $Y$.
    Suppose $f\in\surj{\carrier[X]}{\carrier[Y]}$.
    Then $Y$ is path-connected.
\end{theorem}
\begin{proof}
    Show that for all $a,b\in\carrier[Y]$ there exists $q$ such that
        $q$ is a path from $a$ to $b$ in $Y$.
    \begin{subproof}
        Fix $a,b\in\carrier[Y]$.
        Take $x\in\carrier[X]$ such that $f(x)=a$ by \obv{surj}.
        Take $y\in\carrier[X]$ such that $f(y)=b$ by \obv{surj}.
        Take $p$ such that $p$ is a path from $x$ to $y$ in $X$ by \obv{path_connected}.
        $f\circ p$ is a path from $f(x)$ to $f(y)$ in $Y$ by \obv{continuous_image_of_path}.
        Follows by \auto{path_from_to}.
    \end{subproof}
    Follows by \obv{path_connected}.
\end{proof}

\begin{theorem}\label{singleton_space_is_path_connected}
    Let $X$ be a topological space.
    Suppose $\carrier[X]=\{a\}$.
    Then $X$ is path-connected.
\end{theorem}
\begin{proof}
    $a\in\carrier[X]$ by \auto{singleton_iff}.
    $\RealUnitInterval\times\{a\}$ is a path from $a$ to $a$ in $X$ by
        \obv{constant_path}.
    Follows by \auto{path_connected,singleton_iff}.
\end{proof}

\begin{theorem}\label{homeomorphism_preserves_path_connectedness}
    Let $X,Y$ be topological spaces.
    Suppose $X$ is path-connected and $f$ is a homeomorphism from $X$ to $Y$.
    Then $Y$ is path-connected.
\end{theorem}
\begin{proof}
    $f$ is continuous from $X$ to $Y$ by \obv{homeomorphism}.
    $f\in\surj{\carrier[X]}{\carrier[Y]}$ by \auto{homeomorphism,bijections}.
    Follows by \obv{continuous_surjective_image_is_path_connected}.
\end{proof}
