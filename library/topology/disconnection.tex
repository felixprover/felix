\import{set.tex}
\import{set/bipartition.tex}
\import{topology/topological-space.tex}

\section{Disconnections}

\begin{definition}\label{disconnections}
    $\disconnections{X} = \{ p\in\bipartitions{\carrier[X]} \mid \text{$\fst{p},\snd{p}\in\opens[X]$} \}$.
\end{definition}

\begin{definition}\label{is_a_disconnection}
    Let $X$ be a topological space.
    $D$ is a disconnection of $X$ iff $D\in\disconnections{X}$.
\end{definition}

\begin{definition}\label{disconnected}
    $X$ is disconnected iff $X$ is a topological space and
        there exist $U,V\in\opens[X]$ such that
        $\carrier[X]$ is partitioned by $U$ and $V$.
\end{definition}

\begin{classification}\label{disconnected_is_topological}
    Suppose $X$ is disconnected.
    Then $X$ is a topological space.
\end{classification}
\begin{proof}
    Follows by \obv{disconnected}.
\end{proof}

\begin{abbreviation}\label{disconnected_space}
    $X$ is a disconnected space iff $X$ is disconnected.
\end{abbreviation}

\begin{classification}\label{disconnected_space_is_topological}
    Let $X$ be a disconnected space.
    Then $X$ is a topological space.
\end{classification}
\begin{proof}
    Follows by \obv{disconnected}.
\end{proof}

\begin{proposition}\label{disconnection_from_disconnected}
    Suppose $X$ is a disconnected space.
    Then there exists a disconnection of $X$.
\end{proposition}
\begin{proof}
    $X$ is a topological space by notion.
    Take $U,V$ such that $U,V\in\opens[X]$ and
        $\carrier[X]$ is partitioned by $U$ and $V$
        by \obv{disconnected}.
    $(U,V)$ is a bipartition of $\carrier[X]$ by \obv{bipartition_intro}.
    $\fst{(U,V)}=U$ by \obv{fst_eq}.
    $\snd{(U,V)}=V$ by \obv{snd_eq}.
    $(U,V)\in\disconnections{X}$ by \obv{disconnections}.
    Follows by \auto{is_a_disconnection}.
\end{proof}

\begin{proposition}\label{disconnected_from_disconnection}
    Let $X$ be a topological space.
    Let $D$ be a disconnection of $X$.
    Then $X$ is a disconnected space.
\end{proposition}
\begin{proof}
    $D\in\disconnections{X}$ by \obv{is_a_disconnection}.
    $D\in\bipartitions{\carrier[X]}$ by \obv{disconnections}.
    $\fst{D},\snd{D}\in\opens[X]$ by \obv{disconnections}.
    $\carrier[X]$ is partitioned by $\fst{D}$ and $\snd{D}$
        by \obv{bipartitions_of_a_set}.
    $X$ is disconnected by \auto{disconnected}.
\end{proof}

\begin{definition}\label{connected}
    $X$ is connected iff $X$ is a topological space and
        $X$ is not disconnected.
\end{definition}

\begin{classification}\label{connected_is_topological}
    Suppose $X$ is connected.
    Then $X$ is a topological space.
\end{classification}
\begin{proof}
    Follows by \obv{connected}.
\end{proof}

\begin{abbreviation}\label{connected_space}
    $X$ is a connected space iff $X$ is connected.
\end{abbreviation}

\begin{classification}\label{connected_space_is_topological}
    Let $X$ be a connected space.
    Then $X$ is a topological space.
\end{classification}
\begin{proof}
    Follows by \obv{connected}.
\end{proof}

\begin{theorem}\label{subsingleton_topological_space_is_connected}
    Let $X$ be a topological space.
    Suppose $\carrier[X]$ is a subsingleton.
    Then $X$ is connected.
\end{theorem}
\begin{proof}
    Suppose not.
    Hence $X$ is disconnected by \auto{connected}.
    Take $U,V\in\opens[X]$ such that $\carrier[X]$ is partitioned by $U$ and $V$ by
        \obv{disconnected}.
    $U$ and $V$ are inhabited by \obv{nonempty_inhabited}.
    Take $u$ such that $u\in U$.
    Take $v$ such that $v\in V$.
    $U\subseteq\carrier[X]$ and $V\subseteq\carrier[X]$ by \obv{opens_type}.
    Hence $u,v\in\carrier[X]$ by \obv{subseteq}.
    Hence $u=v$.
    Contradiction by \auto{disjoint}.
\end{proof}
