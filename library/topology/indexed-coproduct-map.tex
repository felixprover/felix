\section{Maps from indexed coproducts}

\import{topology/indexed-coproduct.tex}

The set-theoretic indexed-sum mediator is continuous exactly when each of
its component maps is continuous.  Thus the coproduct topology introduces no
second map construction.

\begin{theorem}\label{continuous_indexed_coproduct_mediator}
    Suppose $X$ is a function on $I$.
    Suppose for every $i\in I$ we have $X(i)$ is a topological space.
    Let $Y$ be a topological space.
    Suppose $f$ is a function on $I$.
    Suppose for every $i\in I$ we have
        $f(i)$ is continuous from $X(i)$ to $Y$.
    Then $\indexedSumMediator{I}{
        \indexedTopologicalCarrierFamily{I}{X}}{\carrier[Y]}{f}$
        is continuous from $\IndexedCoproductTopology{I}{X}$ to $Y$.
\end{theorem}
\begin{proof}
    $\indexedTopologicalCarrierFamily{I}{X}$ is a function on $I$ by
        \obv{indexed_topological_carrier_family_type}.
    Show that for every $i\in I$ we have
        $f(i)\in\funs{
            (\indexedTopologicalCarrierFamily{I}{X})(i)}{\carrier[Y]}$.
    \begin{subproof}
        Fix $i\in I$.
        $f(i)$ is continuous from $X(i)$ to $Y$.
        $f(i)\in\funs{\carrier[X(i)]}{\carrier[Y]}$ by
            \obv{continuous_map}.
        $(\indexedTopologicalCarrierFamily{I}{X})(i)
            =\carrier[X(i)]$ by
            \obv{indexed_topological_carrier_family_apply}.
    \end{subproof}
    Hence $\indexedSumMediator{I}{
        \indexedTopologicalCarrierFamily{I}{X}}{\carrier[Y]}{f}
        \in\funs{
            \indexedSum{I}{
                \indexedTopologicalCarrierFamily{I}{X}}}{\carrier[Y]}$ by
        \obv{indexed_sum_mediator_type}.
    For every $i\in I$ we have
        $\indexedSumMediator{I}{
            \indexedTopologicalCarrierFamily{I}{X}}{\carrier[Y]}{f}
        \circ\indexedSumInjection{I}{
            \indexedTopologicalCarrierFamily{I}{X}}{i}=f(i)$ by
        \obv{indexed_sum_mediator_injection}.
    Hence for every $i\in I$ we have
        $\indexedSumMediator{I}{
            \indexedTopologicalCarrierFamily{I}{X}}{\carrier[Y]}{f}
        \circ\indexedSumInjection{I}{
            \indexedTopologicalCarrierFamily{I}{X}}{i}$
        is continuous from $X(i)$ to $Y$ by \auto.
    Follows by \obv{indexed_coproduct_topology_universal_property}.
\end{proof}

\begin{theorem}\label{indexed_coproduct_mediator_continuous_iff}
    Suppose $X$ is a function on $I$.
    Suppose for every $i\in I$ we have $X(i)$ is a topological space.
    Let $Y$ be a topological space.
    Suppose $f$ is a function on $I$.
    Suppose for every $i\in I$ we have
        $f(i)\in\funs{\carrier[X(i)]}{\carrier[Y]}$.
    Then $\indexedSumMediator{I}{
        \indexedTopologicalCarrierFamily{I}{X}}{\carrier[Y]}{f}$
        is continuous from $\IndexedCoproductTopology{I}{X}$ to $Y$ iff
        for every $i\in I$ we have
            $f(i)$ is continuous from $X(i)$ to $Y$.
\end{theorem}
\begin{proof}
    $\indexedTopologicalCarrierFamily{I}{X}$ is a function on $I$ by
        \obv{indexed_topological_carrier_family_type}.
    Show that for every $i\in I$ we have
        $f(i)\in\funs{
            (\indexedTopologicalCarrierFamily{I}{X})(i)}{\carrier[Y]}$.
    \begin{subproof}
        Fix $i\in I$.
        $f(i)\in\funs{\carrier[X(i)]}{\carrier[Y]}$.
        $(\indexedTopologicalCarrierFamily{I}{X})(i)
            =\carrier[X(i)]$ by
            \obv{indexed_topological_carrier_family_apply}.
    \end{subproof}
    Hence $\indexedSumMediator{I}{
        \indexedTopologicalCarrierFamily{I}{X}}{\carrier[Y]}{f}
        \in\funs{
            \indexedSum{I}{
                \indexedTopologicalCarrierFamily{I}{X}}}{\carrier[Y]}$ by
        \obv{indexed_sum_mediator_type}.
    For every $i\in I$ we have
        $\indexedSumMediator{I}{
            \indexedTopologicalCarrierFamily{I}{X}}{\carrier[Y]}{f}
        \circ\indexedSumInjection{I}{
            \indexedTopologicalCarrierFamily{I}{X}}{i}=f(i)$ by
        \obv{indexed_sum_mediator_injection}.
    $\indexedSumMediator{I}{
        \indexedTopologicalCarrierFamily{I}{X}}{\carrier[Y]}{f}$
        is continuous from $\IndexedCoproductTopology{I}{X}$ to $Y$ iff
        for every $i\in I$ we have
            $\indexedSumMediator{I}{
                \indexedTopologicalCarrierFamily{I}{X}}{\carrier[Y]}{f}
            \circ\indexedSumInjection{I}{
                \indexedTopologicalCarrierFamily{I}{X}}{i}$
            is continuous from $X(i)$ to $Y$ by
        \obv{indexed_coproduct_topology_universal_property}.
    Follows by \auto.
\end{proof}
