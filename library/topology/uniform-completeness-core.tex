\section{Filter and sequential completeness in uniform spaces}

Sequence-tail filters connect Cauchy filters with Cauchy sequences without
any countability assumption on the uniformity.

\begin{theorem}\label{uniform_cauchy_sequence_tail_filter}
    Let $X$ be a uniform space.
    Suppose $\uniformCauchySequence{X}{s}$.
    Then $\uniformCauchyFilter{X}
        {\sequenceTailFilter{\carrier[X]}{s}}$.
\end{theorem}
\begin{proof}
    Follows by \obv{uniform_cauchy_sequence}.
\end{proof}

\begin{theorem}\label{tail_filter_convergence_implies_sequence_convergence}
    Let $X$ be a uniform space and
        $s\in\funs{\naturals}{\carrier[X]}$.
    Suppose
        $\uniformFilterConverges{X}
            {\sequenceTailFilter{\carrier[X]}{s}}{x}$.
    Then $\uniformSequenceConverges{X}{s}{x}$.
\end{theorem}
\begin{proof}
    $\uniformTopology{X}$ is a topological space
        by \obv{uniform_topology_is_topological_space}.
    $\carrier[\uniformTopology{X}]=\carrier[X]$
        by \obv{uniform_topology_carrier}.
    Hence $s\in\funs{\naturals}{\carrier[\uniformTopology{X}]}$.
    $\sequenceTailFilter{\carrier[X]}{s}=
        \sequenceTailFilter{\carrier[\uniformTopology{X}]}{s}$.
    $\filterConverges{\sequenceTailFilter{\carrier[X]}{s}}{x}
        {\uniformTopology{X}}$ by \obv{uniform_filter_convergence}.
    Hence $\filterConverges{
            \sequenceTailFilter{\carrier[\uniformTopology{X}]}{s}}{x}
        {\uniformTopology{X}}$.
    Hence $\topologicalSequenceConverges{\uniformTopology{X}}{s}{x}$ by
        \obv{topological_sequence_convergence}.
    Follows by \obv{uniform_sequence_convergence}.
\end{proof}

\begin{theorem}\label{complete_uniform_implies_sequentially_complete}
    Suppose $X$ is complete uniform.
    Then $X$ is sequentially complete uniform.
\end{theorem}
\begin{proof}
    Show that for every $s$ such that $\uniformCauchySequence{X}{s}$
        there exists $x\in\carrier[X]$ such that
        $\uniformSequenceConverges{X}{s}{x}$.
    \begin{subproof}
        Fix $s$.
        Assume $\uniformCauchySequence{X}{s}$.
        $s\in\funs{\naturals}{\carrier[X]}$
            by \obv{uniform_cauchy_sequence_sequence}.
        $\uniformCauchyFilter{X}
            {\sequenceTailFilter{\carrier[X]}{s}}$
            by \obv{uniform_cauchy_sequence_tail_filter}.
        Take $x\in\carrier[X]$ such that
            $\uniformFilterConverges{X}
                {\sequenceTailFilter{\carrier[X]}{s}}{x}$
            by \obv{complete_uniform_space}.
        $\uniformSequenceConverges{X}{s}{x}$
            by \obv{tail_filter_convergence_implies_sequence_convergence}.
    \end{subproof}
    Follows by \obv{sequentially_complete_uniform_space}.
\end{proof}
