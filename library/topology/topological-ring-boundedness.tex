\section{Bounded subsets of topological rings}

We first keep track of the side on which a small neighbourhood acts.  This
one-sided notion is useful without a commutativity hypothesis; the customary
unqualified notion will be introduced for commutative topological rings.

\begin{definition}\label{left_bounded_in}
    $B$ is left bounded in $R$ iff
        $R$ is a topological ring and $B\subseteq\carrier[R]$ and
        for every $U\in\neighbourhoods{\ringZero[R]}{R}$ there exists
            $V\in\neighbourhoods{\ringZero[R]}{R}$ such that
            $\pointwiseProduct{\MultiplicativeMonoid{R}}{V}{B}\subseteq U$.
\end{definition}

\begin{proposition}\label{left_bounded_in_ambient}
    Suppose $B$ is left bounded in $R$.
    Then $R$ is a topological ring.
\end{proposition}
\begin{proof}
    Follows by \obv{left_bounded_in}.
\end{proof}

\begin{proposition}\label{left_bounded_in_carrier}
    Suppose $B$ is left bounded in $R$.
    Then $B\subseteq\carrier[R]$.
\end{proposition}
\begin{proof}
    Follows by \obv{left_bounded_in}.
\end{proof}

\begin{proposition}\label{left_bounded_in_neighbourhood}
    Suppose $B$ is left bounded in $R$.
    Let $U\in\neighbourhoods{\ringZero[R]}{R}$.
    Then there exists $V\in\neighbourhoods{\ringZero[R]}{R}$ such that
        $\pointwiseProduct{\MultiplicativeMonoid{R}}{V}{B}\subseteq U$.
\end{proposition}
\begin{proof}
    Follows by \obv{left_bounded_in}.
\end{proof}

\begin{lemma}\label{topological_ring_zero_neighbourhood_product}
    Let $R$ be a topological ring.
    Let $U\in\neighbourhoods{\ringZero[R]}{R}$.
    Then there exist $V,W\in\neighbourhoods{\ringZero[R]}{R}$ such that
        $\pointwiseProduct{\MultiplicativeMonoid{R}}{V}{W}\subseteq U$.
\end{lemma}
\begin{proof}
    Let $M=\MultiplicativeMonoid{R}$.
    $R$ is a ring by \obv{topological_ring_is_ring}.
    $\carrier[M]=\carrier[R]$ and $\mul[M]=\ringMul[R]$ by
        \auto{multiplicative_monoid_fields}.
    Take $O\in\opens[R]$ such that $\ringZero[R]\in O\subseteq U$ by
        \obv{neighbourhoods}.
    $\ringZero[R]\in\carrier[R]$ by \obv{ring_zero_type}.
    $\ringMul[R](\ringZero[R],\ringZero[R])=\ringZero[R]$ by
        \obv{ring_mul_zero_left}.
    Hence $\ringMul[R](\ringZero[R],\ringZero[R])\in O$.
    Take $V,W\in\opens[R]$ such that
        $\ringZero[R]\in V$ and $\ringZero[R]\in W$ and
        for all $v,w$ such that $v\in V$ and $w\in W$ we have
            $\ringMul[R](v,w)\in O$ by
        \obv{topological_ring_multiplication}.
    Hence $V,W\in\neighbourhoods{\ringZero[R]}{R}$ by
        \obv{open_set_containing_point_is_neighbourhood}.
    Show that every element of $\pointwiseProduct{M}{V}{W}$ is an element
        of $U$.
    \begin{subproof}
        Fix $z\in\pointwiseProduct{M}{V}{W}$.
        Take $v,w$ such that $v\in V$ and $w\in W$ and
            $z=\mul[M](v,w)$ by \obv{pointwise_product_iff}.
        Hence $z=\ringMul[R](v,w)$.
        $\ringMul[R](v,w)\in O$.
        Hence $\ringMul[R](v,w)\in U$ by \obv{subseteq}.
    \end{subproof}
    Hence $\pointwiseProduct{M}{V}{W}\subseteq U$ by \obv{subseteq}.
    Follows by \auto.
\end{proof}

\subsection{Fixed multiplication near zero}

\begin{lemma}\label{topological_ring_right_multiplication_zero_neighbourhood}
    Let $R$ be a topological ring and $a\in\carrier[R]$.
    Let $U\in\neighbourhoods{\ringZero[R]}{R}$.
    Then there exists $V\in\neighbourhoods{\ringZero[R]}{R}$ such that
        $\pointwiseProduct{\MultiplicativeMonoid{R}}{V}{\{a\}}\subseteq U$.
\end{lemma}
\begin{proof}
    Let $M=\MultiplicativeMonoid{R}$.
    $R$ is a ring by \obv{topological_ring_is_ring}.
    $\carrier[M]=\carrier[R]$ and $\mul[M]=\ringMul[R]$ by
        \auto{multiplicative_monoid_fields}.
    Take $O\in\opens[R]$ such that
        $\ringZero[R]\in O\subseteq U$ by \obv{neighbourhoods}.
    $\ringMul[R](\ringZero[R],a)=\ringZero[R]$ by
        \obv{ring_mul_zero_left}.
    Hence $\ringMul[R](\ringZero[R],a)\in O$.
    Take $V,W\in\opens[R]$ such that
        $\ringZero[R]\in V$ and $a\in W$ and
        for all $v,w$ such that $v\in V$ and $w\in W$ we have
            $\ringMul[R](v,w)\in O$ by
        \obv{topological_ring_multiplication}.
    Hence $V\in\neighbourhoods{\ringZero[R]}{R}$ by
        \obv{open_set_containing_point_is_neighbourhood}.
    Show that every element of $\pointwiseProduct{M}{V}{\{a\}}$ is an
        element of $U$.
    \begin{subproof}
        Fix $z\in\pointwiseProduct{M}{V}{\{a\}}$.
        Take $v,b$ such that $v\in V$ and $b\in\{a\}$ and
            $z=\mul[M](v,b)$ by \obv{pointwise_product_iff}.
        Hence $b=a$ by \obv{singleton_iff}.
        Hence $z=\ringMul[R](v,a)$.
        $\ringMul[R](v,a)\in O$.
        Hence $\ringMul[R](v,a)\in U$ by \obv{subseteq}.
    \end{subproof}
    Hence $\pointwiseProduct{M}{V}{\{a\}}\subseteq U$ by \obv{subseteq}.
\end{proof}

\subsection{Elementary closure properties}

\begin{proposition}\label{left_bounded_singleton}
    Let $R$ be a topological ring and $a\in\carrier[R]$.
    Then $\{a\}$ is left bounded in $R$.
\end{proposition}
\begin{proof}
    $\{a\}\subseteq\carrier[R]$ by \auto{subseteq,singleton_iff}.
    For every $U\in\neighbourhoods{\ringZero[R]}{R}$ there exists
        $V\in\neighbourhoods{\ringZero[R]}{R}$ such that
        $\pointwiseProduct{\MultiplicativeMonoid{R}}{V}{\{a\}}\subseteq U$
        by \obv{topological_ring_right_multiplication_zero_neighbourhood}.
    Follows by \obv{left_bounded_in}.
\end{proof}

\begin{proposition}\label{left_bounded_downward}
    Suppose $A$ is left bounded in $R$ and $B\subseteq A$.
    Then $B$ is left bounded in $R$.
\end{proposition}
\begin{proof}
    $A\subseteq\carrier[R]$ by \obv{left_bounded_in_carrier}.
    Hence $B\subseteq\carrier[R]$ by \obv{subseteq_transitive}.
    Show that for every $U\in\neighbourhoods{\ringZero[R]}{R}$ there exists
        $V\in\neighbourhoods{\ringZero[R]}{R}$ such that
        $\pointwiseProduct{\MultiplicativeMonoid{R}}{V}{B}\subseteq U$.
    \begin{subproof}
        Fix $U\in\neighbourhoods{\ringZero[R]}{R}$.
        Take $V\in\neighbourhoods{\ringZero[R]}{R}$ such that
            $\pointwiseProduct{\MultiplicativeMonoid{R}}{V}{A}\subseteq U$
            by \obv{left_bounded_in_neighbourhood}.
        $\pointwiseProduct{\MultiplicativeMonoid{R}}{V}{B}\subseteq
            \pointwiseProduct{\MultiplicativeMonoid{R}}{V}{A}$ by
            \obv{pointwise_product_monotone,subseteq_refl}.
        Hence $\pointwiseProduct{\MultiplicativeMonoid{R}}{V}{B}\subseteq U$
            by \obv{subseteq_transitive}.
    \end{subproof}
    Follows by \obv{left_bounded_in}.
\end{proof}

\begin{proposition}\label{left_bounded_emptyset}
    Let $R$ be a topological ring.
    Then $\emptyset$ is left bounded in $R$.
\end{proposition}
\begin{proof}
    $\ringZero[R]\in\carrier[R]$ by \obv{ring_zero_type}.
    $\{\ringZero[R]\}$ is left bounded in $R$ by
        \obv{left_bounded_singleton}.
    $\emptyset\subseteq\{\ringZero[R]\}$ by \obv{emptyset_subseteq}.
    Follows by \obv{left_bounded_downward}.
\end{proof}

\begin{proposition}\label{left_bounded_binary_union}
    Suppose $A$ and $B$ are left bounded in $R$.
    Then $A\union B$ is left bounded in $R$.
\end{proposition}
\begin{proof}
    $R$ is a topological ring by \obv{left_bounded_in_ambient}.
    $A,B\subseteq\carrier[R]$ by \obv{left_bounded_in_carrier}.
    Hence $A\union B\subseteq\carrier[R]$ by
        \auto{subseteq,union_iff}.
    Show that for every $U\in\neighbourhoods{\ringZero[R]}{R}$ there exists
        $T\in\neighbourhoods{\ringZero[R]}{R}$ such that
        $\pointwiseProduct{\MultiplicativeMonoid{R}}{T}{A\union B}
            \subseteq U$.
    \begin{subproof}
        Fix $U\in\neighbourhoods{\ringZero[R]}{R}$.
        Take $V\in\neighbourhoods{\ringZero[R]}{R}$ such that
            $\pointwiseProduct{\MultiplicativeMonoid{R}}{V}{A}\subseteq U$ by
            \obv{left_bounded_in_neighbourhood}.
        Take $W\in\neighbourhoods{\ringZero[R]}{R}$ such that
            $\pointwiseProduct{\MultiplicativeMonoid{R}}{W}{B}\subseteq U$ by
            \obv{left_bounded_in_neighbourhood}.
        Let $T=V\inter W$.
        $V,W\subseteq\carrier[R]$ by
            \auto{topological_ring_zero_neighbourhoods_is_filter,filter}.
        Hence $T\in\neighbourhoods{\ringZero[R]}{R}$ by
            \auto{topological_ring_zero_neighbourhoods_is_filter,
                filter_inter_in_iff}.
        Show that every element of
            $\pointwiseProduct{\MultiplicativeMonoid{R}}{T}{A\union B}$ is
            an element of $U$.
        \begin{subproof}
            Fix $z\in
                \pointwiseProduct{\MultiplicativeMonoid{R}}{T}{A\union B}$.
            Take $t,x$ such that $t\in T$ and $x\in A\union B$ and
                $z=\mul[\MultiplicativeMonoid{R}](t,x)$ by
                \obv{pointwise_product_iff}.
            $t\in V$ and $t\in W$ by \obv{inter}.
            If $x\in A$, then $z\in U$ by
                \auto{pointwise_product_iff,subseteq}.
            If $x\in B$, then $z\in U$ by
                \auto{pointwise_product_iff,subseteq}.
            $x\in A\union B$.
            Hence $x\in A$ or $x\in B$ by \obv{union_iff}.
        \end{subproof}
        Hence
            $\pointwiseProduct{\MultiplicativeMonoid{R}}{T}{A\union B}
                \subseteq U$ by \obv{subseteq}.
    \end{subproof}
    Follows by \obv{left_bounded_in}.
\end{proof}

\begin{theorem}\label{finite_union_left_bounded}
    Let $R$ be a topological ring.
    Suppose $H$ is finite and every element of $H$ is left bounded in $R$.
    Then $\unions{H}$ is left bounded in $R$.
\end{theorem}
\begin{proof}
    Let $C=\{A\in\pow{\carrier[R]}\mid
        \text{$A$ is left bounded in $R$}\}$.
    $\emptyset\subseteq\carrier[R]$ by \obv{emptyset_subseteq}.
    Hence $\emptyset\in\pow{\carrier[R]}$ by \obv{powerset_intro}.
    $\emptyset$ is left bounded in $R$ by \obv{left_bounded_emptyset}.
    Hence $\emptyset\in C$.
    Show that for all $A,B\in C$ we have $A\union B\in C$.
    \begin{subproof}
        Fix $A,B\in C$.
        $A$ and $B$ are left bounded in $R$.
        Hence $A\union B$ is left bounded in $R$ by
            \obv{left_bounded_binary_union}.
        Hence $A\union B\subseteq\carrier[R]$ by
            \obv{left_bounded_in_carrier}.
        Hence $A\union B\in\pow{\carrier[R]}$ by \obv{powerset_intro}.
    \end{subproof}
    Show that every element of $H$ is an element of $C$.
    \begin{subproof}
        Fix $A\in H$.
        $A$ is left bounded in $R$.
        Hence $A\subseteq\carrier[R]$ by \obv{left_bounded_in_carrier}.
        Hence $A\in\pow{\carrier[R]}$ by \obv{powerset_intro}.
    \end{subproof}
    Hence $H\subseteq C$ by \obv{subseteq}.
    Hence $\unions{H}\in C$ by
        \obv{finite_family_union_in_binary_union_closed_family}.
\end{proof}

\begin{theorem}\label{finite_subset_left_bounded}
    Let $R$ be a topological ring.
    Suppose $B$ is finite and $B\subseteq\carrier[R]$.
    Then $B$ is left bounded in $R$.
\end{theorem}
\begin{proof}
    Let $C=\{Y\in\pow{\carrier[R]}\mid
        \text{$Y$ is left bounded in $R$}\}$.
    $\emptyset\subseteq\carrier[R]$ by \obv{emptyset_subseteq}.
    Hence $\emptyset\in\pow{\carrier[R]}$ by \obv{powerset_intro}.
    $\emptyset$ is left bounded in $R$ by \obv{left_bounded_emptyset}.
    Hence $\emptyset\in C$.
    Show that for all $Y,y$ such that $Y\in C$ and $y\in B$ we have
        $\cons{y}{Y}\in C$.
    \begin{subproof}
        Fix $Y,y$.
        Assume $Y\in C$ and $y\in B$.
        $Y$ is left bounded in $R$.
        Hence $Y\subseteq\carrier[R]$ by \obv{left_bounded_in_carrier}.
        $y\in\carrier[R]$ by \obv{subseteq}.
        Hence $\{y\}$ is left bounded in $R$ by
            \obv{left_bounded_singleton}.
        Hence $\{y\}\union Y$ is left bounded in $R$ by
            \obv{left_bounded_binary_union}.
        $\cons{y}{Y}=\{y\}\union Y$ by
            \auto{setext,cons_iff,singleton_iff,union_iff}.
        Hence $\cons{y}{Y}$ is left bounded in $R$.
        $\cons{y}{Y}\subseteq\carrier[R]$ by
            \auto{subseteq,cons_iff}.
        Hence $\cons{y}{Y}\in\pow{\carrier[R]}$ by
            \obv{powerset_intro}.
    \end{subproof}
    Hence $B\in C$ by \obv{finite_induction}.
\end{proof}

\begin{theorem}\label{left_bounded_pointwise_product}
    Suppose $A$ and $B$ are left bounded in $R$.
    Then $\pointwiseProduct{\MultiplicativeMonoid{R}}{A}{B}$ is left bounded
        in $R$.
\end{theorem}
\begin{proof}
    Let $M=\MultiplicativeMonoid{R}$.
    $R$ is a topological ring by \obv{left_bounded_in_ambient}.
    Hence $R$ is a ring by \obv{topological_ring_is_ring}.
    $M$ is a monoid by \obv{multiplicative_monoid_is_exact_monoid}.
    Hence $M$ is a semigroup by \obv{monoid_implies_semigroup}.
    $\carrier[M]=\carrier[R]$ by \auto{multiplicative_monoid_fields}.
    $A,B\subseteq\carrier[R]$ by \obv{left_bounded_in_carrier}.
    Hence $A,B\subseteq\carrier[M]$.
    Hence $\pointwiseProduct{M}{A}{B}\subseteq\carrier[M]$ by
        \obv{pointwise_product_type}.
    Hence $\pointwiseProduct{M}{A}{B}\subseteq\carrier[R]$.
    Show that for every $U\in\neighbourhoods{\ringZero[R]}{R}$ there exists
        $V\in\neighbourhoods{\ringZero[R]}{R}$ such that
        $\pointwiseProduct{M}{V}{\pointwiseProduct{M}{A}{B}}\subseteq U$.
    \begin{subproof}
        Fix $U\in\neighbourhoods{\ringZero[R]}{R}$.
        Take $W\in\neighbourhoods{\ringZero[R]}{R}$ such that
            $\pointwiseProduct{\MultiplicativeMonoid{R}}{W}{B}\subseteq U$
            by \obv{left_bounded_in_neighbourhood}.
        Take $V\in\neighbourhoods{\ringZero[R]}{R}$ such that
            $\pointwiseProduct{\MultiplicativeMonoid{R}}{V}{A}\subseteq W$
            by \obv{left_bounded_in_neighbourhood}.
        $V,W\subseteq\carrier[R]$ by
            \auto{topological_ring_zero_neighbourhoods_is_filter,filter}.
        Hence $V,W\subseteq\carrier[M]$.
        $\pointwiseProduct{M}{W}{B}\subseteq U$.
        $\pointwiseProduct{M}{V}{\pointwiseProduct{M}{A}{B}}
            =\pointwiseProduct{M}{\pointwiseProduct{M}{V}{A}}{B}$ by
            \obv{pointwise_product_associative}.
        $\pointwiseProduct{M}{\pointwiseProduct{M}{V}{A}}{B}
            \subseteq\pointwiseProduct{M}{W}{B}$ by
            \obv{pointwise_product_monotone,subseteq_refl}.
        Hence $\pointwiseProduct{M}{\pointwiseProduct{M}{V}{A}}{B}
            \subseteq U$ by \obv{subseteq_transitive}.
        Hence $\pointwiseProduct{M}{V}{\pointwiseProduct{M}{A}{B}}
            \subseteq U$.
    \end{subproof}
    Follows by \auto{left_bounded_in}.
\end{proof}

\begin{corollary}\label{left_bounded_left_fixed_multiplication}
    Suppose $B$ is left bounded in $R$ and $a\in\carrier[R]$.
    Then $\pointwiseProduct{\MultiplicativeMonoid{R}}{\{a\}}{B}$ is left
        bounded in $R$.
\end{corollary}
\begin{proof}
    $\{a\}$ is left bounded in $R$ by \obv{left_bounded_singleton}.
    Follows by \obv{left_bounded_pointwise_product}.
\end{proof}

\begin{corollary}\label{left_bounded_right_fixed_multiplication}
    Suppose $B$ is left bounded in $R$ and $a\in\carrier[R]$.
    Then $\pointwiseProduct{\MultiplicativeMonoid{R}}{B}{\{a\}}$ is left
        bounded in $R$.
\end{corollary}
\begin{proof}
    $\{a\}$ is left bounded in $R$ by \obv{left_bounded_singleton}.
    Follows by \obv{left_bounded_pointwise_product}.
\end{proof}

\subsection{Additive generation}

\begin{theorem}\label{nonarchimedean_generated_subgroup_left_bounded}
    Let $R$ be a nonarchimedean topological ring.
    Suppose $A$ is left bounded in $R$.
    Then $\generatedSubgroup{A}{\AdditiveGroup{R}}$ is left bounded in $R$.
\end{theorem}
\begin{proof}
    Let $G=\AdditiveGroup{R}$ and $M=\MultiplicativeMonoid{R}$.
    $R$ is a topological ring by
        \obv{nonarchimedean_topological_ring_is_topological_ring}.
    Hence $R$ is a ring by \obv{topological_ring_is_ring}.
    $G$ is a group by \obv{additive_group_is_group}.
    $M$ is a monoid by \obv{multiplicative_monoid_is_exact_monoid}.
    $\carrier[G]=\carrier[R]$ and $\carrier[M]=\carrier[R]$ and
        $\mul[M]=\ringMul[R]$ by
        \auto{additive_group_fields,multiplicative_monoid_fields}.
    $A\subseteq\carrier[R]$ by \obv{left_bounded_in_carrier}.
    Hence $A\subseteq\carrier[G]$.
    $\generatedSubgroup{A}{G}\subseteq\carrier[G]$ by
        \obv{generated_subgroup_subseteq_carrier}.
    Hence $\generatedSubgroup{A}{G}\subseteq\carrier[R]$.
    Show that for every $U\in\neighbourhoods{\ringZero[R]}{R}$ there exists
        $V\in\neighbourhoods{\ringZero[R]}{R}$ such that
        $\pointwiseProduct{M}{V}{\generatedSubgroup{A}{G}}\subseteq U$.
    \begin{subproof}
        Fix $U\in\neighbourhoods{\ringZero[R]}{R}$.
        Take $H$ such that $\subgroup{H}{G}$ and $H$ is open in $R$ and
            $H\subseteq U$ by
            \auto{nonarchimedean_topological_ring_iff_open_subgroup_refinement}.
        Hence $H\in\neighbourhoods{\ringZero[R]}{R}$ by
            \obv{additive_subgroup_open_iff_neighbourhood_of_zero}.
        Take $V\in\neighbourhoods{\ringZero[R]}{R}$ such that
            $\pointwiseProduct{\MultiplicativeMonoid{R}}{V}{A}\subseteq H$
            by \obv{left_bounded_in_neighbourhood}.
        $V\subseteq\carrier[R]$ by
            \auto{topological_ring_zero_neighbourhoods_is_filter,filter}.
        $\pointwiseProduct{M}{V}{A}\subseteq H$.
        Show that every element of
            $\pointwiseProduct{M}{V}{\generatedSubgroup{A}{G}}$ is an
            element of $H$.
        \begin{subproof}
            Fix $z\in\pointwiseProduct{M}{V}{\generatedSubgroup{A}{G}}$.
            Take $v,y$ such that $v\in V$ and
                $y\in\generatedSubgroup{A}{G}$ and $z=\mul[M](v,y)$ by
                \obv{pointwise_product_iff}.
            $v\in\carrier[R]$ by \obv{subseteq}.
            Show that for every $a\in A$ we have
                $\ringMul[R](v,a)\in H$.
            \begin{subproof}
                Fix $a\in A$.
                $\mul[M](v,a)\in\pointwiseProduct{M}{V}{A}$ by
                    \obv{pointwise_product_iff}.
                Hence $\mul[M](v,a)\in H$ by \obv{subseteq}.
            \end{subproof}
            Hence $\ringMul[R](v,y)\in H$ by
                \obv{generated_subgroup_closed_under_left_ring_multiplication}.
        \end{subproof}
        Hence $\pointwiseProduct{M}{V}{\generatedSubgroup{A}{G}}\subseteq H$
            by \obv{subseteq}.
        Hence $\pointwiseProduct{M}{V}{\generatedSubgroup{A}{G}}\subseteq U$
            by \obv{subseteq_transitive}.
    \end{subproof}
    Follows by \auto{left_bounded_in}.
\end{proof}

\subsection{Bounded subsets of commutative topological rings}

\begin{definition}\label{bounded_in_commutative_topological_ring}
    $\Bounded{R}{B}$ iff
        $R$ is a commutative topological ring and $B$ is left bounded in $R$.
\end{definition}

\begin{proposition}\label{bounded_iff_left_bounded}
    Let $R$ be a commutative topological ring.
    Then $\Bounded{R}{B}$ iff $B$ is left bounded in $R$.
\end{proposition}
\begin{proof}
    Follows by \auto{bounded_in_commutative_topological_ring}.
\end{proof}

\begin{corollary}\label{bounded_downward}
    Suppose $\Bounded{R}{A}$ and $B\subseteq A$.
    Then $\Bounded{R}{B}$.
\end{corollary}
\begin{proof}
    $R$ is a commutative topological ring and $A$ is left bounded in $R$ by
        \obv{bounded_in_commutative_topological_ring}.
    Hence $B$ is left bounded in $R$ by \obv{left_bounded_downward}.
    Follows by \obv{bounded_in_commutative_topological_ring}.
\end{proof}

\begin{corollary}\label{bounded_binary_union}
    Suppose $\Bounded{R}{A}$ and $\Bounded{R}{B}$.
    Then $\Bounded{R}{A\union B}$.
\end{corollary}
\begin{proof}
    $R$ is a commutative topological ring and
        $A$ and $B$ are left bounded in $R$ by
        \obv{bounded_in_commutative_topological_ring}.
    Hence $A\union B$ is left bounded in $R$ by
        \obv{left_bounded_binary_union}.
    Follows by \obv{bounded_in_commutative_topological_ring}.
\end{proof}

\begin{theorem}\label{bounded_iff_right_neighbourhood}
    Let $R$ be a commutative topological ring.
    Then $\Bounded{R}{B}$ iff
        $B\subseteq\carrier[R]$ and
        for every $U\in\neighbourhoods{\ringZero[R]}{R}$ there exists
            $V\in\neighbourhoods{\ringZero[R]}{R}$ such that
            $\pointwiseProduct{\MultiplicativeMonoid{R}}{B}{V}\subseteq U$.
\end{theorem}
\begin{proof}
    $R$ is a topological ring by
        \obv{commutative_topological_ring_is_topological_ring}.
    $R$ is a commutative ring by
        \obv{commutative_topological_ring_is_commutative_ring}.
    Show that if $\Bounded{R}{B}$, then
        $B\subseteq\carrier[R]$ and
        for every $U\in\neighbourhoods{\ringZero[R]}{R}$ there exists
            $V\in\neighbourhoods{\ringZero[R]}{R}$ such that
            $\pointwiseProduct{\MultiplicativeMonoid{R}}{B}{V}\subseteq U$.
    \begin{subproof}
        Assume $\Bounded{R}{B}$.
        Hence $B$ is left bounded in $R$ by \obv{bounded_iff_left_bounded}.
        Hence $B\subseteq\carrier[R]$ by \obv{left_bounded_in_carrier}.
        Show that for every $U\in\neighbourhoods{\ringZero[R]}{R}$ there
            exists $V\in\neighbourhoods{\ringZero[R]}{R}$ such that
            $\pointwiseProduct{\MultiplicativeMonoid{R}}{B}{V}\subseteq U$.
        \begin{subproof}
            Fix $U\in\neighbourhoods{\ringZero[R]}{R}$.
            Take $V\in\neighbourhoods{\ringZero[R]}{R}$ such that
                $\pointwiseProduct{\MultiplicativeMonoid{R}}{V}{B}
                    \subseteq U$ by \obv{left_bounded_in_neighbourhood}.
            $V\subseteq\carrier[R]$ by
                \auto{topological_ring_zero_neighbourhoods_is_filter,filter}.
            $\pointwiseProduct{\MultiplicativeMonoid{R}}{B}{V}
                =\pointwiseProduct{\MultiplicativeMonoid{R}}{V}{B}$ by
                \obv{commutative_ring_pointwise_product_commutative}.
            Hence $\pointwiseProduct{\MultiplicativeMonoid{R}}{B}{V}
                \subseteq U$.
        \end{subproof}
    \end{subproof}
    Show that if $B\subseteq\carrier[R]$ and
        for every $U\in\neighbourhoods{\ringZero[R]}{R}$ there exists
            $V\in\neighbourhoods{\ringZero[R]}{R}$ such that
            $\pointwiseProduct{\MultiplicativeMonoid{R}}{B}{V}\subseteq U$,
        then $\Bounded{R}{B}$.
    \begin{subproof}
        Assume $B\subseteq\carrier[R]$ and
            for every $U\in\neighbourhoods{\ringZero[R]}{R}$ there exists
                $V\in\neighbourhoods{\ringZero[R]}{R}$ such that
                $\pointwiseProduct{\MultiplicativeMonoid{R}}{B}{V}
                    \subseteq U$.
        Show that for every $U\in\neighbourhoods{\ringZero[R]}{R}$ there
            exists $V\in\neighbourhoods{\ringZero[R]}{R}$ such that
            $\pointwiseProduct{\MultiplicativeMonoid{R}}{V}{B}\subseteq U$.
        \begin{subproof}
            Fix $U\in\neighbourhoods{\ringZero[R]}{R}$.
            Take $V\in\neighbourhoods{\ringZero[R]}{R}$ such that
                $\pointwiseProduct{\MultiplicativeMonoid{R}}{B}{V}
                    \subseteq U$.
            $V\subseteq\carrier[R]$ by
                \auto{topological_ring_zero_neighbourhoods_is_filter,filter}.
            $\pointwiseProduct{\MultiplicativeMonoid{R}}{V}{B}
                =\pointwiseProduct{\MultiplicativeMonoid{R}}{B}{V}$ by
                \obv{commutative_ring_pointwise_product_commutative}.
            Hence $\pointwiseProduct{\MultiplicativeMonoid{R}}{V}{B}
                \subseteq U$.
        \end{subproof}
        Hence $B$ is left bounded in $R$ by \obv{left_bounded_in}.
        Follows by \obv{bounded_in_commutative_topological_ring}.
    \end{subproof}
    Follows by \auto.
\end{proof}
