\section{Continuity at a point}

\import{topology/continuity.tex}
\import{topology/neighbourhood-filter.tex}
\import{topology/filter-convergence.tex}
\import{set/filter-tendsto.tex}

Continuity at a point is tendency between the corresponding neighbourhood
filters.  The endpoint and graph-typing conditions remain part of the public
predicate, while the familiar open-neighbourhood condition is its local proof
interface.

\begin{definition}\label{continuous_at}
    $f$ is continuous at $x$ from $X$ to $Y$ iff
        $X$ is a topological space and
        $Y$ is a topological space and
        $f\in\funs{\carrier[X]}{\carrier[Y]}$ and
        $x\in\carrier[X]$ and
        $f$ tends from $\neighbourhoods{x}{X}$ on $\carrier[X]$ to
            $\neighbourhoods{f(x)}{Y}$ on $\carrier[Y]$.
\end{definition}

\begin{theorem}\label{continuous_at_iff_open_local}
    Let $X,Y$ be topological spaces.
    Let $f\in\funs{\carrier[X]}{\carrier[Y]}$ and
        $x\in\carrier[X]$.
    Then $f$ is continuous at $x$ from $X$ to $Y$ iff
        for every $V\in\opens[Y]$ such that $f(x)\in V$ there exists
        $U\in\opens[X]$ such that $x\in U$ and
        for every $y\in U$ we have $f(y)\in V$.
\end{theorem}
\begin{proof}
    $f(x)\in\carrier[Y]$ by \obv{funs_type_apply}.
    $f$ is a function and $\dom{f}=\carrier[X]$ by
        \obv{funs_is_function,funs}.
    $\neighbourhoods{x}{X}$ is a filter on $\carrier[X]$ and
        $\neighbourhoods{f(x)}{Y}$ is a filter on $\carrier[Y]$ by
        \obv{neighbourhoods_is_filter}.

    Show that if $f$ is continuous at $x$ from $X$ to $Y$, then
        for every $V\in\opens[Y]$ such that $f(x)\in V$ there exists
        $U\in\opens[X]$ such that $x\in U$ and
        for every $y\in U$ we have $f(y)\in V$.
    \begin{subproof}
        Assume $f$ is continuous at $x$ from $X$ to $Y$.
        Hence $f$ tends from $\neighbourhoods{x}{X}$ on $\carrier[X]$ to
            $\neighbourhoods{f(x)}{Y}$ on $\carrier[Y]$ by
            \obv{continuous_at}.
        $f$ tends from $\neighbourhoods{x}{X}$ on $\carrier[X]$ to
            $\neighbourhoods{f(x)}{Y}$ on $\carrier[Y]$ iff
            for every $N\in\neighbourhoods{f(x)}{Y}$ we have
            $\preimg{f}{N}\in\neighbourhoods{x}{X}$ by
            \obv{filter_tendsto_iff_preimage}.
        Hence for every $N\in\neighbourhoods{f(x)}{Y}$ we have
            $\preimg{f}{N}\in\neighbourhoods{x}{X}$.
        Fix $V\in\opens[Y]$.
        Assume $f(x)\in V$.
        Hence $V\in\neighbourhoods{f(x)}{Y}$ by
            \obv{open_set_containing_point_is_neighbourhood}.
        Thus $\preimg{f}{V}\in\neighbourhoods{x}{X}$.
        Take $U\in\opens[X]$ such that
            $x\in U\subseteq\preimg{f}{V}$ by
            \obv{neighbourhoods}.
        Show that for every $y\in U$ we have $f(y)\in V$.
        \begin{subproof}
            Fix $y\in U$.
            Hence $y\in\preimg{f}{V}$ by \obv{subseteq}.
            Take $v\in V$ such that $(y,v)\in f$ by \obv{preimg_iff}.
            Hence $f(y)=v$ by \obv{function_apply_intro}.
        \end{subproof}
        Follows by \auto.
    \end{subproof}

    Show that if for every $V\in\opens[Y]$ such that $f(x)\in V$ there
        exists $U\in\opens[X]$ such that $x\in U$ and
        for every $y\in U$ we have $f(y)\in V$, then
        $f$ is continuous at $x$ from $X$ to $Y$.
    \begin{subproof}
        Assume for every $V\in\opens[Y]$ such that $f(x)\in V$ there
            exists $U\in\opens[X]$ such that $x\in U$ and
            for every $y\in U$ we have $f(y)\in V$.
        Show that for every $N\in\neighbourhoods{f(x)}{Y}$ we have
            $\preimg{f}{N}\in\neighbourhoods{x}{X}$.
        \begin{subproof}
            Fix $N\in\neighbourhoods{f(x)}{Y}$.
            Take $V\in\opens[Y]$ such that $f(x)\in V\subseteq N$ by
                \obv{neighbourhoods}.
            Take $U\in\opens[X]$ such that $x\in U$ and
                for every $y\in U$ we have $f(y)\in V$.
            Show that $U\subseteq\preimg{f}{N}$.
            \begin{subproof}
                It suffices to show that every element of $U$ is an element
                    of $\preimg{f}{N}$ by \obv{subseteq}.
                Fix $y\in U$.
                $U\subseteq\carrier[X]$ by \obv{opens_type}.
                Hence $y\in\carrier[X]$ by \obv{subseteq}.
                $\dom{f}=\carrier[X]$ by \obv{funs}.
                Hence $y\in\dom{f}$.
                Thus $(y,f(y))\in f$ by \obv{function_apply_elim}.
                $f(y)\in V$ and $V\subseteq N$.
                Hence $f(y)\in N$ by \obv{subseteq}.
                Follows by \obv{preimg_iff}.
            \end{subproof}
            $\preimg{f}{N}\subseteq\carrier[X]$ by
                \obv{preimg_subseteq_dom,funs}.
            Hence $\preimg{f}{N}\in\pow{\carrier[X]}$ by
                \obv{powerset_intro}.
            Follows by \obv{neighbourhoods}.
        \end{subproof}
        Hence $f$ tends from $\neighbourhoods{x}{X}$ on $\carrier[X]$ to
            $\neighbourhoods{f(x)}{Y}$ on $\carrier[Y]$ by
            \obv{filter_tendsto_iff_preimage}.
        Follows by \obv{continuous_at}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{theorem}\label{continuous_iff_continuous_at_every_point}
    Let $X,Y$ be topological spaces.
    Let $f\in\funs{\carrier[X]}{\carrier[Y]}$.
    Then $f$ is continuous from $X$ to $Y$ iff
        for every $x\in\carrier[X]$ we have
        $f$ is continuous at $x$ from $X$ to $Y$.
\end{theorem}
\begin{proof}
    Show that if $f$ is continuous from $X$ to $Y$, then
        for every $x\in\carrier[X]$ we have
        $f$ is continuous at $x$ from $X$ to $Y$.
    \begin{subproof}
        Assume $f$ is continuous from $X$ to $Y$.
        For every $z\in\carrier[X]$ we have
            for every $W\in\opens[Y]$ we have if $f(z)\in W$, then
            there exists $O\in\opens[X]$ such that $z\in O$ and
                for every $w\in O$ we have $f(w)\in W$ by
            \obv{continuous_iff_open_local}.
        Fix $x\in\carrier[X]$.
        Show that for every $V\in\opens[Y]$ such that $f(x)\in V$ there
            exists $U\in\opens[X]$ such that $x\in U$ and
            for every $y\in U$ we have $f(y)\in V$.
        \begin{subproof}
            Fix $V\in\opens[Y]$.
            Assume $f(x)\in V$.
        \end{subproof}
        $f$ is continuous at $x$ from $X$ to $Y$ iff
            for every $V\in\opens[Y]$ such that $f(x)\in V$ there exists
            $U\in\opens[X]$ such that $x\in U$ and
            for every $y\in U$ we have $f(y)\in V$ by
            \obv{continuous_at_iff_open_local}.
    \end{subproof}

    Show that if for every $x\in\carrier[X]$ we have
        $f$ is continuous at $x$ from $X$ to $Y$, then
        $f$ is continuous from $X$ to $Y$.
    \begin{subproof}
        Assume for every $x\in\carrier[X]$ we have
            $f$ is continuous at $x$ from $X$ to $Y$.
        Show that for every $x\in\carrier[X]$ we have
            for every $V\in\opens[Y]$ we have if $f(x)\in V$, then
            there exists $U\in\opens[X]$ such that $x\in U$ and
                for every $y\in U$ we have $f(y)\in V$.
        \begin{subproof}
            Fix $x\in\carrier[X]$.
            $f$ is continuous at $x$ from $X$ to $Y$.
            $f$ is continuous at $x$ from $X$ to $Y$ iff
                for every $V\in\opens[Y]$ such that $f(x)\in V$ there exists
                $U\in\opens[X]$ such that $x\in U$ and
                for every $y\in U$ we have $f(y)\in V$ by
                \obv{continuous_at_iff_open_local}.
            Hence for every $V\in\opens[Y]$ such that $f(x)\in V$ there
                exists $U\in\opens[X]$ such that $x\in U$ and
                for every $y\in U$ we have $f(y)\in V$.
        \end{subproof}
        $f$ is continuous from $X$ to $Y$ iff
            for every $x\in\carrier[X]$ we have
            for every $V\in\opens[Y]$ we have if $f(x)\in V$, then there
            exists $U\in\opens[X]$ such that $x\in U$ and
                for every $y\in U$ we have $f(y)\in V$ by
            \obv{continuous_iff_open_local}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{theorem}\label{continuous_map_preserves_filter_limits}
    Let $X,Y$ be topological spaces.
    Let $F$ be a filter on $\carrier[X]$.
    Suppose $f$ is continuous from $X$ to $Y$ and
        $\filterConverges{F}{x}{X}$.
    Then
        $\filterConverges{
            \filterPushforward{
                \carrier[X]}{\carrier[Y]}{f}{F}
        }{f(x)}{Y}$.
\end{theorem}
\begin{proof}
    $f\in\funs{\carrier[X]}{\carrier[Y]}$ by
        \obv{continuous_map}.
    $x\in\carrier[X]$ by \obv{filter_converges}.
    Hence $f(x)\in\carrier[Y]$ by \obv{funs_type_apply}.
    $f$ is continuous at $x$ from $X$ to $Y$ by
        \auto{continuous_iff_continuous_at_every_point}.
    Hence $f$ tends from $\neighbourhoods{x}{X}$ on $\carrier[X]$ to
        $\neighbourhoods{f(x)}{Y}$ on $\carrier[Y]$ by
        \obv{continuous_at}.
    $\neighbourhoods{x}{X}$ is a filter on $\carrier[X]$ and
        $\neighbourhoods{f(x)}{Y}$ is a filter on $\carrier[Y]$ by
        \obv{neighbourhoods_is_filter}.
    Hence for every $N\in\neighbourhoods{f(x)}{Y}$ we have
        $\preimg{f}{N}\in\neighbourhoods{x}{X}$ by
        \obv{filter_tendsto_iff_preimage}.
    $\neighbourhoods{x}{X}\subseteq F$ by
        \obv{filter_converges}.

    Show that $\neighbourhoods{f(x)}{Y}\subseteq
        \filterPushforward{\carrier[X]}{\carrier[Y]}{f}{F}$.
    \begin{subproof}
        It suffices to show that every element of
            $\neighbourhoods{f(x)}{Y}$ is an element of
            $\filterPushforward{\carrier[X]}{\carrier[Y]}{f}{F}$ by
            \obv{subseteq}.
        Fix $N\in\neighbourhoods{f(x)}{Y}$.
        $N\subseteq\carrier[Y]$ by \obv{filter}.
        $\preimg{f}{N}\in\neighbourhoods{x}{X}$.
        Hence $\preimg{f}{N}\in F$ by \obv{subseteq}.
        Follows by \obv{filter_pushforward_iff}.
    \end{subproof}
    Follows by \obv{filter_converges}.
\end{proof}
