\section{Paths in real intervals}

An order-convex real subset contains the straight path between ordered
endpoints. Reversal removes the ordering restriction. This also covers empty
subsets and singletons.

\begin{theorem}\label{real_order_convex_set_is_path_connected}
    Suppose $A$ is order convex in $\RealField$.
    Then $\subspaceTopology{\RealTopology}{A}$ is path-connected.
\end{theorem}
\begin{proof}
    Let $S=\subspaceTopology{\RealTopology}{A}$.
    $A\subseteq\reals$ by \auto{order_convex,surreal_real_field_carrier}.
    $\RealTopology$ is a topological space and $\carrier[\RealTopology]=\reals$ by
        \obv{real_topology_is_topological_space,real_topology_carrier}.
    $S$ is a topological space and $\carrier[S]=A$ by
        \auto{subspace_topology_is_topological_space,subspace_topology_carrier_of_subset}.
    Show that for all $a,b\in A$ such that $\surrealLe{a}{b}$ there exists $p$
        such that $p$ is a path from $a$ to $b$ in $S$.
    \begin{subproof}
        Fix $a,b\in A$.
        Assume $\surrealLe{a}{b}$.
        $a,b\in\reals$ by \obv{subseteq}.
        Let $B=\realClosedInterval{a}{b}$.
        $B\subseteq A$ by
            \auto{subseteq,real_closed_interval_member_iff,order_convex,
                surreal_real_field_carrier,surreal_real_field_order,surreal_le}.
        Let $p=\realMapAffine{\RealUnitInterval}{a}{\surrealAdd{b}{\surrealNeg{a}}}
            {\identity{\RealUnitInterval}}$.
        $p$ is a path from $a$ to $b$ in $\subspaceTopology{\RealTopology}{B}$ by
            \auto{real_affine_path_in_closed_interval}.
        $p$ is a path from $a$ to $b$ in $\RealTopology$ by
            \obv{path_subspace_codomain_inclusion}.
        $p\in\funs{\RealUnitInterval}{\carrier[\subspaceTopology{\RealTopology}{B}]}$ by
            \auto{path_from_to,continuous_map,real_unit_interval_topology_carrier}.
        Hence $\ran{p}\subseteq\carrier[\subspaceTopology{\RealTopology}{B}]$ by \obv{funs_ran}.
        $\ran{p}\subseteq A$ by \auto{subspace_topology_carrier,inter,subseteq}.
        $p$ is a path from $a$ to $b$ in $S$ by \obv{path_codomain_corestriction}.
    \end{subproof}
    Show that for all $a,b\in\carrier[S]$ there exists $p$ such that
        $p$ is a path from $a$ to $b$ in $S$.
    \begin{subproof}
        Fix $a,b\in\carrier[S]$.
        $a,b\in A$ and $a,b\in\reals$ by \auto{subseteq}.
        $\surrealLe{a}{b}$ or $\surrealLe{b}{a}$ by \obv{surreal_le_total,surreal_real_is_surreal}.
        \begin{byCase}
            \caseOf{$\surrealLe{a}{b}$.}
                Follows.
            \caseOf{$\surrealLe{b}{a}$.}
                Take $q$ such that $q$ is a path from $b$ to $a$ in $S$.
                Follows by \obv{path_can_be_reversed}.
        \end{byCase}
    \end{subproof}
    Follows by \auto{path_connected}.
\end{proof}

\begin{theorem}\label{real_open_interval_is_path_connected}
    Let $a,b\in\reals$.
    Then $\subspaceTopology{\RealTopology}{\realOpenInterval{a}{b}}$ is path-connected.
\end{theorem}
\begin{proof}
    $\RealField$ is an ordered set by
        \obv{surreal_real_field_is_linearly_ordered_set,linearly_ordered_set_is_ordered_set}.
    $a,b\in\carrier[\RealField]$ by \obv{surreal_real_field_carrier}.
    $\orderOpenInterval{\RealField}{a}{b}$ is order convex in $\RealField$ by
        \obv{order_open_interval_is_order_convex}.
    $\realOpenInterval{a}{b}$ is order convex in $\RealField$ by
        \auto{real_order_open_interval_eq}.
    Follows by \obv{real_order_convex_set_is_path_connected}.
\end{proof}

\begin{theorem}\label{real_closed_interval_is_path_connected}
    Let $a,b\in\reals$.
    Then $\subspaceTopology{\RealTopology}{\realClosedInterval{a}{b}}$ is path-connected.
\end{theorem}
\begin{proof}
    $\RealField$ is an ordered set by
        \obv{surreal_real_field_is_linearly_ordered_set,linearly_ordered_set_is_ordered_set}.
    $a,b\in\carrier[\RealField]$ by \obv{surreal_real_field_carrier}.
    $\orderClosedInterval{\RealField}{a}{b}$ is order convex in $\RealField$ by
        \obv{order_closed_interval_is_order_convex}.
    $\realClosedInterval{a}{b}$ is order convex in $\RealField$ by
        \auto{real_order_closed_interval_eq}.
    Follows by \obv{real_order_convex_set_is_path_connected}.
\end{proof}

\begin{theorem}\label{real_topology_is_locally_path_connected}
    $\RealTopology$ is locally path-connected.
\end{theorem}
\begin{proof}
    $\RealTopology$ is a topological space by \obv{real_topology_is_topological_space}.
    Show that for all $U,x$ such that $U\in\opens[\RealTopology]$ and $x\in U$ there exists
        $V\in\opens[\RealTopology]$ such that $x\in V\subseteq U$ and
        $\subspaceTopology{\RealTopology}{V}$ is path-connected.
    \begin{subproof}
        Fix $U,x$.
        Assume $U\in\opens[\RealTopology]$ and $x\in U$.
        Take $a,b$ such that $a,b\in\reals$ and $x\in\realOpenInterval{a}{b}\subseteq U$ by
            \obv{real_open_neighbourhood_contains_interval}.
        $\realOpenInterval{a}{b}\in\opens[\RealTopology]$ by \obv{real_open_interval_is_open}.
        $\subspaceTopology{\RealTopology}{\realOpenInterval{a}{b}}$ is path-connected by
            \obv{real_open_interval_is_path_connected}.
    \end{subproof}
    Follows by \obv{locally_path_connected}.
\end{proof}
