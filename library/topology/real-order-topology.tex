The generic intervals of the ordered real field agree with the established
real interval sets.  We use those literal set equalities to identify the
generated order topology with the canonical real metric topology; no second
real-topology value is introduced.

\begin{proposition}\label{real_field_order_strict_iff}
    Let $x,y\in\reals$.
    Then $x\mathrel{\orderStrictRelation{\RealField}}y$ iff
        $\surrealLt{x}{y}$.
\end{proposition}
\begin{proof}
    Follows by
        \obv{ordered_set_strict_relation,
            ordered_algebra_strict_relation,
            surreal_real_field_strict_order}.
\end{proof}

\begin{theorem}\label{real_metric_ball_eq_order_open_interval}
    Let $x,r\in\reals$.
    Then $\metricBall{\RealMetric}{x}{r}
        =\orderOpenInterval{\RealField}
            {\surrealAdd{x}{\surrealNeg{r}}}{\surrealAdd{x}{r}}$.
\end{theorem}
\begin{proof}
    Let $l=\surrealAdd{x}{\surrealNeg{r}}$.
    Let $u=\surrealAdd{x}{r}$.
    $\surrealNeg{r}\in\reals$ by \obv{surreal_real_negation_closed}.
    Hence $l,u\in\reals$ by \auto{surreal_real_addition_closed}.
    Hence $l,u\in\carrier[\RealField]$ by
        \obv{surreal_real_field_carrier}.

    Show that every element of $\metricBall{\RealMetric}{x}{r}$ is an
        element of $\orderOpenInterval{\RealField}{l}{u}$.
    \begin{subproof}
        Fix $y\in\metricBall{\RealMetric}{x}{r}$.
        Hence $y\in\carrier[\RealMetric]$ and
            $\surrealLt{\metricDistance[\RealMetric](x,y)}{r}$ by
            \obv{metric_ball}.
        Hence $y\in\reals$ by \obv{surreal_real_metric_carrier}.
        $\metricDistance[\RealMetric](x,y)
            =\abs{\surrealAdd{x}{\surrealNeg{y}}}$ by
            \obv{surreal_real_metric_distance}.
        Hence $\surrealLt
            {\abs{\surrealAdd{x}{\surrealNeg{y}}}}{r}$.
        Hence $\surrealLt{x}{\surrealAdd{y}{r}}$ and
            $\surrealLt{y}{u}$ by
            \auto{surreal_abs_difference_lt_iff,surreal_real_is_surreal}.
        $\surrealLt{l}{y}$ iff
            $\surrealLt{x}{\surrealAdd{y}{r}}$ by
            \auto{surreal_difference_lt_iff,surreal_add_commutative,
                surreal_real_is_surreal}.
        Hence $\surrealLt{l}{y}$.
        $y\in\carrier[\RealField]$ by
            \obv{surreal_real_field_carrier}.
        Follows by
            \obv{order_open_interval_iff,real_field_order_strict_iff}.
    \end{subproof}
    Hence $\metricBall{\RealMetric}{x}{r}\subseteq
        \orderOpenInterval{\RealField}{l}{u}$ by \obv{subseteq}.

    Show that every element of $\orderOpenInterval{\RealField}{l}{u}$
        is an element of $\metricBall{\RealMetric}{x}{r}$.
    \begin{subproof}
        Fix $y\in\orderOpenInterval{\RealField}{l}{u}$.
        Hence $y\in\carrier[\RealField]$ by
            \obv{order_open_interval}.
        Hence $y\in\reals$ by \obv{surreal_real_field_carrier}.
        Hence $\surrealLt{l}{y}$ and $\surrealLt{y}{u}$ by
            \auto{order_open_interval_iff,real_field_order_strict_iff}.
        $\surrealLt{l}{y}$ iff
            $\surrealLt{x}{\surrealAdd{y}{r}}$ by
            \auto{surreal_difference_lt_iff,surreal_add_commutative,
                surreal_real_is_surreal}.
        Hence $\surrealLt{x}{\surrealAdd{y}{r}}$.
        Hence $\surrealLt
            {\abs{\surrealAdd{x}{\surrealNeg{y}}}}{r}$ by
            \auto{surreal_abs_difference_lt_iff,surreal_real_is_surreal}.
        $\metricDistance[\RealMetric](x,y)
            =\abs{\surrealAdd{x}{\surrealNeg{y}}}$ by
            \obv{surreal_real_metric_distance}.
        Hence $\surrealLt{\metricDistance[\RealMetric](x,y)}{r}$.
        $y\in\carrier[\RealMetric]$ by
            \obv{surreal_real_metric_carrier}.
        Follows by \obv{metric_ball}.
    \end{subproof}
    Hence $\orderOpenInterval{\RealField}{l}{u}\subseteq
        \metricBall{\RealMetric}{x}{r}$ by \obv{subseteq}.
    Hence $\metricBall{\RealMetric}{x}{r}
        =\orderOpenInterval{\RealField}{l}{u}$ by
        \obv{subseteq_antisymmetric}.
\end{proof}

\begin{theorem}\label{real_order_topology_prebasis_is_metric_open}
    $\orderTopologyPrebasis{\RealField}\subseteq\opens[\RealTopology]$.
\end{theorem}
\begin{proof}
    Show that every element of $\orderTopologyPrebasis{\RealField}$ is
        an element of $\opens[\RealTopology]$.
    \begin{subproof}
        Fix $U\in\orderTopologyPrebasis{\RealField}$.
        Take $a\in\carrier[\RealField]$ such that
            $U=\orderOpenLowerRay{\RealField}{a}$ or
            $U=\orderOpenUpperRay{\RealField}{a}$ by
            \obv{order_topology_prebasis}.
        Hence $a\in\reals$ by \obv{surreal_real_field_carrier}.
        $\orderOpenLowerRay{\RealField}{a}=\realOpenLowerRay{a}$ and
            $\orderOpenUpperRay{\RealField}{a}=\realOpenUpperRay{a}$ by
            \obv{real_order_open_lower_ray_eq,
                real_order_open_upper_ray_eq}.
        $\realOpenLowerRay{a},\realOpenUpperRay{a}
            \in\opens[\RealTopology]$ by
            \obv{real_open_lower_ray_is_open,
                real_open_upper_ray_is_open}.
    \end{subproof}
    Follows by \obv{subseteq}.
\end{proof}

\begin{theorem}\label{real_order_opens_subset_metric_opens}
    $\opens[\orderTopology{\RealField}]\subseteq
        \opens[\RealTopology]$.
\end{theorem}
\begin{proof}
    $\opens[\RealTopology]$ is a topology on
        $\carrier[\RealTopology]$ by
        \obv{topological_space_opens_form_topology,
            real_topology_is_topological_space}.
    $\carrier[\RealTopology]=\reals=\carrier[\RealField]$ by
        \obv{real_topology_carrier,surreal_real_field_carrier}.
    Hence $\opens[\RealTopology]$ is a topology on
        $\carrier[\RealField]$.
    $\orderTopologyPrebasis{\RealField}$ is a family of subsets of
        $\carrier[\RealField]$ by \obv{order_topology_prebasis_type}.
    $\orderTopologyPrebasis{\RealField}\subseteq
        \opens[\RealTopology]$ by
        \obv{real_order_topology_prebasis_is_metric_open}.
    Hence $\generatedOpens{\carrier[\RealField]}
        {\orderTopologyPrebasis{\RealField}}
        \subseteq\opens[\RealTopology]$ by
        \obv{generated_opens_least}.
    $\opens[\orderTopology{\RealField}]
        =\generatedOpens{\carrier[\RealField]}
            {\orderTopologyPrebasis{\RealField}}$ by
        \obv{order_topology_opens}.
\end{proof}

\begin{theorem}\label{real_metric_opens_subset_order_opens}
    $\metricOpens{\RealMetric}\subseteq
        \opens[\orderTopology{\RealField}]$.
\end{theorem}
\begin{proof}
    $\RealField$ is a linearly ordered set by
        \obv{surreal_real_field_is_linearly_ordered_set}.
    $\orderTopology{\RealField}$ is a topological space by
        \obv{order_topology_is_topological_space}.
    $\carrier[\orderTopology{\RealField}]
        =\carrier[\RealField]=\reals$ by
        \obv{order_topology_carrier,surreal_real_field_carrier}.
    $\RealMetric$ is a pseudometric space by
        \obv{surreal_real_pseudometric_space}.

    Show that every element of $\metricOpens{\RealMetric}$ is an element
        of $\opens[\orderTopology{\RealField}]$.
    \begin{subproof}
        Fix $A\in\metricOpens{\RealMetric}$.
        Hence $\metricOpen{\RealMetric}{A}$ by
            \obv{metric_opens_iff}.
        Hence $A\subseteq\carrier[\RealMetric]$ by \obv{metric_open}.
        $\carrier[\RealMetric]=\reals
            =\carrier[\orderTopology{\RealField}]$ by
            \obv{surreal_real_metric_carrier,order_topology_carrier,
                surreal_real_field_carrier}.
        Hence $A\subseteq\carrier[\orderTopology{\RealField}]$.
        Show that for every $x\in A$ we have
            $A\in\neighbourhoods{x}{\orderTopology{\RealField}}$.
        \begin{subproof}
            Fix $x\in A$.
            Hence $x\in\carrier[\RealMetric]$ and $x\in\reals$ by
                \obv{subseteq,surreal_real_metric_carrier}.
            Take $r\in\reals$ such that
                $\surrealLt{\surrealZero}{r}$ and
                $\metricBall{\RealMetric}{x}{r}\subseteq A$ by
                \obv{metric_open}.
            $\surrealNeg{r}\in\reals$ by
                \obv{surreal_real_negation_closed}.
            Hence $\surrealAdd{x}{\surrealNeg{r}},\surrealAdd{x}{r}
                \in\reals$ by \obv{surreal_real_addition_closed}.
            Hence
                $\surrealAdd{x}{\surrealNeg{r}},\surrealAdd{x}{r}
                    \in\carrier[\RealField]$ by
                \obv{surreal_real_field_carrier}.
            $\orderOpenInterval{\RealField}
                {\surrealAdd{x}{\surrealNeg{r}}}{\surrealAdd{x}{r}}
                \in\opens[\orderTopology{\RealField}]$ by
                \obv{order_open_interval_is_open}.
            $\metricBall{\RealMetric}{x}{r}
                =\orderOpenInterval{\RealField}
                    {\surrealAdd{x}{\surrealNeg{r}}}
                    {\surrealAdd{x}{r}}$ by
                \obv{real_metric_ball_eq_order_open_interval}.
            Hence $\metricBall{\RealMetric}{x}{r}
                \in\opens[\orderTopology{\RealField}]$.
            $x\in\metricBall{\RealMetric}{x}{r}$ by
                \obv{metric_ball_contains_centre}.
            $A\in\pow{\carrier[\orderTopology{\RealField}]}$ by
                \obv{powerset_intro}.
            Follows by \obv{neighbourhoods}.
        \end{subproof}
        $A\in\opens[\orderTopology{\RealField}]$ iff
            $A\subseteq\carrier[\orderTopology{\RealField}]$ and
            for every $x\in A$ we have
                $A\in\neighbourhoods{x}{\orderTopology{\RealField}}$ by
            \obv{open_iff_neighbourhood_of_each_point}.
    \end{subproof}
    Follows by \obv{subseteq}.
\end{proof}

\begin{theorem}\label{real_metric_opens_subset_order_topology_opens}
    $\opens[\RealTopology]\subseteq
        \opens[\orderTopology{\RealField}]$.
\end{theorem}
\begin{proof}
    $\opens[\RealTopology]=\metricOpens{\RealMetric}$ by
        \auto{real_topology,metric_topology_opens}.
    Follows by \obv{real_metric_opens_subset_order_opens}.
\end{proof}

\begin{theorem}\label{real_order_topology_eq_metric_topology}
    $\orderTopology{\RealField}=\RealTopology$.
\end{theorem}
\begin{proof}
    $\RealField$ is a linearly ordered set by
        \obv{surreal_real_field_is_linearly_ordered_set}.
    $\RealMetric$ is a pseudometric space by
        \obv{surreal_real_pseudometric_space}.
    $\orderTopology{\RealField}$ is an exact topological space by
        \obv{order_topology_is_exact}.
    $\RealTopology$ is an exact topological space by
        \auto{real_topology,metric_topology_is_exact}.
    $\carrier[\orderTopology{\RealField}]=\carrier[\RealField]=\reals$
        by \obv{order_topology_carrier,surreal_real_field_carrier}.
    $\carrier[\RealTopology]=\reals$ by \obv{real_topology_carrier}.
    Hence $\carrier[\orderTopology{\RealField}]
        =\carrier[\RealTopology]$.
    $\opens[\orderTopology{\RealField}]\subseteq
        \opens[\RealTopology]$ by
        \obv{real_order_opens_subset_metric_opens}.
    $\opens[\RealTopology]\subseteq
        \opens[\orderTopology{\RealField}]$ by
        \obv{real_metric_opens_subset_order_topology_opens}.
    Hence $\opens[\orderTopology{\RealField}]
        =\opens[\RealTopology]$ by \obv{subseteq_antisymmetric}.
    Follows by structure extensionality.
\end{proof}

\begin{theorem}\label{real_open_neighbourhood_contains_interval}
    Let $U\in\opens[\RealTopology]$ and $x\in U$.
    Then there exist $a,b$ such that $a,b\in\reals$ and
        $x\in\realOpenInterval{a}{b}\subseteq U$.
\end{theorem}
\begin{proof}
    $\metricOpen{\RealMetric}{U}$ by
        \auto{real_topology,metric_topology_opens,metric_opens_iff}.
    $x\in\reals$ by \auto{metric_open,subseteq,surreal_real_metric_carrier}.
    Take $r\in\reals$ such that $\surrealLt{\surrealZero}{r}$ and
        $\metricBall{\RealMetric}{x}{r}\subseteq U$ by \obv{metric_open}.
    Let $a=\surrealAdd{x}{\surrealNeg{r}}$ and $b=\surrealAdd{x}{r}$.
    $a,b\in\reals$ by \auto{surreal_real_negation_closed,surreal_real_addition_closed}.
    $\metricBall{\RealMetric}{x}{r}=\orderOpenInterval{\RealField}{a}{b}$ by
        \auto{real_metric_ball_eq_order_open_interval}.
    $\orderOpenInterval{\RealField}{a}{b}=\realOpenInterval{a}{b}$ by
        \obv{real_order_open_interval_eq}.
    $x\in\metricBall{\RealMetric}{x}{r}$ by
        \auto{metric_ball_contains_centre,surreal_real_pseudometric_space,surreal_real_metric_carrier}.
    Follows by \auto{real_order_open_interval_eq}.
\end{proof}
