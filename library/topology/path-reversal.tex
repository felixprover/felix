\section{Reversing paths}

\import{topology/real-path.tex}

Reversal uses the affine self-map $t\mapsto1-t$ of the unit interval.
Composition changes the direction of a path without leaving its space.

\begin{theorem}\label{unit_interval_reverse_path}
    Let $r=\realMapAffine{\RealUnitInterval}{\surrealOne}
        {\surrealAdd{\surrealZero}{\surrealNeg{\surrealOne}}}{\identity{\RealUnitInterval}}$.
    Then $r$ is a path from $\surrealOne$ to $\surrealZero$ in $\RealUnitIntervalTopology$.
\end{theorem}
\begin{proof}
    $\surrealZero,\surrealOne\in\reals$ by \obv{surreal_zero_is_real,surreal_one_is_real}.
    $r$ is a path from $\surrealOne$ to $\surrealZero$ in $\RealTopology$ by
        \auto{real_affine_path}.
    $\RealTopology$ is a topological space and $\carrier[\RealTopology]=\reals$ by
        \obv{real_topology_is_topological_space,real_topology_carrier}.
    $\RealUnitInterval\subseteq\reals$ by \obv{real_unit_interval_subset_reals}.
    $\identity{\RealUnitInterval}\in\funs{\RealUnitInterval}{\reals}$ by
        \auto{id_elem_funs,funs_weaken_codom}.
    $\surrealNeg{\surrealOne}\in\reals$ and
        $\surrealAdd{\surrealZero}{\surrealNeg{\surrealOne}}\in\reals$ by
        \auto{surreal_real_negation_closed,surreal_real_addition_closed}.
    Show that for every $t\in\RealUnitInterval$ we have $r(t)\in\RealUnitInterval$.
    \begin{subproof}
        Fix $t\in\RealUnitInterval$.
        $t\in\reals$ and $\surrealLe{\surrealZero}{t}$ and $\surrealLe{t}{\surrealOne}$ by
            \obv{real_unit_interval_iff}.
        $\surreal{t}$ and $\surreal{\surrealOne}$ and $\surreal{\surrealZero}$ by
            \obv{surreal_real_is_surreal}.
        $r(t)=\surrealAdd{\surrealOne}{\surrealNeg{t}}$ by
            \auto{real_map_affine_apply,id_apply,surreal_add_zero_left,
                surreal_negation_is_surreal,surreal_mul_neg_left,surreal_mul_one_left}.
        $\surrealLe{\surrealNeg{\surrealOne}}{\surrealNeg{t}}$ and
            $\surrealLe{\surrealNeg{t}}{\surrealZero}$ by
            \auto{surreal_negation_le_reverses,surreal_negation_empty}.
        $\surrealLe{\surrealZero}{r(t)}$ and $\surrealLe{r(t)}{\surrealOne}$ by
            \auto{surreal_add_le_right,surreal_add_inverse_right,surreal_add_zero_right,
                surreal_negation_is_surreal}.
        $r(t)\in\reals$ by \auto{surreal_real_negation_closed,surreal_real_addition_closed}.
        Follows by \obv{real_unit_interval_iff}.
    \end{subproof}
    $r\in\funs{\RealUnitInterval}{\reals}$ by
        \auto{path_from_to,continuous_map,real_unit_interval_topology_carrier}.
    Hence $r$ is a function and $\dom{r}=\RealUnitInterval$ by \obv{funs_is_function,funs}.
    Hence $r$ is a function to $\RealUnitInterval$.
    $r\in\funs{\RealUnitInterval}{\RealUnitInterval}$ by \obv{funs_intro}.
    Hence $\ran{r}\subseteq\RealUnitInterval$ by \obv{funs_ran}.
    $r$ is a path from $\surrealOne$ to $\surrealZero$ in
        $\subspaceTopology{\RealTopology}{\RealUnitInterval}$ by \obv{path_codomain_corestriction}.
    Follows by \obv{real_unit_interval_topology}.
\end{proof}

\begin{theorem}\label{path_can_be_reversed}
    Let $X$ be a topological space.
    Suppose $p$ is a path from $a$ to $b$ in $X$.
    Then there exists $q$ such that $q$ is a path from $b$ to $a$ in $X$.
\end{theorem}
\begin{proof}
    Let $r=\realMapAffine{\RealUnitInterval}{\surrealOne}
        {\surrealAdd{\surrealZero}{\surrealNeg{\surrealOne}}}{\identity{\RealUnitInterval}}$.
    $r$ is a path from $\surrealOne$ to $\surrealZero$ in $\RealUnitIntervalTopology$ by
        \obv{unit_interval_reverse_path}.
    $p$ is continuous from $\RealUnitIntervalTopology$ to $X$ by \obv{path_from_to}.
    $p\circ r$ is a path from $p(\surrealOne)$ to $p(\surrealZero)$ in $X$ by
        \auto{continuous_image_of_path,real_unit_interval_topology_is_topological_space}.
    Follows by \auto{path_from_to}.
\end{proof}
