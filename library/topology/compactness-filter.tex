\section{Compactness through filters}

\import{topology/compactness.tex}
\import{set/filter-eventual.tex}

The open-cover predicate remains the definition of compactness.  The
following theorems expose its equivalent filter interface.

\begin{theorem}\label{filter_cluster_iff_meshes_neighbourhoods}
    Let $X$ be a topological space and
        $F$ be a filter on $\carrier[X]$.
    Let $x\in\carrier[X]$.
    Then $\filterClusters{F}{x}{X}$ iff
        $F$ meshes with $\neighbourhoods{x}{X}$ on $\carrier[X]$.
\end{theorem}
\begin{proof}
    $\neighbourhoods{x}{X}$ is a filter on $\carrier[X]$ by
        \obv{neighbourhoods_is_filter}.
    Follows by \auto{filter_cluster_point,filters_mesh}.
\end{proof}

\begin{theorem}\label{filter_cluster_iff_common_refinement_proper}
    Let $X$ be a topological space and
        $F$ be a filter on $\carrier[X]$.
    Let $x\in\carrier[X]$.
    Then $\filterClusters{F}{x}{X}$ iff
        $\binaryCommonFilterRefinement{\carrier[X]}{F}{
            \neighbourhoods{x}{X}}$
        is a proper filter on $\carrier[X]$.
\end{theorem}
\begin{proof}
    $\neighbourhoods{x}{X}$ is a filter on $\carrier[X]$ by
        \obv{neighbourhoods_is_filter}.
    $F$ meshes with $\neighbourhoods{x}{X}$ on $\carrier[X]$ iff
        $\binaryCommonFilterRefinement{\carrier[X]}{F}{
            \neighbourhoods{x}{X}}$
        is a proper filter on $\carrier[X]$ by
        \obv{filters_mesh_iff_common_refinement_proper}.
    Follows by \obv{filter_cluster_iff_meshes_neighbourhoods}.
\end{proof}

\begin{proposition}\label{filter_cluster_point_under_coarsening}
    Let $X$ be a topological space.
    Suppose $F'$ refines $F$ on $\carrier[X]$.
    Suppose $\filterClusters{F'}{x}{X}$.
    Then $\filterClusters{F}{x}{X}$.
\end{proposition}
\begin{proof}
    $F\subseteq F'$ by \obv{filter_refines}.
    $x\in\carrier[X]$ by \obv{filter_cluster_point}.
    Show that for every $A,N$ such that $A\in F$ and
        $N\in\neighbourhoods{x}{X}$ we have $A\inter N$ is inhabited.
    \begin{subproof}
        Fix $A,N$.
        Assume $A\in F$ and $N\in\neighbourhoods{x}{X}$.
        Hence $A\in F'$ by \obv{subseteq}.
        Follows by \obv{filter_cluster_point}.
    \end{subproof}
    Follows by \obv{filter_cluster_point}.
\end{proof}

\begin{theorem}\label{compact_iff_every_proper_filter_has_cluster_point}
    Let $X$ be a topological space.
    Then $X$ is compact iff for every $F$ such that
        $F$ is a proper filter on $\carrier[X]$ there exists
        $x\in\carrier[X]$ such that $\filterClusters{F}{x}{X}$.
\end{theorem}
\begin{proof}
    $X$ is compact iff for every $U$ such that
        $U$ is an ultrafilter on $\carrier[X]$ there exists
        $z\in\carrier[X]$ such that $\filterConverges{U}{z}{X}$ by
        \obv{compact_iff_every_ultrafilter_converges}.

    Show that if $X$ is compact, then for every $F$ such that
        $F$ is a proper filter on $\carrier[X]$ there exists
        $x\in\carrier[X]$ such that $\filterClusters{F}{x}{X}$.
    \begin{subproof}
        Assume $X$ is compact.
        Hence for every $U$ such that
            $U$ is an ultrafilter on $\carrier[X]$ there exists
            $z\in\carrier[X]$ such that $\filterConverges{U}{z}{X}$.
        Fix $F$.
        Assume $F$ is a proper filter on $\carrier[X]$.
        Take $U$ such that $F\subseteq U$ and
            $U$ is an ultrafilter on $\carrier[X]$ by
            \obv{filter_extension_to_ultrafilter}.
        Take $x\in\carrier[X]$ such that
            $\filterConverges{U}{x}{X}$.
        $U$ is a proper filter on $\carrier[X]$ by
            \auto{ultrafilter,maximalfilter}.
        Hence $U$ is a filter on $\carrier[X]$ by
            \obv{proper_filter_is_filter}.
        $\neighbourhoods{x}{X}\subseteq U$ by
            \obv{filter_converges}.
        Show that $\filterClusters{F}{x}{X}$.
        \begin{subproof}
            Show that for every $A,N$ such that $A\in F$ and
                $N\in\neighbourhoods{x}{X}$ we have
                $A\inter N$ is inhabited.
            \begin{subproof}
                Fix $A,N$.
                Assume $A\in F$ and $N\in\neighbourhoods{x}{X}$.
                Hence $A,N\in U$ by \obv{subseteq}.
                Hence $A\inter N\in U$ by \obv{filter}.
                $\emptyset\notin U$ by
                    \obv{proper_filter_excludes_emptyset}.
                Hence $A\inter N\neq\emptyset$.
                Follows by \obv{nonempty_inhabited}.
            \end{subproof}
            Follows by \obv{filter_cluster_point}.
        \end{subproof}
    \end{subproof}

    Show that if for every $F$ such that
        $F$ is a proper filter on $\carrier[X]$ there exists
        $x\in\carrier[X]$ such that $\filterClusters{F}{x}{X}$, then
        $X$ is compact.
    \begin{subproof}
        Assume for every $F$ such that
            $F$ is a proper filter on $\carrier[X]$ there exists
            $x\in\carrier[X]$ such that $\filterClusters{F}{x}{X}$.
        Show that for every $U$ such that
            $U$ is an ultrafilter on $\carrier[X]$ there exists
            $z\in\carrier[X]$ such that $\filterConverges{U}{z}{X}$.
        \begin{subproof}
            Fix $U$.
            Assume $U$ is an ultrafilter on $\carrier[X]$.
            Hence $U$ is a proper filter on $\carrier[X]$ by
                \auto{ultrafilter,maximalfilter}.
            Take $x\in\carrier[X]$ such that
                $\filterClusters{U}{x}{X}$.
            Hence $\filterConverges{U}{x}{X}$ by
                \obv{ultrafilter_cluster_iff_converges}.
        \end{subproof}
    \end{subproof}
    Follows by \auto.
\end{proof}
