\section{Path connectedness of finite powers}

\import{topology/power-split.tex}
\import{topology/path-product.tex}

Finite-set induction adds one coordinate at a time through the split/join
homeomorphism. The empty power is a singleton, even when the original space
is empty.

\begin{theorem}\label{empty_topological_power_is_path_connected}
    Let $X$ be a topological space.
    Then $\TopologicalPower{X}{\emptyset}$ is path-connected.
\end{theorem}
\begin{proof}
    $\TopologicalPower{X}{\emptyset}$ is a topological space by
        \obv{topological_power_is_topological_space}.
    $\carrier[\TopologicalPower{X}{\emptyset}]=\{\emptyset\}$ by
        \auto{topological_power_carrier,empty_domain_function_space}.
    Follows by \obv{singleton_space_is_path_connected}.
\end{proof}

\begin{theorem}\label{singleton_topological_power_is_path_connected}
    Let $X$ be a path-connected topological space.
    Then $\TopologicalPower{X}{\{i\}}$ is path-connected.
\end{theorem}
\begin{proof}
    $\TopologicalPower{X}{\{i\}}$ is a topological space by
        \obv{topological_power_is_topological_space}.
    $\functionEvaluation{\{i\}}{\carrier[X]}{i}$ is a homeomorphism from
        $\TopologicalPower{X}{\{i\}}$ to $X$ by \obv{singleton_power_evaluation_homeomorphism}.
    Hence $\converse{\functionEvaluation{\{i\}}{\carrier[X]}{i}}$ is a homeomorphism
        from $X$ to $\TopologicalPower{X}{\{i\}}$ by \obv{homeomorphism_inverse}.
    Follows by \obv{homeomorphism_preserves_path_connectedness}.
\end{proof}

\begin{theorem}\label{finite_topological_power_is_path_connected}
    Let $X$ be a path-connected topological space.
    Suppose $I$ is finite.
    Then $\TopologicalPower{X}{I}$ is path-connected.
\end{theorem}
\begin{proof}
    Let $C=\{J\in\pow{I}\mid\text{$\TopologicalPower{X}{J}$ is path-connected}\}$.
    $\TopologicalPower{X}{\emptyset}$ is path-connected by
        \obv{empty_topological_power_is_path_connected}.
    Hence $\emptyset\in C$ by \obv{powerset_bottom}.
    Show that for all $J,i$ such that $J\in C$ and $i\in I$ we have $\cons{i}{J}\in C$.
    \begin{subproof}
        Fix $J,i$.
        Assume $J\in C$ and $i\in I$.
        $J\subseteq I$ by \obv{pow_iff}.
        Hence $\cons{i}{J}\in\pow{I}$ by \auto{cons_subseteq_intro,powerset_intro}.
        \begin{byCase}
            \caseOf{$i\in J$.}
                $\cons{i}{J}=J$ by \obv{cons_absorb}.
            \caseOf{$i\notin J$.}
                $J$ is disjoint from $\{i\}$ by \auto{disjoint,singleton_iff}.
                $J\union\{i\}=\cons{i}{J}$ by \auto{union_eq_cons,union_comm}.
                $\TopologicalPower{X}{J}$ is a path-connected topological space by
                    \auto{topological_power_is_topological_space}.
                $\TopologicalPower{X}{\{i\}}$ is a path-connected topological space by
                    \auto{topological_power_is_topological_space,singleton_topological_power_is_path_connected}.
                Hence $\BinaryProductTopology{\TopologicalPower{X}{J}}{\TopologicalPower{X}{\{i\}}}$
                    is path-connected by \obv{binary_product_of_path_connected_spaces}.
                $\functionJoin{J}{\{i\}}{\carrier[X]}$ is a homeomorphism from
                    $\BinaryProductTopology{\TopologicalPower{X}{J}}{\TopologicalPower{X}{\{i\}}}$
                    to $\TopologicalPower{X}{J\union\{i\}}$ by \obv{topological_power_join_homeomorphism}.
                $\TopologicalPower{X}{J\union\{i\}}$ is path-connected by
                    \auto{homeomorphism_preserves_path_connectedness,
                        topological_power_is_topological_space,binary_product_topology_is_topological_space}.
        \end{byCase}
    \end{subproof}
    $I\in C$ by \obv{finite_induction}.
\end{proof}
