\section{Second countability under open continuous maps}

\import{topology/countability-axioms.tex}
\import{topology/basis-continuity.tex}

\begin{theorem}\label{open_continuous_surjection_preserves_second_countability}
    Let $X,Y$ be topological spaces.
    Suppose $X$ is second-countable.
    Suppose $f$ is continuous from $X$ to $Y$ and $f$ is an open map from $X$ to $Y$.
    Suppose $f\in\surj{\carrier[X]}{\carrier[Y]}$.
    Then $Y$ is second-countable.
\end{theorem}
\begin{proof}
    Take $B$ such that $B$ is countable and $B$ is an open basis of $X$
        by \obv{is_second_countable}.
    Let $C=\{\img{f}{U}\mid U\in B\}$.
    $\cardLe{C}{\naturals}$ by \auto{countable_replacement}.
    Hence $C$ is countable by \obv{cardinal_le,at_most_countable}.
    $C$ is an open basis of $Y$ by \auto{open_continuous_surjection_maps_basis}.
    Follows by \obv{is_second_countable}.
\end{proof}

\begin{corollary}\label{homeomorphism_preserves_second_countability}
    Let $X,Y$ be topological spaces.
    Suppose $X$ is second-countable and $f$ is a homeomorphism from $X$ to $Y$.
    Then $Y$ is second-countable.
\end{corollary}
\begin{proof}
    $f$ is continuous from $X$ to $Y$ by \obv{homeomorphism}.
    $f$ is an open map from $X$ to $Y$ by \obv{homeomorphism_is_open_map}.
    $f\in\surj{\carrier[X]}{\carrier[Y]}$ by \auto{homeomorphism,bijections}.
    Follows by \obv{open_continuous_surjection_preserves_second_countability}.
\end{proof}
