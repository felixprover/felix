\section{Local connectedness}

\import{topology/connectedness.tex}
\import{topology/local-path.tex}
\import{topology/path-connectedness.tex}

\begin{definition}\label{locally_connected}
    Let $X$ be a topological space.
    $X$ is locally connected iff for all $U,x$ such that
        $U\in\opens[X]$ and $x\in U$ there exists $V\in\opens[X]$ such that
        $x\in V\subseteq U$ and $\subspaceTopology{X}{V}$ is connected.
\end{definition}

\begin{theorem}\label{locally_path_connected_implies_locally_connected}
    Let $X$ be a locally path-connected topological space.
    Then $X$ is locally connected.
\end{theorem}
\begin{proof}
    Show that for all $U,x$ such that $U\in\opens[X]$ and $x\in U$ there exists
        $V\in\opens[X]$ such that $x\in V\subseteq U$ and
        $\subspaceTopology{X}{V}$ is connected.
    \begin{subproof}
        Fix $U,x$.
        Assume $U\in\opens[X]$ and $x\in U$.
        Take $V\in\opens[X]$ such that $x\in V\subseteq U$ and
            $\subspaceTopology{X}{V}$ is path-connected by \obv{locally_path_connected}.
        Hence $\subspaceTopology{X}{V}$ is connected by \obv{path_connected_implies_connected}.
    \end{subproof}
    Follows by \obv{locally_connected}.
\end{proof}
