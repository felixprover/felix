The constant zero map is admissible although its image is not open.
We use the identity presentation at the source and the reflection presentation
at the target; both generate the same full coordinate structure.

\begin{theorem}\label{real_zero_coordinate_map}
    Then $\realZeroMap{\reals}$ is coordinate-admissible
        from $\Equip{\RealTopology}{\GeneratedCoordinates{\FullPseudogroup{\RealTopology}}{\RealTopology}{\RealOneChart}}$
        under $\FullPseudogroup{\RealTopology}$
        to $\Equip{\RealTopology}{\GeneratedCoordinates{\FullPseudogroup{\RealTopology}}{\RealTopology}{\RealReflectionChart}}$
        under $\FullPseudogroup{\RealTopology}$ in $\ContinuousCalculus{\{\RealTopology\}}$.
\end{theorem}
\begin{proof}
    Let $X=\RealTopology$ and $G=\FullPseudogroup{\RealTopology}$.
    $X$ is a topological space by \obv{real_topology_is_topological_space}.
    $G$ is a coordinate pseudogroup by \auto{full_pseudogroup}.
    $\model[G]=X$ by \auto{full_pseudogroup_model}.
    $\{X\}$ is a topological model family by \obv{singleton_topological_model_family}.
    $\model[G]\in\{X\}$ by \obv{singleton_iff}.
    $\RealOneChart$ is a coordinate atlas on $X$ under $G$ by \obv{real_identity_full_atlas}.
    $\RealReflectionChart$ is a coordinate atlas on $X$ under $G$ by \obv{real_reflection_full_atlas}.
    Let $S=\GeneratedCoordinates{G}{X}{\RealOneChart}$.
    $S$ is a relative coordinate structure on $X$ under $G$ by \auto{generated_coordinates}.
    $S$ is a coordinate datum by \obv{relative_coordinate_structure}.
    Let $M=\Equip{X}{S}$.
    Let $N=\Equip{X}{\GeneratedCoordinates{G}{X}{\RealReflectionChart}}$.
    $M=N$ by \auto{real_full_coordinate_presentations_equal}.
    $M$ is an equipped coordinate space by \auto{equip}.
    $M$ has coordinates under $G$ by \auto{equip_preserves_coordinates}.
    $M$ is a topological space by \obv{equipped_coordinate_space}.
    $\carrier[M]=\carrier[X]$ and $\opens[M]=\opens[X]$ by
        \auto{equipprojection_carrier,equipprojection_opens}.
    $\carrier[X]=\reals$ by \obv{real_topology_carrier}.
    $\surrealZero\in\reals$ by \obv{surreal_zero_is_real}.
    $\carrier[M]\times\{\surrealZero\}$ is continuous from $M$ to $X$ by
        \auto{real_constant_graph_continuous}.
    $\realZeroMap{\reals}$ is continuous from $M$ to $M$ by
        \auto{real_zero_map,continuous_map}.
    $\realZeroMap{\reals}$ is a continuous partial map from $M$ to $N$ by
        \obv{continuous_map_is_partial}.
    Follows by \auto{continuous_calculus_coordinate_maps}.
\end{proof}

\begin{theorem}\label{real_zero_map_image}
    Then $\ran{\realZeroMap{\reals}}=\{\surrealZero\}$.
\end{theorem}
\begin{proof}
    $\surrealZero\in\reals$ by \obv{surreal_zero_is_real}.
    $\reals$ is inhabited.
    Follows by \auto{real_zero_map,ran_times_inhabited}.
\end{proof}

\begin{theorem}\label{real_zero_map_image_not_open}
    Then $\ran{\realZeroMap{\reals}}\notin\opens[\RealTopology]$.
\end{theorem}
\begin{proof}
    Suppose $\ran{\realZeroMap{\reals}}\in\opens[\RealTopology]$.
    $\{\surrealZero\}\in\opens[\RealTopology]$ by \obv{real_zero_map_image}.
    $\opens[\RealTopology]=\metricOpens{\RealMetric}$ by
        \auto{real_topology,metric_topology_opens}.
    $\metricOpen{\RealMetric}{\{\surrealZero\}}$ by \auto{metric_opens_iff}.
    $\surrealZero\in\{\surrealZero\}$ by \auto{singleton_iff}.
    Take $r\in\reals$ such that $\surrealLt{\surrealZero}{r}$ and
        $\metricBall{\RealMetric}{\surrealZero}{r}\subseteq\{\surrealZero\}$ by \obv{metric_open}.
    $\surrealZero,\surrealOne\in\reals$ by \obv{surreal_zero_is_real,surreal_one_is_real}.
    $\surrealLt{\surrealZero}{\surrealOne}$ by \obv{surreal_zero_lt_one}.
    Take $t\in\reals$ such that $\surrealLt{\surrealZero}{t}$ and
        $\surrealLt{t}{\surrealOne}$ and $t\in\metricBall{\RealMetric}{\surrealZero}{r}$ by
        \obv{real_ball_meets_right_open_interval}.
    $t=\surrealZero$ by \auto{subseteq,singleton_iff}.
    $\surreal{t}$ by \obv{surreal_real_is_surreal}.
    $\surrealLt{t}{t}$ by \auto.
    Contradiction by \obv{surreal_lt_irreflexive}.
\end{proof}

The empty graph has empty exact domain in the same equipped real line.
Its endpoint and calculus requirements remain in force.

\begin{theorem}\label{real_empty_coordinate_map}
    Then $\emptyset$ is coordinate-admissible
        from $\Equip{\RealTopology}{\GeneratedCoordinates{\FullPseudogroup{\RealTopology}}{\RealTopology}{\RealOneChart}}$
        under $\FullPseudogroup{\RealTopology}$
        to $\Equip{\RealTopology}{\GeneratedCoordinates{\FullPseudogroup{\RealTopology}}{\RealTopology}{\RealReflectionChart}}$
        under $\FullPseudogroup{\RealTopology}$ in $\ContinuousCalculus{\{\RealTopology\}}$
        and $\dom{\emptyset}=\emptyset$.
\end{theorem}
\begin{proof}
    Let $X=\RealTopology$ and $G=\FullPseudogroup{\RealTopology}$.
    $X$ is a topological space by \obv{real_topology_is_topological_space}.
    $G$ is a coordinate pseudogroup by \auto{full_pseudogroup}.
    $\model[G]=X$ by \auto{full_pseudogroup_model}.
    $\{X\}$ is a topological model family by \obv{singleton_topological_model_family}.
    $\model[G]\in\{X\}$ by \obv{singleton_iff}.
    $\ContinuousCalculus{\{X\}}$ is a local-map calculus by \obv{continuous_calculus}.
    $G$ is compatible with the calculus $\ContinuousCalculus{\{X\}}$ by
        \obv{pseudogroup_continuous_calculus_compatible}.
    $\RealOneChart$ is a coordinate atlas on $X$ under $G$ by \obv{real_identity_full_atlas}.
    Let $S=\GeneratedCoordinates{G}{X}{\RealOneChart}$.
    $S$ is a relative coordinate structure on $X$ under $G$ by \auto{generated_coordinates}.
    $S$ is a coordinate datum by \obv{relative_coordinate_structure}.
    Let $M=\Equip{X}{S}$.
    Let $N=\Equip{X}{\GeneratedCoordinates{G}{X}{\RealReflectionChart}}$.
    $M=N$ by \auto{real_full_coordinate_presentations_equal}.
    $M$ is an equipped coordinate space by \auto{equip}.
    $M$ has coordinates under $G$ by \auto{equip_preserves_coordinates}.
    $N$ is an equipped coordinate space.
    $N$ has coordinates under $G$.
    $\emptyset$ is coordinate-admissible from $M$ under $G$ to $N$ under $G$
        in $\ContinuousCalculus{\{X\}}$ by \obv{empty_coordinate_admissible_map}.
    $\dom{\emptyset}=\emptyset$ by \obv{dom_emptyset}.
\end{proof}
