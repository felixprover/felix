\section{The additive uniformity of a topological ring}

The additive uniformity is the canonical left uniformity of the exact
additive topological group.  Thus all uniform-space facts remain owned by the
generic topological-group construction.

\begin{definition}\label{additive_uniform_space}
    $\AdditiveUniformSpace{R}=
        \leftUniformSpace{\AdditiveTopologicalGroup{R}}$.
\end{definition}

\begin{theorem}\label{additive_uniform_space_is_uniform_space}
    Let $R$ be a topological ring.
    Then $\AdditiveUniformSpace{R}$ is a uniform space.
\end{theorem}
\begin{proof}
    $\AdditiveTopologicalGroup{R}$ is a topological group by
        \obv{additive_topological_group_is_exact}.
    Follows by
        \auto{additive_uniform_space,left_uniform_space_is_uniform_space}.
\end{proof}

\begin{proposition}\label{additive_uniform_space_carrier}
    Let $R$ be a topological ring.
    Then $\carrier[\AdditiveUniformSpace{R}]=\carrier[R]$.
\end{proposition}
\begin{proof}
    $\carrier[\AdditiveUniformSpace{R}]
        =\carrier[\AdditiveTopologicalGroup{R}]$ by
        \auto{additive_uniform_space,left_uniform_space_carrier}.
    Follows by \auto{additive_topological_group_fields}.
\end{proof}

\begin{proposition}\label{additive_uniform_space_uniformity}
    Let $R$ be a topological ring.
    Then $\uniformity[\AdditiveUniformSpace{R}]
        =\generatedFilter{
            \carrier[R]\times\carrier[R]}{
            \leftUniformityBasis{\AdditiveTopologicalGroup{R}}}$.
\end{proposition}
\begin{proof}
    $\carrier[\AdditiveTopologicalGroup{R}]=\carrier[R]$ by
        \obv{additive_topological_group_fields}.
    Follows by
        \auto{additive_uniform_space,left_uniform_space_uniformity}.
\end{proof}

\begin{proposition}\label{additive_uniform_topology_carrier}
    Let $R$ be a topological ring.
    Then $\carrier[\uniformTopology{\AdditiveUniformSpace{R}}]
        =\carrier[R]$.
\end{proposition}
\begin{proof}
    $\carrier[\uniformTopology{\AdditiveUniformSpace{R}}]
        =\carrier[\AdditiveUniformSpace{R}]$ by
        \obv{uniform_topology_carrier}.
    Follows by \obv{additive_uniform_space_carrier}.
\end{proof}

\begin{theorem}\label{additive_uniform_topology_opens}
    Let $R$ be a topological ring.
    Then $\opens[\uniformTopology{\AdditiveUniformSpace{R}}]
        =\opens[R]$.
\end{theorem}
\begin{proof}
    Let $G=\AdditiveTopologicalGroup{R}$.
    $\opens[\uniformTopology{\leftUniformSpace{G}}]=\opens[G]$ by
        \obv{left_uniform_topology_opens}.
    $\opens[G]=\opens[R]$ by
        \auto{additive_topological_group_fields}.
    Follows by \auto{additive_uniform_space}.
\end{proof}

\begin{theorem}\label{additive_topological_group_hausdorff_iff_ring_hausdorff}
    Let $R$ be a topological ring.
    Then $\AdditiveTopologicalGroup{R}$ is Hausdorff iff $R$ is Hausdorff.
\end{theorem}
\begin{proof}
    $R$ is a topological space by
        \obv{topological_ring_is_topological_space}.
    Hence $\AdditiveTopologicalGroup{R}$ is a topological space by
        \obv{topological_group_is_topological_space}.
    $\carrier[\AdditiveTopologicalGroup{R}]=\carrier[R]$ and
        $\opens[\AdditiveTopologicalGroup{R}]=\opens[R]$ by
        \obv{additive_topological_group_fields}.
    Follows by \auto{is_hausdorff}.
\end{proof}

\begin{theorem}\label{additive_uniform_space_separated_iff_hausdorff}
    Let $R$ be a topological ring.
    Then $\AdditiveUniformSpace{R}$ is separated uniform iff
        $R$ is Hausdorff.
\end{theorem}
\begin{proof}
    Let $G=\AdditiveTopologicalGroup{R}$.
    $\leftUniformSpace{G}$ is separated uniform iff $G$ is Hausdorff by
        \obv{left_uniform_space_separated_iff_hausdorff}.
    $G$ is Hausdorff iff $R$ is Hausdorff by
        \obv{additive_topological_group_hausdorff_iff_ring_hausdorff}.
    Follows by \auto{additive_uniform_space}.
\end{proof}

Completeness includes Hausdorffness explicitly because the generic
complete-uniform predicate does not include separatedness.

\begin{definition}\label{additively_complete_topological_ring}
    Let $R$ be a topological ring.
    $R$ is additively complete iff
        $\AdditiveUniformSpace{R}$ is complete uniform and
        $R$ is Hausdorff.
\end{definition}

\begin{theorem}\label{additively_complete_iff_complete_separated}
    Let $R$ be a topological ring.
    Then $R$ is additively complete iff
        $\AdditiveUniformSpace{R}$ is complete uniform and
        $\AdditiveUniformSpace{R}$ is separated uniform.
\end{theorem}
\begin{proof}
    $\AdditiveUniformSpace{R}$ is separated uniform iff
        $R$ is Hausdorff by
        \obv{additive_uniform_space_separated_iff_hausdorff}.
    Follows by \auto{additively_complete_topological_ring}.
\end{proof}

\begin{theorem}\label{additive_uniform_complete_iff_filter_basis_convergence}
    Let $R$ be a topological ring.
    Then $\AdditiveUniformSpace{R}$ is complete uniform iff
        for every $B$ such that
            $B$ is a proper filter basis on $\carrier[R]$, if
            for every $U\in\uniformity[\AdditiveUniformSpace{R}]$ there
                exists $A\in B$ such that $A\times A\subseteq U$, then
            there exists $x\in\carrier[R]$ such that for every
                $U\in\uniformity[\AdditiveUniformSpace{R}]$ there exists
                $A\in B$ such that
                    $A\subseteq\uniformBall{\AdditiveUniformSpace{R}}{U}{x}$.
\end{theorem}
\begin{proof}
    $\AdditiveUniformSpace{R}$ is a uniform space by
        \obv{additive_uniform_space_is_uniform_space}.
    $\carrier[\AdditiveUniformSpace{R}]=\carrier[R]$ by
        \obv{additive_uniform_space_carrier}.
    Follows by \auto{complete_uniform_iff_filter_basis_convergence}.
\end{proof}

\begin{theorem}\label{additively_complete_iff_filter_basis_convergence}
    Let $R$ be a topological ring.
    Then $R$ is additively complete iff
        $R$ is Hausdorff and for every $B$ such that
            $B$ is a proper filter basis on $\carrier[R]$, if
            for every $U\in\uniformity[\AdditiveUniformSpace{R}]$ there
                exists $A\in B$ such that $A\times A\subseteq U$, then
            there exists $x\in\carrier[R]$ such that for every
                $U\in\uniformity[\AdditiveUniformSpace{R}]$ there exists
                $A\in B$ such that
                    $A\subseteq\uniformBall{\AdditiveUniformSpace{R}}{U}{x}$.
\end{theorem}
\begin{proof}
    $R$ is additively complete iff
        $\AdditiveUniformSpace{R}$ is complete uniform and
        $R$ is Hausdorff by \auto{additively_complete_topological_ring}.
    $\AdditiveUniformSpace{R}$ is complete uniform iff
        for every $B$ such that
            $B$ is a proper filter basis on $\carrier[R]$, if
            for every $U\in\uniformity[\AdditiveUniformSpace{R}]$ there
                exists $A\in B$ such that $A\times A\subseteq U$, then
            there exists $x\in\carrier[R]$ such that for every
                $U\in\uniformity[\AdditiveUniformSpace{R}]$ there exists
                $A\in B$ such that
                    $A\subseteq\uniformBall{\AdditiveUniformSpace{R}}{U}{x}$
        by \obv{additive_uniform_complete_iff_filter_basis_convergence}.
    Follows by \auto.
\end{proof}
