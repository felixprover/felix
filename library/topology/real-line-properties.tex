\section{Separation and local compactness of the real line}

\import{topology/real-compactness.tex}
\import{topology/metric-uniform-topology.tex}

The established metric gives Hausdorff separation. Every real point has a
unit ball contained in a compact closed interval, which gives local
compactness without choosing any new topology.

\begin{theorem}\label{real_topology_is_hausdorff}
    $\RealTopology$ is Hausdorff.
\end{theorem}
\begin{proof}
    $\RealMetric$ is a metric space by \obv{surreal_real_metric_space}.
    Follows by \auto{real_topology,metric_topology_is_hausdorff}.
\end{proof}

\begin{theorem}\label{real_topology_is_locally_compact}
    $\RealTopology$ is locally compact.
\end{theorem}
\begin{proof}
    $\RealTopology$ is a topological space by \obv{real_topology_is_topological_space}.
    Show that for every $x\in\carrier[\RealTopology]$ there exist $U,K$ such that
        $U\in\opens[\RealTopology]$ and $x\in U\subseteq K$ and $K$ is compact in $\RealTopology$.
    \begin{subproof}
        Fix $x\in\carrier[\RealTopology]$.
        $x\in\reals$ by \obv{real_topology_carrier}.
        Let $U=\metricBall{\RealMetric}{x}{\surrealOne}$.
        Let $a=\surrealAdd{x}{\surrealNeg{\surrealOne}}$ and $b=\surrealAdd{x}{\surrealOne}$.
        Let $K=\realClosedInterval{a}{b}$.
        $\surrealOne\in\reals$ and $\surrealNeg{\surrealOne}\in\reals$ by
            \obv{surreal_one_is_real,surreal_real_negation_closed}.
        Hence $a,b\in\reals$ by \auto{surreal_real_addition_closed}.
        $U\in\opens[\RealTopology]$ and $x\in U$ by \auto{real_unit_ball_is_open_neighbourhood}.
        $U=\orderOpenInterval{\RealField}{a}{b}$ by \auto{real_metric_ball_eq_order_open_interval}.
        $K=\orderClosedInterval{\RealField}{a}{b}$ by \auto{real_order_closed_interval_eq}.
        Hence $U\subseteq K$ by
            \auto{subseteq,order_open_interval_iff,order_closed_interval_iff,ordered_set_strict_relation_iff}.
        Follows by \auto{real_closed_interval_is_compact}.
    \end{subproof}
    Follows by \obv{is_locally_compact}.
\end{proof}
