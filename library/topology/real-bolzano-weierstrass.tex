\section{The Bolzano--Weierstrass theorem}

\import{surreal/real-boundedness.tex}
\import{topology/compact-metric-sequences.tex}
\import{topology/real-compactness.tex}

A bounded real sequence lies in a compact symmetric interval.  Sequential
compactness of that interval then supplies the familiar convergent
subsequence, while keeping the extraction argument entirely in the generic
metric layer.

\begin{theorem}[Bolzano--Weierstrass]\label{real_bolzano_weierstrass}
    Let $s\in\funs{\naturals}{\reals}$.
    Suppose $\ran{s}$ is real bounded.
    Then there exist $p,t$ such that $p\in\reals$ and
        $t$ is a subsequence of $s$ in $\reals$ and
        $\metricConverges{\RealMetric}{t}{p}$.
\end{theorem}
\begin{proof}
    Take $r\in\reals$ such that $\surrealLe{\surrealZero}{r}$ and
        for every $x\in\ran{s}$ we have $\surrealLe{\abs{x}}{r}$ by
        \obv{real_bounded}.
    Let $K=\realClosedSymmetricInterval{r}$.
    Show that $\ran{s}\subseteq K$.
    \begin{subproof}
        Show that every element of $\ran{s}$ is an element of $K$.
        \begin{subproof}
            Fix $x\in\ran{s}$.
            $\ran{s}\subseteq\reals$ by \obv{real_bounded}.
            Hence $x\in\reals$ by \obv{subseteq}.
            $\surrealLe{\abs{x}}{r}$.
            Hence $x\in\realClosedSymmetricInterval{r}$ by
                \auto{real_closed_symmetric_interval_iff_abs_bound}.
        \end{subproof}
    \end{subproof}

    $\surrealNeg{r}\in\reals$ by
        \obv{surreal_real_negation_closed}.
    $\RealTopology$ is a topological space by
        \obv{real_topology_is_topological_space}.
    $\realClosedInterval{\surrealNeg{r}}{r}$ is compact in
        $\RealTopology$ by \obv{real_closed_interval_is_compact}.
    $\realClosedSymmetricInterval{r}
        =\realClosedInterval{\surrealNeg{r}}{r}$ by
        \obv{real_closed_symmetric_interval}.
    Hence $K$ is compact in $\RealTopology$.

    $s$ is a function by \obv{funs_is_function}.
    Hence $s\in\funs{\dom{s}}{\ran{s}}$ by
        \obv{function_exact_type_to_range}.
    Hence $s\in\funs{\dom{s}}{K}$ by
        \obv{funs_weaken_codom}.
    $\dom{s}=\naturals$ by \obv{funs}.
    Hence $s\in\funs{\naturals}{K}$.

    $\RealMetric$ is a metric space by
        \obv{surreal_real_metric_space}.
    $\RealTopology=\metricTopology{\RealMetric}$ by
        \obv{real_topology}.
    Take $p,t$ such that $p\in K$ and
        $t$ is a subsequence of $s$ in $K$ and
        $\metricConverges{\RealMetric}{t}{p}$ by
        \obv{compact_metric_subspace_is_sequentially_compact}.
    $K\subseteq\reals$ by
        \auto{real_closed_symmetric_interval,
            real_closed_interval_subset_reals}.
    Hence $p\in\reals$ by \obv{subseteq}.
    Take $h$ such that $h$ is a natural subsequence selector and
        $t=s\circ h$ by \obv{subsequence_selector_exists}.
    $s\circ h$ is a subsequence of $s$ in $\reals$ by
        \obv{subsequence_from_selector}.
    Hence $t$ is a subsequence of $s$ in $\reals$.
    Follows by \auto.
\end{proof}
