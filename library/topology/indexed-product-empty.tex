\section{The empty indexed product}

\import{topology/indexed-product.tex}
\import{topology/discrete.tex}

The dependent product over no coordinates contains exactly the empty
function.  Since there are no coordinate maps, its initial topology is
indiscrete; on this singleton carrier it is also discrete.

\begin{proposition}\label{empty_indexed_product_carrier}
    $\carrier[\IndexedProductTopology{\emptyset}{X}]
        =\{\emptyset\}$.
\end{proposition}
\begin{proof}
    $\carrier[\IndexedProductTopology{\emptyset}{X}]
        =\dependentProduct{\emptyset}{
            \indexedTopologicalCarrierFamily{\emptyset}{X}}$ by
        \obv{indexed_product_topology_carrier}.
    $\dependentProduct{\emptyset}{
        \indexedTopologicalCarrierFamily{\emptyset}{X}}
        =\{\emptyset\}$ by \obv{empty_dependent_product}.
\end{proof}

\begin{theorem}\label{empty_indexed_product_is_indiscrete}
    $\IndexedProductTopology{\emptyset}{X}
        =\indiscreteSpace{\{\emptyset\}}$.
\end{theorem}
\begin{proof}
    $\IndexedProductTopology{\emptyset}{X}
        =\InitialTopology{
            \dependentProduct{\emptyset}{
                \indexedTopologicalCarrierFamily{\emptyset}{X}}}{
            \emptyset}{X}{
            \indexedProductProjectionFamily{\emptyset}{X}}$ by
        \obv{indexed_product_topology}.
    $\InitialTopology{
        \dependentProduct{\emptyset}{
            \indexedTopologicalCarrierFamily{\emptyset}{X}}}{
        \emptyset}{X}{
        \indexedProductProjectionFamily{\emptyset}{X}}
        =\indiscreteSpace{
            \dependentProduct{\emptyset}{
                \indexedTopologicalCarrierFamily{\emptyset}{X}}}$ by
        \obv{empty_initial_topology_is_indiscrete}.
    $\dependentProduct{\emptyset}{
        \indexedTopologicalCarrierFamily{\emptyset}{X}}
        =\{\emptyset\}$ by \obv{empty_dependent_product}.
    Hence $\indiscreteSpace{
        \dependentProduct{\emptyset}{
            \indexedTopologicalCarrierFamily{\emptyset}{X}}}
        =\indiscreteSpace{\{\emptyset\}}$.
    Hence $\InitialTopology{
        \dependentProduct{\emptyset}{
            \indexedTopologicalCarrierFamily{\emptyset}{X}}}{
        \emptyset}{X}{
        \indexedProductProjectionFamily{\emptyset}{X}}
        =\indiscreteSpace{\{\emptyset\}}$.
\end{proof}

\begin{theorem}\label{empty_indexed_product_is_discrete}
    $\IndexedProductTopology{\emptyset}{X}
        =\discreteTopology{\{\emptyset\}}$.
\end{theorem}
\begin{proof}
    $\IndexedProductTopology{\emptyset}{X}
        =\indiscreteSpace{\{\emptyset\}}$ by
        \obv{empty_indexed_product_is_indiscrete}.
    $\{\emptyset\}$ is a subsingleton by
        \auto{singleton_iff}.
    Hence $\indiscreteSpace{\{\emptyset\}}
        =\discreteTopology{\{\emptyset\}}$ by
        \obv{subsingleton_indiscrete_space_is_discrete_topology}.
\end{proof}
