\section{Paths in subspaces}

\import{topology/path.tex}
\import{topology/subspace-continuity.tex}

A path whose range lies in a subspace is already a path there: only its
codomain changes, not its graph or its parameter interval.

\begin{theorem}\label{path_codomain_corestriction}
    Let $X$ be a topological space.
    Suppose $p$ is a path from $a$ to $b$ in $X$.
    Suppose $A\subseteq\carrier[X]$ and $\ran{p}\subseteq A$.
    Then $p$ is a path from $a$ to $b$ in $\subspaceTopology{X}{A}$.
\end{theorem}
\begin{proof}
    Let $S=\subspaceTopology{X}{A}$.
    $S$ is a topological space and $\carrier[S]=A$ by
        \auto{subspace_topology_is_topological_space,subspace_topology_carrier_of_subset}.
    $p$ is continuous from $\RealUnitIntervalTopology$ to $X$ by \obv{path_from_to}.
    Hence $p$ is continuous from $\RealUnitIntervalTopology$ to $S$ by
        \obv{continuous_codomain_corestriction}.
    $p\in\funs{\RealUnitInterval}{\carrier[S]}$ by
        \auto{continuous_map,real_unit_interval_topology_carrier}.
    $\surrealZero,\surrealOne\in\RealUnitInterval$ by
        \obv{surreal_zero_in_real_unit_interval,surreal_one_in_real_unit_interval}.
    $p(\surrealZero),p(\surrealOne)\in\carrier[S]$ by \obv{funs_type_apply}.
    Follows by \auto{path_from_to}.
\end{proof}

\begin{theorem}\label{path_subspace_codomain_inclusion}
    Let $X$ be a topological space.
    Suppose $p$ is a path from $a$ to $b$ in $\subspaceTopology{X}{A}$.
    Then $p$ is a path from $a$ to $b$ in $X$.
\end{theorem}
\begin{proof}
    $p$ is continuous from $\RealUnitIntervalTopology$ to $\subspaceTopology{X}{A}$ by
        \obv{path_from_to}.
    Hence $p$ is continuous from $\RealUnitIntervalTopology$ to $X$ by
        \auto{continuous_subspace_codomain_inclusion,real_unit_interval_topology_is_topological_space}.
    $a,b\in\carrier[\subspaceTopology{X}{A}]$ by \obv{path_from_to}.
    Hence $a,b\in\carrier[X]$ by \auto{subspace_topology_carrier,inter}.
    Follows by \obv{path_from_to}.
\end{proof}
