\section{Empty and singleton indexed coproducts}

\import{topology/indexed-coproduct-open.tex}
\import{set/indexed-sum-degenerate.tex}

The empty coproduct has the empty carrier and hence its unique topology.  A
coproduct with one summand retains a tag in its carrier, so its canonical
comparison with that summand is a homeomorphism rather than a literal
equality of topological-space records.

\begin{proposition}\label{empty_indexed_coproduct_carrier}
    $\carrier[\IndexedCoproductTopology{\emptyset}{X}]=\emptyset$.
\end{proposition}
\begin{proof}
    $\carrier[\IndexedCoproductTopology{\emptyset}{X}]
        =\indexedSum{\emptyset}{
            \indexedTopologicalCarrierFamily{\emptyset}{X}}$ by
        \obv{indexed_coproduct_topology_carrier}.
    $\indexedSum{\emptyset}{
        \indexedTopologicalCarrierFamily{\emptyset}{X}}=\emptyset$ by
        \obv{empty_indexed_sum}.
\end{proof}

\begin{theorem}\label{empty_indexed_coproduct_is_discrete}
    $\IndexedCoproductTopology{\emptyset}{X}
        =\discreteTopology{\emptyset}$.
\end{theorem}
\begin{proof}
    $\IndexedCoproductTopology{\emptyset}{X}
        =\FinalTopology{
            \indexedSum{\emptyset}{
                \indexedTopologicalCarrierFamily{\emptyset}{X}}}{
            \emptyset}{X}{
            \indexedCoproductInjectionFamily{\emptyset}{X}}$ by
        \obv{indexed_coproduct_topology}.
    $\indexedSum{\emptyset}{
        \indexedTopologicalCarrierFamily{\emptyset}{X}}=\emptyset$ by
        \obv{empty_indexed_sum}.
    Hence $\FinalTopology{
        \indexedSum{\emptyset}{
            \indexedTopologicalCarrierFamily{\emptyset}{X}}}{
        \emptyset}{X}{
        \indexedCoproductInjectionFamily{\emptyset}{X}}
        =\FinalTopology{\emptyset}{\emptyset}{X}{
            \indexedCoproductInjectionFamily{\emptyset}{X}}$.
    $\FinalTopology{\emptyset}{\emptyset}{X}{
        \indexedCoproductInjectionFamily{\emptyset}{X}}
        =\discreteTopology{\emptyset}$ by
        \obv{empty_final_topology_is_discrete}.
\end{proof}

\begin{theorem}\label{singleton_indexed_coproduct_injection_is_homeomorphism}
    Suppose $X$ is a function on $\{i\}$.
    Suppose $X(i)$ is a topological space.
    Then $\indexedSumInjection{\{i\}}{
        \indexedTopologicalCarrierFamily{\{i\}}{X}}{i}$
        is a homeomorphism from $X(i)$ to
        $\IndexedCoproductTopology{\{i\}}{X}$.
\end{theorem}
\begin{proof}
    Show that for every $j\in\{i\}$ we have $X(j)$ is a topological
        space.
    \begin{subproof}
        Fix $j\in\{i\}$.
        Hence $j=i$ by \obv{singleton_iff}.
    \end{subproof}
    Hence $\indexedSumInjection{\{i\}}{
        \indexedTopologicalCarrierFamily{\{i\}}{X}}{i}$
        is continuous from $X(i)$ to
            $\IndexedCoproductTopology{\{i\}}{X}$ and
        $\indexedSumInjection{\{i\}}{
            \indexedTopologicalCarrierFamily{\{i\}}{X}}{i}$
        is an open map from $X(i)$ to
            $\IndexedCoproductTopology{\{i\}}{X}$ by
        \auto{indexed_coproduct_injection_is_continuous_open_injection}.
    $\indexedTopologicalCarrierFamily{\{i\}}{X}$ is a function on
        $\{i\}$ by \obv{indexed_topological_carrier_family_type}.
    Hence $\indexedSumInjection{\{i\}}{
        \indexedTopologicalCarrierFamily{\{i\}}{X}}{i}$
        is a bijection from
            $(\indexedTopologicalCarrierFamily{\{i\}}{X})(i)$ to
            $\indexedSum{\{i\}}{
                \indexedTopologicalCarrierFamily{\{i\}}{X}}$ by
        \obv{singleton_indexed_sum_injection_is_bijection}.
    $i\in\{i\}$ by \auto{singleton_iff}.
    $(\indexedTopologicalCarrierFamily{\{i\}}{X})(i)
        =\carrier[X(i)]$ by
        \obv{indexed_topological_carrier_family_apply}.
    $\carrier[\IndexedCoproductTopology{\{i\}}{X}]
        =\indexedSum{\{i\}}{
            \indexedTopologicalCarrierFamily{\{i\}}{X}}$ by
        \obv{indexed_coproduct_topology_carrier}.
    Hence $\indexedSumInjection{\{i\}}{
        \indexedTopologicalCarrierFamily{\{i\}}{X}}{i}$
        is a bijection from $\carrier[X(i)]$ to
            $\carrier[\IndexedCoproductTopology{\{i\}}{X}]$.
    Follows by
        \obv{continuous_bijection_open_map_is_homeomorphism}.
\end{proof}
