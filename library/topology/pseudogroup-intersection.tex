\section{Intersections of coordinate pseudogroups}

\import{topology/full-pseudogroup.tex}

All pseudogroups in an intersection have the same displayed model.
The ambient set of open partial homeomorphisms bounds the construction,
including the intersection of an empty family.

\begin{definition}\label{pseudogroup_family}
    Let $E$ be a topological space.
    $P$ is a pseudogroup family on $E$ iff
        for every $G\in P$ we have $G$ is a pseudogroup on $E$.
\end{definition}

\begin{classification}\label{pseudogroup_family_member_on}
    Let $E$ be a topological space.
    Let $P$ be a pseudogroup family on $E$.
    Let $G\in P$.
    Then $G$ is a pseudogroup on $E$.
\end{classification}
\begin{proof}
    Follows by \obv{pseudogroup_family}.
\end{proof}

\begin{classification}\label{pseudogroup_family_member}
    Let $E$ be a topological space.
    Let $P$ be a pseudogroup family on $E$.
    Let $G\in P$.
    Then $G$ is a coordinate pseudogroup.
\end{classification}
\begin{proof}
    $G$ is a pseudogroup on $E$ by \obv{pseudogroup_family_member_on}.
    Follows by \obv{pseudogroup_on}.
\end{proof}

\begin{definition}\label{common_coordinate_changes}
    $\commonCoordinateChanges{E}{P}=\{h\in\openPartialHomeomorphisms{E}{E}\mid
        \text{for every $G\in P$ we have $h\in\changes[G]$}\}$.
\end{definition}

\begin{theorem}\label{common_coordinate_changes_iff}
    Then $h\in\commonCoordinateChanges{E}{P}$ iff
        $h$ is an open partial homeomorphism from $E$ to $E$ and
        for every $G\in P$ we have $h\in\changes[G]$.
\end{theorem}
\begin{proof}
    Follows by \auto{common_coordinate_changes,open_partial_homeomorphisms_iff}.
\end{proof}

\begin{construction}\label{pseudogroup_intersection}
    Let $E$ be a topological space.
    Let $P$ be a pseudogroup family on $E$.
    The coordinate pseudogroup $\PseudogroupIntersection{E}{P}=
        \structure{\model=E,\changes=\commonCoordinateChanges{E}{P}}$.
\end{construction}
\begin{proof}
    Let $H=\PseudogroupIntersection{E}{P}$.
    $\model[H]=E$ and $\changes[H]=\commonCoordinateChanges{E}{P}$ by
        \auto{coordinate_pseudogroupaggregateprojection_model,
            coordinate_pseudogroupaggregateprojection_changes}.
    $H$ is a record by \obv{coordinate_pseudogroupaggregateexact}.
    For every $G\in P$ we have $G$ is a coordinate pseudogroup and $\model[G]=E$
        by \auto{pseudogroup_family,pseudogroup_on}.
    Every element of $\changes[H]$ is an open partial homeomorphism from
        $\model[H]$ to $\model[H]$ by \auto{common_coordinate_changes_iff}.
    $\carrier[E]\in\opens[E]$ by \obv{carrier_open}.
    $\identity{\carrier[E]}$ is an open partial homeomorphism from $E$ to $E$ by
        \obv{identity_open_partial_homeomorphism}.
    For every $G\in P$ we have $\identity{\carrier[E]}\in\changes[G]$ by
        \auto{pseudogroup_identity}.
    $\identity{\carrier[\model[H]]}\in\changes[H]$ by
        \auto{common_coordinate_changes_iff}.
    Show that for every $h\in\changes[H]$ we have for every $U\in\opens[\model[H]]$
        we have $\restrl{h}{U}\in\changes[H]$.
    \begin{subproof}
        Fix $h\in\changes[H]$ and $U\in\opens[\model[H]]$.
        $h$ is an open partial homeomorphism from $E$ to $E$ and
            for every $G\in P$ we have $h\in\changes[G]$ by
            \auto{common_coordinate_changes_iff}.
        $\restrl{h}{U}$ is an open partial homeomorphism from $E$ to $E$ by
            \obv{open_partial_homeomorphism_restriction}.
        For every $G\in P$ we have $\restrl{h}{U}\in\changes[G]$ by
            \auto{pseudogroup_restriction}.
        Follows by \auto{common_coordinate_changes_iff}.
    \end{subproof}
    Show that for every $h\in\changes[H]$ we have $\converse{h}\in\changes[H]$.
    \begin{subproof}
        Fix $h\in\changes[H]$.
        $h$ is an open partial homeomorphism from $E$ to $E$ and
            for every $G\in P$ we have $h\in\changes[G]$ by
            \auto{common_coordinate_changes_iff}.
        $\converse{h}$ is an open partial homeomorphism from $E$ to $E$ by
            \obv{open_partial_homeomorphism_converse}.
        For every $G\in P$ we have $\converse{h}\in\changes[G]$ by
            \auto{pseudogroup_converse}.
        Follows by \auto{common_coordinate_changes_iff}.
    \end{subproof}
    Show that for all $f,g\in\changes[H]$ we have $g\circ f\in\changes[H]$.
    \begin{subproof}
        Fix $f,g\in\changes[H]$.
        $f$ is an open partial homeomorphism from $E$ to $E$ and
            $g$ is an open partial homeomorphism from $E$ to $E$ by
            \auto{common_coordinate_changes_iff}.
        For every $G\in P$ we have $f,g\in\changes[G]$ by
            \auto{common_coordinate_changes_iff}.
        $g\circ f$ is an open partial homeomorphism from $E$ to $E$ by
            \obv{open_partial_homeomorphism_composition}.
        For every $G\in P$ we have $g\circ f\in\changes[G]$ by
            \auto{pseudogroup_composition}.
        Follows by \auto{common_coordinate_changes_iff}.
    \end{subproof}
    Show that for every $h$ such that $h$ is an open partial homeomorphism
        from $\model[H]$ to $\model[H]$ we have
        for every $B\subseteq\opens[\model[H]]$ such that
            (for every $x\in\dom{h}$ there exists $U\in B$ such that $x\in U$) and
            (for every $U\in B$ we have $\restrl{h}{U}\in\changes[H]$)
        we have $h\in\changes[H]$.
    \begin{subproof}
        Fix $h$ such that $h$ is an open partial homeomorphism
            from $\model[H]$ to $\model[H]$.
        Fix $B\subseteq\opens[\model[H]]$ such that
            (for every $x\in\dom{h}$ there exists $U\in B$ such that $x\in U$) and
            (for every $U\in B$ we have $\restrl{h}{U}\in\changes[H]$).
        For every $G\in P$ we have for every $U\in B$ we have
            $\restrl{h}{U}\in\changes[G]$ by \auto{common_coordinate_changes_iff}.
        For every $G\in P$ we have $h\in\changes[G]$ by \auto{pseudogroup_locality}.
        Follows by \auto{common_coordinate_changes_iff}.
    \end{subproof}
    $H$ is a coordinate pseudogroup by \auto{coordinate_pseudogroup}.
    Follows by \auto{coordinate_pseudogroup}.
\end{proof}

\begin{theorem}\label{pseudogroup_intersection_model}
    Let $E$ be a topological space.
    Let $P$ be a pseudogroup family on $E$.
    Then $\model[\PseudogroupIntersection{E}{P}]=E$.
\end{theorem}
\begin{proof}
    Follows by \obv{pseudogroup_intersectionprojection_model}.
\end{proof}

\begin{classification}\label{pseudogroup_intersection_on}
    Let $E$ be a topological space.
    Let $P$ be a pseudogroup family on $E$.
    Then $\PseudogroupIntersection{E}{P}$ is a pseudogroup on $E$.
\end{classification}
\begin{proof}
    $\PseudogroupIntersection{E}{P}$ is a coordinate pseudogroup by \obv{pseudogroup_intersection}.
    $\model[\PseudogroupIntersection{E}{P}]=E$ by \obv{pseudogroup_intersection_model}.
    Follows by \obv{pseudogroup_on}.
\end{proof}

\begin{theorem}\label{pseudogroup_intersection_changes}
    Let $E$ be a topological space.
    Let $P$ be a pseudogroup family on $E$.
    Then $h\in\changes[\PseudogroupIntersection{E}{P}]$ iff
        $h$ is an open partial homeomorphism from $E$ to $E$ and
        for every $G\in P$ we have $h\in\changes[G]$.
\end{theorem}
\begin{proof}
    $\changes[\PseudogroupIntersection{E}{P}]=\commonCoordinateChanges{E}{P}$ by
        \obv{pseudogroup_intersectionprojection_changes}.
    Follows by \auto{common_coordinate_changes_iff}.
\end{proof}

\begin{theorem}\label{pseudogroup_intersection_is_exact}
    Let $E$ be a topological space.
    Let $P$ be a pseudogroup family on $E$.
    Then $\PseudogroupIntersection{E}{P}$ is an exact coordinate pseudogroup.
\end{theorem}
\begin{proof}
    Follows by \auto{pseudogroup_intersection,pseudogroup_intersectionshape}.
\end{proof}

\begin{theorem}\label{pseudogroup_intersection_lower_bound}
    Let $E$ be a topological space.
    Let $P$ be a pseudogroup family on $E$.
    Let $G\in P$.
    Then $\PseudogroupIntersection{E}{P}$ is a subpseudogroup of $G$.
\end{theorem}
\begin{proof}
    $G$ is a pseudogroup on $E$ by \obv{pseudogroup_family_member_on}.
    $G$ is a coordinate pseudogroup and $\model[G]=E$ by \obv{pseudogroup_on}.
    $\PseudogroupIntersection{E}{P}$ is a coordinate pseudogroup by
        \obv{pseudogroup_intersection}.
    $\model[\PseudogroupIntersection{E}{P}]=E$ by
        \obv{pseudogroup_intersection_model}.
    $\changes[\PseudogroupIntersection{E}{P}]\subseteq\changes[G]$ by
        \auto{pseudogroup_intersection_changes,subseteq}.
    Follows by \obv{subpseudogroup}.
\end{proof}

\begin{theorem}\label{pseudogroup_intersection_greatest}
    Let $E$ be a topological space.
    Let $P$ be a pseudogroup family on $E$.
    Let $H$ be a coordinate pseudogroup.
    Suppose $\model[H]=E$.
    Suppose for every $G\in P$ we have $H$ is a subpseudogroup of $G$.
    Then $H$ is a subpseudogroup of $\PseudogroupIntersection{E}{P}$.
\end{theorem}
\begin{proof}
    $\PseudogroupIntersection{E}{P}$ is a coordinate pseudogroup by
        \obv{pseudogroup_intersection}.
    $\model[\PseudogroupIntersection{E}{P}]=E$ by
        \obv{pseudogroup_intersection_model}.
    For every $h\in\changes[H]$ we have
        $h$ is an open partial homeomorphism from $E$ to $E$ by
        \auto{pseudogroup_changes}.
    For every $h\in\changes[H]$ we have for every $G\in P$ we have
        $h\in\changes[G]$ by \auto{subpseudogroup,subseteq}.
    $\changes[H]\subseteq\changes[\PseudogroupIntersection{E}{P}]$ by
        \auto{pseudogroup_intersection_changes,subseteq}.
    Follows by \obv{subpseudogroup}.
\end{proof}

\subsection{Finite families}

\begin{classification}\label{empty_pseudogroup_family}
    Let $E$ be a topological space.
    Then $\emptyset$ is a pseudogroup family on $E$.
\end{classification}
\begin{proof}
    Follows by \auto{pseudogroup_family}.
\end{proof}

\begin{classification}\label{singleton_pseudogroup_family}
    Let $E$ be a topological space.
    Let $G$ be a pseudogroup on $E$.
    Then $\{G\}$ is a pseudogroup family on $E$.
\end{classification}
\begin{proof}
    Follows by \auto{pseudogroup_family,singleton_iff}.
\end{proof}

\begin{classification}\label{pair_pseudogroup_family}
    Let $E$ be a topological space.
    Let $G$ be a pseudogroup on $E$.
    Let $H$ be a pseudogroup on $E$.
    Then $\{G,H\}$ is a pseudogroup family on $E$.
\end{classification}
\begin{proof}
    Follows by \auto{pseudogroup_family,cons_iff}.
\end{proof}

\begin{theorem}\label{empty_pseudogroup_intersection}
    Let $E$ be a topological space.
    Then $\PseudogroupIntersection{E}{\emptyset}=\FullPseudogroup{E}$.
\end{theorem}
\begin{proof}
    $\emptyset$ is a pseudogroup family on $E$ by \obv{empty_pseudogroup_family}.
    $\PseudogroupIntersection{E}{\emptyset}$ is an exact coordinate pseudogroup and
        $\FullPseudogroup{E}$ is an exact coordinate pseudogroup by
        \obv{pseudogroup_intersection_is_exact,full_pseudogroup_is_exact}.
    $\model[\PseudogroupIntersection{E}{\emptyset}]=E$ and
        $\model[\FullPseudogroup{E}]=E$ by
        \obv{pseudogroup_intersection_model,full_pseudogroup_model}.
    For every $h$ we have $h\in\changes[\PseudogroupIntersection{E}{\emptyset}]$ iff
        $h\in\changes[\FullPseudogroup{E}]$ by
        \auto{pseudogroup_intersection_changes,full_pseudogroup_changes}.
    $\changes[\PseudogroupIntersection{E}{\emptyset}]=\changes[\FullPseudogroup{E}]$ by
        \obv{setext}.
    Follows by \obv{exact_coordinate_pseudogroup_extensionality}.
\end{proof}

\begin{theorem}\label{binary_pseudogroup_intersection_changes}
    Let $E$ be a topological space.
    Let $G$ be a pseudogroup on $E$.
    Let $H$ be a pseudogroup on $E$.
    Then $h\in\changes[\PseudogroupIntersection{E}{\{G,H\}}]$ iff
        $h\in\changes[G]\inter\changes[H]$.
\end{theorem}
\begin{proof}
    $\{G,H\}$ is a pseudogroup family on $E$ by \obv{pair_pseudogroup_family}.
    Follows by \auto{pseudogroup_intersection_changes,pseudogroup_changes,
        pseudogroup_on,cons_iff,inter}.
\end{proof}
