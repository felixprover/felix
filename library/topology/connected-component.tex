\section{Connected components}

\import{topology/connected-union.tex}
\import{topology/local-connectedness.tex}

The component of a point is the union of all connected subsets containing
it. This is a bounded family of carrier subsets, with no choice of paths
or neighbourhoods. At a point outside the carrier the defining family is empty.

\begin{definition}\label{connected_component}
    Let $X$ be a topological space.
    $\connectedComponent{X}{p}=\unions{\{A\in\pow{\carrier[X]}\mid
        \text{$A$ is connected in $X$ and $p\in A$}\}}$.
\end{definition}

\begin{theorem}\label{connected_component_member_iff}
    Let $X$ be a topological space.
    Then $q\in\connectedComponent{X}{p}$ iff there exists $A$ such that
        $A$ is connected in $X$ and $p\in A$ and $q\in A$.
\end{theorem}
\begin{proof}
    Let $F=\{A\in\pow{\carrier[X]}\mid\text{$A$ is connected in $X$ and $p\in A$}\}$.
    $\connectedComponent{X}{p}=\unions{F}$ by \obv{connected_component}.
    Show that if $q\in\connectedComponent{X}{p}$, then there exists $A$ such that
        $A$ is connected in $X$ and $p\in A$ and $q\in A$.
    \begin{subproof}
        Assume $q\in\connectedComponent{X}{p}$.
        Take $A\in F$ such that $q\in A$ by \auto{unions_iff}.
        Follows by \auto.
    \end{subproof}
    Show that if there exists $A$ such that $A$ is connected in $X$ and
        $p\in A$ and $q\in A$, then $q\in\connectedComponent{X}{p}$.
    \begin{subproof}
        Assume there exists $A$ such that $A$ is connected in $X$ and $p\in A$ and $q\in A$.
        Take $A$ such that $A$ is connected in $X$ and $p\in A$ and $q\in A$.
        $A\subseteq\carrier[X]$ by \obv{connected_in_subseteq_carrier}.
        $A\in\pow{\carrier[X]}$ by \obv{pow_iff}.
        $A\in F$.
        Follows by \auto{unions_intro}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{theorem}\label{connected_component_is_connected}
    Let $X$ be a topological space.
    Then $\connectedComponent{X}{p}$ is connected in $X$.
\end{theorem}
\begin{proof}
    Let $F=\{A\in\pow{\carrier[X]}\mid\text{$A$ is connected in $X$ and $p\in A$}\}$.
    For every $A\in F$ we have $A$ is connected in $X$ and $p\in A$.
    $\unions{F}$ is connected in $X$ by \obv{union_of_connected_subsets_with_common_point}.
    Follows by \auto{connected_component}.
\end{proof}

\begin{theorem}\label{point_in_connected_component}
    Let $X$ be a topological space and $p\in\carrier[X]$.
    Then $p\in\connectedComponent{X}{p}$.
\end{theorem}
\begin{proof}
    $\{p\}\subseteq\carrier[X]$ by \obv{singleton_subset_intro}.
    $\carrier[\subspaceTopology{X}{\{p\}}]=\{p\}$ by \obv{subspace_topology_carrier_of_subset}.
    $\{p\}$ is a subsingleton by \auto{singleton_iff}.
    $\subspaceTopology{X}{\{p\}}$ is connected by
        \auto{subsingleton_topological_space_is_connected,subspace_topology_is_topological_space}.
    $\{p\}$ is connected in $X$ by \obv{connected_in}.
    Follows by \auto{connected_component_member_iff,singleton_intro}.
\end{proof}

\begin{theorem}\label{connected_component_contains_connected_subset}
    Let $X$ be a topological space.
    Suppose $A$ is connected in $X$ and $p\in A$.
    Then $A\subseteq\connectedComponent{X}{p}$.
\end{theorem}
\begin{proof}
    Follows by \auto{connected_component_member_iff,subseteq}.
\end{proof}

\begin{theorem}\label{connected_component_eq_at_member}
    Let $X$ be a topological space.
    Suppose $q\in\connectedComponent{X}{p}$.
    Then $\connectedComponent{X}{q}=\connectedComponent{X}{p}$.
\end{theorem}
\begin{proof}
    Take $A$ such that $A$ is connected in $X$ and $p\in A$ and $q\in A$ by
        \auto{connected_component_member_iff}.
    $A\subseteq\carrier[X]$ by \obv{connected_in_subseteq_carrier}.
    $p\in\carrier[X]$ by \obv{subseteq}.
    $p\in\connectedComponent{X}{p}$ by \obv{point_in_connected_component}.
    $\connectedComponent{X}{p}$ is connected in $X$ by \obv{connected_component_is_connected}.
    $\connectedComponent{X}{p}\subseteq\connectedComponent{X}{q}$ by
        \obv{connected_component_contains_connected_subset}.
    $p\in\connectedComponent{X}{q}$ by \obv{subseteq}.
    $\connectedComponent{X}{q}$ is connected in $X$ by \obv{connected_component_is_connected}.
    $\connectedComponent{X}{q}\subseteq\connectedComponent{X}{p}$ by
        \obv{connected_component_contains_connected_subset}.
    Follows by \obv{subseteq_antisymmetric}.
\end{proof}

\begin{theorem}\label{connected_component_outside_carrier}
    Let $X$ be a topological space.
    Suppose $p\notin\carrier[X]$.
    Then $\connectedComponent{X}{p}=\emptyset$.
\end{theorem}
\begin{proof}
    For every $q$ we have $q\notin\connectedComponent{X}{p}$ by
        \auto{connected_component_member_iff,connected_in_subseteq_carrier,subseteq}.
    Follows by \auto{setext,emptyset}.
\end{proof}

\begin{theorem}\label{locally_connected_components_are_open}
    Let $X$ be a locally connected topological space.
    Then $\connectedComponent{X}{p}\in\opens[X]$.
\end{theorem}
\begin{proof}
    $\connectedComponent{X}{p}$ is connected in $X$ by \obv{connected_component_is_connected}.
    $\connectedComponent{X}{p}\subseteq\carrier[X]$ by \obv{connected_in_subseteq_carrier}.
    $\carrier[X]\in\opens[X]$ by \obv{carrier_open}.
    Show that for every $q\in\connectedComponent{X}{p}$ there exists $V\in\opens[X]$ such that
        $q\in V$ and $V\subseteq\connectedComponent{X}{p}$.
    \begin{subproof}
        Fix $q\in\connectedComponent{X}{p}$.
        $q\in\carrier[X]$ by \obv{subseteq}.
        Take $V\in\opens[X]$ such that $q\in V\subseteq\carrier[X]$ and
            $\subspaceTopology{X}{V}$ is connected by \obv{locally_connected}.
        $V$ is connected in $X$ by \obv{connected_in}.
        $V\subseteq\connectedComponent{X}{q}$ by \obv{connected_component_contains_connected_subset}.
        $\connectedComponent{X}{q}=\connectedComponent{X}{p}$ by \obv{connected_component_eq_at_member}.
        Follows by \auto.
    \end{subproof}
    Follows by \auto{open_iff_local_open_subsets}.
\end{proof}
