\section{Raw topology families}

A raw topology family separates the mathematical closure conditions from the
native structure packaging. This is useful for constructing and reasoning
about explicit open families independently of the record that later packages
them as a topological space.

\begin{definition}\label{topology_family}
    $T$ is a topology on $A$ iff
        $T\subseteq\pow{A}$ and
        $A\in T$ and
        (for all $U,V\in T$ we have $U\inter V\in T$) and
        (for all $F\subseteq T$ we have $\unions{F}\in T$).
\end{definition}

\begin{proposition}\label{topology_family_empty}
    Suppose $T$ is a topology on $A$.
    Then $\emptyset\in T$.
\end{proposition}
\begin{proof}
    $\emptyset\subseteq T$ by \obv{emptyset_subseteq}.
    $\unions{\emptyset}=\emptyset$ by \obv{unions_emptyset}.
    Follows by \auto{topology_family}.
\end{proof}

\begin{proposition}\label{topology_family_binary_union}
    Suppose $T$ is a topology on $A$.
    Let $U,V\in T$.
    Then $U\union V\in T$.
\end{proposition}
\begin{proof}
    $\{U,V\}\subseteq T$ by \auto{cons_iff,subseteq}.
    $\unions{\{U,V\}}\in T$ by \obv{topology_family}.
    $\unions{\{U,V\}}=U\union V$ by \obv{union_as_unions}.
\end{proof}

\begin{proposition}\label{topology_family_members_subset_carrier}
    Suppose $T$ is a topology on $A$.
    Let $U\in T$.
    Then $U\subseteq A$.
\end{proposition}
\begin{proof}
    $U\in\pow{A}$ by \auto{topology_family,subseteq}.
    Follows by \obv{pow_iff,subseteq}.
\end{proof}
