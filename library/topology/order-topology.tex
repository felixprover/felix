\section{The topology induced by a linear order}

\import{order/interval.tex}
\import{topology/basis.tex}

The order topology is generated by the two kinds of open ray.  Retaining the
rays as a prebasis handles endpoints and empty carriers uniformly; their
finite intersections form the canonical basis.

\begin{definition}\label{order_topology_prebasis}
    $\orderTopologyPrebasis{L}
        =\{U\in\pow{\carrier[L]}\mid
            \text{there exists $a\in\carrier[L]$ such that
                $U=\orderOpenLowerRay{L}{a}$ or
                $U=\orderOpenUpperRay{L}{a}$}\}$.
\end{definition}

\begin{proposition}\label{order_topology_prebasis_iff}
    $U\in\orderTopologyPrebasis{L}$ iff
        $U\subseteq\carrier[L]$ and there exists $a\in\carrier[L]$
        such that $U=\orderOpenLowerRay{L}{a}$ or
        $U=\orderOpenUpperRay{L}{a}$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{order_topology_prebasis,pow_iff,subseteq}.
\end{proof}

\begin{proposition}\label{order_topology_prebasis_type}
    $\orderTopologyPrebasis{L}$ is a family of subsets of $\carrier[L]$.
\end{proposition}
\begin{proof}
    Follows by \auto{order_topology_prebasis_iff,subseteq}.
\end{proof}

\begin{definition}\label{order_topology}
    $\orderTopology{L}
        =\GeneratedTopology{\carrier[L]}{\orderTopologyPrebasis{L}}$.
\end{definition}

\begin{classification}\label{order_topology_is_topological_space}
    Let $L$ be a linearly ordered set.
    Then $\orderTopology{L}$ is a topological space.
\end{classification}
\begin{proof}
    $\orderTopologyPrebasis{L}$ is a family of subsets of $\carrier[L]$
        by \obv{order_topology_prebasis_type}.
    Follows by
        \auto{order_topology,generated_topology_is_topological_space}.
\end{proof}

\begin{theorem}\label{order_topology_is_exact}
    Let $L$ be a linearly ordered set.
    Then $\orderTopology{L}$ is an exact topological space.
\end{theorem}
\begin{proof}
    $\orderTopologyPrebasis{L}$ is a family of subsets of $\carrier[L]$
        by \obv{order_topology_prebasis_type}.
    Follows by \auto{order_topology,generated_topology_is_exact}.
\end{proof}

\begin{proposition}\label{order_topology_carrier}
    $\carrier[\orderTopology{L}]=\carrier[L]$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{order_topology,generated_topology_carrier}.
\end{proof}

\begin{proposition}\label{order_topology_opens}
    $\opens[\orderTopology{L}]
        =\generatedOpens{\carrier[L]}{\orderTopologyPrebasis{L}}$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{order_topology,generated_topology_opens}.
\end{proof}

\begin{theorem}\label{order_topology_prebasis_is_open_prebasis}
    Let $L$ be a linearly ordered set.
    Then $\orderTopologyPrebasis{L}$ is an open prebasis of
        $\orderTopology{L}$.
\end{theorem}
\begin{proof}
    $\orderTopologyPrebasis{L}$ is a family of subsets of $\carrier[L]$
        by \obv{order_topology_prebasis_type}.
    Follows by
        \auto{order_topology,
            generated_topology_generators_form_open_prebasis}.
\end{proof}

\begin{theorem}\label{order_topology_finite_intersections_are_open_basis}
    Let $L$ be a linearly ordered set.
    Then
        $\finiteIntersectionsOn{\carrier[L]}{\orderTopologyPrebasis{L}}$
        is an open basis of $\orderTopology{L}$.
\end{theorem}
\begin{proof}
    $\orderTopologyPrebasis{L}$ is an open prebasis of
        $\orderTopology{L}$ by
        \obv{order_topology_prebasis_is_open_prebasis}.
    $\carrier[\orderTopology{L}]=\carrier[L]$ by
        \obv{order_topology_carrier}.
    $\finiteIntersectionsOn{\carrier[\orderTopology{L}]}
        {\orderTopologyPrebasis{L}}$ is an open basis of
        $\orderTopology{L}$ by
        \obv{finite_intersections_of_open_prebasis_form_open_basis}.
\end{proof}

\begin{theorem}\label{order_open_lower_ray_is_open}
    Let $L$ be a linearly ordered set and $b\in\carrier[L]$.
    Then $\orderOpenLowerRay{L}{b}\in\opens[\orderTopology{L}]$.
\end{theorem}
\begin{proof}
    $\orderOpenLowerRay{L}{b}\in\orderTopologyPrebasis{L}$ by
        \auto{order_topology_prebasis_iff,
            order_interval_and_ray_types}.
    $\orderTopologyPrebasis{L}\subseteq
        \generatedOpens{\carrier[L]}{\orderTopologyPrebasis{L}}$ by
        \obv{generated_opens_contains,order_topology_prebasis_type}.
    Hence $\orderOpenLowerRay{L}{b}\in
        \generatedOpens{\carrier[L]}{\orderTopologyPrebasis{L}}$ by
        \obv{subseteq}.
    $\opens[\orderTopology{L}]
        =\generatedOpens{\carrier[L]}{\orderTopologyPrebasis{L}}$ by
        \obv{order_topology_opens}.
\end{proof}

\begin{theorem}\label{order_open_upper_ray_is_open}
    Let $L$ be a linearly ordered set and $a\in\carrier[L]$.
    Then $\orderOpenUpperRay{L}{a}\in\opens[\orderTopology{L}]$.
\end{theorem}
\begin{proof}

        \auto{order_topology_prebasis_iff,
            order_interval_and_ray_types}.
    $\orderTopologyPrebasis{L}\subseteq
        \generatedOpens{\carrier[L]}{\orderTopologyPrebasis{L}}$ by
        \obv{generated_opens_contains,order_topology_prebasis_type}.
    Hence $\orderOpenUpperRay{L}{a}\in
        \generatedOpens{\carrier[L]}{\orderTopologyPrebasis{L}}$ by

    $\opens[\orderTopology{L}]
        =\generatedOpens{\carrier[L]}{\orderTopologyPrebasis{L}}$ by

\end{proof}

\begin{theorem}\label{order_open_interval_is_open}
    Let $L$ be a linearly ordered set and $a,b\in\carrier[L]$.
    Then $\orderOpenInterval{L}{a}{b}\in\opens[\orderTopology{L}]$.
\end{theorem}
\begin{proof}
    $\orderOpenUpperRay{L}{a},\orderOpenLowerRay{L}{b}
        \in\opens[\orderTopology{L}]$ by
        \obv{order_open_upper_ray_is_open,order_open_lower_ray_is_open}.
    Hence $\orderOpenUpperRay{L}{a}\inter\orderOpenLowerRay{L}{b}
        \in\opens[\orderTopology{L}]$ by
        \obv{opens_inter,order_topology_is_topological_space}.
    $\orderOpenInterval{L}{a}{b}
        =\orderOpenUpperRay{L}{a}\inter\orderOpenLowerRay{L}{b}$ by
        \obv{order_open_interval_eq_intersection_rays}.
\end{proof}

\begin{theorem}\label{order_closed_lower_ray_is_closed}
    Let $L$ be a linearly ordered set and $b\in\carrier[L]$.
    Then $\orderClosedLowerRay{L}{b}$ is closed in $\orderTopology{L}$.
\end{theorem}
\begin{proof}
    $\orderTopology{L}$ is a topological space by
        \obv{order_topology_is_topological_space}.
    $\carrier[\orderTopology{L}]=\carrier[L]$ by
        \obv{order_topology_carrier}.
    $\orderOpenUpperRay{L}{b}\in\opens[\orderTopology{L}]$ by
        \obv{order_open_upper_ray_is_open}.
    Hence $\orderOpenUpperRay{L}{b}$ is open in $\orderTopology{L}$ by
        \obv{is_open}.
    $\orderClosedLowerRay{L}{b}\subseteq\carrier[L]$ by
        \obv{order_closed_lower_ray,subseteq}.
    $\carrier[L]\setminus\orderClosedLowerRay{L}{b}
        =\orderOpenUpperRay{L}{b}$ by
        \obv{order_complement_closed_lower_ray}.
    Follows by
        \auto{is_closed_in,closeds,pow_iff,subseteq}.
\end{proof}

\begin{theorem}\label{order_closed_upper_ray_is_closed}
    Let $L$ be a linearly ordered set and $a\in\carrier[L]$.

\end{theorem}
\begin{proof}

    $\carrier[\orderTopology{L}]=\carrier[L]$ by
        \obv{order_topology_carrier}.

    Follows by

\end{proof}

\begin{theorem}\label{order_closed_interval_is_closed}
    Let $L$ be a linearly ordered set and $a,b\in\carrier[L]$.
    Then $\orderClosedInterval{L}{a}{b}$ is closed in
        $\orderTopology{L}$.
\end{theorem}
\begin{proof}
    $\orderClosedUpperRay{L}{a}$ and $\orderClosedLowerRay{L}{b}$ are
        closed in $\orderTopology{L}$ by
        \obv{order_closed_upper_ray_is_closed,
            order_closed_lower_ray_is_closed}.
    $\orderClosedUpperRay{L}{a},\orderClosedLowerRay{L}{b}
        \subseteq\carrier[\orderTopology{L}]$ by
        \obv{order_topology_carrier,order_interval_and_ray_types}.
    Hence $\orderClosedUpperRay{L}{a}\inter
        \orderClosedLowerRay{L}{b}$ is closed in $\orderTopology{L}$ by
        \obv{intersection_of_closed_is_closed}.
    $\orderClosedInterval{L}{a}{b}
        =\orderClosedUpperRay{L}{a}\inter
            \orderClosedLowerRay{L}{b}$ by
        \obv{order_closed_interval_eq_intersection_rays}.
\end{proof}
