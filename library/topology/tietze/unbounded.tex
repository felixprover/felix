\section{The real-valued Tietze extension theorem}

\begin{theorem}\label{tietze_extension_real_from_normal_shrinker}
    Let $X$ be a topological space.
    Let $h$ be a normal shrinker for $X$.
    Let $A\in\closeds{X}$.
    Suppose $f\in\funs{A}{\reals}$ and $f$ is continuous from
        $\subspaceTopology{X}{A}$ to $\RealTopology$.
    Then there exists $F\in\funs{\carrier[X]}{\reals}$ such that
        $F$ is continuous from $X$ to $\RealTopology$ and
        $\restrl{F}{A}=f$.
\end{theorem}
\begin{proof}
    Let $S=\subspaceTopology{X}{A}$.
    Let $I=\RealOpenSymmetricUnitInterval$.
    Let $J=\RealOpenSymmetricUnitIntervalTopology$.
    Let $C=\realClosedInterval
        {\surrealNeg{\surrealOne}}{\surrealOne}$.
    Let $K=\realClosedIntervalTopology
        {\surrealNeg{\surrealOne}}{\surrealOne}$.
    Let $s=\realSquash\circ f$.
    $A\subseteq\carrier[X]$ by \obv{closeds,pow_iff,subseteq}.
    Hence $\carrier[S]=A$
        by \auto{subspace_topology_carrier_of_subset}.
    $\realSquash\in\funs{\reals}{I}$
        by \obv{real_squash_range_open_symmetric_unit}.
    Hence $s\in\funs{A}{I}$ by \auto{funs_circ}.
    $\realSquash$ is continuous from $\RealTopology$ to $J$
        by \obv{real_squash_continuous_open_interval}.
    Hence $s$ is continuous from $S$ to $J$
        by \auto{continuous_composition}.
    $I\subseteq C$
        by
        \auto{real_open_symmetric_unit_interval,real_open_interval_subset_closed_interval}.
    Hence $s\in\funs{A}{C}$ by \obv{funs_weaken_codom}.
    $J=\subspaceTopology{\RealTopology}{I}$
        by
        \obv{real_open_symmetric_unit_interval_topology,real_open_interval_topology,real_open_symmetric_unit_interval}.
    $K=\subspaceTopology{\RealTopology}{C}$
        by \obv{real_closed_interval_topology}.
    Hence $s$ is continuous from $S$ to $K$
        by
        \obv{continuous_codomain_weaken_to_nested_subspace,real_topology_is_topological_space}.
    $\surrealNeg{\surrealOne},\surrealOne\in\reals$
        by
        \obv{surreal_one_is_real,surreal_real_negation_closed}.
    $\surrealLt{\surrealNeg{\surrealOne}}{\surrealZero}$
        by
        \obv{surreal_negation_lt_reverses,surreal_negation_empty,surreal_zero_lt_one,surreal_one_is_surreal,empty_surreal}.
    Hence $\surrealLt{\surrealNeg{\surrealOne}}{\surrealOne}$
        by
        \obv{surreal_lt_transitive,surreal_zero_lt_one,surreal_one_is_surreal,surreal_negation_is_surreal,empty_surreal}.
    Hence $\surrealLe{\surrealNeg{\surrealOne}}{\surrealOne}$
        by \obv{surreal_le}.
    $s\in\funs{A}{C}$ and $s$ is continuous from $S$ to $K$.
    Take $g\in\funs{\carrier[X]}{C}$ such that
        $g$ is continuous from $X$ to $K$ and
        $\restrl{g}{A}=s$
        by
        \auto{tietze_extension_closed_interval_from_normal_shrinker}.
    Show that for every $x\in A$ we have $g(x)\in I$.
    \begin{subproof}
        Fix $x\in A$.
        Hence $x\in\carrier[X]$ by \obv{subseteq}.
        $A\subseteq\dom{g}$ by \auto{funs,subseteq}.
        Hence $(\restrl{g}{A})(x)=g(x)$
            by
            \obv{restrl_of_function_apply,funs_is_function}.
        $s(x)\in I$ by \obv{funs_type_apply}.
    \end{subproof}
    Take $k\in\funs{\carrier[X]}{I}$ such that
        $k$ is continuous from $X$ to $J$ and
        $\restrl{k}{A}=\restrl{g}{A}$
        by
        \auto{normal_shrinker_push_closed_unit_interval_into_open}.
    Let $F=\realUnsquash\circ k$.
    $\realUnsquash\in\funs{I}{\reals}$
        by \obv{real_unsquash_typed_real}.
    Hence $F\in\funs{\carrier[X]}{\reals}$ by \auto{funs_circ}.
    $\realUnsquash$ is continuous from $J$ to $\RealTopology$
        by \obv{real_unsquash_continuous}.
    Hence $F$ is continuous from $X$ to $\RealTopology$
        by \auto{continuous_composition}.
    $\restrl{F}{A}=\realUnsquash\circ\restrl{k}{A}$
        by \auto{restrl_circ}.
    Hence $\restrl{F}{A}=\realUnsquash\circ\restrl{g}{A}$.
    Hence $\restrl{F}{A}=\realUnsquash\circ s$.
    Hence $\restrl{F}{A}=\realUnsquash\circ(\realSquash\circ f)$.
    $(\realUnsquash\circ\realSquash)\circ f
        =\realUnsquash\circ(\realSquash\circ f)$
        by \obv{circ_assoc}.
    Hence $\restrl{F}{A}=(\realUnsquash\circ\realSquash)\circ f$.
    $\realUnsquash\circ\realSquash=\identity{\reals}$
        by \obv{real_unsquash_after_squash}.
    Hence $\restrl{F}{A}=\identity{\reals}\circ f$.
    $\identity{\reals}\circ f=f$
        by \obv{funs_identity_circ_left}.
    Hence $\restrl{F}{A}=f$.
\end{proof}

\begin{theorem}\label{tietze_extension_real}
    Let $X$ be a normal space.
    Let $A\in\closeds{X}$.
    Suppose $f\in\funs{A}{\reals}$ and $f$ is continuous from
        $\subspaceTopology{X}{A}$ to $\RealTopology$.
    Then there exists $F\in\funs{\carrier[X]}{\reals}$ such that
        $F$ is continuous from $X$ to $\RealTopology$ and
        $\restrl{F}{A}=f$.
\end{theorem}
\begin{proof}
    Take $h$ such that $h$ is a normal shrinker for $X$
        by \obv{normal_shrinker_exists}.
    Follows by \obv{tietze_extension_real_from_normal_shrinker}.
\end{proof}
