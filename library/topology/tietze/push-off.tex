\section{Removing accidental endpoint values}

\subsection{The endpoint set}

\begin{definition}\label{closed_unit_endpoints}
    $\closedUnitEndpoints
        =\{\surrealNeg{\surrealOne}\}\union\{\surrealOne\}$.
\end{definition}

\begin{definition}\label{endpoint_set}
    $\endpointSet{g}=\preimg{g}{\closedUnitEndpoints}$.
\end{definition}

\begin{theorem}\label{closed_unit_endpoints_is_closed}
    $\closedUnitEndpoints\in\closeds{\RealTopology}$.
\end{theorem}
\begin{proof}
    $\RealTopology$ is a topological space
        by \obv{real_topology_is_topological_space}.
    $\surrealNeg{\surrealOne}\in\reals$
        by
        \obv{surreal_one_is_real,surreal_real_negation_closed}.
    Hence $\{\surrealNeg{\surrealOne}\}\in\closeds{\RealTopology}$
        by \obv{real_singleton_is_closed}.
    $\{\surrealOne\}\in\closeds{\RealTopology}$
        by \obv{real_singleton_is_closed,surreal_one_is_real}.
    Hence $\{\surrealNeg{\surrealOne}\}$ and $\{\surrealOne\}$
        are closed in $\RealTopology$ by \obv{is_closed_in}.
    $\{\surrealNeg{\surrealOne}\},\{\surrealOne\}
        \subseteq\carrier[\RealTopology]$
        by \auto{closeds,pow_iff,subseteq}.
    Hence
        $\{\surrealNeg{\surrealOne}\}\union\{\surrealOne\}$
            is closed in $\RealTopology$
        by \obv{union_of_closed_is_closed}.
    Follows by \obv{closed_unit_endpoints,is_closed_in}.
\end{proof}

\begin{proposition}\label{endpoint_set_member_iff}
    Suppose $g\in\funs{D}{\reals}$.
    Then $x\in\endpointSet{g}$ iff
        $x\in D$ and $g(x)\in\closedUnitEndpoints$.
\end{proposition}
\begin{proof}
    $g$ is a function and $\dom{g}=D$
        by \obv{funs_is_function,funs}.
    If $x\in\endpointSet{g}$, then
        $x\in D$ and $g(x)\in\closedUnitEndpoints$
        by \auto{endpoint_set,preimg_iff,function_apply_intro,dom_iff}.
    Show that if $x\in D$ and $g(x)\in\closedUnitEndpoints$, then
        $x\in\endpointSet{g}$.
    \begin{subproof}
        Assume $x\in D$ and $g(x)\in\closedUnitEndpoints$.
        $(x,g(x))\in g$ by \obv{function_apply_elim}.
        Hence there exists $b\in\closedUnitEndpoints$ such that
            $x\mathrel{g}b$.
        Follows by \obv{endpoint_set,preimg_iff}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{theorem}\label{endpoint_set_closed}
    Let $X$ be a topological space.
    Suppose $g$ is continuous from $X$ to
        $\realClosedIntervalTopology
            {\surrealNeg{\surrealOne}}{\surrealOne}$.
    Then $\endpointSet{g}\in\closeds{X}$.
\end{theorem}
\begin{proof}
    $\realClosedIntervalTopology
        {\surrealNeg{\surrealOne}}{\surrealOne}
        =\subspaceTopology{\RealTopology}
            {\realClosedInterval
                {\surrealNeg{\surrealOne}}{\surrealOne}}$
        by \obv{real_closed_interval_topology}.
    Hence $g$ is continuous from $X$ to $\RealTopology$
        by
        \auto{continuous_subspace_codomain_inclusion,real_topology_is_topological_space}.
    $\closedUnitEndpoints\in\closeds{\RealTopology}$
        by \obv{closed_unit_endpoints_is_closed}.
    Follows by
        \auto{endpoint_set,continuous_preimage_closed,real_topology_is_topological_space}.
\end{proof}

\begin{theorem}\label{endpoint_set_disjoint_from_fixed_closed_set}
    Let $X$ be a topological space.
    Let $A\in\closeds{X}$.
    Suppose
        $g\in\funs{\carrier[X]}
            {\realClosedInterval
                {\surrealNeg{\surrealOne}}{\surrealOne}}$.
    Suppose for every $x\in A$ we have
        $g(x)\in\RealOpenSymmetricUnitInterval$.
    Then $A$ is disjoint from $\endpointSet{g}$.
\end{theorem}
\begin{proof}
    $\realClosedInterval
        {\surrealNeg{\surrealOne}}{\surrealOne}\subseteq\reals$
        by \obv{real_closed_interval_subset_reals}.
    Hence $g\in\funs{\carrier[X]}{\reals}$
        by \obv{funs_weaken_codom}.
    It suffices to show that there exists no $x$ such that
        $x\in A,\endpointSet{g}$ by \obv{disjoint}.
    Suppose there exists $x$ such that $x\in A,\endpointSet{g}$.
    Take $x$ such that $x\in A,\endpointSet{g}$.
    $g(x)\in\closedUnitEndpoints$
        by \auto{endpoint_set_member_iff}.
    Hence $g(x)=\surrealNeg{\surrealOne}$ or $g(x)=\surrealOne$
        by
        \auto{closed_unit_endpoints,union_iff,singleton_iff}.
    $g(x)\in\RealOpenSymmetricUnitInterval$.
    Hence $g(x)\in\realOpenInterval
        {\surrealNeg{\surrealOne}}{\surrealOne}$
        by \obv{real_open_symmetric_unit_interval}.
    Hence
        $\surrealLt{\surrealNeg{\surrealOne}}{g(x)}$ and
        $\surrealLt{g(x)}{\surrealOne}$
        by \obv{real_open_interval}.
    Contradiction by
        \auto{surreal_lt_irreflexive,surreal_one_is_surreal,surreal_negation_is_surreal}.
\end{proof}

\begin{theorem}\label{endpoint_push_off_map_strict_abs_bound}
    Let $X$ be a topological space.
    Let $h$ be a normal shrinker for $X$.
    Let $A\in\closeds{X}$.
    Suppose
        $g\in\funs{\carrier[X]}
            {\realClosedInterval
                {\surrealNeg{\surrealOne}}{\surrealOne}}$.
    Suppose $g$ is continuous from $X$ to
        $\realClosedIntervalTopology
            {\surrealNeg{\surrealOne}}{\surrealOne}$.
    Suppose for every $x\in A$ we have
        $g(x)\in\RealOpenSymmetricUnitInterval$.
    Then for every $x\in\carrier[X]$ we have
        $\surrealLt{
            \abs{\apply{\endpointPushOffMap{X}{h}{A}{g}}{x}}}
            {\surrealOne}$.
\end{theorem}
\begin{proof}
    Let $E=\endpointSet{g}$.
    Let $q=\normalShrinkerSeparator{X}{h}{A}{E}$.
    $E\in\closeds{X}$ by \auto{endpoint_set_closed}.
    $A$ is disjoint from $E$
        by \obv{endpoint_set_disjoint_from_fixed_closed_set}.
    $q\in\funs{\carrier[X]}{\reals}$
        by \auto{normal_shrinker_separator_typed_real}.
    $\realClosedInterval
        {\surrealNeg{\surrealOne}}{\surrealOne}\subseteq\reals$
        by \obv{real_closed_interval_subset_reals}.
    Hence $g\in\funs{\carrier[X]}{\reals}$
        by \obv{funs_weaken_codom}.
    Show that for every $x\in\carrier[X]$ we have
        $\surrealLt{
            \abs{\apply{\endpointPushOffMap{X}{h}{A}{g}}{x}}}
            {\surrealOne}$.
    \begin{subproof}
        Fix $x\in\carrier[X]$.
        $q(x),g(x)\in\reals$ by \obv{funs_type_apply}.
        $\apply{\endpointPushOffMap{X}{h}{A}{g}}{x}
            =\surrealMul{q(x)}{g(x)}$
            by \auto{endpoint_push_off_map,real_map_mul_apply}.
        \begin{byCase}
        \caseOf{$x\in E$.}
            $\apply{\endpointPushOffMap{X}{h}{A}{g}}{x}
                =\surrealZero$
                by \obv{endpoint_push_off_map_eq_zero_on_endpoints}.
            Follows by
                \obv{surreal_abs_zero,surreal_zero_lt_one}.
        \caseOf{$x\notin E$.}
            $g(x)\in\realClosedInterval
                {\surrealNeg{\surrealOne}}{\surrealOne}$
                by \obv{funs_type_apply}.
            Show that $g(x)\neq\surrealNeg{\surrealOne}$.
            \begin{subproof}
                Assume $g(x)=\surrealNeg{\surrealOne}$.
                Hence $g(x)\in\closedUnitEndpoints$
                    by
                    \auto{closed_unit_endpoints,union_iff,singleton_iff}.
                Hence $x\in E$ by \auto{endpoint_set_member_iff}.
                Contradiction.
            \end{subproof}
            Show that $g(x)\neq\surrealOne$.
            \begin{subproof}
                Assume $g(x)=\surrealOne$.
                Hence $g(x)\in\closedUnitEndpoints$
                    by
                    \auto{closed_unit_endpoints,union_iff,singleton_iff}.
                Hence $x\in E$ by \auto{endpoint_set_member_iff}.
                Contradiction.
            \end{subproof}
            Hence $g(x)\in\realOpenInterval
                {\surrealNeg{\surrealOne}}{\surrealOne}$
                by
                \obv{real_closed_interval_nonendpoint_in_open_interval,surreal_one_is_real,surreal_real_negation_closed}.
            Hence $g(x)\in\RealOpenSymmetricUnitInterval$
                by \obv{real_open_symmetric_unit_interval}.
            Hence $\surrealLt{\abs{g(x)}}{\surrealOne}$
                by
                \auto{real_open_symmetric_interval_iff_abs_bound,real_open_symmetric_unit_interval,real_open_symmetric_interval,surreal_one_is_real,surreal_zero_lt_one,surreal_le}.
            $\surrealLe{\surrealZero}{q(x)}$ and
                $\surrealLe{q(x)}{\surrealOne}$
                by
                \auto{normal_shrinker_separator_between_zero_and_one}.
            Hence $\abs{q(x)}=q(x)$
                by
                \obv{surreal_abs_nonnegative,surreal_real_is_surreal}.
            $\surrealLe{\surrealZero}{\abs{g(x)}}$
                by
                \obv{surreal_abs_nonnegative_value,surreal_real_is_surreal}.
            Hence
                $\surrealLe
                    {\surrealMul{q(x)}{\abs{g(x)}}}
                    {\surrealMul{\surrealOne}{\abs{g(x)}}}$
                by
                \obv{surreal_nonneg_mul_le_right,surreal_real_is_surreal,surreal_abs_is_surreal,surreal_one_is_surreal}.
            $\surrealMul{\surrealOne}{\abs{g(x)}}=\abs{g(x)}$
                by
                \obv{surreal_mul_one_left,surreal_abs_is_surreal,surreal_real_is_surreal,funs_type_apply}.
            Hence
                $\surrealLe
                    {\surrealMul{q(x)}{\abs{g(x)}}}{\abs{g(x)}}$.
            $\abs{\surrealMul{q(x)}{g(x)}}
                =\surrealMul{\abs{q(x)}}{\abs{g(x)}}$
                by
                \obv{surreal_abs_multiplicative,surreal_real_is_surreal}.
            Hence
                $\surrealLe{\abs{\surrealMul{q(x)}{g(x)}}}
                    {\abs{g(x)}}$.
            $\surrealLe{\abs{\surrealMul{q(x)}{g(x)}}}
                {\abs{g(x)}}$ and
                $\surrealLt{\abs{g(x)}}{\surrealOne}$.
            $\surreal{\abs{\surrealMul{q(x)}{g(x)}}}$ and
                $\surreal{\abs{g(x)}}$ and $\surreal{\surrealOne}$
                by
                \obv{funs_type_apply,surreal_real_is_surreal,surreal_mul_is_surreal,surreal_abs_is_surreal,surreal_one_is_surreal}.
            $\surrealLe{\abs{\surrealMul{q(x)}{g(x)}}}
                {\abs{g(x)}}$ and
                $\surrealLt{\abs{g(x)}}{\surrealOne}$.
            Hence
                $\surrealLt{\abs{\surrealMul{q(x)}{g(x)}}}
                    {\surrealOne}$
                by
                \obv{surreal_le_lt_transitive,surreal_abs_is_surreal,surreal_mul_is_surreal,surreal_one_is_surreal}.
        \end{byCase}
    \end{subproof}
\end{proof}

\begin{theorem}\label{endpoint_push_off_map_restriction}
    Let $X$ be a topological space.
    Let $h$ be a normal shrinker for $X$.
    Let $A\in\closeds{X}$.
    Suppose
        $g\in\funs{\carrier[X]}
            {\realClosedInterval
                {\surrealNeg{\surrealOne}}{\surrealOne}}$.
    Suppose $g$ is continuous from $X$ to
        $\realClosedIntervalTopology
            {\surrealNeg{\surrealOne}}{\surrealOne}$.
    Suppose for every $x\in A$ we have
        $g(x)\in\RealOpenSymmetricUnitInterval$.
    Then
        $\restrl{\endpointPushOffMap{X}{h}{A}{g}}{A}
            =\restrl{g}{A}$.
\end{theorem}
\begin{proof}
    Let $k=\endpointPushOffMap{X}{h}{A}{g}$.
    $A\subseteq\carrier[X]$ by \auto{closeds,pow_iff,subseteq}.
    $k\in\funs{\carrier[X]}{\reals}$
        by \auto{endpoint_push_off_map_typed_real}.
    $\realClosedInterval
        {\surrealNeg{\surrealOne}}{\surrealOne}\subseteq\reals$
        by \obv{real_closed_interval_subset_reals}.
    Hence $g\in\funs{\carrier[X]}{\reals}$
        by \obv{funs_weaken_codom}.
    Hence $\restrl{k}{A},\restrl{g}{A}\in\funs{A}{\reals}$
        by \obv{funs_restrl}.
    $A\subseteq\dom{k}$ and $A\subseteq\dom{g}$
        by \auto{funs,subseteq}.
    Show that for every $x\in A$ we have
        $(\restrl{k}{A})(x)=(\restrl{g}{A})(x)$.
    \begin{subproof}
        Fix $x\in A$.
        Hence $x\in\carrier[X]$ by \obv{subseteq}.
        $(\restrl{k}{A})(x)=k(x)$ and
            $(\restrl{g}{A})(x)=g(x)$
            by
            \obv{restrl_of_function_apply,funs_is_function,endpoint_push_off_map_typed_real}.
        $k(x)=g(x)$ by \auto{endpoint_push_off_map_agrees_on_A}.
    \end{subproof}
    Follows by \auto{typed_functions_equal_from_agreement}.
\end{proof}

\begin{theorem}\label{normal_shrinker_push_closed_unit_interval_into_open}
    Let $X$ be a topological space.
    Let $h$ be a normal shrinker for $X$.
    Let $A\in\closeds{X}$.
    Suppose
        $g\in\funs{\carrier[X]}
            {\realClosedInterval
                {\surrealNeg{\surrealOne}}{\surrealOne}}$.
    Suppose $g$ is continuous from $X$ to
        $\realClosedIntervalTopology
            {\surrealNeg{\surrealOne}}{\surrealOne}$.
    Suppose for every $x\in A$ we have
        $g(x)\in\RealOpenSymmetricUnitInterval$.
    Then there exists
        $k\in\funs{\carrier[X]}{\RealOpenSymmetricUnitInterval}$
        such that $k$ is continuous from $X$ to
            $\RealOpenSymmetricUnitIntervalTopology$ and
            $\restrl{k}{A}=\restrl{g}{A}$.
\end{theorem}
\begin{proof}
    Let $k=\endpointPushOffMap{X}{h}{A}{g}$.
    $k\in\funs{\carrier[X]}{\RealOpenSymmetricUnitInterval}$
        by \auto{endpoint_push_off_map_typed_open_interval}.
    $k$ is continuous from $X$ to
        $\RealOpenSymmetricUnitIntervalTopology$
        by \auto{endpoint_push_off_map_continuous_open_interval}.
    $\restrl{k}{A}=\restrl{g}{A}$
        by \auto{endpoint_push_off_map_restriction}.
    Follows by \auto.
\end{proof}
