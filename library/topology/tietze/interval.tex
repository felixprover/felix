\section{Tietze extension into closed real intervals}

\subsection{A symmetric bound for a closed interval}

\begin{theorem}\label{real_closed_interval_abs_bound_by_endpoint_sum}
    Let $a,b\in\reals$.
    Suppose $\surrealLe{a}{b}$.
    Let $x\in\realClosedInterval{a}{b}$.
    Then
        $\surrealLe{\abs{x}}{\surrealAdd{\abs{a}}{\abs{b}}}$.
\end{theorem}
\begin{proof}
    $x\in\reals$ and $\surrealLe{a}{x}$ and $\surrealLe{x}{b}$
        by \obv{real_closed_interval}.
    Hence $\surreal{a}$ and $\surreal{b}$ and $\surreal{x}$
        by \obv{surreal_real_is_surreal}.
    $\surrealLe{b}{\abs{b}}$ and
        $\surrealLe{\surrealNeg{a}}{\abs{a}}$
        by \obv{surreal_le_abs,surreal_neg_le_abs}.
    Hence $\surrealLe{x}{\abs{b}}$
        by \obv{surreal_le_transitive,surreal_abs_is_surreal}.
    $\surrealLe{\surrealZero}{\abs{a}}$
        by \obv{surreal_abs_nonnegative_value}.
    Hence
        $\surrealLe{\abs{b}}{\surrealAdd{\abs{a}}{\abs{b}}}$
        by
        \auto{surreal_add_le_left,surreal_add_zero_left,surreal_add_commutative,surreal_abs_is_surreal,empty_surreal}.
    Hence
        $\surrealLe{x}{\surrealAdd{\abs{a}}{\abs{b}}}$
        by
        \auto{surreal_le_transitive,surreal_abs_is_surreal,surreal_add_is_surreal}.
    $\surrealLe{\surrealNeg{x}}{\surrealNeg{a}}$
        by \obv{surreal_negation_le_reverses}.
    Hence $\surrealLe{\surrealNeg{x}}{\abs{a}}$
        by
        \auto{surreal_le_transitive,surreal_negation_is_surreal,surreal_abs_is_surreal}.
    $\surrealLe{\surrealZero}{\abs{b}}$
        by \obv{surreal_abs_nonnegative_value}.
    Hence
        $\surrealLe{\abs{a}}{\surrealAdd{\abs{a}}{\abs{b}}}$
        by
        \auto{surreal_add_le_right,surreal_add_zero_right,surreal_abs_is_surreal,empty_surreal}.
    Hence
        $\surrealLe{\surrealNeg{x}}
            {\surrealAdd{\abs{a}}{\abs{b}}}$
        by
        \auto{surreal_le_transitive,surreal_negation_is_surreal,surreal_abs_is_surreal,surreal_add_is_surreal}.
    Follows by
        \auto{surreal_abs_le_iff,surreal_abs_is_surreal,surreal_add_is_surreal}.
\end{proof}

\subsection{Real-valued extension with interval range}

\begin{theorem}\label{tietze_extension_closed_interval_real_from_normal_shrinker}
    Let $X$ be a topological space.
    Let $h$ be a normal shrinker for $X$.
    Let $A\in\closeds{X}$.
    Let $a,b\in\reals$.
    Suppose $\surrealLe{a}{b}$.
    Suppose $f\in\funs{A}{\reals}$ and $f$ is continuous from
        $\subspaceTopology{X}{A}$ to $\RealTopology$.
    Suppose for every $x\in A$ we have
        $f(x)\in\realClosedInterval{a}{b}$.
    Then there exists $F\in\funs{\carrier[X]}{\reals}$ such that
        $F$ is continuous from $X$ to $\RealTopology$ and
        $\restrl{F}{A}=f$ and
        for every $x\in\carrier[X]$ we have
            $F(x)\in\realClosedInterval{a}{b}$.
\end{theorem}
\begin{proof}
    Let $r=\surrealAdd{\abs{a}}{\abs{b}}$.
    $\abs{a},\abs{b}\in\reals$ by \obv{surreal_real_abs_closed}.
    Hence $r\in\reals$ by \auto{surreal_real_addition_closed}.
    $\surrealLe{\surrealZero}{\abs{a}}$
        by
        \obv{surreal_abs_nonnegative_value,surreal_real_is_surreal}.
    $\surrealLe{\surrealZero}{\abs{b}}$
        by
        \obv{surreal_abs_nonnegative_value,surreal_real_is_surreal}.
    $\surreal{\abs{a}}$ and $\surreal{\abs{b}}$
        by \obv{surreal_real_is_surreal}.
    $\surrealLe{\surrealAdd{\abs{a}}{\surrealZero}}
        {\surrealAdd{\abs{a}}{\abs{b}}}$
        by \obv{surreal_add_le_right,empty_surreal}.
    $\surrealAdd{\abs{a}}{\surrealZero}=\abs{a}$
        by \obv{surreal_add_zero_right,surreal_abs_is_surreal}.
    Hence $\surrealLe{\abs{a}}{r}$.
    $\surrealLe{\surrealAdd{\surrealZero}{\surrealZero}}
        {\surrealAdd{\abs{a}}{\abs{b}}}$
        by \obv{surreal_add_le_both,empty_surreal}.
    $\surrealAdd{\surrealZero}{\surrealZero}=\surrealZero$
        by \obv{surreal_add_zero_left,empty_surreal}.
    Hence $\surrealLe{\surrealZero}{r}$.
    For every $x\in A$ we have $\surrealLe{\abs{f(x)}}{r}$
        by \obv{real_closed_interval_abs_bound_by_endpoint_sum}.
    Take $G\in\funs{\carrier[X]}{\reals}$ such that
        $G$ is continuous from $X$ to $\RealTopology$ and
        $\restrl{G}{A}=f$ and
        for every $x\in\carrier[X]$ we have
            $\surrealLe{\abs{G(x)}}{r}$
        by
        \obv{tietze_extension_bounded_real_from_normal_shrinker}.
    Let $F=\realClamp{a}{b}\circ G$.
    $F\in\funs{\carrier[X]}{\reals}$
        by \auto{real_clamp_typed_real,funs_circ}.
    $F$ is continuous from $X$ to $\RealTopology$
        by \auto{real_clamp_continuous,continuous_composition}.
    $\realClamp{a}{b}\in
        \funs{\reals}{\realClosedInterval{a}{b}}$
        by \obv{real_clamp_range}.
    Hence $F\in\funs{\carrier[X]}{\realClosedInterval{a}{b}}$
        by \auto{funs_circ}.
    Hence for every $x\in\carrier[X]$ we have
        $F(x)\in\realClosedInterval{a}{b}$
        by \obv{funs_type_apply}.
    $f$ is a function and $\dom{f}=A$
        by \obv{funs_is_function,funs}.
    Hence $f$ is a function to $\realClosedInterval{a}{b}$.
    Hence $f\in\funs{A}{\realClosedInterval{a}{b}}$
        by \obv{funs_intro}.
    $\restrl{F}{A}=\realClamp{a}{b}\circ\restrl{G}{A}$
        by \auto{restrl_circ}.
    Hence $\restrl{F}{A}=\realClamp{a}{b}\circ f$.
    Hence $\restrl{F}{A}=f$
        by \obv{real_clamp_circ_fixes_interval_map}.
\end{proof}

\subsection{Interval-valued extension}

\begin{theorem}\label{tietze_extension_closed_interval_from_normal_shrinker}
    Let $X$ be a topological space.
    Let $h$ be a normal shrinker for $X$.
    Let $A\in\closeds{X}$.
    Let $a,b\in\reals$.
    Suppose $\surrealLe{a}{b}$.
    Suppose $f\in\funs{A}{\realClosedInterval{a}{b}}$ and
        $f$ is continuous from $\subspaceTopology{X}{A}$ to
            $\realClosedIntervalTopology{a}{b}$.
    Then there exists
        $F\in\funs{\carrier[X]}{\realClosedInterval{a}{b}}$ such that
        $F$ is continuous from $X$ to
            $\realClosedIntervalTopology{a}{b}$ and
        $\restrl{F}{A}=f$.
\end{theorem}
\begin{proof}
    $\realClosedIntervalTopology{a}{b}
        =\subspaceTopology{\RealTopology}{\realClosedInterval{a}{b}}$
        by \obv{real_closed_interval_topology}.
    $\subspaceTopology{X}{A}$ is a topological space and
        $\RealTopology$ is a topological space
        by
        \obv{subspace_topology_is_topological_space,real_topology_is_topological_space}.
    Hence $f$ is continuous from $\subspaceTopology{X}{A}$ to
        $\RealTopology$
        by \auto{continuous_subspace_codomain_inclusion}.
    $f$ is a function and $\dom{f}=A$
        by \obv{funs_is_function,funs}.
    $f$ is a function to $\realClosedInterval{a}{b}$
        by \obv{funs_elim}.
    $\realClosedInterval{a}{b}\subseteq\reals$
        by \obv{real_closed_interval_subset_reals}.
    Hence $f$ is a function to $\reals$
        by \obv{function_on_weaken_codom}.
    Hence $f\in\funs{A}{\reals}$ by \obv{funs_intro}.
    For every $x\in A$ we have
        $f(x)\in\realClosedInterval{a}{b}$
        by \obv{funs_type_apply}.
    Take $F\in\funs{\carrier[X]}{\reals}$ such that
        $F$ is continuous from $X$ to $\RealTopology$ and
        $\restrl{F}{A}=f$ and
        for every $x\in\carrier[X]$ we have
            $F(x)\in\realClosedInterval{a}{b}$
        by
        \obv{tietze_extension_closed_interval_real_from_normal_shrinker}.
    $F$ is a function and $\dom{F}=\carrier[X]$
        by \obv{funs_is_function,funs}.
    Hence $F$ is a function to $\realClosedInterval{a}{b}$.
    Hence
        $F\in\funs{\carrier[X]}{\realClosedInterval{a}{b}}$
        by \obv{funs_intro}.
    Hence $\ran{F}\subseteq\realClosedInterval{a}{b}$
        by \obv{funs_ran}.
    $\carrier[\realClosedIntervalTopology{a}{b}]
        =\realClosedInterval{a}{b}$
        by \obv{real_closed_interval_topology_carrier}.
    Hence
        $\ran{F}\subseteq\carrier[\realClosedIntervalTopology{a}{b}]$.
    Hence $F$ is continuous from $X$ to
        $\realClosedIntervalTopology{a}{b}$
        by \obv{continuous_codomain_corestriction}.
\end{proof}

\begin{theorem}\label{tietze_extension_unit_interval}
    Let $X$ be a normal space.
    Let $A\in\closeds{X}$.
    Suppose $f\in\funs{A}{\RealUnitInterval}$ and
        $f$ is continuous from $\subspaceTopology{X}{A}$ to
            $\RealUnitIntervalTopology$.
    Then there exists
        $F\in\funs{\carrier[X]}{\RealUnitInterval}$ such that
        $F$ is continuous from $X$ to $\RealUnitIntervalTopology$ and
        $\restrl{F}{A}=f$.
\end{theorem}
\begin{proof}
    $\RealUnitInterval=
        \realClosedInterval{\surrealZero}{\surrealOne}$
        by \obv{real_unit_interval_eq_closed_interval}.
    $\RealUnitIntervalTopology=
        \realClosedIntervalTopology{\surrealZero}{\surrealOne}$
        by
        \obv{real_unit_interval_topology,real_closed_interval_topology}.
    $\surrealZero,\surrealOne\in\reals$
        by \obv{surreal_zero_is_real,surreal_one_is_real}.
    $\surrealLe{\surrealZero}{\surrealOne}$
        by \obv{surreal_le,surreal_zero_lt_one}.
    Follows by \auto{tietze_extension_closed_interval}.
\end{proof}
