\section{Transport of topological manifolds}

\import{topology/manifold.tex}
\import{topology/embedding.tex}
\import{topology/separation-continuity.tex}
\import{topology/countability-continuity.tex}

A homeomorphism moves each Euclidean chart to the image of its domain.
The model dimension is unchanged; no atlas is chosen as part of the space.

\begin{theorem}\label{homeomorphism_preserves_locally_euclidean}
    Let $n$ be a natural number and $X,Y$ be topological spaces.
    Suppose $X$ is locally Euclidean of dimension $n$ and
        $f$ is a homeomorphism from $X$ to $Y$.
    Then $Y$ is locally Euclidean of dimension $n$.
\end{theorem}
\begin{proof}
    $f\in\bijections{\carrier[X]}{\carrier[Y]}$ by \obv{homeomorphism}.
    $f$ is a function and $\dom{f}=\carrier[X]$ by \obv{bijection_is_function,bijections_dom}.
    $f$ is an open map from $X$ to $Y$ by \obv{homeomorphism_is_open_map}.
    Show that for every $y\in\carrier[Y]$ there exists $W\in\opens[Y]$ such that
        $y\in W$ and there exists $V\in\opens[\RealCoordinateSpace{n}]$ such that
        there exists $k$ such that $k$ is a homeomorphism from $\subspaceTopology{Y}{W}$ to
            $\subspaceTopology{\RealCoordinateSpace{n}}{V}$.
    \begin{subproof}
        Fix $y\in\carrier[Y]$.
        Take $x\in\carrier[X]$ such that $f(x)=y$ by \auto{bijections,surj}.
        Take $U,V,h$ such that $U\in\opens[X]$ and $x\in U$ and
            $V\in\opens[\RealCoordinateSpace{n}]$ and $h$ is a homeomorphism from
            $\subspaceTopology{X}{U}$ to $\subspaceTopology{\RealCoordinateSpace{n}}{V}$ by
            \auto{locally_euclidean}.
        $U\subseteq\carrier[X]$ by \auto{opens_type,subseteq}.
        $\img{f}{U}\in\opens[Y]$ by \obv{open_map}.
        $y\in\img{f}{U}$ by \auto{img_iff,function_apply_elim}.
        $\restrl{f}{U}$ is a homeomorphism from $\subspaceTopology{X}{U}$ to
            $\subspaceTopology{Y}{\img{f}{U}}$ by \obv{homeomorphism_restriction_to_image}.
        Hence $\converse{\restrl{f}{U}}$ is a homeomorphism from
            $\subspaceTopology{Y}{\img{f}{U}}$ to $\subspaceTopology{X}{U}$ by
            \obv{homeomorphism_inverse}.
        Hence $h\circ\converse{\restrl{f}{U}}$ is a homeomorphism from
            $\subspaceTopology{Y}{\img{f}{U}}$ to $\subspaceTopology{\RealCoordinateSpace{n}}{V}$ by
            \obv{homeomorphism_composition}.
        Follows by \auto{homeomorphism}.
    \end{subproof}
    Follows by \obv{locally_euclidean}.
\end{proof}

\begin{theorem}\label{homeomorphism_preserves_topological_manifold}
    Let $n$ be a natural number and $X,Y$ be topological spaces.
    Suppose $X$ is a topological manifold of dimension $n$ and
        $f$ is a homeomorphism from $X$ to $Y$.
    Then $Y$ is a topological manifold of dimension $n$.
\end{theorem}
\begin{proof}
    $X$ is locally Euclidean of dimension $n$ and $X$ is Hausdorff and $X$ is second-countable by
        \obv{topological_manifold}.
    $Y$ is locally Euclidean of dimension $n$ by \obv{homeomorphism_preserves_locally_euclidean}.
    $Y$ is Hausdorff by \obv{homeomorphism_preserves_hausdorff}.
    $Y$ is second-countable by \obv{homeomorphism_preserves_second_countability}.
    Follows by \obv{topological_manifold}.
\end{proof}
