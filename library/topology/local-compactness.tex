\section{Local compactness}

\import{topology/compactness.tex}
\import{topology/subspace.tex}
\import{topology/basis.tex}

\begin{definition}\label{compact_in}
    Let $X$ be a topological space.
    $K$ is compact in $X$ iff
        $K\subseteq\carrier[X]$ and
        $\subspaceTopology{X}{K}$ is compact.
\end{definition}

\begin{theorem}\label{compact_space_carrier_is_compact}
    Let $X$ be a compact topological space.
    Then $\carrier[X]$ is compact in $X$.
\end{theorem}
\begin{proof}
    $\subspaceTopology{X}{\carrier[X]}$ is a topological space by
        \obv{subspace_topology_is_topological_space}.
    $\carrier[\subspaceTopology{X}{\carrier[X]}]=\carrier[X]$ by
        \auto{subspace_topology_carrier_of_subset,subseteq_refl}.
    $\opens[\subspaceTopology{X}{\carrier[X]}]=\opens[X]$ by
        \auto{subspace_topology_opens,subspace_opens_on_full_carrier}.
    Hence $\subspaceTopology{X}{\carrier[X]}$ is compact by
        \auto{is_compact,open_cover}.
    Follows by \auto{compact_in,subseteq_refl}.
\end{proof}

\begin{definition}\label{is_locally_compact}
    Let $X$ be a topological space.
    $X$ is locally compact iff
        for every $x\in\carrier[X]$ there exist $U,K$ such that
            $U\in\opens[X]$ and $x\in U\subseteq K$ and
            $K$ is compact in $X$.
\end{definition}

\begin{theorem}\label{compact_space_is_locally_compact}
    Let $X$ be a compact topological space.
    Then $X$ is locally compact.
\end{theorem}
\begin{proof}
    $\carrier[X]$ is compact in $X$ by \obv{compact_space_carrier_is_compact}.
    $\carrier[X]\in\opens[X]$ by \obv{carrier_open}.
    Follows by \auto{is_locally_compact,subseteq_refl}.
\end{proof}

\begin{theorem}\label{compact_open_basis_implies_locally_compact}
    Let $X$ be a topological space.
    Suppose $B$ is an open basis of $X$.
    Suppose for every $V\in B$ we have $V$ is compact in $X$.
    Then $X$ is locally compact.
\end{theorem}
\begin{proof}
    Show that for every $x\in\carrier[X]$ there exist $U,K$ such that
        $U\in\opens[X]$ and $x\in U\subseteq K$ and
        $K$ is compact in $X$.
    \begin{subproof}
        Fix $x\in\carrier[X]$.
        $\carrier[X]\in\opens[X]$ by \obv{carrier_open}.
        Take $V\in B$ such that $x\in V\subseteq\carrier[X]$
            by \obv{open_basis}.
        $B\subseteq\opens[X]$ by \obv{open_basis}.
        Hence $V\in\opens[X]$ by \obv{subseteq}.
        $V\subseteq V$ by \obv{subseteq_refl}.
        $V$ is compact in $X$.
        Follows by \auto.
    \end{subproof}
    Follows by \obv{is_locally_compact}.
\end{proof}
