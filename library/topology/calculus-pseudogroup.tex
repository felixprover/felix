\section{Coordinate changes from a local calculus}

\import{topology/local-map-calculus.tex}
\import{topology/continuous-calculus.tex}
\import{topology/full-pseudogroup.tex}

An invertible coordinate change must be admissible in both directions.
Its source and target are open parts of the models; neither direction is
extended outside its natural domain.

\begin{definition}\label{invertibly_admissible}
    Let $C$ be a local-map calculus.
    $h$ is invertibly admissible from $E$ to $F$ in $C$ iff
        $h$ is an open partial homeomorphism from $E$ to $F$ and
        $h$ is admissible from $E$ to $F$ in $C$ and
        $\converse{h}$ is admissible from $F$ to $E$ in $C$.
\end{definition}

\begin{theorem}\label{invertibly_admissible_converse}
    Let $C$ be a local-map calculus.
    Suppose $h$ is invertibly admissible from $E$ to $F$ in $C$.
    Then $\converse{h}$ is invertibly admissible from $F$ to $E$ in $C$.
\end{theorem}
\begin{proof}
    $h$ is an open partial homeomorphism from $E$ to $F$ and
        $h$ is admissible from $E$ to $F$ in $C$ and
        $\converse{h}$ is admissible from $F$ to $E$ in $C$ by
        \obv{invertibly_admissible}.
    $\converse{h}$ is an open partial homeomorphism from $F$ to $E$ by
        \obv{open_partial_homeomorphism_converse}.
    $h$ is a function by \auto{open_partial_homeomorphism_bijection,bijection_is_function}.
    $\converse{\converse{h}}=h$ by \obv{converse_converse_eq}.
    Follows by \obv{invertibly_admissible}.
\end{proof}

\begin{theorem}\label{invertibly_admissible_locality}
    Let $C$ be a local-map calculus.
    Let $E$ be a model of $C$.
    Let $F$ be a model of $C$.
    Suppose $h$ is an open partial homeomorphism from $E$ to $F$.
    Suppose $B\subseteq\opens[E]$.
    Suppose for every $x\in\dom{h}$ there exists $U\in B$ such that $x\in U$.
    Suppose for every $U\in B$ we have
        $\restrl{h}{U}$ is invertibly admissible from $E$ to $F$ in $C$.
    Then $h$ is invertibly admissible from $E$ to $F$ in $C$.
\end{theorem}
\begin{proof}
    $E,F\in\models[C]$ by \obv{calculus_model}.
    $h$ is a continuous partial map from $E$ to $F$ by
        \obv{open_partial_homeomorphism_is_continuous_partial_map}.
    $h\in\funs{\dom{h}}{\carrier[F]}$ by \obv{continuous_partial_map_type}.
    $\dom{h}\in\opens[E]$ by \obv{open_partial_homeomorphism}.
    For every $U\in B$ we have $\restrl{h}{U}$ is admissible from $E$ to $F$ in $C$
        by \auto{invertibly_admissible}.
    $h$ is admissible from $E$ to $F$ in $C$ by \obv{calculus_locality}.

    Let $k=\converse{h}$.
    $k$ is an open partial homeomorphism from $F$ to $E$ by
        \auto{open_partial_homeomorphism_converse}.
    $k$ is a continuous partial map from $F$ to $E$ by
        \obv{open_partial_homeomorphism_is_continuous_partial_map}.
    $k\in\funs{\dom{k}}{\carrier[E]}$ by \obv{continuous_partial_map_type}.
    $\dom{k}\in\opens[F]$ by \obv{open_partial_homeomorphism}.
    $h$ is an injection by
        \auto{open_partial_homeomorphism_bijection,bijections,bijection_is_function}.
    $\dom{k}=\ran{h}$ by \auto{dom_converse}.
    Show that for every $y\in\dom{k}$ there exists $V\in\opens[F]$ such that
        $y\in V$ and $\restrl{k}{V}$ is admissible from $F$ to $E$ in $C$.
    \begin{subproof}
        Fix $y\in\dom{k}$.
        Take $x\in\dom{h}$ such that $h(x)=y$ by \auto{fun_ran_iff}.
        Take $U\in B$ such that $x\in U$.
        $U\in\opens[E]$ by \obv{subseteq}.
        Let $V=\img{h}{U}$.
        $V\in\opens[F]$ by \auto{open_partial_homeomorphism_image_open}.
        $y\in V$ by \auto{img_iff,function_apply_elim}.
        $\converse{\restrl{h}{U}}$ is admissible from $F$ to $E$ in $C$ by
            \auto{invertibly_admissible}.
        $\converse{\restrl{h}{U}}=\restrl{k}{V}$ by \auto{restriction_converse}.
        Follows by \auto.
    \end{subproof}
    $k$ is admissible from $F$ to $E$ in $C$ by \obv{admissible_map_iff_locally}.
    Follows by \auto{invertibly_admissible}.
\end{proof}

\begin{theorem}\label{invertibly_admissible_composition}
    Let $C$ be a local-map calculus.
    Suppose $f$ is invertibly admissible from $E$ to $F$ in $C$.
    Suppose $g$ is invertibly admissible from $F$ to $H$ in $C$.
    Then $g\circ f$ is invertibly admissible from $E$ to $H$ in $C$.
\end{theorem}
\begin{proof}
    $f$ is an open partial homeomorphism from $E$ to $F$ and
        $g$ is an open partial homeomorphism from $F$ to $H$ by
        \obv{invertibly_admissible}.
    $f$ is admissible from $E$ to $F$ in $C$ and
        $g$ is admissible from $F$ to $H$ in $C$ and
        $\converse{f}$ is admissible from $F$ to $E$ in $C$ and
        $\converse{g}$ is admissible from $H$ to $F$ in $C$ by
        \obv{invertibly_admissible}.
    $g\circ f$ is an open partial homeomorphism from $E$ to $H$ by
        \obv{open_partial_homeomorphism_composition}.
    $g\circ f$ is admissible from $E$ to $H$ in $C$ and
        $\converse{f}\circ\converse{g}$ is admissible from $H$ to $E$ in $C$ by
        \obv{admissible_map_composition}.
    $\converse{g\circ f}=\converse{f}\circ\converse{g}$ by \obv{circ_converse}.
    Follows by \obv{invertibly_admissible}.
\end{proof}

\begin{theorem}\label{invertibly_admissible_identity}
    Let $C$ be a local-map calculus.
    Let $E$ be a model of $C$.
    Suppose $U\in\opens[E]$.
    Then $\identity{U}$ is invertibly admissible from $E$ to $E$ in $C$.
\end{theorem}
\begin{proof}
    $E\in\models[C]$ by \obv{calculus_model}.
    $\identity{U}$ is an open partial homeomorphism from $E$ to $E$ by
        \obv{identity_open_partial_homeomorphism}.
    $\identity{U}$ is admissible from $E$ to $E$ in $C$ by
        \obv{admissible_open_identity}.
    $\converse{\identity{U}}=\identity{U}$ by \obv{converse_identity}.
    Follows by \obv{invertibly_admissible}.
\end{proof}

\begin{theorem}\label{invertibly_admissible_restriction}
    Let $C$ be a local-map calculus.
    Suppose $h$ is invertibly admissible from $E$ to $F$ in $C$.
    Suppose $U\in\opens[E]$.
    Then $\restrl{h}{U}$ is invertibly admissible from $E$ to $F$ in $C$.
\end{theorem}
\begin{proof}
    $h$ is an open partial homeomorphism from $E$ to $F$ and
        $h$ is admissible from $E$ to $F$ in $C$ and
        $\converse{h}$ is admissible from $F$ to $E$ in $C$ by
        \obv{invertibly_admissible}.
    $\restrl{h}{U}$ is an open partial homeomorphism from $E$ to $F$ by
        \obv{open_partial_homeomorphism_restriction}.
    $\restrl{h}{U}$ is admissible from $E$ to $F$ in $C$ by
        \obv{admissible_map_restriction}.
    $\img{h}{U}\in\opens[F]$ by \obv{open_partial_homeomorphism_image_open}.
    $\restrl{\converse{h}}{\img{h}{U}}$ is admissible from $F$ to $E$ in $C$ by
        \obv{admissible_map_restriction}.
    $h$ is an injection by \auto{open_partial_homeomorphism_bijection,bijections,bijection_is_function}.
    $\converse{\restrl{h}{U}}=\restrl{\converse{h}}{\img{h}{U}}$ by
        \obv{restriction_converse}.
    Follows by \obv{invertibly_admissible}.
\end{proof}

\subsection{The associated pseudogroup}

\begin{definition}\label{invertible_changes}
    Let $C$ be a local-map calculus.
    $\invertibleChanges{C}{E}=
        \{h\in\openPartialHomeomorphisms{E}{E}\mid
            \text{$h$ is invertibly admissible from $E$ to $E$ in $C$}\}$.
\end{definition}

\begin{theorem}\label{invertible_changes_iff}
    Let $C$ be a local-map calculus.
    Then $h\in\invertibleChanges{C}{E}$ iff
        $h$ is invertibly admissible from $E$ to $E$ in $C$.
\end{theorem}
\begin{proof}
    Follows by \auto{invertible_changes,invertibly_admissible,
        open_partial_homeomorphisms_iff}.
\end{proof}

\begin{construction}\label{diffeomorphism_pseudogroup}
    Let $C$ be a local-map calculus.
    Let $E$ be a model of $C$.
    The coordinate pseudogroup $\DiffeomorphismPseudogroup{C}{E}=
        \structure{\model=E,\changes=\invertibleChanges{C}{E}}$.
\end{construction}
\begin{proof}
    Let $G=\DiffeomorphismPseudogroup{C}{E}$.
    $\model[G]=E$ and $\changes[G]=\invertibleChanges{C}{E}$ by
        \auto{coordinate_pseudogroupaggregateprojection_model,
            coordinate_pseudogroupaggregateprojection_changes}.
    $G$ is a record by \obv{coordinate_pseudogroupaggregateexact}.
    $\model[G]$ is a topological space.
    Every element of $\changes[G]$ is an open partial homeomorphism from
        $\model[G]$ to $\model[G]$ by
        \auto{invertible_changes_iff,invertibly_admissible}.
    $\carrier[E]\in\opens[E]$ by \obv{carrier_open}.
    $\identity{\carrier[E]}$ is invertibly admissible from $E$ to $E$ in $C$ by
        \obv{invertibly_admissible_identity}.
    $\identity{\carrier[\model[G]]}\in\changes[G]$ by
        \auto{invertible_changes_iff}.
    For every $h\in\changes[G]$ we have for every $U\in\opens[\model[G]]$
        we have $\restrl{h}{U}\in\changes[G]$ by
        \auto{invertible_changes_iff,invertibly_admissible_restriction}.
    For every $h\in\changes[G]$ we have $\converse{h}\in\changes[G]$ by
        \auto{invertible_changes_iff,invertibly_admissible_converse}.
    For all $f,g\in\changes[G]$ we have $g\circ f\in\changes[G]$ by
        \auto{invertible_changes_iff,invertibly_admissible_composition}.
    Show that for every $h$ such that $h$ is an open partial homeomorphism
        from $\model[G]$ to $\model[G]$ we have
        for every $B\subseteq\opens[\model[G]]$ such that
            (for every $x\in\dom{h}$ there exists $U\in B$ such that $x\in U$) and
            (for every $U\in B$ we have $\restrl{h}{U}\in\changes[G]$)
        we have $h\in\changes[G]$.
    \begin{subproof}
        Fix $h$ such that $h$ is an open partial homeomorphism
            from $\model[G]$ to $\model[G]$.
        Fix $B\subseteq\opens[\model[G]]$ such that
            (for every $x\in\dom{h}$ there exists $U\in B$ such that $x\in U$) and
            (for every $U\in B$ we have $\restrl{h}{U}\in\changes[G]$).
        For every $U\in B$ we have
            $\restrl{h}{U}$ is invertibly admissible from $E$ to $E$ in $C$ by
            \auto{invertible_changes_iff}.
        $h$ is invertibly admissible from $E$ to $E$ in $C$ by
            \auto{invertibly_admissible_locality}.
        Follows by \auto{invertible_changes_iff}.
    \end{subproof}
    $G$ is a coordinate pseudogroup by \auto{coordinate_pseudogroup}.
    Follows by \auto{coordinate_pseudogroup}.
\end{proof}

\begin{theorem}\label{diffeomorphism_pseudogroup_model}
    Let $C$ be a local-map calculus.
    Let $E$ be a model of $C$.
    Then $\model[\DiffeomorphismPseudogroup{C}{E}]=E$.
\end{theorem}
\begin{proof}
    Follows by \obv{diffeomorphism_pseudogroupprojection_model}.
\end{proof}

\begin{classification}\label{diffeomorphism_pseudogroup_on}
    Let $C$ be a local-map calculus.
    Let $E$ be a model of $C$.
    Then $\DiffeomorphismPseudogroup{C}{E}$ is a pseudogroup on $E$.
\end{classification}
\begin{proof}
    $\DiffeomorphismPseudogroup{C}{E}$ is a coordinate pseudogroup by
        \obv{diffeomorphism_pseudogroup}.
    $\model[\DiffeomorphismPseudogroup{C}{E}]=E$ by \obv{diffeomorphism_pseudogroup_model}.
    Follows by \obv{pseudogroup_on}.
\end{proof}

\begin{theorem}\label{diffeomorphism_pseudogroup_changes}
    Let $C$ be a local-map calculus.
    Let $E$ be a model of $C$.
    Then $h\in\changes[\DiffeomorphismPseudogroup{C}{E}]$ iff
        $h$ is invertibly admissible from $E$ to $E$ in $C$.
\end{theorem}
\begin{proof}
    $\changes[\DiffeomorphismPseudogroup{C}{E}]=\invertibleChanges{C}{E}$ by
        \obv{diffeomorphism_pseudogroupprojection_changes}.
    Follows by \auto{invertible_changes_iff}.
\end{proof}

\begin{theorem}\label{diffeomorphism_pseudogroup_is_exact}
    Let $C$ be a local-map calculus.
    Let $E$ be a model of $C$.
    Then $\DiffeomorphismPseudogroup{C}{E}$ is an exact coordinate pseudogroup.
\end{theorem}
\begin{proof}
    Follows by \auto{diffeomorphism_pseudogroup,diffeomorphism_pseudogroupshape}.
\end{proof}

\begin{theorem}\label{subcalculus_pseudogroup}
    Let $C$ be a local-map calculus.
    Let $D$ be a local-map calculus.
    Let $E$ be a model of $C$.
    Suppose $C$ is a subcalculus of $D$.
    Then $\DiffeomorphismPseudogroup{C}{E}$ is a subpseudogroup of
        $\DiffeomorphismPseudogroup{D}{E}$.
\end{theorem}
\begin{proof}
    $E$ is a model of $D$ by \obv{subcalculus_model}.
    Let $G=\DiffeomorphismPseudogroup{C}{E}$.
    Let $H=\DiffeomorphismPseudogroup{D}{E}$.
    $G$ is a coordinate pseudogroup and $H$ is a coordinate pseudogroup by
        \auto{diffeomorphism_pseudogroup}.
    $\model[G]=E$ and $\model[H]=E$ by \auto{diffeomorphism_pseudogroup_model}.
    Show that for every $h\in\changes[G]$ we have $h\in\changes[H]$.
    \begin{subproof}
        Fix $h\in\changes[G]$.
        $h$ is invertibly admissible from $E$ to $E$ in $C$ by
            \auto{diffeomorphism_pseudogroup_changes}.
        $h$ is an open partial homeomorphism from $E$ to $E$ and
            $h$ is admissible from $E$ to $E$ in $C$ and
            $\converse{h}$ is admissible from $E$ to $E$ in $C$ by
            \obv{invertibly_admissible}.
        $h$ is admissible from $E$ to $E$ in $D$ and
            $\converse{h}$ is admissible from $E$ to $E$ in $D$ by
            \obv{subcalculus_admissibility}.
        $h$ is invertibly admissible from $E$ to $E$ in $D$ by
            \obv{invertibly_admissible}.
        Follows by \auto{diffeomorphism_pseudogroup_changes}.
    \end{subproof}
    Follows by \auto{subpseudogroup,subseteq}.
\end{proof}

\begin{theorem}\label{continuous_calculus_pseudogroup}
    Let $M$ be a topological model family.
    Let $E\in M$.
    Then $\DiffeomorphismPseudogroup{\ContinuousCalculus{M}}{E}
        =\FullPseudogroup{E}$.
\end{theorem}
\begin{proof}
    Let $C=\ContinuousCalculus{M}$.
    $C$ is a local-map calculus by \auto{continuous_calculus}.
    $E$ is a model of $C$ by \auto{continuous_calculus_model}.
    $E$ is a topological space by \obv{topological_model_family}.
    Let $G=\DiffeomorphismPseudogroup{C}{E}$.
    Let $H=\FullPseudogroup{E}$.
    $G$ is an exact coordinate pseudogroup and $H$ is an exact coordinate pseudogroup by
        \auto{diffeomorphism_pseudogroup_is_exact,full_pseudogroup_is_exact}.
    $\model[G]=E$ and $\model[H]=E$ by
        \auto{diffeomorphism_pseudogroup_model,full_pseudogroup_model}.
    Show that for every $h$ we have $h\in\changes[G]$ iff $h\in\changes[H]$.
    \begin{subproof}
        Fix $h$.
        If $h\in\changes[G]$, then $h\in\changes[H]$ by
            \auto{diffeomorphism_pseudogroup_changes,invertibly_admissible,
                full_pseudogroup_changes}.
        Show that if $h\in\changes[H]$, then $h\in\changes[G]$.
        \begin{subproof}
            Assume $h\in\changes[H]$.
            $h$ is an open partial homeomorphism from $E$ to $E$ by
                \auto{full_pseudogroup_changes}.
            $\converse{h}$ is an open partial homeomorphism from $E$ to $E$ by
                \obv{open_partial_homeomorphism_converse}.
            $h$ is a continuous partial map from $E$ to $E$ and
                $\converse{h}$ is a continuous partial map from $E$ to $E$ by
                \obv{open_partial_homeomorphism_is_continuous_partial_map}.
            $h$ is admissible from $E$ to $E$ in $C$ and
                $\converse{h}$ is admissible from $E$ to $E$ in $C$ by
                \auto{continuous_calculus_admissible_iff}.
            $h$ is invertibly admissible from $E$ to $E$ in $C$ by
                \obv{invertibly_admissible}.
            Follows by \auto{diffeomorphism_pseudogroup_changes}.
        \end{subproof}
        Follows by \auto.
    \end{subproof}
    $\changes[G]=\changes[H]$ by \obv{setext}.
    $G=H$ by \obv{exact_coordinate_pseudogroup_extensionality}.
\end{proof}

\subsection{Calculus-compatible coordinate changes}

A coordinate pseudogroup may impose additional restrictions on admissible
changes. Compatibility asks only that its model belongs to the calculus and
that its changes are admissible. Closure under converse then handles inverses.

\begin{definition}\label{pseudogroup_calculus_compatible}
    Let $C$ be a local-map calculus.
    Let $G$ be a coordinate pseudogroup.
    $G$ is compatible with the calculus $C$ iff
        $\model[G]$ is a model of $C$ and
        for every $h\in\changes[G]$ we have
            $h$ is admissible from $\model[G]$ to $\model[G]$ in $C$.
\end{definition}

\begin{classification}\label{compatible_pseudogroup_model}
    Let $C$ be a local-map calculus.
    Let $G$ be a coordinate pseudogroup.
    Suppose $G$ is compatible with the calculus $C$.
    Then $\model[G]$ is a model of $C$.
\end{classification}
\begin{proof}
    Follows by \obv{pseudogroup_calculus_compatible}.
\end{proof}

\begin{theorem}\label{compatible_pseudogroup_changes}
    Let $C$ be a local-map calculus.
    Let $G$ be a coordinate pseudogroup.
    Suppose $G$ is compatible with the calculus $C$.
    Let $h\in\changes[G]$.
    Then $h$ is invertibly admissible from $\model[G]$ to $\model[G]$ in $C$.
\end{theorem}
\begin{proof}
    $h$ is an open partial homeomorphism from $\model[G]$ to $\model[G]$ by
        \obv{pseudogroup_changes}.
    $\converse{h}\in\changes[G]$ by \obv{pseudogroup_converse}.
    $h$ is admissible from $\model[G]$ to $\model[G]$ in $C$ and
        $\converse{h}$ is admissible from $\model[G]$ to $\model[G]$ in $C$ by
        \auto{pseudogroup_calculus_compatible}.
    Follows by \obv{invertibly_admissible}.
\end{proof}

\begin{classification}\label{diffeomorphism_pseudogroup_compatible}
    Let $C$ be a local-map calculus.
    Let $E$ be a model of $C$.
    Then $\DiffeomorphismPseudogroup{C}{E}$ is compatible with the calculus $C$.
\end{classification}
\begin{proof}
    $\DiffeomorphismPseudogroup{C}{E}$ is a coordinate pseudogroup by
        \obv{diffeomorphism_pseudogroup}.
    $\model[\DiffeomorphismPseudogroup{C}{E}]=E$ by
        \obv{diffeomorphism_pseudogroup_model}.
    For every $h\in\changes[\DiffeomorphismPseudogroup{C}{E}]$ we have
        $h$ is admissible from $E$ to $E$ in $C$ by
        \auto{diffeomorphism_pseudogroup_changes,invertibly_admissible}.
    Follows by \auto{pseudogroup_calculus_compatible}.
\end{proof}

\begin{classification}\label{pseudogroup_continuous_calculus_compatible}
    Let $P$ be a topological model family.
    Let $G$ be a coordinate pseudogroup.
    Suppose $\model[G]\in P$.
    Then $G$ is compatible with the calculus $\ContinuousCalculus{P}$.
\end{classification}
\begin{proof}
    $\ContinuousCalculus{P}$ is a local-map calculus by \obv{continuous_calculus}.
    $\model[G]$ is a model of $\ContinuousCalculus{P}$ by \obv{continuous_calculus_model}.
    For every $h\in\changes[G]$ we have
        $h$ is a continuous partial map from $\model[G]$ to $\model[G]$ by
        \auto{pseudogroup_changes,open_partial_homeomorphism_is_continuous_partial_map}.
    Follows by \auto{pseudogroup_calculus_compatible,continuous_calculus_admissible_iff}.
\end{proof}

\begin{classification}\label{subcalculus_pseudogroup_compatible}
    Let $C$ be a local-map calculus.
    Let $D$ be a local-map calculus.
    Let $G$ be a coordinate pseudogroup.
    Suppose $C$ is a subcalculus of $D$.
    Suppose $G$ is compatible with the calculus $C$.
    Then $G$ is compatible with the calculus $D$.
\end{classification}
\begin{proof}
    $\model[G]$ is a model of $C$ by \obv{compatible_pseudogroup_model}.
    $\model[G]$ is a model of $D$ by \obv{subcalculus_model}.
    For every $h\in\changes[G]$ we have
        $h$ is admissible from $\model[G]$ to $\model[G]$ in $C$ by
        \obv{pseudogroup_calculus_compatible}.
    For every $h\in\changes[G]$ we have
        $h$ is admissible from $\model[G]$ to $\model[G]$ in $D$ by
        \obv{subcalculus_admissibility}.
    Follows by \obv{pseudogroup_calculus_compatible}.
\end{proof}

\begin{classification}\label{subpseudogroup_calculus_compatible}
    Let $C$ be a local-map calculus.
    Let $G,H$ be coordinate pseudogroups.
    Suppose $H$ is compatible with the calculus $C$.
    Suppose $G$ is a subpseudogroup of $H$.
    Then $G$ is compatible with the calculus $C$.
\end{classification}
\begin{proof}
    $\model[G]=\model[H]$ and $\changes[G]\subseteq\changes[H]$ by \obv{subpseudogroup}.
    $\model[H]$ is a model of $C$ by \obv{compatible_pseudogroup_model}.
    For every $h\in\changes[G]$ we have
        $h$ is admissible from $\model[G]$ to $\model[G]$ in $C$ by
        \auto{pseudogroup_calculus_compatible,subseteq}.
    Follows by \auto{pseudogroup_calculus_compatible}.
\end{proof}
