Separation holds for every index set, and finite coordinate spaces are locally
compact and second-countable. Dimension zero is the one-point space consisting
of the empty function; its compactness and connectedness do not involve
coordinate evaluation.

\begin{theorem}\label{real_coordinate_space_is_hausdorff}
    $\RealCoordinateSpace{I}$ is Hausdorff.
\end{theorem}
\begin{proof}
    $\RealTopology$ is Hausdorff by \obv{real_topology_is_hausdorff}.
    Follows by \auto{real_coordinate_space,topological_power_is_hausdorff}.
\end{proof}

\begin{theorem}\label{real_coordinate_space_is_locally_compact}
    Suppose $I$ is finite.
    Then $\RealCoordinateSpace{I}$ is locally compact.
\end{theorem}
\begin{proof}
    $\RealTopology$ is a topological space by \obv{real_topology_is_topological_space}.
    $\RealTopology$ is locally compact by \obv{real_topology_is_locally_compact}.
    Follows by \auto{real_coordinate_space,finite_topological_power_is_locally_compact}.
\end{proof}
