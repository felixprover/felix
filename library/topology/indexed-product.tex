\section{Indexed product topologies}

\import{topology/initial.tex}
\import{topology/indexed-family.tex}

The carrier of an indexed topological product is the existing set-theoretic
dependent product of the carrier family.  Its topology is initial with
respect to the existing dependent coordinate projections.

\subsection{The projection family}

\begin{definition}\label{indexed_product_projection_op}
    $\indexedProductProjectionOp:\Set\to\Set\to\Set\to\Set
        =\fun{I,X,i:\Set}{
            \dependentProjection{I}{
                \indexedTopologicalCarrierFamily{I}{X}}{i}}$.
\end{definition}

\begin{definition}\label{indexed_product_projection_family}
    $\indexedProductProjectionFamily{I}{X}
        =\Apply{\graphOnDomainOp}{
            I,\Apply{\indexedProductProjectionOp}{I,X}}$.
\end{definition}

\begin{proposition}\label{indexed_product_projection_family_type}
    $\indexedProductProjectionFamily{I}{X}$ is a function on $I$.
\end{proposition}
\begin{proof}
    $\indexedProductProjectionFamily{I}{X}\in
        \funs{I}{\ran{\indexedProductProjectionFamily{I}{X}}}$ by
        \auto{indexed_product_projection_family,
            logical_graph_on_domain_function}.
    Follows by \obv{funs_is_function,funs}.
\end{proof}

\begin{proposition}\label{indexed_product_projection_family_apply}
    Let $i\in I$.
    Then $(\indexedProductProjectionFamily{I}{X})(i)
        =\dependentProjection{I}{
            \indexedTopologicalCarrierFamily{I}{X}}{i}$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{indexed_product_projection_family,
            indexed_product_projection_op,
            logical_graph_on_domain_apply}.
\end{proof}

\subsection{The product topology}

\begin{definition}\label{indexed_product_topology}
    $\IndexedProductTopology{I}{X}
        =\InitialTopology{
            \dependentProduct{I}{
                \indexedTopologicalCarrierFamily{I}{X}}}{I}{X}{
            \indexedProductProjectionFamily{I}{X}}$.
\end{definition}

\begin{proposition}\label{indexed_product_initial_data}
    Suppose $X$ is a function on $I$.
    Suppose for every $i\in I$ we have $X(i)$ is a topological space.
    Then $\indexedProductProjectionFamily{I}{X}$ is an initial mapping
        family from
        $\dependentProduct{I}{\indexedTopologicalCarrierFamily{I}{X}}$
        to $X$ indexed by $I$.
\end{proposition}
\begin{proof}
    $\indexedProductProjectionFamily{I}{X}$ is a function on $I$ by
        \obv{indexed_product_projection_family_type}.
    $\indexedTopologicalCarrierFamily{I}{X}$ is a function on $I$ by
        \obv{indexed_topological_carrier_family_type}.
    Show that for every $i\in I$ we have $X(i)$ is a topological space
        and $(\indexedProductProjectionFamily{I}{X})(i)\in
            \funs{
                \dependentProduct{I}{
                    \indexedTopologicalCarrierFamily{I}{X}}}{
                \carrier[X(i)]}$.
    \begin{subproof}
        Fix $i\in I$.
        $(\indexedProductProjectionFamily{I}{X})(i)
            =\dependentProjection{I}{
                \indexedTopologicalCarrierFamily{I}{X}}{i}$ by
            \obv{indexed_product_projection_family_apply}.
        $\dependentProjection{I}{
            \indexedTopologicalCarrierFamily{I}{X}}{i}\in
            \funs{
                \dependentProduct{I}{
                    \indexedTopologicalCarrierFamily{I}{X}}}{
                (\indexedTopologicalCarrierFamily{I}{X})(i)}$ by
            \obv{dependent_projection_type}.
        $(\indexedTopologicalCarrierFamily{I}{X})(i)
            =\carrier[X(i)]$ by
            \obv{indexed_topological_carrier_family_apply}.
    \end{subproof}
    Follows by \obv{initial_topology_data}.
\end{proof}

\begin{classification}\label{indexed_product_topology_is_topological_space}
    $\IndexedProductTopology{I}{X}$ is a topological space.
\end{classification}
\begin{proof}
    Follows by
        \auto{indexed_product_topology,
            initial_topology_is_topological_space}.
\end{proof}

\begin{theorem}\label{indexed_product_topology_is_exact}
    $\IndexedProductTopology{I}{X}$ is an exact topological space.
\end{theorem}
\begin{proof}
    Follows by
        \auto{indexed_product_topology,initial_topology_is_exact}.
\end{proof}

\begin{proposition}\label{indexed_product_topology_carrier}
    $\carrier[\IndexedProductTopology{I}{X}]
        =\dependentProduct{I}{
            \indexedTopologicalCarrierFamily{I}{X}}$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{indexed_product_topology,initial_topology_carrier}.
\end{proof}

\begin{theorem}\label{indexed_product_coordinate_continuous}
    Suppose $X$ is a function on $I$.
    Suppose for every $j\in I$ we have $X(j)$ is a topological space.
    Let $i\in I$.
    Then $\dependentProjection{I}{
        \indexedTopologicalCarrierFamily{I}{X}}{i}$ is continuous from
        $\IndexedProductTopology{I}{X}$ to $X(i)$.
\end{theorem}
\begin{proof}
    $\indexedProductProjectionFamily{I}{X}$ is an initial mapping family
        from
        $\dependentProduct{I}{\indexedTopologicalCarrierFamily{I}{X}}$
        to $X$ indexed by $I$ by \obv{indexed_product_initial_data}.
    Hence $(\indexedProductProjectionFamily{I}{X})(i)$ is continuous
        from
        $\InitialTopology{
            \dependentProduct{I}{
                \indexedTopologicalCarrierFamily{I}{X}}}{I}{X}{
            \indexedProductProjectionFamily{I}{X}}$ to $X(i)$ by
        \obv{initial_topology_projection_continuous}.
    $(\indexedProductProjectionFamily{I}{X})(i)
        =\dependentProjection{I}{
            \indexedTopologicalCarrierFamily{I}{X}}{i}$ by
        \obv{indexed_product_projection_family_apply}.
    $\IndexedProductTopology{I}{X}
        =\InitialTopology{
            \dependentProduct{I}{
                \indexedTopologicalCarrierFamily{I}{X}}}{I}{X}{
            \indexedProductProjectionFamily{I}{X}}$ by
        \obv{indexed_product_topology}.
\end{proof}

\begin{theorem}\label{indexed_product_topology_universal_property}
    Suppose $X$ is a function on $I$.
    Suppose for every $i\in I$ we have $X(i)$ is a topological space.
    Let $Z$ be a topological space.
    Let $g\in\funs{\carrier[Z]}{
        \dependentProduct{I}{\indexedTopologicalCarrierFamily{I}{X}}}$.
    Then $g$ is continuous from $Z$ to $\IndexedProductTopology{I}{X}$
        iff for every $i\in I$ we have
            $\dependentProjection{I}{
                \indexedTopologicalCarrierFamily{I}{X}}{i}\circ g$
            is continuous from $Z$ to $X(i)$.
\end{theorem}
\begin{proof}
    $\indexedProductProjectionFamily{I}{X}$ is an initial mapping family
        from
        $\dependentProduct{I}{\indexedTopologicalCarrierFamily{I}{X}}$
        to $X$ indexed by $I$ by \obv{indexed_product_initial_data}.
    Hence $g$ is continuous from $Z$ to
        $\InitialTopology{
            \dependentProduct{I}{
                \indexedTopologicalCarrierFamily{I}{X}}}{I}{X}{
            \indexedProductProjectionFamily{I}{X}}$ iff
        for every $i\in I$ we have
            $(\indexedProductProjectionFamily{I}{X})(i)\circ g$
                is continuous from $Z$ to $X(i)$ by
        \obv{initial_topology_universal_property}.
    For every $i\in I$ we have
        $(\indexedProductProjectionFamily{I}{X})(i)
        =\dependentProjection{I}{
            \indexedTopologicalCarrierFamily{I}{X}}{i}$ by
        \obv{indexed_product_projection_family_apply}.
    $\IndexedProductTopology{I}{X}
        =\InitialTopology{
            \dependentProduct{I}{
                \indexedTopologicalCarrierFamily{I}{X}}}{I}{X}{
            \indexedProductProjectionFamily{I}{X}}$ by
        \obv{indexed_product_topology}.
    Follows by \auto.
\end{proof}

\subsection{Mediators}

\begin{theorem}\label{indexed_product_mediator_coordinate}
    Suppose $A$ is a function on $I$.
    Suppose $f$ is a function on $I$.
    Suppose for every $j\in I$ we have $f(j)\in\funs{D}{A(j)}$.
    Let $i\in I$.
    Then $\dependentProjection{I}{A}{i}\circ
        \dependentMediator{D}{I}{A}{f}=f(i)$.
\end{theorem}
\begin{proof}
    $f(i)\in\funs{D}{A(i)}$.
    $\dependentProjection{I}{A}{i}\in
        \funs{\dependentProduct{I}{A}}{A(i)}$ by
        \obv{dependent_projection_type}.
    $\dependentMediator{D}{I}{A}{f}\in
        \funs{D}{\dependentProduct{I}{A}}$ by
        \obv{dependent_mediator_type}.
    Hence $\dependentProjection{I}{A}{i}\circ
        \dependentMediator{D}{I}{A}{f}\in\funs{D}{A(i)}$ by
        \obv{funs_circ}.
    Show that for every $x\in D$ we have
        $(\dependentProjection{I}{A}{i}\circ
            \dependentMediator{D}{I}{A}{f})(x)=\apply{f(i)}{x}$.
    \begin{subproof}
        Fix $x\in D$.
        $\dependentMediator{D}{I}{A}{f}$ is a function and
            $\dependentProjection{I}{A}{i}$ is a function by
            \obv{funs_is_function}.
        $\dependentProjection{I}{A}{i}$ is composable with
            $\dependentMediator{D}{I}{A}{f}$ by
            \auto{funs_ran,funs}.
        $x\in\dom{\dependentMediator{D}{I}{A}{f}}$ by \auto{funs}.
        Hence $(\dependentProjection{I}{A}{i}\circ
            \dependentMediator{D}{I}{A}{f})(x)
            =(\dependentProjection{I}{A}{i})
                ((\dependentMediator{D}{I}{A}{f})(x))$ by
            \obv{circ_apply}.
        $(\dependentMediator{D}{I}{A}{f})(x)
            \in\dependentProduct{I}{A}$ by \obv{funs_type_apply}.
        Hence $(\dependentProjection{I}{A}{i})
            ((\dependentMediator{D}{I}{A}{f})(x))
            =\apply{\apply{\dependentMediator{D}{I}{A}{f}}{x}}{i}$ by
            \obv{dependent_projection_apply}.
        $\apply{\apply{\dependentMediator{D}{I}{A}{f}}{x}}{i}
            =\apply{f(i)}{x}$ by
            \obv{dependent_mediator_coordinate}.
    \end{subproof}
    Follows by \obv{typed_functions_equal_from_agreement}.
\end{proof}

\begin{theorem}\label{continuous_indexed_product_mediator}
    Suppose $X$ is a function on $I$.
    Suppose for every $i\in I$ we have $X(i)$ is a topological space.
    Let $Z$ be a topological space.
    Suppose $f$ is a function on $I$.
    Suppose for every $i\in I$ we have
        $f(i)$ is continuous from $Z$ to $X(i)$.
    Then $\dependentMediator{\carrier[Z]}{I}{
        \indexedTopologicalCarrierFamily{I}{X}}{f}$ is continuous from
        $Z$ to $\IndexedProductTopology{I}{X}$.
\end{theorem}
\begin{proof}
    $\indexedTopologicalCarrierFamily{I}{X}$ is a function on $I$ by
        \obv{indexed_topological_carrier_family_type}.
    For every $i\in I$ we have
        $f(i)\in\funs{\carrier[Z]}{
            (\indexedTopologicalCarrierFamily{I}{X})(i)}$ by
        \auto{continuous_map,indexed_topological_carrier_family_apply}.
    Hence $\dependentMediator{\carrier[Z]}{I}{
        \indexedTopologicalCarrierFamily{I}{X}}{f}\in
        \funs{\carrier[Z]}{
            \dependentProduct{I}{
                \indexedTopologicalCarrierFamily{I}{X}}}$ by
        \obv{dependent_mediator_type}.
    For every $i\in I$ we have
        $\dependentProjection{I}{
            \indexedTopologicalCarrierFamily{I}{X}}{i}\circ
        \dependentMediator{\carrier[Z]}{I}{
            \indexedTopologicalCarrierFamily{I}{X}}{f}=f(i)$ by
        \obv{indexed_product_mediator_coordinate}.
    Hence for every $i\in I$ we have
        $\dependentProjection{I}{
            \indexedTopologicalCarrierFamily{I}{X}}{i}\circ
        \dependentMediator{\carrier[Z]}{I}{
            \indexedTopologicalCarrierFamily{I}{X}}{f}$ is continuous
        from $Z$ to $X(i)$ by \auto.
    Follows by \obv{indexed_product_topology_universal_property}.
\end{proof}
