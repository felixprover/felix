\section{Coordinate pseudogroups}

\import{topology/partial-homeomorphism.tex}

A coordinate pseudogroup specifies allowable changes of coordinates on one
model. Locality tests an already bijective partial homeomorphism; it does not
claim that locally injective maps can be glued to a globally injective map.

\begin{struct}\label{coordinate_pseudogroup}
    A coordinate pseudogroup $G$ is a record equipped with
    \begin{enumerate}
        \item a model $\model[G]$;
        \item a set $\changes[G]$;
    \end{enumerate}
    such that
    \begin{enumerate}
        \item\label{pseudogroup_model} $\model[G]$ is a topological space.
        \item\label{pseudogroup_changes}
            every element of $\changes[G]$ is an open partial homeomorphism
                from $\model[G]$ to $\model[G]$.
        \item\label{pseudogroup_identity}
            $\identity{\carrier[\model[G]]}\in\changes[G]$.
        \item\label{pseudogroup_restriction}
            for every $h\in\changes[G]$ we have for every $U\in\opens[\model[G]]$
                we have $\restrl{h}{U}\in\changes[G]$.
        \item\label{pseudogroup_converse}
            for every $h\in\changes[G]$ we have $\converse{h}\in\changes[G]$.
        \item\label{pseudogroup_composition}
            for all $f,g\in\changes[G]$ we have $g\circ f\in\changes[G]$.
        \item\label{pseudogroup_locality}
            for every $h$ such that $h$ is an open partial homeomorphism
                from $\model[G]$ to $\model[G]$ we have
                for every $B\subseteq\opens[\model[G]]$ such that
                    (for every $x\in\dom{h}$ there exists $U\in B$ such that
                        $x\in U$) and
                    (for every $U\in B$ we have $\restrl{h}{U}\in\changes[G]$)
                we have $h\in\changes[G]$.
    \end{enumerate}
\end{struct}

\begin{definition}\label{pseudogroup_on}
    Let $E$ be a topological space.
    $G$ is a pseudogroup on $E$ iff
        $G$ is a coordinate pseudogroup and $\model[G]=E$.
\end{definition}

\begin{theorem}\label{pseudogroup_open_identity}
    Let $G$ be a coordinate pseudogroup.
    Suppose $U\in\opens[\model[G]]$.
    Then $\identity{U}\in\changes[G]$.
\end{theorem}
\begin{proof}
    $\identity{\carrier[\model[G]]}\in\changes[G]$ by \obv{pseudogroup_identity}.
    $\restrl{\identity{\carrier[\model[G]]}}{U}\in\changes[G]$ by
        \obv{pseudogroup_restriction}.
    $U\subseteq\carrier[\model[G]]$ by \obv{opens_type}.
    $\carrier[\model[G]]\inter U=U$ by \obv{inter_absorb_supseteq_left}.
    $\restrl{\identity{\carrier[\model[G]]}}{U}
        =\identity{\carrier[\model[G]]\inter U}$ by \obv{identity_restriction}.
\end{proof}

\begin{classification}\label{pseudogroup_model_is_topological}
    Let $G$ be a coordinate pseudogroup.
    Then $\model[G]$ is a topological space.
\end{classification}
\begin{proof}
    Follows by \obv{pseudogroup_model}.
\end{proof}

\begin{theorem}\label{pseudogroup_change_iff_locally}
    Let $G$ be a coordinate pseudogroup.
    Suppose $h$ is an open partial homeomorphism from $\model[G]$ to $\model[G]$.
    Then $h\in\changes[G]$ iff
        for every $x\in\dom{h}$ there exists $U\in\opens[\model[G]]$ such that
            $x\in U$ and $\restrl{h}{U}\in\changes[G]$.
\end{theorem}
\begin{proof}
    $\dom{h}\in\opens[\model[G]]$ by \obv{open_partial_homeomorphism}.
    $h$ is a function by
        \auto{open_partial_homeomorphism_bijection,bijection_is_function}.
    $\restrl{h}{\dom{h}}=h$ by \obv{restrl_by_dom}.
    If $h\in\changes[G]$, then
        for every $x\in\dom{h}$ there exists $U\in\opens[\model[G]]$ such that
            $x\in U$ and $\restrl{h}{U}\in\changes[G]$ by \auto.
    Show that if for every $x\in\dom{h}$ there exists $U\in\opens[\model[G]]$
        such that $x\in U$ and $\restrl{h}{U}\in\changes[G]$,
        then $h\in\changes[G]$.
    \begin{subproof}
        Assume for every $x\in\dom{h}$ there exists $U\in\opens[\model[G]]$
            such that $x\in U$ and $\restrl{h}{U}\in\changes[G]$.
        Let $B=\{U\in\opens[\model[G]]\mid\restrl{h}{U}\in\changes[G]\}$.
        $B\subseteq\opens[\model[G]]$ by \obv{subseteq}.
        For every $x\in\dom{h}$ there exists $U\in B$ such that $x\in U$ by \auto.
        For every $U\in B$ we have $\restrl{h}{U}\in\changes[G]$.
        Follows by \obv{pseudogroup_locality}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{theorem}\label{pseudogroup_empty_change}
    Let $G$ be a coordinate pseudogroup.
    Then $\emptyset\in\changes[G]$.
\end{theorem}
\begin{proof}
    $\emptyset$ is an open partial homeomorphism from $\model[G]$ to $\model[G]$
        by \obv{empty_open_partial_homeomorphism}.
    $\dom{\emptyset}=\emptyset$ by \obv{dom_emptyset}.
    For every $x\in\dom{\emptyset}$ there exists $U\in\opens[\model[G]]$
        such that $x\in U$ and $\restrl{\emptyset}{U}\in\changes[G]$
        by \auto.
    Follows by \auto{pseudogroup_change_iff_locally}.
\end{proof}

\subsection{Inclusion and equality}

\begin{definition}\label{subpseudogroup}
    Let $G,H$ be coordinate pseudogroups.
    $G$ is a subpseudogroup of $H$ iff
        $\model[G]=\model[H]$ and $\changes[G]\subseteq\changes[H]$.
\end{definition}

\begin{classification}\label{subpseudogroup_reflexive}
    Let $G$ be a coordinate pseudogroup.
    Then $G$ is a subpseudogroup of $G$.
\end{classification}
\begin{proof}
    Follows by \obv{subpseudogroup,subseteq_refl}.
\end{proof}

\begin{classification}\label{subpseudogroup_transitive}
    Let $G,H,K$ be coordinate pseudogroups.
    Suppose $G$ is a subpseudogroup of $H$ and $H$ is a subpseudogroup of $K$.
    Then $G$ is a subpseudogroup of $K$.
\end{classification}
\begin{proof}
    Follows by \auto{subpseudogroup,subseteq_transitive}.
\end{proof}

\begin{theorem}\label{exact_coordinate_pseudogroup_extensionality}
    Suppose $G$ is an exact coordinate pseudogroup and
        $H$ is an exact coordinate pseudogroup.
    Suppose $\model[G]=\model[H]$ and $\changes[G]=\changes[H]$.
    Then $G=H$.
\end{theorem}
\begin{proof}
    Follows by structure extensionality.
\end{proof}
