\section{Coordinates on open subspaces}

\import{topology/coordinate-atlas-restriction.tex}
\import{topology/coordinate-map.tex}

The relative structure on an open subspace retains precisely the ambient
charts whose domains lie in that subspace. Its defining atlas is already
maximal, so the construction requires no further maximalization.

\begin{construction}\label{restricted_coordinates}
    Let $G$ be a coordinate pseudogroup.
    Let $X$ be a topological space.
    Let $S$ be a relative coordinate structure on $X$ under $G$.
    Suppose $U$ is open in $X$.
    The relative coordinate structure $\RestrictedCoordinates{G}{X}{S}{U}=
        \structure{\atlas=\RestrictedAtlas{\atlas[S]}{U}}$
        on $\subspaceTopology{X}{U}$ under $G$.
\end{construction}
\begin{proof}
    $U\in\opens[X]$ by \obv{is_open}.
    $\atlas[S]$ is a maximal coordinate atlas on $X$ under $G$ by \obv{relative_atlas_maximal}.
    $\RestrictedAtlas{\atlas[S]}{U}$ is a maximal coordinate atlas on $\subspaceTopology{X}{U}$ under $G$ by
        \obv{restricted_maximal_atlas_is_maximal}.
    Follows by \auto{relative_coordinate_structureaggregateprojection_atlas}.
\end{proof}

\begin{theorem}\label{restricted_coordinates_is_exact}
    Let $G$ be a coordinate pseudogroup and $X$ be a topological space.
    Let $S$ be a relative coordinate structure on $X$ under $G$.
    Let $U\in\opens[X]$.
    Then $\RestrictedCoordinates{G}{X}{S}{U}$ is an exact relative coordinate structure
        on $\subspaceTopology{X}{U}$ under $G$.
\end{theorem}
\begin{proof}
    $U$ is open in $X$ by \obv{is_open}.
    Follows by \auto{restricted_coordinates,restricted_coordinatesshape}.
\end{proof}

\begin{theorem}\label{restricted_coordinates_chart_iff}
    Let $G$ be a coordinate pseudogroup and $X$ be a topological space.
    Let $S$ be a relative coordinate structure on $X$ under $G$.
    Let $U\in\opens[X]$.
    Then $c\in\atlas[\RestrictedCoordinates{G}{X}{S}{U}]$ iff
        $c\in\atlas[S]$ and $\dom{c}\subseteq U$.
\end{theorem}
\begin{proof}
    $U$ is open in $X$ by \obv{is_open}.
    $\atlas[\RestrictedCoordinates{G}{X}{S}{U}]=\RestrictedAtlas{\atlas[S]}{U}$ by
        \obv{restricted_coordinatesprojection_atlas}.
    $\atlas[S]$ is a maximal coordinate atlas on $X$ under $G$ by \obv{relative_atlas_maximal}.
    Follows by \auto{restricted_maximal_atlas_iff}.
\end{proof}

\subsection{Restriction through presentations}

Any presenting atlas computes the restricted structure. Restricting a
presentation and then generating coordinates gives the same record as
restricting the original maximal atlas.

\begin{theorem}\label{restricted_coordinates_presentation}
    Let $G$ be a coordinate pseudogroup and $X$ be a topological space.
    Let $S$ be a coordinate datum.
    Suppose $A$ presents $S$ on $X$ under $G$.
    Suppose $U$ is open in $X$.
    Then $\RestrictedAtlas{A}{U}$ presents $\RestrictedCoordinates{G}{X}{S}{U}$
        on $\subspaceTopology{X}{U}$ under $G$.
\end{theorem}
\begin{proof}
    $S$ is a relative coordinate structure on $X$ under $G$ by \obv{presented_coordinates}.
    $U\in\opens[X]$ by \obv{is_open}.
    $A$ is a coordinate atlas on $X$ under $G$ and $\atlas[S]=\MaxAtlas{G}{X}{A}$ by
        \obv{coordinate_presentation}.
    $\RestrictedAtlas{A}{U}$ is a coordinate atlas on $\subspaceTopology{X}{U}$ under $G$ by
        \obv{restricted_atlas_is_atlas}.
    $\RestrictedCoordinates{G}{X}{S}{U}$ is a coordinate datum by
        \auto{restricted_coordinates,relative_coordinate_structure}.
    $\atlas[\RestrictedCoordinates{G}{X}{S}{U}]=\RestrictedAtlas{\atlas[S]}{U}$ by
        \obv{restricted_coordinatesprojection_atlas}.
    $\MaxAtlas{G}{\subspaceTopology{X}{U}}{\RestrictedAtlas{A}{U}}
        =\RestrictedAtlas{\atlas[S]}{U}$ by \auto{restricted_atlas_maximalization}.
    Follows by \obv{coordinate_presentation}.
\end{proof}

\begin{theorem}\label{restricted_coordinates_from_presentation}
    Let $G$ be a coordinate pseudogroup and $X$ be a topological space.
    Let $S$ be a coordinate datum.
    Suppose $A$ presents $S$ on $X$ under $G$.
    Suppose $U$ is open in $X$.
    Then $\RestrictedCoordinates{G}{X}{S}{U}
        =\GeneratedCoordinates{G}{\subspaceTopology{X}{U}}{\RestrictedAtlas{A}{U}}$.
\end{theorem}
\begin{proof}
    $S$ is a relative coordinate structure on $X$ under $G$ by \obv{presented_coordinates}.
    $U\in\opens[X]$ by \obv{is_open}.
    $\RestrictedCoordinates{G}{X}{S}{U}$ is an exact relative coordinate structure
        on $\subspaceTopology{X}{U}$ under $G$ by \obv{restricted_coordinates_is_exact}.
    $\RestrictedCoordinates{G}{X}{S}{U}$ is an exact coordinate datum by
        \auto{restricted_coordinatesshape,relative_coordinate_structure}.
    $\RestrictedAtlas{A}{U}$ presents $\RestrictedCoordinates{G}{X}{S}{U}$
        on $\subspaceTopology{X}{U}$ under $G$ by \obv{restricted_coordinates_presentation}.
    $\subspaceTopology{X}{U}$ is a topological space by \obv{subspace_topology_is_topological_space}.
    Follows by \obv{exact_presented_coordinates}.
\end{proof}

\subsection{Restriction laws}

\begin{theorem}\label{restricted_coordinates_nested}
    Let $G$ be a coordinate pseudogroup and $X$ be a topological space.
    Let $S$ be a relative coordinate structure on $X$ under $G$.
    Suppose $U$ is open in $X$ and $V$ is open in $\subspaceTopology{X}{U}$.
    Then $\RestrictedCoordinates{G}{\subspaceTopology{X}{U}}{\RestrictedCoordinates{G}{X}{S}{U}}{V}
        =\RestrictedCoordinates{G}{X}{S}{V}$.
\end{theorem}
\begin{proof}
    $U\in\opens[X]$ and $V\in\opens[\subspaceTopology{X}{U}]$ by \obv{is_open}.
    $V\in\opens[X]$ and $V\subseteq U$ by \auto{open_subspace_open_iff}.
    $\subspaceTopology{X}{U}$ is a topological space by \obv{subspace_topology_is_topological_space}.
    $U\subseteq\carrier[X]$ by \obv{opens_type}.
    $\subspaceTopology{\subspaceTopology{X}{U}}{V}=\subspaceTopology{X}{V}$ by
        \obv{nested_subspace_topology}.
    Let $T=\RestrictedCoordinates{G}{X}{S}{U}$.
    $T$ is a relative coordinate structure on $\subspaceTopology{X}{U}$ under $G$ by
        \auto{restricted_coordinates}.
    $\RestrictedCoordinates{G}{\subspaceTopology{X}{U}}{T}{V}$ is an exact relative coordinate structure
        on $\subspaceTopology{X}{V}$ under $G$ by \auto{restricted_coordinates_is_exact}.
    $\RestrictedCoordinates{G}{X}{S}{V}$ is an exact relative coordinate structure
        on $\subspaceTopology{X}{V}$ under $G$ by \auto{restricted_coordinates_is_exact}.
    For every $c$ we have $c\in\atlas[\RestrictedCoordinates{G}{\subspaceTopology{X}{U}}{T}{V}]$ iff
        $c\in\atlas[\RestrictedCoordinates{G}{X}{S}{V}]$ by
        \auto{restricted_coordinates_chart_iff,subseteq_transitive}.
    $\atlas[\RestrictedCoordinates{G}{\subspaceTopology{X}{U}}{T}{V}]
        =\atlas[\RestrictedCoordinates{G}{X}{S}{V}]$ by set extensionality.
    Follows by structure extensionality.
\end{proof}

\begin{theorem}\label{restricted_coordinates_empty_atlas}
    Let $G$ be a coordinate pseudogroup and $X$ be a topological space.
    Let $S$ be a relative coordinate structure on $X$ under $G$.
    Then $\atlas[\RestrictedCoordinates{G}{X}{S}{\emptyset}]=\{\emptyset\}$.
\end{theorem}
\begin{proof}
    $\emptyset$ is open in $X$ by \obv{emptyset_open}.
    Let $Y=\subspaceTopology{X}{\emptyset}$ and $T=\RestrictedCoordinates{G}{X}{S}{\emptyset}$.
    $Y$ is a topological space by \obv{subspace_topology_is_topological_space}.
    $\carrier[Y]=\emptyset$ by \auto{subspace_topology_carrier,inter_emptyset,inter_comm}.
    $T$ is a relative coordinate structure on $Y$ under $G$ by \auto{restricted_coordinates}.
    $\atlas[T]$ is a maximal coordinate atlas on $Y$ under $G$ by \obv{relative_atlas_maximal}.
    $\atlas[T]$ is a coordinate atlas on $Y$ under $G$ by \obv{maximal_coordinate_atlas}.
    $\emptyset$ is a coordinate atlas on $Y$ under $G$ by \obv{empty_coordinate_atlas_iff}.
    $\emptyset\subseteq\atlas[T]$ by \obv{emptyset_subseteq}.
    $\MaxAtlas{G}{Y}{\emptyset}=\MaxAtlas{G}{Y}{\atlas[T]}$ by
        \obv{atlas_extension_same_max_atlas}.
    $\atlas[T]=\MaxAtlas{G}{Y}{\atlas[T]}$ by \obv{maximal_coordinate_atlas_fixed_point}.
    Follows by \obv{empty_max_atlas}.
\end{proof}

Full restriction recovers an exact relative datum. No exactness of the
ambient topological record is needed: its full subspace is its topological
reduct, with the same carrier and opens.

\begin{theorem}\label{restricted_coordinates_full}
    Let $G$ be a coordinate pseudogroup and $X$ be a topological space.
    Suppose $S$ is an exact relative coordinate structure on $X$ under $G$.
    Then $\RestrictedCoordinates{G}{X}{S}{\carrier[X]}=S$.
\end{theorem}
\begin{proof}
    $\carrier[X]\in\opens[X]$ by \obv{carrier_open}.
    $\RestrictedCoordinates{G}{X}{S}{\carrier[X]}$ is an exact relative coordinate structure
        on $\subspaceTopology{X}{\carrier[X]}$ under $G$ by \obv{restricted_coordinates_is_exact}.
    $\atlas[S]$ is a coordinate atlas on $X$ under $G$ by
        \auto{relative_atlas_maximal,maximal_coordinate_atlas}.
    For every $c\in\atlas[S]$ we have $\dom{c}\subseteq\carrier[X]$ by
        \auto{coordinate_atlas,open_partial_homeomorphism,opens_type}.
    For every $c$ we have $c\in\atlas[\RestrictedCoordinates{G}{X}{S}{\carrier[X]}]$ iff
        $c\in\atlas[S]$ by \auto{restricted_coordinates_chart_iff}.
    $\atlas[\RestrictedCoordinates{G}{X}{S}{\carrier[X]}]=\atlas[S]$ by set extensionality.
    Follows by structure extensionality.
\end{proof}

\subsection{Equipping the subspace}

Equipping uses the subspace topology and the restricted atlas. The inclusion
will be admissible because in a restricted chart it is the identity.

\begin{theorem}\label{equipped_restricted_coordinates}
    Let $G$ be a coordinate pseudogroup and $X$ be a topological space.
    Let $S$ be a relative coordinate structure on $X$ under $G$.
    Suppose $U$ is open in $X$.
    Then $\Equip{\subspaceTopology{X}{U}}{\RestrictedCoordinates{G}{X}{S}{U}}$ has coordinates under $G$
        and $\carrier[\Equip{\subspaceTopology{X}{U}}{\RestrictedCoordinates{G}{X}{S}{U}}]=U$
        and $\TopologicalReduct{\Equip{\subspaceTopology{X}{U}}{\RestrictedCoordinates{G}{X}{S}{U}}}
            =\subspaceTopology{X}{U}$.
\end{theorem}
\begin{proof}
    Let $Y=\subspaceTopology{X}{U}$ and $T=\RestrictedCoordinates{G}{X}{S}{U}$.
    $Y$ is an exact topological space by \auto{subspace_topology_is_exact}.
    $T$ is a relative coordinate structure on $Y$ under $G$ by \auto{restricted_coordinates}.
    $T$ is a coordinate datum by \obv{relative_coordinate_structure}.
    $\Equip{Y}{T}$ has coordinates under $G$ by \obv{equip_preserves_coordinates}.
    $U\in\opens[X]$ by \obv{is_open}.
    $U\subseteq\carrier[X]$ by \obv{opens_type}.
    $\carrier[Y]=U$ by \auto{subspace_topology_carrier_of_subset}.
    $\carrier[\Equip{Y}{T}]=U$ by \auto{equipprojection_carrier}.
    $\TopologicalReduct{\Equip{Y}{T}}=\TopologicalReduct{Y}$ by \obv{equip_topological_reduct}.
    $\TopologicalReduct{Y}=Y$ by \obv{topological_reduct_of_exact}.
\end{proof}

\begin{theorem}\label{equipped_restricted_coordinates_presentation}
    Let $G$ be a coordinate pseudogroup and $X$ be a topological space.
    Let $S$ be a coordinate datum.
    Suppose $A$ presents $S$ on $X$ under $G$.
    Suppose $U$ is open in $X$.
    Then $\RestrictedAtlas{A}{U}$ presents
        $\Equip{\subspaceTopology{X}{U}}{\RestrictedCoordinates{G}{X}{S}{U}}$ on
        $\Equip{\subspaceTopology{X}{U}}{\RestrictedCoordinates{G}{X}{S}{U}}$ under $G$.
\end{theorem}
\begin{proof}
    $S$ is a relative coordinate structure on $X$ under $G$ by \obv{presented_coordinates}.
    $\subspaceTopology{X}{U}$ is a topological space by \obv{subspace_topology_is_topological_space}.
    $\RestrictedCoordinates{G}{X}{S}{U}$ is a coordinate datum by
        \auto{restricted_coordinates,relative_coordinate_structure}.
    $\RestrictedAtlas{A}{U}$ presents $\RestrictedCoordinates{G}{X}{S}{U}$
        on $\subspaceTopology{X}{U}$ under $G$ by \obv{restricted_coordinates_presentation}.
    Follows by \obv{equip_presentation_iff}.
\end{proof}

\begin{theorem}\label{restricted_coordinates_inclusion}
    Let $C$ be a local-map calculus and $G$ be a coordinate pseudogroup.
    Let $X$ be a topological space.
    Let $S$ be a relative coordinate structure on $X$ under $G$.
    Suppose $G$ is compatible with the calculus $C$ and $U$ is open in $X$.
    Then $\subspaceInclusion{X}{U}$ is coordinate-admissible from
        $\Equip{\subspaceTopology{X}{U}}{\RestrictedCoordinates{G}{X}{S}{U}}$ under $G$
        to $\Equip{X}{S}$ under $G$ in $C$.
\end{theorem}
\begin{proof}
    Let $Y=\subspaceTopology{X}{U}$ and $T=\RestrictedCoordinates{G}{X}{S}{U}$.
    Let $f=\subspaceInclusion{X}{U}$.
    $Y$ is a topological space by \obv{subspace_topology_is_topological_space}.
    $T$ is a relative coordinate structure on $Y$ under $G$ by \auto{restricted_coordinates}.
    $\atlas[T]$ is a coordinate atlas on $Y$ under $G$ by
        \auto{relative_atlas_maximal,maximal_coordinate_atlas}.
    $\model[G]\in\models[C]$ by \auto{compatible_pseudogroup_model,calculus_model}.
    $U\in\opens[X]$ and $U\subseteq\carrier[X]$ by \auto{is_open,opens_type}.
    $\carrier[Y]=U$ by \auto{subspace_topology_carrier_of_subset}.
    $f=\identity{U}$ by \auto{subspace_inclusion,inter_absorb_supseteq_right}.
    $f$ is a function and $\dom{f}=U$ by \auto{id_is_function,id_dom}.
    $f\in\funs{\dom{f}}{\carrier[X]}$ by \auto{id_elem_funs,funs_weaken_codom}.
    $\dom{f}\in\opens[Y]$ by \auto{carrier_open}.
    Show that for every $p\in\dom{f}$ there exists $c\in\ChartsAt{T}{p}$ such that
        there exists $d\in\atlas[S]$ such that there exists $V\in\opens[Y]$ such that
            $p\in V$ and $V\subseteq\dom{f}$ and $V\subseteq\dom{c}$ and
            $\img{f}{V}\subseteq\dom{d}$ and
            $\CoordExpr{d}{f}{c}{V}$ is admissible from $\model[G]$ to $\model[G]$ in $C$.
    \begin{subproof}
        Fix $p\in\dom{f}$.
        $p\in\carrier[Y]$.
        Take $c\in\atlas[T]$ such that $p\in\dom{c}$ by \obv{coordinate_atlas}.
        $c\in\ChartsAt{T}{p}$ by \obv{charts_at_iff}.
        $c\in\atlas[S]$ and $\dom{c}\subseteq U$ by \auto{restricted_coordinates_chart_iff}.
        $c$ is an open partial homeomorphism from $Y$ to $\model[G]$ by \obv{coordinate_atlas}.
        $c$ is a function by \auto{open_partial_homeomorphism_bijection,bijection_is_function}.
        $\dom{c}\in\opens[Y]$ by \obv{open_partial_homeomorphism}.
        $\img{f}{\dom{c}}=\dom{c}$ by \auto{id_img,inter_absorb_supseteq_left}.
        $\CoordExpr{c}{f}{c}{\dom{c}}=\identity{\img{c}{\dom{c}}}$ by
            \auto{coordinate_expression_partial_identity}.
        $\img{c}{\dom{c}}\in\opens[\model[G]]$ by \obv{open_partial_homeomorphism_image_open}.
        $\identity{\img{c}{\dom{c}}}$ is admissible from $\model[G]$ to $\model[G]$ in $C$ by
            \obv{admissible_open_identity}.
        Follows by \auto{subseteq_refl}.
    \end{subproof}
    Follows by \auto{coordinate_admissible_equip_iff}.
\end{proof}
