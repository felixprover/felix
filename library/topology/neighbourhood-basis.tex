\subsection{Local bases}

\begin{definition}\label{local_basis}
    Let $X$ be a topological space.
    Let $x\in\carrier[X]$.
    $B$ is a local basis at $x$ in $X$ iff
        $B\subseteq\opens[X]$ and
        (for every $V\in B$ we have $x\in V$) and
        (for all $U$ such that $U\in\opens[X]$ and $x\in U$
            there exists $V\in B$ such that $V\subseteq U$).
\end{definition}

\begin{theorem}\label{open_basis_point_slice_is_local_basis}
    Let $X$ be a topological space.
    Suppose $B$ is an open basis of $X$.
    Let $x\in\carrier[X]$.
    Then $\{V\in B\mid x\in V\}$ is a local basis at $x$ in $X$.
\end{theorem}
\begin{proof}
    Let $C=\{V\in B\mid x\in V\}$.
    $C\subseteq B$ by \obv{subseteq}.
    Hence $C\subseteq\opens[X]$
        by \auto{open_basis,subseteq_transitive}.
    For every $V\in C$ we have $x\in V$.
    Show that for all $U$ such that $U\in\opens[X]$ and $x\in U$
        there exists $V\in C$ such that $V\subseteq U$.
    \begin{subproof}
        Fix $U$.
        Assume $U\in\opens[X]$ and $x\in U$.
        Take $V\in B$ such that $x\in V\subseteq U$
            by \obv{open_basis}.
        Hence $V\in C$.
    \end{subproof}
    Follows by \auto{local_basis}.
\end{proof}

\begin{theorem}\label{local_basis_iff_neighbourhood_filter_basis}
    Let $X$ be a topological space and $x\in\carrier[X]$.
    Then $B$ is a local basis at $x$ in $X$ iff
        $B\subseteq\opens[X]$ and
        $B$ is a basis of $\neighbourhoods{x}{X}$ on $\carrier[X]$.
\end{theorem}
\begin{proof}
    Show that if $B$ is a local basis at $x$ in $X$, then
        $B\subseteq\opens[X]$ and
        $B$ is a basis of $\neighbourhoods{x}{X}$ on $\carrier[X]$.
    \begin{subproof}
        Assume $B$ is a local basis at $x$ in $X$.
        $B\subseteq\opens[X]$ by \obv{local_basis}.
        $\neighbourhoods{x}{X}$ is a filter on $\carrier[X]$
            by \obv{neighbourhoods_is_filter}.
        Show that $B\subseteq\neighbourhoods{x}{X}$.
        \begin{subproof}
            It suffices to show that every element of $B$ is an element of
                $\neighbourhoods{x}{X}$ by \obv{subseteq}.
            Fix $V\in B$.
            $V\in\opens[X]$ and $x\in V$ by \obv{local_basis,subseteq}.
            $V\subseteq\carrier[X]$ by \obv{opens_type}.
            Hence $V\in\pow{\carrier[X]}$ by \obv{powerset_intro}.
            Follows by \obv{neighbourhoods,subseteq_refl}.
        \end{subproof}
        Show that $B$ refines $\neighbourhoods{x}{X}$
            as filter families on $\carrier[X]$.
        \begin{subproof}
            $B$ is a family of subsets of $\carrier[X]$
                by \obv{opens_type,subseteq}.
            $\neighbourhoods{x}{X}$ is a family of subsets of $\carrier[X]$
                by \obv{filter}.
            Show that for every $N\in\neighbourhoods{x}{X}$ there exists
                $V\in B$ such that $V\subseteq N$.
            \begin{subproof}
                Fix $N\in\neighbourhoods{x}{X}$.
                Take $U\in\opens[X]$ such that $x\in U\subseteq N$
                    by \obv{neighbourhoods}.
                Take $V\in B$ such that $V\subseteq U$
                    by \obv{local_basis}.
                Hence $V\subseteq N$ by \obv{subseteq_transitive}.
            \end{subproof}
            Follows by \obv{filter_family_refinement}.
        \end{subproof}
        Hence $B$ is a basis of $\neighbourhoods{x}{X}$ on $\carrier[X]$
            by \obv{filter_basis_of}.
    \end{subproof}

    Show that if $B\subseteq\opens[X]$ and
        $B$ is a basis of $\neighbourhoods{x}{X}$ on $\carrier[X]$, then
        $B$ is a local basis at $x$ in $X$.
    \begin{subproof}
        Assume $B\subseteq\opens[X]$ and
            $B$ is a basis of $\neighbourhoods{x}{X}$ on $\carrier[X]$.
        $B\subseteq\neighbourhoods{x}{X}$ and
            $B$ refines $\neighbourhoods{x}{X}$
                as filter families on $\carrier[X]$
            by \obv{filter_basis_of}.
        Show that for every $V\in B$ we have $x\in V$.
        \begin{subproof}
            Fix $V\in B$.
            Hence $V\in\neighbourhoods{x}{X}$ by \obv{subseteq}.
            Take $U\in\opens[X]$ such that $x\in U\subseteq V$
                by \obv{neighbourhoods}.
            Follows by \obv{subseteq}.
        \end{subproof}
        Show that for all $U$ such that $U\in\opens[X]$ and $x\in U$
            there exists $V\in B$ such that $V\subseteq U$.
        \begin{subproof}
            Fix $U$.
            Assume $U\in\opens[X]$ and $x\in U$.
            $U\subseteq\carrier[X]$ by \obv{opens_type}.
            Hence $U\in\pow{\carrier[X]}$ by \obv{powerset_intro}.
            Hence $U\in\neighbourhoods{x}{X}$
                by \obv{neighbourhoods,subseteq_refl}.
            Follows by \obv{filter_family_refinement}.
        \end{subproof}
        Follows by \obv{local_basis}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{theorem}\label{neighbourhood_filter_basis_has_open_local_refinement}
    Let $X$ be a topological space and $x\in\carrier[X]$.
    Suppose $B$ is a basis of $\neighbourhoods{x}{X}$ on $\carrier[X]$.
    Then
        $\{U\in\opens[X]\mid
            \text{$x\in U$ and there exists $A\in B$ such that
                $U\subseteq A$}\}$
            is a local basis at $x$ in $X$ and
        $\{U\in\opens[X]\mid
            \text{$x\in U$ and there exists $A\in B$ such that
                $U\subseteq A$}\}$
            refines $B$ as filter families on $\carrier[X]$.
\end{theorem}
\begin{proof}
    Let $C=\{U\in\opens[X]\mid
        \text{$x\in U$ and there exists $A\in B$ such that
            $U\subseteq A$}\}$.
    $B\subseteq\neighbourhoods{x}{X}$ and
        $B$ refines $\neighbourhoods{x}{X}$
            as filter families on $\carrier[X]$
        by \obv{filter_basis_of}.
    $C\subseteq\opens[X]$ by \obv{subseteq}.
    Hence $C$ is a family of subsets of $\carrier[X]$
        by \obv{opens_type,subseteq}.
    $B$ is a family of subsets of $\carrier[X]$
        by \obv{filter_family_refinement}.

    Show that $C$ refines $B$ as filter families on $\carrier[X]$.
    \begin{subproof}
        Show that for every $A\in B$ there exists $U\in C$ such that
            $U\subseteq A$.
        \begin{subproof}
            Fix $A\in B$.
            Hence $A\in\neighbourhoods{x}{X}$ by \obv{subseteq}.
            Take $U\in\opens[X]$ such that $x\in U\subseteq A$
                by \obv{neighbourhoods}.
            Hence $U\in C$.
        \end{subproof}
        Follows by \obv{filter_family_refinement}.
    \end{subproof}

    Show that $C$ is a local basis at $x$ in $X$.
    \begin{subproof}
        For every $U\in C$ we have $x\in U$.
        Show that for all $N$ such that $N\in\opens[X]$ and $x\in N$
            there exists $U\in C$ such that $U\subseteq N$.
        \begin{subproof}
            Fix $N$.
            Assume $N\in\opens[X]$ and $x\in N$.
            $N\subseteq\carrier[X]$ by \obv{opens_type}.
            Hence $N\in\pow{\carrier[X]}$ by \obv{powerset_intro}.
            Hence $N\in\neighbourhoods{x}{X}$
                by \obv{neighbourhoods,subseteq_refl}.
            Take $A\in B$ such that $A\subseteq N$
                by \obv{filter_family_refinement}.
            $A\in\neighbourhoods{x}{X}$ by \obv{subseteq}.
            Take $U\in\opens[X]$ such that $x\in U\subseteq A$
                by \obv{neighbourhoods}.
            Hence $U\in C$ by \auto.
            $U\subseteq N$ by \obv{subseteq_transitive}.
        \end{subproof}
        Follows by \obv{local_basis}.
    \end{subproof}
\end{proof}

\subsection{Transport}

\begin{proposition}\label{homeomorphism_image_open}
    Suppose $f$ is a homeomorphism from $X$ to $Y$.
    Let $U\in\opens[X]$.
    Then $\img{f}{U}\in\opens[Y]$.
\end{proposition}
\begin{proof}
    $\converse{f}$ is continuous from $Y$ to $X$
        by \obv{homeomorphism}.
    Hence $\preimg{\converse{f}}{U}\in\opens[Y]$
        by \obv{continuous_map}.
    $\img{f}{U}=\preimg{\converse{f}}{U}$
        by \auto{setext,img_iff,preimg_iff,converse_iff}.
\end{proof}

\begin{proposition}\label{homeomorphism_image_neighbourhood}
    Suppose $f$ is a homeomorphism from $X$ to $Y$.
    Let $x\in\carrier[X]$ and $N\in\neighbourhoods{x}{X}$.
    Then $\img{f}{N}\in\neighbourhoods{f(x)}{Y}$.
\end{proposition}
\begin{proof}
    $f$ is continuous from $X$ to $Y$ by \obv{homeomorphism}.
    Hence $f\in\funs{\carrier[X]}{\carrier[Y]}$ by \obv{continuous_map}.
    Take $U\in\opens[X]$ such that $x\in U\subseteq N$
        by \obv{neighbourhoods}.
    $\img{f}{U}\in\opens[Y]$ by \obv{homeomorphism_image_open}.
    $(x,f(x))\in f$ by \auto{funs_elim,function_apply_iff}.
    Hence $f(x)\in\img{f}{U}$ by \auto{img_iff}.
    $\img{f}{U}\subseteq\img{f}{N}$ by \obv{img_subseteq}.
    $\img{f}{N}\subseteq\ran{f}$ by \obv{img_subseteq_ran}.
    $\ran{f}\subseteq\carrier[Y]$ by \obv{funs_ran}.
    Hence $\img{f}{N}\subseteq\carrier[Y]$
        by \obv{subseteq_transitive}.
    Hence $\img{f}{N}\in\pow{\carrier[Y]}$ by \obv{powerset_intro}.
    Follows by \obv{neighbourhoods}.
\end{proof}

\begin{theorem}\label{neighbourhood_filter_basis_homeomorphism_image}
    Let $X,Y$ be topological spaces.
    Let $f\in\funs{\carrier[X]}{\carrier[Y]}$.
    Suppose $f$ is a homeomorphism from $X$ to $Y$.
    Let $x\in\carrier[X]$.
    Suppose $B$ is a basis of $\neighbourhoods{x}{X}$ on $\carrier[X]$.
    Then $\{\img{f}{A}\mid A\in B\}$ is a basis of
        $\neighbourhoods{f(x)}{Y}$ on $\carrier[Y]$.
\end{theorem}
\begin{proof}
    $f$ is continuous from $X$ to $Y$ by \obv{homeomorphism}.
    $f(x)\in\carrier[Y]$ by \obv{funs_type_apply}.
    $\neighbourhoods{f(x)}{Y}$ is a filter on $\carrier[Y]$
        by \obv{neighbourhoods_is_filter}.
    Let $C=\{\img{f}{A}\mid A\in B\}$.
    $B\subseteq\neighbourhoods{x}{X}$ and
        $B$ refines $\neighbourhoods{x}{X}$
            as filter families on $\carrier[X]$
        by \obv{filter_basis_of}.

    Show that $C\subseteq\neighbourhoods{f(x)}{Y}$.
    \begin{subproof}
        It suffices to show that every element of $C$ is an element of
            $\neighbourhoods{f(x)}{Y}$ by \obv{subseteq}.
        Fix $D\in C$.
        Take $A\in B$ such that $D=\img{f}{A}$.
        Hence $A\in\neighbourhoods{x}{X}$ by \obv{subseteq}.
        $\img{f}{A}\in\neighbourhoods{f(x)}{Y}$
            by \obv{homeomorphism_image_neighbourhood}.
    \end{subproof}

    Show that $C$ refines $\neighbourhoods{f(x)}{Y}$
        as filter families on $\carrier[Y]$.
    \begin{subproof}
        $C$ is a family of subsets of $\carrier[Y]$
            by \obv{neighbourhoods_is_filter,filter,subseteq}.
        $\neighbourhoods{f(x)}{Y}$ is a family of subsets of $\carrier[Y]$
            by \obv{filter}.
        Show that for every $N\in\neighbourhoods{f(x)}{Y}$ there exists
            $D\in C$ such that $D\subseteq N$.
        \begin{subproof}
            Fix $N\in\neighbourhoods{f(x)}{Y}$.
            Take $V\in\opens[Y]$ such that $f(x)\in V\subseteq N$
                by \obv{neighbourhoods}.
            Let $P=\preimg{f}{V}$.
            $P\in\opens[X]$ by \auto{continuous_map}.
            $(x,f(x))\in f$ by \auto{funs_elim,function_apply_iff}.
            Hence $x\in P$ by \auto{preimg_iff}.
            $P\subseteq\carrier[X]$ by \obv{opens_type}.
            Hence $P\in\pow{\carrier[X]}$ by \obv{powerset_intro}.
            Hence $P\in\neighbourhoods{x}{X}$
                by \obv{neighbourhoods,subseteq_refl}.
            Take $A\in B$ such that $A\subseteq P$
                by \obv{filter_family_refinement}.
            Hence $\img{f}{A}\in C$.
            $\img{f}{A}\subseteq\img{f}{P}$ by \obv{img_subseteq}.
            Show that $\img{f}{P}\subseteq V$.
            \begin{subproof}
                It suffices to show that every element of $\img{f}{P}$
                    is an element of $V$ by \obv{subseteq}.
                Fix $y\in\img{f}{P}$.
                Take $a\in P$ such that $(a,y)\in f$
                    by \obv{img_iff}.
                Take $v\in V$ such that $(a,v)\in f$
                    by \obv{preimg_iff}.
                $f$ is right-unique by \obv{funs_is_function,rightunique}.
                Hence $y=v$ by \obv{rightunique}.
            \end{subproof}
            Hence $\img{f}{A}\subseteq V$
                by \obv{subseteq_transitive}.
            Hence $\img{f}{A}\subseteq N$
                by \obv{subseteq_transitive}.
        \end{subproof}
        Follows by \obv{filter_family_refinement}.
    \end{subproof}
    Follows by \obv{filter_basis_of}.
\end{proof}

\begin{corollary}\label{local_basis_homeomorphism_image}
    Let $X,Y$ be topological spaces.
    Let $f\in\funs{\carrier[X]}{\carrier[Y]}$.
    Suppose $f$ is a homeomorphism from $X$ to $Y$.
    Let $x\in\carrier[X]$.
    Suppose $B$ is a local basis at $x$ in $X$.
    Then $\{\img{f}{A}\mid A\in B\}$ is a local basis at $f(x)$ in $Y$.
\end{corollary}
\begin{proof}
    $f(x)\in\carrier[Y]$ by \obv{funs_type_apply}.
    Let $C=\{\img{f}{A}\mid A\in B\}$.
    $B\subseteq\opens[X]$ and
        $B$ is a basis of $\neighbourhoods{x}{X}$ on $\carrier[X]$
        by \obv{local_basis_iff_neighbourhood_filter_basis}.
    Hence $C$ is a basis of $\neighbourhoods{f(x)}{Y}$ on $\carrier[Y]$
        by \auto{neighbourhood_filter_basis_homeomorphism_image}.
    Show that $C\subseteq\opens[Y]$.
    \begin{subproof}
        It suffices to show that every element of $C$ is an element of $\opens[Y]$
            by \obv{subseteq}.
        Fix $D\in C$.
        Take $A\in B$ such that $D=\img{f}{A}$.
        Hence $A\in\opens[X]$ by \obv{subseteq}.
        Follows by \auto{homeomorphism_image_open}.
    \end{subproof}
    Hence $C$ is a local basis at $f(x)$ in $Y$
        by \auto{local_basis_iff_neighbourhood_filter_basis}.
\end{proof}

\subsection{Nested natural bases}

\begin{corollary}\label{antitone_natural_tail_range_is_neighbourhood_filter_basis}
    Let $X$ be a topological space and $x\in\carrier[X]$.
    Let $b\in\funs{\naturals}{\pow{\carrier[X]}}$.
    Suppose for all $K,L$ such that $K,L\in\naturals$ and $K\subseteq L$
        we have $b(L)\subseteq b(K)$.
    Suppose $\ran{b}$ is a basis of $\neighbourhoods{x}{X}$
        on $\carrier[X]$.
    Let $N$ be a natural number.
    Then $\img{b}{\naturalTail{N}}$ is a basis of
        $\neighbourhoods{x}{X}$ on $\carrier[X]$.
\end{corollary}
\begin{proof}
    $\img{b}{\naturalTail{N}}\subseteq\ran{b}$
        by \obv{img_subseteq_ran}.
    $\img{b}{\naturalTail{N}}$ refines $\ran{b}$
        as filter families on $\carrier[X]$
        by \auto{antitone_natural_tail_range_mutual_refinement}.
    $\neighbourhoods{x}{X}$ is a filter on $\carrier[X]$
        by \obv{neighbourhoods_is_filter}.
    Follows by \obv{refining_subfamily_is_filter_basis}.
\end{proof}
