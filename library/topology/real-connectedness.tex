The density of the finite-stage surreals supplies points on either side of a
real point inside every metric ball.  These two elementary lemmas are the
local ingredients in the supremum proof of connectedness.

\begin{lemma}\label{real_ball_meets_right_open_interval}
    Suppose $s,v,r\in\reals$.
    Suppose $\surrealLt{s}{v}$ and
        $\surrealLt{\surrealZero}{r}$.
    Then there exists $t\in\reals$ such that
        $\surrealLt{s}{t}$ and $\surrealLt{t}{v}$ and
        $t\in\metricBall{\RealMetric}{s}{r}$.
\end{lemma}
\begin{proof}
    Let $l=\surrealAdd{s}{\surrealNeg{r}}$.
    Let $u=\surrealAdd{s}{r}$.
    $\surrealNeg{r}\in\reals$ by
        \obv{surreal_real_negation_closed}.
    Hence $l,u\in\reals$ by
        \auto{surreal_real_addition_closed}.
    $\metricBall{\RealMetric}{s}{r}
        =\orderOpenInterval{\RealField}
            {\surrealAdd{s}{\surrealNeg{r}}}{\surrealAdd{s}{r}}$ by
        \obv{real_metric_ball_eq_order_open_interval}.
    Hence $\metricBall{\RealMetric}{s}{r}
        =\orderOpenInterval{\RealField}{l}{u}$.
    $\surreal{r}$ and $\surreal{s}$ by
        \obv{surreal_real_is_surreal}.
    $\surrealLt{l}{s}$ by
        \auto{surreal_negation_lt_reverses,surreal_negation_empty,
            surreal_add_lt_right,surreal_add_zero_right,empty_surreal,
            surreal_negation_is_surreal}.
    $\surrealLt{s}{u}$ by
        \auto{surreal_add_lt_right,surreal_add_zero_right,empty_surreal}.
    $\surrealLe{v}{u}$ or $\surrealLe{u}{v}$ by
        \obv{surreal_le_total,surreal_real_is_surreal}.
    \begin{byCase}
    \caseOf{$\surrealLe{v}{u}$.}
        Take $t\in\surrealStage{\naturals}$ such that
            $\surrealLt{s}{t}$ and $\surrealLt{t}{v}$ by
            \obv{finite_surreal_between_reals}.
        Hence $t\in\reals$ by
            \obv{finite_surreal_stage_subset_reals,subseteq}.
        $\surrealLt{l}{t}$ by
            \auto{surreal_lt_transitive,surreal_real_is_surreal}.
        $\surrealLt{t}{u}$ by
            \auto{surreal_lt_le_transitive,surreal_real_is_surreal}.
        Thus $t\in\orderOpenInterval{\RealField}{l}{u}$ by
            \auto{order_open_interval_iff,
                real_field_order_strict_iff,surreal_real_field_carrier}.
        Hence $t\in\metricBall{\RealMetric}{s}{r}$.
    \caseOf{not $\surrealLe{v}{u}$.}
        Hence $\surrealLe{u}{v}$.
        $u\neq v$.
        Hence $\surrealLt{u}{v}$ by \auto{surreal_le}.
        Take $t\in\surrealStage{\naturals}$ such that
            $\surrealLt{s}{t}$ and $\surrealLt{t}{u}$ by
            \obv{finite_surreal_between_reals}.
        Hence $t\in\reals$ by
            \obv{finite_surreal_stage_subset_reals,subseteq}.
        $\surrealLt{t}{v}$ by
            \auto{surreal_lt_transitive,surreal_real_is_surreal}.
        $\surrealLt{l}{t}$ by
            \auto{surreal_lt_transitive,surreal_real_is_surreal}.
        Thus $t\in\orderOpenInterval{\RealField}{l}{u}$ by
            \auto{order_open_interval_iff,
                real_field_order_strict_iff,surreal_real_field_carrier}.
        Hence $t\in\metricBall{\RealMetric}{s}{r}$.
    \end{byCase}
\end{proof}

\begin{theorem}\label{real_order_convex_set_is_connected}
    Let $X$ be a topological space.
    Suppose $X=\RealTopology$.
    Let $L$ be a linearly ordered set.
    Suppose $L=\RealField$.
    Suppose $A$ is order convex in $L$.
    Then $A$ is connected in $X$.
\end{theorem}
\begin{proof}
    $A\subseteq\carrier[L]$ by \obv{order_convex}.
    $\carrier[L]=\reals$ by
        \auto{surreal_real_field_carrier}.
    Hence $A\subseteq\reals$.
    $\carrier[X]=\reals$ by \auto{real_topology_carrier}.
    Hence $A\subseteq\carrier[X]$.
    Let $S=\subspaceTopology{X}{A}$.
    $\subspaceTopology{X}{A}$ is a topological space by
        \obv{subspace_topology_is_topological_space}.
    $\carrier[\subspaceTopology{X}{A}]=A$ by
        \obv{subspace_topology_carrier_of_subset}.
    Hence $\carrier[S]=A$.

    Show that $S$ is connected.
    \begin{subproof}
        Suppose not.
        Hence $S$ is disconnected by \auto{connected}.
        Take $G,H\in\opens[S]$ such that
            $\carrier[S]$ is partitioned by $G$ and $H$ by
            \obv{disconnected}.
        Hence $A$ is partitioned by $G$ and $H$ by \auto.
        $G$ and $H$ are inhabited by
            \auto{nonempty_inhabited}.
        Take $g$ such that $g\in G$.
        Take $h$ such that $h\in H$.
        $G,H\subseteq A$ by
            \auto{opens_type,subseteq}.
        Hence $g,h\in A$ by \obv{subseteq}.
        Hence $g,h\in\reals$ by \obv{subseteq}.
        $G$ is disjoint from $H$.
        Hence $g\neq h$ by \auto{disjoint}.
        $\surrealLt{g}{h}$ or $g=h$ or $\surrealLt{h}{g}$ by
            \auto{surreal_lt_trichotomy,surreal_real_is_surreal}.

        Show that there exist $u,v,U,V$ such that
            $u,v\in A$ and $U,V\in\opens[S]$ and
            $A$ is partitioned by $U$ and $V$ and
            $u\in U$ and $v\in V$ and $\surrealLt{u}{v}$.
        \begin{subproof}
            \begin{byCase}
            \caseOf{$\surrealLt{g}{h}$.}
                Follows by \auto.
            \caseOf{not $\surrealLt{g}{h}$.}
                Hence $\surrealLt{h}{g}$ by \auto.
                $A$ is partitioned by $H$ and $G$ by
                    \auto{disjoint,union_comm}.
                Follows by \auto.
            \end{byCase}
        \end{subproof}
        Take $u,v,U,V$ such that
            $u,v\in A$ and $U,V\in\opens[S]$ and
            $A$ is partitioned by $U$ and $V$ and
            $u\in U$ and $v\in V$ and $\surrealLt{u}{v}$.
        Hence $u,v\in\reals$ by \obv{subseteq}.
        $U$ is disjoint from $V$.

        Let $B=\{x\in U\mid
            \text{$\surrealLe{u}{x}$ and $\surrealLe{x}{v}$}\}$.
        $B\subseteq U$ by \obv{subseteq}.
        $U\subseteq\carrier[S]$ by \obv{opens_type}.
        Hence $U\subseteq A$ by \auto.
        Hence $B\subseteq A$ by \auto{subseteq_transitive}.
        Hence $B\subseteq\reals$ by \auto{subseteq_transitive}.
        $\surrealLe{u}{u}$ and $\surrealLe{u}{v}$ by
            \auto{surreal_le_reflexive,surreal_le,
                surreal_real_is_surreal}.
        Hence $u\in B$.
        Thus $B$ is inhabited by \auto.
        For every $b\in B$ we have $\surrealLe{b}{v}$.
        Let $s=\realSupremum{B}$.
        $\realSupremum{B}\in\reals$ by
            \auto{real_supremum_is_real}.
        Hence $s\in\reals$.
        $\surrealLe{u}{\realSupremum{B}}$ by
            \auto{real_supremum_upper_bound}.
        Hence $\surrealLe{u}{s}$.
        $\surrealLe{\realSupremum{B}}{v}$ by
            \auto{real_supremum_least}.
        Hence $\surrealLe{s}{v}$.

        $u,v,s\in\carrier[L]$ by \auto.
        $u\mathrel{\leq[L]}s$ and $s\mathrel{\leq[L]}v$ by
            \auto{surreal_real_field_order}.
        Hence $s\in A$ by \auto{order_convex}.
        Thus $s\in\carrier[S]$.
        Hence $s\in U$ or $s\in V$ by \auto{union_iff}.

        \begin{byCase}
        \caseOf{$s\in U$.}
            $s\neq v$ by \auto{disjoint}.
            Hence $\surrealLt{s}{v}$ by \auto{surreal_le}.
            Take $O\in\opens[X]$ such that
                $U=O\inter\carrier[S]$ by
                \auto{subspace_open_iff}.
            Hence $s\in O$ by \auto{inter}.
            $O\in\opens[\RealTopology]$ by \auto.
            $\opens[\RealTopology]=\metricOpens{\RealMetric}$ by
                \auto{real_topology,metric_topology_opens}.
            Hence $O\in\metricOpens{\RealMetric}$.
            Thus $\metricOpen{\RealMetric}{O}$ by
                \auto{metric_opens_iff}.
            Take $r\in\reals$ such that
                $\surrealLt{\surrealZero}{r}$ and
                $\metricBall{\RealMetric}{s}{r}\subseteq O$ by
                \obv{metric_open}.
            Take $t\in\reals$ such that
                $\surrealLt{s}{t}$ and $\surrealLt{t}{v}$ and
                $t\in\metricBall{\RealMetric}{s}{r}$ by
                \obv{real_ball_meets_right_open_interval}.
            $s,t,v\in\carrier[L]$ by \auto.
            $s\mathrel{\leq[L]}t$ and $t\mathrel{\leq[L]}v$ by
                \auto{surreal_le,surreal_real_field_order}.
            Hence $t\in A$ by \auto{order_convex}.
            Hence $t\in\carrier[S]$.
            $t\in O$ by \obv{subseteq}.
            Hence $t\in U$ by \auto{inter}.
            $\surrealLt{u}{t}$ by
                \auto{surreal_le_lt_transitive,
                    surreal_real_is_surreal}.
            Hence $\surrealLe{u}{t}$ by \auto{surreal_le}.
            $\surrealLe{t}{v}$ by \auto{surreal_le}.
            Hence $t\in B$.
            Thus $\surrealLe{t}{s}$ by
                \auto{real_supremum_upper_bound}.
            Contradiction by
                \auto{surreal_le,surreal_lt_asymmetric,
                    surreal_real_is_surreal}.

        \caseOf{$s\notin U$.}
            Hence $s\in V$ by \auto.
            $s\neq u$ by \auto{disjoint}.
            Hence $\surrealLt{u}{s}$ by \auto{surreal_le}.
            Take $O\in\opens[X]$ such that
                $V=O\inter\carrier[S]$ by
                \auto{subspace_open_iff}.
            Hence $s\in O$ by \auto{inter}.
            $O\in\opens[\RealTopology]$ by \auto.
            $\opens[\RealTopology]=\metricOpens{\RealMetric}$ by
                \auto{real_topology,metric_topology_opens}.
            Hence $O\in\metricOpens{\RealMetric}$.
            Thus $\metricOpen{\RealMetric}{O}$ by
                \auto{metric_opens_iff}.
            Take $r\in\reals$ such that
                $\surrealLt{\surrealZero}{r}$ and
                $\metricBall{\RealMetric}{s}{r}\subseteq O$ by
                \obv{metric_open}.
            Take $t\in\reals$ such that
                $\surrealLt{u}{t}$ and $\surrealLt{t}{s}$ and
                $t\in\metricBall{\RealMetric}{s}{r}$ by
                \obv{real_ball_meets_left_open_interval}.
            $u,t,s\in\carrier[L]$ by \auto.
            $u\mathrel{\leq[L]}t$ and $t\mathrel{\leq[L]}s$ by
                \auto{surreal_le,surreal_real_field_order}.
            Hence $t\in A$ by \auto{order_convex}.
            Hence $t\in\carrier[S]$.
            $t\in O$ by \obv{subseteq}.
            Hence $t\in V$ by \auto{inter}.

            Show that there exists $b\in B$ such that
                $\surrealLt{t}{b}$.
            \begin{subproof}
                Suppose not.
                Show that for every $b\in B$ we have
                    $\surrealLe{b}{t}$.
                \begin{subproof}
                    Fix $b\in B$.
                    Hence $b\in U$ by \auto.
                    Hence $b\in\carrier[S]$ by
                        \auto{opens_type,pow_iff,subseteq}.
                    Hence $b\in A$ by \auto.
                    Hence $b\in\carrier[L]$ by
                        \auto{order_convex,subseteq}.
                    Hence $b\in\reals$ by
                        \auto{surreal_real_field_carrier}.
                    not $\surrealLt{t}{b}$ by \auto.
                    Follows by
                        \auto{surreal_lt_trichotomy,surreal_le,
                            surreal_real_is_surreal}.
                \end{subproof}
                Hence $\surrealLe{s}{t}$ by
                    \auto{real_supremum_least}.
                Contradiction by
                    \auto{surreal_le,surreal_lt_asymmetric,
                        surreal_real_is_surreal}.
            \end{subproof}
            Take $b\in B$ such that $\surrealLt{t}{b}$.
            Hence $b\in U$ and $b\in A$ and $b\in\reals$ by
                \auto{subseteq}.
            $\surrealLe{b}{s}$ by
                \auto{real_supremum_upper_bound}.
            $t\in\carrier[\RealField]$ by
                \auto{surreal_real_field_carrier}.
            $b\in\carrier[\RealField]$ by
                \auto{surreal_real_field_carrier}.
            $s\in\carrier[\RealField]$ by
                \auto{surreal_real_field_carrier}.
            $\surrealNeg{r}\in\reals$ by
                \auto{surreal_real_negation_closed}.
            Hence $\surrealAdd{s}{\surrealNeg{r}}\in\reals$ by
                \auto{surreal_real_addition_closed}.
            $\surrealAdd{s}{r}\in\reals$ by
                \auto{surreal_real_addition_closed}.
            Hence $\surrealAdd{s}{\surrealNeg{r}},
                \surrealAdd{s}{r}\in\carrier[\RealField]$ by
                \auto{surreal_real_field_carrier}.
            $\RealField$ is an ordered set by
                \auto{surreal_real_field_is_linearly_ordered_set,
                    linearly_ordered_set_is_ordered_set}.
            $\orderOpenInterval{\RealField}
                {\surrealAdd{s}{\surrealNeg{r}}}
                {\surrealAdd{s}{r}}$ is order convex in $\RealField$ by
                \obv{order_open_interval_is_order_convex}.
            $\metricBall{\RealMetric}{s}{r}
                =\orderOpenInterval{\RealField}
                    {\surrealAdd{s}{\surrealNeg{r}}}
                    {\surrealAdd{s}{r}}$ by
                \obv{real_metric_ball_eq_order_open_interval}.
            Hence $t\in\orderOpenInterval{\RealField}
                {\surrealAdd{s}{\surrealNeg{r}}}
                {\surrealAdd{s}{r}}$ by \auto.
            $\RealMetric$ is a metric space by
                \obv{surreal_real_metric_space}.
            $\RealMetric$ is a pseudometric space by
                \obv{metric_is_pseudometric_space}.
            $s\in\carrier[\RealMetric]$ by
                \auto{surreal_real_metric_carrier}.
            $s\in\metricBall{\RealMetric}{s}{r}$ by
                \auto{metric_ball_contains_centre,
                    surreal_real_metric_space,
                    surreal_real_metric_carrier}.
            Hence $s\in\orderOpenInterval{\RealField}
                {\surrealAdd{s}{\surrealNeg{r}}}
                {\surrealAdd{s}{r}}$ by \auto.
            $t\mathrel{\leq[\RealField]}b$ and
                $b\mathrel{\leq[\RealField]}s$ by
                \auto{surreal_le,surreal_real_field_order}.
            Hence $b\in\orderOpenInterval{\RealField}
                {\surrealAdd{s}{\surrealNeg{r}}}
                {\surrealAdd{s}{r}}$ by
                \auto{order_convex}.
            Hence $b\in\metricBall{\RealMetric}{s}{r}$ by
                \auto.
            Hence $b\in O$ by \obv{subseteq}.
            Thus $b\in V$ by \auto{inter}.
            Contradiction by \auto{disjoint}.
        \end{byCase}
    \end{subproof}
    Hence $\subspaceTopology{X}{A}$ is connected by \auto.
    Follows by \obv{connected_in}.
\end{proof}

\begin{corollary}\label{real_closed_interval_is_connected}
    Let $X$ be a topological space.
    Suppose $X=\RealTopology$.
    Let $L$ be a linearly ordered set.
    Suppose $L=\RealField$.
    Let $a,b\in\carrier[L]$.
    Then $\orderClosedInterval{L}{a}{b}$ is connected in $X$.
\end{corollary}
\begin{proof}
    $\orderClosedInterval{L}{a}{b}$ is order convex in $L$ by
        \obv{order_closed_interval_is_order_convex}.
    Follows by \obv{real_order_convex_set_is_connected}.
\end{proof}
