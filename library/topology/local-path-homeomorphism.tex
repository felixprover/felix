\section{Local path connectedness under homeomorphisms}

\import{topology/local-path.tex}
\import{topology/embedding.tex}

\begin{theorem}\label{homeomorphism_preserves_locally_path_connectedness}
    Let $X,Y$ be topological spaces.
    Suppose $X$ is locally path-connected and $f$ is a homeomorphism from $X$ to $Y$.
    Then $Y$ is locally path-connected.
\end{theorem}
\begin{proof}
    $f$ is continuous from $X$ to $Y$ and $f\in\bijections{\carrier[X]}{\carrier[Y]}$ by
        \obv{homeomorphism}.
    $f$ is a function and $\dom{f}=\carrier[X]$ by \obv{bijection_is_function,bijections_dom}.
    $f$ is an open map from $X$ to $Y$ by \obv{homeomorphism_is_open_map}.
    Show that for all $U,y$ such that $U\in\opens[Y]$ and $y\in U$ there exists
        $V\in\opens[Y]$ such that $y\in V\subseteq U$ and
        $\subspaceTopology{Y}{V}$ is path-connected.
    \begin{subproof}
        Fix $U,y$.
        Assume $U\in\opens[Y]$ and $y\in U$.
        $y\in\carrier[Y]$ by \auto{opens_type,subseteq}.
        Take $x\in\carrier[X]$ such that $f(x)=y$ by \auto{bijections,surj}.
        $\preimg{f}{U}\in\opens[X]$ by \obv{continuous_map}.
        $x\in\preimg{f}{U}$ by \auto{preimg_iff,function_apply_elim}.
        Take $W\in\opens[X]$ such that $x\in W\subseteq\preimg{f}{U}$ and
            $\subspaceTopology{X}{W}$ is path-connected by \obv{locally_path_connected}.
        $W\subseteq\carrier[X]$ by \auto{opens_type,subseteq}.
        $\img{f}{W}\in\opens[Y]$ by \obv{open_map}.
        $y\in\img{f}{W}$ by \auto{img_iff,function_apply_elim}.
        $\img{f}{W}\subseteq U$ by \auto{img_iff,preimg_iff,subseteq,rightunique}.
        $\restrl{f}{W}$ is a homeomorphism from $\subspaceTopology{X}{W}$ to
            $\subspaceTopology{Y}{\img{f}{W}}$ by \obv{homeomorphism_restriction_to_image}.
        $\subspaceTopology{Y}{\img{f}{W}}$ is path-connected by
            \auto{homeomorphism_preserves_path_connectedness,subspace_topology_is_topological_space}.
    \end{subproof}
    Follows by \obv{locally_path_connected}.
\end{proof}
