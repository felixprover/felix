\section{Binary product topologies}

\import{topology/initial.tex}
\import{set/product-map.tex}
\import{set/filter-product.tex}
\import{set/dependent-operation.tex}

The binary topological product uses the literal Cartesian product carrier.
Its two projections are collected into the same two-coordinate set-coded
family used elsewhere in the library.

\begin{definition}\label{binary_product_space_family}
    $\binaryProductSpaceFamily{X}{Y}=\binaryFamily{X}{Y}$.
\end{definition}

\begin{definition}\label{binary_product_projection_family}
    $\binaryProductProjectionFamily{X}{Y}
        =\binaryFamily{
            \firstProductProjection{\carrier[X]}{\carrier[Y]}}{
            \secondProductProjection{\carrier[X]}{\carrier[Y]}}$.
\end{definition}

\begin{definition}\label{binary_product_topology}
    $\BinaryProductTopology{X}{Y}
        =\InitialTopology{
            \carrier[X]\times\carrier[Y]}{
            \suc{\suc{\emptyset}}}{
            \binaryProductSpaceFamily{X}{Y}}{
            \binaryProductProjectionFamily{X}{Y}}$.
\end{definition}

\begin{proposition}\label{binary_product_initial_data}
    Let $X,Y$ be topological spaces.
    Then $\binaryProductProjectionFamily{X}{Y}$ is an initial mapping
        family from $\carrier[X]\times\carrier[Y]$ to
        $\binaryProductSpaceFamily{X}{Y}$ indexed by
        $\suc{\suc{\emptyset}}$.
\end{proposition}
\begin{proof}
    Let $I=\suc{\suc{\emptyset}}$.
    $\binaryProductSpaceFamily{X}{Y}\in\funs{I}{\{X,Y\}}$ by
        \auto{binary_product_space_family,binary_family_type}.
    Hence $\binaryProductSpaceFamily{X}{Y}$ is a function on $I$ by
        \obv{funs_is_function,funs}.
    $\binaryProductProjectionFamily{X}{Y}\in
        \funs{I}{\{
            \firstProductProjection{\carrier[X]}{\carrier[Y]},
            \secondProductProjection{\carrier[X]}{\carrier[Y]}\}}$ by
        \auto{binary_product_projection_family,binary_family_type}.
    Hence $\binaryProductProjectionFamily{X}{Y}$ is a function on $I$ by
        \obv{funs_is_function,funs}.
    Show that for every $i\in I$ we have
        $(\binaryProductSpaceFamily{X}{Y})(i)$ is a topological space and
        $(\binaryProductProjectionFamily{X}{Y})(i)
            \in\funs{\carrier[X]\times\carrier[Y]}{
                \carrier[(\binaryProductSpaceFamily{X}{Y})(i)]}$.
    \begin{subproof}
        Fix $i\in I$.
        $i=\emptyset$ or $i=\suc{\emptyset}$ by
            \auto{suc_elim,emptyset}.
        Follows by
            \auto{binary_product_space_family,
                binary_product_projection_family,
                binary_family_left,binary_family_right,
                first_product_projection_type,
                second_product_projection_type}.
    \end{subproof}
    Follows by \obv{initial_topology_data}.
\end{proof}

\begin{classification}\label{binary_product_topology_is_topological_space}
    $\BinaryProductTopology{X}{Y}$ is a topological space.
\end{classification}
\begin{proof}
    Follows by
        \auto{binary_product_topology,
            initial_topology_is_topological_space}.
\end{proof}

\begin{theorem}\label{binary_product_topology_is_exact}
    $\BinaryProductTopology{X}{Y}$ is an exact topological space.
\end{theorem}
\begin{proof}
    Follows by
        \auto{binary_product_topology,initial_topology_is_exact}.
\end{proof}

\begin{proposition}\label{binary_product_topology_carrier}
    $\carrier[\BinaryProductTopology{X}{Y}]
        =\carrier[X]\times\carrier[Y]$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{binary_product_topology,initial_topology_carrier}.
\end{proof}

\begin{theorem}\label{binary_product_coordinate_continuous}
    Let $X,Y$ be topological spaces.
    Let $i\in\suc{\suc{\emptyset}}$.
    Then $(\binaryProductProjectionFamily{X}{Y})(i)$ is continuous from
        $\BinaryProductTopology{X}{Y}$ to
        $(\binaryProductSpaceFamily{X}{Y})(i)$.
\end{theorem}
\begin{proof}
    $\binaryProductProjectionFamily{X}{Y}$ is an initial mapping
        family from $\carrier[X]\times\carrier[Y]$ to
        $\binaryProductSpaceFamily{X}{Y}$ indexed by
        $\suc{\suc{\emptyset}}$ by
        \obv{binary_product_initial_data}.
    Hence $(\binaryProductProjectionFamily{X}{Y})(i)$ is continuous from
        $\InitialTopology{
            \carrier[X]\times\carrier[Y]}{
            \suc{\suc{\emptyset}}}{
            \binaryProductSpaceFamily{X}{Y}}{
            \binaryProductProjectionFamily{X}{Y}}$ to
        $(\binaryProductSpaceFamily{X}{Y})(i)$ by
        \obv{initial_topology_projection_continuous}.
    $\BinaryProductTopology{X}{Y}
        =\InitialTopology{
            \carrier[X]\times\carrier[Y]}{
            \suc{\suc{\emptyset}}}{
            \binaryProductSpaceFamily{X}{Y}}{
            \binaryProductProjectionFamily{X}{Y}}$ by
        \obv{binary_product_topology}.
\end{proof}

\begin{theorem}\label{first_product_projection_continuous}
    Let $X,Y$ be topological spaces.
    Then $\firstProductProjection{\carrier[X]}{\carrier[Y]}$ is continuous
        from $\BinaryProductTopology{X}{Y}$ to $X$.
\end{theorem}
\begin{proof}
    $\emptyset\in\suc{\suc{\emptyset}}$ by
        \auto{suc_iff,emptyset}.
    Hence $(\binaryProductProjectionFamily{X}{Y})(\emptyset)$ is continuous
        from $\BinaryProductTopology{X}{Y}$ to
        $(\binaryProductSpaceFamily{X}{Y})(\emptyset)$ by
        \obv{binary_product_coordinate_continuous}.
    $(\binaryProductProjectionFamily{X}{Y})(\emptyset)
        =\firstProductProjection{\carrier[X]}{\carrier[Y]}$ and
        $(\binaryProductSpaceFamily{X}{Y})(\emptyset)=X$ by
        \auto{binary_product_projection_family,
            binary_product_space_family,binary_family_left}.
\end{proof}

\begin{theorem}\label{second_product_projection_continuous}
    Let $X,Y$ be topological spaces.
    Then $\secondProductProjection{\carrier[X]}{\carrier[Y]}$ is continuous
        from $\BinaryProductTopology{X}{Y}$ to $Y$.
\end{theorem}
\begin{proof}
    $\suc{\emptyset}\in\suc{\suc{\emptyset}}$ by
        \auto{suc_iff}.
    Hence $(\binaryProductProjectionFamily{X}{Y})(\suc{\emptyset})$
        is continuous from $\BinaryProductTopology{X}{Y}$ to
        $(\binaryProductSpaceFamily{X}{Y})(\suc{\emptyset})$ by
        \obv{binary_product_coordinate_continuous}.
    $(\binaryProductProjectionFamily{X}{Y})(\suc{\emptyset})
        =\secondProductProjection{\carrier[X]}{\carrier[Y]}$ and
        $(\binaryProductSpaceFamily{X}{Y})(\suc{\emptyset})=Y$ by
        \auto{binary_product_projection_family,
            binary_product_space_family,binary_family_right}.
\end{proof}

\begin{theorem}\label{binary_product_topology_universal_property}
    Let $X,Y,Z$ be topological spaces.
    Let $h\in\funs{\carrier[Z]}{
        \carrier[X]\times\carrier[Y]}$.
    Then $h$ is continuous from $Z$ to $\BinaryProductTopology{X}{Y}$ iff
        $\firstProductProjection{\carrier[X]}{\carrier[Y]}\circ h$
            is continuous from $Z$ to $X$ and
        $\secondProductProjection{\carrier[X]}{\carrier[Y]}\circ h$
            is continuous from $Z$ to $Y$.
\end{theorem}
\begin{proof}
    Show that if $h$ is continuous from $Z$ to
        $\BinaryProductTopology{X}{Y}$, then
        $\firstProductProjection{\carrier[X]}{\carrier[Y]}\circ h$
            is continuous from $Z$ to $X$ and
        $\secondProductProjection{\carrier[X]}{\carrier[Y]}\circ h$
            is continuous from $Z$ to $Y$.
    \begin{subproof}
        Assume $h$ is continuous from $Z$ to
            $\BinaryProductTopology{X}{Y}$.
        $\firstProductProjection{\carrier[X]}{\carrier[Y]}$ is continuous
            from $\BinaryProductTopology{X}{Y}$ to $X$ by
            \obv{first_product_projection_continuous}.
        Hence $\firstProductProjection{\carrier[X]}{\carrier[Y]}\circ h$
            is continuous from $Z$ to $X$ by
            \obv{continuous_composition}.
        $\secondProductProjection{\carrier[X]}{\carrier[Y]}$ is continuous
            from $\BinaryProductTopology{X}{Y}$ to $Y$ by
            \obv{second_product_projection_continuous}.
        Hence $\secondProductProjection{\carrier[X]}{\carrier[Y]}\circ h$
            is continuous from $Z$ to $Y$ by
            \obv{continuous_composition}.
    \end{subproof}

    Show that if
        $\firstProductProjection{\carrier[X]}{\carrier[Y]}\circ h$
            is continuous from $Z$ to $X$ and
        $\secondProductProjection{\carrier[X]}{\carrier[Y]}\circ h$
            is continuous from $Z$ to $Y$, then
        $h$ is continuous from $Z$ to $\BinaryProductTopology{X}{Y}$.
    \begin{subproof}
        Assume
            $\firstProductProjection{\carrier[X]}{\carrier[Y]}\circ h$
                is continuous from $Z$ to $X$ and
            $\secondProductProjection{\carrier[X]}{\carrier[Y]}\circ h$
                is continuous from $Z$ to $Y$.
        $\binaryProductProjectionFamily{X}{Y}$ is an initial mapping
            family from $\carrier[X]\times\carrier[Y]$ to
            $\binaryProductSpaceFamily{X}{Y}$ indexed by
            $\suc{\suc{\emptyset}}$ by
            \obv{binary_product_initial_data}.
        Show that for every $i\in\suc{\suc{\emptyset}}$ we have
            $(\binaryProductProjectionFamily{X}{Y})(i)\circ h$
                is continuous from $Z$ to
                $(\binaryProductSpaceFamily{X}{Y})(i)$.
        \begin{subproof}
            Fix $i\in\suc{\suc{\emptyset}}$.
            $i=\emptyset$ or $i=\suc{\emptyset}$ by
                \auto{suc_elim,emptyset}.
            Follows by
                \auto{binary_product_projection_family,
                    binary_product_space_family,
                    binary_family_left,binary_family_right}.
        \end{subproof}
        Hence $h$ is continuous from $Z$ to
            $\InitialTopology{
                \carrier[X]\times\carrier[Y]}{
                \suc{\suc{\emptyset}}}{
                \binaryProductSpaceFamily{X}{Y}}{
                \binaryProductProjectionFamily{X}{Y}}$ by
            \obv{initial_topology_universal_property}.
        $\BinaryProductTopology{X}{Y}
            =\InitialTopology{
                \carrier[X]\times\carrier[Y]}{
                \suc{\suc{\emptyset}}}{
                \binaryProductSpaceFamily{X}{Y}}{
                \binaryProductProjectionFamily{X}{Y}}$ by
            \obv{binary_product_topology}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\subsection{Pairing maps}


\begin{theorem}\label{continuous_product_pairing}
    Suppose $f$ is continuous from $Z$ to $X$ and
        $g$ is continuous from $Z$ to $Y$.
    Then $\productPairing{\carrier[Z]}{f}{g}$ is continuous from $Z$ to
        $\BinaryProductTopology{X}{Y}$.
\end{theorem}
\begin{proof}
    $X$ is a topological space by \obv{continuous_map}.
    $Z$ is a topological space by \obv{continuous_map}.
    $Y$ is a topological space by \obv{continuous_map}.
    $f\in\funs{\carrier[Z]}{\carrier[X]}$ and
        $g\in\funs{\carrier[Z]}{\carrier[Y]}$ by
        \obv{continuous_map}.
    Hence $\productPairing{\carrier[Z]}{f}{g}\in
        \funs{\carrier[Z]}{\carrier[X]\times\carrier[Y]}$ by
        \obv{product_pairing_type}.
    $\firstProductProjection{\carrier[X]}{\carrier[Y]}\circ
        \productPairing{\carrier[Z]}{f}{g}=f$ by
        \obv{first_projection_of_product_pairing}.
    $\secondProductProjection{\carrier[X]}{\carrier[Y]}\circ
        \productPairing{\carrier[Z]}{f}{g}=g$ by
        \obv{second_projection_of_product_pairing}.
    Follows by \auto{binary_product_topology_universal_property}.
\end{proof}
