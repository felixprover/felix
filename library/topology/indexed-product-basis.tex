\section{The finite-coordinate basis of an indexed product}

\import{topology/indexed-product.tex}
\import{topology/basis.tex}

The coordinate slabs form the canonical prebasis of an indexed product.
Finite intersections of these slabs are the open boxes: only finitely many
coordinates are constrained, and the empty intersection is the whole product.

\begin{definition}\label{indexed_product_prebasis}
    $\indexedProductPrebasis{I}{X}
        =\initialPrebasis{
            \dependentProduct{I}{
                \indexedTopologicalCarrierFamily{I}{X}}}{I}{X}{
            \indexedProductProjectionFamily{I}{X}}$.
\end{definition}

\begin{definition}\label{indexed_product_boxes}
    $\indexedProductBoxes{I}{X}
        =\finiteIntersectionsOn{
            \dependentProduct{I}{
                \indexedTopologicalCarrierFamily{I}{X}}}{
            \indexedProductPrebasis{I}{X}}$.
\end{definition}

\begin{proposition}\label{indexed_product_prebasis_iff}
    $W\in\indexedProductPrebasis{I}{X}$ iff
        $W\subseteq\dependentProduct{I}{
            \indexedTopologicalCarrierFamily{I}{X}}$ and
        there exists $i\in I$ such that there exists $U\in\opens[X(i)]$
            such that
            $W=\preimg{
                \dependentProjection{I}{
                    \indexedTopologicalCarrierFamily{I}{X}}{i}}{U}$.
\end{proposition}
\begin{proof}
    Let $S=\dependentProduct{I}{
        \indexedTopologicalCarrierFamily{I}{X}}$.
    Let $f=\indexedProductProjectionFamily{I}{X}$.
    Show that if $W\in\indexedProductPrebasis{I}{X}$, then
        $W\subseteq S$ and there exists $i\in I$ such that
        there exists $U\in\opens[X(i)]$ such that
        $W=\preimg{\dependentProjection{I}{
            \indexedTopologicalCarrierFamily{I}{X}}{i}}{U}$.
    \begin{subproof}
        Assume $W\in\indexedProductPrebasis{I}{X}$.
        Hence $W\in\initialPrebasis{S}{I}{X}{f}$ by
            \obv{indexed_product_prebasis}.
        Hence $W\in\pow{S}$ and there exists $i\in I$ such that
            there exists $U\in\opens[X(i)]$ such that
                $W=\preimg{f(i)}{U}$ by \obv{initial_prebasis}.
        Take $i\in I$ such that there exists $U\in\opens[X(i)]$
            such that $W=\preimg{f(i)}{U}$.
        Take $U\in\opens[X(i)]$ such that $W=\preimg{f(i)}{U}$.
        $W\subseteq S$ by \obv{powerset_elim}.
        $f(i)=\dependentProjection{I}{
            \indexedTopologicalCarrierFamily{I}{X}}{i}$ by
            \auto{indexed_product_projection_family_apply}.
        Follows by \auto.
    \end{subproof}

    Show that if $W\subseteq S$ and there exists $i\in I$ such that
        there exists $U\in\opens[X(i)]$ such that
        $W=\preimg{\dependentProjection{I}{
            \indexedTopologicalCarrierFamily{I}{X}}{i}}{U}$,
        then $W\in\indexedProductPrebasis{I}{X}$.
    \begin{subproof}
        Assume $W\subseteq S$ and there exists $i\in I$ such that
            there exists $U\in\opens[X(i)]$ such that
            $W=\preimg{\dependentProjection{I}{
                \indexedTopologicalCarrierFamily{I}{X}}{i}}{U}$.
        Take $i\in I$ such that there exists $U\in\opens[X(i)]$
            such that $W=\preimg{\dependentProjection{I}{
                \indexedTopologicalCarrierFamily{I}{X}}{i}}{U}$.
        Take $U\in\opens[X(i)]$ such that
            $W=\preimg{\dependentProjection{I}{
                \indexedTopologicalCarrierFamily{I}{X}}{i}}{U}$.
        $W\in\pow{S}$ by \obv{powerset_intro}.
        $f(i)=\dependentProjection{I}{
            \indexedTopologicalCarrierFamily{I}{X}}{i}$ by
            \auto{indexed_product_projection_family_apply}.
        Hence $W=\preimg{f(i)}{U}$.
        Hence $W\in\initialPrebasis{S}{I}{X}{f}$ by
            \auto{initial_prebasis}.
        Follows by \obv{indexed_product_prebasis}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{proposition}\label{indexed_product_prebasis_type}
    $\indexedProductPrebasis{I}{X}$ is a family of subsets of
        $\dependentProduct{I}{
            \indexedTopologicalCarrierFamily{I}{X}}$.
\end{proposition}
\begin{proof}
    Follows by
        \obv{indexed_product_prebasis,initial_prebasis_type}.
\end{proof}

\begin{proposition}\label{indexed_product_boxes_iff}
    $W\in\indexedProductBoxes{I}{X}$ iff
        $W\subseteq\dependentProduct{I}{
            \indexedTopologicalCarrierFamily{I}{X}}$ and
        there exists a finite set $D$ such that
            $D\subseteq\indexedProductPrebasis{I}{X}$ and
            $W=\boundedIntersection{
                \dependentProduct{I}{
                    \indexedTopologicalCarrierFamily{I}{X}}}{D}$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{indexed_product_boxes,finite_intersections_on_iff}.
\end{proof}

\begin{theorem}\label{indexed_product_prebasis_is_open_prebasis}
    $\indexedProductPrebasis{I}{X}$ is an open prebasis of
        $\IndexedProductTopology{I}{X}$.
\end{theorem}
\begin{proof}
    $\indexedProductPrebasis{I}{X}$ is a family of subsets of
        $\dependentProduct{I}{
            \indexedTopologicalCarrierFamily{I}{X}}$ by
        \obv{indexed_product_prebasis_type}.
    Hence $\indexedProductPrebasis{I}{X}$ is an open prebasis of
        $\GeneratedTopology{
            \dependentProduct{I}{
                \indexedTopologicalCarrierFamily{I}{X}}}{
            \indexedProductPrebasis{I}{X}}$ by
        \obv{generated_topology_generators_form_open_prebasis}.
    $\IndexedProductTopology{I}{X}=\InitialTopology{
        \dependentProduct{I}{
            \indexedTopologicalCarrierFamily{I}{X}}}{I}{X}{
        \indexedProductProjectionFamily{I}{X}}$ by
        \obv{indexed_product_topology}.
    $\InitialTopology{
        \dependentProduct{I}{
            \indexedTopologicalCarrierFamily{I}{X}}}{I}{X}{
        \indexedProductProjectionFamily{I}{X}}
        =\GeneratedTopology{
            \dependentProduct{I}{
                \indexedTopologicalCarrierFamily{I}{X}}}{
            \indexedProductPrebasis{I}{X}}$ by
        \obv{initial_topology,indexed_product_prebasis}.
    Hence $\IndexedProductTopology{I}{X}
        =\GeneratedTopology{
            \dependentProduct{I}{
                \indexedTopologicalCarrierFamily{I}{X}}}{
            \indexedProductPrebasis{I}{X}}$.
\end{proof}

\begin{theorem}\label{indexed_product_boxes_form_open_basis}
    $\indexedProductBoxes{I}{X}$ is an open basis of
        $\IndexedProductTopology{I}{X}$.
\end{theorem}
\begin{proof}
    $\indexedProductPrebasis{I}{X}$ is an open prebasis of
        $\IndexedProductTopology{I}{X}$ by
        \obv{indexed_product_prebasis_is_open_prebasis}.
    Hence
        $\finiteIntersectionsOn{
            \carrier[\IndexedProductTopology{I}{X}]}{
            \indexedProductPrebasis{I}{X}}$
        is an open basis of $\IndexedProductTopology{I}{X}$ by
        \obv{finite_intersections_of_open_prebasis_form_open_basis}.
    $\carrier[\IndexedProductTopology{I}{X}]
        =\dependentProduct{I}{
            \indexedTopologicalCarrierFamily{I}{X}}$ by
        \obv{indexed_product_topology_carrier}.
    Follows by \auto{indexed_product_boxes}.
\end{proof}
