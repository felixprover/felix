\section{Countable rectangle bases}

\import{topology/countability-axioms.tex}
\import{topology/product-basis.tex}

\begin{theorem}\label{product_of_countable_bases_is_countable}
    Suppose $B$ and $C$ are countable.
    Then $\productBasis{B}{C}$ is countable.
\end{theorem}
\begin{proof}
    $B\times C$ is countable by \obv{product_of_countable_sets_is_countable}.
    Let $D=\{\fst{p}\times\snd{p}\mid p\in B\times C\}$.
    $\cardLe{D}{\naturals}$ by \auto{countable_replacement}.
    $\productBasis{B}{C}=D$ by \obv{product_basis}.
    Follows by \auto{cardinal_le,at_most_countable}.
\end{proof}

\begin{theorem}\label{binary_product_of_second_countable_spaces}
    Let $X,Y$ be second-countable topological spaces.
    Then $\BinaryProductTopology{X}{Y}$ is second-countable.
\end{theorem}
\begin{proof}
    Take $B$ such that $B$ is countable and $B$ is an open basis of $X$
        by \obv{is_second_countable}.
    Take $C$ such that $C$ is countable and $C$ is an open basis of $Y$
        by \obv{is_second_countable}.
    $\productBasis{B}{C}$ is countable by \obv{product_of_countable_bases_is_countable}.
    $\productBasis{B}{C}$ is an open basis of $\BinaryProductTopology{X}{Y}$
        by \obv{product_of_open_bases}.
    $\BinaryProductTopology{X}{Y}$ is a topological space by
        \obv{binary_product_topology_is_topological_space}.
    Follows by \obv{is_second_countable}.
\end{proof}
