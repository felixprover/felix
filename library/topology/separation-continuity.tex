\section{Hausdorff separation through continuous maps}

\import{topology/separation.tex}
\import{topology/continuity.tex}

Distinct images in a Hausdorff space have disjoint open neighbourhoods.
Their preimages separate the original points. Injectivity is needed only
when one map is to separate every pair of distinct points.

\begin{theorem}\label{continuous_map_separates_points}
    Suppose $f$ is continuous from $X$ to $Y$ and $Y$ is Hausdorff.
    Let $x,y\in\carrier[X]$.
    Suppose $f(x)\neq f(y)$.
    Then there exist $U,V\in\opens[X]$ such that
        $x\in U$ and $y\in V$ and $U$ is disjoint from $V$.
\end{theorem}
\begin{proof}
    $f\in\funs{\carrier[X]}{\carrier[Y]}$ by \obv{continuous_map}.
    Hence $f(x),f(y)\in\carrier[Y]$ by \obv{funs_type_apply}.
    Take $A,B\in\opens[Y]$ such that
        $f(x)\in A$ and $f(y)\in B$ and $A$ is disjoint from $B$ by
        \obv{is_hausdorff}.
    Let $U=\preimg{f}{A}$ and $V=\preimg{f}{B}$.
    $U\in\opens[X]$ and $V\in\opens[X]$ by \auto{continuous_map}.
    $f$ is a function and $\dom{f}=\carrier[X]$ by \obv{funs_is_function,funs}.
    $(x,f(x))\in f$ and $(y,f(y))\in f$ by \obv{function_apply_elim}.
    Hence $x\in U$ and $y\in V$ by \auto{preimg_iff}.
    Follows by \auto{disjoint,preimg_iff,rightunique}.
\end{proof}

\begin{theorem}\label{continuous_injection_into_hausdorff}
    Suppose $f$ is continuous from $X$ to $Y$ and $Y$ is Hausdorff.
    Suppose $f$ is injective.
    Then $X$ is Hausdorff.
\end{theorem}
\begin{proof}
    $X$ is a topological space and $f\in\funs{\carrier[X]}{\carrier[Y]}$ by
        \obv{continuous_map}.
    Show that for all $x,y\in\carrier[X]$ such that $x\neq y$ there exist
        $U,V\in\opens[X]$ such that $x\in U$ and $y\in V$ and $U$ is disjoint from $V$.
    \begin{subproof}
        Fix $x,y\in\carrier[X]$.
        Assume $x\neq y$.
        $f(x)\neq f(y)$ by \auto{injective_function,funs_is_function,funs}.
        Follows by \obv{continuous_map_separates_points}.
    \end{subproof}
    Follows by \obv{is_hausdorff}.
\end{proof}

\begin{theorem}\label{homeomorphism_preserves_hausdorff}
    Suppose $f$ is a homeomorphism from $X$ to $Y$ and $X$ is Hausdorff.
    Then $Y$ is Hausdorff.
\end{theorem}
\begin{proof}
    $\converse{f}$ is a homeomorphism from $Y$ to $X$ by \obv{homeomorphism_inverse}.
    Hence $\converse{f}$ is continuous from $Y$ to $X$ and
        $\converse{f}\in\bijections{\carrier[Y]}{\carrier[X]}$ by \obv{homeomorphism}.
    Hence $\converse{f}$ is injective by \obv{bijections}.
    Follows by \obv{continuous_injection_into_hausdorff}.
\end{proof}
