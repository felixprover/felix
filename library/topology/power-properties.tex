\section{Separation and compactness of powers}

\import{topology/power.tex}
\import{topology/separation-product.tex}
\import{topology/compactness-indexed-product.tex}
\import{topology/disconnection.tex}

The general product theorems specialize to powers through their constant
families. The empty power is compact and connected regardless of these
properties of the original space.

\begin{theorem}\label{topological_power_is_hausdorff}
    Suppose $X$ is Hausdorff.
    Then $\TopologicalPower{X}{I}$ is Hausdorff.
\end{theorem}
\begin{proof}
    $I\times\{X\}$ is a function on $I$ by \auto{constant_graph_type,funs_is_function,funs}.
    For every $i\in I$ we have $(I\times\{X\})(i)$ is Hausdorff by
        \auto{constant_graph_application}.
    Follows by \auto{topological_power,indexed_product_of_hausdorff_spaces}.
\end{proof}

\begin{theorem}\label{finite_topological_power_is_compact}
    Let $X$ be a topological space.
    Suppose $X$ is compact and $I$ is finite.
    Then $\TopologicalPower{X}{I}$ is compact.
\end{theorem}
\begin{proof}
    $I\times\{X\}$ is a function on $I$ by \auto{constant_graph_type,funs_is_function,funs}.
    For every $i\in I$ we have $(I\times\{X\})(i)$ is a compact topological space by
        \auto{constant_graph_application}.
    Follows by \auto{topological_power,finite_indexed_product_of_compact_spaces_is_compact}.
\end{proof}

\begin{theorem}\label{empty_topological_power_is_compact}
    Let $X$ be a topological space.
    Then $\TopologicalPower{X}{\emptyset}$ is compact.
\end{theorem}
\begin{proof}
    $\TopologicalPower{X}{\emptyset}$ is a topological space by
        \obv{topological_power_is_topological_space}.
    $\carrier[\TopologicalPower{X}{\emptyset}]=\{\emptyset\}$ by
        \auto{topological_power_carrier,empty_domain_function_space}.
    Hence $\carrier[\TopologicalPower{X}{\emptyset}]$ is finite by \auto{singleton_is_finite}.
    Follows by \obv{finite_topological_space_is_compact}.
\end{proof}

\begin{theorem}\label{empty_topological_power_is_connected}
    Let $X$ be a topological space.
    Then $\TopologicalPower{X}{\emptyset}$ is connected.
\end{theorem}
\begin{proof}
    $\TopologicalPower{X}{\emptyset}$ is a topological space by
        \obv{topological_power_is_topological_space}.
    $\carrier[\TopologicalPower{X}{\emptyset}]=\{\emptyset\}$ by
        \auto{topological_power_carrier,empty_domain_function_space}.
    Hence $\carrier[\TopologicalPower{X}{\emptyset}]$ is a subsingleton by \auto{singleton_iff}.
    Follows by \obv{subsingleton_topological_space_is_connected}.
\end{proof}
