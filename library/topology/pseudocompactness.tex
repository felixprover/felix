\section{Pseudocompact spaces}

\import{topology/continuity.tex}

\begin{definition}\label{is_pseudocompact}
    Let $X$ be a topological space.
    $X$ is pseudocompact iff for every $f$ such that
        $f$ is continuous from $X$ to $\RealTopology$
        we have $\ran{f}$ is real bounded.
\end{definition}

\begin{theorem}\label{pseudocompact_continuous_real_map_bounded}
    Let $X$ be a topological space.
    Suppose $X$ is pseudocompact.
    Suppose $f$ is continuous from $X$ to $\RealTopology$.
    Then $\ran{f}$ is real bounded.
\end{theorem}
\begin{proof}
    Follows by \obv{is_pseudocompact}.
\end{proof}

\begin{theorem}\label{compact_implies_pseudocompact}
    Let $X$ be a topological space.
    Suppose $X$ is compact.
    Then $X$ is pseudocompact.
\end{theorem}
\begin{proof}
    Show that for every $f$ such that
        $f$ is continuous from $X$ to $\RealTopology$
        we have $\ran{f}$ is real bounded.
    \begin{subproof}
        Fix $f$.
        Assume $f$ is continuous from $X$ to $\RealTopology$.
        $\RealTopology$ is a topological space by
            \obv{real_topology_is_topological_space}.
        $\ran{f}$ is compact in $\RealTopology$ by
            \obv{continuous_image_is_compact}.
        Follows by \obv{compact_real_subset_is_bounded}.
    \end{subproof}
    Follows by \obv{is_pseudocompact}.
\end{proof}

\begin{theorem}\label{not_pseudocompact_iff_unbounded_continuous_real_map}
    Let $X$ be a topological space.
    $X$ is not pseudocompact iff there exists $f$ such that
        $f$ is continuous from $X$ to $\RealTopology$ and
        $\ran{f}$ is not real bounded.
\end{theorem}
\begin{proof}
    Show that if $X$ is not pseudocompact, then there exists $f$ such that
        $f$ is continuous from $X$ to $\RealTopology$ and
        $\ran{f}$ is not real bounded.
    \begin{subproof}
        Assume $X$ is not pseudocompact.
        Suppose not.
        Show that for every $f$ such that
            $f$ is continuous from $X$ to $\RealTopology$
            we have $\ran{f}$ is real bounded.
        \begin{subproof}
            Fix $f$.
            Assume $f$ is continuous from $X$ to $\RealTopology$.
            Suppose not.
            Contradiction by \auto.
        \end{subproof}
        Hence $X$ is pseudocompact by \obv{is_pseudocompact}.
        Contradiction.
    \end{subproof}
    Show that if there exists $f$ such that
        $f$ is continuous from $X$ to $\RealTopology$ and
        $\ran{f}$ is not real bounded, then $X$ is not pseudocompact.
    \begin{subproof}
        Assume there exists $f$ such that
            $f$ is continuous from $X$ to $\RealTopology$ and
            $\ran{f}$ is not real bounded.
        Take $f$ such that
            $f$ is continuous from $X$ to $\RealTopology$ and
            $\ran{f}$ is not real bounded.
        Suppose $X$ is pseudocompact.
        Hence $\ran{f}$ is real bounded by
            \obv{pseudocompact_continuous_real_map_bounded}.
        Contradiction.
    \end{subproof}
    Follows by \auto.
\end{proof}
