\section{The topology induced by a pseudometric}

A pseudometric structure already determines a checked family of open sets.
The following transparent construction packages that family as an ordinary
topological-space record.  It is a parameterized term, not an instance
declaration.  The adjacent import preserves this module's established access
to the topology induced by a uniform structure.

\begin{definition}\label{metric_topology}
    $\metricTopology{M}=
        \TopologicalSpaceOn{\carrier[M]}{\metricOpens{M}}$.
\end{definition}

\begin{theorem}\label{metric_topology_is_topological_space}
    Let $M$ be a pseudometric space.
    Then $\metricTopology{M}$ is a topological space.
\end{theorem}
\begin{proof}
    $\metricOpens{M}$ is a topology on $\carrier[M]$
        by \obv{metric_opens_form_topology}.
    Follows by
        \auto{metric_topology,topological_space_on_is_topological_space}.
\end{proof}

\begin{theorem}\label{metric_topology_is_exact}
    Let $M$ be a pseudometric space.
    Then $\metricTopology{M}$ is an exact topological space.
\end{theorem}
\begin{proof}
    $\metricOpens{M}$ is a topology on $\carrier[M]$ by
        \obv{metric_opens_form_topology}.
    Follows by \auto{metric_topology,topological_space_on_is_exact}.
\end{proof}

\begin{proposition}\label{metric_topology_carrier}
    $\carrier[\metricTopology{M}]=\carrier[M]$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{metric_topology,topological_space_on_carrier}.
\end{proof}

\begin{proposition}\label{metric_topology_opens}
    $\opens[\metricTopology{M}]=\metricOpens{M}$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{metric_topology,topological_space_on_opens}.
\end{proof}
