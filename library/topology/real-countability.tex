\section{A countable basis of the real line}

\import{topology/real-order-topology.tex}
\import{topology/countability-axioms.tex}
\import{surreal/countability.tex}
\import{surreal/real-completeness.tex}

Intervals with finite-birthday endpoints form a countable basis for the
existing real topology. No new real carrier or topology is needed.

\begin{definition}\label{real_countable_basis}
    $\RealCountableBasis=\{\realOpenInterval{\fst{p}}{\snd{p}}
        \mid p\in\surrealStage{\naturals}\times\surrealStage{\naturals}\}$.
\end{definition}

\begin{proposition}\label{real_countable_basis_iff}
    $U\in\RealCountableBasis$ iff there exists $a\in\surrealStage{\naturals}$ such that
        there exists $b\in\surrealStage{\naturals}$ such that $U=\realOpenInterval{a}{b}$.
\end{proposition}
\begin{proof}
    Follows by \auto{real_countable_basis,times,fst_eq,snd_eq}.
\end{proof}

\begin{theorem}\label{real_countable_basis_is_countable}
    $\RealCountableBasis$ is countable.
\end{theorem}
\begin{proof}
    $\surrealStage{\naturals}$ is countable by \obv{finite_birthday_surreals_are_countable}.
    Hence $\surrealStage{\naturals}\times\surrealStage{\naturals}$ is countable by
        \obv{product_of_countable_sets_is_countable}.
    Let $B=\{\realOpenInterval{\fst{p}}{\snd{p}}
        \mid p\in\surrealStage{\naturals}\times\surrealStage{\naturals}\}$.
    $\cardLe{B}{\naturals}$ by \auto{countable_replacement}.
    $\RealCountableBasis=B$ by \obv{real_countable_basis}.
    Follows by \auto{cardinal_le,at_most_countable}.
\end{proof}

\begin{theorem}\label{real_countable_basis_is_open_basis}
    $\RealCountableBasis$ is an open basis of $\RealTopology$.
\end{theorem}
\begin{proof}
    $\RealTopology$ is a topological space by \obv{real_topology_is_topological_space}.
    $\surrealStage{\naturals}\subseteq\reals$ by \obv{finite_surreal_stage_subset_reals}.
    $\RealCountableBasis\subseteq\opens[\RealTopology]$ by
        \auto{real_countable_basis_iff,real_open_interval_is_open,subseteq}.
    Show that for all $U,x$ such that $U\in\opens[\RealTopology]$ and $x\in U$
        there exists $V\in\RealCountableBasis$ such that $x\in V\subseteq U$.
    \begin{subproof}
        Fix $U,x$.
        Assume $U\in\opens[\RealTopology]$ and $x\in U$.
        Take $a,b$ such that $a,b\in\reals$ and
            $x\in\realOpenInterval{a}{b}\subseteq U$ by
            \obv{real_open_neighbourhood_contains_interval}.
        $x\in\reals$ and $\surrealLt{a}{x}$ and $\surrealLt{x}{b}$ by
            \obv{real_open_interval_member_iff}.
        Take $q\in\surrealStage{\naturals}$ such that $\surrealLt{a}{q}$ and
            $\surrealLt{q}{x}$ by \obv{finite_surreal_between_reals}.
        Take $r\in\surrealStage{\naturals}$ such that $\surrealLt{x}{r}$ and
            $\surrealLt{r}{b}$ by \obv{finite_surreal_between_reals}.
        $q,r\in\reals$ by \obv{subseteq}.
        $x\in\realOpenInterval{q}{r}$ by \obv{real_open_interval_member_iff}.
        $\realOpenInterval{q}{r}\subseteq\realOpenInterval{a}{b}$ by
            \auto{subseteq,real_open_interval_member_iff,surreal_lt_transitive,surreal_real_is_surreal}.
        Hence $\realOpenInterval{q}{r}\subseteq U$ by \obv{subseteq_transitive}.
        $\realOpenInterval{q}{r}\in\RealCountableBasis$ by \auto{real_countable_basis_iff}.
    \end{subproof}
    Follows by \obv{open_basis}.
\end{proof}

\begin{theorem}\label{real_topology_is_second_countable}
    $\RealTopology$ is second-countable.
\end{theorem}
\begin{proof}
    $\RealCountableBasis$ is countable by \obv{real_countable_basis_is_countable}.
    $\RealCountableBasis$ is an open basis of $\RealTopology$ by \obv{real_countable_basis_is_open_basis}.
    Follows by \obv{is_second_countable,real_topology_is_topological_space}.
\end{proof}
