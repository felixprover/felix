\section{Universal properties of binary coproducts}

\import{topology/coproduct.tex}

The two-element indexed statements reduce to the familiar binary rules: a
set is open when both tagged traces are open, and a map out of a coproduct is
continuous when both restrictions are continuous.

\begin{theorem}\label{binary_coproduct_open_iff}
    Let $X,Y$ be topological spaces.
    Then $U\in\opens[\BinaryCoproductTopology{X}{Y}]$ iff
        $U\subseteq\carrier[\BinaryCoproductTopology{X}{Y}]$ and
        $\preimg{\firstCoproductInjection{X}{Y}}{U}\in\opens[X]$ and
        $\preimg{\secondCoproductInjection{X}{Y}}{U}\in\opens[Y]$.
\end{theorem}
\begin{proof}
    Let $I=\suc{\suc{\emptyset}}$.
    Let $F=\binaryCoproductSpaceFamily{X}{Y}$.
    $F$ is a function on $I$ by
        \auto{binary_coproduct_space_family_type}.
    For every $i\in I$ we have $F(i)$ is a topological space by
        \auto{binary_coproduct_family_is_topological}.
    $\carrier[\BinaryCoproductTopology{X}{Y}]
        =\indexedSum{I}{\binaryCoproductCarrierFamily{X}{Y}}$ by
        \auto{binary_coproduct_topology_carrier}.
    $\binaryCoproductCarrierFamily{X}{Y}
        =\indexedTopologicalCarrierFamily{I}{F}$ by
        \obv{binary_coproduct_carrier_family}.
    $\BinaryCoproductTopology{X}{Y}
        =\IndexedCoproductTopology{I}{F}$ by
        \obv{binary_coproduct_topology}.

    Show that if $U\in\opens[\BinaryCoproductTopology{X}{Y}]$, then
        $U\subseteq\carrier[\BinaryCoproductTopology{X}{Y}]$ and
        $\preimg{\firstCoproductInjection{X}{Y}}{U}\in\opens[X]$ and
        $\preimg{\secondCoproductInjection{X}{Y}}{U}\in\opens[Y]$.
    \begin{subproof}
        Assume $U\in\opens[\BinaryCoproductTopology{X}{Y}]$.
        Hence $U\in\opens[\IndexedCoproductTopology{I}{F}]$.
        Hence $U\subseteq\indexedSum{I}{
            \indexedTopologicalCarrierFamily{I}{F}}$ and
            for every $i\in I$ we have
                $\preimg{\indexedSumInjection{I}{
                    \indexedTopologicalCarrierFamily{I}{F}}{i}}{U}
                \in\opens[F(i)]$ by
            \obv{indexed_coproduct_open_iff}.
        $\emptyset\in I$ and $\suc{\emptyset}\in I$ by
            \auto{suc_iff,emptyset}.
        Hence $\preimg{\indexedSumInjection{I}{
            \indexedTopologicalCarrierFamily{I}{F}}{\emptyset}}{U}
            \in\opens[F(\emptyset)]$ and
            $\preimg{\indexedSumInjection{I}{
                \indexedTopologicalCarrierFamily{I}{F}}{
                \suc{\emptyset}}}{U}
            \in\opens[F(\suc{\emptyset})]$.
        Follows by
            \auto{first_coproduct_injection,second_coproduct_injection,
                binary_coproduct_space_family_left,
                binary_coproduct_space_family_right}.
    \end{subproof}

    Show that if
        $U\subseteq\carrier[\BinaryCoproductTopology{X}{Y}]$ and
        $\preimg{\firstCoproductInjection{X}{Y}}{U}\in\opens[X]$ and
        $\preimg{\secondCoproductInjection{X}{Y}}{U}\in\opens[Y]$, then
        $U\in\opens[\BinaryCoproductTopology{X}{Y}]$.
    \begin{subproof}
        Assume $U\subseteq\carrier[\BinaryCoproductTopology{X}{Y}]$ and
            $\preimg{\firstCoproductInjection{X}{Y}}{U}\in\opens[X]$ and
            $\preimg{\secondCoproductInjection{X}{Y}}{U}\in\opens[Y]$.
        Hence $U\subseteq\indexedSum{I}{
            \indexedTopologicalCarrierFamily{I}{F}}$.
        Show that for every $i\in I$ we have
            $\preimg{\indexedSumInjection{I}{
                \indexedTopologicalCarrierFamily{I}{F}}{i}}{U}
            \in\opens[F(i)]$.
        \begin{subproof}
            Fix $i\in I$.
            $i=\emptyset$ or $i=\suc{\emptyset}$ by
                \auto{suc_elim,emptyset}.
            Follows by
                \auto{first_coproduct_injection,
                    second_coproduct_injection,
                    binary_coproduct_space_family_left,
                    binary_coproduct_space_family_right}.
        \end{subproof}
        Hence $U\in\opens[\IndexedCoproductTopology{I}{F}]$ by
            \auto{indexed_coproduct_open_iff}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{theorem}\label{binary_coproduct_topology_universal_property}
    Let $X,Y,Z$ be topological spaces.
    Let $h\in\funs{\carrier[\BinaryCoproductTopology{X}{Y}]}{
        \carrier[Z]}$.
    Then $h$ is continuous from $\BinaryCoproductTopology{X}{Y}$ to $Z$
        iff $h\circ\firstCoproductInjection{X}{Y}$ is continuous from
            $X$ to $Z$ and
            $h\circ\secondCoproductInjection{X}{Y}$ is continuous from
            $Y$ to $Z$.
\end{theorem}
\begin{proof}
    Let $I=\suc{\suc{\emptyset}}$.
    Let $F=\binaryCoproductSpaceFamily{X}{Y}$.
    $F$ is a function on $I$ by
        \auto{binary_coproduct_space_family_type}.
    For every $i\in I$ we have $F(i)$ is a topological space by
        \auto{binary_coproduct_family_is_topological}.
    $\carrier[\BinaryCoproductTopology{X}{Y}]
        =\indexedSum{I}{\binaryCoproductCarrierFamily{X}{Y}}$ by
        \auto{binary_coproduct_topology_carrier}.
    $\binaryCoproductCarrierFamily{X}{Y}
        =\indexedTopologicalCarrierFamily{I}{F}$ by
        \obv{binary_coproduct_carrier_family}.
    Hence $h\in\funs{
        \indexedSum{I}{\indexedTopologicalCarrierFamily{I}{F}}}{
        \carrier[Z]}$.
    $\BinaryCoproductTopology{X}{Y}
        =\IndexedCoproductTopology{I}{F}$ by
        \obv{binary_coproduct_topology}.

    Show that if $h$ is continuous from
        $\BinaryCoproductTopology{X}{Y}$ to $Z$, then
        $h\circ\firstCoproductInjection{X}{Y}$ is continuous from
            $X$ to $Z$ and
        $h\circ\secondCoproductInjection{X}{Y}$ is continuous from
            $Y$ to $Z$.
    \begin{subproof}
        Assume $h$ is continuous from
            $\BinaryCoproductTopology{X}{Y}$ to $Z$.
        Hence $h$ is continuous from $\IndexedCoproductTopology{I}{F}$
            to $Z$.
        Hence for every $i\in I$ we have
            $h\circ\indexedSumInjection{I}{
                \indexedTopologicalCarrierFamily{I}{F}}{i}$
            is continuous from $F(i)$ to $Z$ by
            \obv{indexed_coproduct_topology_universal_property}.
        $\emptyset\in I$ and $\suc{\emptyset}\in I$ by
            \auto{suc_iff,emptyset}.
        Hence $h\circ\indexedSumInjection{I}{
            \indexedTopologicalCarrierFamily{I}{F}}{\emptyset}$
            is continuous from $F(\emptyset)$ to $Z$ and
            $h\circ\indexedSumInjection{I}{
                \indexedTopologicalCarrierFamily{I}{F}}{
                \suc{\emptyset}}$
            is continuous from $F(\suc{\emptyset})$ to $Z$.
        Follows by
            \auto{first_coproduct_injection,second_coproduct_injection,
                binary_coproduct_space_family_left,
                binary_coproduct_space_family_right}.
    \end{subproof}

    Show that if
        $h\circ\firstCoproductInjection{X}{Y}$ is continuous from
            $X$ to $Z$ and
        $h\circ\secondCoproductInjection{X}{Y}$ is continuous from
            $Y$ to $Z$, then
        $h$ is continuous from $\BinaryCoproductTopology{X}{Y}$ to $Z$.
    \begin{subproof}
        Assume $h\circ\firstCoproductInjection{X}{Y}$ is continuous from
            $X$ to $Z$ and
            $h\circ\secondCoproductInjection{X}{Y}$ is continuous from
            $Y$ to $Z$.
        Show that for every $i\in I$ we have
            $h\circ\indexedSumInjection{I}{
                \indexedTopologicalCarrierFamily{I}{F}}{i}$
            is continuous from $F(i)$ to $Z$.
        \begin{subproof}
            Fix $i\in I$.
            $i=\emptyset$ or $i=\suc{\emptyset}$ by
                \auto{suc_elim,emptyset}.
            Follows by
                \auto{first_coproduct_injection,
                    second_coproduct_injection,
                    binary_coproduct_space_family_left,
                    binary_coproduct_space_family_right}.
        \end{subproof}
        Hence $h$ is continuous from $\IndexedCoproductTopology{I}{F}$
            to $Z$ by
            \obv{indexed_coproduct_topology_universal_property}.
    \end{subproof}
    Follows by \auto.
\end{proof}
