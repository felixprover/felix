The real line can be described by the identity chart, or by two proper
open rays with overlap $(0,1)$. The coordinate construction identifies these
presentations as the same equipped object.

\begin{definition}\label{real_two_charts}
    $\RealTwoCharts=
        \{\identity{\realOpenLowerRay{\surrealOne}},
            \identity{\realOpenUpperRay{\surrealZero}}\}$.
\end{definition}

\begin{classification}\label{real_two_charts_presentation}
    Then $\RealTwoCharts$ is a topological coordinate presentation on $\reals$
        modelled on $\RealTopology$.
\end{classification}
\begin{proof}
    $\RealTopology$ is a topological space by \obv{real_topology_is_topological_space}.
    $\surrealZero,\surrealOne\in\reals$ by
        \obv{surreal_zero_is_real,surreal_one_is_real}.
    $\surrealLt{\surrealZero}{\surrealOne}$ by \obv{surreal_zero_lt_one}.
    $\realOpenLowerRay{\surrealOne}\union\realOpenUpperRay{\surrealZero}=\reals$ by
        \obv{real_open_rays_cover}.
    $\carrier[\RealTopology]=\reals$ by \obv{real_topology_carrier}.
    $\realOpenLowerRay{\surrealOne}\in\opens[\RealTopology]$ and
        $\realOpenUpperRay{\surrealZero}\in\opens[\RealTopology]$ by
        \obv{real_open_lower_ray_is_open,real_open_upper_ray_is_open}.
    Follows by \auto{real_two_charts,two_open_identity_charts}.
\end{proof}

\begin{definition}\label{real_one_chart}
    $\RealOneChart=\{\identity{\reals}\}$.
\end{definition}

\begin{classification}\label{real_one_chart_presentation}
    Then $\RealOneChart$ is a topological coordinate presentation on $\reals$
        modelled on $\RealTopology$.
\end{classification}
\begin{proof}
    $\RealTopology$ is a topological space by \obv{real_topology_is_topological_space}.
    $\carrier[\RealTopology]=\reals$ by \obv{real_topology_carrier}.
    $\reals\in\opens[\RealTopology]$ by \obv{carrier_open}.
    $\identity{\reals}$ is an open partial homeomorphism from $\RealTopology$
        to $\RealTopology$ by \obv{identity_open_partial_homeomorphism}.
    Every element of $\RealOneChart$ is an open partial homeomorphism
        from $\RealTopology$ to $\RealTopology$ by \auto{real_one_chart,singleton_iff}.
    For every $x\in\carrier[\RealTopology]$ there exists $c\in\RealOneChart$
        such that $x\in\dom{c}$ by \auto{real_one_chart,singleton_iff,id_dom}.
    Follows by \auto{open_chart_cover_is_coordinate_presentation}.
\end{proof}

\begin{theorem}\label{real_presentations_cross_compatible}
    Then for every $c\in\RealTwoCharts$ we have for every $d\in\RealOneChart$ we have
        $c$ is topologically compatible with $d$ on $\reals$ modelled on $\RealTopology$.
\end{theorem}
\begin{proof}
    $\RealTopology$ is a topological space by \obv{real_topology_is_topological_space}.
    $\carrier[\RealTopology]=\reals$ by \obv{real_topology_carrier}.
    $\surrealZero,\surrealOne\in\reals$ by
        \obv{surreal_zero_is_real,surreal_one_is_real}.
    $\realOpenLowerRay{\surrealOne}\in\opens[\RealTopology]$ and
        $\realOpenUpperRay{\surrealZero}\in\opens[\RealTopology]$ by
        \obv{real_open_lower_ray_is_open,real_open_upper_ray_is_open}.
    $\reals\in\opens[\RealTopology]$ by \obv{carrier_open}.
    Every element of $\RealTwoCharts$ is an open partial homeomorphism
        from $\RealTopology$ to $\RealTopology$ by
        \auto{real_two_charts,cons_iff,identity_open_partial_homeomorphism}.
    Every element of $\RealOneChart$ is an open partial homeomorphism
        from $\RealTopology$ to $\RealTopology$ by
        \auto{real_one_chart,singleton_iff,identity_open_partial_homeomorphism}.
    Follows by \auto{open_partial_homeomorphisms_are_compatible}.
\end{proof}

\begin{theorem}\label{real_chart_spaces_equal}
    Then $\SpaceFromCharts{\reals}{\RealTopology}{\RealTwoCharts}
        =\SpaceFromCharts{\reals}{\RealTopology}{\RealOneChart}$.
\end{theorem}
\begin{proof}
    $\RealTwoCharts$ is a topological coordinate presentation on $\reals$
        modelled on $\RealTopology$ by \obv{real_two_charts_presentation}.
    $\RealOneChart$ is a topological coordinate presentation on $\reals$
        modelled on $\RealTopology$ by \obv{real_one_chart_presentation}.
    For every $c\in\RealTwoCharts$ we have for every $d\in\RealOneChart$ we have
        $c$ is topologically compatible with $d$ on $\reals$ modelled on $\RealTopology$
        by \obv{real_presentations_cross_compatible}.
    Follows by \obv{space_from_cross_compatible_charts}.
\end{proof}

\begin{theorem}\label{real_two_chart_space_coordinates}
    Then $\SpaceFromCharts{\reals}{\RealTopology}{\RealTwoCharts}$
        has coordinates under $\FullPseudogroup{\RealTopology}$.
\end{theorem}
\begin{proof}
    $\RealTwoCharts$ is a topological coordinate presentation on $\reals$
        modelled on $\RealTopology$ by \obv{real_two_charts_presentation}.
    Follows by \obv{space_from_charts_coordinates}.
\end{proof}

\begin{theorem}\label{real_continuous_calculus}
    Then $f$ is admissible from $\RealTopology$ to $\RealTopology$ in
        $\ContinuousCalculus{\{\RealTopology\}}$ iff
        $f$ is a continuous partial map from $\RealTopology$ to $\RealTopology$.
\end{theorem}
\begin{proof}
    $\RealTopology$ is a topological space by \obv{real_topology_is_topological_space}.
    Follows by \obv{continuous_single_model_admissibility}.
\end{proof}

\subsection{The zero-dimensional model}

A zero-coordinate vector is the empty function. Its function space is a
singleton, equipped here with its discrete topology. No coordinate is evaluated.

\begin{definition}\label{zero_coordinate_model}
    $\ZeroCoordinateModel=\discreteTopology{\funs{\emptyset}{\reals}}$.
\end{definition}

\begin{theorem}\label{zero_coordinate_model_is_real_coordinate_space}
    $\ZeroCoordinateModel=\RealCoordinateSpace{\emptyset}$.
\end{theorem}
\begin{proof}
    Follows by \auto{zero_coordinate_model,real_coordinate_space_zero,
        empty_domain_function_space}.
\end{proof}

\begin{classification}\label{zero_coordinate_model_is_topological_space}
    Then $\ZeroCoordinateModel$ is a topological space.
\end{classification}
\begin{proof}
    Follows by \auto{zero_coordinate_model,discrete_topology_is_topological_space}.
\end{proof}

\begin{theorem}\label{zero_coordinate_model_carrier}
    Then $\carrier[\ZeroCoordinateModel]=\{\emptyset\}$.
\end{theorem}
\begin{proof}
    Follows by \auto{zero_coordinate_model,discrete_topology_carrier,
        empty_domain_function_space}.
\end{proof}

\begin{definition}\label{zero_coordinate_charts}
    $\ZeroCoordinateCharts=\{\identity{\{\emptyset\}}\}$.
\end{definition}

\begin{classification}\label{zero_coordinate_charts_presentation}
    Then $\ZeroCoordinateCharts$ is a topological coordinate presentation on
        $\{\emptyset\}$ modelled on $\ZeroCoordinateModel$.
\end{classification}
\begin{proof}
    $\ZeroCoordinateModel$ is a topological space by
        \obv{zero_coordinate_model_is_topological_space}.
    $\carrier[\ZeroCoordinateModel]=\{\emptyset\}$ by
        \obv{zero_coordinate_model_carrier}.
    $\{\emptyset\}\in\opens[\ZeroCoordinateModel]$ by \obv{carrier_open}.
    $\identity{\{\emptyset\}}$ is an open partial homeomorphism from
        $\ZeroCoordinateModel$ to $\ZeroCoordinateModel$ by
        \obv{identity_open_partial_homeomorphism}.
    Every element of $\ZeroCoordinateCharts$ is an open partial homeomorphism
        from $\ZeroCoordinateModel$ to $\ZeroCoordinateModel$ by
        \auto{zero_coordinate_charts,singleton_iff}.
    For every $x\in\carrier[\ZeroCoordinateModel]$ there exists
        $c\in\ZeroCoordinateCharts$ such that $x\in\dom{c}$ by
        \auto{zero_coordinate_charts,singleton_iff,id_dom}.
    Follows by \auto{open_chart_cover_is_coordinate_presentation}.
\end{proof}

\begin{theorem}\label{zero_coordinate_chart_space}
    Then $\SpaceFromCharts{\{\emptyset\}}{\ZeroCoordinateModel}{\ZeroCoordinateCharts}$
        has coordinates under $\FullPseudogroup{\ZeroCoordinateModel}$ and
        $\carrier[\SpaceFromCharts{\{\emptyset\}}{\ZeroCoordinateModel}{\ZeroCoordinateCharts}]
            =\{\emptyset\}$.
\end{theorem}
\begin{proof}
    $\ZeroCoordinateCharts$ is a topological coordinate presentation on
        $\{\emptyset\}$ modelled on $\ZeroCoordinateModel$ by
        \obv{zero_coordinate_charts_presentation}.
    Follows by \obv{space_from_charts_coordinates,space_from_charts_carrier}.
\end{proof}

\subsection{The empty source}

\begin{theorem}\label{empty_chart_space}
    Let $E$ be a topological space.
    Then $\SpaceFromCharts{\emptyset}{E}{\emptyset}$ has coordinates under
        $\FullPseudogroup{E}$ and
        $\carrier[\SpaceFromCharts{\emptyset}{E}{\emptyset}]=\emptyset$ and
        $\atlas[\SpaceFromCharts{\emptyset}{E}{\emptyset}]=\{\emptyset\}$.
\end{theorem}
\begin{proof}
    $\emptyset$ is a topological coordinate presentation on $\emptyset$ modelled on $E$
        by \obv{empty_coordinate_presentation}.
    $\SpaceFromCharts{\emptyset}{E}{\emptyset}$ has coordinates under
        $\FullPseudogroup{E}$ by \obv{space_from_charts_coordinates}.
    $\carrier[\SpaceFromCharts{\emptyset}{E}{\emptyset}]=\emptyset$ by
        \obv{space_from_charts_carrier}.
    Let $X=\CoordinateTopology{\emptyset}{E}{\emptyset}$.
    $X$ is a topological space by \auto{coordinate_presentation_topology}.
    $\carrier[X]=\emptyset$ by \auto{coordinate_topology_carrier}.
    $\atlas[\SpaceFromCharts{\emptyset}{E}{\emptyset}]
        =\openPartialHomeomorphisms{X}{E}$ by \auto{space_from_charts_atlas}.
    Every element of $\openPartialHomeomorphisms{X}{E}$ is equal to $\emptyset$ by
        \auto{open_partial_homeomorphisms_iff,empty_source_chart}.
    $\emptyset$ is an open partial homeomorphism from $X$ to $E$ by
        \obv{empty_open_partial_homeomorphism}.
    $\emptyset\in\openPartialHomeomorphisms{X}{E}$ by
        \auto{open_partial_homeomorphisms_iff}.
    $\openPartialHomeomorphisms{X}{E}=\{\emptyset\}$ by \auto{setext,singleton_iff}.
    Follows by \auto.
\end{proof}
