\section{Presentations and charts at a point}

\import{topology/coordinate-structure.tex}

A presentation is a covering atlas that generates the stored maximal atlas.
It is working data, not an additional field of the coordinate structure.

\begin{definition}\label{coordinate_presentation}
    Let $G$ be a coordinate pseudogroup.
    Let $X$ be a topological space.
    Let $S$ be a coordinate datum.
    $A$ presents $S$ on $X$ under $G$ iff
        $A$ is a coordinate atlas on $X$ under $G$ and
        $\atlas[S]=\MaxAtlas{G}{X}{A}$.
\end{definition}

\begin{classification}\label{presentation_is_atlas}
    Let $G$ be a coordinate pseudogroup.
    Let $X$ be a topological space.
    Let $S$ be a coordinate datum.
    Suppose $A$ presents $S$ on $X$ under $G$.
    Then $A$ is a coordinate atlas on $X$ under $G$.
\end{classification}
\begin{proof}
    Follows by \obv{coordinate_presentation}.
\end{proof}

\begin{classification}\label{presented_coordinates}
    Let $G$ be a coordinate pseudogroup.
    Let $X$ be a topological space.
    Let $S$ be a coordinate datum.
    Suppose $A$ presents $S$ on $X$ under $G$.
    Then $S$ is a relative coordinate structure on $X$ under $G$.
\end{classification}
\begin{proof}
    $A$ is a coordinate atlas on $X$ under $G$ and
        $\atlas[S]=\MaxAtlas{G}{X}{A}$ by \obv{coordinate_presentation}.
    $\MaxAtlas{G}{X}{A}$ is a maximal coordinate atlas on $X$ under $G$ by
        \obv{max_atlas_is_maximal}.
    Follows by \auto{relative_coordinate_structure}.
\end{proof}

\begin{theorem}\label{coordinate_presentation_same_fields}
    Let $G$ be a coordinate pseudogroup.
    Let $X,Y$ be topological spaces.
    Let $S,T$ be coordinate data.
    Suppose $\carrier[X]=\carrier[Y]$ and $\opens[X]=\opens[Y]$ and $\atlas[S]=\atlas[T]$.
    Then $A$ presents $S$ on $X$ under $G$ iff $A$ presents $T$ on $Y$ under $G$.
\end{theorem}
\begin{proof}
    $A$ is a coordinate atlas on $X$ under $G$ iff $A$ is a coordinate atlas on $Y$ under $G$ by
        \obv{coordinate_atlas_same_fields}.
    $\compatibleHull{G}{X}{A}=\compatibleHull{G}{Y}{A}$ by \obv{compatible_hull_same_fields}.
    Follows by \auto{coordinate_presentation,max_atlas}.
\end{proof}

\begin{theorem}\label{equip_presentation_iff}
    Let $G$ be a coordinate pseudogroup.
    Let $X$ be a topological space.
    Let $S$ be a coordinate datum.
    Then $A$ presents $S$ on $X$ under $G$ iff
        $A$ presents $\Equip{X}{S}$ on $\Equip{X}{S}$ under $G$.
\end{theorem}
\begin{proof}
    $\Equip{X}{S}$ is a topological space and $\Equip{X}{S}$ is a coordinate datum by
        \obv{equip,equipped_coordinate_space}.
    $\carrier[\Equip{X}{S}]=\carrier[X]$ and $\opens[\Equip{X}{S}]=\opens[X]$ and
        $\atlas[\Equip{X}{S}]=\atlas[S]$ by
        \obv{equipprojection_carrier,equipprojection_opens,equipprojection_atlas}.
    Follows by \obv{coordinate_presentation_same_fields}.
\end{proof}

\begin{theorem}\label{generated_coordinates_presentation}
    Let $G$ be a coordinate pseudogroup.
    Let $X$ be a topological space.
    Let $A$ be a coordinate atlas on $X$ under $G$.
    Then $A$ presents $\GeneratedCoordinates{G}{X}{A}$ on $X$ under $G$.
\end{theorem}
\begin{proof}
    $\GeneratedCoordinates{G}{X}{A}$ is a coordinate datum by
        \auto{generated_coordinates,relative_coordinate_structure}.
    Follows by \obv{coordinate_presentation,generated_coordinatesprojection_atlas}.
\end{proof}

\begin{theorem}\label{presentation_chart_inclusion}
    Let $G$ be a coordinate pseudogroup.
    Let $X$ be a topological space.
    Let $S$ be a coordinate datum.
    Suppose $A$ presents $S$ on $X$ under $G$.
    Then $A\subseteq\atlas[S]$.
\end{theorem}
\begin{proof}
    $A$ is a coordinate atlas on $X$ under $G$ and
        $\atlas[S]=\MaxAtlas{G}{X}{A}$ by \obv{coordinate_presentation}.
    Follows by \auto{atlas_subset_max_atlas}.
\end{proof}

\begin{theorem}\label{presented_chart_iff}
    Let $G$ be a coordinate pseudogroup.
    Let $X$ be a topological space.
    Let $S$ be a coordinate datum.
    Suppose $A$ presents $S$ on $X$ under $G$.
    Then $c\in\atlas[S]$ iff
        $c$ is an open partial homeomorphism from $X$ to $\model[G]$ and
        for every $a\in A$ we have $c$ is compatible with $a$ on $X$ under $G$.
\end{theorem}
\begin{proof}
    $A$ is a coordinate atlas on $X$ under $G$ and
        $\atlas[S]=\MaxAtlas{G}{X}{A}$ by \obv{coordinate_presentation}.
    Follows by \auto{max_atlas_member_iff}.
\end{proof}

\begin{theorem}\label{exact_presented_coordinates}
    Let $G$ be a coordinate pseudogroup.
    Let $X$ be a topological space.
    Suppose $S$ is an exact coordinate datum.
    Suppose $A$ presents $S$ on $X$ under $G$.
    Then $\GeneratedCoordinates{G}{X}{A}=S$.
\end{theorem}
\begin{proof}
    $A$ is a coordinate atlas on $X$ under $G$ and
        $\atlas[S]=\MaxAtlas{G}{X}{A}$ by \obv{coordinate_presentation}.
    $\GeneratedCoordinates{G}{X}{A}$ is an exact relative coordinate structure
        on $X$ under $G$ by \obv{generated_coordinates_is_exact}.
    $\atlas[\GeneratedCoordinates{G}{X}{A}]=\atlas[S]$ by
        \auto{generated_coordinatesprojection_atlas}.
    Follows by structure extensionality.
\end{proof}

\begin{theorem}\label{relative_atlas_presentation}
    Let $G$ be a coordinate pseudogroup.
    Let $X$ be a topological space.
    Let $S$ be a relative coordinate structure on $X$ under $G$.
    Then $\atlas[S]$ presents $S$ on $X$ under $G$.
\end{theorem}
\begin{proof}
    $\atlas[S]$ is a maximal coordinate atlas on $X$ under $G$ by
        \obv{relative_atlas_maximal}.
    $\atlas[S]$ is a coordinate atlas on $X$ under $G$ by
        \obv{maximal_coordinate_atlas}.
    $\atlas[S]=\MaxAtlas{G}{X}{\atlas[S]}$ by
        \obv{maximal_coordinate_atlas_fixed_point}.
    Follows by \obv{coordinate_presentation}.
\end{proof}

\begin{theorem}\label{same_coordinate_presentations_iff}
    Let $G$ be a coordinate pseudogroup.
    Let $X$ be a topological space.
    Let $S$ be a coordinate datum.
    Suppose $A$ presents $S$ on $X$ under $G$.
    Let $B$ be a coordinate atlas on $X$ under $G$.
    Then $B$ presents $S$ on $X$ under $G$ iff
        for every $a\in A$ we have for every $b\in B$ we have
            $a$ is compatible with $b$ on $X$ under $G$.
\end{theorem}
\begin{proof}
    $A$ is a coordinate atlas on $X$ under $G$ and
        $\atlas[S]=\MaxAtlas{G}{X}{A}$ by \obv{coordinate_presentation}.
    Follows by \auto{coordinate_presentation,max_atlas_equality_iff}.
\end{proof}

\subsection{Charts at a point}

\begin{definition}\label{charts_at}
    Let $S$ be a coordinate datum.
    $\ChartsAt{S}{p}=\{c\in\atlas[S]\mid p\in\dom{c}\}$.
\end{definition}

\begin{theorem}\label{charts_at_iff}
    Let $S$ be a coordinate datum.
    Then $c\in\ChartsAt{S}{p}$ iff $c\in\atlas[S]$ and $p\in\dom{c}$.
\end{theorem}
\begin{proof}
    Follows by \obv{charts_at}.
\end{proof}

\begin{theorem}\label{charts_at_inhabited_iff}
    Let $G$ be a coordinate pseudogroup.
    Let $X$ be a topological space.
    Let $S$ be a relative coordinate structure on $X$ under $G$.
    Then $\ChartsAt{S}{p}$ is inhabited iff $p\in\carrier[X]$.
\end{theorem}
\begin{proof}
    $S$ is a coordinate datum by \obv{relative_coordinate_structure}.
    $\atlas[S]$ is a coordinate atlas on $X$ under $G$ by
        \auto{relative_atlas_maximal,maximal_coordinate_atlas}.
    If $p\in\carrier[X]$, then $\ChartsAt{S}{p}$ is inhabited by
        \auto{coordinate_atlas,charts_at_iff}.
    Show that if $\ChartsAt{S}{p}$ is inhabited, then $p\in\carrier[X]$.
    \begin{subproof}
        Assume $\ChartsAt{S}{p}$ is inhabited.
        Take $c$ such that $c\in\ChartsAt{S}{p}$.
        $c\in\atlas[S]$ and $p\in\dom{c}$ by \obv{charts_at_iff}.
        $c$ is an open partial homeomorphism from $X$ to $\model[G]$ by
            \obv{coordinate_atlas}.
        $\dom{c}\in\opens[X]$ by \obv{open_partial_homeomorphism}.
        $\dom{c}\subseteq\carrier[X]$ by \obv{opens_type}.
        Follows by \obv{subseteq}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{theorem}\label{presentation_chart_at_point}
    Let $G$ be a coordinate pseudogroup.
    Let $X$ be a topological space.
    Let $S$ be a coordinate datum.
    Suppose $A$ presents $S$ on $X$ under $G$.
    Let $p\in\carrier[X]$.
    Then there exists $c\in A$ such that $c\in\ChartsAt{S}{p}$.
\end{theorem}
\begin{proof}
    $A$ is a coordinate atlas on $X$ under $G$ by \obv{presentation_is_atlas}.
    Take $c\in A$ such that $p\in\dom{c}$ by \obv{coordinate_atlas}.
    $A\subseteq\atlas[S]$ by \obv{presentation_chart_inclusion}.
    Follows by \auto{charts_at_iff,subseteq}.
\end{proof}

\begin{theorem}\label{equipped_charts_at_inhabited_iff}
    Let $G$ be a coordinate pseudogroup.
    Let $M$ be an equipped coordinate space.
    Suppose $M$ has coordinates under $G$.
    Then $\ChartsAt{M}{p}$ is inhabited iff $p\in\carrier[M]$.
\end{theorem}
\begin{proof}
    $\TopologicalReduct{M}$ is a topological space by
        \obv{topological_reduct_is_topological_space}.
    $\CoordinateStructure{M}$ is a relative coordinate structure
        on $\TopologicalReduct{M}$ under $G$ by \obv{coordinate_structure_relative}.
    $\atlas[\CoordinateStructure{M}]=\atlas[M]$ by
        \obv{coordinate_structureprojection_atlas}.
    $\ChartsAt{\CoordinateStructure{M}}{p}=\ChartsAt{M}{p}$ by \obv{charts_at}.
    $\carrier[\TopologicalReduct{M}]=\carrier[M]$ by
        \obv{topological_reduct_carrier}.
    Follows by \auto{charts_at_inhabited_iff}.
\end{proof}
