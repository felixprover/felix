\section{Second countability of subspaces}

\import{topology/countability-axioms.tex}
\import{topology/subspace-basis.tex}

\begin{theorem}\label{subspace_of_second_countable_space}
    Let $X$ be a second-countable topological space.
    Then $\subspaceTopology{X}{A}$ is second-countable.
\end{theorem}
\begin{proof}
    Take $B$ such that $B$ is countable and $B$ is an open basis of $X$
        by \obv{is_second_countable}.
    Let $S=\subspaceTopology{X}{A}$.
    Let $C=\{U\inter\carrier[S]\mid U\in B\}$.
    $\cardLe{C}{\naturals}$ by \auto{countable_replacement}.
    Hence $C$ is countable by \obv{cardinal_le,at_most_countable}.
    $C=\subspaceBasis{X}{A}{B}$ by \auto{subspace_basis_iff,setext,inter,subseteq}.
    $\subspaceBasis{X}{A}{B}$ is an open basis of $S$ by
        \auto{open_basis_restricts_to_subspace}.
    $S$ is a topological space by \obv{subspace_topology_is_topological_space}.
    Follows by \auto{is_second_countable}.
\end{proof}
