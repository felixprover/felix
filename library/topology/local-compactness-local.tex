\section{Local compactness on open neighbourhoods}

\import{topology/local-compactness.tex}

Compactness of a subset does not depend on whether it is viewed inside a
subspace or the ambient space. Consequently, local compactness can be checked
on an open neighbourhood of each point, without a separation hypothesis.

\begin{theorem}\label{compact_in_subspace_iff}
    Let $X$ be a topological space.
    Suppose $K\subseteq A$ and $A\subseteq\carrier[X]$.
    Then $K$ is compact in $\subspaceTopology{X}{A}$ iff $K$ is compact in $X$.
\end{theorem}
\begin{proof}
    $\carrier[\subspaceTopology{X}{A}]=A$ by
        \obv{subspace_topology_carrier_of_subset}.
    $K\subseteq\carrier[X]$ by \obv{subseteq_transitive}.
    $\subspaceTopology{\subspaceTopology{X}{A}}{K}=\subspaceTopology{X}{K}$ by
        \obv{nested_subspace_topology}.
    Follows by \auto{compact_in,subspace_topology_is_topological_space}.
\end{proof}

\begin{theorem}\label{locally_compact_from_open_neighbourhoods}
    Let $X$ be a topological space.
    Suppose for every $x\in\carrier[X]$ there exists $A\in\opens[X]$ such that
        $x\in A$ and $\subspaceTopology{X}{A}$ is locally compact.
    Then $X$ is locally compact.
\end{theorem}
\begin{proof}
    Show that for every $x\in\carrier[X]$ there exist $U,K$ such that
        $U\in\opens[X]$ and $x\in U\subseteq K$ and $K$ is compact in $X$.
    \begin{subproof}
        Fix $x\in\carrier[X]$.
        Take $A\in\opens[X]$ such that $x\in A$ and
            $\subspaceTopology{X}{A}$ is locally compact.
        Let $S=\subspaceTopology{X}{A}$.
        $A\subseteq\carrier[X]$ by \auto{opens_type,subseteq}.
        $\carrier[S]=A$ by \auto{subspace_topology_carrier_of_subset}.
        Take $U,K$ such that $U\in\opens[S]$ and $x\in U\subseteq K$ and
            $K$ is compact in $S$ by \auto{is_locally_compact}.
        $U\in\opens[X]$ by \auto{open_subspace_open_iff}.
        $K\subseteq A$ by \obv{compact_in}.
        $K$ is compact in $X$ by \auto{compact_in_subspace_iff}.
        Follows by \auto.
    \end{subproof}
    Follows by \obv{is_locally_compact}.
\end{proof}
