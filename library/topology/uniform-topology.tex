\section{The topology induced by a uniform structure}

\begin{definition}\label{uniform_topology}
    $\uniformTopology{X}=
        \TopologicalSpaceOn{\carrier[X]}{\uniformOpens{X}}$.
\end{definition}

\begin{theorem}\label{uniform_topology_is_topological_space}
    Let $X$ be a uniform space.
    Then $\uniformTopology{X}$ is a topological space.
\end{theorem}
\begin{proof}
    $\uniformOpens{X}$ is a topology on $\carrier[X]$
        by \obv{uniform_opens_form_topology}.
    Follows by
        \auto{uniform_topology,topological_space_on_is_topological_space}.
\end{proof}

\begin{proposition}\label{uniform_topology_carrier}
    $\carrier[\uniformTopology{X}]=\carrier[X]$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{uniform_topology,topological_space_on_carrier}.
\end{proof}

\begin{proposition}\label{uniform_topology_opens}
    $\opens[\uniformTopology{X}]=\uniformOpens{X}$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{uniform_topology,topological_space_on_opens}.
\end{proof}

\begin{theorem}\label{uniform_ball_contains_open_neighbourhood}
    Let $X$ be a uniform space, $U\in\uniformity[X]$, and
        $x\in\carrier[X]$.
    Then there exists $A\in\opens[\uniformTopology{X}]$ such that
        $x\in A\subseteq\uniformBall{X}{U}{x}$.
\end{theorem}
\begin{proof}
    Let $A=\{y\in\carrier[X]\mid
        \text{there exists $V\in\uniformity[X]$ such that
            $\uniformBall{X}{V}{y}\subseteq\uniformBall{X}{U}{x}$}\}$.
    $A\subseteq\carrier[X]$ by \obv{subseteq}.

    Show that $x\in A$.
    \begin{subproof}
        Take $V\in\uniformity[X]$ such that $V\circ V\subseteq U$
            by \obv{uniformity_square}.
        Show that $\uniformBall{X}{V}{x}\subseteq\uniformBall{X}{U}{x}$.
        \begin{subproof}
            It suffices to show that every element of
                $\uniformBall{X}{V}{x}$ is an element of
                $\uniformBall{X}{U}{x}$ by \obv{subseteq}.
            Fix $y\in\uniformBall{X}{V}{x}$.
            $(x,x)\in V$ by \obv{uniform_entourage_reflexive}.
            $(x,y)\in V$ by \obv{uniform_ball_iff}.
            Hence $(x,y)\in V\circ V$ by \obv{circ_elem_intro}.
            Hence $(x,y)\in U$ by \obv{subseteq}.
            Follows by \obv{uniform_ball_iff}.
        \end{subproof}
    \end{subproof}

    Show that $A\subseteq\uniformBall{X}{U}{x}$.
    \begin{subproof}
        It suffices to show that every element of $A$ is an element of
            $\uniformBall{X}{U}{x}$ by \obv{subseteq}.
        Fix $y\in A$.
        Take $V\in\uniformity[X]$ such that
            $\uniformBall{X}{V}{y}\subseteq\uniformBall{X}{U}{x}$.
        $y\in\uniformBall{X}{V}{y}$
            by \obv{uniform_ball_contains_centre}.
        Follows by \obv{subseteq}.
    \end{subproof}

    Show that $\uniformOpen{X}{A}$.
    \begin{subproof}
        Show that for every $y\in A$ there exists $W\in\uniformity[X]$
            such that $\uniformBall{X}{W}{y}\subseteq A$.
        \begin{subproof}
            Fix $y\in A$.
            Take $V\in\uniformity[X]$ such that
                $\uniformBall{X}{V}{y}\subseteq\uniformBall{X}{U}{x}$.
            Take $W\in\uniformity[X]$ such that $W\circ W\subseteq V$
                by \obv{uniformity_square}.
            Show that $\uniformBall{X}{W}{y}\subseteq A$.
            \begin{subproof}
                It suffices to show that every element of
                    $\uniformBall{X}{W}{y}$ is an element of $A$
                    by \obv{subseteq}.
                Fix $z\in\uniformBall{X}{W}{y}$.
                $\uniformBall{X}{W}{y}\subseteq\carrier[X]$
                    by \obv{uniform_ball_subset_carrier}.
                Hence $z\in\carrier[X]$ by \obv{subseteq}.
                Show that $\uniformBall{X}{W}{z}
                    \subseteq\uniformBall{X}{U}{x}$.
                \begin{subproof}
                    It suffices to show that every element of
                        $\uniformBall{X}{W}{z}$ is an element of
                        $\uniformBall{X}{U}{x}$ by \obv{subseteq}.
                    Fix $w\in\uniformBall{X}{W}{z}$.
                    $(y,z)\in W$ and $(z,w)\in W$
                        by \obv{uniform_ball_iff}.
                    Hence $(y,w)\in W\circ W$ by \obv{circ_elem_intro}.
                    Hence $(y,w)\in V$ by \obv{subseteq}.
                    Hence $w\in\uniformBall{X}{V}{y}$
                        by \obv{uniform_ball_iff}.
                    Follows by \obv{subseteq}.
                \end{subproof}
            \end{subproof}
        \end{subproof}
        Follows by \obv{uniform_open}.
    \end{subproof}
    Hence $A\in\uniformOpens{X}$ by \obv{uniform_opens_iff}.
    Hence $A\in\opens[\uniformTopology{X}]$
        by \obv{uniform_topology_opens}.
\end{proof}

\begin{theorem}\label{separated_uniform_iff_uniform_topology_hausdorff}
    Let $X$ be a uniform space.
    Then $X$ is separated uniform iff $\uniformTopology{X}$ is Hausdorff.
\end{theorem}
\begin{proof}
    $\uniformTopology{X}$ is a topological space
        by \obv{uniform_topology_is_topological_space}.
    $\carrier[\uniformTopology{X}]=\carrier[X]$
        by \obv{uniform_topology_carrier}.
    $\opens[\uniformTopology{X}]=\uniformOpens{X}$
        by \obv{uniform_topology_opens}.
    $\carrier[X]\times\carrier[X]\in\uniformity[X]$
        by \obv{uniform_top_entourage}.
    Hence $\uniformity[X]$ is inhabited.

    Show that if $X$ is separated uniform, then
        $\uniformTopology{X}$ is Hausdorff.
    \begin{subproof}
        Assume $X$ is separated uniform.
        $\inters{\uniformity[X]}=\identity{\carrier[X]}$
            by \obv{separated_uniform_space}.
        Show that for all $x,y\in\carrier[\uniformTopology{X}]$ such that
            $x\neq y$ there exist $A,B\in\opens[\uniformTopology{X}]$
            such that $x\in A$ and $y\in B$ and $A$ is disjoint from $B$.
        \begin{subproof}
            Fix $x,y\in\carrier[\uniformTopology{X}]$.
            Assume $x\neq y$.
            Hence $x,y\in\carrier[X]$.
            $(x,y)\notin\identity{\carrier[X]}$ by \auto{id_iff}.
            Hence $(x,y)\notin\inters{\uniformity[X]}$.
            Take $U\in\uniformity[X]$ such that $(x,y)\notin U$
                by \auto{inters_iff_forall}.
            Take $W\in\uniformity[X]$ such that $W\circ W\subseteq U$
                by \obv{uniformity_square}.
            Let $V=W\inter\converse{W}$.
            $\converse{W}\in\uniformity[X]$ by \obv{uniformity_converse}.
            Hence $V\in\uniformity[X]$
                by \auto{uniform_entourage_intersection}.

            Show that $\uniformBall{X}{V}{x}$ is disjoint from
                $\uniformBall{X}{V}{y}$.
            \begin{subproof}
                Suppose not.
                Take $z$ such that $z\in\uniformBall{X}{V}{x}$ and
                    $z\in\uniformBall{X}{V}{y}$ by \auto{disjoint}.
                Hence $(x,z)\in V$ and $(y,z)\in V$
                    by \obv{uniform_ball_iff}.
                $(x,z)\in W$ by \auto{inter}.
                $(y,z)\in\converse{W}$ by \auto{inter}.
                Hence $(z,y)\in W$ by \obv{converse_iff}.
                Hence $(x,y)\in W\circ W$ by \obv{circ_elem_intro}.
                Hence $(x,y)\in U$ by \obv{subseteq}.
                Contradiction.
            \end{subproof}

            Take $A\in\opens[\uniformTopology{X}]$ such that
                $x\in A\subseteq\uniformBall{X}{V}{x}$
                by \obv{uniform_ball_contains_open_neighbourhood}.
            Take $B\in\opens[\uniformTopology{X}]$ such that
                $y\in B\subseteq\uniformBall{X}{V}{y}$
                by \obv{uniform_ball_contains_open_neighbourhood}.
            $A$ is disjoint from $B$ by \auto{disjoint,subseteq,emptyset}.
            Follows by \auto.
        \end{subproof}
        Follows by \obv{is_hausdorff}.
    \end{subproof}

    Show that if $\uniformTopology{X}$ is Hausdorff, then
        $X$ is separated uniform.
    \begin{subproof}
        Assume $\uniformTopology{X}$ is Hausdorff.
        $\identity{\carrier[X]}\subseteq\inters{\uniformity[X]}$
            by \obv{inters_greatest,uniformity_diagonal}.
        Show that $\inters{\uniformity[X]}
            \subseteq\identity{\carrier[X]}$.
        \begin{subproof}
            It suffices to show that every element of
                $\inters{\uniformity[X]}$ is an element of
                $\identity{\carrier[X]}$ by \obv{subseteq}.
            Fix $p\in\inters{\uniformity[X]}$.
            Hence $p\in\carrier[X]\times\carrier[X]$
                by \obv{inters_destr}.
            Take $x,y\in\carrier[X]$ such that $p=(x,y)$
                by \obv{times_elem_is_tuple}.
            Show that $x=y$.
            \begin{subproof}
                Suppose not.
                Hence $x,y\in\carrier[\uniformTopology{X}]$.
                Take $A,B\in\opens[\uniformTopology{X}]$ such that
                    $x\in A$ and $y\in B$ and $A$ is disjoint from $B$
                    by \obv{is_hausdorff}.
                $A\in\uniformOpens{X}$ by \obv{uniform_topology_opens}.
                Hence $\uniformOpen{X}{A}$ by \obv{uniform_opens_iff}.
                Take $U\in\uniformity[X]$ such that
                    $\uniformBall{X}{U}{x}\subseteq A$
                    by \obv{uniform_open}.
                $(x,y)\in\inters{\uniformity[X]}$.
                Hence $(x,y)\in U$ by \obv{inters_destr}.
                Hence $y\in\uniformBall{X}{U}{x}$
                    by \obv{uniform_ball_iff}.
                Hence $y\in A$ by \obv{subseteq}.
                Contradiction by \auto{disjoint,emptyset}.
            \end{subproof}
            Hence $p=(x,x)$.
            Follows by \obv{id_iff}.
        \end{subproof}
        Hence $\inters{\uniformity[X]}=\identity{\carrier[X]}$
            by \obv{subseteq_antisymmetric}.
        Follows by \obv{separated_uniform_space}.
    \end{subproof}
    Follows by \auto.
\end{proof}
