The boundedness half of Heine--Borel is an open-cover argument.  Traces of
unit balls cover a compact real subspace; a finite union of those balls is
bounded.  This formulation keeps the finite-union fact available for later
analysis rather than hiding it inside the final equivalence.

\begin{theorem}\label{compact_real_subset_is_bounded}
    Let $X$ be a topological space.
    Suppose $X=\RealTopology$.
    Suppose $K$ is compact in $X$.
    Then $K$ is real bounded.
\end{theorem}
\begin{proof}
    $K\subseteq\carrier[X]$ and $\subspaceTopology{X}{K}$ is compact by
        \obv{compact_in}.
    Let $S=\subspaceTopology{X}{K}$.
    $\carrier[X]=\reals$ by \obv{real_topology_carrier}.
    Hence $K\subseteq\reals$.
    $\carrier[\subspaceTopology{X}{K}]=K$ by
        \obv{subspace_topology_carrier_of_subset}.
    Hence $\carrier[S]=K$.
    $S$ is compact.

    Let $C=\{U\in\opens[S]\mid
        \text{there exists $a\in K$ such that
            $U=\metricBall{\RealMetric}{a}{\surrealOne}\inter K$}\}$.
    $C\subseteq\opens[S]$ by \obv{subseteq}.
    Show that $C$ covers $\carrier[S]$.
    \begin{subproof}
        Show that for every $x\in\carrier[S]$ there exists $U\in C$
            such that $x\in U$.
        \begin{subproof}
            Fix $x\in\carrier[S]$.
            Hence $x\in K$.
            Thus $x\in\reals$ by \obv{subseteq}.
            Let $U=\metricBall{\RealMetric}{x}{\surrealOne}\inter K$.
            $\metricBall{\RealMetric}{x}{\surrealOne}
                \in\opens[\RealTopology]$ and
                $x\in\metricBall{\RealMetric}{x}{\surrealOne}$ by
                \obv{real_unit_ball_is_open_neighbourhood}.
            Hence $\metricBall{\RealMetric}{x}{\surrealOne}
                \in\opens[X]$.
            $U=\metricBall{\RealMetric}{x}{\surrealOne}\inter
                \carrier[S]$.
            Hence $U\in\opens[S]$ by
                \auto{subspace_open_iff}.
            There exists $a\in K$ such that
                $U=\metricBall{\RealMetric}{a}{\surrealOne}\inter K$.
            Thus $U\in C$.
            $x\in U$ by \obv{inter}.
        \end{subproof}
    \end{subproof}
    Hence $C$ is an open cover of $S$ by \obv{open_cover}.
    Take $D$ such that $D$ is finite and $D\subseteq C$ and
        $D$ covers $\carrier[S]$ by \auto{is_compact}.
    Show that every element of $D$ is real bounded.
    \begin{subproof}
        Fix $U\in D$.
        Hence $U\in C$ by \obv{subseteq}.
        Take $a\in K$ such that
            $U=\metricBall{\RealMetric}{a}{\surrealOne}\inter K$.
        $a\in\reals$ by \obv{subseteq}.
        $\metricBall{\RealMetric}{a}{\surrealOne}$ is real bounded by
            \obv{real_unit_ball_is_bounded}.
        $U\subseteq\metricBall{\RealMetric}{a}{\surrealOne}$ by
            \auto{inter_lower_left}.
        Follows by \obv{subset_of_real_bounded_is_bounded}.
    \end{subproof}
    $\unions{D}$ is real bounded by
        \obv{unions_of_finite_real_bounded_family}.
    $K\subseteq\unions{D}$ by
        \obv{covers_unions_intro}.
    Follows by \obv{subset_of_real_bounded_is_bounded}.
\end{proof}

\begin{theorem}\label{closed_bounded_real_subset_is_compact}
    Let $X$ be a topological space.
    Suppose $X=\RealTopology$.
    Suppose $K$ is closed in $X$ and $K$ is real bounded.
    Then $K$ is compact in $X$.
\end{theorem}
\begin{proof}
    $K\subseteq\reals$ by \obv{real_bounded}.
    $\carrier[X]=\reals$ by \obv{real_topology_carrier}.
    Hence $K\subseteq\carrier[X]$.
    Take $M\in\reals$ such that $\surrealLe{\surrealZero}{M}$ and
        for every $x\in K$ we have $\surrealLe{\abs{x}}{M}$ by
        \obv{real_bounded}.
    Let $I=\realClosedSymmetricInterval{M}$.
    Show that $K\subseteq I$.
    \begin{subproof}
        Show that every element of $K$ is an element of $I$.
        \begin{subproof}
            Fix $x\in K$.
            Hence $x\in\reals$ by \obv{subseteq}.
            Hence $\surreal{x}$ and $\surreal{M}$ by
                \obv{surreal_real_is_surreal}.
            $\surrealLe{\abs{x}}{M}$.
            $\surrealLe{x}{M}$ and
                $\surrealLe{\surrealNeg{x}}{M}$ by
                \obv{surreal_abs_le_iff}.
            Hence $\surrealLe{\surrealNeg{M}}
                {\surrealNeg{\surrealNeg{x}}}$ by
                \obv{surreal_negation_le_reverses,
                    surreal_negation_is_surreal}.
            $\surrealNeg{\surrealNeg{x}}=x$ by
                \obv{surreal_negation_involutive}.
            Hence $\surrealLe{\surrealNeg{M}}{x}$.
            Thus $x\in\realClosedInterval{\surrealNeg{M}}{M}$ by
                \obv{real_closed_interval}.
            $\realClosedSymmetricInterval{M}
                =\realClosedInterval{\surrealNeg{M}}{M}$ by
                \obv{real_closed_symmetric_interval}.
            Hence $x\in\realClosedSymmetricInterval{M}$.
            Hence $x\in I$.
        \end{subproof}
    \end{subproof}
    $\surrealNeg{M}\in\reals$ by
        \obv{surreal_real_negation_closed}.
    $\realClosedInterval{\surrealNeg{M}}{M}$ is compact in $X$ by
        \obv{real_closed_interval_is_compact}.
    $I=\realClosedInterval{\surrealNeg{M}}{M}$ by
        \obv{real_closed_symmetric_interval}.
    Hence $I$ is compact in $X$.
    Thus $I\subseteq\carrier[X]$ and
        $\subspaceTopology{X}{I}$ is compact by \obv{compact_in}.

    $\carrier[\subspaceTopology{X}{I}]=I$ by
        \obv{subspace_topology_carrier_of_subset}.
    $K\in\closeds{X}$ by \obv{is_closed_in}.
    $K\inter\carrier[\subspaceTopology{X}{I}]=K$ by
        \obv{inter_absorb_supseteq_right}.
    Hence there exists $F\in\closeds{X}$ such that
        $K=F\inter\carrier[\subspaceTopology{X}{I}]$.
    Hence $K$ is closed in $\subspaceTopology{X}{I}$ by
        \auto{subspace_closed_iff}.
    Thus $\subspaceTopology{\subspaceTopology{X}{I}}{K}$ is compact by
        \obv{closed_subspace_of_compact_is_compact}.

    $I\inter\carrier[X]=I$ by
        \obv{inter_absorb_supseteq_right}.
    $I\inter K=K$ by
        \obv{inter_absorb_supseteq_left}.
    Hence $(I\inter\carrier[X])\inter K=K$.
    $\subspaceTopology{\subspaceTopology{X}{I}}{K}
        =\subspaceTopology{X}{(I\inter\carrier[X])\inter K}$ by
        \obv{subspace_topology_transitive}.
    Hence $\subspaceTopology{\subspaceTopology{X}{I}}{K}
        =\subspaceTopology{X}{K}$.
    Thus $\subspaceTopology{X}{K}$ is compact.
    Follows by \obv{compact_in}.
\end{proof}

\begin{theorem}[Heine--Borel]\label{real_heine_borel}
    Let $X$ be a topological space.
    Suppose $X=\RealTopology$.
    Then $K$ is compact in $X$ iff
        $K$ is closed in $X$ and $K$ is real bounded.
\end{theorem}
\begin{proof}
    $\RealMetric$ is a metric space by
        \obv{surreal_real_metric_space}.
    Hence $\metricTopology{\RealMetric}$ is Hausdorff by
        \obv{metric_topology_is_hausdorff}.
    $\RealTopology=\metricTopology{\RealMetric}$ by
        \obv{real_topology}.
    Hence $X$ is Hausdorff.
    Show that if $K$ is compact in $X$, then
        $K$ is closed in $X$ and $K$ is real bounded.
    \begin{subproof}
        Assume $K$ is compact in $X$.
        $K$ is closed in $X$ by
            \obv{compact_subset_of_hausdorff_is_closed}.
        $K$ is real bounded by
            \obv{compact_real_subset_is_bounded}.
    \end{subproof}
    Show that if $K$ is closed in $X$ and $K$ is real bounded, then
        $K$ is compact in $X$.
    \begin{subproof}
        Assume $K$ is closed in $X$ and $K$ is real bounded.
        Follows by \obv{closed_bounded_real_subset_is_compact}.
    \end{subproof}
    Follows by \auto.
\end{proof}
