\section{Intermediate values of real maps}

\import{topology/real-connectedness.tex}

The connected subsets of the real line are exactly its order-convex subsets.
Together with preservation of connectedness by continuous maps, this turns
the intermediate value theorem into a direct range argument.

\begin{theorem}\label{real_connected_set_is_order_convex}
    Let $X$ be a topological space.
    Suppose $X=\RealTopology$.
    Let $L$ be a linearly ordered set.
    Suppose $L=\RealField$.
    Suppose $A$ is connected in $X$.
    Then $A$ is order convex in $L$.
\end{theorem}
\begin{proof}
    $A\subseteq\carrier[X]$ by
        \obv{connected_in_subseteq_carrier}.
    $\subspaceTopology{X}{A}$ is connected by
        \obv{connected_in}.
    Hence $\subspaceTopology{X}{A}$ is not disconnected by
        \obv{connected}.
    $\carrier[X]=\reals$ by \obv{real_topology_carrier}.
    Hence $A\subseteq\reals$.
    $\carrier[L]=\reals$ by
        \obv{surreal_real_field_carrier}.
    Hence $A\subseteq\carrier[L]$.

    Show that for all $x,z\in A$ we have
        for every $y\in\carrier[L]$ we have if
            $x\mathrel{\leq[L]}y$ and $y\mathrel{\leq[L]}z$, then
            $y\in A$.
    \begin{subproof}
        Fix $x,z\in A$.
        Fix $y\in\carrier[L]$.
        Assume $x\mathrel{\leq[L]}y$ and
            $y\mathrel{\leq[L]}z$.
        Suppose not.
        Hence $y\notin A$.
        $x,z\in\carrier[L]$ by \obv{subseteq}.
        $x,y,z\in\reals$.
        $\surrealLe{x}{y}$ and $\surrealLe{y}{z}$ by
            \obv{surreal_real_field_order}.
        $x\neq y$ and $y\neq z$.
        Hence $\surrealLt{x}{y}$ and $\surrealLt{y}{z}$ by
            \auto{surreal_le}.

        Let $S=\subspaceTopology{X}{A}$.
        Hence $S$ is a topological space.
        $\carrier[\subspaceTopology{X}{A}]=A$ by
            \obv{subspace_topology_carrier_of_subset}.
        Hence $\carrier[S]=A$.

        Let $U=\realOpenLowerRay{y}\inter A$.
        Let $V=\realOpenUpperRay{y}\inter A$.
        $\realOpenLowerRay{y},\realOpenUpperRay{y}
            \in\opens[\RealTopology]$ by
            \obv{real_open_lower_ray_is_open,
                real_open_upper_ray_is_open}.
        Hence $\realOpenLowerRay{y},\realOpenUpperRay{y}
            \in\opens[X]$.
        Hence $U,V\in\opens[S]$ by
            \auto{subspace_open_iff}.
        $x\in\realOpenLowerRay{y}$ by
            \obv{real_open_lower_ray_member_iff}.
        Hence $x\in U$ by \obv{inter}.
        Thus $U$ is inhabited.
        $z\in\realOpenUpperRay{y}$ by
            \obv{real_open_upper_ray_member_iff}.
        Hence $z\in V$ by \obv{inter}.
        Thus $V$ is inhabited.

        Show that $U$ is disjoint from $V$.
        \begin{subproof}
            Suppose not.
            Take $a$ such that $a\in U,V$ by \auto{disjoint}.
            Hence $a\in\realOpenLowerRay{y}$ and
                $a\in\realOpenUpperRay{y}$ by \auto{inter}.
            Hence $\surrealLt{a}{y}$ and $\surrealLt{y}{a}$ by
                \auto{real_open_lower_ray_member_iff,
                    real_open_upper_ray_member_iff}.
            $\surreal{a}$ and $\surreal{y}$ by
                \auto{real_open_lower_ray_member_iff,
                    surreal_real_is_surreal}.
            Contradiction by \obv{surreal_lt_asymmetric}.
        \end{subproof}
        Show that $U\union V=A$.
        \begin{subproof}
            Show that for every $a\in U\union V$ we have $a\in A$.
            \begin{subproof}
                Fix $a\in U\union V$.
                Follows by \auto{union_iff,inter}.
            \end{subproof}
            Show that for every $a\in A$ we have $a\in U\union V$.
            \begin{subproof}
                Fix $a\in A$.
                Hence $a\in\reals$ by \obv{subseteq}.
                $a\neq y$.
                $\surreal{a}$ and $\surreal{y}$ by
                    \obv{surreal_real_is_surreal}.
                Hence $\surrealLt{a}{y}$ or
                    $\surrealLt{y}{a}$ by
                    \obv{surreal_lt_trichotomy}.
                Hence $a\in\realOpenLowerRay{y}$ or
                    $a\in\realOpenUpperRay{y}$ by
                    \auto{real_open_lower_ray_member_iff,
                        real_open_upper_ray_member_iff}.
                Follows by \auto{union_iff,inter}.
            \end{subproof}
            Follows by set extensionality.
        \end{subproof}
        Hence $A$ is partitioned by $U$ and $V$ by \auto.
        Hence $\carrier[S]$ is partitioned by $U$ and $V$.
        Thus $S$ is disconnected by \auto{disconnected}.
        Hence $\subspaceTopology{X}{A}$ is disconnected.
        Contradiction.
    \end{subproof}
    Follows by \obv{order_convex}.
\end{proof}

\begin{theorem}\label{continuous_real_map_intermediate_value_ordered}
    Let $X,Y$ be topological spaces.
    Suppose $Y=\RealTopology$.
    Let $L$ be a linearly ordered set.
    Suppose $L=\RealField$.
    Suppose $X$ is connected.
    Suppose $f$ is continuous from $X$ to $Y$.
    Let $x,z\in\carrier[X]$ and $c\in\carrier[L]$.
    Suppose $f(x)\mathrel{\leq[L]}c$ and
        $c\mathrel{\leq[L]}f(z)$.
    Then there exists $y\in\carrier[X]$ such that $f(y)=c$.
\end{theorem}
\begin{proof}
    $\ran{f}$ is connected in $Y$ by
        \obv{continuous_image_is_connected}.
    Hence $\ran{f}$ is order convex in $L$ by
        \obv{real_connected_set_is_order_convex}.
    $f\in\funs{\carrier[X]}{\carrier[Y]}$ by
        \obv{continuous_map}.
    Hence $f$ is a function and $\dom{f}=\carrier[X]$ by
        \obv{funs_is_function,funs}.
    $f(x),f(z)\in\ran{f}$ by \auto{fun_ran_iff}.
    Hence $c\in\ran{f}$ by \obv{order_convex}.
    Take $y\in\dom{f}$ such that $f(y)=c$ by
        \obv{fun_ran_iff}.
    Hence $y\in\carrier[X]$.
\end{proof}

\begin{theorem}\label{continuous_real_map_intermediate_value}
    Let $X,Y$ be topological spaces.
    Suppose $Y=\RealTopology$.
    Let $L$ be a linearly ordered set.
    Suppose $L=\RealField$.
    Suppose $X$ is connected.
    Suppose $f$ is continuous from $X$ to $Y$.
    Let $x,z\in\carrier[X]$ and $c\in\carrier[L]$.
    Suppose
        ($f(x)\mathrel{\leq[L]}c$ and
            $c\mathrel{\leq[L]}f(z)$) or
        ($f(z)\mathrel{\leq[L]}c$ and
            $c\mathrel{\leq[L]}f(x)$).
    Then there exists $y\in\carrier[X]$ such that $f(y)=c$.
\end{theorem}
\begin{proof}
    Follows by
        \obv{continuous_real_map_intermediate_value_ordered}.
\end{proof}

\begin{corollary}\label{real_connected_iff_order_convex}
    Let $X$ be a topological space.
    Suppose $X=\RealTopology$.
    Let $L$ be a linearly ordered set.
    Suppose $L=\RealField$.
    Then $A$ is connected in $X$ iff $A$ is order convex in $L$.
\end{corollary}
\begin{proof}
    Follows by
        \auto{real_connected_set_is_order_convex,
            real_order_convex_set_is_connected}.
\end{proof}

\begin{corollary}\label{real_closed_interval_intermediate_value}
    Let $X$ be a topological space.
    Suppose $X=\RealTopology$.
    Let $L$ be a linearly ordered set.
    Suppose $L=\RealField$.
    Let $a,b\in\carrier[L]$.
    Suppose $a\mathrel{\leq[L]}b$.
    Let $S$ be a topological space.
    Suppose
        $S=\subspaceTopology{X}{\orderClosedInterval{L}{a}{b}}$.
    Suppose $f$ is continuous from $S$ to $X$.
    Let $c\in\carrier[L]$.
    Suppose
        ($f(a)\mathrel{\leq[L]}c$ and
            $c\mathrel{\leq[L]}f(b)$) or
        ($f(b)\mathrel{\leq[L]}c$ and
            $c\mathrel{\leq[L]}f(a)$).
    Then there exists $y\in\orderClosedInterval{L}{a}{b}$ such that
        $f(y)=c$.
\end{corollary}
\begin{proof}
    $L$ is an ordered set by
        \obv{linearly_ordered_set_is_ordered_set}.
    $\orderClosedInterval{L}{a}{b}$ is connected in $X$ by
        \obv{real_closed_interval_is_connected}.
    Hence
        $\subspaceTopology{X}{\orderClosedInterval{L}{a}{b}}$
        is connected by \obv{connected_in}.
    Hence $S$ is connected.
    $a,b\in\orderClosedInterval{L}{a}{b}$ by
        \auto{order_closed_interval_iff,
            ordered_set_order_reflexive,reflexive_on}.
    $\orderClosedInterval{L}{a}{b}\subseteq\carrier[L]$ by
        \obv{order_interval_and_ray_types}.
    $\carrier[L]=\reals$ by
        \obv{surreal_real_field_carrier}.
    $\carrier[X]=\reals$ by \obv{real_topology_carrier}.
    Hence
        $\orderClosedInterval{L}{a}{b}\subseteq\carrier[X]$.
    $\carrier[\subspaceTopology{X}
        {\orderClosedInterval{L}{a}{b}}]
        =\orderClosedInterval{L}{a}{b}$ by
        \obv{subspace_topology_carrier_of_subset}.
    Hence $\carrier[S]=\orderClosedInterval{L}{a}{b}$.
    Hence $a,b\in\carrier[S]$.
    Take $y\in\carrier[S]$ such that $f(y)=c$ by
        \obv{continuous_real_map_intermediate_value}.
    Hence $y\in\orderClosedInterval{L}{a}{b}$.
\end{proof}
