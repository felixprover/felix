\section{Binary coproduct topologies}

\import{topology/indexed-coproduct-open.tex}
\import{topology/indexed-coproduct-map.tex}

The binary coproduct is the two-coordinate instance of the indexed
coproduct.  Its carrier is therefore a tagged indexed sum, and the two
canonical injections retain their set-theoretic definitions.

\begin{definition}\label{binary_coproduct_space_family}
    $\binaryCoproductSpaceFamily{X}{Y}=\binaryFamily{X}{Y}$.
\end{definition}

\begin{definition}\label{binary_coproduct_carrier_family}
    $\binaryCoproductCarrierFamily{X}{Y}
        =\indexedTopologicalCarrierFamily{
            \suc{\suc{\emptyset}}}{
            \binaryCoproductSpaceFamily{X}{Y}}$.
\end{definition}

\begin{definition}\label{first_coproduct_injection}
    $\firstCoproductInjection{X}{Y}
        =\indexedSumInjection{\suc{\suc{\emptyset}}}{
            \binaryCoproductCarrierFamily{X}{Y}}{\emptyset}$.
\end{definition}

\begin{definition}\label{second_coproduct_injection}
    $\secondCoproductInjection{X}{Y}
        =\indexedSumInjection{\suc{\suc{\emptyset}}}{
            \binaryCoproductCarrierFamily{X}{Y}}{\suc{\emptyset}}$.
\end{definition}

\begin{definition}\label{binary_coproduct_topology}
    $\BinaryCoproductTopology{X}{Y}
        =\IndexedCoproductTopology{
            \suc{\suc{\emptyset}}}{
            \binaryCoproductSpaceFamily{X}{Y}}$.
\end{definition}

\begin{proposition}\label{binary_coproduct_space_family_type}
    $\binaryCoproductSpaceFamily{X}{Y}$ is a function on
        $\suc{\suc{\emptyset}}$.
\end{proposition}
\begin{proof}
    $\binaryCoproductSpaceFamily{X}{Y}
        \in\funs{\suc{\suc{\emptyset}}}{\{X,Y\}}$ by
        \auto{binary_coproduct_space_family,binary_family_type}.
    Follows by \obv{funs_is_function,funs}.
\end{proof}

\begin{proposition}\label{binary_coproduct_space_family_left}
    $(\binaryCoproductSpaceFamily{X}{Y})(\emptyset)=X$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{binary_coproduct_space_family,binary_family_left}.
\end{proof}

\begin{proposition}\label{binary_coproduct_space_family_right}
    $(\binaryCoproductSpaceFamily{X}{Y})(\suc{\emptyset})=Y$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{binary_coproduct_space_family,binary_family_right}.
\end{proof}

\begin{proposition}\label{binary_coproduct_carrier_family_left}
    $(\binaryCoproductCarrierFamily{X}{Y})(\emptyset)=\carrier[X]$.
\end{proposition}
\begin{proof}
    $\emptyset\in\suc{\suc{\emptyset}}$ by
        \auto{suc_iff,emptyset}.
    Hence $(\indexedTopologicalCarrierFamily{
        \suc{\suc{\emptyset}}}{
        \binaryCoproductSpaceFamily{X}{Y}})(\emptyset)
        =\carrier[(\binaryCoproductSpaceFamily{X}{Y})(\emptyset)]$ by
        \obv{indexed_topological_carrier_family_apply}.
    Follows by
        \auto{binary_coproduct_carrier_family,
            binary_coproduct_space_family_left}.
\end{proof}

\begin{proposition}\label{binary_coproduct_carrier_family_right}
    $(\binaryCoproductCarrierFamily{X}{Y})(\suc{\emptyset})
        =\carrier[Y]$.
\end{proposition}
\begin{proof}
    $\suc{\emptyset}\in\suc{\suc{\emptyset}}$ by \auto{suc_iff}.
    Hence $(\indexedTopologicalCarrierFamily{
        \suc{\suc{\emptyset}}}{
        \binaryCoproductSpaceFamily{X}{Y}})(\suc{\emptyset})
        =\carrier[(\binaryCoproductSpaceFamily{X}{Y})(
            \suc{\emptyset})]$ by
        \obv{indexed_topological_carrier_family_apply}.
    Follows by
        \auto{binary_coproduct_carrier_family,
            binary_coproduct_space_family_right}.
\end{proof}

\begin{proposition}\label{binary_coproduct_family_is_topological}
    Let $X,Y$ be topological spaces.
    Then for every $i\in\suc{\suc{\emptyset}}$ we have
        $(\binaryCoproductSpaceFamily{X}{Y})(i)$ is a topological space.
\end{proposition}
\begin{proof}
    Fix $i\in\suc{\suc{\emptyset}}$.
    $i=\emptyset$ or $i=\suc{\emptyset}$ by
        \auto{suc_elim,emptyset}.
    Follows by
        \auto{binary_coproduct_space_family_left,
            binary_coproduct_space_family_right}.
\end{proof}

\begin{theorem}\label{binary_coproduct_topology_is_topological_space}
    Let $X,Y$ be topological spaces.
    Then $\BinaryCoproductTopology{X}{Y}$ is a topological space.
\end{theorem}
\begin{proof}
    $\binaryCoproductSpaceFamily{X}{Y}$ is a function on
        $\suc{\suc{\emptyset}}$ by
        \obv{binary_coproduct_space_family_type}.
    For every $i\in\suc{\suc{\emptyset}}$ we have
        $(\binaryCoproductSpaceFamily{X}{Y})(i)$ is a topological space by
        \obv{binary_coproduct_family_is_topological}.
    Hence $\IndexedCoproductTopology{
        \suc{\suc{\emptyset}}}{
        \binaryCoproductSpaceFamily{X}{Y}}$ is a topological space by
        \obv{indexed_coproduct_topology_is_topological_space}.
    Follows by \obv{binary_coproduct_topology}.
\end{proof}

\begin{theorem}\label{binary_coproduct_topology_is_exact}
    Let $X,Y$ be topological spaces.
    Then $\BinaryCoproductTopology{X}{Y}$ is an exact topological space.
\end{theorem}
\begin{proof}
    $\binaryCoproductSpaceFamily{X}{Y}$ is a function on
        $\suc{\suc{\emptyset}}$ by
        \obv{binary_coproduct_space_family_type}.
    For every $i\in\suc{\suc{\emptyset}}$ we have
        $(\binaryCoproductSpaceFamily{X}{Y})(i)$ is a topological space by
        \obv{binary_coproduct_family_is_topological}.
    Hence $\IndexedCoproductTopology{
        \suc{\suc{\emptyset}}}{
        \binaryCoproductSpaceFamily{X}{Y}}$
        is an exact topological space by
        \obv{indexed_coproduct_topology_is_exact}.
    Follows by \auto{binary_coproduct_topology}.
\end{proof}

\begin{proposition}\label{binary_coproduct_topology_carrier}
    $\carrier[\BinaryCoproductTopology{X}{Y}]
        =\indexedSum{\suc{\suc{\emptyset}}}{
            \binaryCoproductCarrierFamily{X}{Y}}$.
\end{proposition}
\begin{proof}
    $\carrier[\IndexedCoproductTopology{
        \suc{\suc{\emptyset}}}{
        \binaryCoproductSpaceFamily{X}{Y}}]
        =\indexedSum{\suc{\suc{\emptyset}}}{
            \indexedTopologicalCarrierFamily{
                \suc{\suc{\emptyset}}}{
                \binaryCoproductSpaceFamily{X}{Y}}}$ by
        \obv{indexed_coproduct_topology_carrier}.
    Follows by
        \auto{binary_coproduct_topology,binary_coproduct_carrier_family}.
\end{proof}

\begin{theorem}\label{first_coproduct_injection_is_continuous_open_injection}
    Let $X,Y$ be topological spaces.
    Then $\firstCoproductInjection{X}{Y}$ is continuous from $X$ to
        $\BinaryCoproductTopology{X}{Y}$ and
        $\firstCoproductInjection{X}{Y}$ is an open map from $X$ to
        $\BinaryCoproductTopology{X}{Y}$ and
        $\firstCoproductInjection{X}{Y}
            \in\inj{\carrier[X]}{
                \carrier[\BinaryCoproductTopology{X}{Y}]}$.
\end{theorem}
\begin{proof}
    $\binaryCoproductSpaceFamily{X}{Y}$ is a function on
        $\suc{\suc{\emptyset}}$ by
        \obv{binary_coproduct_space_family_type}.
    For every $i\in\suc{\suc{\emptyset}}$ we have
        $(\binaryCoproductSpaceFamily{X}{Y})(i)$ is a topological space by
        \obv{binary_coproduct_family_is_topological}.
    $\emptyset\in\suc{\suc{\emptyset}}$ by
        \auto{suc_iff,emptyset}.
    Hence $\indexedSumInjection{\suc{\suc{\emptyset}}}{
        \indexedTopologicalCarrierFamily{
            \suc{\suc{\emptyset}}}{
            \binaryCoproductSpaceFamily{X}{Y}}}{\emptyset}$
        is continuous from
        $(\binaryCoproductSpaceFamily{X}{Y})(\emptyset)$ to
        $\IndexedCoproductTopology{\suc{\suc{\emptyset}}}{
            \binaryCoproductSpaceFamily{X}{Y}}$ and
        $\indexedSumInjection{\suc{\suc{\emptyset}}}{
            \indexedTopologicalCarrierFamily{
                \suc{\suc{\emptyset}}}{
                \binaryCoproductSpaceFamily{X}{Y}}}{\emptyset}$
        is an open map from
        $(\binaryCoproductSpaceFamily{X}{Y})(\emptyset)$ to
        $\IndexedCoproductTopology{\suc{\suc{\emptyset}}}{
            \binaryCoproductSpaceFamily{X}{Y}}$ and
        $\indexedSumInjection{\suc{\suc{\emptyset}}}{
            \indexedTopologicalCarrierFamily{
                \suc{\suc{\emptyset}}}{
                \binaryCoproductSpaceFamily{X}{Y}}}{\emptyset}
        \in\inj{
            \carrier[(\binaryCoproductSpaceFamily{X}{Y})(\emptyset)]}{
            \carrier[\IndexedCoproductTopology{
                \suc{\suc{\emptyset}}}{
                \binaryCoproductSpaceFamily{X}{Y}}]}$ by
        \obv{indexed_coproduct_injection_is_continuous_open_injection}.
    Follows by
        \auto{first_coproduct_injection,binary_coproduct_carrier_family,
            binary_coproduct_space_family_left,binary_coproduct_topology}.
\end{proof}

\begin{theorem}\label{second_coproduct_injection_is_continuous_open_injection}
    Let $X,Y$ be topological spaces.
    Then $\secondCoproductInjection{X}{Y}$ is continuous from $Y$ to
        $\BinaryCoproductTopology{X}{Y}$ and
        $\secondCoproductInjection{X}{Y}$ is an open map from $Y$ to
        $\BinaryCoproductTopology{X}{Y}$ and
        $\secondCoproductInjection{X}{Y}
            \in\inj{\carrier[Y]}{
                \carrier[\BinaryCoproductTopology{X}{Y}]}$.
\end{theorem}
\begin{proof}
    $\binaryCoproductSpaceFamily{X}{Y}$ is a function on
        $\suc{\suc{\emptyset}}$ by
        \obv{binary_coproduct_space_family_type}.
    For every $i\in\suc{\suc{\emptyset}}$ we have
        $(\binaryCoproductSpaceFamily{X}{Y})(i)$ is a topological space by
        \obv{binary_coproduct_family_is_topological}.
    $\suc{\emptyset}\in\suc{\suc{\emptyset}}$ by \auto{suc_iff}.
    Hence $\indexedSumInjection{\suc{\suc{\emptyset}}}{
        \indexedTopologicalCarrierFamily{
            \suc{\suc{\emptyset}}}{
            \binaryCoproductSpaceFamily{X}{Y}}}{\suc{\emptyset}}$
        is continuous from
        $(\binaryCoproductSpaceFamily{X}{Y})(\suc{\emptyset})$ to
        $\IndexedCoproductTopology{\suc{\suc{\emptyset}}}{
            \binaryCoproductSpaceFamily{X}{Y}}$ and
        $\indexedSumInjection{\suc{\suc{\emptyset}}}{
            \indexedTopologicalCarrierFamily{
                \suc{\suc{\emptyset}}}{
                \binaryCoproductSpaceFamily{X}{Y}}}{\suc{\emptyset}}$
        is an open map from
        $(\binaryCoproductSpaceFamily{X}{Y})(\suc{\emptyset})$ to
        $\IndexedCoproductTopology{\suc{\suc{\emptyset}}}{
            \binaryCoproductSpaceFamily{X}{Y}}$ and
        $\indexedSumInjection{\suc{\suc{\emptyset}}}{
            \indexedTopologicalCarrierFamily{
                \suc{\suc{\emptyset}}}{
                \binaryCoproductSpaceFamily{X}{Y}}}{\suc{\emptyset}}
        \in\inj{
            \carrier[(\binaryCoproductSpaceFamily{X}{Y})(
                \suc{\emptyset})]}{
            \carrier[\IndexedCoproductTopology{
                \suc{\suc{\emptyset}}}{
                \binaryCoproductSpaceFamily{X}{Y}}]}$ by
        \obv{indexed_coproduct_injection_is_continuous_open_injection}.
    Follows by
        \auto{second_coproduct_injection,binary_coproduct_carrier_family,
            binary_coproduct_space_family_right,binary_coproduct_topology}.
\end{proof}
