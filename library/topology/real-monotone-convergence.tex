\section{Monotone convergence on the real line}

A nondecreasing real sequence is an order-preserving map from the natural
index order to the ordered real line.  The definition below keeps the
familiar sequence language while retaining the generic order-morphism
interface.

\begin{definition}\label{nondecreasing_real_sequence}
    $s$ is a nondecreasing real sequence iff
        $\orderPreserving{s}{\naturals}{\subseteqrel{\naturals}}
            {\reals}{\surrealRealNonstrictOrder}$.
\end{definition}

\begin{proposition}\label{nondecreasing_real_sequence_type}
    Suppose $s$ is a nondecreasing real sequence.
    Then $s\in\funs{\naturals}{\reals}$.
\end{proposition}
\begin{proof}
    Follows by
        \obv{nondecreasing_real_sequence,order_preserving_type}.
\end{proof}

\begin{proposition}\label{nondecreasing_real_sequence_mono}
    Suppose $s$ is a nondecreasing real sequence.
    Let $m,n\in\naturals$.
    Suppose $m\subseteq n$.
    Then $\surrealLe{s(m)}{s(n)}$.
\end{proposition}
\begin{proof}
    $s\in\funs{\naturals}{\reals}$ by
        \obv{nondecreasing_real_sequence_type}.
    Hence $s(m),s(n)\in\reals$ by \obv{funs_type_apply}.
    $m\mathrel{\subseteqrel{\naturals}}n$ by \obv{subseteqrel}.
    Hence $s(m)\mathrel{\surrealRealNonstrictOrder}s(n)$ by
        \auto{nondecreasing_real_sequence,order_preserving}.
    Follows by \obv{surreal_real_nonstrict_order_iff}.
\end{proof}

\begin{definition}\label{nonincreasing_real_sequence}
    $s$ is a nonincreasing real sequence iff
        $\orderPreserving{s}{\naturals}{\subseteqrel{\naturals}}
            {\reals}{\converse{\surrealRealNonstrictOrder}}$.
\end{definition}

\begin{proposition}\label{nonincreasing_real_sequence_type}
    Suppose $s$ is a nonincreasing real sequence.
    Then $s\in\funs{\naturals}{\reals}$.
\end{proposition}
\begin{proof}
    Follows by
        \obv{nonincreasing_real_sequence,order_preserving_type}.
\end{proof}

\begin{proposition}\label{nonincreasing_real_sequence_mono}
    Suppose $s$ is a nonincreasing real sequence.
    Let $m,n\in\naturals$.
    Suppose $m\subseteq n$.
    Then $\surrealLe{s(n)}{s(m)}$.
\end{proposition}
\begin{proof}
    $s\in\funs{\naturals}{\reals}$ by
        \obv{nonincreasing_real_sequence_type}.
    Hence $s(m),s(n)\in\reals$ by \obv{funs_type_apply}.
    $m\mathrel{\subseteqrel{\naturals}}n$ by \obv{subseteqrel}.
    Hence $s(m)\mathrel{\converse{\surrealRealNonstrictOrder}}s(n)$ by
        \auto{nonincreasing_real_sequence,order_preserving}.
    Hence $s(n)\mathrel{\surrealRealNonstrictOrder}s(m)$ by
        \obv{converse_iff}.
    Follows by \obv{surreal_real_nonstrict_order_iff}.
\end{proof}

\begin{proposition}\label{negated_nonincreasing_real_sequence}
    Suppose $s$ is a nonincreasing real sequence.
    Then $\realMapNeg{\naturals}{s}$ is a nondecreasing real sequence.
\end{proposition}
\begin{proof}
    Let $g=\realMapNeg{\naturals}{s}$.
    $s\in\funs{\naturals}{\reals}$ by
        \obv{nonincreasing_real_sequence_type}.
    Hence $\realMapNeg{\naturals}{s}\in\funs{\naturals}{\reals}$ by
        \obv{real_map_neg_type}.
    Hence $g\in\funs{\naturals}{\reals}$.
    $\subseteqrel{\naturals}$ is a quasiorder on $\naturals$ by
        \obv{subseteqrel_is_quasiorder}.
    Hence $\subseteqrel{\naturals}$ is a binary relation on
        $\naturals$ by definition.
    $\surrealRealNonstrictOrder$ is a binary relation on $\reals$ by
        \obv{surreal_real_nonstrict_order_type}.
    Show that for all $m,n\in\naturals$ such that
        $m\mathrel{\subseteqrel{\naturals}}n$ we have
        $g(m)\mathrel{\surrealRealNonstrictOrder}g(n)$.
    \begin{subproof}
        Fix $m,n\in\naturals$.
        Assume $m\mathrel{\subseteqrel{\naturals}}n$.
        Hence $m\subseteq n$ by \auto{subseteqrel}.
        Hence $\surrealLe{s(n)}{s(m)}$ by
            \obv{nonincreasing_real_sequence_mono}.
        $s(m),s(n)\in\reals$ by \obv{funs_type_apply}.
        Hence $\surrealLe{\surrealNeg{s(m)}}{\surrealNeg{s(n)}}$ by
            \obv{surreal_negation_le_reverses,surreal_real_is_surreal}.
        $(\realMapNeg{\naturals}{s})(m)=\surrealNeg{s(m)}$ and
            $(\realMapNeg{\naturals}{s})(n)=\surrealNeg{s(n)}$ by
            \obv{real_map_neg_apply}.
        Hence $g(m)=\surrealNeg{s(m)}$ and
            $g(n)=\surrealNeg{s(n)}$.
        $g(m),g(n)\in\reals$ by \obv{funs_type_apply}.
        Follows by \auto{surreal_real_nonstrict_order_iff}.
    \end{subproof}
    $\orderPreserving{g}{\naturals}{\subseteqrel{\naturals}}
        {\reals}{\surrealRealNonstrictOrder}$ by
        \obv{order_preserving}.
    Follows by \obv{nondecreasing_real_sequence}.
\end{proof}

\begin{theorem}\label{nondecreasing_real_sequence_converges_to_supremum}
    Let $s$ be a nondecreasing real sequence.
    Suppose there exists $b\in\reals$ such that for every
        $n\in\naturals$ we have $\surrealLe{s(n)}{b}$.
    Then $\metricConverges{\RealMetric}{s}{\realSupremum{\ran{s}}}$.
\end{theorem}
\begin{proof}
    $s\in\funs{\naturals}{\reals}$ by
        \obv{nondecreasing_real_sequence_type}.
    Hence $s$ is a function and $\dom{s}=\naturals$ by
        \obv{funs_is_function,funs}.
    $\ran{s}\subseteq\reals$ by \obv{funs_ran}.
    $\emptyset\in\naturals$ by \obv{omega_empty}.
    Hence $s(\emptyset)\in\ran{s}$ by \auto{fun_ran_iff}.
    Hence $\ran{s}$ is inhabited.
    Take $b\in\reals$ such that for every $n\in\naturals$ we have
        $\surrealLe{s(n)}{b}$.
    Show that for every $y\in\ran{s}$ we have $\surrealLe{y}{b}$.
    \begin{subproof}
        Fix $y\in\ran{s}$.
        Take $n\in\dom{s}$ such that $s(n)=y$ by \obv{fun_ran_iff}.
        Hence $n\in\naturals$.
        Follows by \auto.
    \end{subproof}

    Let $q=\realSupremum{\ran{s}}$.
    $\realSupremum{\ran{s}}\in\reals$ by
        \obv{real_supremum_is_real}.
    Hence $q\in\reals$.
    Show that for every $n\in\naturals$ we have
        $\surrealLe{s(n)}{q}$.
    \begin{subproof}
        Fix $n\in\naturals$.
        $s(n)\in\ran{s}$ by \auto{fun_ran_iff}.
        $\surrealLe{s(n)}{\realSupremum{\ran{s}}}$ by
            \obv{real_supremum_upper_bound}.
    \end{subproof}

    $\RealMetric$ is a metric space by \obv{surreal_real_metric_space}.
    $\carrier[\RealMetric]=\reals$ by
        \obv{surreal_real_metric_carrier}.
    Hence $s\in\funs{\naturals}{\carrier[\RealMetric]}$ and
        $q\in\carrier[\RealMetric]$.

    Show that for every $e\in\reals$ such that
        $\surrealLt{\surrealZero}{e}$ there exists $N\in\naturals$
        such that for every $n\in\naturals$ such that $N\subseteq n$
        we have $s(n)\in\metricBall{\RealMetric}{q}{e}$.
    \begin{subproof}
        Fix $e\in\reals$.
        Assume $\surrealLt{\surrealZero}{e}$.
        $\surrealNeg{e}\in\reals$ by
            \obv{surreal_real_negation_closed}.
        Hence $\surrealAdd{q}{\surrealNeg{e}}\in\reals$ by
            \obv{surreal_real_addition_closed}.
        $\surrealLt{q}{\surrealAdd{q}{e}}$ by
            \auto{surreal_add_lt_right,surreal_add_zero_right,
                surreal_real_is_surreal,empty_surreal}.
        Hence $\surrealLt{\surrealAdd{q}{\surrealNeg{e}}}{q}$ by
            \auto{surreal_difference_lt_iff,surreal_add_commutative,
                surreal_real_is_surreal}.

        Show that there exists $N\in\naturals$ such that
            $\surrealLt{\surrealAdd{q}{\surrealNeg{e}}}{s(N)}$.
        \begin{subproof}
            Suppose not.
            Hence for every $n\in\naturals$ we have
                not $\surrealLt{\surrealAdd{q}{\surrealNeg{e}}}{s(n)}$
                by \auto.
            Show that for every $n\in\naturals$ we have
                $\surrealLe{s(n)}
                    {\surrealAdd{q}{\surrealNeg{e}}}$.
            \begin{subproof}
                Fix $n\in\naturals$.
                $s(n)\in\reals$ by \obv{funs_type_apply}.
                Follows by
                    \auto{surreal_lt_trichotomy,surreal_le,
                        surreal_lt_asymmetric,surreal_real_is_surreal}.
            \end{subproof}
            Show that for every $y\in\ran{s}$ we have
                $\surrealLe{y}{\surrealAdd{q}{\surrealNeg{e}}}$.
            \begin{subproof}
                Fix $y\in\ran{s}$.
                Take $n\in\dom{s}$ such that $s(n)=y$ by
                    \obv{fun_ran_iff}.
                Hence $n\in\naturals$.
                Follows by \auto.
            \end{subproof}
            Hence $\surrealLe{\realSupremum{\ran{s}}}
                {\surrealAdd{q}{\surrealNeg{e}}}$ by
                \obv{real_supremum_least}.
            Hence $\surrealLe{q}{\surrealAdd{q}{\surrealNeg{e}}}$.
            Contradiction by
                \obv{surreal_le,surreal_lt_asymmetric,
                    surreal_real_is_surreal}.
        \end{subproof}

        Take $N\in\naturals$ such that
            $\surrealLt{\surrealAdd{q}{\surrealNeg{e}}}{s(N)}$.
        Show that for every $n\in\naturals$ such that $N\subseteq n$
            we have $s(n)\in\metricBall{\RealMetric}{q}{e}$.
        \begin{subproof}
            Fix $n\in\naturals$.
            Assume $N\subseteq n$.
            $\surrealLe{s(N)}{s(n)}$ by
                \obv{nondecreasing_real_sequence_mono}.
            $s(N),s(n)\in\reals$ by \obv{funs_type_apply}.
            Hence $\surrealLt{\surrealAdd{q}{\surrealNeg{e}}}{s(n)}$ by
                \auto{surreal_lt_le_transitive,surreal_add_is_surreal,
                    surreal_negation_is_surreal,surreal_real_is_surreal}.
            $\surrealLe{s(n)}{q}$.
            $\surrealLt{q}{\surrealAdd{q}{e}}$ by
                \obv{surreal_add_lt_right,surreal_add_zero_right,
                    surreal_real_is_surreal,empty_surreal}.
            Hence $\surrealLt{s(n)}{\surrealAdd{q}{e}}$ by
                \auto{surreal_le_lt_transitive,surreal_add_is_surreal,
                    surreal_real_is_surreal}.
            $s(n)\in\carrier[\RealField]$ by
                \obv{funs_type_apply,surreal_real_field_carrier}.
            $\surrealAdd{q}{\surrealNeg{e}},\surrealAdd{q}{e}\in\reals$ by
                \obv{surreal_real_negation_closed,
                    surreal_real_addition_closed}.
            Hence $\surrealAdd{q}{\surrealNeg{e}},
                \surrealAdd{q}{e}\in\carrier[\RealField]$ by
                \obv{surreal_real_field_carrier}.
            Hence $s(n)\in\orderOpenInterval{\RealField}
                {\surrealAdd{q}{\surrealNeg{e}}}
                {\surrealAdd{q}{e}}$ by
                \auto{order_open_interval_iff,
                    real_field_order_strict_iff}.
            $\metricBall{\RealMetric}{q}{e}
                =\orderOpenInterval{\RealField}
                    {\surrealAdd{q}{\surrealNeg{e}}}
                    {\surrealAdd{q}{e}}$ by
                \obv{real_metric_ball_eq_order_open_interval}.
        \end{subproof}
        Follows by \auto.
    \end{subproof}
    Hence $\metricConverges{\RealMetric}{s}{q}$ by
        \obv{metric_convergence}.
\end{proof}

\begin{theorem}\label{nonincreasing_real_sequence_converges_to_infimum}
    Let $s$ be a nonincreasing real sequence.
    Suppose there exists $b\in\reals$ such that for every
        $n\in\naturals$ we have $\surrealLe{b}{s(n)}$.
    Then $\metricConverges{\RealMetric}{s}{\realInfimum{\ran{s}}}$.
\end{theorem}
\begin{proof}
    Let $g=\realMapNeg{\naturals}{s}$.
    $s\in\funs{\naturals}{\reals}$ by
        \obv{nonincreasing_real_sequence_type}.
    Hence $\realMapNeg{\naturals}{s}\in\funs{\naturals}{\reals}$ by
        \obv{real_map_neg_type}.
    Hence $g\in\funs{\naturals}{\reals}$.
    $\realMapNeg{\naturals}{s}$ is a nondecreasing real sequence by
        \obv{negated_nonincreasing_real_sequence}.
    Hence $g$ is a nondecreasing real sequence.
    Take $b\in\reals$ such that for every $n\in\naturals$ we have
        $\surrealLe{b}{s(n)}$.
    $\surrealNeg{b}\in\reals$ by
        \obv{surreal_real_negation_closed}.
    Show that for every $n\in\naturals$ we have
        $\surrealLe{g(n)}{\surrealNeg{b}}$.
    \begin{subproof}
        Fix $n\in\naturals$.
        $s(n)\in\reals$ by \obv{funs_type_apply}.
        $\surrealLe{\surrealNeg{s(n)}}{\surrealNeg{b}}$ by
            \obv{surreal_negation_le_reverses,surreal_real_is_surreal}.
        $(\realMapNeg{\naturals}{s})(n)=\surrealNeg{s(n)}$ by
            \obv{real_map_neg_apply}.
        Hence $\surrealLe{(\realMapNeg{\naturals}{s})(n)}
            {\surrealNeg{b}}$.
    \end{subproof}
    Hence there exists $c\in\reals$ such that for every
        $n\in\naturals$ we have
        $\surrealLe{(\realMapNeg{\naturals}{s})(n)}{c}$ by
        \auto.
    $\metricConverges{\RealMetric}{\realMapNeg{\naturals}{s}}
        {\realSupremum{\ran{\realMapNeg{\naturals}{s}}}}$ by
        \auto{nondecreasing_real_sequence_converges_to_supremum}.
    Hence $\metricConverges{\RealMetric}{g}
        {\realSupremum{\ran{g}}}$ by \auto.
    Let $q=\realSupremum{\ran{g}}$.
    $g\in\funs{\naturals}{\carrier[\RealMetric]}$ by
        \obv{metric_convergence}.
    $\realSupremum{\ran{g}}\in\carrier[\RealMetric]$ by
        \obv{metric_convergence}.
    Hence $q\in\carrier[\RealMetric]$.
    Hence $q\in\reals$ by \obv{surreal_real_metric_carrier}.
    $\surrealNeg{q}\in\reals$ by
        \obv{surreal_real_negation_closed}.
    Hence $s\in\funs{\naturals}{\carrier[\RealMetric]}$ and
        $\surrealNeg{q}\in\carrier[\RealMetric]$ by
        \auto{surreal_real_metric_carrier}.
    Show that for every $e\in\reals$ such that
        $\surrealLt{\surrealZero}{e}$ there exists $N\in\naturals$
        such that for every $n\in\naturals$ such that $N\subseteq n$
        we have $s(n)\in\metricBall{\RealMetric}{\surrealNeg{q}}{e}$.
    \begin{subproof}
        Fix $e\in\reals$.
        Assume $\surrealLt{\surrealZero}{e}$.
        Take $N\in\naturals$ such that for every $n\in\naturals$
            such that $N\subseteq n$ we have
            $(\realMapNeg{\naturals}{s})(n)\in
                \metricBall{\RealMetric}
                    {\realSupremum{
                        \ran{\realMapNeg{\naturals}{s}}}}{e}$ by
            \obv{metric_convergence}.
        Show that for every $n\in\naturals$ such that $N\subseteq n$
            we have $s(n)\in
                \metricBall{\RealMetric}{\surrealNeg{q}}{e}$.
        \begin{subproof}
            Fix $n\in\naturals$.
            Assume $N\subseteq n$.
            Hence $(\realMapNeg{\naturals}{s})(n)\in
                \metricBall{\RealMetric}
                    {\realSupremum{
                        \ran{\realMapNeg{\naturals}{s}}}}{e}$.
            Hence $\surrealLt{\metricDistance[\RealMetric]
                (\realSupremum{\ran{\realMapNeg{\naturals}{s}}},
                    (\realMapNeg{\naturals}{s})(n))}{e}$
                by \auto{metric_ball_iff}.
            Hence $\surrealLt{\metricDistance[\RealMetric](q,g(n))}{e}$
                by \auto.
            $s(n),g(n)\in\reals$ by \obv{funs_type_apply}.
            $(\realMapNeg{\naturals}{s})(n)=\surrealNeg{s(n)}$ by
                \obv{real_map_neg_apply}.
            Hence $g(n)=\surrealNeg{s(n)}$.
            $\surrealNeg{g(n)}=s(n)$ by
                \auto{surreal_negation_involutive,
                    surreal_real_is_surreal}.
            $\metricDistance[\RealMetric]
                    (\surrealNeg{q},\surrealNeg{g(n)})
                =\metricDistance[\RealMetric](q,g(n))$ by
                \obv{real_metric_negation_invariant}.
            Hence $\surrealLt{\metricDistance[\RealMetric]
                (\surrealNeg{q},s(n))}{e}$ by \auto.
            $s(n)\in\carrier[\RealMetric]$ by \obv{funs_type_apply}.
            Follows by \obv{metric_ball}.
        \end{subproof}
        Follows by \auto.
    \end{subproof}
    Hence $\metricConverges{\RealMetric}{s}{\surrealNeg{q}}$ by
        \obv{metric_convergence}.
    $\ran{g}=\surrealNegSet{\ran{s}}$ by
        \obv{real_map_neg_range}.
    Hence $\surrealNeg{q}=\realInfimum{\ran{s}}$ by
        \auto{real_infimum}.
\end{proof}
