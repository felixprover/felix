\section{Splitting finite topological powers}

\import{topology/power-split.tex}
\import{topology/product-map.tex}
\import{set/finite-function-coherence.tex}

The topological identification uses the same explicit split and join maps as
the set-level construction. No norm or algebraic structure is required.

\begin{theorem}\label{finite_power_split_homeomorphism}
    Let $X$ be a topological space and $m,n$ be natural numbers.
    Then $\finiteFunctionSplit{m}{n}{\carrier[X]}$ is a homeomorphism from
        $\TopologicalPower{X}{m+n}$ to
        $\BinaryProductTopology{\TopologicalPower{X}{m}}{\TopologicalPower{X}{n}}$.
\end{theorem}
\begin{proof}
    Let $D=(m+n)\setminus m$ and $e=\finiteIndexShift{m}{n}$.
    Let $P=\TopologicalPower{X}{m}$ and $Q=\TopologicalPower{X}{D}$.
    Let $R=\TopologicalPower{X}{n}$.
    $m\subseteq m+n$ by \obv{natural_addition_left_subseteq_sum}.
    Hence $m\union D=m+n$ by \auto{setminus_partition}.
    $m$ is disjoint from $D$ by \auto{disjoint,setminus}.
    Hence $\functionSplit{m}{D}{\carrier[X]}$ is a homeomorphism from
        $\TopologicalPower{X}{m+n}$ to $\BinaryProductTopology{P}{Q}$ by
        \auto{topological_power_split_homeomorphism}.
    $P$ is a topological space and $Q$ is a topological space by
        \auto{topological_power_is_topological_space}.
    $\identity{\carrier[P]}$ is a homeomorphism from $P$ to $P$ by
        \obv{identity_homeomorphism}.
    $e\in\bijections{n}{D}$ by \auto{finite_index_shift_bijection}.
    Hence $\functionReindex{D}{n}{\carrier[X]}{e}$ is a homeomorphism from $Q$ to $R$ by
        \auto{topological_power_reindex_homeomorphism}.
    Hence $\productMap{\carrier[P]}{\carrier[Q]}{
        \identity{\carrier[P]}}{\functionReindex{D}{n}{\carrier[X]}{e}}$ is a homeomorphism
        from $\BinaryProductTopology{P}{Q}$ to $\BinaryProductTopology{P}{R}$ by
        \obv{product_map_homeomorphism}.
    $\carrier[P]=\funs{m}{\carrier[X]}$ and $\carrier[Q]=\funs{D}{\carrier[X]}$ by
        \auto{topological_power_carrier}.
    Follows by \auto{finite_function_split,homeomorphism_composition}.
\end{proof}

\begin{theorem}\label{finite_power_join_homeomorphism}
    Let $X$ be a topological space and $m,n$ be natural numbers.
    Then $\finiteFunctionJoin{m}{n}{\carrier[X]}$ is a homeomorphism from
        $\BinaryProductTopology{\TopologicalPower{X}{m}}{\TopologicalPower{X}{n}}$ to
        $\TopologicalPower{X}{m+n}$.
\end{theorem}
\begin{proof}
    Follows by \auto{finite_power_split_homeomorphism,finite_function_split_converse,
        homeomorphism_inverse}.
\end{proof}
