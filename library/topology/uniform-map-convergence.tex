\section{Uniform convergence of map sequences}

These are ordinary uniform convergence and Cauchy predicates in the uniform
space of maps on an explicit domain.  Their familiar entourage formulas are
derived below.

\begin{definition}\label{uniform_map_convergence}
    $\uniformMapConverges{Y}{D}{S}{g}$ iff
        $Y$ is a uniform space and
        $\uniformSequenceConverges{\uniformMapSpace{Y}{D}}{S}{g}$.
\end{definition}

\begin{definition}\label{uniform_map_cauchy}
    $\uniformMapCauchy{Y}{D}{S}$ iff
        $Y$ is a uniform space and
        $\uniformCauchySequence{\uniformMapSpace{Y}{D}}{S}$.
\end{definition}

\begin{proposition}\label{uniform_map_convergence_target}
    Suppose $\uniformMapConverges{Y}{D}{S}{g}$.
    Then $Y$ is a uniform space.
\end{proposition}
\begin{proof}
    Follows by \obv{uniform_map_convergence}.
\end{proof}

\begin{proposition}\label{uniform_map_convergence_sequence}
    Suppose $\uniformMapConverges{Y}{D}{S}{g}$.
    Then $S\in\funs{\naturals}{\funs{D}{\carrier[Y]}}$.
\end{proposition}
\begin{proof}
    $\uniformSequenceConverges{\uniformMapSpace{Y}{D}}{S}{g}$ by
        \obv{uniform_map_convergence}.
    Hence $S\in\funs{\naturals}{\carrier[\uniformMapSpace{Y}{D}]}$ by
        \obv{uniform_sequence_convergence_sequence}.
    $\carrier[\uniformMapSpace{Y}{D}]=\funs{D}{\carrier[Y]}$ by
        \obv{uniform_map_space_carrier}.
\end{proof}

\begin{proposition}\label{uniform_map_convergence_limit}
    Suppose $\uniformMapConverges{Y}{D}{S}{g}$.
    Then $g\in\funs{D}{\carrier[Y]}$.
\end{proposition}
\begin{proof}
    $\uniformSequenceConverges{\uniformMapSpace{Y}{D}}{S}{g}$ by
        \obv{uniform_map_convergence}.
    Hence $g\in\carrier[\uniformMapSpace{Y}{D}]$ by
        \obv{uniform_sequence_convergence_point}.
    $\carrier[\uniformMapSpace{Y}{D}]=\funs{D}{\carrier[Y]}$ by
        \obv{uniform_map_space_carrier}.
\end{proof}

\begin{proposition}\label{uniform_map_cauchy_target}
    Suppose $\uniformMapCauchy{Y}{D}{S}$.
    Then $Y$ is a uniform space.
\end{proposition}
\begin{proof}
    Follows by \obv{uniform_map_cauchy}.
\end{proof}

\begin{proposition}\label{uniform_map_cauchy_sequence}
    Suppose $\uniformMapCauchy{Y}{D}{S}$.
    Then $S\in\funs{\naturals}{\funs{D}{\carrier[Y]}}$.
\end{proposition}
\begin{proof}
    $\uniformCauchySequence{\uniformMapSpace{Y}{D}}{S}$ by
        \obv{uniform_map_cauchy}.
    Hence $S\in\funs{\naturals}{\carrier[\uniformMapSpace{Y}{D}]}$ by
        \obv{uniform_cauchy_sequence_sequence}.
    $\carrier[\uniformMapSpace{Y}{D}]=\funs{D}{\carrier[Y]}$ by
        \obv{uniform_map_space_carrier}.
\end{proof}

\begin{theorem}\label{uniform_map_convergence_iff_entourages}
    Let $Y$ be a uniform space.
    Then $\uniformMapConverges{Y}{D}{S}{g}$ iff
        $S\in\funs{\naturals}{\funs{D}{\carrier[Y]}}$ and
        $g\in\funs{D}{\carrier[Y]}$ and for every $U\in\uniformity[Y]$
        there exists $N\in\naturals$ such that for every $n\in\naturals$
        such that $N\subseteq n$ we have for every $x\in D$ we have
            $(g(x),\apply{\apply{S}{n}}{x})\in U$.
\end{theorem}
\begin{proof}
    $\uniformMapSpace{Y}{D}$ is a uniform space by
        \obv{uniform_map_space_is_uniform_space}.
    $\carrier[\uniformMapSpace{Y}{D}]=\funs{D}{\carrier[Y]}$ by
        \obv{uniform_map_space_carrier}.

    Show that if $\uniformMapConverges{Y}{D}{S}{g}$, then
        $S\in\funs{\naturals}{\funs{D}{\carrier[Y]}}$ and
        $g\in\funs{D}{\carrier[Y]}$ and for every $U\in\uniformity[Y]$
        there exists $N\in\naturals$ such that for every $n\in\naturals$
        such that $N\subseteq n$ we have for every $x\in D$ we have
            $(g(x),\apply{\apply{S}{n}}{x})\in U$.
    \begin{subproof}
        Assume $\uniformMapConverges{Y}{D}{S}{g}$.
        $S\in\funs{\naturals}{\funs{D}{\carrier[Y]}}$ by
            \obv{uniform_map_convergence_sequence}.
        $g\in\funs{D}{\carrier[Y]}$ by
            \obv{uniform_map_convergence_limit}.
        $\uniformSequenceConverges{\uniformMapSpace{Y}{D}}{S}{g}$ by
            \obv{uniform_map_convergence}.
        Show that for every $U\in\uniformity[Y]$ there exists
            $N\in\naturals$ such that for every $n\in\naturals$ such that
            $N\subseteq n$ we have for every $x\in D$ we have
            $(g(x),\apply{\apply{S}{n}}{x})\in U$.
        \begin{subproof}
            Fix $U\in\uniformity[Y]$.
            $\uniformMapEntourage{Y}{D}{U}\subseteq
                \funs{D}{\carrier[Y]}\times\funs{D}{\carrier[Y]}$ by
                \obv{uniform_map_entourage_type}.
            Hence $\uniformMapEntourage{Y}{D}{U}\in
                \uniformity[\uniformMapSpace{Y}{D}]$ by
                \auto{uniform_map_space_uniformity_iff,subseteq_refl}.
            Take $N\in\naturals$ such that for every $n\in\naturals$ such
                that $N\subseteq n$ we have
                $(g,\apply{S}{n})\in\uniformMapEntourage{Y}{D}{U}$ by
                \obv{uniform_sequence_convergence_entourage}.
            Show that for every $n\in\naturals$ such that $N\subseteq n$
                we have for every $x\in D$ we have
                $(g(x),\apply{\apply{S}{n}}{x})\in U$.
            \begin{subproof}
                Fix $n\in\naturals$.
                Assume $N\subseteq n$.
                $(g,\apply{S}{n})\in\uniformMapEntourage{Y}{D}{U}$.
                Follows by \auto{uniform_map_entourage_iff}.
            \end{subproof}
            Follows by \auto.
        \end{subproof}
    \end{subproof}

    Show that if $S\in\funs{\naturals}{\funs{D}{\carrier[Y]}}$ and
        $g\in\funs{D}{\carrier[Y]}$ and for every $U\in\uniformity[Y]$
        there exists $N\in\naturals$ such that for every $n\in\naturals$
        such that $N\subseteq n$ we have for every $x\in D$ we have
            $(g(x),\apply{\apply{S}{n}}{x})\in U$, then
        $\uniformMapConverges{Y}{D}{S}{g}$.
    \begin{subproof}
        Assume $S\in\funs{\naturals}{\funs{D}{\carrier[Y]}}$ and
            $g\in\funs{D}{\carrier[Y]}$ and for every
            $U\in\uniformity[Y]$ there exists $N\in\naturals$ such that for
            every $n\in\naturals$ such that $N\subseteq n$ we have for
            every $x\in D$ we have
            $(g(x),\apply{\apply{S}{n}}{x})\in U$.
        Hence $S\in\funs{\naturals}
            {\carrier[\uniformMapSpace{Y}{D}]}$ and
            $g\in\carrier[\uniformMapSpace{Y}{D}]$.
        Show that for every $E\in\uniformity[\uniformMapSpace{Y}{D}]$
            there exists $N\in\naturals$ such that for every
            $n\in\naturals$ such that $N\subseteq n$ we have
            $(g,\apply{S}{n})\in E$.
        \begin{subproof}
            Fix $E\in\uniformity[\uniformMapSpace{Y}{D}]$.
            Take $U\in\uniformity[Y]$ such that
                $\uniformMapEntourage{Y}{D}{U}\subseteq E$ by
                \auto{uniform_map_space_uniformity_iff}.
            Take $N\in\naturals$ such that for every $n\in\naturals$ such
                that $N\subseteq n$ we have for every $x\in D$ we have
                $(g(x),\apply{\apply{S}{n}}{x})\in U$.
            Show that for every $n\in\naturals$ such that $N\subseteq n$
                we have $(g,\apply{S}{n})\in E$.
            \begin{subproof}
                Fix $n\in\naturals$.
                Assume $N\subseteq n$.
                $\apply{S}{n}\in\funs{D}{\carrier[Y]}$ by
                    \obv{funs_type_apply}.
                For every $x\in D$ we have
                    $(g(x),\apply{\apply{S}{n}}{x})\in U$.
                Hence $(g,\apply{S}{n})\in
                    \uniformMapEntourage{Y}{D}{U}$ by
                    \auto{uniform_map_entourage_iff}.
                Follows by \obv{subseteq}.
            \end{subproof}
            Follows by \auto.
        \end{subproof}
        $\uniformSequenceConverges{\uniformMapSpace{Y}{D}}{S}{g}$ iff
            $S\in\funs{\naturals}
                {\carrier[\uniformMapSpace{Y}{D}]}$ and
            $g\in\carrier[\uniformMapSpace{Y}{D}]$ and for every
            $E\in\uniformity[\uniformMapSpace{Y}{D}]$ there exists
            $N\in\naturals$ such that for every $n\in\naturals$ such that
            $N\subseteq n$ we have $(g,\apply{S}{n})\in E$ by
            \obv{uniform_sequence_convergence_iff_entourages}.
        Hence $\uniformSequenceConverges{\uniformMapSpace{Y}{D}}{S}{g}$.
        Follows by \obv{uniform_map_convergence}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{proposition}\label{uniform_map_convergence_entourage}
    Suppose $\uniformMapConverges{Y}{D}{S}{g}$ and $U\in\uniformity[Y]$.
    Then there exists $N\in\naturals$ such that for every
        $n\in\naturals$ such that $N\subseteq n$ we have for every
        $x\in D$ we have $(g(x),\apply{\apply{S}{n}}{x})\in U$.
\end{proposition}
\begin{proof}
    $S\in\funs{\naturals}{\funs{D}{\carrier[Y]}}$ by
        \obv{uniform_map_convergence_sequence}.
    $g\in\funs{D}{\carrier[Y]}$ by
        \obv{uniform_map_convergence_limit}.
    $\uniformMapConverges{Y}{D}{S}{g}$ iff
        $S\in\funs{\naturals}{\funs{D}{\carrier[Y]}}$ and
        $g\in\funs{D}{\carrier[Y]}$ and for every $V\in\uniformity[Y]$
        there exists $N\in\naturals$ such that for every $n\in\naturals$
        such that $N\subseteq n$ we have for every $x\in D$ we have
            $(g(x),\apply{\apply{S}{n}}{x})\in V$ by
        \obv{uniform_map_convergence_iff_entourages}.
    Follows by \auto.
\end{proof}

        $S\in\funs{\naturals}{\funs{D}{\carrier[Y]}}$ by
            \obv{uniform_map_cauchy_sequence}.
        $\uniformCauchySequence{\uniformMapSpace{Y}{D}}{S}$ by
            \obv{uniform_map_cauchy}.
        Show that for every $U\in\uniformity[Y]$ there exists
            $N\in\naturals$ such that for all $m,n\in\naturals$ such that
            $N\subseteq m$ and $N\subseteq n$ we have for every $x\in D$

                $m,n\in\naturals$ such that $N\subseteq m$ and

                and $N\subseteq n$ we have for every $x\in D$ we have

            \begin{subproof}

                Assume $N\subseteq m$ and $N\subseteq n$.
                $(\apply{S}{m},\apply{S}{n})
                    \in\uniformMapEntourage{Y}{D}{U}$.
                Follows by \auto{uniform_map_entourage_iff}.
            \end{subproof}
            Follows by \auto.
        \end{subproof}
    \end{subproof}

    Show that if $S\in\funs{\naturals}{\funs{D}{\carrier[Y]}}$ and for
        every $U\in\uniformity[Y]$ there exists $N\in\naturals$ such that
        for all $m,n\in\naturals$ such that $N\subseteq m$ and
        $N\subseteq n$ we have for every $x\in D$ we have

    \begin{subproof}
        Assume $S\in\funs{\naturals}{\funs{D}{\carrier[Y]}}$ and for every

             \apply{\apply{S}{n}}{x})\in U$.
        Hence $S\in\funs{\naturals}
            {\carrier[\uniformMapSpace{Y}{D}]}$.

\begin{theorem}\label{metric_uniform_map_convergence_iff}
    Let $M$ be a metric space.
    Suppose $S\in\funs{\naturals}{\funs{D}{\carrier[M]}}$ and
        $g\in\funs{D}{\carrier[M]}$.
    Then
        $\uniformMapConverges{\metricUniformSpace{M}}{D}{S}{g}$ iff
        for every $e\in\reals$ such that
            $\surrealLt{\surrealZero}{e}$ there exists $N\in\naturals$
            such that for every $n\in\naturals$ such that
                $N\subseteq n$ we have for every $x\in D$ we have
                $\surrealLt{\metricDistance[M]
                    (g(x),\apply{\apply{S}{n}}{x})}{e}$.
\end{theorem}
\begin{proof}
    $M$ is a pseudometric space by
        \obv{metric_is_pseudometric_space}.
    $\metricUniformSpace{M}$ is a uniform space by
        \obv{metric_uniform_space_is_uniform_space}.
    $\carrier[\metricUniformSpace{M}]=\carrier[M]$ and
        $\uniformity[\metricUniformSpace{M}]=\metricUniformity{M}$
        by
        \obv{metric_uniform_space_carrier,metric_uniform_space_uniformity}.
    Show that if
        $\uniformMapConverges{\metricUniformSpace{M}}{D}{S}{g}$, then
        for every $e\in\reals$ such that
            $\surrealLt{\surrealZero}{e}$ there exists $N\in\naturals$
            such that for every $n\in\naturals$ such that
                $N\subseteq n$ we have for every $x\in D$ we have
                $\surrealLt{\metricDistance[M]
                    (g(x),\apply{\apply{S}{n}}{x})}{e}$.
    \begin{subproof}
        Assume $\uniformMapConverges
            {\metricUniformSpace{M}}{D}{S}{g}$.
        Fix $e\in\reals$.
        Assume $\surrealLt{\surrealZero}{e}$.
        $\metricEntourage{M}{e}\in
            \uniformity[\metricUniformSpace{M}]$
            by
            \obv{metric_entourage_is_entourage,metric_uniform_space_uniformity}.
        Take $N\in\naturals$ such that for every $n\in\naturals$
            such that $N\subseteq n$ we have for every $x\in D$ we have
            $(g(x),\apply{\apply{S}{n}}{x})
                \in\metricEntourage{M}{e}$
            by \obv{uniform_map_convergence_entourage}.
        Show that for every $n\in\naturals$ such that $N\subseteq n$
            we have for every $x\in D$ we have
            $\surrealLt{\metricDistance[M]
                (g(x),\apply{\apply{S}{n}}{x})}{e}$.
        \begin{subproof}
            Fix $n\in\naturals$.
            Assume $N\subseteq n$.
            Fix $x\in D$.
            $(g(x),\apply{\apply{S}{n}}{x})
                \in\metricEntourage{M}{e}$.
            Follows by \auto{metric_entourage_iff}.
        \end{subproof}
        Follows by \auto.
    \end{subproof}
    Show that if for every $e\in\reals$ such that
        $\surrealLt{\surrealZero}{e}$ there exists $N\in\naturals$
        such that for every $n\in\naturals$ such that
            $N\subseteq n$ we have for every $x\in D$ we have
            $\surrealLt{\metricDistance[M]
                (g(x),\apply{\apply{S}{n}}{x})}{e}$, then
        $\uniformMapConverges{\metricUniformSpace{M}}{D}{S}{g}$.
    \begin{subproof}
        Assume for every $e\in\reals$ such that
            $\surrealLt{\surrealZero}{e}$ there exists $N\in\naturals$
            such that for every $n\in\naturals$ such that
                $N\subseteq n$ we have for every $x\in D$ we have
                $\surrealLt{\metricDistance[M]
                    (g(x),\apply{\apply{S}{n}}{x})}{e}$.
        Show that for every
            $U\in\uniformity[\metricUniformSpace{M}]$ there exists
            $N\in\naturals$ such that for every $n\in\naturals$
            such that $N\subseteq n$ we have for every $x\in D$ we have
            $(g(x),\apply{\apply{S}{n}}{x})\in U$.
        \begin{subproof}
            Fix $U\in\uniformity[\metricUniformSpace{M}]$.
            $U\in\metricUniformity{M}$.
            Take $e\in\reals$ such that
                $\surrealLt{\surrealZero}{e}$ and
                $\metricEntourage{M}{e}\subseteq U$
                by \obv{metric_uniformity}.
            Take $N\in\naturals$ such that for every $n\in\naturals$
                such that $N\subseteq n$ we have for every $x\in D$ we have
                $\surrealLt{\metricDistance[M]
                    (g(x),\apply{\apply{S}{n}}{x})}{e}$.
            Show that for every $n\in\naturals$ such that
                $N\subseteq n$ we have for every $x\in D$ we have
                $(g(x),\apply{\apply{S}{n}}{x})\in U$.
            \begin{subproof}
                Fix $n\in\naturals$.
                Assume $N\subseteq n$.
                Fix $x\in D$.
                $g(x),\apply{\apply{S}{n}}{x}\in\carrier[M]$
                    by \auto{funs_type_apply}.
                $(g(x),\apply{\apply{S}{n}}{x})
                    \in\metricEntourage{M}{e}$
                    by \auto{metric_entourage_iff}.
                Follows by \obv{subseteq}.
            \end{subproof}
            Follows by \auto.
        \end{subproof}
        Follows by \auto{uniform_map_convergence_iff_entourages}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\subsection{Canonical pointwise limits}

\begin{definition}\label{map_sequence_at}
    $\mapSequenceAt{S}{x}=\Apply{\graphOnDomainOp}
        {\naturals,\fun{n:\Set}{\apply{\apply{S}{n}}{x}}}$.
\end{definition}

\begin{proposition}\label{map_sequence_at_typed}
    Suppose $S\in\funs{\naturals}{\funs{D}{C}}$ and $x\in D$.
    Then $\mapSequenceAt{S}{x}\in\funs{\naturals}{C}$.
\end{proposition}
\begin{proof}
    For every $n\in\naturals$ we have
        $\apply{\apply{S}{n}}{x}\in C$
        by \auto{funs_type_apply}.
    Follows by \auto{map_sequence_at,logical_graph_on_domain_type}.
\end{proof}

\begin{proposition}\label{map_sequence_at_apply}
    Let $n\in\naturals$.
    Then $\apply{\mapSequenceAt{S}{x}}{n}
        =\apply{\apply{S}{n}}{x}$.
\end{proposition}
\begin{proof}
    Follows by \auto{map_sequence_at,logical_graph_on_domain_apply}.
\end{proof}

\begin{theorem}\label{uniform_map_convergence_pointwise}
    Let $M$ be a metric space.

                by \obv{metric_ball}.
