\section{Clopen bases}

\import{topology/basis.tex}

We formulate the clopen-basis property locally, without choosing a particular
basis.

\begin{definition}\label{has_clopen_basis}
    Let $X$ be a topological space.
    $X$ has a clopen basis iff
        for every $U\in\opens[X]$ we have for every $x\in U$
        there exists $V$ such that
            $V$ is clopen in $X$ and $x\in V\subseteq U$.
\end{definition}

\begin{theorem}\label{clopen_open_basis_implies_has_clopen_basis}
    Let $X$ be a topological space.
    Suppose $B$ is an open basis of $X$.
    Suppose for every $V\in B$ we have $V$ is clopen in $X$.
    Then $X$ has a clopen basis.
\end{theorem}
\begin{proof}
    Show that for every $U\in\opens[X]$ we have for every $x\in U$
        there exists $V$ such that
            $V$ is clopen in $X$ and $x\in V\subseteq U$.
    \begin{subproof}
        Fix $U\in\opens[X]$.
        Fix $x\in U$.
        Take $V\in B$ such that $x\in V\subseteq U$
            by \obv{open_basis}.
        $V$ is clopen in $X$.
    \end{subproof}
    Follows by \obv{has_clopen_basis}.
\end{proof}
