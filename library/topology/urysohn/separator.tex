\section{The deterministic Urysohn separator}

For disjoint closed sets $A$ and $B$, shrink the open complement of $B$
around $A$. The nested-open construction then gives a continuous function
that is one on $A$ and zero on $B$.

We fix the shrinker as a parameter. The separator is a total term; its
separation properties hold under the hypotheses below. In Urysohn's lemma
we will interchange $A$ and $B$ to obtain the usual endpoint order.

\begin{definition}\label{normal_shrinker_separator}
    $\normalShrinkerSeparator{X}{h}{A}{B}
        =\openChainLimitMap{X}{h}
            {h((A,\carrier[X]\setminus B))}
            {\carrier[X]\setminus B}$.
\end{definition}

\begin{proposition}\label{normal_shrinker_separator_apply}
    Let $x\in\carrier[X]$.
    Then
        $\apply{\normalShrinkerSeparator{X}{h}{A}{B}}{x}
            =\openChainLimitValue{X}{h}
                {h((A,\carrier[X]\setminus B))}
                {\carrier[X]\setminus B}{x}$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{normal_shrinker_separator,open_chain_limit_map_apply}.
\end{proof}

\begin{proposition}\label{normal_shrinker_separator_open_data}
    Let $X$ be a topological space.
    Let $h$ be a normal shrinker for $X$.
    Let $A,B\in\closeds{X}$.
    Suppose $A$ is disjoint from $B$.
    Let $W=\carrier[X]\setminus B$ and $U=h((A,W))$.
    Then $U,W\in\opens[X]$ and
        $A\subseteq U$ and $\closure{U}{X}\subseteq W$ and
        $\carrier[X]\setminus W=B$.
\end{proposition}
\begin{proof}
    $A,B\subseteq\carrier[X]$ by \obv{closeds,pow_iff,subseteq}.
    $W\in\opens[X]$ by \obv{closeds,is_closed_in}.
    $A\subseteq W$ by \auto{subseteq,setminus,disjoint}.
    $U\in\opens[X]$ and
        $A\subseteq U\subseteq\closure{U}{X}\subseteq W$
        by \auto{normal_shrinker_apply_pair}.
    $\carrier[X]\setminus W=B$
        by \auto{double_relative_complement}.
\end{proof}

\begin{theorem}\label{normal_shrinker_separator_typed_real}
    Let $X$ be a topological space.
    Let $h$ be a normal shrinker for $X$.
    Let $A,B\in\closeds{X}$.
    Suppose $A$ is disjoint from $B$.
    Then $\normalShrinkerSeparator{X}{h}{A}{B}
        \in\funs{\carrier[X]}{\reals}$.
\end{theorem}
\begin{proof}
    Follows by
        \auto{normal_shrinker_separator,normal_shrinker_separator_open_data,
            open_chain_limit_map_typed}.
\end{proof}

\begin{theorem}\label{normal_shrinker_separator_between_zero_and_one}
    Let $X$ be a topological space.
    Let $h$ be a normal shrinker for $X$.
    Let $A,B\in\closeds{X}$.
    Suppose $A$ is disjoint from $B$.
    Then for every $x\in\carrier[X]$ we have
        $\surrealLe{\surrealZero}
            {\apply{\normalShrinkerSeparator{X}{h}{A}{B}}{x}}$ and
        $\surrealLe{\apply{\normalShrinkerSeparator{X}{h}{A}{B}}{x}}
            {\surrealOne}$.
\end{theorem}
\begin{proof}
    Follows by
        \auto{normal_shrinker_separator,normal_shrinker_separator_open_data,
            open_chain_limit_map_between_zero_and_one}.
\end{proof}

\begin{theorem}\label{normal_shrinker_separator_one_on_first}
    Let $X$ be a topological space.
    Let $h$ be a normal shrinker for $X$.
    Let $A,B\in\closeds{X}$.
    Suppose $A$ is disjoint from $B$.
    Then for every $x\in A$ we have
        $\apply{\normalShrinkerSeparator{X}{h}{A}{B}}{x}=\surrealOne$.
\end{theorem}
\begin{proof}
    Let $W=\carrier[X]\setminus B$ and $U=h((A,W))$.
    $U,W\in\opens[X]$ and $A\subseteq U$ and
        $\closure{U}{X}\subseteq W$
        by \auto{normal_shrinker_separator_open_data}.
    Follows by
        \auto{normal_shrinker_separator,open_chain_limit_map_one_on_first,subseteq}.
\end{proof}

\begin{theorem}\label{normal_shrinker_separator_zero_on_second}
    Let $X$ be a topological space.
    Let $h$ be a normal shrinker for $X$.
    Let $A,B\in\closeds{X}$.
    Suppose $A$ is disjoint from $B$.
    Then for every $x\in B$ we have
        $\apply{\normalShrinkerSeparator{X}{h}{A}{B}}{x}=\surrealZero$.
\end{theorem}
\begin{proof}
    Let $W=\carrier[X]\setminus B$ and $U=h((A,W))$.
    $U,W\in\opens[X]$ and $\closure{U}{X}\subseteq W$ and
        $\carrier[X]\setminus W=B$
        by \auto{normal_shrinker_separator_open_data}.
    Follows by
        \auto{normal_shrinker_separator,open_chain_limit_map_zero_outside_last}.
\end{proof}
