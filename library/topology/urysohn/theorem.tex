\section{Urysohn's lemma}

\import{topology/urysohn/separator.tex}

The construction is now complete: a fixed normal shrinker gives a continuous
separator for any two disjoint closed sets. Normality supplies such a
shrinker, yielding Urysohn's lemma. We then prove the converse and the
bump-function form.

\subsection{Closed sets and a supplied normal shrinker}

\begin{theorem}[Separation from a fixed shrinker]\label{urysohn_map_from_normal_shrinker}
    Let $X$ be a topological space.
    Let $h$ be a normal shrinker for $X$.
    Let $A,B\in\closeds{X}$.
    Suppose $A$ is disjoint from $B$.
    Then there exists $f\in\funs{\carrier[X]}{\RealUnitInterval}$
        such that $f$ is continuous from $X$ to
        $\RealUnitIntervalTopology$ and
        $f$ is continuous from $X$ to $\RealTopology$ and
        (for every $x\in\carrier[X]$ we have
            $\surrealLe{\surrealZero}{f(x)}$ and
            $\surrealLe{f(x)}{\surrealOne}$) and
        $f$ is constantly $\surrealOne$ on $A$ and
        $f$ is constantly $\surrealZero$ on $B$.
\end{theorem}
\begin{proof}
    Let $f=\normalShrinkerSeparator{X}{h}{A}{B}$.
    $f\in\funs{\carrier[X]}{\RealUnitInterval}$
        by \auto{normal_shrinker_separator_typed_unit_interval}.
    $f$ is continuous from $X$ to $\RealUnitIntervalTopology$
        by \auto{normal_shrinker_separator_continuous_unit_interval}.
    $f$ is continuous from $X$ to $\RealTopology$
        by \auto{normal_shrinker_separator_continuous_real}.
    For every $x\in\carrier[X]$ we have
        $\surrealLe{\surrealZero}{f(x)}$ and
        $\surrealLe{f(x)}{\surrealOne}$
        by \auto{normal_shrinker_separator_between_zero_and_one}.
    $f$ is constantly $\surrealOne$ on $A$
        by \auto{normal_shrinker_separator_one_on_first}.
    $f$ is constantly $\surrealZero$ on $B$
        by \auto{normal_shrinker_separator_zero_on_second}.
\end{proof}

\subsection{Separation by a continuous function}

\begin{theorem}[Urysohn's lemma]\label{urysohn_lemma}
    Let $X$ be a topological space.
    Suppose $X$ is normal.
    Let $A,B\in\closeds{X}$.
    Suppose $A$ is disjoint from $B$.
    Then there exists $f\in\funs{\carrier[X]}{\RealUnitInterval}$
        such that $f$ is continuous from $X$ to
        $\RealUnitIntervalTopology$ and
        $f$ is constantly $\surrealZero$ on $A$ and
        $f$ is constantly $\surrealOne$ on $B$.
\end{theorem}
\begin{proof}
    Take a normal shrinker $h$ for $X$
        by \obv{normal_shrinker_exists}.
    $B$ is disjoint from $A$ by \obv{disjoint_symmetric}.
    Take $f\in\funs{\carrier[X]}{\RealUnitInterval}$ such that
        $f$ is continuous from $X$ to $\RealUnitIntervalTopology$ and
        $f$ is constantly $\surrealOne$ on $B$ and
        $f$ is constantly $\surrealZero$ on $A$
        by \auto{urysohn_map_from_normal_shrinker}.
    Follows by \auto.
\end{proof}

\subsection{Separation implies normality}

\begin{theorem}\label{unit_interval_separation_implies_normal}
    Let $X$ be a topological space.
    Suppose for all $C,D\in\closeds{X}$ such that
        $C$ is disjoint from $D$ there exists
        $f\in\funs{\carrier[X]}{\RealUnitInterval}$ such that
        $f$ is continuous from $X$ to $\RealUnitIntervalTopology$ and
        $f$ is constantly $\surrealZero$ on $C$ and
        $f$ is constantly $\surrealOne$ on $D$.
    Then $X$ is normal.
\end{theorem}
\begin{proof}
    Show that for all $C,D\in\closeds{X}$ such that
        $C$ is disjoint from $D$ there exist $O,V\in\opens[X]$ such that
        $C\subseteq O$ and $D\subseteq V$ and
        $O$ is disjoint from $V$.
    \begin{subproof}
        Fix $C,D\in\closeds{X}$.
        Assume $C$ is disjoint from $D$.
        Take $f\in\funs{\carrier[X]}{\RealUnitInterval}$ such that
            $f$ is continuous from $X$ to
            $\RealUnitIntervalTopology$ and
            $f$ is constantly $\surrealZero$ on $C$ and
            $f$ is constantly $\surrealOne$ on $D$.
        Let $n=\suc{\emptyset}$.
        $\emptyset\in\naturals$ by \obv{omega_empty}.
        Hence $n$ is a natural number by \auto{omega_successor}.
        Let $e=\surrealEpsilon{n}$.
        $e\in\reals$ by \auto{surreal_epsilon_is_real}.
        $\surrealLt{\surrealZero}{e}$
            by \obv{surreal_epsilon_positive,natural_number_is_ordinal}.
        $\surrealZero,\surrealOne\in\carrier[\RealMetric]$
            by
            \obv{surreal_real_metric_carrier,surreal_zero_is_real,surreal_one_is_real}.
        Let $P_0=\metricBall{\RealMetric}{\surrealZero}{e}$.
        Let $Q_0=\metricBall{\RealMetric}{\surrealOne}{e}$.
        $P_0,Q_0\in\metricOpens{\RealMetric}$
            by \auto{metric_ball_is_open,surreal_real_pseudometric_space}.
        $\opens[\RealTopology]=\metricOpens{\RealMetric}$
            by \auto{real_topology,metric_topology_opens}.
        Hence $P_0,Q_0\in\opens[\RealTopology]$.
        Let $P=P_0\inter\RealUnitInterval$.
        Let $Q=Q_0\inter\RealUnitInterval$.
        $\carrier[\RealUnitIntervalTopology]=\RealUnitInterval$
            by \obv{real_unit_interval_topology_carrier}.
        Hence
            $P=P_0\inter\carrier[\RealUnitIntervalTopology]$ and
            $Q=Q_0\inter\carrier[\RealUnitIntervalTopology]$.
        $\RealUnitIntervalTopology
            =\subspaceTopology{\RealTopology}{\RealUnitInterval}$
            by \obv{real_unit_interval_topology}.
        Hence $P,Q\in\opens[\RealUnitIntervalTopology]$
            by \auto{subspace_open_iff,real_topology_is_topological_space}.
        Let $O=\preimg{f}{P}$.
        Let $V=\preimg{f}{Q}$.
        $O,V\in\opens[X]$ by \auto{continuous_map}.
        $\surrealZero\in P_0$ and $\surrealOne\in Q_0$
            by
            \obv{metric_ball_contains_centre,surreal_real_pseudometric_space}.
        $\surrealZero,\surrealOne\in\RealUnitInterval$
            by
            \obv{surreal_zero_in_real_unit_interval,surreal_one_in_real_unit_interval}.
        Hence $\surrealZero\in P$ and $\surrealOne\in Q$
            by \obv{inter}.
        $C,D\subseteq\carrier[X]$
            by \auto{closeds,pow_iff,subseteq}.
        $f$ is a function and $\dom{f}=\carrier[X]$
            by \obv{funs_is_function,funs}.
        $C\subseteq O$ by \auto{function_preimage_iff,subseteq}.
        $D\subseteq V$ by \auto{function_preimage_iff,subseteq}.
        $\surrealAdd{e}{e}=\surrealEpsilon{\emptyset}$
            by \auto{surreal_epsilon_halving,emptyset_is_ordinal}.
        Hence $\surrealAdd{e}{e}=\surrealOne$
            by \obv{surreal_epsilon_zero}.
        $\metricDistance[\RealMetric](\surrealZero,\surrealOne)
            =\surrealOne$
            by \obv{surreal_real_metric_zero_one_distance}.
        Hence
            $\surrealLe{\surrealAdd{e}{e}}
                {\metricDistance[\RealMetric]
                    (\surrealZero,\surrealOne)}$
            by \obv{surreal_le_reflexive,surreal_one_is_surreal}.
        Hence $P_0\inter Q_0=\emptyset$
            by
            \auto{metric_balls_disjoint,surreal_real_pseudometric_space}.
        Hence $P$ is disjoint from $Q$
            by \auto{disjoint,inter,emptyset}.
        Hence $O$ is disjoint from $V$
            by \auto{disjoint,function_preimage_iff}.
        Follows by \auto.
    \end{subproof}
    Follows by \obv{is_normal}.
\end{proof}

\subsection{Functional characterization of normality}

\begin{theorem}\label{normal_iff_unit_interval_separation}
    Let $X$ be a topological space.
    Then $X$ is normal iff for all $A,B\in\closeds{X}$ such that
        $A$ is disjoint from $B$ there exists
        $f\in\funs{\carrier[X]}{\RealUnitInterval}$ such that
        $f$ is continuous from $X$ to $\RealUnitIntervalTopology$ and
        $f$ is constantly $\surrealZero$ on $A$ and
        $f$ is constantly $\surrealOne$ on $B$.
\end{theorem}
\begin{proof}
    Follows by \obv{urysohn_lemma,unit_interval_separation_implies_normal}.
\end{proof}

\subsection{The bump-function form}

\begin{corollary}[A bump function]\label{urysohn_bump_function}
    Let $X$ be a topological space.
    Suppose $X$ is normal.
    Let $C\in\closeds{X}$.
    Let $U\in\opens[X]$.
    Suppose $C\subseteq U$.
    Then there exists $f\in\funs{\carrier[X]}{\RealUnitInterval}$
        such that $f$ is continuous from $X$ to
        $\RealUnitIntervalTopology$ and
        $f$ is constantly $\surrealOne$ on $C$ and
        $f$ is constantly $\surrealZero$ on $\carrier[X]\setminus U$.
\end{corollary}
\begin{proof}
    Let $A=\carrier[X]\setminus U$.
    $A\in\closeds{X}$ by \auto{complement_of_open_elem_closeds}.
    $C$ is disjoint from $A$
        by \obv{disjoint,setminus,subseteq}.
    Hence $A$ is disjoint from $C$ by \obv{disjoint_symmetric}.
    Take $f\in\funs{\carrier[X]}{\RealUnitInterval}$ such that
        $f$ is continuous from $X$ to $\RealUnitIntervalTopology$ and
        $f$ is constantly $\surrealZero$ on $A$ and
        $f$ is constantly $\surrealOne$ on $C$
        by \obv{urysohn_lemma}.
    Follows by \auto.
\end{proof}

\subsection{Real-valued separation}

\begin{theorem}[Real-valued separation]\label{urysohn_lemma_real}
    Let $X$ be a topological space.
    Suppose $X$ is normal.
    Let $A,B\in\closeds{X}$.
    Suppose $A$ is disjoint from $B$.
    Then there exists $f\in\funs{\carrier[X]}{\reals}$
        such that $f$ is continuous from $X$ to $\RealTopology$ and
        (for every $x\in\carrier[X]$ we have
            $\surrealLe{\surrealZero}{f(x)}$ and
            $\surrealLe{f(x)}{\surrealOne}$) and
        $f$ is constantly $\surrealZero$ on $A$ and
        $f$ is constantly $\surrealOne$ on $B$.
\end{theorem}
\begin{proof}
    Take a normal shrinker $h$ for $X$
        by \obv{normal_shrinker_exists}.
    $B$ is disjoint from $A$ by \obv{disjoint_symmetric}.
    Take $f\in\funs{\carrier[X]}{\RealUnitInterval}$ such that
        $f$ is continuous from $X$ to $\RealTopology$ and
        (for every $x\in\carrier[X]$ we have
            $\surrealLe{\surrealZero}{f(x)}$ and
            $\surrealLe{f(x)}{\surrealOne}$) and
        $f$ is constantly $\surrealOne$ on $B$ and
        $f$ is constantly $\surrealZero$ on $A$
        by \auto{urysohn_map_from_normal_shrinker}.
    $\RealUnitInterval\subseteq\reals$
        by \obv{real_unit_interval_subset_reals}.
    Hence $f\in\funs{\carrier[X]}{\reals}$
        by \obv{funs_weaken_codom}.
\end{proof}
