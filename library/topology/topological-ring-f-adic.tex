\section{F-adic presentations}

The powers in an f-adic presentation are the existing positive powers in the
semigroup of additively generated subsets.  In particular, they are not raw
pointwise powers.

\begin{definition}\label{f_adic_power}
    $\fAdicPower{R}{U}{n}=\additiveSubsetPower{R}{U}{n}$.
\end{definition}

\begin{proposition}\label{f_adic_power_type}
    Let $R$ be a ring. Suppose $U\subseteq\carrier[R]$.
    Let $n$ be a natural number such that $n\neq\emptyset$.
    Then $\fAdicPower{R}{U}{n}\subseteq\carrier[R]$.
\end{proposition}
\begin{proof}
    Follows by \auto{f_adic_power,additive_subset_power_type}.
\end{proof}

\begin{proposition}\label{f_adic_power_one}
    Let $R$ be a ring. Suppose $U\subseteq\carrier[R]$.
    Then $\fAdicPower{R}{U}{\suc{\emptyset}}=U$.
\end{proposition}
\begin{proof}
    Follows by \obv{f_adic_power,additive_subset_power_one}.
\end{proof}

\begin{proposition}\label{f_adic_power_successor}
    Let $R$ be a ring. Suppose $U\subseteq\carrier[R]$.
    Let $n$ be a natural number such that $n\neq\emptyset$.
    Then $\fAdicPower{R}{U}{\suc{n}}
        =\additiveSubsetProduct{R}{\fAdicPower{R}{U}{n}}{U}$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{f_adic_power,additive_subset_power_successor}.
\end{proof}

\begin{theorem}\label{f_adic_power_antitone}
    Let $R$ be a ring. Suppose $U\subseteq\carrier[R]$.
    Suppose $\additiveSubsetProduct{R}{U}{U}\subseteq U$.
    Let $m,n$ be natural numbers.
    Suppose $m,n\neq\emptyset$ and $m\subseteq n$.
    Then $\fAdicPower{R}{U}{n}\subseteq\fAdicPower{R}{U}{m}$.
\end{theorem}
\begin{proof}
    Follows by \auto{f_adic_power,additive_subset_power_antitone}.
\end{proof}

\begin{theorem}\label{f_adic_power_subset_of_contraction}
    Let $R$ be a ring. Suppose $U\subseteq\carrier[R]$.
    Suppose $\additiveSubsetProduct{R}{U}{U}\subseteq U$.
    Let $n$ be a natural number such that $n\neq\emptyset$.
    Then $\fAdicPower{R}{U}{n}\subseteq U$.
\end{theorem}
\begin{proof}
    Follows by
        \auto{f_adic_power,additive_subset_power_subset_of_square_subset}.
\end{proof}

\begin{theorem}\label{f_adic_power_is_subgroup}
    Let $R$ be a ring. Suppose $U\subseteq\carrier[R]$.
    Let $n$ be a natural number such that
        $\suc{\suc{\emptyset}}\subseteq n$.
    Then $\subgroup{\fAdicPower{R}{U}{n}}{\AdditiveGroup{R}}$.
\end{theorem}
\begin{proof}
    Follows by \auto{f_adic_power,additive_subset_power_is_subgroup}.
\end{proof}

\begin{proposition}\label{f_adic_power_family_type}
    Let $R$ be a ring. Suppose $U\subseteq\carrier[R]$.
    Then
        $\Apply{\graphOnOp}{
            \naturals,\pow{\carrier[R]},
            \fun{n:\Set}{\fAdicPower{R}{U}{\suc{n}}}}
        \in\funs{\naturals}{\pow{\carrier[R]}}$.
\end{proposition}
\begin{proof}
    Show that for every $n\in\naturals$ we have
        $\fAdicPower{R}{U}{\suc{n}}\subseteq\carrier[R]$.
    \begin{subproof}
        Fix $n\in\naturals$.
        $\suc{n}\in\naturals$ and $\suc{n}\neq\emptyset$ by
            \obv{omega_successor,suc_neq_emptyset}.
        Follows by \obv{f_adic_power_type}.
    \end{subproof}
    Hence for every $n\in\naturals$ we have
        $\fAdicPower{R}{U}{\suc{n}}\in\pow{\carrier[R]}$ by
        \obv{powerset_intro}.
    Follows by \obv{logical_graph_on_function}.
\end{proof}

\begin{proposition}\label{f_adic_power_family_apply}
    Let $R$ be a ring. Suppose $U\subseteq\carrier[R]$.
    Let $n$ be a natural number.
    Then
        $(\Apply{\graphOnOp}{
            \naturals,\pow{\carrier[R]},
            \fun{k:\Set}{\fAdicPower{R}{U}{\suc{k}}}})(n)
        =\fAdicPower{R}{U}{\suc{n}}$.
\end{proposition}
\begin{proof}
    Follows by \auto{logical_graph_on_apply}.
\end{proof}

\begin{definition}\label{f_adic_presentation}
    $\fAdicPresentation{R}{U}{T}$ iff
        $R$ is a commutative topological ring and
        $U\subseteq\carrier[R]$ and $T$ is finite and $T\subseteq U$ and
        $\additiveSubsetProduct{R}{T}{U}
            =\additiveSubsetProduct{R}{U}{U}$ and
        $\additiveSubsetProduct{R}{U}{U}\subseteq U$ and
        (for every $n\in\naturals$ we have
            $\fAdicPower{R}{U}{\suc{n}}
                \in\neighbourhoods{\ringZero[R]}{R}$) and
        (for every $V\in\neighbourhoods{\ringZero[R]}{R}$ there exists
            $n\in\naturals$ such that
                $\fAdicPower{R}{U}{\suc{n}}\subseteq V$).
\end{definition}

\begin{proposition}\label{f_adic_presentation_ambient}
    Suppose $\fAdicPresentation{R}{U}{T}$.
    Then $R$ is a commutative topological ring and
        $U\subseteq\carrier[R]$ and $T$ is finite and $T\subseteq U$.
\end{proposition}
\begin{proof}
    Follows by \obv{f_adic_presentation}.
\end{proof}

\begin{proposition}\label{f_adic_presentation_contraction}
    Suppose $\fAdicPresentation{R}{U}{T}$.
    Then $\additiveSubsetProduct{R}{T}{U}
            =\additiveSubsetProduct{R}{U}{U}$ and
        $\additiveSubsetProduct{R}{U}{U}\subseteq U$.
\end{proposition}
\begin{proof}
    Follows by \obv{f_adic_presentation}.
\end{proof}

\begin{proposition}\label{f_adic_presentation_power_neighbourhood}
    Suppose $\fAdicPresentation{R}{U}{T}$.
    Let $n$ be a natural number.
    Then $\fAdicPower{R}{U}{\suc{n}}
        \in\neighbourhoods{\ringZero[R]}{R}$.
\end{proposition}
\begin{proof}
    Follows by \obv{f_adic_presentation}.
\end{proof}

\begin{proposition}\label{f_adic_presentation_iff}
    Let $R$ be a commutative topological ring.
    Then $\fAdicPresentation{R}{U}{T}$ iff
        $U\subseteq\carrier[R]$ and $T$ is finite and $T\subseteq U$ and
        $\additiveSubsetProduct{R}{T}{U}
            =\additiveSubsetProduct{R}{U}{U}$ and
        $\additiveSubsetProduct{R}{U}{U}\subseteq U$ and
        $\ran{\Apply{\graphOnOp}{
            \naturals,\pow{\carrier[R]},
            \fun{n:\Set}{\fAdicPower{R}{U}{\suc{n}}}}}$
            is a basis of $\neighbourhoods{\ringZero[R]}{R}$ on
                $\carrier[R]$.
\end{proposition}
\begin{proof}
    Let $b=\Apply{\graphOnOp}{
        \naturals,\pow{\carrier[R]},
        \fun{n:\Set}{\fAdicPower{R}{U}{\suc{n}}}}$.
    $R$ is a commutative ring by
        \obv{commutative_topological_ring_is_commutative_ring}.
    Hence $R$ is a ring by \obv{commutative_ring_is_ring}.
    $R$ is a topological ring by
        \obv{commutative_topological_ring_is_topological_ring}.
    Hence $R$ is a topological space by
        \obv{topological_ring_is_topological_space}.
    $\ringZero[R]\in\carrier[R]$ by \obv{ring_zero_type}.
    $\neighbourhoods{\ringZero[R]}{R}$ is a filter on $\carrier[R]$ by
        \obv{neighbourhoods_is_filter}.

    Show that if $\fAdicPresentation{R}{U}{T}$, then
        $U\subseteq\carrier[R]$ and $T$ is finite and $T\subseteq U$ and
        $\additiveSubsetProduct{R}{T}{U}
            =\additiveSubsetProduct{R}{U}{U}$ and
        $\additiveSubsetProduct{R}{U}{U}\subseteq U$ and
        $\ran{b}$ is a basis of
            $\neighbourhoods{\ringZero[R]}{R}$ on $\carrier[R]$.
    \begin{subproof}
        Assume $\fAdicPresentation{R}{U}{T}$.
        $U\subseteq\carrier[R]$ and $T$ is finite and $T\subseteq U$ by
            \obv{f_adic_presentation_ambient}.
        $\additiveSubsetProduct{R}{T}{U}
            =\additiveSubsetProduct{R}{U}{U}$ and
            $\additiveSubsetProduct{R}{U}{U}\subseteq U$ by
            \obv{f_adic_presentation_contraction}.
        $b\in\funs{\naturals}{\pow{\carrier[R]}}$ by
            \auto{f_adic_power_family_type}.
        Hence $\ran{b}\subseteq\pow{\carrier[R]}$ by \obv{funs_ran}.
        Show that $\ran{b}\subseteq
            \neighbourhoods{\ringZero[R]}{R}$.
        \begin{subproof}
            It suffices to show that every element of $\ran{b}$ is an
                element of $\neighbourhoods{\ringZero[R]}{R}$ by
                \obv{subseteq}.
            Fix $H\in\ran{b}$.
            Take $n$ such that $(n,H)\in b$ by \obv{ran_iff}.
            Hence $n\in\dom{b}$ by \obv{dom_intro}.
            Hence $n\in\naturals$ by \auto{funs}.
            Hence $H=b(n)$ by \auto{funs_is_function,function_apply_iff}.
            Hence $H=\fAdicPower{R}{U}{\suc{n}}$ by
                \auto{f_adic_power_family_apply}.
            $\fAdicPower{R}{U}{\suc{n}}
                \in\neighbourhoods{\ringZero[R]}{R}$ by
                \obv{f_adic_presentation}.
        \end{subproof}
        Show that $\ran{b}$ refines
            $\neighbourhoods{\ringZero[R]}{R}$ as filter families on
                $\carrier[R]$.
        \begin{subproof}
            $\ran{b}$ is a family of subsets of $\carrier[R]$ by
                \obv{pow_iff,subseteq}.
            $\neighbourhoods{\ringZero[R]}{R}$ is a family of subsets of
                $\carrier[R]$ by \obv{filter}.
            Show that for every
                $V\in\neighbourhoods{\ringZero[R]}{R}$ there exists
                $H\in\ran{b}$ such that $H\subseteq V$.
            \begin{subproof}
                Fix $V\in\neighbourhoods{\ringZero[R]}{R}$.
                Take $n\in\naturals$ such that
                    $\fAdicPower{R}{U}{\suc{n}}\subseteq V$ by
                    \obv{f_adic_presentation}.
                $b(n)=\fAdicPower{R}{U}{\suc{n}}$ by
                    \auto{f_adic_power_family_apply}.
                $b(n)\in\ran{b}$ by
                    \auto{funs_is_function,funs,img_of_function_intro,
                        ran_iff,function_apply_iff}.
            \end{subproof}
            Follows by \obv{filter_family_refinement}.
        \end{subproof}
        Hence $\ran{b}$ is a basis of
            $\neighbourhoods{\ringZero[R]}{R}$ on $\carrier[R]$ by
            \obv{filter_basis_of}.
    \end{subproof}

    Show that if $U\subseteq\carrier[R]$ and $T$ is finite and
        $T\subseteq U$ and
        $\additiveSubsetProduct{R}{T}{U}
            =\additiveSubsetProduct{R}{U}{U}$ and
        $\additiveSubsetProduct{R}{U}{U}\subseteq U$ and
        $\ran{b}$ is a basis of
            $\neighbourhoods{\ringZero[R]}{R}$ on $\carrier[R]$, then
        $\fAdicPresentation{R}{U}{T}$.
    \begin{subproof}
        Assume $U\subseteq\carrier[R]$ and $T$ is finite and
            $T\subseteq U$ and
            $\additiveSubsetProduct{R}{T}{U}
                =\additiveSubsetProduct{R}{U}{U}$ and
            $\additiveSubsetProduct{R}{U}{U}\subseteq U$ and
            $\ran{b}$ is a basis of
                $\neighbourhoods{\ringZero[R]}{R}$ on $\carrier[R]$.
        $b\in\funs{\naturals}{\pow{\carrier[R]}}$ by
            \auto{f_adic_power_family_type}.
        $\ran{b}\subseteq\neighbourhoods{\ringZero[R]}{R}$ and
            $\ran{b}$ refines $\neighbourhoods{\ringZero[R]}{R}$ as
                filter families on $\carrier[R]$ by
            \obv{filter_basis_of}.
        Show that for every $n\in\naturals$ we have
            $\fAdicPower{R}{U}{\suc{n}}
                \in\neighbourhoods{\ringZero[R]}{R}$.
        \begin{subproof}
            Fix $n\in\naturals$.
            $b(n)=\fAdicPower{R}{U}{\suc{n}}$ by
                \auto{f_adic_power_family_apply}.
            $b(n)\in\ran{b}$ by
                \auto{funs_is_function,funs,ran_iff,function_apply_iff}.
            Hence $b(n)\in\neighbourhoods{\ringZero[R]}{R}$ by
                \obv{subseteq}.
        \end{subproof}
        Show that for every $V\in\neighbourhoods{\ringZero[R]}{R}$ there
            exists $n\in\naturals$ such that
                $\fAdicPower{R}{U}{\suc{n}}\subseteq V$.
        \begin{subproof}
            Fix $V\in\neighbourhoods{\ringZero[R]}{R}$.
            Take $H\in\ran{b}$ such that $H\subseteq V$ by
                \obv{filter_family_refinement}.
            Take $n$ such that $(n,H)\in b$ by \obv{ran_iff}.
            Hence $n\in\dom{b}$ by \obv{dom_intro}.
            Hence $n\in\naturals$ by \auto{funs}.
            Hence $H=b(n)$ by \auto{funs_is_function,function_apply_iff}.
            Hence $H=\fAdicPower{R}{U}{\suc{n}}$ by
                \auto{f_adic_power_family_apply}.
        \end{subproof}
        Follows by \obv{f_adic_presentation}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{proposition}\label{f_adic_presentation_power_basis}
    Let $R$ be a commutative topological ring.
    Suppose $\fAdicPresentation{R}{U}{T}$.
    Then
        $\ran{\Apply{\graphOnOp}{
            \naturals,\pow{\carrier[R]},
            \fun{n:\Set}{\fAdicPower{R}{U}{\suc{n}}}}}$
            is a basis of $\neighbourhoods{\ringZero[R]}{R}$ on
                $\carrier[R]$.
\end{proposition}
\begin{proof}
    Follows by \auto{f_adic_presentation_iff}.
\end{proof}

\begin{theorem}\label{f_adic_presentation_tail_basis}
    Let $R$ be a commutative topological ring.
    Suppose $\fAdicPresentation{R}{U}{T}$.
    Let $b=\Apply{\graphOnOp}{
        \naturals,\pow{\carrier[R]},
        \fun{n:\Set}{\fAdicPower{R}{U}{\suc{n}}}}$.
    Then $\img{b}{\naturalTail{\suc{\emptyset}}}$ is a basis of
        $\neighbourhoods{\ringZero[R]}{R}$ on $\carrier[R]$.
\end{theorem}
\begin{proof}
    $R$ is a commutative topological ring and
        $U\subseteq\carrier[R]$ by \obv{f_adic_presentation_ambient}.
    Hence $R$ is a commutative ring by
        \obv{commutative_topological_ring_is_commutative_ring}.
    Hence $R$ is a ring by \obv{commutative_ring_is_ring}.
    $b\in\funs{\naturals}{\pow{\carrier[R]}}$ by
        \auto{f_adic_power_family_type}.
    $\additiveSubsetProduct{R}{U}{U}\subseteq U$ by
        \obv{f_adic_presentation_contraction}.
    Show that for all $K,L$ such that $K,L\in\naturals$ and
        $K\subseteq L$ we have $b(L)\subseteq b(K)$.
    \begin{subproof}
        Fix $K,L$.
        Assume $K,L\in\naturals$ and $K\subseteq L$.
        $K$ and $L$ are ordinals by \obv{natural_number_is_ordinal}.
        Hence $K\in L$ or $K=L$ by
            \obv{ordinal_subseteq_implies_elem_or_eq}.
        Hence $K\in\suc{L}$ by \auto{suc_iff}.
        $L\subseteq\suc{L}$ by \obv{subseteq_self_suc_intro}.
        Hence $K\subseteq\suc{L}$ by
            \obv{subseteq_transitive,subseteq_refl}.
        Hence $\suc{K}\subseteq\suc{L}$ by \obv{suc_subseteq_intro}.
        $\suc{K},\suc{L}\in\naturals$ and
            $\suc{K},\suc{L}\neq\emptyset$ by
            \obv{omega_successor,suc_neq_emptyset}.
        $b(K)=\fAdicPower{R}{U}{\suc{K}}$ and
            $b(L)=\fAdicPower{R}{U}{\suc{L}}$ by
            \auto{f_adic_power_family_apply}.
        Follows by \auto{f_adic_power_antitone}.
    \end{subproof}
    $\ran{b}$ is a basis of $\neighbourhoods{\ringZero[R]}{R}$ on
        $\carrier[R]$ by
        \auto{f_adic_presentation_power_basis}.
    $R$ is a topological space by
        \obv{topological_ring_is_topological_space}.
    $\ringZero[R]\in\carrier[R]$ by \obv{ring_zero_type}.
    $\suc{\emptyset}$ is a natural number by
        \obv{omega_empty,omega_successor}.
    Follows by
        \obv{antitone_natural_tail_range_is_neighbourhood_filter_basis}.
\end{proof}

\begin{theorem}\label{f_adic_presentation_tail_subgroups}
    Suppose $\fAdicPresentation{R}{U}{T}$.
    Let $b=\Apply{\graphOnOp}{
        \naturals,\pow{\carrier[R]},
        \fun{n:\Set}{\fAdicPower{R}{U}{\suc{n}}}}$.
    Then for every $H\in\img{b}{\naturalTail{\suc{\emptyset}}}$ we have
        $\subgroup{H}{\AdditiveGroup{R}}$.
\end{theorem}
\begin{proof}
    $R$ is a commutative topological ring and
        $U\subseteq\carrier[R]$ by \obv{f_adic_presentation_ambient}.
    Hence $R$ is a commutative ring by
        \obv{commutative_topological_ring_is_commutative_ring}.
    Hence $R$ is a ring by \obv{commutative_ring_is_ring}.
    $b\in\funs{\naturals}{\pow{\carrier[R]}}$ by
        \auto{f_adic_power_family_type}.
    Fix $H\in\img{b}{\naturalTail{\suc{\emptyset}}}$.
    Take $n\in\naturalTail{\suc{\emptyset}}$ such that $(n,H)\in b$ by
        \obv{img_iff}.
    Hence $n\in\naturals$ and $\suc{\emptyset}\subseteq n$ by
        \obv{natural_tail}.
    Hence $H=b(n)$ by \auto{funs_is_function,function_apply_iff}.
    Hence $H=\fAdicPower{R}{U}{\suc{n}}$ by
        \auto{f_adic_power_family_apply}.
    $n$ is a natural number.
    $\suc{\emptyset}$ is a natural number by
        \obv{omega_empty,omega_successor}.
    Hence $\suc{\emptyset}$ is an ordinal by
        \obv{natural_number_is_ordinal}.
    $n$ is an ordinal by \obv{natural_number_is_ordinal}.
    Hence $\suc{\emptyset}\in n$ or $\suc{\emptyset}=n$ by
        \obv{ordinal_subseteq_implies_elem_or_eq}.
    Hence $\suc{\emptyset}\in\suc{n}$ by \auto{suc_iff}.
    $\suc{\emptyset}\subseteq n$ and
        $n\subseteq\suc{n}$ by \obv{subseteq_self_suc_intro}.
    Hence $\suc{\emptyset}\subseteq\suc{n}$ by
        \obv{subseteq_transitive}.
    Hence $\suc{\suc{\emptyset}}\subseteq\suc{n}$ by
        \obv{suc_subseteq_intro}.
    $\suc{n}$ is a natural number by \obv{omega_successor}.
    $\subgroup{\fAdicPower{R}{U}{\suc{n}}}{\AdditiveGroup{R}}$ by
        \obv{f_adic_power_is_subgroup}.
\end{proof}

\begin{definition}\label{f_adic_ring}
    $R$ is an f-adic ring iff
        $R$ is a commutative topological ring and
        there exist $U,T$ such that $\fAdicPresentation{R}{U}{T}$.
\end{definition}

\begin{proposition}\label{f_adic_ring_is_commutative_topological_ring}
    Let $R$ be an f-adic ring.
    Then $R$ is a commutative topological ring.
\end{proposition}
\begin{proof}
    Follows by \obv{f_adic_ring}.
\end{proof}

\begin{proposition}\label{f_adic_ring_has_presentation}
    Let $R$ be an f-adic ring.
    Then there exist $U,T$ such that $\fAdicPresentation{R}{U}{T}$.
\end{proposition}
\begin{proof}
    Follows by \obv{f_adic_ring}.
\end{proof}

\begin{theorem}\label{f_adic_ring_is_nonarchimedean}
    Let $R$ be an f-adic ring.
    Then $R$ is a nonarchimedean topological ring.
\end{theorem}
\begin{proof}
    Hence $R$ is a topological ring by
        \obv{commutative_topological_ring_is_topological_ring}.
    Take $U,T$ such that $\fAdicPresentation{R}{U}{T}$ by
        \obv{f_adic_ring_has_presentation}.
    Let $b=\Apply{\graphOnOp}{
        \naturals,\pow{\carrier[R]},
        \fun{n:\Set}{\fAdicPower{R}{U}{\suc{n}}}}$.
    Let $B=\img{b}{\naturalTail{\suc{\emptyset}}}$.
    $\img{b}{\naturalTail{\suc{\emptyset}}}$ is a basis of
        $\neighbourhoods{\ringZero[R]}{R}$ on $\carrier[R]$ by
        \obv{f_adic_presentation_tail_basis}.
    Hence $B$ is a basis of $\neighbourhoods{\ringZero[R]}{R}$ on
        $\carrier[R]$.
    For every $H\in B$ we have
        $\subgroup{H}{\AdditiveGroup{R}}$ by
        \obv{f_adic_presentation_tail_subgroups}.
    Follows by
        \obv{topological_ring_with_subgroup_basis_is_nonarchimedean}.
\end{proof}

\subsection{Bounded presentation subsets}

\begin{theorem}\label{f_adic_presentation_subset_bounded}
    Suppose $\fAdicPresentation{R}{U}{T}$.
    Then $\Bounded{R}{U}$.
\end{theorem}
\begin{proof}
    $R$ is a commutative topological ring and
        $U\subseteq\carrier[R]$ by \obv{f_adic_presentation_ambient}.
    Hence $R$ is a topological ring by
        \obv{commutative_topological_ring_is_topological_ring}.
    $R$ is a commutative ring by
        \obv{commutative_topological_ring_is_commutative_ring}.
    Hence $R$ is a ring by \obv{commutative_ring_is_ring}.
    Let $M=\MultiplicativeMonoid{R}$.
    Let $b=\Apply{\graphOnOp}{
        \naturals,\pow{\carrier[R]},
        \fun{n:\Set}{\fAdicPower{R}{U}{\suc{n}}}}$.
    Let $B=\img{b}{\naturalTail{\suc{\emptyset}}}$.
    $M$ is a monoid by \obv{multiplicative_monoid_is_exact_monoid}.
    $\carrier[M]=\carrier[R]$ by \auto{multiplicative_monoid_fields}.
    Hence $U\subseteq\carrier[M]$.
    $b\in\funs{\naturals}{\pow{\carrier[R]}}$ by
        \auto{f_adic_power_family_type}.
    $\img{b}{\naturalTail{\suc{\emptyset}}}$ is a basis of
        $\neighbourhoods{\ringZero[R]}{R}$ on $\carrier[R]$ by
        \obv{f_adic_presentation_tail_basis}.
    Hence $B$ is a basis of $\neighbourhoods{\ringZero[R]}{R}$ on
        $\carrier[R]$.
    Hence $B\subseteq\neighbourhoods{\ringZero[R]}{R}$ and
        $B$ refines $\neighbourhoods{\ringZero[R]}{R}$ as filter families
            on $\carrier[R]$ by \obv{filter_basis_of}.
    $\additiveSubsetProduct{R}{U}{U}\subseteq U$ by
        \obv{f_adic_presentation_contraction}.

    Show that for every $V\in\neighbourhoods{\ringZero[R]}{R}$ there
        exists $Q\in\neighbourhoods{\ringZero[R]}{R}$ such that
            $\pointwiseProduct{M}{Q}{U}\subseteq V$.
    \begin{subproof}
        Fix $V\in\neighbourhoods{\ringZero[R]}{R}$.
        Take $H\in B$ such that $H\subseteq V$ by
            \obv{filter_family_refinement}.
        Hence $H\in\neighbourhoods{\ringZero[R]}{R}$ by \obv{subseteq}.
        Take $n\in\naturalTail{\suc{\emptyset}}$ such that $(n,H)\in b$ by
            \obv{img_iff}.
        Hence $n\in\naturals$ and $\suc{\emptyset}\subseteq n$ by
            \obv{natural_tail}.
        Hence $H=b(n)$ by \auto{funs_is_function,function_apply_iff}.
        Hence $H=\fAdicPower{R}{U}{\suc{n}}$ by
            \auto{f_adic_power_family_apply}.
        $\suc{n}$ is a natural number and $\suc{n}\neq\emptyset$ by
            \obv{omega_successor,suc_neq_emptyset}.
        Hence $\fAdicPower{R}{U}{\suc{n}}\subseteq\carrier[R]$ by
            \obv{f_adic_power_type}.
        $\pointwiseProduct{\MultiplicativeMonoid{R}}{
                \fAdicPower{R}{U}{\suc{n}}}{U}
            \subseteq\additiveSubsetProduct{R}{
                \fAdicPower{R}{U}{\suc{n}}}{U}$ by
            \obv{pointwise_product_subseteq_additive_subset_product}.
        Hence $\pointwiseProduct{M}{H}{U}
            \subseteq\additiveSubsetProduct{R}{
                \fAdicPower{R}{U}{\suc{n}}}{U}$.
        $\fAdicPower{R}{U}{\suc{\suc{n}}}
            =\additiveSubsetProduct{R}{
                \fAdicPower{R}{U}{\suc{n}}}{U}$ by
            \obv{f_adic_power_successor}.
        $\suc{n}\subseteq\suc{\suc{n}}$ by
            \obv{subseteq_self_suc_intro}.
        $\suc{\suc{n}}$ is a natural number and
            $\suc{\suc{n}}\neq\emptyset$ by
            \obv{omega_successor,suc_neq_emptyset}.
        Hence $\fAdicPower{R}{U}{\suc{\suc{n}}}
            \subseteq\fAdicPower{R}{U}{\suc{n}}$ by
            \obv{f_adic_power_antitone}.
        Hence $\pointwiseProduct{M}{H}{U}\subseteq H$ by
            \obv{subseteq_transitive}.
        Hence $\pointwiseProduct{M}{H}{U}\subseteq V$ by
            \obv{subseteq_transitive}.
    \end{subproof}
    Hence $U$ is left bounded in $R$ by \auto{left_bounded_in}.
    Follows by \obv{bounded_in_commutative_topological_ring}.
\end{proof}

\begin{theorem}\label{f_adic_presentation_subset_power_bounded}
    Suppose $\fAdicPresentation{R}{U}{T}$.
    Then $\PowerBounded{R}{U}$.
\end{theorem}
\begin{proof}
    $R$ is a commutative topological ring and
        $U\subseteq\carrier[R]$ by \obv{f_adic_presentation_ambient}.
    Hence $R$ is a commutative ring by
        \obv{commutative_topological_ring_is_commutative_ring}.
    Hence $R$ is a ring by \obv{commutative_ring_is_ring}.
    Let $M=\MultiplicativeMonoid{R}$.
    Let $E=\pointwiseSubsetPowers{M}{U}$.
    $M$ is a monoid by \obv{multiplicative_monoid_is_exact_monoid}.
    $\carrier[M]=\carrier[R]$ and $\neutral[M]=\ringOne[R]$ by
        \auto{multiplicative_monoid_fields}.
    Hence $U\subseteq\carrier[M]$.
    $\additiveSubsetProduct{R}{U}{U}\subseteq U$ by
        \obv{f_adic_presentation_contraction}.
    $\Bounded{R}{U}$ by \obv{f_adic_presentation_subset_bounded}.
    $\{\ringOne[R]\}$ is finite by \obv{singleton_is_finite}.
    $\ringOne[R]\in\carrier[R]$ by \obv{ring_one_type}.
    Hence $\{\ringOne[R]\}\subseteq\carrier[R]$ by
        \auto{subseteq,singleton_iff}.
    Hence $\Bounded{R}{\{\ringOne[R]\}}$ by
        \obv{finite_subset_bounded}.
    Hence $\Bounded{R}{\{\ringOne[R]\}\union U}$ by
        \obv{bounded_binary_union}.

    Show that every element of $E$ is an element of
        $\{\ringOne[R]\}\union U$.
    \begin{subproof}
        Fix $a\in E$.
        Take a natural number $n$ such that
            $a\in\pointwiseSubsetPower{M}{U}{n}$ by
            \obv{pointwise_subset_powers_iff}.
        \begin{byCase}
            \caseOf{$n=\emptyset$.}
                $\pointwiseSubsetPower{M}{U}{\emptyset}
                    =\{\neutral[M]\}$ by
                    \obv{pointwise_subset_power_zero}.
                Hence $a=\ringOne[R]$ by
                    \obv{singleton_iff}.
                Follows by \auto{union_iff,singleton_iff}.
            \caseOf{$n\neq\emptyset$.}
                $\pointwiseSubsetPower{\MultiplicativeMonoid{R}}{U}{n}
                    \subseteq\additiveSubsetPower{R}{U}{n}$ by
                    \obv{pointwise_subset_power_subseteq_additive_subset_power}.
                Hence $\pointwiseSubsetPower{M}{U}{n}
                    \subseteq\additiveSubsetPower{R}{U}{n}$.
                Hence $a\in\additiveSubsetPower{R}{U}{n}$ by
                    \obv{subseteq}.
                $\additiveSubsetPower{R}{U}{n}
                    =\fAdicPower{R}{U}{n}$ by \obv{f_adic_power}.
                Hence $a\in\fAdicPower{R}{U}{n}$.
                $\fAdicPower{R}{U}{n}\subseteq U$ by
                    \obv{f_adic_power_subset_of_contraction}.
                Hence $a\in U$ by \obv{subseteq}.
                Follows by \auto{union_iff}.
        \end{byCase}
    \end{subproof}
    Hence $E\subseteq\{\ringOne[R]\}\union U$ by \obv{subseteq}.
    Hence $\Bounded{R}{E}$ by \obv{bounded_downward}.
    Follows by \obv{power_bounded_set}.
\end{proof}

\begin{corollary}\label{f_adic_presentation_element_power_bounded}
    Suppose $\fAdicPresentation{R}{U}{T}$ and $x\in U$.
    Then $x$ is power bounded in $R$.
\end{corollary}
\begin{proof}
    $R$ is a commutative topological ring and
        $U\subseteq\carrier[R]$ by \obv{f_adic_presentation_ambient}.
    $\PowerBounded{R}{U}$ by
        \obv{f_adic_presentation_subset_power_bounded}.
    $\{x\}\subseteq U$ by \auto{subseteq,singleton_iff}.
    Hence $\PowerBounded{R}{\{x\}}$ by
        \obv{power_bounded_set_downward}.
    $x\in\carrier[R]$ by \obv{subseteq}.
    Follows by \obv{power_bounded_element}.
\end{proof}

\subsection{An open bounded subring}

\begin{theorem}\label{f_adic_presentation_generated_subring_open_bounded}
    Let $R$ be a commutative topological ring.
    Suppose $U\subseteq\carrier[R]$.
    Suppose $\fAdicPresentation{R}{U}{T}$.
    Then $\generatedSubring{R}{U}$ is a subring of $R$ and
        $\generatedSubring{R}{U}$ is open in $R$ and
        $\Bounded{R}{\generatedSubring{R}{U}}$.
\end{theorem}
\begin{proof}
    $R$ is a commutative topological ring and
        $U\subseteq\carrier[R]$ and $T$ is finite and $T\subseteq U$ by
        \obv{f_adic_presentation_ambient}.
    Hence $R$ is a topological ring by
        \obv{commutative_topological_ring_is_topological_ring}.
    $R$ is a commutative ring by
        \obv{commutative_topological_ring_is_commutative_ring}.
    Hence $R$ is a ring by \obv{commutative_ring_is_ring}.
    Let $G=\AdditiveGroup{R}$.
    Let $M=\MultiplicativeMonoid{R}$.
    Let $A=\{\ringOne[R]\}\union U$.
    Let $H=\generatedSubgroup{A}{G}$.
    Let $S=\generatedSubring{R}{U}$.
    $G$ is a group by \obv{additive_group_is_group}.
    $\carrier[G]=\carrier[R]$ by \auto{additive_group_fields}.
    $M$ is a monoid by \obv{multiplicative_monoid_is_exact_monoid}.
    $\carrier[M]=\carrier[R]$ and $\mul[M]=\ringMul[R]$ by
        \auto{multiplicative_monoid_fields}.
    $\ringOne[R]\in\carrier[R]$ by \obv{ring_one_type}.
    Hence $\{\ringOne[R]\}\subseteq\carrier[R]$ by
        \auto{subseteq,singleton_iff}.
    Hence $A\subseteq\carrier[R]$ by \auto{union_iff,subseteq}.
    Hence $A\subseteq\carrier[G]$.
    $\subgroup{H}{G}$ by
        \auto{generated_subgroup_is_subgroup}.
    $A\subseteq H$ by \obv{generated_subgroup_contains_generators}.
    $U\subseteq A$ by \auto{union_iff,subseteq}.
    Hence $U\subseteq H$ by \obv{subseteq_transitive}.
    $\ringOne[R]\in A$ by \auto{union_iff,singleton_iff}.
    Hence $\ringOne[R]\in H$ by \obv{subseteq}.
    $\additiveSubsetProduct{R}{U}{U}\subseteq U$ by
        \obv{f_adic_presentation_contraction}.

    Show that for all $x,y\in H$ we have $\ringMul[R](x,y)\in H$.
    \begin{subproof}
        Fix $x,y\in H$.
        $H\subseteq\carrier[G]$ by \obv{subgroup_subseteq}.
        Hence $y\in\carrier[G]$ by \obv{subseteq}.
        Hence $y\in\carrier[R]$ by \auto.
        Show that for every $a\in A$ we have $\ringMul[R](a,y)\in H$.
        \begin{subproof}
            Fix $a\in A$.
            Hence $a\in\carrier[R]$ by \obv{subseteq}.
            Show that for every $c\in A$ we have
                $\ringMul[R](a,c)\in H$.
            \begin{subproof}
                Fix $c\in A$.
                Hence $c\in\carrier[R]$ by \obv{subseteq}.
                $a=\ringOne[R]$ or $a\in U$ by
                    \auto{union_iff,singleton_iff}.
                $c=\ringOne[R]$ or $c\in U$ by
                    \auto{union_iff,singleton_iff}.
                \begin{byCase}
                    \caseOf{$a=\ringOne[R]$.}
                        $\ringMul[R](a,c)=c$ by
                            \auto{ring_mul_one_left}.
                        $c\in H$ by \obv{subseteq}.
                        Follows by \auto.
                    \caseOf{$a\neq\ringOne[R]$.}
                        Hence $a\in U$ by \auto.
                        \begin{byCase}
                            \caseOf{$c=\ringOne[R]$.}
                                $\ringMul[R](a,c)=a$ by
                                    \auto{ring_mul_one_right}.
                                $a\in H$ by \obv{subseteq}.
                                Follows by \auto.
                            \caseOf{$c\neq\ringOne[R]$.}
                                Hence $c\in U$ by \auto.
                                $\mul[M](a,c)
                                    \in\pointwiseProduct{M}{U}{U}$ by
                                    \auto{pointwise_product_iff}.
                                Hence $\ringMul[R](a,c)
                                    \in\pointwiseProduct{
                                        \MultiplicativeMonoid{R}}{U}{U}$ by
                                    \auto.
                                $\pointwiseProduct{
                                        \MultiplicativeMonoid{R}}{U}{U}
                                    \subseteq
                                        \additiveSubsetProduct{R}{U}{U}$ by
                                    \obv{pointwise_product_subseteq_additive_subset_product}.
                                Hence $\ringMul[R](a,c)
                                    \in\additiveSubsetProduct{R}{U}{U}$ by
                                    \obv{subseteq}.
                                Hence $\ringMul[R](a,c)\in U$ by
                                    \obv{subseteq}.
                                Follows by \obv{subseteq}.
                        \end{byCase}
                \end{byCase}
            \end{subproof}
            Hence for every $z\in H$ we have
                $\ringMul[R](a,z)\in H$ by
                \auto{generated_subgroup_closed_under_left_ring_multiplication}.
            Follows by \auto.
        \end{subproof}
        Hence for every $z\in H$ we have
            $\ringMul[R](z,y)\in H$ by
            \auto{generated_subgroup_closed_under_right_ring_multiplication}.
        Follows by \auto.
    \end{subproof}
    Hence $H$ is a subring of $R$ by \auto{subring}.
    Hence $S\subseteq H$ by \auto{generated_subring_least}.
    $S$ is a subring of $R$ by \auto{generated_subring_is_subring}.

    $R$ is an f-adic ring by \auto{f_adic_ring}.
    Hence $R$ is a nonarchimedean topological ring by
        \obv{f_adic_ring_is_nonarchimedean}.
    $\Bounded{R}{U}$ by \obv{f_adic_presentation_subset_bounded}.
    $\{\ringOne[R]\}$ is finite by \obv{singleton_is_finite}.
    Hence $\Bounded{R}{\{\ringOne[R]\}}$ by
        \obv{finite_subset_bounded}.
    Hence $\Bounded{R}{A}$ by \auto{bounded_binary_union}.
    Hence $\Bounded{R}{H}$ by
        \auto{nonarchimedean_generated_subgroup_bounded}.
    Hence $\Bounded{R}{S}$ by \auto{bounded_downward}.

    $\fAdicPower{R}{U}{\suc{\emptyset}}
        \in\neighbourhoods{\ringZero[R]}{R}$ by
        \obv{f_adic_presentation_power_neighbourhood,omega_empty}.
    $\fAdicPower{R}{U}{\suc{\emptyset}}=U$ by
        \obv{f_adic_power_one}.
    Hence $U\in\neighbourhoods{\ringZero[R]}{R}$ by \auto.
    $U\subseteq\generatedSubring{R}{U}$ by
        \obv{generated_subring_contains_generators}.
    Hence $U\subseteq S$ by \auto.
    $S\subseteq\carrier[R]$ by \obv{subring_subseteq_carrier}.
    $\neighbourhoods{\ringZero[R]}{R}$ is a filter on $\carrier[R]$ by
        \obv{topological_ring_zero_neighbourhoods_is_filter}.
    Hence $S\in\neighbourhoods{\ringZero[R]}{R}$ by \auto{filter}.
    $\subgroup{S}{\AdditiveGroup{R}}$ by
        \obv{subring_additive_subgroup}.
    Hence $\subgroup{S}{G}$ by \auto.
    Hence $S$ is open in $R$ by
        \auto{additive_subgroup_open_iff_neighbourhood_of_zero}.
    Follows by \auto.
\end{proof}

\begin{corollary}\label{f_adic_ring_has_open_bounded_subring}
    Let $R$ be a commutative topological ring.
    Suppose $R$ is an f-adic ring.
    Then there exists $S$ such that $S$ is a subring of $R$ and
        $S$ is open in $R$ and $\Bounded{R}{S}$.
\end{corollary}
\begin{proof}
    Take $U,T$ such that $\fAdicPresentation{R}{U}{T}$ by
        \obv{f_adic_ring_has_presentation}.
    $U\subseteq\carrier[R]$ by \obv{f_adic_presentation_ambient}.
    $\generatedSubring{R}{U}$ is a subring of $R$ and
        $\generatedSubring{R}{U}$ is open in $R$ and
        $\Bounded{R}{\generatedSubring{R}{U}}$ by
        \obv{f_adic_presentation_generated_subring_open_bounded}.
    Follows by \auto.
\end{proof}
