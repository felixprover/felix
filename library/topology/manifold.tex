\section{Topological manifolds}

\import{topology/real-coordinate-properties.tex}

Local Euclidean structure is a property of a topological space, not a choice
of atlas. A topological manifold is additionally Hausdorff and
second-countable. Dimensions are explicit parameters; no uniqueness of
dimension is asserted, and empty spaces are allowed.

\begin{definition}\label{locally_euclidean}
    Let $n$ be a natural number and $X$ be a topological space.
    $X$ is locally Euclidean of dimension $n$ iff for every $x\in\carrier[X]$
        there exists $U\in\opens[X]$ such that $x\in U$ and there exists
        $V\in\opens[\RealCoordinateSpace{n}]$ such that there exists $h$ such that
        $h$ is a homeomorphism from $\subspaceTopology{X}{U}$ to
            $\subspaceTopology{\RealCoordinateSpace{n}}{V}$.
\end{definition}

\begin{definition}\label{topological_manifold}
    Let $n$ be a natural number and $X$ be a topological space.
    $X$ is a topological manifold of dimension $n$ iff
        $X$ is locally Euclidean of dimension $n$ and
        $X$ is Hausdorff and $X$ is second-countable.
\end{definition}

\begin{theorem}\label{real_coordinate_space_is_locally_euclidean}
    Let $n$ be a natural number.
    Then $\RealCoordinateSpace{n}$ is locally Euclidean of dimension $n$.
\end{theorem}
\begin{proof}
    Let $X=\RealCoordinateSpace{n}$.
    $X$ is a topological space by \obv{real_coordinate_space_is_topological_space}.
    $\carrier[X]\in\opens[X]$ by \obv{carrier_open}.
    Let $S=\subspaceTopology{X}{\carrier[X]}$.
    $S$ is a topological space by \obv{subspace_topology_is_topological_space}.
    $\identity{\carrier[S]}$ is a homeomorphism from $S$ to $S$ by \obv{identity_homeomorphism}.
    Follows by \auto{locally_euclidean}.
\end{proof}

\begin{theorem}\label{empty_space_is_locally_euclidean}
    Let $n$ be a natural number and $X$ be a topological space.
    Suppose $\carrier[X]=\emptyset$.
    Then $X$ is locally Euclidean of dimension $n$.
\end{theorem}
\begin{proof}
    Follows by \auto{locally_euclidean}.
\end{proof}

\begin{theorem}\label{real_coordinate_space_is_topological_manifold}
    Let $n$ be a natural number.
    Then $\RealCoordinateSpace{n}$ is a topological manifold of dimension $n$.
\end{theorem}
\begin{proof}
    $n$ is finite by \obv{natural_number_is_finite}.
    $\RealCoordinateSpace{n}$ is locally Euclidean of dimension $n$ by
        \obv{real_coordinate_space_is_locally_euclidean}.
    $\RealCoordinateSpace{n}$ is Hausdorff and $\RealCoordinateSpace{n}$ is second-countable by
        \obv{real_coordinate_space_is_hausdorff,real_coordinate_space_is_second_countable}.
    Follows by \obv{topological_manifold,real_coordinate_space_is_topological_space}.
\end{proof}

\begin{theorem}\label{empty_space_is_topological_manifold}
    Let $n$ be a natural number and $X$ be a topological space.
    Suppose $\carrier[X]=\emptyset$.
    Then $X$ is a topological manifold of dimension $n$.
\end{theorem}
\begin{proof}
    $X$ is locally Euclidean of dimension $n$ by \obv{empty_space_is_locally_euclidean}.
    $X$ is Hausdorff by \auto{is_hausdorff}.
    $\carrier[X]$ is finite by \auto{emptyset_is_finite}.
    Hence $X$ is second-countable by \obv{finite_topological_space_is_second_countable}.
    Follows by \obv{topological_manifold}.
\end{proof}
