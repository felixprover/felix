\section{Topological countability axioms}

\import{topology/neighbourhood-basis.tex}
\import{topology/discrete.tex}
\import{cardinal/countability.tex}

\begin{definition}\label{is_dense_in}
    Let $X$ be a topological space.
    $D$ is dense in $X$ iff
        $D\subseteq\carrier[X]$ and
        $\closure{D}{X}=\carrier[X]$.
\end{definition}

\begin{definition}\label{is_separable}
    Let $X$ be a topological space.
    $X$ is separable iff there exists $D$ such that
        $D$ is countable and $D$ is dense in $X$.
\end{definition}

\begin{definition}\label{is_first_countable}
    Let $X$ be a topological space.
    $X$ is first-countable iff for every $x\in\carrier[X]$
        there exists a countable set $B$ such that
            $B$ is a local basis at $x$ in $X$.
\end{definition}

\begin{definition}\label{is_second_countable}
    Let $X$ be a topological space.
    $X$ is second-countable iff there exists a countable set $B$ such that
        $B$ is an open basis of $X$.
\end{definition}

\begin{theorem}\label{second_countable_implies_first_countable}
    Let $X$ be a topological space.
    Suppose $X$ is second-countable.
    Then $X$ is first-countable.
\end{theorem}
\begin{proof}
    Take $B$ such that $B$ is countable and $B$ is an open basis of $X$
        by \obv{is_second_countable}.
    Show that for every $x\in\carrier[X]$ there exists a countable set $C$
        such that $C$ is a local basis at $x$ in $X$.
    \begin{subproof}
        Fix $x\in\carrier[X]$.
        Let $C=\{V\in B\mid x\in V\}$.
        $C\subseteq B$ by \obv{subseteq}.
        Hence $C$ is countable by \obv{subseteq_countable_is_countable}.
        $C$ is a local basis at $x$ in $X$
            by \auto{open_basis_point_slice_is_local_basis}.
    \end{subproof}
    Follows by \obv{is_first_countable}.
\end{proof}

\begin{theorem}\label{finite_topological_space_is_second_countable}
    Let $X$ be a topological space.
    Suppose $\carrier[X]$ is finite.
    Then $X$ is second-countable.
\end{theorem}
\begin{proof}
    $\pow{\carrier[X]}$ is finite by \obv{finite_powerset}.
    $\opens[X]\subseteq\pow{\carrier[X]}$ by \obv{opens_family}.
    Hence $\opens[X]$ is finite by \obv{subseteq_finite_is_finite}.
    Hence $\opens[X]$ is countable by \obv{finite_is_countable}.
    $\opens[X]$ is an open basis of $X$ by \obv{opens_form_open_basis}.
    Follows by \obv{is_second_countable}.
\end{proof}
