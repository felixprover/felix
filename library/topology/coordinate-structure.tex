\section{Relative and equipped coordinate structures}

\import{topology/coordinate-transport.tex}

A relative structure stores only its maximal atlas on a specified space.
An equipped record stores that same atlas together with a carrier and a
topology. The shared datum declares the atlas field once. The equipped schema
supplies these fields; having coordinates under a pseudogroup is a separate
mathematical property, just as the model is a parameter rather than a field.

\begin{struct}\label{coordinate_datum}
    A coordinate datum $S$ is a record equipped with
    \begin{enumerate}
        \item a set $\atlas[S]$;
    \end{enumerate}.
\end{struct}

\begin{struct}\label{relative_coordinate_structure}
    Let $X$ be a topological space.
    Let $G$ be a coordinate pseudogroup.
    A relative coordinate structure $S$ on $X$ under $G$ is a coordinate datum
    such that
    \begin{enumerate}
        \item\label{relative_atlas_maximal}
            $\atlas[S]$ is a maximal coordinate atlas on $X$ under $G$.
    \end{enumerate}
\end{struct}

\begin{classification}\label{relative_atlas_is_maximal}
    Let $G$ be a coordinate pseudogroup and $X$ be a topological space.
    Let $S$ be a relative coordinate structure on $X$ under $G$.
    Then $\atlas[S]$ is a maximal coordinate atlas on $X$ under $G$.
\end{classification}
\begin{proof}
    Follows by \obv{relative_atlas_maximal}.
\end{proof}

\begin{struct}\label{equipped_coordinate_space}
    An equipped coordinate space $M$ is a topological space and a coordinate datum.
\end{struct}

\begin{definition}\label{equipped_coordinates}
    Let $G$ be a coordinate pseudogroup and $M$ be an equipped coordinate space.
    $M$ has coordinates under $G$ iff
        $\atlas[M]$ is a maximal coordinate atlas on $\TopologicalReduct{M}$ under $G$.
\end{definition}

\subsection{Generation, equipping, and extraction}

\begin{construction}\label{generated_coordinates}
    Let $G$ be a coordinate pseudogroup.
    Let $X$ be a topological space.
    Let $A$ be a coordinate atlas on $X$ under $G$.
    The relative coordinate structure $\GeneratedCoordinates{G}{X}{A}=
        \structure{\atlas=\MaxAtlas{G}{X}{A}}$ on $X$ under $G$.
\end{construction}
\begin{proof}
    $\MaxAtlas{G}{X}{A}$ is a maximal coordinate atlas on $X$ under $G$ by
        \obv{max_atlas_is_maximal}.
    Follows by \auto{relative_coordinate_structureaggregateprojection_atlas}.
\end{proof}

\begin{construction}\label{equip}
    Let $X$ be a topological space.
    Let $S$ be a coordinate datum.
    The equipped coordinate space $\Equip{X}{S}=
        equipped coordinate space based on X with \atlas=\atlas[S]$.
\end{construction}
\begin{proof}
    $\opens[X]$ is a topology on $\carrier[X]$ by
        \obv{topological_space_opens_form_topology}.
    Follows by \auto{topology_family,
        equipped_coordinate_spaceaggregateprojection_carrier,
        equipped_coordinate_spaceaggregateprojection_opens}.
\end{proof}

\begin{construction}\label{coordinate_structure}
    Let $M$ be an equipped coordinate space.
    The coordinate datum $\CoordinateStructure{M}=coordinate datum based on M$.
\end{construction}
\begin{proof}
    Follows.
\end{proof}

\begin{theorem}\label{equip_is_exact}
    Let $X$ be a topological space and $S$ be a coordinate datum.
    Then $\Equip{X}{S}$ is an exact equipped coordinate space.
\end{theorem}
\begin{proof}
    Follows by \auto{equip,equipshape}.
\end{proof}

\begin{theorem}\label{coordinate_structure_is_exact}
    Let $M$ be an equipped coordinate space.
    Then $\CoordinateStructure{M}$ is an exact coordinate datum.
\end{theorem}
\begin{proof}
    Follows by \auto{coordinate_structure,coordinate_structureshape}.
\end{proof}

\begin{theorem}\label{equip_same_fields}
    Let $X,Y$ be topological spaces and $S,T$ be coordinate data.
    Suppose $\carrier[X]=\carrier[Y]$ and $\opens[X]=\opens[Y]$ and $\atlas[S]=\atlas[T]$.
    Then $\Equip{X}{S}=\Equip{Y}{T}$.
\end{theorem}
\begin{proof}
    $\Equip{X}{S}$ is an exact equipped coordinate space by \obv{equip_is_exact}.
    $\Equip{Y}{T}$ is an exact equipped coordinate space by \obv{equip_is_exact}.
    $\carrier[\Equip{X}{S}]=\carrier[\Equip{Y}{T}]$ and
        $\opens[\Equip{X}{S}]=\opens[\Equip{Y}{T}]$ and
        $\atlas[\Equip{X}{S}]=\atlas[\Equip{Y}{T}]$ by
        \auto{equipprojection_carrier,equipprojection_opens,equipprojection_atlas}.
    Follows by structure extensionality.
\end{proof}

\begin{theorem}\label{equip_topological_reduct}
    Let $X$ be a topological space and $S$ be a coordinate datum.
    Then $\TopologicalReduct{\Equip{X}{S}}=\TopologicalReduct{X}$.
\end{theorem}
\begin{proof}
    $\carrier[\Equip{X}{S}]=\carrier[X]$ and $\opens[\Equip{X}{S}]=\opens[X]$ by
        \obv{equipprojection_carrier,equipprojection_opens}.
    Follows by \obv{topological_reduct}.
\end{proof}

\begin{classification}\label{equip_preserves_coordinates}
    Let $G$ be a coordinate pseudogroup.
    Let $X$ be a topological space.
    Let $S$ be a relative coordinate structure on $X$ under $G$.
    Then $\Equip{X}{S}$ has coordinates under $G$.
\end{classification}
\begin{proof}
    $\atlas[S]$ is a maximal coordinate atlas on $X$ under $G$ by
        \obv{relative_atlas_maximal}.
    $\TopologicalReduct{X}$ is a topological space by
        \obv{topological_reduct_is_topological_space}.
    $\carrier[\TopologicalReduct{X}]=\carrier[X]$ and
        $\opens[\TopologicalReduct{X}]=\opens[X]$ by
        \obv{topological_reduct_carrier,topological_reduct_opens}.
    $\atlas[S]$ is a maximal coordinate atlas on $\TopologicalReduct{X}$ under $G$
        by \auto{maximal_coordinate_atlas_same_fields}.
    $\Equip{X}{S}$ is an equipped coordinate space by \obv{equip}.
    $\atlas[\Equip{X}{S}]=\atlas[S]$ by \obv{equipprojection_atlas}.
    $\TopologicalReduct{\Equip{X}{S}}=\TopologicalReduct{X}$ by
        \obv{equip_topological_reduct}.
    Follows by \auto{equipped_coordinates}.
\end{proof}

\begin{classification}\label{coordinate_structure_relative}
    Let $G$ be a coordinate pseudogroup.
    Let $M$ be an equipped coordinate space.
    Suppose $M$ has coordinates under $G$.
    Then $\CoordinateStructure{M}$ is a relative coordinate structure
        on $\TopologicalReduct{M}$ under $G$.
\end{classification}
\begin{proof}
    $\CoordinateStructure{M}$ is a coordinate datum by \obv{coordinate_structure}.
    $\atlas[\CoordinateStructure{M}]=\atlas[M]$ by
        \obv{coordinate_structureprojection_atlas}.
    $\atlas[M]$ is a maximal coordinate atlas on $\TopologicalReduct{M}$ under $G$
        by \obv{equipped_coordinates}.
    Follows by \auto{relative_coordinate_structure}.
\end{proof}

\begin{theorem}\label{coordinate_structure_of_equip}
    Let $X$ be a topological space.
    Suppose $S$ is an exact coordinate datum.
    Then $\CoordinateStructure{\Equip{X}{S}}=S$.
\end{theorem}
\begin{proof}
    $\Equip{X}{S}$ is an equipped coordinate space by \obv{equip}.
    $\CoordinateStructure{\Equip{X}{S}}$ is an exact coordinate datum by
        \obv{coordinate_structure_is_exact}.
    $\atlas[\CoordinateStructure{\Equip{X}{S}}]=\atlas[\Equip{X}{S}]$ by
        \obv{coordinate_structureprojection_atlas}.
    $\atlas[\Equip{X}{S}]=\atlas[S]$ by \obv{equipprojection_atlas}.
    $\atlas[\CoordinateStructure{\Equip{X}{S}}]=\atlas[S]$.
    Follows by structure extensionality.
\end{proof}

\begin{theorem}\label{equip_coordinate_structure}
    Suppose $M$ is an exact equipped coordinate space.
    Then $\Equip{\TopologicalReduct{M}}{\CoordinateStructure{M}}=M$.
\end{theorem}
\begin{proof}
    $\TopologicalReduct{M}$ is a topological space by
        \obv{topological_reduct_is_topological_space}.
    $\CoordinateStructure{M}$ is a coordinate datum by \obv{coordinate_structure}.
    Let $N=\Equip{\TopologicalReduct{M}}{\CoordinateStructure{M}}$.
    $N$ is an exact equipped coordinate space by \auto{equip_is_exact}.
    $\carrier[\TopologicalReduct{M}]=\carrier[M]$ and
        $\opens[\TopologicalReduct{M}]=\opens[M]$ by
        \obv{topological_reduct_carrier,topological_reduct_opens}.
    $\carrier[N]=\carrier[M]$ and $\opens[N]=\opens[M]$ by
        \auto{equipprojection_carrier,equipprojection_opens}.
    $\atlas[\CoordinateStructure{M}]=\atlas[M]$ by
        \obv{coordinate_structureprojection_atlas}.
    $\atlas[N]=\atlas[M]$ by \auto{equipprojection_atlas}.
    Follows by structure extensionality.
\end{proof}

\subsection{Equality from presentations}

\begin{theorem}\label{generated_coordinates_is_exact}
    Let $G$ be a coordinate pseudogroup and $X$ be a topological space.
    Suppose $A$ is a coordinate atlas on $X$ under $G$.
    Then $\GeneratedCoordinates{G}{X}{A}$ is an exact relative coordinate structure
        on $X$ under $G$.
\end{theorem}
\begin{proof}
    Follows by \auto{generated_coordinates,generated_coordinatesshape}.
\end{proof}

\begin{theorem}\label{generated_coordinates_equal}
    Let $G$ be a coordinate pseudogroup and $X$ be a topological space.
    Suppose $A$ is a coordinate atlas on $X$ under $G$ and
        $B$ is a coordinate atlas on $X$ under $G$.
    Suppose for every $a\in A$ we have for every $b\in B$ we have
        $a$ is compatible with $b$ on $X$ under $G$.
    Then $\GeneratedCoordinates{G}{X}{A}=\GeneratedCoordinates{G}{X}{B}$.
\end{theorem}
\begin{proof}
    $\GeneratedCoordinates{G}{X}{A}$ is an exact relative coordinate structure
        on $X$ under $G$ by \obv{generated_coordinates_is_exact}.
    $\GeneratedCoordinates{G}{X}{B}$ is an exact relative coordinate structure
        on $X$ under $G$ by \obv{generated_coordinates_is_exact}.
    $\MaxAtlas{G}{X}{A}=\MaxAtlas{G}{X}{B}$ by \auto{max_atlas_equality_iff}.
    $\atlas[\GeneratedCoordinates{G}{X}{A}]=\atlas[\GeneratedCoordinates{G}{X}{B}]$
        by \auto{generated_coordinatesprojection_atlas}.
    Follows by structure extensionality.
\end{proof}

\begin{theorem}\label{equipped_presentations_equal}
    Let $G$ be a coordinate pseudogroup and $X$ be a topological space.
    Suppose $A$ is a coordinate atlas on $X$ under $G$ and
        $B$ is a coordinate atlas on $X$ under $G$.
    Suppose for every $a\in A$ we have for every $b\in B$ we have
        $a$ is compatible with $b$ on $X$ under $G$.
    Then $\Equip{X}{\GeneratedCoordinates{G}{X}{A}}
        =\Equip{X}{\GeneratedCoordinates{G}{X}{B}}$.
\end{theorem}
\begin{proof}
    $\GeneratedCoordinates{G}{X}{A}=\GeneratedCoordinates{G}{X}{B}$ by
        \obv{generated_coordinates_equal}.
\end{proof}

\begin{theorem}\label{generated_coordinates_chart_iff}
    Let $G$ be a coordinate pseudogroup and $X$ be a topological space.
    Suppose $A$ is a coordinate atlas on $X$ under $G$.
    Then $c\in\atlas[\GeneratedCoordinates{G}{X}{A}]$ iff
        $c$ is an open partial homeomorphism from $X$ to $\model[G]$ and
        for every $a\in A$ we have $c$ is compatible with $a$ on $X$ under $G$.
\end{theorem}
\begin{proof}
    $\atlas[\GeneratedCoordinates{G}{X}{A}]=\MaxAtlas{G}{X}{A}$ by
        \obv{generated_coordinatesprojection_atlas}.
    Follows by \auto{max_atlas_member_iff}.
\end{proof}
