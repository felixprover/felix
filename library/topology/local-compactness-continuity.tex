\section{Local compactness under open continuous maps}

\import{topology/compactness-continuity.tex}
\import{topology/open-map.tex}

An open continuous surjection carries a compact neighbourhood to a compact
neighbourhood. In particular, local compactness is preserved by homeomorphisms.

\begin{theorem}\label{open_continuous_surjection_preserves_local_compactness}
    Let $X,Y$ be topological spaces.
    Suppose $X$ is locally compact.
    Suppose $f$ is continuous from $X$ to $Y$ and $f$ is an open map from $X$ to $Y$.
    Suppose $f\in\surj{\carrier[X]}{\carrier[Y]}$.
    Then $Y$ is locally compact.
\end{theorem}
\begin{proof}
    $f$ is a function and $\dom{f}=\carrier[X]$ by
        \auto{continuous_map,funs,funs_is_function}.
    Show that for every $y\in\carrier[Y]$ there exist $V,L$ such that
        $V\in\opens[Y]$ and $y\in V\subseteq L$ and $L$ is compact in $Y$.
    \begin{subproof}
        Fix $y\in\carrier[Y]$.
        Take $x\in\carrier[X]$ such that $f(x)=y$ by \obv{surj}.
        Take $U,K$ such that $U\in\opens[X]$ and $x\in U\subseteq K$ and
            $K$ is compact in $X$ by \obv{is_locally_compact}.
        $\img{f}{U}\in\opens[Y]$ by \obv{open_map_image_open}.
        $y\in\img{f}{U}$ by \auto{img_of_function_intro,inter_intro}.
        $\img{f}{K}$ is compact in $Y$ by \obv{continuous_image_of_compact_subset}.
        Follows by \auto{img_subseteq}.
    \end{subproof}
    Follows by \obv{is_locally_compact}.
\end{proof}

\begin{corollary}\label{homeomorphism_preserves_local_compactness}
    Let $X,Y$ be topological spaces.
    Suppose $X$ is locally compact and $f$ is a homeomorphism from $X$ to $Y$.
    Then $Y$ is locally compact.
\end{corollary}
\begin{proof}
    $f$ is continuous from $X$ to $Y$ by \obv{homeomorphism}.
    $f$ is an open map from $X$ to $Y$ by \obv{homeomorphism_is_open_map}.
    $f\in\surj{\carrier[X]}{\carrier[Y]}$ by \auto{homeomorphism,bijections}.
    Follows by \obv{open_continuous_surjection_preserves_local_compactness}.
\end{proof}
