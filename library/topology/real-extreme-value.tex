We first isolate the order-topological step behind the extreme-value theorem.
A nonempty closed set contains its supremum whenever that supremum exists.
The proof uses only the local metric presentation of the real topology.

\begin{theorem}\label{closed_real_set_contains_supremum}
    Let $X$ be a topological space.
    Suppose $X=\RealTopology$.
    Let $P$ be a linearly ordered set.
    Suppose $P=\RealField$.
    Let $q\in\carrier[P]$.
    Suppose $A$ is closed in $X$ and $q$ is a supremum of $A$ in $P$.
    Then $q\in A$.
\end{theorem}
\begin{proof}
    $\carrier[P]=\reals$ by
        \obv{surreal_real_field_carrier}.
    For all $x,y\in\reals$ we have
        $x\mathrel{\leq[P]}y$ iff $\surrealLe{x}{y}$ by
        \obv{surreal_real_field_order}.
    $q$ is a least upper bound of $A$ in $\carrier[P]$
        with respect to $\leq[P]$ by \obv{supremum_element}.
    $\leq[P]$ is an order on $\carrier[P]$ by
        \obv{ordered_set_order_on_carrier}.
    $\leq[P]$ is a binary relation on $\carrier[P]$ by
        \obv{quasiorder_type}.
    Hence $q\in\reals$.
    Thus $q\in\carrier[X]$ by \obv{real_topology_carrier}.

    Suppose not.
    Hence $q\notin A$.
    Let $U=\carrier[X]\setminus A$.
    $A\subseteq\carrier[X]$ by \obv{closed_in_subseteq_carrier}.
    $\carrier[X]\setminus A$ is open in $X$ by
        \obv{complement_of_closed_is_open}.
    Hence $U$ is open in $X$.
    Hence $U\in\opens[X]$ by \obv{is_open}.
    $q\in U$ by \obv{setminus}.
    Hence $U\in\opens[\RealTopology]$.
    $\opens[\RealTopology]=\metricOpens{\RealMetric}$ by
        \auto{real_topology,metric_topology_opens}.
    Hence $U\in\metricOpens{\RealMetric}$.
    Thus $\metricOpen{\RealMetric}{U}$ by \obv{metric_opens_iff}.
    Take $r\in\reals$ such that $\surrealLt{\surrealZero}{r}$ and
        $\metricBall{\RealMetric}{q}{r}\subseteq U$ by
        \obv{metric_open}.
    Hence $\metricBall{\RealMetric}{q}{r}
        \subseteq\carrier[X]\setminus A$.

    Let $l=\surrealAdd{q}{\surrealNeg{r}}$.
    Let $u=\surrealAdd{q}{r}$.
    $\surrealNeg{r}\in\reals$ by
        \obv{surreal_real_negation_closed}.
    Hence $l,u\in\reals$ by \auto{surreal_real_addition_closed}.
    Hence $l,u\in\carrier[\RealField]$ by
        \obv{surreal_real_field_carrier}.
    Hence $l,u\in\carrier[P]$.
    $\metricBall{\RealMetric}{q}{r}
        =\orderOpenInterval{\RealField}
            {\surrealAdd{q}{\surrealNeg{r}}}{\surrealAdd{q}{r}}$ by
        \obv{real_metric_ball_eq_order_open_interval}.
    Hence $\metricBall{\RealMetric}{q}{r}
        =\orderOpenInterval{\RealField}{l}{u}$.
    $\surreal{q}$ and $\surreal{r}$ by
        \obv{surreal_real_is_surreal}.
    $\surrealLt{q}{u}$ by
        \auto{surreal_add_lt_right,surreal_add_zero_right,empty_surreal}.
    $\surrealLt{l}{q}$ by
        \auto{surreal_negation_lt_reverses,surreal_negation_empty,
            surreal_add_lt_right,surreal_add_zero_right,empty_surreal,
            surreal_negation_is_surreal}.

    Show that for every $a\in A$ we have
        $a\mathrel{\leq[P]}l$.
    \begin{subproof}
        Fix $a\in A$.
        $a\mathrel{\leq[P]}q$ by
            \auto{least_upper_bound_iff_order_on}.
        Hence $(a,q)\in\carrier[P]\times\carrier[P]$ by
            \obv{subseteq}.
        Hence $a\in\carrier[P]$ by \obv{times_tuple_elim}.
        Hence $a\in\reals$.
        $a\mathrel{\leq[P]}q$ iff $\surrealLe{a}{q}$.
        Hence $\surrealLe{a}{q}$.
        Show that $a\notin\metricBall{\RealMetric}{q}{r}$.
        \begin{subproof}
            Suppose $a\in\metricBall{\RealMetric}{q}{r}$.
            Hence $a\in\carrier[X]\setminus A$ by \obv{subseteq}.
            Hence $a\notin A$ by \obv{setminus}.
            Contradiction.
        \end{subproof}
        Show that not $\surrealLt{l}{a}$.
        \begin{subproof}
            Suppose $\surrealLt{l}{a}$.
            $\surrealLt{a}{u}$ by
                \obv{surreal_le_lt_transitive,
                    surreal_real_is_surreal}.
            $a\in\orderOpenInterval{\RealField}{l}{u}$ by
                \auto{order_open_interval_iff,
                    real_field_order_strict_iff}.
            Hence $a\in\metricBall{\RealMetric}{q}{r}$.
            Contradiction.
        \end{subproof}
        Hence $\surrealLe{a}{l}$ by
            \auto{surreal_lt_trichotomy,surreal_le,
                surreal_real_is_surreal}.
        $a\mathrel{\leq[P]}l$ iff $\surrealLe{a}{l}$.
    \end{subproof}
    Hence $q\mathrel{\leq[P]}l$ by
        \auto{least_upper_bound_iff_order_on}.
    $q\mathrel{\leq[P]}l$ iff $\surrealLe{q}{l}$.
    Hence $\surrealLe{q}{l}$.
    Contradiction by
        \obv{surreal_le,surreal_lt_asymmetric,
            surreal_real_is_surreal}.
\end{proof}

\begin{theorem}\label{closed_real_set_contains_infimum}
    Let $X$ be a topological space.
    Suppose $X=\RealTopology$.
    Let $P$ be a linearly ordered set.
    Suppose $P=\RealField$.
    Let $q\in\carrier[P]$.
    Suppose $A$ is closed in $X$ and $q$ is an infimum of $A$ in $P$.
    Then $q\in A$.
\end{theorem}
\begin{proof}
    $\carrier[P]=\reals$ by
        \obv{surreal_real_field_carrier}.
    For all $x,y\in\reals$ we have
        $x\mathrel{\leq[P]}y$ iff $\surrealLe{x}{y}$ by
        \obv{surreal_real_field_order}.
    $q$ is a greatest lower bound of $A$ in $\carrier[P]$
        with respect to $\leq[P]$ by \obv{infimum_element}.
    $\leq[P]$ is an order on $\carrier[P]$ by
        \obv{ordered_set_order_on_carrier}.
    $\leq[P]$ is a binary relation on $\carrier[P]$ by
        \obv{quasiorder_type}.
    Hence $q\in\reals$.
    Thus $q\in\carrier[X]$ by \obv{real_topology_carrier}.

    Suppose not.
    Hence $q\notin A$.
    Let $U=\carrier[X]\setminus A$.
    $A\subseteq\carrier[X]$ by \obv{closed_in_subseteq_carrier}.
    $\carrier[X]\setminus A$ is open in $X$ by
        \obv{complement_of_closed_is_open}.
    Hence $U$ is open in $X$.
    Hence $U\in\opens[X]$ by \obv{is_open}.
    $q\in U$ by \obv{setminus}.
    Hence $U\in\opens[\RealTopology]$.
    $\opens[\RealTopology]=\metricOpens{\RealMetric}$ by
        \auto{real_topology,metric_topology_opens}.
    Hence $U\in\metricOpens{\RealMetric}$.
    Thus $\metricOpen{\RealMetric}{U}$ by \obv{metric_opens_iff}.
    Take $r\in\reals$ such that $\surrealLt{\surrealZero}{r}$ and
        $\metricBall{\RealMetric}{q}{r}\subseteq U$ by
        \obv{metric_open}.
    Hence $\metricBall{\RealMetric}{q}{r}
        \subseteq\carrier[X]\setminus A$.

    Let $l=\surrealAdd{q}{\surrealNeg{r}}$.
    Let $u=\surrealAdd{q}{r}$.
    $\surrealNeg{r}\in\reals$ by
        \obv{surreal_real_negation_closed}.
    Hence $l,u\in\reals$ by \auto{surreal_real_addition_closed}.
    Hence $l,u\in\carrier[\RealField]$ by
        \obv{surreal_real_field_carrier}.
    Hence $l,u\in\carrier[P]$.
    $\metricBall{\RealMetric}{q}{r}
        =\orderOpenInterval{\RealField}
            {\surrealAdd{q}{\surrealNeg{r}}}{\surrealAdd{q}{r}}$ by
        \obv{real_metric_ball_eq_order_open_interval}.
    Hence $\metricBall{\RealMetric}{q}{r}
        =\orderOpenInterval{\RealField}{l}{u}$.
    $\surreal{q}$ and $\surreal{r}$ by
        \obv{surreal_real_is_surreal}.
    $\surrealLt{q}{u}$ by
        \auto{surreal_add_lt_right,surreal_add_zero_right,empty_surreal}.
    $\surrealLt{l}{q}$ by
        \auto{surreal_negation_lt_reverses,surreal_negation_empty,
            surreal_add_lt_right,surreal_add_zero_right,empty_surreal,
            surreal_negation_is_surreal}.

    Show that for every $a\in A$ we have
        $u\mathrel{\leq[P]}a$.
    \begin{subproof}
        Fix $a\in A$.
        $q\mathrel{\leq[P]}a$ by
            \auto{greatest_lower_bound_iff_order_on}.
        Hence $(q,a)\in\carrier[P]\times\carrier[P]$ by
            \obv{subseteq}.
        Hence $a\in\carrier[P]$ by \obv{times_tuple_elim}.
        Hence $a\in\reals$.
        $q\mathrel{\leq[P]}a$ iff $\surrealLe{q}{a}$.
        Hence $\surrealLe{q}{a}$.
        Show that $a\notin\metricBall{\RealMetric}{q}{r}$.
        \begin{subproof}
            Suppose $a\in\metricBall{\RealMetric}{q}{r}$.
            Hence $a\in\carrier[X]\setminus A$ by \obv{subseteq}.
            Hence $a\notin A$ by \obv{setminus}.
            Contradiction.
        \end{subproof}
        Show that not $\surrealLt{a}{u}$.
        \begin{subproof}
            Suppose $\surrealLt{a}{u}$.
            $\surrealLt{l}{a}$ by
                \obv{surreal_lt_le_transitive,
                    surreal_real_is_surreal}.
            $a\in\orderOpenInterval{\RealField}{l}{u}$ by
                \auto{order_open_interval_iff,
                    real_field_order_strict_iff}.
            Hence $a\in\metricBall{\RealMetric}{q}{r}$.
            Contradiction.
        \end{subproof}
        Hence $\surrealLe{u}{a}$ by
            \auto{surreal_lt_trichotomy,surreal_le,
                surreal_real_is_surreal}.
        $u\mathrel{\leq[P]}a$ iff $\surrealLe{u}{a}$.
    \end{subproof}
    Hence $u\mathrel{\leq[P]}q$ by
        \auto{greatest_lower_bound_iff_order_on}.
    $u\mathrel{\leq[P]}q$ iff $\surrealLe{u}{q}$.
    Hence $\surrealLe{u}{q}$.
    Contradiction by
        \auto{surreal_le,surreal_lt_asymmetric,
            surreal_real_is_surreal}.
\end{proof}

\begin{theorem}\label{inhabited_compact_real_set_has_extrema}
    Let $X$ be a topological space.
    Suppose $X=\RealTopology$.
    Let $P$ be a conditionally complete linearly ordered set.
    Suppose $P=\RealField$.
    Suppose $K$ is compact in $X$ and $K$ is inhabited.
    Then there exist $m,M\in K$ such that for every $x\in K$ we have
        $\surrealLe{m}{x}$ and $\surrealLe{x}{M}$.
\end{theorem}
\begin{proof}
    $K$ is compact in $X$ iff
        $K$ is closed in $X$ and $K$ is real bounded by
        \obv{real_heine_borel}.
    Hence $K$ is closed in $X$ and $K$ is real bounded.
    $K\subseteq\reals$ by \obv{real_bounded}.
    $\carrier[P]=\reals$ by
        \obv{surreal_real_field_carrier}.
    For all $x,y\in\reals$ we have
        $x\mathrel{\leq[P]}y$ iff $\surrealLe{x}{y}$ by
        \obv{surreal_real_field_order}.
    Take $B\in\reals$ such that $\surrealLe{\surrealZero}{B}$ and
        for every $x\in K$ we have $\surrealLe{\abs{x}}{B}$ by
        \obv{real_bounded}.
    Hence $B\in\carrier[P]$.

    Show that $K$ is bounded above in $P$.
    \begin{subproof}
        $K\subseteq\carrier[P]$.
        Show that for every $x\in K$ we have $x\mathrel{\leq[P]}B$.
        \begin{subproof}
            Fix $x\in K$.
            Hence $x\in\reals$ by \obv{subseteq}.
            Hence $\surreal{x}$ and $\surreal{B}$ by
                \obv{surreal_real_is_surreal}.
            $\surrealLe{\abs{x}}{B}$.
            Hence $\surrealLe{x}{B}$ by
                \auto{surreal_abs_le_iff,surreal_real_is_surreal}.
            $x\mathrel{\leq[P]}B$ iff $\surrealLe{x}{B}$.
        \end{subproof}
        Follows by \obv{bounded_above_in_ordered_set}.
    \end{subproof}
    $\hasSupremum{P}{K}$ by
        \obv{conditionally_complete_linear_order}.
    $\supremum{P}{K}\in\carrier[P]$ by \obv{supremum_type}.
    $\supremum{P}{K}$ is a supremum of $K$ in $P$ by
        \obv{supremum_spec}.
    Hence $\supremum{P}{K}\in K$ by
        \obv{closed_real_set_contains_supremum}.
    Let $M=\supremum{P}{K}$.
    Hence $M\in\carrier[P]$ and $M\in K$.
    Show that for every $x\in K$ we have $\surrealLe{x}{M}$.
    \begin{subproof}
        Fix $x\in K$.
        $x\mathrel{\leq[P]}\supremum{P}{K}$ by
            \obv{supremum_upper}.
        Hence $x\mathrel{\leq[P]}M$.
        $x\in\reals$ and $M\in\reals$ by \obv{subseteq}.
        $x\mathrel{\leq[P]}M$ iff $\surrealLe{x}{M}$.
    \end{subproof}

    $\surrealNeg{B}\in\reals$ by
        \obv{surreal_real_negation_closed}.
    Hence $\surrealNeg{B}\in\carrier[P]$.
    Show that $K$ is bounded below in $P$.
    \begin{subproof}
        $K\subseteq\carrier[P]$.
        Show that for every $x\in K$ we have
            $\surrealNeg{B}\mathrel{\leq[P]}x$.
        \begin{subproof}
            Fix $x\in K$.
            Hence $x\in\reals$ by \obv{subseteq}.
            Hence $\surreal{x}$ and $\surreal{B}$ by
                \obv{surreal_real_is_surreal}.
            $\surrealLe{\abs{x}}{B}$.
            Hence $\surrealLe{\surrealNeg{x}}{B}$ by
                \auto{surreal_abs_le_iff,surreal_real_is_surreal}.
            Hence $\surrealLe{\surrealNeg{B}}
                {\surrealNeg{\surrealNeg{x}}}$ by
                \obv{surreal_negation_le_reverses,
                    surreal_negation_is_surreal}.
            $\surrealNeg{\surrealNeg{x}}=x$ by
                \obv{surreal_negation_involutive}.
            Hence $\surrealLe{\surrealNeg{B}}{x}$.
            $\surrealNeg{B}\mathrel{\leq[P]}x$ iff
                $\surrealLe{\surrealNeg{B}}{x}$.
        \end{subproof}
        Follows by \obv{bounded_below_in_ordered_set}.
    \end{subproof}
    $\hasInfimum{P}{K}$ by \obv{conditionally_complete_has_infimum}.
    $\infimum{P}{K}\in\carrier[P]$ by \obv{infimum_type}.
    $\infimum{P}{K}$ is an infimum of $K$ in $P$ by
        \obv{infimum_spec}.
    Hence $\infimum{P}{K}\in K$ by
        \obv{closed_real_set_contains_infimum}.
    Let $m=\infimum{P}{K}$.
    Hence $m\in\carrier[P]$ and $m\in K$.
    Show that for every $x\in K$ we have $\surrealLe{m}{x}$.
    \begin{subproof}
        Fix $x\in K$.
        $\infimum{P}{K}\mathrel{\leq[P]}x$ by
            \obv{infimum_lower}.
        Hence $m\mathrel{\leq[P]}x$.
        $m\in\reals$ and $x\in\reals$ by \obv{subseteq}.
        $m\mathrel{\leq[P]}x$ iff $\surrealLe{m}{x}$.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{theorem}[Extreme-value theorem]\label{continuous_real_map_attains_extrema}
    Let $X,Y$ be topological spaces.
    Suppose $Y=\RealTopology$.
    Suppose $X$ is compact and $\carrier[X]$ is inhabited.
    Suppose $f$ is continuous from $X$ to $Y$.
    Then there exist $p,q\in\carrier[X]$ such that for every
        $x\in\carrier[X]$ we have
        $\surrealLe{f(p)}{f(x)}$ and $\surrealLe{f(x)}{f(q)}$.
\end{theorem}
\begin{proof}
    $f\in\funs{\carrier[X]}{\carrier[Y]}$ by
        \obv{continuous_map}.
    Hence $f$ is a function and $\dom{f}=\carrier[X]$ by
        \obv{funs_is_function,funs}.
    $\ran{f}$ is compact in $Y$ by
        \obv{continuous_image_is_compact}.
    Take $z$ such that $z\in\carrier[X]$.
    $f(z)\in\ran{f}$ by \auto{fun_ran_iff}.
    Hence $\ran{f}$ is inhabited.
    $\RealField$ is a conditionally complete linearly ordered set by
        \obv{surreal_real_conditionally_complete_linear_order}.
    Take $m,M\in\ran{f}$ such that for every $y\in\ran{f}$ we have
        $\surrealLe{m}{y}$ and $\surrealLe{y}{M}$ by
        \obv{inhabited_compact_real_set_has_extrema}.
    Take $p\in\dom{f}$ such that $f(p)=m$ by
        \obv{fun_ran_iff}.
    Take $q\in\dom{f}$ such that $f(q)=M$ by
        \obv{fun_ran_iff}.
    Hence $p,q\in\carrier[X]$.
    Show that for every $x\in\carrier[X]$ we have
        $\surrealLe{f(p)}{f(x)}$ and $\surrealLe{f(x)}{f(q)}$.
    \begin{subproof}
        Fix $x\in\carrier[X]$.
        $f(x)\in\ran{f}$ by \auto{fun_ran_iff}.
        Hence $\surrealLe{m}{f(x)}$ and $\surrealLe{f(x)}{M}$.
    \end{subproof}
    Follows by \auto.
\end{proof}
