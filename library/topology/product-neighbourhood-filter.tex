\section{Neighbourhood filters of binary products}

\import{topology/product-basis.tex}
\import{topology/neighbourhood-filter.tex}

The rectangle basis identifies the neighbourhood filter at a pair with the
product of the two coordinate neighbourhood filters.

\begin{theorem}\label{neighbourhood_filter_of_product}
    Let $X,Y$ be topological spaces.
    Let $x\in\carrier[X]$ and $y\in\carrier[Y]$.
    Then
        $\neighbourhoods{(x,y)}{\BinaryProductTopology{X}{Y}}
        =\productFilter{\carrier[X]}{\carrier[Y]}{
            \neighbourhoods{x}{X}}{\neighbourhoods{y}{Y}}$.
\end{theorem}
\begin{proof}
    Let $P=\BinaryProductTopology{X}{Y}$.
    $\carrier[P]=\carrier[X]\times\carrier[Y]$ by
        \auto{binary_product_topology_carrier}.
    Hence $(x,y)\in\carrier[P]$ by \obv{times_tuple_intro}.
    $\neighbourhoods{x}{X}$ is a filter on $\carrier[X]$ by
        \obv{neighbourhoods_is_filter}.
    $\neighbourhoods{y}{Y}$ is a filter on $\carrier[Y]$ by
        \obv{neighbourhoods_is_filter}.
    $\productRectangles{X}{Y}$ is an open basis of $P$ by
        \obv{product_rectangles_form_open_basis}.

    Show that every element of $\neighbourhoods{(x,y)}{P}$ is an
        element of
        $\productFilter{\carrier[X]}{\carrier[Y]}{
            \neighbourhoods{x}{X}}{\neighbourhoods{y}{Y}}$.
    \begin{subproof}
        Fix $N\in\neighbourhoods{(x,y)}{P}$.
        Hence $N\in\pow{\carrier[P]}$ and there exists $O\in\opens[P]$
            such that $(x,y)\in O\subseteq N$ by
            \obv{neighbourhoods}.
        Hence $N\subseteq\carrier[P]$ by \obv{powerset_elim}.
        Take $O\in\opens[P]$ such that $(x,y)\in O\subseteq N$.
        Take $R\in\productRectangles{X}{Y}$ such that
            $(x,y)\in R\subseteq O$ by \obv{open_basis}.
        Take $U\in\opens[X]$ such that there exists $V\in\opens[Y]$
            such that $R=U\times V$ by \obv{product_rectangles}.
        Take $V\in\opens[Y]$ such that $R=U\times V$.
        Hence $(x,y)\in U\times V$.
        Hence $x\in U$ and $y\in V$ by \obv{times_tuple_elim}.
        Hence $U\in\neighbourhoods{x}{X}$ and
            $V\in\neighbourhoods{y}{Y}$ by
            \obv{open_set_containing_point_is_neighbourhood}.
        $U\times V\subseteq O$.
        Hence $U\times V\subseteq N$ by \obv{subseteq_transitive}.
        $N\subseteq\carrier[X]\times\carrier[Y]$.
        Follows by \auto{product_filter_iff}.
    \end{subproof}
    Hence $\neighbourhoods{(x,y)}{P}\subseteq
        \productFilter{\carrier[X]}{\carrier[Y]}{
            \neighbourhoods{x}{X}}{\neighbourhoods{y}{Y}}$ by
        \obv{subseteq}.

    Show that every element of
        $\productFilter{\carrier[X]}{\carrier[Y]}{
            \neighbourhoods{x}{X}}{\neighbourhoods{y}{Y}}$ is an
        element of $\neighbourhoods{(x,y)}{P}$.
    \begin{subproof}
        Fix $N\in\productFilter{\carrier[X]}{\carrier[Y]}{
            \neighbourhoods{x}{X}}{\neighbourhoods{y}{Y}}$.
        $N\in\productFilter{\carrier[X]}{\carrier[Y]}{
            \neighbourhoods{x}{X}}{\neighbourhoods{y}{Y}}$ iff
            $N\subseteq\carrier[X]\times\carrier[Y]$ and there
            exist $A,B$ such that
                $A\in\neighbourhoods{x}{X}$ and
                $B\in\neighbourhoods{y}{Y}$ and
                $A\times B\subseteq N$ by \obv{product_filter_iff}.
        Hence $N\subseteq\carrier[X]\times\carrier[Y]$ and there
            exist $A,B$ such that
                $A\in\neighbourhoods{x}{X}$ and
                $B\in\neighbourhoods{y}{Y}$ and
                $A\times B\subseteq N$.
        Take $A,B$ such that $A\in\neighbourhoods{x}{X}$ and
            $B\in\neighbourhoods{y}{Y}$ and $A\times B\subseteq N$.
        Take $U\in\opens[X]$ such that $x\in U\subseteq A$ by
            \obv{neighbourhoods}.
        Take $V\in\opens[Y]$ such that $y\in V\subseteq B$ by
            \obv{neighbourhoods}.
        $U\subseteq\carrier[X]$ and $V\subseteq\carrier[Y]$ by
            \obv{opens_type}.
        Hence $U\times V\subseteq\carrier[X]\times\carrier[Y]$ by
            \auto{times,subseteq}.
        Hence $U\times V\in\pow{\carrier[X]\times\carrier[Y]}$ by
            \obv{powerset_intro}.
        Hence $U\times V\in\productRectangles{X}{Y}$ by
            \auto{product_rectangles}.
        Hence $U\times V\in\opens[P]$ by \auto{open_basis,subseteq}.
        $(x,y)\in U\times V$ by \obv{times_tuple_intro}.
        $U\times V\subseteq A\times B$ by \auto{times,subseteq}.
        Hence $U\times V\subseteq N$ by \obv{subseteq_transitive}.
        $N\subseteq\carrier[P]$.
        Hence $N\in\pow{\carrier[P]}$ by \obv{powerset_intro}.
        Follows by \obv{neighbourhoods}.
    \end{subproof}
    Hence $\productFilter{\carrier[X]}{\carrier[Y]}{
        \neighbourhoods{x}{X}}{\neighbourhoods{y}{Y}}
        \subseteq\neighbourhoods{(x,y)}{P}$ by \obv{subseteq}.
    Hence $\neighbourhoods{(x,y)}{P}
        =\productFilter{\carrier[X]}{\carrier[Y]}{
            \neighbourhoods{x}{X}}{\neighbourhoods{y}{Y}}$ by
        \obv{subseteq_antisymmetric}.
    $P=\BinaryProductTopology{X}{Y}$.
\end{proof}
