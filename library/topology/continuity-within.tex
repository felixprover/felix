\section{Continuity within a set}

\import{topology/continuity-at.tex}
\import{topology/subspace-neighbourhood-filter.tex}
\import{topology/subspace-continuity.tex}

For a total graph, continuity within an arbitrary set uses the ambient
neighbourhood-within filter.  Continuity on the set means continuity within
at each point of its carrier trace.

\begin{definition}\label{continuous_at_within}
    $f$ is continuous at $x$ within $A$ from $X$ to $Y$ iff
        $X$ is a topological space and
        $Y$ is a topological space and
        $f\in\funs{\carrier[X]}{\carrier[Y]}$ and
        $x\in\carrier[X]$ and
        $f$ tends from $\neighbourhoodsWithin{x}{A}{X}$ on
            $\carrier[X]$ to
            $\neighbourhoods{f(x)}{Y}$ on $\carrier[Y]$.
\end{definition}

\begin{definition}\label{continuous_on}
    $f$ is continuous on $A$ from $X$ to $Y$ iff
        $X$ is a topological space and
        $Y$ is a topological space and
        $f\in\funs{\carrier[X]}{\carrier[Y]}$ and
        for every $x\in A\inter\carrier[X]$ we have
        $f$ is continuous at $x$ within $A$ from $X$ to $Y$.
\end{definition}

\begin{theorem}\label{continuous_at_within_iff_preimages}
    Let $X,Y$ be topological spaces.
    Let $f\in\funs{\carrier[X]}{\carrier[Y]}$ and
        $x\in\carrier[X]$.
    Then $f$ is continuous at $x$ within $A$ from $X$ to $Y$ iff
        for every $N\in\neighbourhoods{f(x)}{Y}$ we have
        $\preimg{f}{N}\in\neighbourhoodsWithin{x}{A}{X}$.
\end{theorem}
\begin{proof}
    $f(x)\in\carrier[Y]$ by \obv{funs_type_apply}.
    $\neighbourhoodsWithin{x}{A}{X}$ is a filter on $\carrier[X]$ by
        \obv{neighbourhoods_within_is_filter}.
    $\neighbourhoods{f(x)}{Y}$ is a filter on $\carrier[Y]$ by
        \obv{neighbourhoods_is_filter}.
    Follows by \auto{continuous_at_within,filter_tendsto_iff_preimage}.
\end{proof}

\begin{theorem}\label{continuous_implies_continuous_at_within}
    Suppose $f$ is continuous from $X$ to $Y$.
    Let $x\in\carrier[X]$.
    Then $f$ is continuous at $x$ within $A$ from $X$ to $Y$.
\end{theorem}
\begin{proof}
    $X$ is a topological space and $Y$ is a topological space and
        $f\in\funs{\carrier[X]}{\carrier[Y]}$ by
        \obv{continuous_map}.
    $f$ is continuous from $X$ to $Y$ iff
        for every $z\in\carrier[X]$ we have
        $f$ is continuous at $z$ from $X$ to $Y$ by
        \obv{continuous_iff_continuous_at_every_point}.
    Hence for every $z\in\carrier[X]$ we have
        $f$ is continuous at $z$ from $X$ to $Y$.
    Thus $f$ is continuous at $x$ from $X$ to $Y$.
    Hence $f$ tends from $\neighbourhoods{x}{X}$ on $\carrier[X]$ to
        $\neighbourhoods{f(x)}{Y}$ on $\carrier[Y]$ by
        \obv{continuous_at}.
    $\neighbourhoodsWithin{x}{A}{X}$ refines
        $\neighbourhoods{x}{X}$ on $\carrier[X]$ by
        \obv{neighbourhoods_within_refines_neighbourhoods}.
    $f(x)\in\carrier[Y]$ by \obv{funs_type_apply}.
    $\neighbourhoods{f(x)}{Y}$ is a filter on $\carrier[Y]$ by
        \obv{neighbourhoods_is_filter}.
    Hence $\neighbourhoods{f(x)}{Y}$ refines
        $\neighbourhoods{f(x)}{Y}$ on $\carrier[Y]$ by
        \obv{filter_refines_reflexive}.
    Thus $f$ tends from $\neighbourhoodsWithin{x}{A}{X}$ on
        $\carrier[X]$ to $\neighbourhoods{f(x)}{Y}$ on $\carrier[Y]$ by
        \obv{filter_tendsto_mono}.
    Follows by \obv{continuous_at_within}.
\end{proof}

\begin{theorem}\label{continuous_at_within_carrier_iff}
    Let $X,Y$ be topological spaces.
    Let $f\in\funs{\carrier[X]}{\carrier[Y]}$ and
        $x\in\carrier[X]$.
    Then $f$ is continuous at $x$ within $\carrier[X]$ from $X$ to $Y$
        iff $f$ is continuous at $x$ from $X$ to $Y$.
\end{theorem}
\begin{proof}
    $\neighbourhoodsWithin{x}{\carrier[X]}{X}
        =\neighbourhoods{x}{X}$ by \obv{neighbourhoods_within_carrier}.
    Follows by \auto{continuous_at_within,continuous_at}.
\end{proof}

\begin{corollary}\label{continuous_implies_continuous_on}
    Suppose $f$ is continuous from $X$ to $Y$.
    Then $f$ is continuous on $A$ from $X$ to $Y$.
\end{corollary}
\begin{proof}
    $X$ is a topological space and $Y$ is a topological space and
        $f\in\funs{\carrier[X]}{\carrier[Y]}$ by
        \obv{continuous_map}.
    Show that for every $x\in A\inter\carrier[X]$ we have
        $f$ is continuous at $x$ within $A$ from $X$ to $Y$.
    \begin{subproof}
        Fix $x\in A\inter\carrier[X]$.
        Hence $x\in\carrier[X]$ by \obv{inter}.
        Follows by \obv{continuous_implies_continuous_at_within}.
    \end{subproof}
    Follows by \obv{continuous_on}.
\end{proof}

\begin{theorem}\label{continuous_on_carrier_iff_continuous}
    Let $X,Y$ be topological spaces.
    Let $f\in\funs{\carrier[X]}{\carrier[Y]}$.
    Then $f$ is continuous on $\carrier[X]$ from $X$ to $Y$ iff
        $f$ is continuous from $X$ to $Y$.
\end{theorem}
\begin{proof}
    $\carrier[X]\inter\carrier[X]=\carrier[X]$ by
        \obv{inter_idempotent}.
    Show that if $f$ is continuous on $\carrier[X]$ from $X$ to $Y$,
        then $f$ is continuous from $X$ to $Y$.
    \begin{subproof}
        Assume $f$ is continuous on $\carrier[X]$ from $X$ to $Y$.
        Show that for every $x\in\carrier[X]$ we have
            $f$ is continuous at $x$ from $X$ to $Y$.
        \begin{subproof}
            Fix $x\in\carrier[X]$.
            Hence $x\in\carrier[X]\inter\carrier[X]$ by
                \obv{inter_intro}.
            Thus $f$ is continuous at $x$ within $\carrier[X]$ from
                $X$ to $Y$ by \obv{continuous_on}.
            Follows by \obv{continuous_at_within_carrier_iff}.
        \end{subproof}
        Follows by \obv{continuous_iff_continuous_at_every_point}.
    \end{subproof}
    If $f$ is continuous from $X$ to $Y$, then
        $f$ is continuous on $\carrier[X]$ from $X$ to $Y$ by
        \obv{continuous_implies_continuous_on}.
    Follows by \auto.
\end{proof}

\subsection{Restriction to a subspace}

The following predicates are the exact-domain interface.  They apply directly
to a graph whose domain is the carrier trace, without choosing values outside
that trace.

\begin{definition}\label{continuous_at_as_map_on}
    $f$ is a continuous partial map at $x$ on $A$ from $X$ to $Y$ iff
        $X$ is a topological space and
        $Y$ is a topological space and
        $f\in\funs{A\inter\carrier[X]}{\carrier[Y]}$ and
        $x\in A\inter\carrier[X]$ and
        $f$ is continuous at $x$ from $\subspaceTopology{X}{A}$ to $Y$.
\end{definition}

\begin{definition}\label{continuous_as_map_on}
    $f$ is a continuous partial map on $A$ from $X$ to $Y$ iff
        $X$ is a topological space and
        $Y$ is a topological space and
        $f\in\funs{A\inter\carrier[X]}{\carrier[Y]}$ and
        $f$ is continuous from $\subspaceTopology{X}{A}$ to $Y$.
\end{definition}

\begin{proposition}\label{restriction_is_composition_with_subspace_inclusion}
    Let $X,Y$ be topological spaces.
    Let $f\in\funs{\carrier[X]}{\carrier[Y]}$.
    Then
        $\restrl{f}{A\inter\carrier[X]}
            =f\circ\subspaceInclusion{X}{A}$.
\end{proposition}
\begin{proof}
    $\restrl{f}{A\inter\carrier[X]}$ is a relation by
        \auto{funs_is_relation,relation,restrl}.
    $f\circ\subspaceInclusion{X}{A}$ is a relation by
        \obv{circ_is_relation}.
    For all $x,y$ we have
        $x\mathrel{\restrl{f}{A\inter\carrier[X]}}y$ iff
        $x\mathrel{f\circ\subspaceInclusion{X}{A}}y$ by
        \auto{subspace_inclusion,restrl_iff,circ_iff,id_iff}.
    Follows by \obv{relext}.
\end{proof}

\begin{theorem}\label{continuous_at_within_iff_restriction_continuous_at}
    Let $X,Y$ be topological spaces.
    Let $f\in\funs{\carrier[X]}{\carrier[Y]}$ and
        $x\in A\inter\carrier[X]$.
    Then $f$ is continuous at $x$ within $A$ from $X$ to $Y$ iff
        $\restrl{f}{A\inter\carrier[X]}$ is continuous at $x$ from
        $\subspaceTopology{X}{A}$ to $Y$.
\end{theorem}
\begin{proof}
    $x\in\carrier[X]$ by \obv{inter}.
    $\subspaceTopology{X}{A}$ is a topological space by
        \obv{subspace_topology_is_topological_space}.
    $\carrier[\subspaceTopology{X}{A}]=A\inter\carrier[X]$ by
        \obv{subspace_topology_carrier}.
    $f$ is a function and $\dom{f}=\carrier[X]$ by
        \obv{funs_is_function,funs}.
    $A\inter\carrier[X]\subseteq\carrier[X]$ by
        \obv{inter_lower_right}.
    Hence $A\inter\carrier[X]\subseteq\dom{f}$.
    $\restrl{f}{A\inter\carrier[X]}
        \in\funs{A\inter\carrier[X]}{\carrier[Y]}$ by
        \obv{funs_restrl}.
    $(\restrl{f}{A\inter\carrier[X]})(x)=f(x)$ by
        \obv{restrl_of_function_apply}.
    $\subspaceInclusion{X}{A}\in
        \funs{A\inter\carrier[X]}{\carrier[X]}$ by
        \obv{subspace_inclusion_type}.
    $x\in\carrier[\subspaceTopology{X}{A}]$.
    Hence $\neighbourhoods{x}{\subspaceTopology{X}{A}}$ is a filter on
        $A\inter\carrier[X]$ by
        \obv{neighbourhoods_is_filter}.
    $f(x)\in\carrier[Y]$ by \obv{funs_type_apply}.
    Hence $\neighbourhoods{f(x)}{Y}$ is a filter on $\carrier[Y]$ by
        \obv{neighbourhoods_is_filter}.
    $\filterPushforward{A\inter\carrier[X]}{\carrier[X]}{
        \subspaceInclusion{X}{A}}{
        \neighbourhoods{x}{\subspaceTopology{X}{A}}}
        =\neighbourhoodsWithin{x}{A}{X}$ by
        \obv{subspace_neighbourhood_pushforward_is_within}.
    $f$ tends from
        $\filterPushforward{A\inter\carrier[X]}{\carrier[X]}{
            \subspaceInclusion{X}{A}}{
            \neighbourhoods{x}{\subspaceTopology{X}{A}}}$ on
        $\carrier[X]$ to $\neighbourhoods{f(x)}{Y}$ on $\carrier[Y]$ iff
        $f\circ\subspaceInclusion{X}{A}$ tends from
            $\neighbourhoods{x}{\subspaceTopology{X}{A}}$ on
            $A\inter\carrier[X]$ to
            $\neighbourhoods{f(x)}{Y}$ on $\carrier[Y]$ by
        \obv{filter_tendsto_from_pushforward_iff_composition}.
    $\restrl{f}{A\inter\carrier[X]}
        =f\circ\subspaceInclusion{X}{A}$ by
        \obv{restriction_is_composition_with_subspace_inclusion}.
    Follows by \auto{continuous_at_within,continuous_at}.
\end{proof}

\begin{theorem}\label{continuous_on_iff_restriction_continuous_from_subspace}
    Let $X,Y$ be topological spaces.
    Let $f\in\funs{\carrier[X]}{\carrier[Y]}$.
    Then $f$ is continuous on $A$ from $X$ to $Y$ iff
        $\restrl{f}{A\inter\carrier[X]}$ is continuous from
        $\subspaceTopology{X}{A}$ to $Y$.
\end{theorem}
\begin{proof}
    $\subspaceTopology{X}{A}$ is a topological space by
        \obv{subspace_topology_is_topological_space}.
    $\carrier[\subspaceTopology{X}{A}]=A\inter\carrier[X]$ by
        \obv{subspace_topology_carrier}.
    $\restrl{f}{A\inter\carrier[X]}
        \in\funs{A\inter\carrier[X]}{\carrier[Y]}$ by
        \obv{funs_restrl}.
    Hence $\restrl{f}{A\inter\carrier[X]}\in
        \funs{\carrier[\subspaceTopology{X}{A}]}{\carrier[Y]}$.

    Show that if $f$ is continuous on $A$ from $X$ to $Y$, then
        $\restrl{f}{A\inter\carrier[X]}$ is continuous from
            $\subspaceTopology{X}{A}$ to $Y$.
    \begin{subproof}
        Assume $f$ is continuous on $A$ from $X$ to $Y$.
        Show that for every
            $x\in\carrier[\subspaceTopology{X}{A}]$ we have
            $\restrl{f}{A\inter\carrier[X]}$ is continuous at $x$ from
            $\subspaceTopology{X}{A}$ to $Y$.
        \begin{subproof}
            Fix $x\in\carrier[\subspaceTopology{X}{A}]$.
            Hence $x\in A\inter\carrier[X]$.
            Thus $f$ is continuous at $x$ within $A$ from $X$ to $Y$ by
                \obv{continuous_on}.
            Follows by
                \obv{continuous_at_within_iff_restriction_continuous_at}.
        \end{subproof}
        $\restrl{f}{A\inter\carrier[X]}$ is continuous from
            $\subspaceTopology{X}{A}$ to $Y$ iff for every
            $z\in\carrier[\subspaceTopology{X}{A}]$ we have
            $\restrl{f}{A\inter\carrier[X]}$ is continuous at $z$ from
            $\subspaceTopology{X}{A}$ to $Y$ by
            \obv{continuous_iff_continuous_at_every_point}.
    \end{subproof}

    Show that if $\restrl{f}{A\inter\carrier[X]}$ is continuous from
        $\subspaceTopology{X}{A}$ to $Y$, then
        $f$ is continuous on $A$ from $X$ to $Y$.
    \begin{subproof}
        Assume $\restrl{f}{A\inter\carrier[X]}$ is continuous from
            $\subspaceTopology{X}{A}$ to $Y$.
        $\restrl{f}{A\inter\carrier[X]}$ is continuous from
            $\subspaceTopology{X}{A}$ to $Y$ iff for every
            $z\in\carrier[\subspaceTopology{X}{A}]$ we have
            $\restrl{f}{A\inter\carrier[X]}$ is continuous at $z$ from
            $\subspaceTopology{X}{A}$ to $Y$ by
            \obv{continuous_iff_continuous_at_every_point}.
        Hence for every $z\in\carrier[\subspaceTopology{X}{A}]$ we have
            $\restrl{f}{A\inter\carrier[X]}$ is continuous at $z$ from
            $\subspaceTopology{X}{A}$ to $Y$.
        Show that for every $x\in A\inter\carrier[X]$ we have
            $f$ is continuous at $x$ within $A$ from $X$ to $Y$.
        \begin{subproof}
            Fix $x\in A\inter\carrier[X]$.
            Hence $x\in\carrier[\subspaceTopology{X}{A}]$.
            Hence $\restrl{f}{A\inter\carrier[X]}$ is continuous at $x$
                from $\subspaceTopology{X}{A}$ to $Y$.
            Follows by
                \obv{continuous_at_within_iff_restriction_continuous_at}.
        \end{subproof}
        Follows by \obv{continuous_on}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{corollary}\label{continuous_at_within_iff_restriction_continuous_as_map_on}
    Let $X,Y$ be topological spaces.
    Let $f\in\funs{\carrier[X]}{\carrier[Y]}$ and
        $x\in A\inter\carrier[X]$.
    Then $f$ is continuous at $x$ within $A$ from $X$ to $Y$ iff
        $\restrl{f}{A\inter\carrier[X]}$ is a continuous partial map at
            $x$ on $A$ from $X$ to $Y$.
\end{corollary}
\begin{proof}
    $\restrl{f}{A\inter\carrier[X]}
        \in\funs{A\inter\carrier[X]}{\carrier[Y]}$ by
        \obv{funs_restrl}.
    Follows by
        \auto{continuous_at_as_map_on,
            continuous_at_within_iff_restriction_continuous_at}.
\end{proof}

\begin{corollary}\label{continuous_on_iff_restriction_continuous_as_map_on}
    Let $X,Y$ be topological spaces.
    Let $f\in\funs{\carrier[X]}{\carrier[Y]}$.
    Then $f$ is continuous on $A$ from $X$ to $Y$ iff
        $\restrl{f}{A\inter\carrier[X]}$ is a continuous partial map on
            $A$ from $X$ to $Y$.
\end{corollary}
\begin{proof}
    $\restrl{f}{A\inter\carrier[X]}
        \in\funs{A\inter\carrier[X]}{\carrier[Y]}$ by
        \obv{funs_restrl}.
    Follows by
        \auto{continuous_as_map_on,
            continuous_on_iff_restriction_continuous_from_subspace}.
\end{proof}
