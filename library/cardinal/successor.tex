\section{Successor cardinals}

\import{cardinal/comparison.tex}

For a cardinal input, its Hartogs ordinal is exactly the least larger
cardinal.  The term remains total on arbitrary sets, while its mathematical
interface below keeps the cardinal hypothesis explicit.

\begin{definition}\label{cardinal_successor}
    $\cardSucc{\kappa}=\hartogs{\kappa}$.
\end{definition}

\begin{theorem}\label{cardinal_successor_is_cardinal}
    Let $\kappa$ be a cardinal.
    Then $\cardSucc{\kappa}$ is a cardinal.
\end{theorem}
\begin{proof}
    Let $H=\cardSucc{\kappa}$.
    $H=\hartogs{\kappa}$ by \obv{cardinal_successor}.
    Hence $H$ is an ordinal by \obv{hartogs_is_ordinal}.
    Show that for every $\alpha\in H$ we have
        $\lnot(\alpha\approx H)$.
    \begin{subproof}
        Fix $\alpha\in H$.
        Suppose $\alpha\approx H$.
        There exists $f\in\inj{\alpha}{\kappa}$
            by \auto{below_hartogs_injection}.
        Hence $\cardLe{\alpha}{\kappa}$ by \obv{cardinal_le}.
        $\cardLe{H}{\alpha}$
            by \obv{equinum_sym,equinum_cardinal_le}.
        Hence $\cardLe{H}{\kappa}$
            by \obv{cardinal_le_transitive}.
        Hence there exists $g\in\inj{H}{\kappa}$
            by \auto{cardinal_le}.
        Contradiction by \auto{hartogs_theorem}.
    \end{subproof}
    Hence $H$ is an initial ordinal by \obv{initial_ordinal}.
    Follows by \obv{cardinal}.
\end{proof}

\begin{proposition}\label{cardinal_belongs_to_successor}
    Let $\kappa$ be a cardinal.
    Then $\kappa\in\cardSucc{\kappa}$.
\end{proposition}
\begin{proof}
    $\kappa$ is an ordinal by \obv{cardinal_is_ordinal}.
    Hence $\memrel{\kappa}$ is a strict well order on $\kappa$
        by \obv{ordinal_membership_is_strict_well_order}.
    $\orderType{\kappa}{\memrel{\kappa}}$ is an ordinal
        by \obv{order_type_is_ordinal}.
    $\orderIsomorphism{\orderCollapse{\kappa}{\memrel{\kappa}}}
        {\kappa}{\memrel{\kappa}}
        {\orderType{\kappa}{\memrel{\kappa}}}
        {\memrel{\orderType{\kappa}{\memrel{\kappa}}}}$
        by \obv{order_collapse_is_isomorphism}.
    Hence $\kappa=\orderType{\kappa}{\memrel{\kappa}}$
        by \obv{ordinal_order_isomorphism_unique}.
    $\orderType{\kappa}{\memrel{\kappa}}\in\hartogs{\kappa}$
        by \obv{order_type_belongs_to_hartogs,subseteq_refl}.
    Follows by \obv{cardinal_successor}.
\end{proof}

\begin{proposition}\label{cardinal_lt_successor}
    Let $\kappa$ be a cardinal.
    Then $\cardLt{\kappa}{\cardSucc{\kappa}}$.
\end{proposition}
\begin{proof}
    $\cardSucc{\kappa}$ is a cardinal
        by \obv{cardinal_successor_is_cardinal}.
    Hence $\card{\kappa}=\kappa$ and
        $\card{\cardSucc{\kappa}}=\cardSucc{\kappa}$
        by \obv{card_of_cardinal}.
    $\kappa\in\cardSucc{\kappa}$
        by \obv{cardinal_belongs_to_successor}.
    Follows by \auto{card_elem_iff_cardinal_lt}.
\end{proof}

\begin{theorem}\label{cardinal_successor_minimal}
    Let $\kappa,\lambda$ be cardinals.
    Suppose $\cardLt{\kappa}{\lambda}$.
    Then $\cardSucc{\kappa}\subseteq\lambda$.
\end{theorem}
\begin{proof}
    Let $H=\cardSucc{\kappa}$.
    $H$ is a cardinal by \obv{cardinal_successor_is_cardinal}.
    Hence $H$ and $\lambda$ are ordinals
        by \auto{cardinal,initial_ordinal}.
    $\card{\kappa}=\kappa$ and $\card{\lambda}=\lambda$
        by \obv{card_of_cardinal}.
    Hence $\kappa\in\lambda$ by \auto{card_elem_iff_cardinal_lt}.
    Suppose not.
    $\lambda\subseteq H$ by \auto{ordinal_subseteq_or_subseteq}.
    Hence $\lambda\in H$ or $\lambda=H$
        by \obv{ordinal_subseteq_implies_elem_or_eq}.
    Hence $\lambda\in H$.
    There exists $f\in\inj{\lambda}{\kappa}$
        by \auto{cardinal_successor,below_hartogs_injection}.
    Hence $\cardLe{\lambda}{\kappa}$ by \obv{cardinal_le}.
    Hence $\lambda\subseteq\kappa$
        by \obv{cardinal,initial_ordinal_cardinal_le_iff_subseteq}.
    Hence $\kappa\in\kappa$ by \obv{subseteq}.
    Contradiction by \obv{in_irrefl}.
\end{proof}

\begin{proposition}\label{below_cardinal_successor_cardinal_le}
    Let $\kappa$ be a cardinal.
    Let $\alpha\in\cardSucc{\kappa}$.
    Then $\cardLe{\alpha}{\kappa}$.
\end{proposition}
\begin{proof}
    There exists $f\in\inj{\alpha}{\kappa}$
        by \auto{cardinal_successor,below_hartogs_injection}.
    Follows by \obv{cardinal_le}.
\end{proof}

\begin{proposition}\label{card_of_below_cardinal_successor}
    Let $\kappa$ be a cardinal.
    Let $\alpha\in\cardSucc{\kappa}$.
    Then $\card{\alpha}\subseteq\kappa$.
\end{proposition}
\begin{proof}
    $\cardLe{\alpha}{\kappa}$
        by \obv{below_cardinal_successor_cardinal_le}.
    Hence $\card{\alpha}\subseteq\card{\kappa}$
        by \obv{card_subseteq_iff_cardinal_le}.
    $\card{\kappa}=\kappa$ by \obv{card_of_cardinal}.
\end{proof}

\begin{theorem}\label{ordinal_below_cardinal_successor_iff}
    Let $\kappa$ be a cardinal.
    Let $\alpha$ be an ordinal.
    Then $\alpha\in\cardSucc{\kappa}$ iff
        $\card{\alpha}\subseteq\kappa$.
\end{theorem}
\begin{proof}
    If $\alpha\in\cardSucc{\kappa}$, then
        $\card{\alpha}\subseteq\kappa$
        by \obv{card_of_below_cardinal_successor}.
    Show that if $\card{\alpha}\subseteq\kappa$, then
        $\alpha\in\cardSucc{\kappa}$.
    \begin{subproof}
        Assume $\card{\alpha}\subseteq\kappa$.
        $\card{\kappa}=\kappa$ by \obv{card_of_cardinal}.
        Hence $\card{\alpha}\subseteq\card{\kappa}$.
        Hence $\cardLe{\alpha}{\kappa}$
            by \obv{card_subseteq_iff_cardinal_le}.
        Take $f$ such that $f\in\inj{\alpha}{\kappa}$
            by \auto{cardinal_le}.
        Let $H=\cardSucc{\kappa}$.
        $H$ is an ordinal
            by \auto{cardinal_successor,hartogs_is_ordinal}.
        Suppose not.
        $H\subseteq\alpha$ or $\alpha\subseteq H$
            by \obv{ordinal_subseteq_or_subseteq}.
        Hence $H\subseteq\alpha$
            by \obv{ordinal_subseteq_implies_elem_or_eq}.
        Hence $\restrl{f}{H}\in\inj{H}{\kappa}$
            by \obv{inj_restrl}.
        Contradiction by \auto{cardinal_successor,hartogs_theorem}.
    \end{subproof}
    Follows by \auto.
\end{proof}
