\section{Exponentiation at infinite cardinals}

\import{cardinal/exponentiation.tex}
\import{cardinal/infinite.tex}

\begin{theorem}\label{infinite_cardinal_self_power}
    Let $\theta$ be an infinite cardinal.
    Then $\cardPower{\theta}{\theta}
        =\cardPower{\cardTwo}{\theta}$.
\end{theorem}
\begin{proof}
    $\cardOne\in\naturals$
        by \auto{cardinal_one,omega_empty,omega_successor}.
    Hence $\cardTwo\in\naturals$
        by \auto{cardinal_two_successor,omega_successor}.
    $\naturals\subseteq\theta$
        by \obv{naturals_subseteq_infinite_cardinal}.
    Hence $\cardTwo\in\theta$ by \obv{subseteq}.
    $\cardTwo$ and $\theta$ are ordinals
        by \auto{cardinal_two_is_cardinal,cardinal,initial_ordinal}.
    Hence $\cardTwo\subseteq\theta$
        by \obv{ordinal_elem_implies_subseteq}.
    Hence $\cardLe{\cardTwo}{\theta}$
        by \obv{subseteq_cardinal_le}.
    Hence $\cardLe{\cardPower{\cardTwo}{\theta}}
        {\cardPower{\theta}{\theta}}$
        by \obv{cardinal_power_monotone_base}.
    $\theta\in\cardPower{\cardTwo}{\theta}$
        by \obv{cantor_cardinal_power_for_cardinal}.
    $\cardPower{\cardTwo}{\theta}$ is an ordinal
        by \auto{cardinal_power_is_cardinal,cardinal,initial_ordinal}.
    Hence $\theta\subseteq\cardPower{\cardTwo}{\theta}$
        by \obv{ordinal_elem_implies_subseteq}.
    Hence $\cardLe{\theta}{\cardPower{\cardTwo}{\theta}}$
        by \obv{subseteq_cardinal_le}.
    Hence $\cardLe{\cardPower{\theta}{\theta}}
        {\cardPower{\cardPower{\cardTwo}{\theta}}{\theta}}$
        by \obv{cardinal_power_monotone_base}.
    $\cardOne\in\cardTwo$
        by \obv{cardinal_two_successor,suc_intro_self}.
    Hence $\cardTwo$ is inhabited.
    Hence $\cardPower{\cardTwo}{\cardProduct{\theta}{\theta}}
        =\cardPower{\cardPower{\cardTwo}{\theta}}{\theta}$
        by \obv{cardinal_power_product_exponent}.
    $\cardProduct{\theta}{\theta}=\theta$
        by \obv{infinite_cardinal_product_square}.
    Hence $\cardPower{\cardPower{\cardTwo}{\theta}}{\theta}
        =\cardPower{\cardTwo}{\theta}$.
    Hence $\cardLe{\cardPower{\theta}{\theta}}
        {\cardPower{\cardTwo}{\theta}}$.
    Hence $\cardPower{\theta}{\theta}
        \approx\cardPower{\cardTwo}{\theta}$
        by \obv{cardinal_le_antisymmetric_up_to_equinum}.
    $\cardPower{\theta}{\theta}$ and
        $\cardPower{\cardTwo}{\theta}$ are initial ordinals
        by \auto{cardinal_power_is_cardinal,cardinal}.
    Follows by \obv{equinumerous_initial_ordinals_are_equal}.
\end{proof}
