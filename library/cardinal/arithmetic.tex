\section{Cardinal sum and product}

\begin{definition}\label{cardinal_zero}
    $\cardZero=0$.
\end{definition}

\begin{definition}\label{cardinal_one}
    $\cardOne=1$.
\end{definition}

\begin{proposition}\label{cardinal_zero_emptyset}
    $\cardZero=\emptyset$.
\end{proposition}
\begin{proof}
    Follows by \obv{cardinal_zero}.
\end{proof}

\begin{proposition}\label{cardinal_one_successor}
    $\cardOne=\suc{\emptyset}$.
\end{proposition}
\begin{proof}
    Follows by \obv{cardinal_one}.
\end{proof}

\begin{definition}\label{cardinal_sum}
    $\cardSum{\kappa}{\lambda}
        =\card{\taggedSum{\kappa}{\lambda}}$.
\end{definition}

\begin{definition}\label{cardinal_product}
    $\cardProduct{\kappa}{\lambda}=\card{\kappa\times\lambda}$.
\end{definition}

\begin{proposition}\label{cardinal_zero_is_cardinal}
    $\cardZero$ is a cardinal.
\end{proposition}
\begin{proof}
    $\cardZero$ is an ordinal by
        \obv{cardinal_zero,emptyset_is_ordinal}.
    For every $\alpha\in\cardZero$ we have
        $\lnot(\alpha\approx\cardZero)$ by \obv{cardinal_zero,emptyset}.
    Hence $\cardZero$ is an initial ordinal by \obv{initial_ordinal}.
    Follows by \obv{cardinal}.
\end{proof}

\begin{proposition}\label{cardinal_one_is_cardinal}
    $\cardOne$ is a cardinal.
\end{proposition}
\begin{proof}
    $\emptyset$ is an ordinal by \obv{emptyset_is_ordinal}.
    Hence $\cardOne$ is an ordinal
        by \obv{cardinal_one,suc_ordinal}.
    Show that for every $\alpha\in\cardOne$ we have
        $\lnot(\alpha\approx\cardOne)$.
    \begin{subproof}
        Fix $\alpha\in\cardOne$.
        Hence $\alpha=\emptyset$ by
            \auto{cardinal_one,suc_elim,emptyset}.
        Suppose $\alpha\approx\cardOne$.
        Hence $\emptyset\approx\cardOne$.
        Take $f$ such that $f\in\bijections{\emptyset}{\cardOne}$
            by \obv{equinum}.
        $\emptyset\in\cardOne$
            by \obv{cardinal_one,suc_intro_self}.
        Take $a\in\emptyset$ such that $f(a)=\emptyset$
            by \auto{bijections,surj}.
        Contradiction by \obv{emptyset}.
    \end{subproof}
    Hence $\cardOne$ is an initial ordinal by \obv{initial_ordinal}.
    Follows by \obv{cardinal}.
\end{proof}

\begin{proposition}\label{card_of_cardinal_zero}
    $\card{\emptyset}=\cardZero$.
\end{proposition}
\begin{proof}
    Follows by \obv{cardinal_zero,cardinal_zero_is_cardinal,card_of_cardinal}.
\end{proof}

\begin{proposition}\label{card_of_cardinal_one}
    $\card{\cardOne}=\cardOne$.
\end{proposition}
\begin{proof}
    Follows by \obv{cardinal_one_is_cardinal,card_of_cardinal}.
\end{proof}

\begin{proposition}\label{cardinal_sum_is_cardinal}
    $\cardSum{\kappa}{\lambda}$ is a cardinal.
\end{proposition}
\begin{proof}
    Follows by \auto{cardinal_sum,card_is_cardinal}.
\end{proof}

\begin{proposition}\label{cardinal_product_is_cardinal}
    $\cardProduct{\kappa}{\lambda}$ is a cardinal.
\end{proposition}
\begin{proof}
    Follows by \auto{cardinal_product,card_is_cardinal}.
\end{proof}

\begin{proposition}\label{cardinal_sum_congruent}
    Suppose $A\approx C$ and $B\approx D$.
    Then $\cardSum{A}{B}=\cardSum{C}{D}$.
\end{proposition}
\begin{proof}
    $\taggedSum{A}{B}\approx\taggedSum{C}{D}$
        by \obv{tagged_sum_equinum_congruent}.
    Follows by \auto{cardinal_sum,card_eq_iff_equinumerous}.
\end{proof}

\begin{proposition}\label{cardinal_product_congruent}
    Suppose $A\approx C$ and $B\approx D$.
    Then $\cardProduct{A}{B}=\cardProduct{C}{D}$.
\end{proposition}
\begin{proof}
    $A\times B\approx C\times D$ by \obv{times_equinum_congruent}.
    Follows by \auto{cardinal_product,card_eq_iff_equinumerous}.
\end{proof}

\begin{proposition}\label{cardinal_sum_commutative}
    $\cardSum{\kappa}{\lambda}=\cardSum{\lambda}{\kappa}$.
\end{proposition}
\begin{proof}
    $\taggedSum{\kappa}{\lambda}\approx\taggedSum{\lambda}{\kappa}$
        by \obv{tagged_sum_equinum_commutative}.
    Follows by \auto{cardinal_sum,card_eq_iff_equinumerous}.
\end{proof}

\begin{proposition}\label{cardinal_product_commutative}
    $\cardProduct{\kappa}{\lambda}=\cardProduct{\lambda}{\kappa}$.
\end{proposition}
\begin{proof}
    $\kappa\times\lambda\approx\lambda\times\kappa$
        by \obv{times_equinum_commutative}.
    Follows by \auto{cardinal_product,card_eq_iff_equinumerous}.
\end{proof}

\begin{proposition}\label{cardinal_sum_zero_right}
    Let $\kappa$ be a cardinal.
    Then $\cardSum{\kappa}{\cardZero}=\kappa$.
\end{proposition}
\begin{proof}
    $\taggedSum{\kappa}{\emptyset}\approx\kappa$
        by \obv{tagged_sum_empty_right_equinum}.
    Hence $\card{\taggedSum{\kappa}{\emptyset}}=\card{\kappa}$
        by \obv{card_eq_iff_equinumerous}.
    $\card{\kappa}=\kappa$ by \obv{card_of_cardinal}.
    Follows by \obv{cardinal_sum,cardinal_zero}.
\end{proof}

\begin{corollary}\label{cardinal_sum_zero_left}
    Let $\kappa$ be a cardinal.
    Then $\cardSum{\cardZero}{\kappa}=\kappa$.
\end{corollary}
\begin{proof}
    Follows by \auto{cardinal_sum_commutative,cardinal_sum_zero_right}.
\end{proof}

\begin{proposition}\label{cardinal_product_one_right}
    Let $\kappa$ be a cardinal.
    Then $\cardProduct{\kappa}{\cardOne}=\kappa$.
\end{proposition}
\begin{proof}
    $\cardOne=\{\emptyset\}$
        by \obv{cardinal_one,suc,cons_iff,setext}.
    Hence $\kappa\times\cardOne\approx\kappa$
        by \auto{times_singleton_right_equinum}.
    Hence $\card{\kappa\times\cardOne}=\card{\kappa}$
        by \obv{card_eq_iff_equinumerous}.
    $\card{\kappa}=\kappa$ by \obv{card_of_cardinal}.
    Follows by \obv{cardinal_product}.
\end{proof}

\begin{corollary}\label{cardinal_product_one_left}
    Let $\kappa$ be a cardinal.
    Then $\cardProduct{\cardOne}{\kappa}=\kappa$.
\end{corollary}
\begin{proof}
    Follows by
        \auto{cardinal_product_commutative,cardinal_product_one_right}.
\end{proof}

\begin{proposition}\label{cardinal_product_zero_right}
    $\cardProduct{\kappa}{\cardZero}=\cardZero$.
\end{proposition}
\begin{proof}
    $\kappa\times\cardZero=\emptyset$
        by \auto{cardinal_zero,times_empty_right}.
    Follows by \obv{cardinal_product,card_of_cardinal_zero}.
\end{proof}

\begin{corollary}\label{cardinal_product_zero_left}
    $\cardProduct{\cardZero}{\kappa}=\cardZero$.
\end{corollary}
\begin{proof}
    Follows by
        \auto{cardinal_product_commutative,cardinal_product_zero_right}.
\end{proof}

\begin{proposition}\label{cardinal_sum_associative}
    $\cardSum{\cardSum{\kappa}{\lambda}}{\mu}
        =\cardSum{\kappa}{\cardSum{\lambda}{\mu}}$.
\end{proposition}
\begin{proof}
    $\cardSum{\kappa}{\lambda}
        \approx\taggedSum{\kappa}{\lambda}$
        by \auto{cardinal_sum,card_equinumerous}.
    Hence
        $\taggedSum{\cardSum{\kappa}{\lambda}}{\mu}
            \approx\taggedSum{\taggedSum{\kappa}{\lambda}}{\mu}$
        by \obv{tagged_sum_equinum_congruent,equinum_refl}.
    $\taggedSum{\taggedSum{\kappa}{\lambda}}{\mu}
        \approx\taggedSum{\kappa}{\taggedSum{\lambda}{\mu}}$
        by \obv{tagged_sum_equinum_associative}.
    $\cardSum{\lambda}{\mu}
        \approx\taggedSum{\lambda}{\mu}$
        by \auto{cardinal_sum,card_equinumerous}.
    Hence $\taggedSum{\kappa}{\taggedSum{\lambda}{\mu}}
        \approx\taggedSum{\kappa}{\cardSum{\lambda}{\mu}}$
        by
        \obv{tagged_sum_equinum_congruent,equinum_refl,equinum_sym}.
    Hence
        $\taggedSum{\cardSum{\kappa}{\lambda}}{\mu}
            \approx\taggedSum{\kappa}{\cardSum{\lambda}{\mu}}$
        by \obv{equinum_tran}.
    Follows by \auto{cardinal_sum,card_eq_iff_equinumerous}.
\end{proof}

\begin{proposition}\label{cardinal_product_associative}
    $\cardProduct{\cardProduct{\kappa}{\lambda}}{\mu}
        =\cardProduct{\kappa}{\cardProduct{\lambda}{\mu}}$.
\end{proposition}
\begin{proof}
    $\cardProduct{\kappa}{\lambda}\approx\kappa\times\lambda$
        by \auto{cardinal_product,card_equinumerous}.
    Hence $\cardProduct{\kappa}{\lambda}\times\mu
        \approx(\kappa\times\lambda)\times\mu$
        by \obv{times_equinum_congruent,equinum_refl}.
    $(\kappa\times\lambda)\times\mu
        \approx\kappa\times(\lambda\times\mu)$
        by \obv{times_equinum_associative}.
    $\cardProduct{\lambda}{\mu}\approx\lambda\times\mu$
        by \auto{cardinal_product,card_equinumerous}.
    Hence $\kappa\times(\lambda\times\mu)
        \approx\kappa\times\cardProduct{\lambda}{\mu}$
        by \obv{times_equinum_congruent,equinum_refl,equinum_sym}.
    Hence $\cardProduct{\kappa}{\lambda}\times\mu
        \approx\kappa\times\cardProduct{\lambda}{\mu}$
        by \obv{equinum_tran}.
    Follows by \auto{cardinal_product,card_eq_iff_equinumerous}.
\end{proof}

\begin{proposition}\label{cardinal_product_distributes_over_sum}
    $\cardProduct{\kappa}{\cardSum{\lambda}{\mu}}
        =\cardSum{\cardProduct{\kappa}{\lambda}}
            {\cardProduct{\kappa}{\mu}}$.
\end{proposition}
\begin{proof}
    $\cardSum{\lambda}{\mu}\approx\taggedSum{\lambda}{\mu}$
        by \auto{cardinal_sum,card_equinumerous}.
    Hence $\kappa\times\cardSum{\lambda}{\mu}
        \approx\kappa\times\taggedSum{\lambda}{\mu}$
        by \obv{times_equinum_congruent,equinum_refl}.
    $\kappa\times\taggedSum{\lambda}{\mu}
        \approx\taggedSum{\kappa\times\lambda}{\kappa\times\mu}$
        by \obv{times_tagged_sum_equinum_distributive}.
    $\cardProduct{\kappa}{\lambda}\approx\kappa\times\lambda$ and
        $\cardProduct{\kappa}{\mu}\approx\kappa\times\mu$
        by \auto{cardinal_product,card_equinumerous}.
    Hence $\taggedSum{\kappa\times\lambda}{\kappa\times\mu}
        \approx
        \taggedSum{\cardProduct{\kappa}{\lambda}}
            {\cardProduct{\kappa}{\mu}}$
        by \obv{tagged_sum_equinum_congruent,equinum_sym}.
    Hence $\kappa\times\cardSum{\lambda}{\mu}
        \approx
        \taggedSum{\cardProduct{\kappa}{\lambda}}
            {\cardProduct{\kappa}{\mu}}$
        by \obv{equinum_tran}.
    Follows by
        \auto{cardinal_product,cardinal_sum,card_eq_iff_equinumerous}.
\end{proof}

\begin{corollary}\label{cardinal_product_distributes_over_sum_right}
    $\cardProduct{\cardSum{\kappa}{\lambda}}{\mu}
        =\cardSum{\cardProduct{\kappa}{\mu}}
            {\cardProduct{\lambda}{\mu}}$.
\end{corollary}
\begin{proof}
    Follows by
        \auto{cardinal_product_commutative,cardinal_product_distributes_over_sum}.
\end{proof}

\begin{theorem}\label{cardinal_sum_monotone}
    Suppose $\cardLe{A}{C}$ and $\cardLe{B}{D}$.
    Then $\cardLe{\cardSum{A}{B}}{\cardSum{C}{D}}$.
\end{theorem}
\begin{proof}
    Take $f$ such that $f\in\inj{A}{C}$ by \auto{cardinal_le}.
    Take $g$ such that $g\in\inj{B}{D}$ by \auto{cardinal_le}.
    Take $h$ such that
        $h\in\inj{\taggedSum{A}{B}}{\taggedSum{C}{D}}$
        by \auto{tagged_sum_injection_congruent}.
    Hence $\cardLe{\taggedSum{A}{B}}{\taggedSum{C}{D}}$
        by \obv{cardinal_le}.
    Hence $\card{\taggedSum{A}{B}}
        \subseteq\card{\taggedSum{C}{D}}$
        by \auto{card_subseteq_iff_cardinal_le}.
    Hence $\cardLe{\card{\taggedSum{A}{B}}}
        {\card{\taggedSum{C}{D}}}$
        by \obv{subseteq_cardinal_le}.
    Follows by \obv{cardinal_sum}.
\end{proof}

\begin{theorem}\label{cardinal_product_monotone}
    Suppose $\cardLe{A}{C}$ and $\cardLe{B}{D}$.
    Then $\cardLe{\cardProduct{A}{B}}{\cardProduct{C}{D}}$.
\end{theorem}
\begin{proof}
    Take $f$ such that $f\in\inj{A}{C}$ by \auto{cardinal_le}.
    Take $g$ such that $g\in\inj{B}{D}$ by \auto{cardinal_le}.
    Take $h$ such that $h\in\inj{A\times B}{C\times D}$
        by \auto{times_injection_congruent}.
    Hence $\cardLe{A\times B}{C\times D}$ by \obv{cardinal_le}.
    Hence $\card{A\times B}\subseteq\card{C\times D}$
        by \auto{card_subseteq_iff_cardinal_le}.
    Hence $\cardLe{\card{A\times B}}{\card{C\times D}}$
        by \obv{subseteq_cardinal_le}.
    Follows by \obv{cardinal_product}.
\end{proof}
