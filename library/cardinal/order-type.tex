\begin{theorem}\label{full_cardinality_subset_order_type}
    Let $\kappa$ be a cardinal.
    Suppose $H\subseteq\kappa$ and $H\approx\kappa$.
    Then $\orderType{H}{\memrel{H}}=\kappa$.
\end{theorem}
\begin{proof}
    $\kappa$ is an initial ordinal by definition.
    Hence $\kappa$ is an ordinal by definition.
    $\memrel{H}$ is a strict well order on $H$ by
        \obv{ordinal_subset_strict_well_order}.
    $\orderType{H}{\memrel{H}}$ is an ordinal by
        \obv{order_type_is_ordinal}.
    $\orderType{H}{\memrel{H}}\subseteq\kappa$ by
        \obv{ordinal_subset_order_type_bound}.
    $\orderIsomorphism{\orderCollapse{H}{\memrel{H}}}
        {H}{\memrel{H}}
        {\orderType{H}{\memrel{H}}}
        {\memrel{\orderType{H}{\memrel{H}}}}$ by
        \obv{order_collapse_is_isomorphism}.
    Let $\beta=\orderType{H}{\memrel{H}}$.
    $\beta$ is an ordinal and $\beta\subseteq\kappa$.
    $\orderIsomorphism{\orderCollapse{H}{\memrel{H}}}
        {H}{\memrel{H}}{\beta}{\memrel{\beta}}$.
    Hence $\orderCollapse{H}{\memrel{H}}\in\bijections{H}{\beta}$ by
        \obv{order_isomorphism_type}.
    Hence $H\approx\beta$ by \obv{equinum}.
    Hence $\beta\approx H$ by \obv{equinum_sym}.
    Hence $\beta\approx\kappa$ by \obv{equinum_tran}.
    $\beta\in\kappa$ or $\beta=\kappa$ by
        \obv{ordinal_subseteq_implies_elem_or_eq}.
    $\beta\notin\kappa$ by \obv{initial_ordinal}.
\end{proof}

\begin{theorem}\label{infinite_well_order_has_omega_suborder}
    Suppose $R$ is a strict well order on $A$ and $A$ is infinite.
    Then there exists $B\subseteq A$ such that
        $\orderType{B}{\relationTrace{R}{B}}=\naturals$.
\end{theorem}
\begin{proof}
    $\orderType{A}{R}$ is an ordinal by \obv{order_type_is_ordinal}.
    $\orderIsomorphism{\orderCollapse{A}{R}}
        {A}{R}{\orderType{A}{R}}{\memrel{\orderType{A}{R}}}$ by
        \obv{order_collapse_is_isomorphism}.
    Let $\alpha=\orderType{A}{R}$.
    $\orderIsomorphism{\orderCollapse{A}{R}}
        {A}{R}{\alpha}{\memrel{\alpha}}$.
    Hence $\orderCollapse{A}{R}\in\bijections{A}{\alpha}$ by
        \obv{order_isomorphism_type}.
    Hence $A\approx\alpha$ by \obv{equinum}.
    Hence $\alpha\approx A$ by \obv{equinum_sym}.
    Show that $\alpha$ is infinite.
    \begin{subproof}
        Suppose not.
        Hence $\alpha$ is finite.
        Hence $A$ is finite by \obv{finite_equinumerous}.
        Contradiction.
    \end{subproof}
    Hence $\naturals\subseteq\alpha$ by
        \obv{omega_below_infinite_ordinal}.
    $\orderIsomorphism{\converse{\orderCollapse{A}{R}}}
        {\alpha}{\memrel{\alpha}}{A}{R}$ by
        \obv{order_isomorphism_inverse}.
    Hence $\converse{\orderCollapse{A}{R}}\in\bijections{\alpha}{A}$ by
        \obv{order_isomorphism_type}.
    Hence $\converse{\orderCollapse{A}{R}}\in\funs{\alpha}{A}$ by
        \obv{bijections_to_funs}.
    $\img{\converse{\orderCollapse{A}{R}}}{\naturals}\subseteq A$ by
        \auto{img_subseteq_ran,funs_ran,subseteq_transitive}.
    Hence
        $\orderIsomorphism{
            \restrl{\converse{\orderCollapse{A}{R}}}{\naturals}}
            {\naturals}{\relationTrace{\memrel{\alpha}}{\naturals}}
            {\img{\converse{\orderCollapse{A}{R}}}{\naturals}}
            {\relationTrace{R}
                {\img{\converse{\orderCollapse{A}{R}}}{\naturals}}}$ by
        \obv{order_isomorphism_restriction_to_image}.
    $\relationTrace{\memrel{\alpha}}{\naturals}=\memrel{\naturals}$ by
        \obv{membership_relation_trace}.
    $\orderIsomorphism{
            \restrl{\converse{\orderCollapse{A}{R}}}{\naturals}}
        {\naturals}{\memrel{\naturals}}
        {\img{\converse{\orderCollapse{A}{R}}}{\naturals}}
        {\relationTrace{R}
            {\img{\converse{\orderCollapse{A}{R}}}{\naturals}}}$.
    Hence
        $\orderIsomorphism{
            \converse{\restrl{\converse{\orderCollapse{A}{R}}}
                {\naturals}}}
            {\img{\converse{\orderCollapse{A}{R}}}{\naturals}}
            {\relationTrace{R}
                {\img{\converse{\orderCollapse{A}{R}}}{\naturals}}}
            {\naturals}{\memrel{\naturals}}$ by
        \obv{order_isomorphism_inverse}.
    $\relationTrace{R}
        {\img{\converse{\orderCollapse{A}{R}}}{\naturals}}$
        is a strict well order on
        $\img{\converse{\orderCollapse{A}{R}}}{\naturals}$ by
        \obv{relation_trace_strict_well_order}.
    $\naturals$ is an ordinal by \obv{omega_is_ordinal}.
    $\orderIsomorphic{
            \img{\converse{\orderCollapse{A}{R}}}{\naturals}}
            {\relationTrace{R}
                {\img{\converse{\orderCollapse{A}{R}}}{\naturals}}}
            {\naturals}{\memrel{\naturals}}$ by
        \obv{order_isomorphic}.
    Hence $\naturals=
        \orderType{\img{\converse{\orderCollapse{A}{R}}}{\naturals}}
            {\relationTrace{R}
                {\img{\converse{\orderCollapse{A}{R}}}{\naturals}}}$ by
        \obv{ordinal_order_type_unique}.
    Let $B=\img{\converse{\orderCollapse{A}{R}}}{\naturals}$.
    $B\subseteq A$ and
        $\naturals=\orderType{B}{\relationTrace{R}{B}}$.
\end{proof}

\begin{corollary}\label{infinite_ordinal_subset_has_omega_order_type}
    Let $\alpha$ be an ordinal.
    Suppose $K\subseteq\alpha$ and $K$ is infinite.
    Then there exists $L\subseteq K$ such that
        $\orderType{L}{\memrel{L}}=\naturals$.
\end{corollary}
\begin{proof}
    $\memrel{K}$ is a strict well order on $K$ by
        \obv{ordinal_subset_strict_well_order}.
    Take $L\subseteq K$ such that
        $\orderType{L}{\relationTrace{\memrel{K}}{L}}=\naturals$ by
        \obv{infinite_well_order_has_omega_suborder}.
    $\relationTrace{\memrel{K}}{L}=\memrel{L}$ by
        \obv{membership_relation_trace}.
    Follows by \auto.
\end{proof}
