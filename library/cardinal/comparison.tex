\section{Comparison of cardinal representatives}

The legacy relation $\cardLe{X}{Y}$ remains the useful assertion that an
injection from $X$ to $Y$ exists.  The following results identify it with
ordinary ordinal inclusion between the canonical representatives.

\begin{theorem}\label{initial_ordinal_cardinal_le_iff_subseteq}
    Let $\kappa,\lambda$ be initial ordinals.
    Then $\cardLe{\kappa}{\lambda}$ iff $\kappa\subseteq\lambda$.
\end{theorem}
\begin{proof}
    Show that if $\cardLe{\kappa}{\lambda}$, then
        $\kappa\subseteq\lambda$.
    \begin{subproof}
        Assume $\cardLe{\kappa}{\lambda}$.
        Suppose not.
        Hence $\lambda\subseteq\kappa$
            by \obv{ordinal_subseteq_or_subseteq}.
        Hence $\cardLe{\lambda}{\kappa}$
            by \obv{subseteq_cardinal_le}.
        Hence $\kappa\approx\lambda$
            by \obv{cardinal_le_antisymmetric_up_to_equinum}.
        Hence $\kappa=\lambda$
            by \obv{equinumerous_initial_ordinals_are_equal}.
        Contradiction by \obv{subseteq_refl}.
    \end{subproof}
    Show that if $\kappa\subseteq\lambda$, then
        $\cardLe{\kappa}{\lambda}$.
    \begin{subproof}
        Follows by \obv{subseteq_cardinal_le}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{theorem}\label{card_subseteq_iff_cardinal_le}
    $\card{X}\subseteq\card{Y}$ iff $\cardLe{X}{Y}$.
\end{theorem}
\begin{proof}
    $\card{X}$ and $\card{Y}$ are initial ordinals
        by \obv{card_is_initial_ordinal}.
    Show that if $\card{X}\subseteq\card{Y}$, then
        $\cardLe{X}{Y}$.
    \begin{subproof}
        Assume $\card{X}\subseteq\card{Y}$.
        Hence $\cardLe{\card{X}}{\card{Y}}$
            by \obv{subseteq_cardinal_le}.
        $\cardLe{X}{\card{X}}$ and
            $\cardLe{\card{Y}}{Y}$
            by
            \auto{equinumerous_card,card_equinumerous,equinum_cardinal_le}.
        Follows by \obv{cardinal_le_transitive}.
    \end{subproof}
    Show that if $\cardLe{X}{Y}$, then
        $\card{X}\subseteq\card{Y}$.
    \begin{subproof}
        Assume $\cardLe{X}{Y}$.
        $\cardLe{\card{X}}{X}$ and
            $\cardLe{Y}{\card{Y}}$
            by
            \auto{card_equinumerous,equinumerous_card,equinum_cardinal_le}.
        Hence $\cardLe{\card{X}}{\card{Y}}$
            by \obv{cardinal_le_transitive}.
        Follows by \obv{initial_ordinal_cardinal_le_iff_subseteq}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{theorem}\label{cardinal_comparability}
    $\cardLe{X}{Y}$ or $\cardLe{Y}{X}$.
\end{theorem}
\begin{proof}
    $\card{X}$ and $\card{Y}$ are ordinals by \obv{card_is_ordinal}.
    $\card{X}\subseteq\card{Y}$ or
        $\card{Y}\subseteq\card{X}$
        by \obv{ordinal_subseteq_or_subseteq}.
    Follows by \auto{card_subseteq_iff_cardinal_le}.
\end{proof}

\begin{theorem}\label{card_eq_iff_mutual_cardinal_le}
    $\card{X}=\card{Y}$ iff
        $\cardLe{X}{Y}$ and $\cardLe{Y}{X}$.
\end{theorem}
\begin{proof}
    Follows by
        \auto{card_eq_iff_equinumerous,equinum_cardinal_le,cardinal_le_antisymmetric_up_to_equinum}.
\end{proof}

\begin{theorem}\label{card_elem_iff_cardinal_lt}
    $\card{X}\in\card{Y}$ iff $\cardLt{X}{Y}$.
\end{theorem}
\begin{proof}
    $\card{X}$ and $\card{Y}$ are ordinals by \obv{card_is_ordinal}.
    Show that if $\card{X}\in\card{Y}$, then $\cardLt{X}{Y}$.
    \begin{subproof}
        Assume $\card{X}\in\card{Y}$.
        Hence $\card{X}\subseteq\card{Y}$
            by \obv{ordinal_elem_implies_subseteq}.
        Hence $\cardLe{X}{Y}$ by \obv{card_subseteq_iff_cardinal_le}.
        Show that $\lnot(X\approx Y)$.
        \begin{subproof}
            Suppose $X\approx Y$.
            Hence $\card{X}=\card{Y}$ by \obv{card_eq_iff_equinumerous}.
            Contradiction by \obv{in_irrefl}.
        \end{subproof}
        Follows by \obv{cardinal_lt}.
    \end{subproof}
    Show that if $\cardLt{X}{Y}$, then
        $\card{X}\in\card{Y}$.
    \begin{subproof}
        Assume $\cardLt{X}{Y}$.
        Hence $\cardLe{X}{Y}$ and $\lnot(X\approx Y)$
            by \obv{cardinal_lt}.
        Hence $\card{X}\subseteq\card{Y}$
            by \obv{card_subseteq_iff_cardinal_le}.
        $\card{X}\neq\card{Y}$ by \obv{card_eq_iff_equinumerous}.
        Follows by \obv{ordinal_subseteq_implies_elem_or_eq}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{theorem}\label{cardinal_lt_transitive}
    Suppose $\cardLt{A}{B}$ and $\cardLt{B}{C}$.
    Then $\cardLt{A}{C}$.
\end{theorem}
\begin{proof}
    $\card{A}\in\card{B}$ by \obv{card_elem_iff_cardinal_lt}.
    $\card{B}\in\card{C}$ by \obv{card_elem_iff_cardinal_lt}.
    $\card{B}$ is an ordinal by \obv{card_is_ordinal}.
    $\card{C}$ is an ordinal by \obv{card_is_ordinal}.
    Hence $\card{B}\subseteq\card{C}$ by
        \obv{ordinal_elem_implies_subseteq}.
    Hence $\card{A}\in\card{C}$ by \obv{elem_subseteq}.
    Follows by \obv{card_elem_iff_cardinal_lt}.
\end{proof}

\begin{theorem}\label{cardinal_le_lt_transitive}
    Suppose $\cardLe{A}{B}$ and $\cardLt{B}{C}$.
    Then $\cardLt{A}{C}$.
\end{theorem}
\begin{proof}
    $\card{A}\subseteq\card{B}$ by
        \obv{card_subseteq_iff_cardinal_le}.
    $\card{B}\in\card{C}$ by \obv{card_elem_iff_cardinal_lt}.
    $\card{A}$ is an ordinal by \obv{card_is_ordinal}.
    $\card{B}$ is an ordinal by \obv{card_is_ordinal}.
    $\card{C}$ is an ordinal by \obv{card_is_ordinal}.
    $\card{B}\subseteq\card{C}$ by
        \obv{ordinal_elem_implies_subseteq}.
    Hence $\card{A}\in\card{B}$ or $\card{A}=\card{B}$ by
        \obv{ordinal_subseteq_implies_elem_or_eq}.
    Hence $\card{A}\in\card{C}$ by \obv{elem_subseteq}.
    Follows by \obv{card_elem_iff_cardinal_lt}.
\end{proof}

\begin{theorem}\label{cardinal_lt_le_transitive}
    Suppose $\cardLt{A}{B}$ and $\cardLe{B}{C}$.
    Then $\cardLt{A}{C}$.
\end{theorem}
\begin{proof}
    $\card{A}\in\card{B}$ by \obv{card_elem_iff_cardinal_lt}.
    $\card{B}\subseteq\card{C}$ by
        \obv{card_subseteq_iff_cardinal_le}.
    Hence $\card{A}\in\card{C}$ by \obv{elem_subseteq}.
    Follows by \obv{card_elem_iff_cardinal_lt}.
\end{proof}

\begin{lemma}\label{card_of_ordinal_below}
    Let $\kappa$ be an ordinal and $\alpha\in\kappa$.
    Then $\card{\alpha}\in\kappa$.
\end{lemma}
\begin{proof}
    $\alpha$ is an ordinal by \obv{elem_of_ordinal_is_ordinal}.
    $\alpha\approx\alpha$ by \obv{equinum_refl}.
    Hence $\card{\alpha}\subseteq\alpha$ by
        \obv{card_least_equinumerous_ordinal}.
    $\card{\alpha}$ is an ordinal by \obv{card_is_ordinal}.
    Hence $\card{\alpha}\in\alpha$ or $\card{\alpha}=\alpha$ by
        \obv{ordinal_subseteq_implies_elem_or_eq}.
    $\alpha\subseteq\kappa$ by \obv{ordinal_elem_implies_subseteq}.
    Follows by \obv{elem_subseteq}.
\end{proof}

\begin{corollary}\label{cardinal_element_small}
    Let $\kappa$ be a cardinal and $\alpha\in\kappa$.
    Then $\cardLt{\alpha}{\kappa}$.
\end{corollary}
\begin{proof}
    $\kappa$ is an initial ordinal by definition.
    Hence $\card{\alpha}\in\kappa$ by \obv{card_of_ordinal_below}.
    $\card{\kappa}=\kappa$ by \obv{card_of_cardinal}.
    Follows by \obv{card_elem_iff_cardinal_lt}.
\end{proof}

\begin{proposition}\label{card_belongs_to_cardinal_implies_cardinal_le}
    Let $\kappa$ be a cardinal.
    Suppose $\card{X}\in\kappa$.
    Then $\cardLe{X}{\kappa}$.
\end{proposition}
\begin{proof}
    $\card{\kappa}=\kappa$ by \obv{card_of_cardinal}.
    Hence $\card{X}\in\card{\kappa}$.
    Hence $\cardLt{X}{\kappa}$ by \obv{card_elem_iff_cardinal_lt}.
    Follows by \obv{cardinal_lt}.
\end{proof}
