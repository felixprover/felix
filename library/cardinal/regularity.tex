\section{Regular and limit cardinals}

\import{cardinal/cofinality.tex}
\import{cardinal/families.tex}

\begin{definition}\label{limit_cardinal}
    $\kappa$ is a limit cardinal iff $\kappa$ is an infinite cardinal and
        there exists no $\lambda$ such that $\lambda$ is a cardinal and
        $\kappa=\cardSucc{\lambda}$.
\end{definition}

\begin{theorem}\label{naturals_are_regular}
    $\naturals$ is a regular cardinal.
\end{theorem}
\begin{proof}
    $\naturals$ is an infinite cardinal
        by \obv{naturals_is_cardinal,naturals_is_infinite}.
    Hence $\naturals$ is a limit ordinal
        by \obv{infinite_cardinal_is_limit_ordinal}.
    Hence $\cofinality{\naturals}$ is an infinite cardinal
        by \obv{cofinality_of_limit_is_infinite_cardinal}.
    Hence $\naturals\subseteq\cofinality{\naturals}$
        by \obv{naturals_subseteq_infinite_cardinal}.
    $\cofinality{\naturals}\subseteq\naturals$
        by \obv{cofinality_subseteq_ordinal}.
    Hence $\cofinality{\naturals}=\naturals$
        by \obv{subseteq_antisymmetric}.
    Follows by \obv{regular_cardinal}.
\end{proof}

\begin{theorem}\label{cofinal_subset_equinumerous_regular_cardinal}
    Let $\kappa$ be a regular cardinal.
    Suppose $\cofinalIn{H}{\kappa}$.
    Then $H\approx\kappa$.
\end{theorem}
\begin{proof}
    $\kappa$ is an infinite cardinal and
        $\cofinality{\kappa}=\kappa$ by \obv{regular_cardinal}.
    $H\subseteq\kappa$ by \obv{cofinal_in}.
    $\identity{H}\in\funs{H}{H}$ by \obv{id_elem_funs}.
    Hence $\identity{H}\in\funs{H}{\kappa}$
        by \obv{funs_weaken_codom}.
    $\ran{\identity{H}}=H$ by \obv{id_ran}.
    Hence $\cofinalMap{\identity{H}}{H}{\kappa}$
        by \obv{cofinal_map}.
    Hence $\cofinality{\kappa}\subseteq\card{H}$
        by \obv{cofinality_le_domain_card}.
    Hence $\kappa\subseteq\card{H}$.
    $\cardLe{H}{\kappa}$ by \obv{subseteq_cardinal_le}.
    Hence $\card{H}\subseteq\card{\kappa}$
        by \obv{card_subseteq_iff_cardinal_le}.
    $\card{\kappa}=\kappa$ by \obv{card_of_cardinal}.
    Hence $\card{H}\subseteq\kappa$.
    Hence $\card{H}=\kappa$ by \obv{subseteq_antisymmetric}.
    Hence $\card{H}=\card{\kappa}$.
    Follows by \obv{card_eq_iff_equinumerous}.
\end{proof}

\begin{theorem}\label{infinite_cardinal_successor_is_regular}
    Let $\kappa$ be an infinite cardinal.
    Then $\cardSucc{\kappa}$ is a regular cardinal.
\end{theorem}
\begin{proof}
    Let $H=\cardSucc{\kappa}$.
    $H$ is a cardinal by \obv{cardinal_successor_is_cardinal}.
    $\kappa\in H$ by \obv{cardinal_belongs_to_successor}.
    $\kappa$ and $H$ are ordinals
        by \auto{cardinal,initial_ordinal}.
    Hence $\kappa\subseteq H$ by \obv{ordinal_elem_implies_subseteq}.
    Show that $H$ is infinite.
    \begin{subproof}
        Suppose not.
        Hence $H$ is finite.
        Hence $\kappa$ is finite by \obv{subseteq_finite_is_finite}.
        Contradiction.
    \end{subproof}
    Hence $H$ is a limit ordinal
        by \obv{infinite_cardinal_is_limit_ordinal}.
    Let $\lambda=\cofinality{H}$.
    $\lambda$ is a cardinal by \obv{cofinality_is_cardinal}.
    Hence $\lambda$ is an ordinal by \auto{cardinal,initial_ordinal}.
    $\lambda\subseteq H$ by \auto{cofinality_subseteq_ordinal}.
    Hence $\lambda\in H$ or $\lambda=H$
        by \obv{ordinal_subseteq_implies_elem_or_eq}.
    Show that $\lambda=H$.
    \begin{subproof}
        Suppose not.
        Hence $\lambda\in H$.
        Hence $\cardLe{\lambda}{\kappa}$
            by \obv{below_cardinal_successor_cardinal_le}.
        Take $f$ such that $\cofinalMap{f}{\lambda}{H}$
            by \auto{cofinality_map_exists}.
        Hence $f\in\funs{\lambda}{H}$ by \obv{cofinal_map}.
        Hence $f$ is a function and $\dom{f}=\lambda$
            by \obv{funs_is_function,funs}.
        Hence $f$ is a function on $\lambda$.
        For every $i\in\lambda$ we have $f(i)\in H$
            by \obv{funs_type_apply}.
        Hence $\unions{\ran{f}}\neq H$
            by \auto{no_small_indexed_cover_of_cardinal_successor}.
        $\unions{\ran{f}}=H$
            by \obv{cofinal_map_into_limit_iff_union_range}.
        Contradiction.
    \end{subproof}
    Hence $\cofinality{H}=H$.
    Follows by \obv{regular_cardinal}.
\end{proof}

\begin{proposition}\label{finite_cardinal_successor}
    Let $\lambda$ be a finite cardinal.
    Then $\cardSucc{\lambda}=\suc{\lambda}$.
\end{proposition}
\begin{proof}
    $\lambda$ is an ordinal by \auto{cardinal,initial_ordinal}.
    Hence $\lambda$ is a natural number
        by \obv{finite_ordinal_is_natural_number}.
    Hence $\suc{\lambda}$ is a natural number
        by \obv{omega_successor}.
    Hence $\suc{\lambda}$ is a cardinal
        by \obv{natural_number_is_cardinal}.
    $\card{\lambda}=\lambda$ and
        $\card{\suc{\lambda}}=\suc{\lambda}$
        by \obv{card_of_cardinal}.
    $\lambda\in\suc{\lambda}$ by \obv{suc_intro_self}.
    Hence $\cardLt{\lambda}{\suc{\lambda}}$
        by \auto{card_elem_iff_cardinal_lt}.
    Hence $\cardSucc{\lambda}\subseteq\suc{\lambda}$
        by \obv{cardinal_successor_minimal}.
    $\lambda\in\cardSucc{\lambda}$
        by \obv{cardinal_belongs_to_successor}.
    $\cardSucc{\lambda}$ is an ordinal
        by \auto{cardinal_successor_is_cardinal,cardinal,initial_ordinal}.
    Hence $\suc{\lambda}\subseteq\cardSucc{\lambda}$
        by \obv{ordinal_suc_subseteq}.
    Follows by \obv{subseteq_antisymmetric}.
\end{proof}

\begin{corollary}\label{finite_cardinal_has_finite_successor}
    Let $\lambda$ be a finite cardinal.
    Then $\cardSucc{\lambda}$ is finite.
\end{corollary}
\begin{proof}
    $\lambda$ is an ordinal by \auto{cardinal,initial_ordinal}.
    Hence $\lambda$ is a natural number
        by \obv{finite_ordinal_is_natural_number}.
    Hence $\suc{\lambda}$ is a natural number
        by \obv{omega_successor}.
    Hence $\suc{\lambda}$ is finite
        by \obv{natural_number_is_finite}.
    Follows by \obv{finite_cardinal_successor}.
\end{proof}

\begin{theorem}\label{singular_cardinal_is_limit_cardinal}
    Let $\kappa$ be a singular cardinal.
    Then $\kappa$ is a limit cardinal.
\end{theorem}
\begin{proof}
    $\kappa$ is an infinite cardinal and $\kappa$ is not a regular cardinal
        by \obv{singular_cardinal}.
    Show that there exists no $\lambda$ such that
        $\lambda$ is a cardinal and $\kappa=\cardSucc{\lambda}$.
    \begin{subproof}
        Suppose there exists $\lambda$ such that
            $\lambda$ is a cardinal and $\kappa=\cardSucc{\lambda}$.
        Take $\lambda$ such that $\lambda$ is a cardinal and
            $\kappa=\cardSucc{\lambda}$.
        \begin{byCase}
            \caseOf{$\lambda$ is finite.}
                Hence $\cardSucc{\lambda}$ is finite
                    by \obv{finite_cardinal_has_finite_successor}.
                Hence $\kappa$ is finite.
                Contradiction.
            \caseOf{$\lambda$ is infinite.}
                Hence $\cardSucc{\lambda}$ is a regular cardinal
                    by \obv{infinite_cardinal_successor_is_regular}.
                Hence $\kappa$ is a regular cardinal.
                Contradiction.
        \end{byCase}
    \end{subproof}
    Follows by \obv{limit_cardinal}.
\end{proof}

\begin{proposition}\label{cofinality_belongs_to_singular_cardinal}
    Let $\kappa$ be a singular cardinal.
    Then $\cofinality{\kappa}\in\kappa$.
\end{proposition}
\begin{proof}
    $\kappa$ is an infinite cardinal and $\kappa$ is not a regular cardinal
        by \obv{singular_cardinal}.
    Hence $\kappa$ is an ordinal by \auto{cardinal,initial_ordinal}.
    $\cofinality{\kappa}\subseteq\kappa$
        by \obv{cofinality_subseteq_ordinal}.
    $\cofinality{\kappa}\neq\kappa$
        by \auto{regular_cardinal}.
    Follows by \auto{cofinality_is_cardinal,cardinal,initial_ordinal,ordinal_subseteq_implies_elem_or_eq}.
\end{proof}

\begin{proposition}\label{infinite_cardinal_successor_belongs_to_singular_cardinal}
    Let $\kappa$ be a singular cardinal.
    Let $\theta\in\kappa$.
    Suppose $\theta$ is an infinite cardinal.
    Then $\cardSucc{\theta}\in\kappa$.
\end{proposition}
\begin{proof}
    $\kappa$ is an infinite cardinal and $\kappa$ is not a regular cardinal
        by \obv{singular_cardinal}.
    $\card{\theta}=\theta$ and $\card{\kappa}=\kappa$
        by \obv{card_of_cardinal}.
    Hence $\cardLt{\theta}{\kappa}$
        by \auto{card_elem_iff_cardinal_lt}.
    Hence $\cardSucc{\theta}\subseteq\kappa$
        by \obv{cardinal_successor_minimal}.
    $\cardSucc{\theta}$ is a regular cardinal
        by \obv{infinite_cardinal_successor_is_regular}.
    Hence $\cardSucc{\theta}\neq\kappa$.
    $\cardSucc{\theta}$ and $\kappa$ are ordinals
        by \auto{cardinal_successor_is_cardinal,cardinal,initial_ordinal}.
    Follows by \obv{ordinal_subseteq_implies_elem_or_eq}.
\end{proof}
