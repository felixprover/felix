\section{Cardinal bounds for indexed families}

The Choice use in this module is concentrated in the following uniformization
theorem.  Later bounds consume the resulting set-coded family of injections.

\begin{definition}\label{indexed_family_apply}
    $\indexedFamilyApply{E}{f}{I}
        =\{(i,(E(i))(f(i)))\mid i\in I\}$.
\end{definition}

\begin{proposition}\label{indexed_family_apply_type}
    Suppose $A$ is a function on $I$.
    Suppose $E$ is a function on $I$.
    Let $f\in\dependentProduct{I}{A}$.
    Suppose for every $i\in I$ we have
        $E(i)\in\inj{A(i)}{\kappa}$.
    Then $\indexedFamilyApply{E}{f}{I}\in\funs{I}{\kappa}$.
\end{proposition}
\begin{proof}
    Let $g=\indexedFamilyApply{E}{f}{I}$.
    $g$ is a relation by \auto{indexed_family_apply,relation}.
    $g$ is right-unique
        by \auto{indexed_family_apply,rightunique,pair_eq_iff}.
    $\dom{g}=I$
        by \auto{indexed_family_apply,setext,dom_iff,pair_eq_iff}.
    For every $i\in I$ we have $f(i)\in A(i)$
        by \obv{dependent_product_coordinate}.
    For every $i\in I$ we have $(E(i))(f(i))\in\kappa$
        by \auto{inj,funs_type_apply}.
    For every $i\in I$ we have $g(i)=(E(i))(f(i))$
        by \obv{indexed_family_apply,function_apply_intro}.
    Hence $g$ is a function to $\kappa$ by \auto.
    Follows by \auto{funs_intro}.
\end{proof}

\begin{proposition}\label{indexed_family_apply_application}
    Suppose $A$ is a function on $I$.
    Suppose $E$ is a function on $I$.
    Let $f\in\dependentProduct{I}{A}$.
    Suppose for every $j\in I$ we have
        $E(j)\in\inj{A(j)}{\kappa}$.
    Let $i\in I$.
    Then $(\indexedFamilyApply{E}{f}{I})(i)=(E(i))(f(i))$.
\end{proposition}
\begin{proof}
    $(i,(E(i))(f(i)))\in\indexedFamilyApply{E}{f}{I}$
        by \obv{indexed_family_apply}.
    $\indexedFamilyApply{E}{f}{I}$ is a function
        by \auto{indexed_family_apply_type,funs_is_function}.
    Follows by \obv{function_apply_intro}.
\end{proof}

\begin{theorem}\label{indexed_family_uniform_injections}
    Suppose $A$ is a function on $I$.
    Suppose for every $i\in I$ we have $\cardLe{A(i)}{\kappa}$.
    Then there exists $E$ such that $E$ is a function on $I$ and
        for every $i\in I$ we have $E(i)\in\inj{A(i)}{\kappa}$.
\end{theorem}
\begin{proof}
    Let $B=\{(i,\inj{A(i)}{\kappa})\mid i\in I\}$.
    $B$ is a relation by \auto{relation}.
    $B$ is right-unique by \auto{rightunique,pair_eq_iff}.
    $\dom{B}=I$ by \auto{setext,dom_iff,pair_eq_iff}.
    Hence $B$ is a function on $I$.
    For every $i\in I$ we have $B(i)=\inj{A(i)}{\kappa}$
        by \obv{function_apply_intro}.
    For every $i\in I$ we have $B(i)$ is inhabited
        by \auto{cardinal_le}.
    Hence $\dependentProduct{I}{B}$ is inhabited
        by \obv{dependent_product_inhabited_iff}.
    Take $E$ such that $E\in\dependentProduct{I}{B}$.
    Hence $E\in\funs{I}{\unions{\ran{B}}}$
        by \obv{dependent_product_iff}.
    Hence $E$ is a function and $\dom{E}=I$
        by \obv{funs_is_function,funs}.
    Hence $E$ is a function on $I$.
    For every $i\in I$ we have $E(i)\in B(i)$
        by \obv{dependent_product_coordinate}.
    Hence for every $i\in I$ we have
        $E(i)\in\inj{A(i)}{\kappa}$ by \auto.
    Follows by \auto.
\end{proof}

\begin{theorem}\label{indexed_sum_injection_into_product}
    Suppose $A$ is a function on $I$.
    Suppose for every $i\in I$ we have $\cardLe{A(i)}{\kappa}$.
    Then there exists $h\in\inj{\indexedSum{I}{A}}{I\times\kappa}$.
\end{theorem}
\begin{proof}
    Take $E$ such that $E$ is a function on $I$ and
        for every $i\in I$ we have $E(i)\in\inj{A(i)}{\kappa}$
        by \obv{indexed_family_uniform_injections}.
    Let $S=\indexedSum{I}{A}$.
    Let $h=\{(p,(\fst{p},(E(\fst{p}))(\snd{p})))\mid p\in S\}$.
    $h$ is a relation by \auto{relation}.
    $h$ is right-unique by \auto{rightunique,pair_eq_iff}.
    $\dom{h}=S$ by \auto{setext,dom_iff,pair_eq_iff}.
    Show that for every $p\in S$ we have
        $(\fst{p},(E(\fst{p}))(\snd{p}))\in I\times\kappa$.
    \begin{subproof}
        Fix $p\in S$.
        Hence $p\in I\times\unions{\ran{A}}$ and
            $\snd{p}\in A(\fst{p})$ by \obv{indexed_sum}.
        Hence $\fst{p}\in I$ by \obv{fst_type}.
        Hence $E(\fst{p})\in\inj{A(\fst{p})}{\kappa}$.
        Hence $(E(\fst{p}))(\snd{p})\in\kappa$
            by \auto{inj,funs_type_apply}.
        Follows by \obv{times_tuple_intro}.
    \end{subproof}
    For every $p\in S$ we have
        $h(p)=(\fst{p},(E(\fst{p}))(\snd{p}))$
        by \obv{function_apply_intro}.
    Hence $h\in\funs{S}{I\times\kappa}$ by \auto{funs_intro}.
    Show that $h$ is injective.
    \begin{subproof}
        Show that for all $p,q$ such that $p\in\dom{h}$ and
            $q\in\dom{h}$ and $h(p)=h(q)$ we have $p=q$.
        \begin{subproof}
            Fix $p,q$.
            Assume $p\in\dom{h}$ and $q\in\dom{h}$ and $h(p)=h(q)$.
            Hence $p,q\in S$.
            Hence $p,q\in I\times\unions{\ran{A}}$ and
                $\snd{p}\in A(\fst{p})$ and
                $\snd{q}\in A(\fst{q})$ by \auto{indexed_sum}.
            $h(p)=(\fst{p},(E(\fst{p}))(\snd{p}))$ and
                $h(q)=(\fst{q},(E(\fst{q}))(\snd{q}))$
                by \obv{function_apply_intro}.
            Hence $\fst{p}=\fst{q}$ and
                $(E(\fst{p}))(\snd{p})
                    =(E(\fst{q}))(\snd{q})$
                by \auto{pair_eq_iff}.
            $\fst{p}\in I$ by \obv{fst_type}.
            Hence $E(\fst{p})\in\inj{A(\fst{p})}{\kappa}$.
            Hence $E(\fst{p})$ is a function and
                $\dom{E(\fst{p})}=A(\fst{p})$
                by \auto{inj,funs_is_function,funs}.
            Hence $E(\fst{p})$ is injective
                by \auto{inj,injective_function}.
            Hence $\snd{p}=\snd{q}$
                by \auto{inj,funs,injective_function}.
            Follows by
                \auto{times_elem_is_tuple,pair_eq_pair_of_proj,pair_eq_iff}.
        \end{subproof}
        Follows by \auto{injective_function}.
    \end{subproof}
    Hence $h\in\inj{S}{I\times\kappa}$
        by \auto{inj,injective_function}.
    Follows by \auto.
\end{proof}

\begin{theorem}\label{dependent_product_injection_into_function_space}
    Suppose $A$ is a function on $I$.
    Suppose for every $i\in I$ we have $\cardLe{A(i)}{\kappa}$.
    Then there exists
        $h\in\inj{\dependentProduct{I}{A}}{\funs{I}{\kappa}}$.
\end{theorem}
\begin{proof}
    Take $E$ such that $E$ is a function on $I$ and
        for every $i\in I$ we have $E(i)\in\inj{A(i)}{\kappa}$
        by \obv{indexed_family_uniform_injections}.
    Let $P=\dependentProduct{I}{A}$.
    Let $h=\{(f,\indexedFamilyApply{E}{f}{I})\mid f\in P\}$.
    $h$ is a relation by \auto{relation}.
    $h$ is right-unique by \auto{rightunique,pair_eq_iff}.
    $\dom{h}=P$ by \auto{setext,dom_iff,pair_eq_iff}.
    For every $f\in P$ we have
        $\indexedFamilyApply{E}{f}{I}\in\funs{I}{\kappa}$
        by \obv{indexed_family_apply_type}.
    For every $f\in P$ we have
        $h(f)=\indexedFamilyApply{E}{f}{I}$
        by \obv{function_apply_intro}.
    Hence $h\in\funs{P}{\funs{I}{\kappa}}$ by \auto{funs_intro}.
    Show that $h$ is injective.
    \begin{subproof}
        Show that for all $f,g$ such that $f\in\dom{h}$ and
            $g\in\dom{h}$ and $h(f)=h(g)$ we have $f=g$.
        \begin{subproof}
            Fix $f,g$.
            Assume $f\in\dom{h}$ and $g\in\dom{h}$ and $h(f)=h(g)$.
            Hence $f,g\in P$.
            Hence $f,g\in\funs{I}{\unions{\ran{A}}}$
                by \auto{dependent_product_iff}.
            Hence $f$ and $g$ are functions and
                $\dom{f}=I=\dom{g}$
                by \obv{funs_is_function,funs}.
            $h(f)=\indexedFamilyApply{E}{f}{I}$ and
                $h(g)=\indexedFamilyApply{E}{g}{I}$
                by \obv{function_apply_intro}.
            Hence $\indexedFamilyApply{E}{f}{I}
                =\indexedFamilyApply{E}{g}{I}$.
            Show that for every $i$ we have $f(i)=g(i)$.
            \begin{subproof}
                Fix $i$.
                \begin{byCase}
                    \caseOf{$i\in I$.}
                        $f(i),g(i)\in A(i)$
                            by \obv{dependent_product_coordinate}.
                        $E(i)\in\inj{A(i)}{\kappa}$.
                        Hence $E(i)$ is a function and
                            $\dom{E(i)}=A(i)$ and $E(i)$ is injective
                            by \auto{inj,funs_is_function,funs,injective_function}.
                        $(\indexedFamilyApply{E}{f}{I})(i)
                            =(E(i))(f(i))$ and
                            $(\indexedFamilyApply{E}{g}{I})(i)
                            =(E(i))(g(i))$
                            by \obv{indexed_family_apply_application}.
                        Hence $(E(i))(f(i))=(E(i))(g(i))$.
                        Follows by \obv{inj}.
                    \caseOf{$i\notin I$.}
                        $f(i)=\emptyset$ and $g(i)=\emptyset$
                            by \obv{function_apply_default}.
                \end{byCase}
            \end{subproof}
            Follows by \obv{funext}.
        \end{subproof}
        Follows by \auto{injective_function}.
    \end{subproof}
    Hence $h\in\inj{P}{\funs{I}{\kappa}}$
        by \auto{inj,injective_function}.
    Follows by \auto.
\end{proof}

\begin{corollary}\label{dependent_product_cardinal_bound}
    Suppose $A$ is a function on $I$.
    Suppose for every $i\in I$ we have $\cardLe{A(i)}{\kappa}$.
    Then $\cardLe{\dependentProduct{I}{A}}
        {\cardPower{\kappa}{I}}$.
\end{corollary}
\begin{proof}
    Take $h$ such that
        $h\in\inj{\dependentProduct{I}{A}}{\funs{I}{\kappa}}$
        by \auto{dependent_product_injection_into_function_space}.
    Hence $\cardLe{\dependentProduct{I}{A}}{\funs{I}{\kappa}}$
        by \obv{cardinal_le}.
    $\funs{I}{\kappa}\approx\card{\funs{I}{\kappa}}$
        by \obv{equinumerous_card}.
    Hence $\cardLe{\funs{I}{\kappa}}{\card{\funs{I}{\kappa}}}$
        by \auto{equinum_cardinal_le}.
    Hence $\cardLe{\dependentProduct{I}{A}}
        {\card{\funs{I}{\kappa}}}$
        by \obv{cardinal_le_transitive}.
    Follows by \obv{cardinal_power}.
\end{proof}

\begin{corollary}\label{indexed_sum_cardinal_product_bound}
    Suppose $A$ is a function on $I$.
    Suppose for every $i\in I$ we have $\cardLe{A(i)}{\kappa}$.
    Then $\cardLe{\indexedSum{I}{A}}{\cardProduct{I}{\kappa}}$.
\end{corollary}
\begin{proof}

        $h\in\inj{\indexedSum{I}{A}}{I\times\kappa}$
        by \auto{indexed_sum_injection_into_product}.
    Hence $\cardLe{\indexedSum{I}{A}}{I\times\kappa}$
        by \obv{cardinal_le}.
    $I\times\kappa\approx\card{I\times\kappa}$
        by \obv{equinumerous_card}.
    Hence $\cardLe{I\times\kappa}{\card{I\times\kappa}}$
        by \auto{equinum_cardinal_le}.
    Hence $\cardLe{\indexedSum{I}{A}}{\card{I\times\kappa}}$
        by \obv{cardinal_le_transitive}.
    Follows by \obv{cardinal_product}.
\end{proof}

\begin{corollary}\label{indexed_sum_infinite_cardinal_bound}
    Let $\kappa$ be an infinite cardinal.
    Suppose $A$ is a function on $I$.
    Suppose $\cardLe{I}{\kappa}$.
    Suppose for every $i\in I$ we have $\cardLe{A(i)}{\kappa}$.
    Then $\cardLe{\indexedSum{I}{A}}{\kappa}$.
\end{corollary}
\begin{proof}
    $\cardLe{\indexedSum{I}{A}}{\cardProduct{I}{\kappa}}$
        by \obv{indexed_sum_cardinal_product_bound}.
    $\cardLe{\kappa}{\kappa}$
        by \obv{subseteq_refl,subseteq_cardinal_le}.
    Hence $\cardLe{\cardProduct{I}{\kappa}}
        {\cardProduct{\kappa}{\kappa}}$
        by \obv{cardinal_product_monotone}.
    $\cardProduct{\kappa}{\kappa}=\kappa$
        by \obv{infinite_cardinal_product_square}.
    Hence $\cardLe{\cardProduct{I}{\kappa}}{\kappa}$.
    Follows by \obv{cardinal_le_transitive}.
\end{proof}

\begin{corollary}\label{indexed_cover_infinite_cardinal_bound}
    Let $\kappa$ be an infinite cardinal.
    Suppose $A$ is a function on $I$.
    Suppose $X\subseteq\unions{\ran{A}}$ and
        $\cardLe{I}{\kappa}$.
    Suppose for every $i\in I$ we have $\cardLe{A(i)}{\kappa}$.
    Then $\cardLe{X}{\kappa}$.
\end{corollary}
\begin{proof}
    $\cardLe{X}{\unions{\ran{A}}}$
        by \obv{subseteq_cardinal_le}.
    $\cardLe{\unions{\ran{A}}}{\kappa}$
        by \obv{indexed_union_infinite_cardinal_bound}.
    Follows by \obv{cardinal_le_transitive}.
\end{proof}

\begin{theorem}\label{no_small_indexed_cover_of_cardinal_successor}
    Let $\kappa$ be an infinite cardinal.
    Suppose $A$ is a function on $I$ and $\cardLe{I}{\kappa}$.
    Suppose for every $i\in I$ we have $A(i)\in\cardSucc{\kappa}$.
    Then $\unions{\ran{A}}\neq\cardSucc{\kappa}$.
\end{theorem}
\begin{proof}
    For every $i\in I$ we have $\cardLe{A(i)}{\kappa}$
        by \obv{below_cardinal_successor_cardinal_le}.
    Hence $\cardLe{\unions{\ran{A}}}{\kappa}$
        by \obv{indexed_union_infinite_cardinal_bound}.
    Suppose $\unions{\ran{A}}=\cardSucc{\kappa}$.
    Hence $\cardLe{\cardSucc{\kappa}}{\kappa}$.
    Take $e$ such that $e\in\inj{\cardSucc{\kappa}}{\kappa}$
        by \auto{cardinal_le}.
    $\cardSucc{\kappa}=\hartogs{\kappa}$
        by \obv{cardinal_successor}.
    Contradiction by \auto{hartogs_theorem}.
\end{proof}
