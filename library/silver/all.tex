\begin{proposition}\label{silver_indexed_family_smoke}
    Suppose $A$ is a function on $I$.
    Let $i\in I$.
    Let $x\in A(i)$.
    Then $(i,x)\in\indexedSum{I}{A}$.
\end{proposition}
\begin{proof}
    Follows by \auto{indexed_sum_iff}.
\end{proof}
