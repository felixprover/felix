\section{Abstract reduction systems}

\begin{struct}\label{abstract_reduction_system}
    An abstract reduction system $A$ is a onesorted structure equipped with
    \begin{enumerate}
        \item\label{abstract_reduction_system_relation}
            a reduction relation
            $\reduction[A]\subseteq
                \carrier[A]\times\carrier[A]$;
    \end{enumerate}.
\end{struct}

\begin{classification}\label{reduction_relation_type}
    Let $A$ be an abstract reduction system.
    Then $\reduction[A]$ is a binary relation on $\carrier[A]$.
\end{classification}
\begin{proof}
    Follows by \obv{abstract_reduction_system_relation}.
\end{proof}

\begin{theorem}\label{exact_abstract_reduction_system_extensionality}
    Let $A,B$ be sets.
    Suppose $A$ is an exact abstract reduction system.
    Suppose $B$ is an exact abstract reduction system.
    Suppose $\carrier[A]=\carrier[B]$ and
        $\reduction[A]=\reduction[B]$.
    Then $A=B$.
\end{theorem}
\begin{proof}
    Follows by structure extensionality.
\end{proof}

\begin{theorem}\label{exact_abstract_reduction_system_reduct_eq}
    Let $A$ be a set.
    Suppose $A$ is an exact abstract reduction system.
    Then $\reduct[abstract reduction system]{A}=A$.
\end{theorem}
\begin{proof}
    Follows by \obv{abstract_reduction_systemreductidentity}.
\end{proof}

\begin{theorem}\label{abstract_reduction_system_reduct_is_exact}
    Let $A$ be an abstract reduction system.
    Then $\reduct[abstract reduction system]{A}$ is an exact
        abstract reduction system.
\end{theorem}
\begin{proof}
    $\carrier[\reduct[abstract reduction system]{A}]=\carrier[A]$
        by \obv{abstract_reduction_systemreductprojection_carrier}.
    $\reduction[\reduct[abstract reduction system]{A}]=\reduction[A]$
        by \obv{abstract_reduction_systemreductprojection_reduction}.
    Hence $\reduction[\reduct[abstract reduction system]{A}]\subseteq
        \carrier[\reduct[abstract reduction system]{A}]\times
        \carrier[\reduct[abstract reduction system]{A}]$
        by \obv{abstract_reduction_system_relation}.
    Hence $\reduct[abstract reduction system]{A}$ is an abstract reduction
        system by \auto{abstract_reduction_system}.
    Follows by \obv{abstract_reduction_systemshape,
        abstract_reduction_systemreductexact}.
\end{proof}

\begin{definition}\label{reduction_system_from_relation}
    $\ReductionSystem{X}{R}=
        \structure[abstract reduction system]{
            \carrier=X,
            \reduction=R
        }$.
\end{definition}

\begin{proposition}\label{reduction_system_from_relation_carrier}
    $\carrier[\ReductionSystem{X}{R}]=X$.
\end{proposition}
\begin{proof}
    Follows by \auto{reduction_system_from_relation,
        abstract_reduction_systemaggregateprojection_carrier}.
\end{proof}

\begin{proposition}\label{reduction_system_from_relation_reduction}
    $\reduction[\ReductionSystem{X}{R}]=R$.
\end{proposition}
\begin{proof}
    Follows by \auto{reduction_system_from_relation,
        abstract_reduction_systemaggregateprojection_reduction}.
\end{proof}

\begin{classification}\label{reduction_system_from_relation_is_system}
    Suppose $R$ is a binary relation on $X$.
    Then $\ReductionSystem{X}{R}$ is an abstract reduction system.
\end{classification}
\begin{proof}
    $R\subseteq X\times X$ by definition.
    Follows by \auto{reduction_system_from_relation,
        abstract_reduction_system,abstract_reduction_systemaggregateexact,
        abstract_reduction_systemaggregateprojection_carrier,
        abstract_reduction_systemaggregateprojection_reduction}.
\end{proof}

\begin{theorem}\label{reduction_system_from_relation_is_exact}
    Suppose $R$ is a binary relation on $X$.
    Then $\ReductionSystem{X}{R}$ is an exact abstract reduction system.
\end{theorem}
\begin{proof}
    $\ReductionSystem{X}{R}$ is an abstract reduction system by
        \obv{reduction_system_from_relation_is_system}.
    Follows by \auto{reduction_system_from_relation,
        abstract_reduction_systemaggregateexact}.
\end{proof}

\begin{theorem}\label{abstract_reduction_system_reduct_eq}
    Let $A$ be an abstract reduction system.
    Then $\reduct[abstract reduction system]{A}
        =\ReductionSystem{\carrier[A]}{\reduction[A]}$.
\end{theorem}
\begin{proof}
    $\reduct[abstract reduction system]{A}$ is an exact abstract reduction
        system by \obv{abstract_reduction_system_reduct_is_exact}.
    $\reduction[A]$ is a binary relation on $\carrier[A]$ by notion.
    $\ReductionSystem{\carrier[A]}{\reduction[A]}$ is an exact abstract
        reduction system by \obv{reduction_system_from_relation_is_exact}.
    $\carrier[\reduct[abstract reduction system]{A}]=
        \carrier[\ReductionSystem{\carrier[A]}{\reduction[A]}]$
        by \obv{abstract_reduction_systemreductprojection_carrier,
            reduction_system_from_relation_carrier}.
    $\reduction[\reduct[abstract reduction system]{A}]=
        \reduction[\ReductionSystem{\carrier[A]}{\reduction[A]}]$
        by \obv{abstract_reduction_systemreductprojection_reduction,
            reduction_system_from_relation_reduction}.
    Follows by \obv{exact_abstract_reduction_system_extensionality}.
\end{proof}

\begin{theorem}\label{reduction_system_from_relation_is_exact_iff}
    $\ReductionSystem{X}{R}$ is an exact abstract reduction system iff
        $R$ is a binary relation on $X$.
\end{theorem}
\begin{proof}
    Show that if $\ReductionSystem{X}{R}$ is an exact abstract reduction
        system, then $R$ is a binary relation on $X$.
    \begin{subproof}
        Suppose $\ReductionSystem{X}{R}$ is an exact abstract reduction
            system.
        $\ReductionSystem{X}{R}$ is an abstract reduction system.
        Hence $\reduction[\ReductionSystem{X}{R}]\subseteq
            \carrier[\ReductionSystem{X}{R}]\times
            \carrier[\ReductionSystem{X}{R}]$
            by \obv{abstract_reduction_system_relation}.
        Hence $R\subseteq X\times X$ by
            \obv{reduction_system_from_relation_carrier,
                reduction_system_from_relation_reduction}.
        Follows by definition.
    \end{subproof}
    Follows by \auto{reduction_system_from_relation_is_exact}.
\end{proof}

\begin{theorem}\label{exact_reduction_system_eta}
    Let $A$ be a set.
    Suppose $A$ is an exact abstract reduction system.
    Then $A=\ReductionSystem{\carrier[A]}{\reduction[A]}$.
\end{theorem}
\begin{proof}
    $\reduct[abstract reduction system]{A}=A$
        by \obv{exact_abstract_reduction_system_reduct_eq}.
    $\reduct[abstract reduction system]{A}=
        \ReductionSystem{\carrier[A]}{\reduction[A]}$
        by \obv{abstract_reduction_system_reduct_eq}.
\end{proof}

\begin{definition}\label{reduction_plus}
    Let $A$ be an abstract reduction system.
    $\reductionPlus{A}=\transitiveClosure{\reduction[A]}$.
\end{definition}

\begin{definition}\label{reduction_star}
    Let $A$ be an abstract reduction system.
    $\reductionStar{A}=

\end{definition}

\begin{theorem}\label{reduction_star_first_step}
    Let $A$ be an abstract reduction system.
    Let $x,y\in\carrier[A]$.
    Then $x\mathrel{\reductionStar{A}}y$ iff
        $x=y$ or there exists $z\in\carrier[A]$ such that
            $x\mathrel{\reduction[A]}z$ and
            $z\mathrel{\reductionStar{A}}y$.
\end{theorem}
\begin{proof}
    $\reduction[A]$ is a binary relation on $\carrier[A]$ by notion.
    Follows by \auto{reduction_star,
        reflexive_transitive_closure_first_step}.
\end{proof}
