\section{Lattice filters}

\import{order/lattice.tex}
\import{relation.tex}

\begin{definition}\label{lattice_filter}
    $\latticeFilter{F}{L}$ iff
        $L$ is a bounded lattice and
        $F\subseteq\carrier[L]$ and
        $\latticeTop[L]\in F$ and
        $\latticeBottom[L]\notin F$ and
        (for all $a,b\in F$ we have $\meet[L](a,b)\in F$) and
        (for every $a\in F$ we have
            for every $b\in\carrier[L]$ such that
                $a\mathrel{\leq[L]}b$ we have $b\in F$).
\end{definition}

\begin{proposition}\label{lattice_filter_ambient}
    Suppose $\latticeFilter{F}{L}$.
    Then $L$ is a bounded lattice.
\end{proposition}
\begin{proof}
    Follows by \obv{lattice_filter}.
\end{proof}

\begin{proposition}\label{lattice_filter_subseteq_carrier}
    Suppose $\latticeFilter{F}{L}$.
    Then $F\subseteq\carrier[L]$.
\end{proposition}
\begin{proof}
    Follows by \obv{lattice_filter}.
\end{proof}

\begin{proposition}\label{lattice_filter_top}
    Suppose $\latticeFilter{F}{L}$.
    Then $\latticeTop[L]\in F$.
\end{proposition}
\begin{proof}
    Follows by \obv{lattice_filter}.
\end{proof}

\begin{proposition}\label{lattice_filter_bottom_notin}
    Suppose $\latticeFilter{F}{L}$.
    Then $\latticeBottom[L]\notin F$.
\end{proposition}
\begin{proof}
    Follows by \obv{lattice_filter}.
\end{proof}

\begin{proposition}\label{lattice_filter_meet}
    Suppose $\latticeFilter{F}{L}$.
    Let $a,b\in F$.
    Then $\meet[L](a,b)\in F$.
\end{proposition}
\begin{proof}
    Follows by \obv{lattice_filter}.
\end{proof}

\begin{proposition}\label{lattice_filter_upward}
    Suppose $\latticeFilter{F}{L}$.
    Let $a\in F$ and $b\in\carrier[L]$.
    Suppose $a\mathrel{\leq[L]}b$.
    Then $b\in F$.
\end{proposition}
\begin{proof}
    Follows by \obv{lattice_filter}.
\end{proof}

\begin{proposition}\label{lattice_filter_implies_bottom_ne_top}
    Suppose $\latticeFilter{F}{L}$.
    Then $\latticeBottom[L]\neq\latticeTop[L]$.
\end{proposition}
\begin{proof}
    $\latticeTop[L]\in F$ and $\latticeBottom[L]\notin F$ by
        \obv{lattice_filter_top,lattice_filter_bottom_notin}.
\end{proof}

\begin{proposition}\label{lattice_filter_meet_in_iff}
    Suppose $\latticeFilter{F}{L}$.
    Let $a,b\in\carrier[L]$.
    Then $\meet[L](a,b)\in F$ iff $a\in F$ and $b\in F$.
\end{proposition}
\begin{proof}
    Hence $L$ is a lattice by \obv{bounded_lattice_is_lattice}.
    Show that if $\meet[L](a,b)\in F$, then $a\in F$ and $b\in F$.
    \begin{subproof}
        Assume $\meet[L](a,b)\in F$.
        $\meet[L](a,b)\mathrel{\leq[L]}a$ and
            $\meet[L](a,b)\mathrel{\leq[L]}b$ by \obv{meet_lb}.
        Hence $a,b\in F$ by \obv{lattice_filter_upward}.
    \end{subproof}
    Show that if $a\in F$ and $b\in F$, then
        $\meet[L](a,b)\in F$.
    \begin{subproof}
        Follows by \obv{lattice_filter_meet}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{proposition}\label{lattice_filter_no_bottom_meet}
    Suppose $\latticeFilter{F}{L}$.
    Let $a,b\in F$.
    Then $\meet[L](a,b)\neq\latticeBottom[L]$.
\end{proposition}
\begin{proof}
    $\meet[L](a,b)\in F$ by \obv{lattice_filter_meet}.
    $\latticeBottom[L]\notin F$ by
        \obv{lattice_filter_bottom_notin}.
\end{proof}

\subsection{Principal lattice filters}

\begin{definition}\label{principal_lattice_filter}
    $\principalLatticeFilter{L}{a}=\upward{\leq[L]}{a}$.
\end{definition}

\begin{proposition}\label{principal_lattice_filter_iff}
    Let $L$ be a bounded lattice.
    Let $a,x\in\carrier[L]$.
    Then $x\in\principalLatticeFilter{L}{a}$ iff
        $a\mathrel{\leq[L]}x$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{principal_lattice_filter,upward_closure_iff}.
\end{proof}

\begin{proposition}\label{principal_lattice_filter_subseteq_carrier}
    Let $L$ be a bounded lattice.
    Then $\principalLatticeFilter{L}{a}\subseteq\carrier[L]$.
\end{proposition}
\begin{proof}
    Hence $\leq[L]$ is a binary relation on $\carrier[L]$ by
        \obv{quasiorder_type}.
    $\upward{\leq[L]}{a}\subseteq\carrier[L]$ by
        \obv{binary_relation_upward_subseteq}.
    Follows by \obv{principal_lattice_filter}.
\end{proof}

\begin{proposition}\label{principal_lattice_filter_contains_generator}
    Let $L$ be a bounded lattice.
    Let $a\in\carrier[L]$.
    Then $a\in\principalLatticeFilter{L}{a}$.
\end{proposition}
\begin{proof}
    Hence $a\mathrel{\leq[L]}a$ by
        \auto{ordered_set_order_reflexive,reflexive_on}.
    Follows by \obv{principal_lattice_filter_iff}.
\end{proof}

\begin{proposition}\label{principal_lattice_filter_contains_top}
    Let $L$ be a bounded lattice.
    Let $a\in\carrier[L]$.
    Then $\latticeTop[L]\in\principalLatticeFilter{L}{a}$.
\end{proposition}
\begin{proof}
    $a\mathrel{\leq[L]}\latticeTop[L]$ by
        \obv{lattice_leq_top}.
    $\latticeTop[L]\in\carrier[L]$ by \obv{lattice_top_type}.
    Follows by \obv{principal_lattice_filter_iff}.
\end{proof}

\begin{theorem}\label{principal_lattice_filter_is_filter_iff_nonbottom}
    Let $L$ be a bounded lattice.
    Let $a\in\carrier[L]$.
    Then $\latticeFilter{\principalLatticeFilter{L}{a}}{L}$ iff
        $a\neq\latticeBottom[L]$.
\end{theorem}
\begin{proof}
    Let $P=\principalLatticeFilter{L}{a}$.
    Hence $L$ is an ordered set and $L$ is a meet semilattice by
        \obv{lattice_is_ordered_set,lattice_is_meet_semilattice}.
    $\latticeBottom[L],\latticeTop[L]\in\carrier[L]$ by
        \obv{lattice_bottom_type,lattice_top_type}.
    $P\subseteq\carrier[L]$ by
        \auto{principal_lattice_filter_subseteq_carrier}.
    $\latticeTop[L]\in P$ by
        \obv{principal_lattice_filter_contains_top}.

    Show that if $\latticeFilter{P}{L}$, then
        $a\neq\latticeBottom[L]$.
    \begin{subproof}
        Assume $\latticeFilter{P}{L}$.
        $a\in P$ by
            \obv{principal_lattice_filter_contains_generator}.
        $\latticeBottom[L]\notin P$ by
            \obv{lattice_filter_bottom_notin}.
    \end{subproof}

    Show that if $a\neq\latticeBottom[L]$, then
        $\latticeFilter{P}{L}$.
    \begin{subproof}
        Assume $a\neq\latticeBottom[L]$.
        Show that $\latticeBottom[L]\notin P$.
        \begin{subproof}
            Assume $\latticeBottom[L]\in P$.
            Hence $a\mathrel{\leq[L]}\latticeBottom[L]$ by
                \obv{principal_lattice_filter_iff}.
            $\latticeBottom[L]\mathrel{\leq[L]}a$ by
                \obv{lattice_bottom_leq}.
            Hence $a=\latticeBottom[L]$ by
                \auto{ordered_set_order_antisymmetric,antisymmetric}.
            Contradiction.
        \end{subproof}
        Show that for all $x,y\in P$ we have $\meet[L](x,y)\in P$.
        \begin{subproof}
            Fix $x,y\in P$.
            Hence $x,y\in\carrier[L]$ by \obv{subseteq}.
            Hence $a\mathrel{\leq[L]}x$ and
                $a\mathrel{\leq[L]}y$ by
                \obv{principal_lattice_filter_iff}.
            Hence $a\mathrel{\leq[L]}\meet[L](x,y)$ by
                \obv{meet_glb}.
            $\meet[L](x,y)\in\carrier[L]$ by \obv{meet_type}.
            Follows by \obv{principal_lattice_filter_iff}.
        \end{subproof}
        Show that for every $x\in P$ we have
            for every $y\in\carrier[L]$ such that
                $x\mathrel{\leq[L]}y$ we have $y\in P$.
        \begin{subproof}
            Fix $x\in P$ and $y\in\carrier[L]$.
            Assume $x\mathrel{\leq[L]}y$.
            Hence $x\in\carrier[L]$ by \obv{subseteq}.
            \begin{align*}
                a
                    &\mathrel{\leq[L]} x
                        \explanation{by \obv{principal_lattice_filter_iff}}
                        \\
                    &\mathrel{\leq[L]} y
                        \explanation{by assumption}
            \end{align*}
            Follows by \obv{principal_lattice_filter_iff}.
        \end{subproof}
        Follows by \obv{lattice_filter}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{theorem}\label{principal_lattice_filter_is_filter}
    Let $L$ be a bounded lattice.
    Let $a\in\carrier[L]$.
    Suppose $a\neq\latticeBottom[L]$.
    Then $\latticeFilter{\principalLatticeFilter{L}{a}}{L}$.
\end{theorem}
\begin{proof}
    Follows by
        \obv{principal_lattice_filter_is_filter_iff_nonbottom}.
\end{proof}

\begin{theorem}\label{principal_lattice_filter_least}
    Suppose $\latticeFilter{F}{L}$ and $a\in F$.
    Then $\principalLatticeFilter{L}{a}\subseteq F$.
\end{theorem}
\begin{proof}
    $L$ is a bounded lattice by \obv{lattice_filter_ambient}.
    $F\subseteq\carrier[L]$ by
        \obv{lattice_filter_subseteq_carrier}.
    Hence $a\in\carrier[L]$ by \obv{subseteq}.
    Show that for every $x\in\principalLatticeFilter{L}{a}$ we have
        $x\in F$.
    \begin{subproof}
        Fix $x\in\principalLatticeFilter{L}{a}$.
        $x\in\carrier[L]$ by
            \auto{principal_lattice_filter_subseteq_carrier,subseteq}.
        $a\mathrel{\leq[L]}x$ by
            \obv{principal_lattice_filter_iff}.
        Follows by \obv{lattice_filter_upward}.
    \end{subproof}
    Follows by \obv{subseteq}.
\end{proof}

\subsection{Adjoining an element}

\begin{definition}\label{lattice_filter_adjoin}
    $\latticeFilterAdjoin{F}{L}{a}=
        \{b\in\carrier[L]\mid
            \text{there exists $c\in F$ such that
                $\meet[L](c,a)\mathrel{\leq[L]}b$}\}$.
\end{definition}

\begin{proposition}\label{lattice_filter_adjoin_iff}
    $b\in\latticeFilterAdjoin{F}{L}{a}$ iff
        $b\in\carrier[L]$ and
        there exists $c\in F$ such that
            $\meet[L](c,a)\mathrel{\leq[L]}b$.
\end{proposition}
\begin{proof}
    Follows by \obv{lattice_filter_adjoin}.
\end{proof}

\begin{proposition}\label{lattice_filter_adjoin_subseteq_carrier}
    $\latticeFilterAdjoin{F}{L}{a}\subseteq\carrier[L]$.
\end{proposition}
\begin{proof}
    Follows by \obv{subseteq,lattice_filter_adjoin_iff}.
\end{proof}

\begin{proposition}\label{lattice_filter_adjoin_contains_filter}
    Suppose $\latticeFilter{F}{L}$ and $a\in\carrier[L]$.
    Then $F\subseteq\latticeFilterAdjoin{F}{L}{a}$.
\end{proposition}
\begin{proof}
    $F\subseteq\carrier[L]$ by
        \obv{lattice_filter_subseteq_carrier}.
    Show that for every $c\in F$ we have
        $c\in\latticeFilterAdjoin{F}{L}{a}$.
    \begin{subproof}
        Fix $c\in F$.
        Hence $c\in\carrier[L]$ by \obv{subseteq}.
        $\meet[L](c,a)\mathrel{\leq[L]}c$ by \auto{meet_lb}.
        Follows by \obv{lattice_filter_adjoin_iff}.
    \end{subproof}
    Follows by \obv{subseteq}.
\end{proof}

\begin{proposition}\label{lattice_filter_adjoin_contains_element}
    Suppose $\latticeFilter{F}{L}$ and $a\in\carrier[L]$.
    Then $a\in\latticeFilterAdjoin{F}{L}{a}$.
\end{proposition}
\begin{proof}
    Hence $L$ is a lattice by \obv{bounded_lattice_is_lattice}.
    $\latticeTop[L]\in F$ by \obv{lattice_filter_top}.
    $\latticeTop[L]\in\carrier[L]$ by \obv{lattice_top_type}.
    $\meet[L](a,\latticeTop[L])=a$ by \obv{meet_lattice_top}.
    $\meet[L](\latticeTop[L],a)
        =\meet[L](a,\latticeTop[L])$ by
        \obv{meet_commutative}.
    Hence $\meet[L](\latticeTop[L],a)=a$.
    $a\mathrel{\leq[L]}a$ by
        \auto{lattice_is_ordered_set,
            ordered_set_order_reflexive,reflexive_on}.
    Follows by \obv{lattice_filter_adjoin_iff}.
\end{proof}

\begin{proposition}\label{lattice_filter_adjoin_top}
    Suppose $\latticeFilter{F}{L}$ and $a\in\carrier[L]$.
    Then $\latticeTop[L]\in\latticeFilterAdjoin{F}{L}{a}$.
\end{proposition}
\begin{proof}
    $F\subseteq\latticeFilterAdjoin{F}{L}{a}$ by
        \obv{lattice_filter_adjoin_contains_filter}.
    $\latticeTop[L]\in F$ by \obv{lattice_filter_top}.
    Follows by \obv{subseteq}.
\end{proof}

\begin{proposition}\label{lattice_filter_adjoin_upward}
    Suppose $\latticeFilter{F}{L}$ and $a\in\carrier[L]$.
    Let $b\in\latticeFilterAdjoin{F}{L}{a}$ and
        $d\in\carrier[L]$.
    Suppose $b\mathrel{\leq[L]}d$.
    Then $d\in\latticeFilterAdjoin{F}{L}{a}$.
\end{proposition}
\begin{proof}
    Hence $L$ is a lattice and $L$ is an ordered set by
        \obv{bounded_lattice_is_lattice,lattice_is_ordered_set}.
    Take $c\in F$ such that
        $\meet[L](c,a)\mathrel{\leq[L]}b$ by
        \obv{lattice_filter_adjoin_iff}.
    Hence $\meet[L](c,a)\mathrel{\leq[L]}d$ by
        \auto{ordered_set_order_transitive,transitive}.
    Follows by \obv{lattice_filter_adjoin_iff}.
\end{proof}

\begin{theorem}\label{lattice_filter_adjoin_meet_closed}
    Suppose $\latticeFilter{F}{L}$ and $a\in\carrier[L]$.
    Let $b,d\in\latticeFilterAdjoin{F}{L}{a}$.
    Then $\meet[L](b,d)\in\latticeFilterAdjoin{F}{L}{a}$.
\end{theorem}
\begin{proof}
    Hence $L$ is a lattice by \obv{bounded_lattice_is_lattice}.
    Hence $L$ is an ordered set and $L$ is a meet semilattice by
        \obv{lattice_is_ordered_set,lattice_is_meet_semilattice}.
    $F\subseteq\carrier[L]$ by
        \obv{lattice_filter_subseteq_carrier}.
    $b,d\in\carrier[L]$ by
        \obv{lattice_filter_adjoin_subseteq_carrier,subseteq}.
    Take $c\in F$ such that
        $\meet[L](c,a)\mathrel{\leq[L]}b$ by
        \obv{lattice_filter_adjoin_iff}.
    Take $e\in F$ such that
        $\meet[L](e,a)\mathrel{\leq[L]}d$ by
        \obv{lattice_filter_adjoin_iff}.
    Hence $c,e\in\carrier[L]$ by \obv{subseteq}.
    Let $w=\meet[L](c,e)$.
    $w\in F$ by \auto{lattice_filter_meet}.
    Hence $w\in\carrier[L]$ by \obv{subseteq}.
    $w\mathrel{\leq[L]}c$ and $w\mathrel{\leq[L]}e$ by
        \auto{meet_lb}.
    Hence $\meet[L](w,a)\mathrel{\leq[L]}\meet[L](c,a)$ and
        $\meet[L](w,a)\mathrel{\leq[L]}\meet[L](e,a)$ by
        \obv{meet_monotone_left}.
    Hence $\meet[L](w,a)\mathrel{\leq[L]}b$ and
        $\meet[L](w,a)\mathrel{\leq[L]}d$ by
        \auto{ordered_set_order_transitive,transitive}.
    $\meet[L](w,a),\meet[L](b,d)\in\carrier[L]$ by
        \obv{meet_type}.
    Hence $\meet[L](w,a)\mathrel{\leq[L]}\meet[L](b,d)$ by
        \obv{leq_meet_iff}.
    Follows by \obv{lattice_filter_adjoin_iff}.
\end{proof}

\begin{theorem}\label{lattice_filter_adjoin_least}
    Suppose $\latticeFilter{F}{L}$ and $a\in\carrier[L]$.
    Suppose $\latticeFilter{G}{L}$ and $F\subseteq G$ and $a\in G$.
    Then $\latticeFilterAdjoin{F}{L}{a}\subseteq G$.
\end{theorem}
\begin{proof}
    Show that for every $b\in\latticeFilterAdjoin{F}{L}{a}$ we have
        $b\in G$.
    \begin{subproof}
        Fix $b\in\latticeFilterAdjoin{F}{L}{a}$.
        Take $c\in F$ such that
            $\meet[L](c,a)\mathrel{\leq[L]}b$ by
            \obv{lattice_filter_adjoin_iff}.
        Hence $c\in G$ by \obv{subseteq}.
        Hence $\meet[L](c,a)\in G$ by
            \obv{lattice_filter_meet}.
        $b\in\carrier[L]$ by
            \obv{lattice_filter_adjoin_subseteq_carrier,subseteq}.
        Follows by \obv{lattice_filter_upward}.
    \end{subproof}
    Follows by \obv{subseteq}.
\end{proof}

\begin{theorem}\label{lattice_filter_adjoin_eq_self_iff}
    Suppose $\latticeFilter{F}{L}$ and $a\in\carrier[L]$.
    Then $a\in F$ iff $\latticeFilterAdjoin{F}{L}{a}=F$.
\end{theorem}
\begin{proof}
    $F\subseteq\latticeFilterAdjoin{F}{L}{a}$ by
        \obv{lattice_filter_adjoin_contains_filter}.
    Show that if $a\in F$, then
        $\latticeFilterAdjoin{F}{L}{a}=F$.
    \begin{subproof}
        Assume $a\in F$.
        $\latticeFilterAdjoin{F}{L}{a}\subseteq F$ by
            \obv{lattice_filter_adjoin_least,subseteq_refl}.
        Follows by \obv{subseteq_antisymmetric}.
    \end{subproof}
    Show that if $\latticeFilterAdjoin{F}{L}{a}=F$, then $a\in F$.
    \begin{subproof}
        Assume $\latticeFilterAdjoin{F}{L}{a}=F$.
        $a\in\latticeFilterAdjoin{F}{L}{a}$ by
            \obv{lattice_filter_adjoin_contains_element}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{theorem}\label{lattice_filter_adjoin_bottom_iff}
    Suppose $\latticeFilter{F}{L}$ and $a\in\carrier[L]$.
    Then $\latticeBottom[L]\in\latticeFilterAdjoin{F}{L}{a}$ iff
        there exists $c\in F$ such that
            $\meet[L](c,a)=\latticeBottom[L]$.
\end{theorem}
\begin{proof}
    $L$ is a bounded lattice by \obv{lattice_filter_ambient}.
    $F\subseteq\carrier[L]$ by
        \obv{lattice_filter_subseteq_carrier}.
    $\latticeBottom[L]\in\carrier[L]$ by
        \obv{lattice_bottom_type}.
    Show that if
        $\latticeBottom[L]\in\latticeFilterAdjoin{F}{L}{a}$,
        then there exists $c\in F$ such that
            $\meet[L](c,a)=\latticeBottom[L]$.
    \begin{subproof}
        Assume $\latticeBottom[L]\in
            \latticeFilterAdjoin{F}{L}{a}$.
        Take $c\in F$ such that
            $\meet[L](c,a)\mathrel{\leq[L]}\latticeBottom[L]$ by
            \obv{lattice_filter_adjoin_iff}.
        Hence $c\in\carrier[L]$ by \obv{subseteq}.
        $\meet[L](c,a)\in\carrier[L]$ by \obv{meet_type}.
        $\latticeBottom[L]\mathrel{\leq[L]}\meet[L](c,a)$ by
            \obv{lattice_bottom_leq}.
        Hence $\meet[L](c,a)=\latticeBottom[L]$ by
            \auto{ordered_set_order_antisymmetric,antisymmetric}.
    \end{subproof}
    Show that if there exists $c\in F$ such that
        $\meet[L](c,a)=\latticeBottom[L]$, then
        $\latticeBottom[L]\in\latticeFilterAdjoin{F}{L}{a}$.
    \begin{subproof}
        Assume that there exists $c\in F$ such that
            $\meet[L](c,a)=\latticeBottom[L]$.
        Take $c\in F$ such that
            $\meet[L](c,a)=\latticeBottom[L]$.
        $\latticeBottom[L]\mathrel{\leq[L]}\latticeBottom[L]$ by
            \obv{lattice_is_ordered_set,
                ordered_set_order_reflexive,reflexive_on}.
        Hence $\meet[L](c,a)
            \mathrel{\leq[L]}\latticeBottom[L]$.
        Follows by \obv{lattice_filter_adjoin_iff}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{theorem}\label{lattice_filter_adjoin_is_filter_iff}
    Suppose $\latticeFilter{F}{L}$ and $a\in\carrier[L]$.
    Then $\latticeFilter{\latticeFilterAdjoin{F}{L}{a}}{L}$ iff
        for every $c\in F$ we have
            $\meet[L](c,a)\neq\latticeBottom[L]$.
\end{theorem}
\begin{proof}
    Let $P=\latticeFilterAdjoin{F}{L}{a}$.
    $P\subseteq\carrier[L]$ by
        \auto{lattice_filter_adjoin_subseteq_carrier}.
    $\latticeTop[L]\in P$ by \obv{lattice_filter_adjoin_top}.
    For all $b,d\in P$ we have $\meet[L](b,d)\in P$ by
        \obv{lattice_filter_adjoin_meet_closed}.
    For every $b\in P$ we have for every $d\in\carrier[L]$ such that
        $b\mathrel{\leq[L]}d$ we have $d\in P$ by
        \obv{lattice_filter_adjoin_upward}.

    Show that if $\latticeFilter{P}{L}$, then
        for every $c\in F$ we have
            $\meet[L](c,a)\neq\latticeBottom[L]$.
    \begin{subproof}
        Assume $\latticeFilter{P}{L}$.
        $\latticeBottom[L]\notin P$ by
            \obv{lattice_filter_bottom_notin}.
        Show that for every $c\in F$ we have
            $\meet[L](c,a)\neq\latticeBottom[L]$.
        \begin{subproof}
            Fix $c\in F$.
            Assume $\meet[L](c,a)=\latticeBottom[L]$.
            Hence $\latticeBottom[L]\in P$ by
                \auto{lattice_filter_adjoin_bottom_iff}.
            Contradiction.
        \end{subproof}
    \end{subproof}
    Show that if for every $c\in F$ we have
        $\meet[L](c,a)\neq\latticeBottom[L]$, then
        $\latticeFilter{P}{L}$.
    \begin{subproof}
        Assume for every $c\in F$ we have
            $\meet[L](c,a)\neq\latticeBottom[L]$.
        $\latticeBottom[L]\notin P$ by
            \auto{lattice_filter_adjoin_bottom_iff}.
        Follows by \obv{lattice_filter}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\subsection{Maximal lattice filters}

\begin{definition}\label{maximal_lattice_filter}
    $\maximalLatticeFilter{F}{L}$ iff
        $\latticeFilter{F}{L}$ and
        there exists no $G$ such that
            $\latticeFilter{G}{L}$ and $F\subset G$.
\end{definition}

\begin{proposition}\label{maximal_lattice_filter_ambient}
    Suppose $\maximalLatticeFilter{F}{L}$.
    Then $L$ is a bounded lattice.
\end{proposition}
\begin{proof}
    $\latticeFilter{F}{L}$ by \obv{maximal_lattice_filter}.
    Follows by \obv{lattice_filter_ambient}.
\end{proof}

\begin{proposition}\label{maximal_lattice_filter_is_filter}
    Suppose $\maximalLatticeFilter{F}{L}$.
    Then $\latticeFilter{F}{L}$.
\end{proposition}
\begin{proof}
    Follows by \obv{maximal_lattice_filter}.
\end{proof}

\begin{proposition}\label{maximal_lattice_filter_no_strict_extension}
    Suppose $\maximalLatticeFilter{F}{L}$.
    Then there exists no $G$ such that
        $\latticeFilter{G}{L}$ and $F\subset G$.
\end{proposition}
\begin{proof}
    Follows by \obv{maximal_lattice_filter}.
\end{proof}

\begin{theorem}\label{maximal_lattice_filter_extension_eq}
    Suppose $\maximalLatticeFilter{F}{L}$ and
        $\latticeFilter{G}{L}$ and $F\subseteq G$.
    Then $G=F$.
\end{theorem}
\begin{proof}
    $F\not\subset G$ by
        \auto{maximal_lattice_filter_no_strict_extension}.
    Follows by \auto{subset}.
\end{proof}

\begin{theorem}\label{maximal_lattice_filter_iff}
    $\maximalLatticeFilter{F}{L}$ iff
        $\latticeFilter{F}{L}$ and
        for every $G$ such that $\latticeFilter{G}{L}$ and
            $F\subseteq G$ we have $G=F$.
\end{theorem}
\begin{proof}
    Show that if $\maximalLatticeFilter{F}{L}$, then
        $\latticeFilter{F}{L}$ and
        for every $G$ such that $\latticeFilter{G}{L}$ and
            $F\subseteq G$ we have $G=F$.
    \begin{subproof}
        Assume $\maximalLatticeFilter{F}{L}$.
        $\latticeFilter{F}{L}$ by
            \obv{maximal_lattice_filter_is_filter}.
        For every $G$ such that $\latticeFilter{G}{L}$ and
            $F\subseteq G$ we have $G=F$ by
            \obv{maximal_lattice_filter_extension_eq}.
    \end{subproof}
    Show that if $\latticeFilter{F}{L}$ and
        for every $G$ such that $\latticeFilter{G}{L}$ and
            $F\subseteq G$ we have $G=F$, then
        $\maximalLatticeFilter{F}{L}$.
    \begin{subproof}
        Assume $\latticeFilter{F}{L}$ and
            for every $G$ such that $\latticeFilter{G}{L}$ and
                $F\subseteq G$ we have $G=F$.
        There exists no $G$ such that
            $\latticeFilter{G}{L}$ and $F\subset G$ by
            \obv{subset}.
        Follows by \obv{maximal_lattice_filter}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\subsection{Prime lattice filters}

\begin{definition}\label{prime_lattice_filter}
    $\primeLatticeFilter{F}{L}$ iff
        $\latticeFilter{F}{L}$ and
        for all $a,b\in\carrier[L]$ such that
            $\join[L](a,b)\in F$ we have $a\in F$ or $b\in F$.
\end{definition}

\begin{proposition}\label{prime_lattice_filter_is_filter}
    Suppose $\primeLatticeFilter{F}{L}$.
    Then $\latticeFilter{F}{L}$.
\end{proposition}
\begin{proof}
    Follows by \obv{prime_lattice_filter}.
\end{proof}

\begin{proposition}\label{prime_lattice_filter_join_elim}
    Suppose $\primeLatticeFilter{F}{L}$.
    Let $a,b\in\carrier[L]$.
    Suppose $\join[L](a,b)\in F$.
    Then $a\in F$ or $b\in F$.
\end{proposition}
\begin{proof}
    Follows by \obv{prime_lattice_filter}.
\end{proof}
