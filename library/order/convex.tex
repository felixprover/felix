\section{Order-convex subsets}

\import{order/interval.tex}

Order convexity isolates the common feature of intervals and rays: together
with two points, the set contains every point lying between them.

\begin{definition}\label{order_convex}
    Let $L$ be an ordered set.
    $A$ is order convex in $L$ iff
        $A\subseteq\carrier[L]$ and for all $x,z\in A$ we have
        for every $y\in\carrier[L]$ we have if
            $x\mathrel{\leq[L]}y$ and $y\mathrel{\leq[L]}z$, then
            $y\in A$.
\end{definition}

\begin{proposition}\label{order_closed_interval_is_order_convex}
    Let $L$ be an ordered set.
    Let $a,b\in\carrier[L]$.
    Then $\orderClosedInterval{L}{a}{b}$ is order convex in $L$.
\end{proposition}
\begin{proof}
    $\orderClosedInterval{L}{a}{b}\subseteq\carrier[L]$ by
        \obv{order_interval_and_ray_types}.
    Show that for all $x,z\in\orderClosedInterval{L}{a}{b}$ we have
        for every $y\in\carrier[L]$ we have if
            $x\mathrel{\leq[L]}y$ and $y\mathrel{\leq[L]}z$, then
            $y\in\orderClosedInterval{L}{a}{b}$.
    \begin{subproof}
        Fix $x,z\in\orderClosedInterval{L}{a}{b}$.
        Fix $y\in\carrier[L]$.
        Assume $x\mathrel{\leq[L]}y$ and $y\mathrel{\leq[L]}z$.
        \begin{align*}
            a
                &\mathrel{\leq[L]} x
                    \explanation{by \obv{order_closed_interval}}
                    \\
                &\mathrel{\leq[L]} y
                    \explanation{by assumption}
        \end{align*}
        \begin{align*}
            y
                &\mathrel{\leq[L]} z
                    \explanation{by assumption}
                    \\
                &\mathrel{\leq[L]} b
                    \explanation{by \obv{order_closed_interval}}
        \end{align*}
        Follows by \obv{order_closed_interval}.
    \end{subproof}
    Follows by \obv{order_convex}.
\end{proof}

\begin{proposition}\label{order_open_interval_is_order_convex}
    Let $L$ be an ordered set.
    Let $a,b\in\carrier[L]$.
    Then $\orderOpenInterval{L}{a}{b}$ is order convex in $L$.
\end{proposition}
\begin{proof}
    $\orderOpenInterval{L}{a}{b}\subseteq\carrier[L]$ by
        \obv{order_interval_and_ray_types}.
    Show that for all $x,z\in\orderOpenInterval{L}{a}{b}$ we have
        for every $y\in\carrier[L]$ we have if
            $x\mathrel{\leq[L]}y$ and $y\mathrel{\leq[L]}z$, then
            $y\in\orderOpenInterval{L}{a}{b}$.
    \begin{subproof}
        Fix $x,z\in\orderOpenInterval{L}{a}{b}$.
        Fix $y\in\carrier[L]$.
        Assume $x\mathrel{\leq[L]}y$ and $y\mathrel{\leq[L]}z$.
        $a\mathrel{\orderStrictRelation{L}}x$ and
            $z\mathrel{\orderStrictRelation{L}}b$ by
            \obv{order_open_interval}.
        $x,z\in\carrier[L]$ by \obv{order_open_interval_iff}.
        Hence $a\mathrel{\orderStrictRelation{L}}y$ by
            \obv{ordered_set_strict_le_transitive}.
        $y\mathrel{\orderStrictRelation{L}}b$ by
            \obv{ordered_set_le_strict_transitive}.
        Follows by \obv{order_open_interval}.
    \end{subproof}
    Follows by \obv{order_convex}.
\end{proof}
