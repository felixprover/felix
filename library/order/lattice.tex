\begin{struct}\label{lattice}
    A lattice $L$ is a meet semilattice and a join semilattice.
\end{struct}

\begin{proposition}\label{lattice_intro}
    Let $L$ be a meet semilattice.
    Suppose $L$ is a join semilattice.
    Then $L$ is a lattice.
\end{proposition}
\begin{proof}
    Follows by definition.
\end{proof}

\begin{proposition}\label{lattice_is_meet_semilattice}
    Let $L$ be a lattice.
    Then $L$ is a meet semilattice.
\end{proposition}
\begin{proof}
    Follows by \obv{lattice}.
\end{proof}

\begin{proposition}\label{lattice_is_join_semilattice}
    Let $L$ be a lattice.
    Then $L$ is a join semilattice.
\end{proposition}
\begin{proof}
    Follows by \obv{lattice}.
\end{proof}

\begin{proposition}\label{lattice_is_ordered_set}
    Let $L$ be a lattice.
    Then $L$ is an ordered set.
\end{proposition}
\begin{proof}
    $L$ is a meet semilattice by \obv{lattice_is_meet_semilattice}.
    Follows by \obv{meet_semilattice}.
\end{proof}

\begin{proposition}\label{meet_join_absorption}
    Let $L$ be a lattice.
    Let $x,y\in\carrier[L]$.
    Then $\meet[L](x,\join[L](x,y))=x$.
\end{proposition}
\begin{proof}
    $\join[L](x,y)\in\carrier[L]$ by
        \obv{lattice_is_join_semilattice,join_type}.
    $x\mathrel{\leq[L]}\join[L](x,y)$ by
        \auto{lattice_is_join_semilattice,join_ub}.
    Follows by
        \obv{lattice_is_meet_semilattice,leq_iff_meet_eq_left}.
\end{proof}

\begin{proposition}\label{join_meet_absorption}
    Let $L$ be a lattice.
    Let $x,y\in\carrier[L]$.
    Then $\join[L](x,\meet[L](x,y))=x$.
\end{proposition}
\begin{proof}
    $\meet[L](x,y)\in\carrier[L]$ by
        \obv{lattice_is_meet_semilattice,meet_type}.
    $\meet[L](x,y)\mathrel{\leq[L]}x$ by
        \auto{lattice_is_meet_semilattice,meet_lb}.
    $\join[L](\meet[L](x,y),x)=x$ by
        \obv{lattice_is_join_semilattice,leq_iff_join_eq_right}.
    Follows by
        \obv{lattice_is_join_semilattice,join_commutative}.
\end{proof}

\begin{proposition}\label{meet_join_absorption_commuted}
    Let $L$ be a lattice.

\end{proposition}
\begin{proof}

        \auto{meet_join_absorption,lattice_is_meet_semilattice,

\end{proof}

\begin{proposition}\label{meet_is_infimum_pair}
    Let $L$ be a lattice.
    Let $x,y\in\carrier[L]$.
    Then $\meet[L](x,y)$ is an infimum of $\{x,y\}$ in $L$.
\end{proposition}
\begin{proof}
    $L$ is a meet semilattice by \obv{lattice_is_meet_semilattice}.
    $L$ is an ordered set by \obv{lattice_is_ordered_set}.
    $\leq[L]$ is an order on $\carrier[L]$ by
        \obv{ordered_set_order_on_carrier}.
    $\meet[L](x,y)\in\carrier[L]$ by \obv{meet_type}.
    For all $z\in\{x,y\}$ we have
        $\meet[L](x,y)\mathrel{\leq[L]}z$ by
        \auto{upair_iff,meet_lb}.
    Show for all $a\in\carrier[L]$ such that
        for all $z\in\{x,y\}$ we have $a\mathrel{\leq[L]}z$
        we have $a\mathrel{\leq[L]}\meet[L](x,y)$.
    \begin{subproof}
        Fix $a\in\carrier[L]$.
        Assume for all $z\in\{x,y\}$ we have
            $a\mathrel{\leq[L]}z$.
        $a\mathrel{\leq[L]}x$ and $a\mathrel{\leq[L]}y$ by
            \auto{upair_iff}.
        Follows by \obv{meet_glb}.
    \end{subproof}
    Follows by
        \auto{infimum_element,greatest_lower_bound_iff_order_on}.
\end{proof}

\begin{proposition}\label{join_is_supremum_pair}
    Let $L$ be a lattice.
    Let $x,y\in\carrier[L]$.
    Then $\join[L](x,y)$ is a supremum of $\{x,y\}$ in $L$.
\end{proposition}
\begin{proof}
    $L$ is a join semilattice by \obv{lattice_is_join_semilattice}.
    $L$ is an ordered set by \obv{lattice_is_ordered_set}.
    $\leq[L]$ is an order on $\carrier[L]$ by
        \obv{ordered_set_order_on_carrier}.
    $\join[L](x,y)\in\carrier[L]$ by \obv{join_type}.
    For all $z\in\{x,y\}$ we have
        $z\mathrel{\leq[L]}\join[L](x,y)$ by
        \auto{upair_iff,join_ub}.
    Show for all $a\in\carrier[L]$ such that
        for all $z\in\{x,y\}$ we have $z\mathrel{\leq[L]}a$
        we have $\join[L](x,y)\mathrel{\leq[L]}a$.
    \begin{subproof}
        Fix $a\in\carrier[L]$.
        Assume for all $z\in\{x,y\}$ we have
            $z\mathrel{\leq[L]}a$.
        $x\mathrel{\leq[L]}a$ and $y\mathrel{\leq[L]}a$ by
            \auto{upair_iff}.
        Follows by \obv{join_lub}.
    \end{subproof}
    Follows by
        \auto{supremum_element,least_upper_bound_iff_order_on}.
\end{proof}

\begin{struct}\label{bounded_lattice}
    A bounded lattice $L$ is a lattice equipped with
    \begin{enumerate}
        \item\label{lattice_bottom_field}
            a bottom element $\latticeBottom[L]\in\carrier[L]$;
        \item\label{lattice_top_field}
            a top element $\latticeTop[L]\in\carrier[L]$;
    \end{enumerate}
    such that
    \begin{enumerate}
        \item\label{lattice_bottom_law}
            for all $x\in\carrier[L]$ we have
                $\latticeBottom[L]\mathrel{\leq[L]}x$.
        \item\label{lattice_top_law}
            for all $x\in\carrier[L]$ we have
                $x\mathrel{\leq[L]}\latticeTop[L]$.
    \end{enumerate}
\end{struct}

\begin{proposition}\label{bounded_lattice_is_lattice}
    Let $L$ be a bounded lattice.
    Then $L$ is a lattice.
\end{proposition}
\begin{proof}
    Follows by \obv{bounded_lattice}.
\end{proof}

            ordered_set_order_antisymmetric,antisymmetric}.

\begin{struct}\label{distributive_lattice}
    A distributive lattice $L$ is a lattice such that
    \begin{enumerate}
        \item\label{meet_distributes_over_join}
            for all $x,y,z\in\carrier[L]$ we have
                $\meet[L](x,\join[L](y,z))
                    =\join[L](\meet[L](x,y),\meet[L](x,z))$.
    \end{enumerate}
\end{struct}

\begin{theorem}\label{join_distributes_over_meet}
    Let $L$ be a distributive lattice.
    Let $x,y,z\in\carrier[L]$.
    Then $\join[L](x,\meet[L](y,z))
        =\meet[L](\join[L](x,y),\join[L](x,z))$.
\end{theorem}
\begin{proof}
    $L$ is a lattice by \obv{distributive_lattice}.
    Hence $L$ is a meet semilattice and $L$ is a join semilattice by
        \obv{lattice_is_meet_semilattice,lattice_is_join_semilattice}.
    Let $a=\join[L](x,y)$.
    Let $b=\join[L](x,z)$.
    $a,b,\meet[L](a,x),\meet[L](a,z),
        \meet[L](z,x),\meet[L](z,y),\meet[L](y,z)
        \in\carrier[L]$ by \auto{join_type,meet_type}.
    $\meet[L](a,b)
        =\join[L](\meet[L](a,x),\meet[L](a,z))$ by
        \auto{meet_distributes_over_join}.
    $\meet[L](a,x)=x$ by
        \auto{meet_join_absorption_commuted}.
    $\meet[L](a,z)=\meet[L](z,a)$ by \obv{meet_commutative}.
    $\meet[L](z,a)
        =\join[L](\meet[L](z,x),\meet[L](z,y))$ by
        \auto{meet_distributes_over_join}.
    Hence $\meet[L](a,b)
        =\join[L]
            (x,\join[L](\meet[L](z,x),\meet[L](z,y)))$.
    $\join[L](x,\join[L](\meet[L](z,x),\meet[L](z,y)))
        =\join[L]
            (\join[L](x,\meet[L](z,x)),\meet[L](z,y))$ by
        \obv{join_associative}.
    $\meet[L](z,x)=\meet[L](x,z)$ by \obv{meet_commutative}.
    Hence $\join[L](x,\meet[L](z,x))=x$ by
        \auto{join_meet_absorption}.
    Thus $\meet[L](a,b)=\join[L](x,\meet[L](z,y))$.
    $\meet[L](z,y)=\meet[L](y,z)$ by \obv{meet_commutative}.
    Hence $\meet[L](a,b)=\join[L](x,\meet[L](y,z))$.
\end{proof}

\begin{theorem}\label{exact_lattice_extensionality}
    Let $A,B$ be sets.
    Suppose $A$ is an exact lattice and $B$ is an exact lattice.
    Suppose $\carrier[A]=\carrier[B]$.
    Suppose $\leq[A]=\leq[B]$.
    Suppose $\meet[A]=\meet[B]$.
    Suppose $\join[A]=\join[B]$.
    Then $A=B$.
\end{theorem}
\begin{proof}
    Follows by structure extensionality.
\end{proof}

\begin{theorem}\label{exact_lattice_reduct_eq}

    Then $\reduct[lattice]{L}=L$.
\end{theorem}
\begin{proof}
    Follows by \obv{latticereductidentity}.
\end{proof}

\begin{theorem}\label{lattice_reduct_is_exact_lattice}
    Let $L$ be a lattice.
    Then $\reduct[lattice]{L}$ is an exact lattice.
\end{theorem}
\begin{proof}
    $\carrier[\reduct[lattice]{L}]=\carrier[L]$ by
        \obv{latticereductprojection_carrier}.
    $\leq[\reduct[lattice]{L}]=\leq[L]$ by
        \obv{latticereductprojection_leq}.
    $\leq[\reduct[lattice]{L}]$ is an order on
        $\carrier[\reduct[lattice]{L}]$ by
        \obv{ordered_set_order_on_carrier}.
    $\leq[\reduct[lattice]{L}]$ is antisymmetric.
    $L$ is a join semilattice by \obv{lattice_is_join_semilattice}.
    $\join[\reduct[lattice]{L}]=\join[L]$ by
        \obv{latticereductprojection_join}.
    $\join[\reduct[lattice]{L}]\in
        \funs{\carrier[\reduct[lattice]{L}]\times
            \carrier[\reduct[lattice]{L}]}{
            \carrier[\reduct[lattice]{L}]}$ by \obv{join_graph}.
    For all $x,y\in\carrier[\reduct[lattice]{L}]$ we have
        $x\mathrel{\leq[\reduct[lattice]{L}]}
            \join[\reduct[lattice]{L}](x,y)$ and
        $y\mathrel{\leq[\reduct[lattice]{L}]}
            \join[\reduct[lattice]{L}](x,y)$ by \auto{join_ub}.
    For all $x,y,a\in\carrier[\reduct[lattice]{L}]$ such that
        $x\mathrel{\leq[\reduct[lattice]{L}]}a$ and
        $y\mathrel{\leq[\reduct[lattice]{L}]}a$ we have
        $\join[\reduct[lattice]{L}](x,y)
            \mathrel{\leq[\reduct[lattice]{L}]}a$ by \auto{join_lub}.
    Hence $\reduct[lattice]{L}$ is a join semilattice by
        \auto{join_semilattice,latticeshape,latticereductexact}.
    $L$ is a meet semilattice by \obv{lattice_is_meet_semilattice}.
    $\meet[\reduct[lattice]{L}]=\meet[L]$ by
        \obv{latticereductprojection_meet}.
    $\meet[\reduct[lattice]{L}]\in
        \funs{\carrier[\reduct[lattice]{L}]\times
            \carrier[\reduct[lattice]{L}]}{
            \carrier[\reduct[lattice]{L}]}$ by \obv{meet_graph}.
    For all $x,y\in\carrier[\reduct[lattice]{L}]$ we have
        $\meet[\reduct[lattice]{L}](x,y)
            \mathrel{\leq[\reduct[lattice]{L}]}x$ and
        $\meet[\reduct[lattice]{L}](x,y)
            \mathrel{\leq[\reduct[lattice]{L}]}y$ by \auto{meet_lb}.
    For all $a,x,y\in\carrier[\reduct[lattice]{L}]$ such that
        $a\mathrel{\leq[\reduct[lattice]{L}]}x$ and
        $a\mathrel{\leq[\reduct[lattice]{L}]}y$ we have
        $a\mathrel{\leq[\reduct[lattice]{L}]}
            \meet[\reduct[lattice]{L}](x,y)$ by \auto{meet_glb}.
    Hence $\reduct[lattice]{L}$ is a meet semilattice by
        \auto{meet_semilattice,latticeshape,latticereductexact}.
    Hence $\reduct[lattice]{L}$ is a lattice by
        \obv{lattice_intro}.
    Follows by \obv{latticeshape,latticereductexact}.
\end{proof}

\begin{theorem}
    \label{bounded_lattice_reduct_is_exact_bounded_lattice}
    Let $L$ be a bounded lattice.
    Then $\reduct[bounded lattice]{L}$ is an exact bounded lattice.
\end{theorem}
\begin{proof}
    $L$ is a lattice by \obv{bounded_lattice_is_lattice}.
    Hence $L$ is an ordered set by \obv{lattice_is_ordered_set}.
    Let $R=\reduct[bounded lattice]{L}$.
    For every $x$ we have $x\in R$ iff
        $x\in\reduct[bounded lattice]{L}$ by \auto.
    Hence $R=\reduct[bounded lattice]{L}$ by set extensionality.
    $\carrier[R]=\carrier[L]$ by
        \auto{bounded_latticereductprojection_carrier}.
    $\leq[R]=\leq[L]$ by
        \auto{bounded_latticereductprojection_leq}.
    $\meet[R]=\meet[L]$ by
        \auto{bounded_latticereductprojection_meet}.
    $\join[R]=\join[L]$ by
        \auto{bounded_latticereductprojection_join}.
    $\latticeBottom[R]=\latticeBottom[L]$ by
        \auto{bounded_latticereductprojection_latticeBottom}.
    $\latticeTop[R]=\latticeTop[L]$ by
        \auto{bounded_latticereductprojection_latticeTop}.

    $\leq[R]$ is an order on $\carrier[R]$ by
        \obv{ordered_set_order_on_carrier}.
    Hence $R$ is a quasiordered set by
        \auto{quasiordered_set,bounded_latticeshape,
            bounded_latticereductexact}.
    $\leq[R]$ is antisymmetric.
    Hence $R$ is an ordered set by
        \auto{orderedset,bounded_latticeshape,
            bounded_latticereductexact}.

    $L$ is a join semilattice by
        \obv{lattice_is_join_semilattice}.
    $\join[R]\in\funs{\carrier[R]\times\carrier[R]}{\carrier[R]}$ by
        \obv{join_graph}.
    For all $x,y\in\carrier[R]$ we have
        $x\mathrel{\leq[R]}\join[R](x,y)$ and
        $y\mathrel{\leq[R]}\join[R](x,y)$ by
        \auto{join_ub}.
    For all $x,y,a\in\carrier[R]$ such that
        $x\mathrel{\leq[R]}a$ and $y\mathrel{\leq[R]}a$ we have
        $\join[R](x,y)\mathrel{\leq[R]}a$ by
        \auto{join_lub}.
    Hence $R$ is a join semilattice by
        \auto{join_semilattice,bounded_latticeshape,
            bounded_latticereductexact}.
    $L$ is a meet semilattice by
        \obv{lattice_is_meet_semilattice}.
    $\meet[R]\in\funs{\carrier[R]\times\carrier[R]}{\carrier[R]}$ by
        \obv{meet_graph}.
    For all $x,y\in\carrier[R]$ we have
        $\meet[R](x,y)\mathrel{\leq[R]}x$ and
        $\meet[R](x,y)\mathrel{\leq[R]}y$ by
        \auto{meet_lb}.
    For all $a,x,y\in\carrier[R]$ such that
        $a\mathrel{\leq[R]}x$ and $a\mathrel{\leq[R]}y$ we have
        $a\mathrel{\leq[R]}\meet[R](x,y)$ by
        \auto{meet_glb}.
    Hence $R$ is a meet semilattice by
        \auto{meet_semilattice,bounded_latticeshape,
            bounded_latticereductexact}.

    Hence $R$ is a lattice by
        \obv{lattice_intro}.

    $\latticeBottom[R],\latticeTop[R]\in\carrier[R]$ by
        \auto{lattice_bottom_type,lattice_top_type}.
    For every $x\in\carrier[R]$ we have
        $\latticeBottom[R]\mathrel{\leq[R]}x$ by
        \auto{lattice_bottom_leq}.
    For every $x\in\carrier[R]$ we have
        $x\mathrel{\leq[R]}\latticeTop[R]$ by
        \auto{lattice_leq_top}.
    Hence $R$ is a bounded lattice by
        \auto{bounded_lattice,bounded_latticeshape,
            bounded_latticereductexact}.
    Follows by
        \auto{bounded_latticeshape,bounded_latticereductexact}.
\end{proof}
