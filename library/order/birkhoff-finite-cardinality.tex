\section{Cardinality consequences of finite Birkhoff representation}

\import{order/birkhoff-finite.tex}
\import{cardinal/comparison.tex}

\begin{theorem}
    \label{finite_distributive_lattice_cardinality_eq_downsets}
    Let $L$ be a finite bounded distributive lattice.
    Then
        $\card{\carrier[L]}
            =\card{\downsets{\joinIrreduciblePoset{L}}}$.
\end{theorem}
\begin{proof}
    $\birkhoffMap{L}\in
        \bijections{\carrier[L]}
            {\downsets{\joinIrreduciblePoset{L}}}$ by
        \obv{birkhoff_map_bijection}.
    Hence $\carrier[L]\approx
        \downsets{\joinIrreduciblePoset{L}}$ by
        \obv{equinum,bijections}.
    Follows by \obv{card_eq_iff_equinumerous}.
\end{proof}

\begin{theorem}
    \label{finite_distributive_lattice_cardinality_leq_powerset_join_irreducibles}
    Let $L$ be a finite bounded distributive lattice.
    Then
        $\card{\carrier[L]}
            \subseteq\card{\pow{\joinIrreducibles{L}}}$.
\end{theorem}
\begin{proof}
    $L$ is a finite bounded lattice by
        \obv{bounded_distributive_lattice}.
    $\card{\carrier[L]}
        =\card{\downsets{\joinIrreduciblePoset{L}}}$ by
        \obv{finite_distributive_lattice_cardinality_eq_downsets}.
    $\downsets{\joinIrreduciblePoset{L}}
        \subseteq\pow{\carrier[\joinIrreduciblePoset{L}]}$ by
        \obv{downsets_subseteq_powerset}.
    $\carrier[\joinIrreduciblePoset{L}]
        =\joinIrreducibles{L}$ by
        \obv{join_irreducible_poset_carrier}.
    Hence $\downsets{\joinIrreduciblePoset{L}}
        \subseteq\pow{\joinIrreducibles{L}}$.
    Hence $\cardLe
        {\downsets{\joinIrreduciblePoset{L}}}
        {\pow{\joinIrreducibles{L}}}$ by
        \obv{subseteq_cardinal_le}.
    Hence $\card{\downsets{\joinIrreduciblePoset{L}}}
        \subseteq\card{\pow{\joinIrreducibles{L}}}$ by
        \obv{card_subseteq_iff_cardinal_le}.
\end{proof}
