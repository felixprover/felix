\section{Meet semilattices}

\begin{struct}\label{meet_semilattice}
    A meet semilattice $X$ is an ordered set
    equipped with
    \begin{enumerate}
        \item\label{meet_graph}
            a meet operation
            $\meet[X]:\carrier[X]\times\carrier[X]\to\carrier[X]$;
    \end{enumerate}
    such that
    \begin{enumerate}
        \item\label{meet_lb}
            for all $x,y\in\carrier[X]$ we have
                $\meet[X](x,y)\mathrel{\leq[X]}x$ and
                $\meet[X](x,y)\mathrel{\leq[X]}y$.
        \item\label{meet_glb}
            for all $a,x,y\in\carrier[X]$ such that
                $a\mathrel{\leq[X]}x$ and $a\mathrel{\leq[X]}y$ we have
                $a\mathrel{\leq[X]}\meet[X](x,y)$.
    \end{enumerate}
\end{struct}

\begin{proposition}\label{meet_type}
    Let $X$ be a meet semilattice.
    Let $x,y\in\carrier[X]$.
    Then $\meet[X](x,y)\in\carrier[X]$.
\end{proposition}
\begin{proof}
    $(x,y)\in\carrier[X]\times\carrier[X]$
        by \obv{times_tuple_intro}.
    Follows by \obv{meet_graph,funs_type_apply}.
\end{proof}


\begin{proposition}\label{meet_idempotent}
    Let $X$ be a meet semilattice.
    Let $x\in\carrier[X]$.
    Then $\meet[X](x,x)=x$.
\end{proposition}
\begin{proof}
    We have $\meet[X](x,x)\in\carrier[X]$ by \obv{meet_type}.
    We have $\meet[X](x,x)\mathrel{\leq[X]}x$ by \obv{meet_lb}.
    We have $x\mathrel{\leq[X]}x$
        by \auto{meet_semilattice,orderedset,quasiorder_refl,reflexive_on}.
    Thus $x\mathrel{\leq[X]}\meet[X](x,x)$ by \obv{meet_glb}.
    Follows by \auto{meet_semilattice,orderedset,orderedset_antisym,antisymmetric}.
\end{proof}

\begin{theorem}\label{meet_commutative}
    Let $X$ be a meet semilattice.
    Let $x,y\in\carrier[X]$.
    Then $\meet[X](x,y)=\meet[X](y,x)$.
\end{theorem}
\begin{proof}
    $\meet[X](x,y),\meet[X](y,x)\in\carrier[X]$
        by \obv{meet_type}.
    $\meet[X](x,y)\mathrel{\leq[X]}x$ and
        $\meet[X](x,y)\mathrel{\leq[X]}y$ and
        $\meet[X](y,x)\mathrel{\leq[X]}y$ and
        $\meet[X](y,x)\mathrel{\leq[X]}x$
        by \obv{meet_lb}.
    Hence $\meet[X](x,y)\mathrel{\leq[X]}\meet[X](y,x)$ and
        $\meet[X](y,x)\mathrel{\leq[X]}\meet[X](x,y)$
        by \obv{meet_glb}.
    Follows by
        \auto{meet_semilattice,orderedset,orderedset_antisym,antisymmetric}.
\end{proof}

\begin{theorem}\label{meet_associative}
    Let $X$ be a meet semilattice.
    Let $x,y,z\in\carrier[X]$.
    Then
        $\meet[X](x,\meet[X](y,z))
            =\meet[X](\meet[X](x,y),z)$.
\end{theorem}
\begin{proof}
    Let $l=\meet[X](x,\meet[X](y,z))$.
    Let $r=\meet[X](\meet[X](x,y),z)$.
    $\meet[X](x,y),\meet[X](y,z),l,r\in\carrier[X]$
        by \auto{meet_type}.
    $l\mathrel{\leq[X]}x$ and
        $l\mathrel{\leq[X]}\meet[X](y,z)$ and
        $\meet[X](y,z)\mathrel{\leq[X]}y$ and
        $\meet[X](y,z)\mathrel{\leq[X]}z$
        by \auto{meet_lb}.
    Hence $l\mathrel{\leq[X]}y$ and $l\mathrel{\leq[X]}z$
        by \auto{meet_semilattice,orderedset,quasiorder_tran,transitive}.
    Hence $l\mathrel{\leq[X]}\meet[X](x,y)$
        by \obv{meet_glb}.
    Hence $l\mathrel{\leq[X]}r$ by \auto{meet_glb}.

    $r\mathrel{\leq[X]}\meet[X](x,y)$ and
        $r\mathrel{\leq[X]}z$ and
        $\meet[X](x,y)\mathrel{\leq[X]}x$ and
        $\meet[X](x,y)\mathrel{\leq[X]}y$
        by \auto{meet_lb}.
    Hence $r\mathrel{\leq[X]}x$ and $r\mathrel{\leq[X]}y$
        by \auto{meet_semilattice,orderedset,quasiorder_tran,transitive}.
    Hence $r\mathrel{\leq[X]}\meet[X](y,z)$
        by \obv{meet_glb}.
    Hence $r\mathrel{\leq[X]}l$ by \auto{meet_glb}.
    Follows by
        \auto{meet_semilattice,orderedset,orderedset_antisym,antisymmetric}.
\end{proof}

\begin{proposition}\label{meet_lower_left}
    Let $X$ be a meet semilattice.
    Let $x,y\in\carrier[X]$.
    Then $\meet[X](x,y)\mathrel{\leq[X]}x$.
\end{proposition}
\begin{proof}
    Follows by \auto{meet_lb}.
\end{proof}

\begin{proposition}\label{meet_lower_right}
    Let $X$ be a meet semilattice.
    Let $x,y\in\carrier[X]$.
    Then $\meet[X](x,y)\mathrel{\leq[X]}y$.
\end{proposition}
\begin{proof}
    Follows by \auto{meet_lb}.
\end{proof}

\begin{proposition}\label{leq_meet_iff}
    Let $X$ be a meet semilattice.
    Let $a,x,y\in\carrier[X]$.
    Then $a\mathrel{\leq[X]}\meet[X](x,y)$ iff
        $a\mathrel{\leq[X]}x$ and $a\mathrel{\leq[X]}y$.
\end{proposition}
\begin{proof}
    Show that if $a\mathrel{\leq[X]}\meet[X](x,y)$, then
        $a\mathrel{\leq[X]}x$ and $a\mathrel{\leq[X]}y$.
    \begin{subproof}
        Assume $a\mathrel{\leq[X]}\meet[X](x,y)$.
        $\meet[X](x,y)\mathrel{\leq[X]}x$ and
            $\meet[X](x,y)\mathrel{\leq[X]}y$ by \obv{meet_lb}.
        Follows by
            \auto{meet_semilattice,ordered_set_order_transitive,transitive}.
    \end{subproof}
    Show that if $a\mathrel{\leq[X]}x$ and $a\mathrel{\leq[X]}y$,
        then $a\mathrel{\leq[X]}\meet[X](x,y)$.
    \begin{subproof}
        Assume $a\mathrel{\leq[X]}x$ and $a\mathrel{\leq[X]}y$.
        Follows by \obv{meet_glb}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{proposition}\label{leq_iff_meet_eq_left}
    Let $X$ be a meet semilattice.
    Let $x,y\in\carrier[X]$.
    Then $x\mathrel{\leq[X]}y$ iff $\meet[X](x,y)=x$.
\end{proposition}
\begin{proof}
    $\meet[X](x,y)\in\carrier[X]$ by \obv{meet_type}.
    Show that if $x\mathrel{\leq[X]}y$, then $\meet[X](x,y)=x$.
    \begin{subproof}
        Assume $x\mathrel{\leq[X]}y$.
        $x\mathrel{\leq[X]}x$ by
            \auto{meet_semilattice,
                ordered_set_order_reflexive,reflexive_on}.
        Hence $x\mathrel{\leq[X]}\meet[X](x,y)$ by \obv{meet_glb}.
        $\meet[X](x,y)\mathrel{\leq[X]}x$ by \obv{meet_lower_left}.
        Follows by
            \auto{meet_semilattice,
                ordered_set_order_antisymmetric,antisymmetric}.
    \end{subproof}
    Show that if $\meet[X](x,y)=x$, then $x\mathrel{\leq[X]}y$.
    \begin{subproof}
        Assume $\meet[X](x,y)=x$.
        $\meet[X](x,y)\mathrel{\leq[X]}y$ by \obv{meet_lower_right}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{proposition}\label{leq_iff_meet_eq_right}
    Let $X$ be a meet semilattice.
    Let $x,y\in\carrier[X]$.
    Then $x\mathrel{\leq[X]}y$ iff $\meet[X](y,x)=x$.
\end{proposition}
\begin{proof}
    $\meet[X](y,x)=\meet[X](x,y)$ by \obv{meet_commutative}.
    Follows by \auto{leq_iff_meet_eq_left}.
\end{proof}

\begin{proposition}\label{meet_eq_left_of_leq}
    Let $X$ be a meet semilattice.
    Let $x,y\in\carrier[X]$.
    Suppose $x\mathrel{\leq[X]}y$.
    Then $\meet[X](x,y)=x$.
\end{proposition}
\begin{proof}
    Follows by \obv{leq_iff_meet_eq_left}.
\end{proof}

\begin{proposition}\label{meet_eq_right_of_leq}
    Let $X$ be a meet semilattice.
    Let $x,y\in\carrier[X]$.
    Suppose $x\mathrel{\leq[X]}y$.
    Then $\meet[X](y,x)=x$.
\end{proposition}
\begin{proof}
    Follows by \obv{leq_iff_meet_eq_right}.
\end{proof}

\begin{proposition}\label{meet_monotone}
    Let $X$ be a meet semilattice.
    Let $x_1,x_2,y_1,y_2\in\carrier[X]$.
    Suppose $x_1\mathrel{\leq[X]}x_2$ and
        $y_1\mathrel{\leq[X]}y_2$.
    Then $\meet[X](x_1,y_1)\mathrel{\leq[X]}\meet[X](x_2,y_2)$.
\end{proposition}
\begin{proof}
    $\meet[X](x_1,y_1),\meet[X](x_2,y_2)\in\carrier[X]$ by
        \obv{meet_type}.
    $\meet[X](x_1,y_1)\mathrel{\leq[X]}x_1$ and
        $\meet[X](x_1,y_1)\mathrel{\leq[X]}y_1$ by \obv{meet_lb}.
    Hence $\meet[X](x_1,y_1)\mathrel{\leq[X]}x_2$ and
        $\meet[X](x_1,y_1)\mathrel{\leq[X]}y_2$ by
        \auto{meet_semilattice,ordered_set_order_transitive,transitive}.
    Follows by \obv{meet_glb}.
\end{proof}

\begin{proposition}\label{meet_monotone_left}
    Let $X$ be a meet semilattice.
    Let $x_1,x_2,y\in\carrier[X]$.
    Suppose $x_1\mathrel{\leq[X]}x_2$.
    Then $\meet[X](x_1,y)\mathrel{\leq[X]}\meet[X](x_2,y)$.
\end{proposition}
\begin{proof}
    $y\mathrel{\leq[X]}y$ by
        \auto{meet_semilattice,
            ordered_set_order_reflexive,reflexive_on}.
    Follows by \obv{meet_monotone}.
\end{proof}

\begin{proposition}\label{meet_monotone_right}
    Let $X$ be a meet semilattice.
    Let $x,y_1,y_2\in\carrier[X]$.
    Suppose $y_1\mathrel{\leq[X]}y_2$.
    Then $\meet[X](x,y_1)\mathrel{\leq[X]}\meet[X](x,y_2)$.
\end{proposition}
\begin{proof}
    $x\mathrel{\leq[X]}x$ by
        \auto{meet_semilattice,
            ordered_set_order_reflexive,reflexive_on}.
    Follows by \obv{meet_monotone}.
\end{proof}
