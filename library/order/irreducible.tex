\section{Join-irreducible and join-prime elements}

\import{order/lattice.tex}

\begin{definition}\label{join_irreducible}
    Let $L$ be a bounded lattice.
    $j$ is join-irreducible in $L$ iff
        $j\in\carrier[L]$ and
        $j\neq\latticeBottom[L]$ and
        for all $a,b\in\carrier[L]$ such that
            $\join[L](a,b)=j$ we have $j=a$ or $j=b$.
\end{definition}

\begin{definition}\label{join_irreducibles}
    Let $L$ be a bounded lattice.
    $\joinIrreducibles{L}=\{j\in\carrier[L]\mid
        \text{$j$ is join-irreducible in $L$}\}$.
\end{definition}

\begin{proposition}\label{join_irreducible_iff}
    Let $L$ be a bounded lattice.
    Then $j$ is join-irreducible in $L$ iff
        $j\in\carrier[L]$ and
        $j\neq\latticeBottom[L]$ and
        for all $a,b\in\carrier[L]$ such that
            $\join[L](a,b)=j$ we have $j=a$ or $j=b$.
\end{proposition}
\begin{proof}
    Follows by \auto{join_irreducible}.
\end{proof}

\begin{proposition}\label{join_irreducible_type}
    Let $L$ be a bounded lattice.
    Suppose $j$ is join-irreducible in $L$.
    Then $j\in\carrier[L]$.
\end{proposition}
\begin{proof}
    Follows by \obv{join_irreducible_iff}.
\end{proof}

\begin{proposition}\label{join_irreducible_nonbottom}
    Let $L$ be a bounded lattice.
    Suppose $j$ is join-irreducible in $L$.
    Then $j\neq\latticeBottom[L]$.
\end{proposition}
\begin{proof}
    Follows by \obv{join_irreducible_iff}.
\end{proof}

\begin{proposition}\label{join_irreducible_elim}
    Let $L$ be a bounded lattice.
    Suppose $j$ is join-irreducible in $L$.
    Let $a,b\in\carrier[L]$.
    Suppose $\join[L](a,b)=j$.
    Then $j=a$ or $j=b$.
\end{proposition}
\begin{proof}
    Follows by \obv{join_irreducible_iff}.
\end{proof}

\begin{proposition}\label{join_irreducibles_iff}
    Let $L$ be a bounded lattice.
    Then $j\in\joinIrreducibles{L}$ iff
        $j$ is join-irreducible in $L$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{join_irreducibles,join_irreducible_iff}.
\end{proof}

\begin{proposition}\label{join_irreducibles_subseteq_carrier}
    Let $L$ be a bounded lattice.
    Then $\joinIrreducibles{L}\subseteq\carrier[L]$.
\end{proposition}
\begin{proof}
    Follows by
        \obv{subseteq,join_irreducibles_iff,join_irreducible_type}.
\end{proof}

\begin{definition}\label{join_prime}
    Let $L$ be a bounded lattice.
    $j$ is join-prime in $L$ iff
        $j\in\carrier[L]$ and
        $j\neq\latticeBottom[L]$ and
        for all $a,b\in\carrier[L]$ such that
            $j\mathrel{\leq[L]}\join[L](a,b)$ we have
            $j\mathrel{\leq[L]}a$ or $j\mathrel{\leq[L]}b$.
\end{definition}

\begin{proposition}\label{join_prime_iff}

    Then $j$ is join-prime in $L$ iff

        $j\neq\latticeBottom[L]$ and

            $j\mathrel{\leq[L]}\join[L](a,b)$ we have
            $j\mathrel{\leq[L]}a$ or $j\mathrel{\leq[L]}b$.
\end{proposition}
\begin{proof}
    Follows by \auto{join_prime}.
\end{proof}

\begin{proposition}\label{join_prime_type}

\end{proposition}
\begin{proof}

\end{proof}

\begin{proposition}\label{join_prime_nonbottom}

\begin{proposition}\label{join_prime_elim}

\end{proof}

\begin{theorem}\label{join_prime_implies_join_irreducible}
    Let $L$ be a bounded lattice.
    Suppose $j$ is join-prime in $L$.
    Then $j$ is join-irreducible in $L$.
\end{theorem}
\begin{proof}
    Hence $L$ is a join semilattice and $L$ is an ordered set by
        \obv{lattice_is_join_semilattice,lattice_is_ordered_set}.
    $j\in\carrier[L]$ and $j\neq\latticeBottom[L]$ by
        \obv{join_prime_type,join_prime_nonbottom}.
    $\leq[L]$ is reflexive on $\carrier[L]$ and antisymmetric by
        \obv{ordered_set_order_reflexive,
            ordered_set_order_antisymmetric}.
    Show for all $a,b\in\carrier[L]$ such that
        $\join[L](a,b)=j$ we have $j=a$ or $j=b$.
    \begin{subproof}
        Fix $a,b\in\carrier[L]$.
        Assume $\join[L](a,b)=j$.
        $j\mathrel{\leq[L]}\join[L](a,b)$ by
            \auto{reflexive_on}.
        Hence $j\mathrel{\leq[L]}a$ or $j\mathrel{\leq[L]}b$ by
            \obv{join_prime_elim}.
        $a\mathrel{\leq[L]}j$ and $b\mathrel{\leq[L]}j$ by
            \auto{join_ub}.
        Follows by \obv{antisymmetric}.
    \end{subproof}
    Follows by \obv{join_irreducible_iff}.
\end{proof}

\begin{theorem}\label{join_irreducible_implies_join_prime}
    Let $L$ be a bounded lattice.
    Suppose $L$ is a distributive lattice.
    Suppose $j$ is join-irreducible in $L$.
    Then $j$ is join-prime in $L$.
\end{theorem}
\begin{proof}
    Hence $L$ is a meet semilattice and $L$ is a join semilattice by
        \obv{lattice_is_meet_semilattice,lattice_is_join_semilattice}.
    $j\in\carrier[L]$ and $j\neq\latticeBottom[L]$ by
        \obv{join_irreducible_type,join_irreducible_nonbottom}.
    Show for all $a,b\in\carrier[L]$ such that
        $j\mathrel{\leq[L]}\join[L](a,b)$ we have
        $j\mathrel{\leq[L]}a$ or $j\mathrel{\leq[L]}b$.
    \begin{subproof}
        Fix $a,b\in\carrier[L]$.
        Assume $j\mathrel{\leq[L]}\join[L](a,b)$.
        $\join[L](a,b),\meet[L](j,a),\meet[L](j,b)
            \in\carrier[L]$ by \obv{join_type,meet_type}.
        $\meet[L](j,\join[L](a,b))=j$ by
            \obv{leq_iff_meet_eq_left}.
        $\meet[L](j,\join[L](a,b))
            =\join[L](\meet[L](j,a),\meet[L](j,b))$ by
            \obv{meet_distributes_over_join}.
        Hence $\join[L](\meet[L](j,a),\meet[L](j,b))=j$.
        Thus $j=\meet[L](j,a)$ or $j=\meet[L](j,b)$ by
            \obv{join_irreducible_elim}.
        Follows by \auto{leq_iff_meet_eq_left}.
    \end{subproof}
    Follows by \obv{join_prime_iff}.
\end{proof}

\begin{theorem}\label{join_irreducible_iff_join_prime}
    Let $L$ be a bounded distributive lattice.
    Then $j$ is join-irreducible in $L$ iff $j$ is join-prime in $L$.
\end{theorem}
\begin{proof}
    $L$ is a bounded lattice by \obv{bounded_distributive_lattice}.
    $L$ is a distributive lattice by
        \obv{bounded_distributive_lattice_is_distributive_lattice}.
    Follows by
        \auto{join_prime_implies_join_irreducible,
            join_irreducible_implies_join_prime}.
\end{proof}

\begin{definition}\label{join_irreducibles_below}
    Let $L$ be a bounded lattice.
    $\joinIrreduciblesBelow{L}{x}
        =\{j\in\joinIrreducibles{L}\mid
            j\mathrel{\leq[L]}x\}$.
\end{definition}

\begin{proposition}\label{join_irreducibles_below_iff}
    Let $L$ be a bounded lattice.
    Then $j\in\joinIrreduciblesBelow{L}{x}$ iff
        $j$ is join-irreducible in $L$ and $j\mathrel{\leq[L]}x$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{join_irreducibles_below,join_irreducibles_iff}.
\end{proof}

\begin{proposition}\label{join_irreducibles_below_subseteq_join_irreducibles}
    Let $L$ be a bounded lattice.
    Then $\joinIrreduciblesBelow{L}{x}\subseteq\joinIrreducibles{L}$.
\end{proposition}
\begin{proof}
    Follows by \obv{join_irreducibles_below,subseteq}.
\end{proof}

\begin{proposition}\label{join_irreducibles_below_subseteq_carrier}
    Let $L$ be a bounded lattice.
    Then $\joinIrreduciblesBelow{L}{x}\subseteq\carrier[L]$.
\end{proposition}
\begin{proof}
    $\joinIrreduciblesBelow{L}{x}\subseteq\joinIrreducibles{L}$ by
        \obv{join_irreducibles_below_subseteq_join_irreducibles}.
    $\joinIrreducibles{L}\subseteq\carrier[L]$ by
        \obv{join_irreducibles_subseteq_carrier}.
    Follows by \obv{subseteq_transitive}.
\end{proof}

\begin{proposition}\label{join_irreducibles_below_monotone}
    Let $L$ be a bounded lattice.
    Let $x,y\in\carrier[L]$.
    Suppose $x\mathrel{\leq[L]}y$.
    Then $\joinIrreduciblesBelow{L}{x}
        \subseteq\joinIrreduciblesBelow{L}{y}$.
\end{proposition}
\begin{proof}
    $\leq[L]$ is transitive by \obv{ordered_set_order_transitive}.
    Show for every $j\in\joinIrreduciblesBelow{L}{x}$ we have
        $j\in\joinIrreduciblesBelow{L}{y}$.
    \begin{subproof}
        Fix $j\in\joinIrreduciblesBelow{L}{x}$.
        $j$ is join-irreducible in $L$ and $j\mathrel{\leq[L]}x$ by
            \auto{join_irreducibles_below_iff}.
        Hence $j\mathrel{\leq[L]}y$ by \obv{transitive}.
        Follows by \obv{join_irreducibles_below_iff}.
    \end{subproof}
    Follows by \obv{subseteq}.
\end{proof}
