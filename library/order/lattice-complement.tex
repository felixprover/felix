\section{Lattice complements}

\import{order/lattice.tex}

\begin{definition}\label{are_lattice_complements}
    Let $L$ be a bounded lattice.
    $\areLatticeComplements{L}{x}{y}$ iff
        $x,y\in\carrier[L]$ and
        $\meet[L](x,y)=\latticeBottom[L]$ and
        $\join[L](x,y)=\latticeTop[L]$.
\end{definition}

\begin{proposition}\label{are_lattice_complements_left_type}
    Let $L$ be a bounded lattice.
    Suppose $\areLatticeComplements{L}{x}{y}$.
    Then $x\in\carrier[L]$.
\end{proposition}
\begin{proof}
    Follows by \obv{are_lattice_complements}.
\end{proof}

\begin{proposition}\label{are_lattice_complements_right_type}
    Let $L$ be a bounded lattice.
    Suppose $\areLatticeComplements{L}{x}{y}$.
    Then $y\in\carrier[L]$.
\end{proposition}
\begin{proof}
    Follows by \obv{are_lattice_complements}.
\end{proof}

\begin{proposition}\label{are_lattice_complements_meet}
    Let $L$ be a bounded lattice.
    Suppose $\areLatticeComplements{L}{x}{y}$.
    Then $\meet[L](x,y)=\latticeBottom[L]$.
\end{proposition}
\begin{proof}
    Follows by \obv{are_lattice_complements}.
\end{proof}

\begin{proposition}\label{are_lattice_complements_join}
    Let $L$ be a bounded lattice.
    Suppose $\areLatticeComplements{L}{x}{y}$.
    Then $\join[L](x,y)=\latticeTop[L]$.
\end{proposition}
\begin{proof}
    Follows by \obv{are_lattice_complements}.
\end{proof}

\begin{proposition}\label{are_lattice_complements_symmetric}
    Let $L$ be a bounded lattice.
    Suppose $\areLatticeComplements{L}{x}{y}$.
    Then $\areLatticeComplements{L}{y}{x}$.
\end{proposition}
\begin{proof}
    $x,y\in\carrier[L]$ by
        \obv{are_lattice_complements_left_type,
            are_lattice_complements_right_type}.
    $\meet[L](y,x)=\meet[L](x,y)$ by \obv{meet_commutative}.
    Hence $\meet[L](y,x)=\latticeBottom[L]$ by
        \obv{are_lattice_complements_meet}.
    $\join[L](y,x)=\join[L](x,y)$ by \obv{join_commutative}.
    Hence $\join[L](y,x)=\latticeTop[L]$ by
        \obv{are_lattice_complements_join}.
    Follows by \obv{are_lattice_complements}.
\end{proof}

\begin{theorem}\label{lattice_complement_unique}
    Let $L$ be a bounded distributive lattice.
    Suppose $\areLatticeComplements{L}{x}{y}$ and
        $\areLatticeComplements{L}{x}{z}$.
    Then $y=z$.
\end{theorem}
\begin{proof}
    $L$ is a bounded lattice and $L$ is a distributive lattice by
        \obv{bounded_distributive_lattice_is_bounded_lattice,
            bounded_distributive_lattice_is_distributive_lattice}.
    Hence $L$ is a meet semilattice and $L$ is a join semilattice by
        \obv{lattice_is_meet_semilattice,lattice_is_join_semilattice}.
    $x,y,z\in\carrier[L]$ by
        \obv{are_lattice_complements_left_type,
            are_lattice_complements_right_type}.
    $\meet[L](y,z),\meet[L](z,y)\in\carrier[L]$ by \obv{meet_type}.

    $\meet[L](y,x)=\meet[L](x,y)$ by \obv{meet_commutative}.
    Hence $\meet[L](y,x)=\latticeBottom[L]$ by
        \obv{are_lattice_complements_meet}.
    $\join[L](x,z)=\latticeTop[L]$ by
        \obv{are_lattice_complements_join}.
    $y=\meet[L](y,\latticeTop[L])$ by \obv{meet_lattice_top}.
    Hence $y=\meet[L](y,\join[L](x,z))$.
    $\meet[L](y,\join[L](x,z))
        =\join[L](\meet[L](y,x),\meet[L](y,z))$ by
        \obv{meet_distributes_over_join}.
    Hence $y=\join[L](\latticeBottom[L],\meet[L](y,z))$.
    $\join[L](\latticeBottom[L],\meet[L](y,z))=\meet[L](y,z)$ by
        \obv{join_lattice_bottom}.
    Hence $\meet[L](y,z)=y$.
    Thus $y\mathrel{\leq[L]}z$ by \obv{leq_iff_meet_eq_left}.

    $\meet[L](x,z)=\latticeBottom[L]$ by
        \obv{are_lattice_complements_meet}.
    $\meet[L](z,x)=\meet[L](x,z)$ by \obv{meet_commutative}.
    Hence $\meet[L](z,x)=\latticeBottom[L]$.
    $\join[L](x,y)=\latticeTop[L]$ by
        \obv{are_lattice_complements_join}.
    $z=\meet[L](z,\latticeTop[L])$ by \obv{meet_lattice_top}.
    Hence $z=\meet[L](z,\join[L](x,y))$.
    $\meet[L](z,\join[L](x,y))
        =\join[L](\meet[L](z,x),\meet[L](z,y))$ by
        \obv{meet_distributes_over_join}.
    Hence $z=\join[L](\latticeBottom[L],\meet[L](z,y))$.
    $\join[L](\latticeBottom[L],\meet[L](z,y))=\meet[L](z,y)$ by
        \obv{join_lattice_bottom}.
    Hence $\meet[L](z,y)=z$.
    Thus $z\mathrel{\leq[L]}y$ by \obv{leq_iff_meet_eq_left}.

    $\leq[L]$ is antisymmetric by
        \obv{lattice_is_ordered_set,ordered_set_order_antisymmetric}.
    Follows by \obv{antisymmetric}.
\end{proof}
