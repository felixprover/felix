\section{Generated well-orders}

\import{order/well-order.tex}
\import{function.tex}
\import{set/cons.tex}

A generated well-order chooses each element from the set of all its
predecessors through an actual entry of one function graph.  Comparing two
orders generated by the same function makes their common initial part
canonical.

\begin{definition}\label{generated_well_ordering}
    $R$ is a well ordering generated by $f$ on $B$ iff
        $f$ is a function and
        $R$ is a strict well order on $B$ and
        for all $b\in B$ we have $(\downward{R}{b},b)\in f$.
\end{definition}

\begin{proposition}\label{generated_well_ordering_entry}
    Suppose $R$ is a well ordering generated by $f$ on $B$.
    Let $b\in B$.
    Then $(\downward{R}{b},b)\in f$.
\end{proposition}
\begin{proof}
    Follows by \obv{generated_well_ordering}.
\end{proof}

\begin{proposition}\label{generated_well_ordering_apply}
    Suppose $R$ is a well ordering generated by $f$ on $B$.
    Let $b\in B$.
    Then $f(\downward{R}{b})=b$.
\end{proposition}
\begin{proof}
    $f$ is a function by \obv{generated_well_ordering}.
    Follows by
        \obv{generated_well_ordering_entry,function_apply_intro}.
\end{proof}

\begin{definition}\label{common_initial_segment}
    $I$ is a common initial segment of $R$ on $B$ and $S$ on $C$ iff
        $I\subseteq B\inter C$ and
        for all $i\in I$ we have
            $\downward{R}{i} = \downward{S}{i}$ and
            $\downward{R}{i}\subseteq I$ and
            $\downward{S}{i}\subseteq I$.
\end{definition}

\begin{lemma}\label{empty_common_initial_segment}
    $\emptyset$ is a common initial segment of $R$ on $B$ and $S$ on $C$.
\end{lemma}
\begin{proof}
    Follows by \auto{common_initial_segment,emptyset_subseteq,emptyset}.
\end{proof}

\begin{lemma}\label{union_common_initial_segments}
    Suppose every element of $K$ is a common initial segment of $R$ on $B$ and $S$ on $C$.
    Then $\unions{K}$ is a common initial segment of $R$ on $B$ and $S$ on $C$.
\end{lemma}
\begin{proof}
    For all $J\in K$ we have $J\subseteq B\inter C$ by \auto{common_initial_segment}.
    Thus $\unions{K}\subseteq B\inter C$ by \obv{unions_family}.

    Show that for all $i\in\unions{K}$ we have $\downward{R}{i}=\downward{S}{i}$.
    \begin{subproof}
        Fix $i\in\unions{K}$.
        Take $J\in K$ such that $i\in J$ by \obv{unions_iff}.
        Follows by \auto{common_initial_segment}.
    \end{subproof}

    Show that for all $i\in\unions{K}$ we have $\downward{R}{i}\subseteq\unions{K}$.
    \begin{subproof}
        Fix $i\in\unions{K}$.
        Take $J\in K$ such that $i\in J$ by \obv{unions_iff}.
        $\downward{R}{i}\subseteq J$ by \auto{common_initial_segment}.
        $J\subseteq\unions{K}$ by \auto{subseteq,unions_iff}.
        Follows by \obv{subseteq_transitive}.
    \end{subproof}

    Show that for all $i\in\unions{K}$ we have $\downward{S}{i}\subseteq\unions{K}$.
    \begin{subproof}
        Fix $i\in\unions{K}$.
        Take $J\in K$ such that $i\in J$ by \obv{unions_iff}.
        $\downward{S}{i}\subseteq J$ by \auto{common_initial_segment}.
        $J\subseteq\unions{K}$ by \auto{subseteq,unions_iff}.
        Follows by \obv{subseteq_transitive}.
    \end{subproof}

    Follows by \obv{common_initial_segment}.
\end{proof}

\begin{lemma}\label{generated_well_orderings_compatible}
    Suppose $R$ is a well ordering generated by $f$ on $B$.
    Suppose $S$ is a well ordering generated by $f$ on $C$.
    Then $B$ is a common initial segment of $R$ on $B$ and $S$ on $C$
    or $C$ is a common initial segment of $R$ on $B$ and $S$ on $C$.
\end{lemma}
\begin{proof}
    Let $K = \{I\in\pow{B\inter C}\mid \text{$I$ is a common initial segment of $R$ on $B$ and $S$ on $C$}\}$.
    $\emptyset\in K$ by \obv{powerset_bottom,empty_common_initial_segment}.
    Thus $K$ is inhabited.
    Every element of $K$ is a common initial segment of $R$ on $B$ and $S$ on $C$.
    Let $I = \unions{K}$.
    $I$ is a common initial segment of $R$ on $B$ and $S$ on $C$
        by \auto{union_common_initial_segments}.

    Show $B\subseteq I$ or $C\subseteq I$.
    \begin{subproof}
        Suppose not.
        $B\setminus I\neq\emptyset$ and $C\setminus I\neq\emptyset$
            by \auto{setminus_eq_emptyset_iff_subseteq}.
        Take $b\in B\setminus I$ such that for all $x\in B\setminus I$
            we have $b\mathrel{R}x$ or $b=x$ by \auto{generated_well_ordering,strict_well_order_iff,setminus_subseteq}.
        Take $c\in C\setminus I$ such that for all $x\in C\setminus I$
            we have $c\mathrel{S}x$ or $c=x$ by \auto{generated_well_ordering,strict_well_order_iff,setminus_subseteq}.

        Show $\downward{R}{b}=I$.
        \begin{subproof}
            Every element of $\downward{R}{b}$ is an element of $I$
                by \auto{generated_well_ordering,strict_well_order_iff,subseteq,times_tuple_elim,downward_closure,setminus,asymmetric}.
            Show that every element of $I$ is an element of $\downward{R}{b}$.
            \begin{subproof}
                Fix $x\in I$.
                $x,b\in B$ and $x\neq b$ by \auto{common_initial_segment,inter_elim_left,subseteq,setminus}.
                $\downward{R}{x}\subseteq I$ by \auto{common_initial_segment}.
                Thus $x\mathrel{R}b$ by \auto{generated_well_ordering,strict_well_order_iff,connex,dom_intro,downward_closure,subseteq,setminus}.
                Follows by \obv{dom_intro,downward_closure}.
            \end{subproof}
            Follows by set extensionality.
        \end{subproof}

        Show $\downward{S}{c}=I$.
        \begin{subproof}
            Every element of $\downward{S}{c}$ is an element of $I$
                by \auto{generated_well_ordering,strict_well_order_iff,subseteq,times_tuple_elim,downward_closure,setminus,asymmetric}.
            Show that every element of $I$ is an element of $\downward{S}{c}$.
            \begin{subproof}
                Fix $x\in I$.
                $x,c\in C$ and $x\neq c$ by \auto{common_initial_segment,inter_elim_right,subseteq,setminus}.
                $\downward{S}{x}\subseteq I$ by \auto{common_initial_segment}.
                Thus $x\mathrel{S}c$ by \auto{generated_well_ordering,strict_well_order_iff,connex,dom_intro,downward_closure,subseteq,setminus}.
                Follows by \obv{dom_intro,downward_closure}.
            \end{subproof}
            Follows by set extensionality.
        \end{subproof}

        $(I,b),(I,c)\in f$ by
            \auto{generated_well_ordering_entry,setminus}.
        Thus $b=c$ by \obv{generated_well_ordering,function_rightunique}.
        Let $J=\cons{b}{I}$.
        $J\subseteq B\inter C$ by \auto{common_initial_segment,setminus,inter_intro,cons_subseteq_intro}.

        Show that for all $j\in J$ we have $\downward{R}{j}=\downward{S}{j}$.
        \begin{subproof}
            Fix $j\in J$.
            Follows by \auto{cons,common_initial_segment}.
        \end{subproof}

        Show that for all $j\in J$ we have $\downward{R}{j}\subseteq J$.
        \begin{subproof}
            Fix $j\in J$.
            Follows by \auto{cons,common_initial_segment,subseteq_cons_self,subseteq_transitive}.
        \end{subproof}

        Show that for all $j\in J$ we have $\downward{S}{j}\subseteq J$.
        \begin{subproof}
            Fix $j\in J$.
            Follows by \auto{cons,common_initial_segment,subseteq_cons_self,subseteq_transitive}.
        \end{subproof}

        Thus $J$ is a common initial segment of $R$ on $B$ and $S$ on $C$
            by \obv{common_initial_segment}.
        Hence $J\in K$ by \obv{powerset_intro}.
        Therefore $b\in I$ by \auto{cons_left,unions_iff}.
        Contradiction by \obv{setminus}.
    \end{subproof}

    \begin{byCase}
    \caseOf{$B\subseteq I$.}
        $I\subseteq B\inter C$ by \obv{common_initial_segment}.
        Thus $I=B$ and $B\subseteq C$ by \auto{inter_elim_left,inter_elim_right,subseteq,subseteq_antisymmetric}.
        Thus $B$ is a common initial segment of $R$ on $B$ and $S$ on $C$.
    \caseOf{$C\subseteq I$.}
        $I\subseteq B\inter C$ by \obv{common_initial_segment}.
        Thus $I=C$ and $C\subseteq B$ by \auto{inter_elim_left,inter_elim_right,subseteq,subseteq_antisymmetric}.
        Thus $C$ is a common initial segment of $R$ on $B$ and $S$ on $C$.
    \end{byCase}
\end{proof}

\begin{lemma}\label{empty_generated_well_ordering}
    Let $f$ be a function.
    $\emptyset$ is a well ordering generated by $f$ on $\emptyset$.
\end{lemma}
\begin{proof}
    $\emptyset$ is transitive by \auto{transitive,emptyset}.
    $\emptyset$ is asymmetric by \auto{asymmetric,emptyset}.
    $\emptyset$ is connex on $\emptyset$ by \auto{connex,emptyset}.
    $\emptyset$ is a strict well order on $\emptyset$
        by \auto{strict_well_order_iff,emptyset,emptyset_subseteq,subseteq_emptyset_iff}.
    Follows by \auto{generated_well_ordering,emptyset}.
\end{proof}

\begin{lemma}\label{generated_well_ordering_adjoin}
    Suppose $R$ is a well ordering generated by $f$ on $B$.
    Suppose $c\notin B$ and $(B,c)\in f$.
    Suppose $S=R\union(B\times\{c\})$.
    Then $S$ is a well ordering generated by $f$ on $\cons{c}{B}$.
\end{lemma}
\begin{proof}
    Let $C=\cons{c}{B}$.
    $R\subseteq B\times B$ by \auto{generated_well_ordering,strict_well_order_iff}.
    $B\subseteq C$ and $c\in C$ by \obv{subseteq_cons_self,cons_left}.
    Thus $B\times B\subseteq C\times C$ by \auto{times_subseteq_left,times_subseteq_right,subseteq_transitive}.
    Hence $R\subseteq C\times C$ by \obv{subseteq_transitive}.
    $\{c\}\subseteq C$ by notion.
    Thus $B\times\{c\}\subseteq C\times C$ by \auto{times_subseteq_left,times_subseteq_right,subseteq_transitive}.
    Hence $S\subseteq C\times C$ by \auto{union_subsets_is_subset}.
    Show $S$ is transitive.
    \begin{subproof}
        It suffices to show that for all $x,y,z$ such that $x\mathrel{S}y\mathrel{S}z$
            we have $x\mathrel{S}z$ by \obv{transitive}.
        Fix $x$.
        Fix $y$.
        Fix $z$.
        Assume $x\mathrel{S}y\mathrel{S}z$.
        Show $x\mathrel{R}y$.
        \begin{subproof}
            Suppose not.
            $(x,y)\in B\times\{c\}$ by \auto{union}.
            Thus $y=c$ by \auto{times_tuple_elim,singleton_iff}.
            $y\in B$ by \auto{union,subseteq,times_tuple_elim}.
            Contradiction.
        \end{subproof}
        \begin{byCase}
        \caseOf{$y\mathrel{R}z$.}
            $x\mathrel{R}z$ by \auto{generated_well_ordering,strict_well_order_iff,transitive}.
            Follows by \auto{union}.
        \caseOf{$y\not\mathrel{R}z$.}
            $(y,z)\in B\times\{c\}$ by \auto{union}.
            Thus $y\in B$ and $z=c$ by \auto{times_tuple_elim,singleton_iff}.
            $x\in B$ by \obv{subseteq,times_tuple_elim}.
            Hence $(x,z)\in B\times\{c\}$ by \auto{times_tuple_intro,singleton_intro}.
            Follows by \auto{union}.
        \end{byCase}
    \end{subproof}

    Show $S$ is asymmetric.
    \begin{subproof}
        It suffices to show that for all $x,y$ such that $x\mathrel{S}y$
            we have $y\not\mathrel{S}x$ by \obv{asymmetric}.
        Fix $x$.
        Fix $y$.
        Assume $x\mathrel{S}y$.
        Show $y\not\mathrel{S}x$.
        \begin{subproof}
            Suppose not.
            $x,y\in B$ by \auto{union,generated_well_ordering,strict_well_order_iff,subseteq,times_tuple_elim}.
            Thus $(x,y),(y,x)\in R$ by \auto{union,times_tuple_elim,singleton_iff}.
            Contradiction by \auto{generated_well_ordering,strict_well_order_iff,asymmetric}.
        \end{subproof}
    \end{subproof}

    Show $S$ is connex on $C$.
    \begin{subproof}
        It suffices to show that for all $x,y$ such that $x\in C$ and $y\in C$ and $x\neq y$
            we have $x\mathrel{S}y$ or $y\mathrel{S}x$ by \obv{connex}.
        Fix $x$.
        Fix $y$.
        Assume $x\in C$ and $y\in C$ and $x\neq y$.
        Follows by \auto{generated_well_ordering,strict_well_order_iff,connex,cons,union,times,pair_eq_iff}.
    \end{subproof}

    Show that for all $D$ such that $D\subseteq C$ and $D\neq\emptyset$
        there exists $d\in D$ such that for all $x\in D$ we have $d\mathrel{S}x$ or $d=x$.
    \begin{subproof}
        Fix $D$.
        Assume $D\subseteq C$ and $D\neq\emptyset$.
        \begin{byCase}
        \caseOf{$D\inter B\neq\emptyset$.}
            Take $d\in D\inter B$ such that for all $b\in D\inter B$
                we have $d\mathrel{R}b$ or $d=b$
                by \auto{generated_well_ordering,strict_well_order_iff,inter_lower_right}.
            $d\in D$ by \obv{inter_elim_left}.
            For all $x\in D$ we have $d\mathrel{S}x$ or $d=x$
                by \auto{subseteq,cons,inter_intro,inter_elim_left,inter_elim_right,union,times,pair_eq_iff}.
        \caseOf{$D\inter B=\emptyset$.}
            $D\subseteq\{c\}$ by \auto{subseteq,cons,inter_intro,emptyset,singleton_iff}.
            Take $d$ such that $d\in D$ by \obv{nonempty_inhabited}.
            Then $d=c$ by \obv{subseteq,singleton_iff}.
            Thus $c\in D$.
            Hence $D=\{c\}$ by \obv{subseteq_antisymmetric,singleton_subset_intro}.
            Hence $c\in D$ and for all $x\in D$ we have $c=x$ by \auto{singleton_iff}.
        \end{byCase}
    \end{subproof}

    Thus $S$ is a strict well order on $C$ by \auto{strict_well_order_iff}.

    Show that for all $b\in B$ we have $\downward{S}{b}=\downward{R}{b}$.
    \begin{subproof}
        Fix $b\in B$.
        Show $\downward{S}{b}\subseteq\downward{R}{b}$.
        \begin{subproof}
            It suffices to show that every element of $\downward{S}{b}$ is an element of $\downward{R}{b}$
                by \obv{subseteq}.
            Fix $x\in\downward{S}{b}$.
            $x\mathrel{S}b$ by \obv{downward_closure}.
            Thus $x\mathrel{R}b$ by \auto{union,times_tuple_elim,singleton_iff}.
            Follows by \obv{dom_intro,downward_closure}.
        \end{subproof}
        Show $\downward{R}{b}\subseteq\downward{S}{b}$.
        \begin{subproof}
            It suffices to show that every element of $\downward{R}{b}$ is an element of $\downward{S}{b}$
                by \obv{subseteq}.
            Fix $x\in\downward{R}{b}$.
            $x\mathrel{R}b$ by \obv{downward_closure}.
            Thus $x\mathrel{S}b$ by \auto{union}.
            Follows by \obv{dom_intro,downward_closure}.
        \end{subproof}
        Follows by \obv{subseteq_antisymmetric}.
    \end{subproof}

    Show $\downward{S}{c}=B$.
    \begin{subproof}
        Show $\downward{S}{c}\subseteq B$.
        \begin{subproof}
            It suffices to show that every element of $\downward{S}{c}$ is an element of $B$
                by \obv{subseteq}.
            Fix $x\in\downward{S}{c}$.
            $x\mathrel{S}c$ by \obv{downward_closure}.
            Follows by \auto{union,subseteq,times_tuple_elim}.
        \end{subproof}
        Show $B\subseteq\downward{S}{c}$.
        \begin{subproof}
            It suffices to show that every element of $B$ is an element of $\downward{S}{c}$
                by \obv{subseteq}.
            Fix $b\in B$.
            $(b,c)\in B\times\{c\}$ by \obv{times_tuple_intro,singleton_intro}.
            Thus $b\mathrel{S}c$ by \auto{union}.
            Follows by \obv{dom_intro,downward_closure}.
        \end{subproof}
        Follows by \obv{subseteq_antisymmetric}.
    \end{subproof}

    For all $x\in C$ we have $(\downward{S}{x},x)\in f$
        by \auto{cons,generated_well_ordering}.
    Follows by \obv{generated_well_ordering}.
\end{proof}

\begin{lemma}\label{common_initial_segment_relation_subseteq}
    Suppose $R$ is a well ordering generated by $f$ on $B$.
    Suppose $S$ is a well ordering generated by $f$ on $C$.
    Suppose $B$ is a common initial segment of $R$ on $B$ and $S$ on $C$.
    Then $R\subseteq S$.
\end{lemma}
\begin{proof}
    It suffices to show that every element of $R$ is an element of $S$ by \obv{subseteq}.
    Fix $w\in R$.
    Take $x,y$ such that $w=(x,y)$ by \auto{generated_well_ordering,strict_well_order_iff,subseteq,times_elem_is_tuple}.
    Then $x\mathrel{R}y$ and $y\in B$ by
        \auto{generated_well_ordering,strict_well_order_iff,subseteq,
            times_tuple_elim}.
    Thus $x\in\downward{R}{y}$ by \obv{dom_intro,downward_closure}.
    $\downward{R}{y}=\downward{S}{y}$ by \auto{common_initial_segment}.
    Hence $x\mathrel{S}y$ by \auto{downward_closure}.
\end{proof}

\begin{lemma}\label{common_initial_segment_symmetric}
    Suppose $I$ is a common initial segment of $R$ on $B$ and $S$ on $C$.
    Then $I$ is a common initial segment of $S$ on $C$ and $R$ on $B$.
\end{lemma}
\begin{proof}
    Follows by \auto{common_initial_segment,inter_comm}.
\end{proof}

\begin{lemma}\label{union_generated_well_orderings}
    Suppose $G$ is inhabited.
    Suppose for all $p\in G$ we have $\snd{p}$ is a well ordering generated by $f$ on $\fst{p}$.
    Suppose $U=\{\fst{p}\mid p\in G\}$ and $V=\{\snd{p}\mid p\in G\}$.
    Suppose $B=\unions{U}$ and $S=\unions{V}$.
    Then $S$ is a well ordering generated by $f$ on $B$.
\end{lemma}
\begin{proof}
    $f$ is a function by
        \auto{nonempty_inhabited,generated_well_ordering}.

    Show that for all $p\in G$ we have for all $q\in G$
        $\snd{p}\subseteq\snd{q}$ or $\snd{q}\subseteq\snd{p}$.
    \begin{subproof}
        Fix $p\in G$.
        Fix $q\in G$.
        Follows by \auto{generated_well_orderings_compatible,common_initial_segment_symmetric,common_initial_segment_relation_subseteq}.
    \end{subproof}

    Show $S\subseteq B\times B$.
    \begin{subproof}
        It suffices to show that every element of $S$ is an element of $B\times B$ by \obv{subseteq}.
        Fix $w\in S$.
        Take $T\in V$ such that $w\in T$ by \obv{unions_iff}.
        Take $p\in G$ such that $T=\snd{p}$ by \auto.
        $T\subseteq\fst{p}\times\fst{p}$ by \auto{generated_well_ordering,strict_well_order_iff}.
        $\fst{p}\in U$ by \auto.
        Thus $\fst{p}\subseteq B$ by \auto{subseteq,unions_iff}.
        Hence $\fst{p}\times\fst{p}\subseteq B\times B$
            by \auto{times_subseteq_left,times_subseteq_right,subseteq_transitive}.
        Follows by \obv{subseteq}.
    \end{subproof}

    Show $S$ is transitive.
    \begin{subproof}
        It suffices to show that for all $x,y,z$ such that $x\mathrel{S}y\mathrel{S}z$
            we have $x\mathrel{S}z$ by \obv{transitive}.
        Fix $x$.
        Fix $y$.
        Fix $z$.
        Assume $x\mathrel{S}y\mathrel{S}z$.
        Take $T\in V$ such that $(x,y)\in T$ by \obv{unions_iff}.
        Take $p\in G$ such that $T=\snd{p}$ by \auto.
        Take $W\in V$ such that $(y,z)\in W$ by \obv{unions_iff}.
        Take $q\in G$ such that $W=\snd{q}$ by \auto.
        \begin{byCase}
        \caseOf{$\snd{p}\subseteq\snd{q}$.}
            $(x,y),(y,z)\in\snd{q}$ by \obv{subseteq}.
            Thus $(x,z)\in\snd{q}$ by \auto{generated_well_ordering,strict_well_order_iff,transitive}.
            Follows by \auto{unions_iff}.
        \caseOf{$\snd{q}\subseteq\snd{p}$.}
            $(x,y),(y,z)\in\snd{p}$ by \obv{subseteq}.
            Thus $(x,z)\in\snd{p}$ by \auto{generated_well_ordering,strict_well_order_iff,transitive}.
            Follows by \auto{unions_iff}.
        \end{byCase}
    \end{subproof}

    Show $S$ is asymmetric.
    \begin{subproof}
        It suffices to show that for all $x,y$ such that $x\mathrel{S}y$
            we have $y\not\mathrel{S}x$ by \obv{asymmetric}.
        Fix $x$.
        Fix $y$.
        Assume $x\mathrel{S}y$.
        Show $y\not\mathrel{S}x$.
        \begin{subproof}
            Suppose not.
            Take $T\in V$ such that $(x,y)\in T$ by \obv{unions_iff}.
            Take $p\in G$ such that $T=\snd{p}$ by \auto.
            Take $W\in V$ such that $(y,x)\in W$ by \obv{unions_iff}.
            Take $q\in G$ such that $W=\snd{q}$ by \auto.
            $\snd{p}\subseteq\snd{q}$ or $\snd{q}\subseteq\snd{p}$.
            Contradiction by \auto{subseteq,generated_well_ordering,strict_well_order_iff,asymmetric}.
        \end{subproof}
    \end{subproof}

    Show $S$ is connex on $B$.
    \begin{subproof}
        It suffices to show that for all $x,y$ such that $x\in B$ and $y\in B$ and $x\neq y$
            we have $x\mathrel{S}y$ or $y\mathrel{S}x$ by \obv{connex}.
        Fix $x$.
        Fix $y$.
        Assume $x\in B$ and $y\in B$ and $x\neq y$.
        Take $C\in U$ such that $x\in C$ by \obv{unions_iff}.
        Take $p\in G$ such that $C=\fst{p}$ by \auto.
        Take $D\in U$ such that $y\in D$ by \obv{unions_iff}.
        Take $q\in G$ such that $D=\fst{q}$ by \auto.
        $\snd{p}\in V$ and $\snd{q}\in V$ by \auto.
        \begin{byCase}
        \caseOf{$\fst{p}$ is a common initial segment of $\snd{p}$ on $\fst{p}$ and $\snd{q}$ on $\fst{q}$.}
            $\fst{p}\subseteq\fst{q}$ by \auto{common_initial_segment,inter_elim_right,subseteq}.
            Thus $x,y\in\fst{q}$ by \obv{subseteq}.
            Hence $x\mathrel{\snd{q}}y$ or $y\mathrel{\snd{q}}x$
                by \auto{generated_well_ordering,strict_well_order_iff,connex}.
            Follows by \auto{unions_iff}.
        \caseOf{$\fst{q}$ is a common initial segment of $\snd{p}$ on $\fst{p}$ and $\snd{q}$ on $\fst{q}$.}
            $\fst{q}\subseteq\fst{p}$ by \auto{common_initial_segment,inter_elim_left,subseteq}.
            Thus $x,y\in\fst{p}$ by \obv{subseteq}.
            Hence $x\mathrel{\snd{p}}y$ or $y\mathrel{\snd{p}}x$
                by \auto{generated_well_ordering,strict_well_order_iff,connex}.
            Follows by \auto{unions_iff}.
        \end{byCase}
    \end{subproof}

    Show that for all $D$ such that $D\subseteq B$ and $D\neq\emptyset$
        there exists $d\in D$ such that for all $x\in D$ we have $d\mathrel{S}x$ or $d=x$.
    \begin{subproof}
        Fix $D$.
        Assume $D\subseteq B$ and $D\neq\emptyset$.
        Take $x$ such that $x\in D$ by \obv{nonempty_inhabited}.
        Take $C\in U$ such that $x\in C$ by \obv{subseteq,unions_iff}.
        Take $p\in G$ such that $C=\fst{p}$ by \auto.
        $\snd{p}\in V$ by \auto.
        $D\inter\fst{p}\neq\emptyset$ by \auto{inter_intro,emptyset}.
        Take $d\in D\inter\fst{p}$ such that for all $a\in D\inter\fst{p}$
            we have $d\mathrel{\snd{p}}a$ or $d=a$
            by \auto{generated_well_ordering,strict_well_order_iff,inter_lower_right}.
        $d\in D$ by \obv{inter_elim_left}.
        Show that for all $y\in D$ we have $d\mathrel{S}y$ or $d=y$.
        \begin{subproof}
            Fix $y\in D$.
            Take $E\in U$ such that $y\in E$ by \obv{subseteq,unions_iff}.
            Take $q\in G$ such that $E=\fst{q}$ by \auto.
            $\snd{q}\in V$ by \auto.
            $\snd{p}$ is a well ordering generated by $f$ on $\fst{p}$ and
                $\snd{q}$ is a well ordering generated by $f$ on $\fst{q}$.
            $\fst{p}$ is a common initial segment of $\snd{p}$ on $\fst{p}$ and $\snd{q}$ on $\fst{q}$
                or $\fst{q}$ is a common initial segment of $\snd{p}$ on $\fst{p}$ and $\snd{q}$ on $\fst{q}$
                by \obv{generated_well_orderings_compatible}.
            \begin{byCase}
            \caseOf{$\fst{p}$ is a common initial segment of $\snd{p}$ on $\fst{p}$ and $\snd{q}$ on $\fst{q}$.}
                Show $d\mathrel{S}y$ or $d=y$.
                \begin{subproof}
                    \begin{byCase}
                    \caseOf{$y\in\fst{p}$.}
                        $y\in D\inter\fst{p}$ by \obv{inter_intro}.
                        Thus $d\mathrel{\snd{p}}y$ or $d=y$.
                        Follows by \auto{unions_iff}.
                    \caseOf{$y\notin\fst{p}$.}
                        $d,y\in\fst{q}$ and $d\neq y$ by \auto{common_initial_segment,inter_elim_right,inter_elim_left,subseteq}.
                        $\downward{\snd{p}}{d}=\downward{\snd{q}}{d}$
                            by \auto{common_initial_segment,inter_elim_right}.
                        $\downward{\snd{p}}{d}\subseteq\fst{p}$
                            by \auto{common_initial_segment,inter_elim_right}.
                        Thus $d\mathrel{\snd{q}}y$ by \auto{generated_well_ordering,strict_well_order_iff,connex,dom_intro,downward_closure,subseteq}.
                        Follows by \auto{unions_iff}.
                    \end{byCase}
                \end{subproof}
            \caseOf{$\fst{q}$ is a common initial segment of $\snd{p}$ on $\fst{p}$ and $\snd{q}$ on $\fst{q}$.}
                $\fst{q}\subseteq\fst{p}$ by \auto{common_initial_segment,inter_elim_left,subseteq}.
                Thus $y\in D\inter\fst{p}$ by \obv{inter_intro,subseteq}.
                Hence $d\mathrel{\snd{p}}y$ or $d=y$.
                Follows by \auto{unions_iff}.
            \end{byCase}
        \end{subproof}
    \end{subproof}

    Thus $S$ is a strict well order on $B$ by \auto{strict_well_order_iff}.

    Show that for all $b\in B$ we have $(\downward{S}{b},b)\in f$.
    \begin{subproof}
        Fix $b\in B$.
        Take $C\in U$ such that $b\in C$ by \obv{unions_iff}.
        Take $p\in G$ such that $C=\fst{p}$ by \auto.
        $\snd{p}$ is a well ordering generated by $f$ on $\fst{p}$.
        Show $\downward{S}{b}=\downward{\snd{p}}{b}$.
        \begin{subproof}
            Show $\downward{S}{b}\subseteq\downward{\snd{p}}{b}$.
            \begin{subproof}
                It suffices to show that every element of $\downward{S}{b}$ is an element of $\downward{\snd{p}}{b}$
                    by \obv{subseteq}.
                Fix $x\in\downward{S}{b}$.
                $(x,b)\in S$ by \obv{downward_closure}.
                Take $T\in V$ such that $(x,b)\in T$ by \obv{unions_iff}.
                Take $q\in G$ such that $T=\snd{q}$ by \auto.
                $\snd{q}$ is a well ordering generated by $f$ on $\fst{q}$.
                \begin{byCase}
                \caseOf{$\fst{p}$ is a common initial segment of $\snd{p}$ on $\fst{p}$ and $\snd{q}$ on $\fst{q}$.}
                    $\downward{\snd{p}}{b}=\downward{\snd{q}}{b}$ by \auto{common_initial_segment}.
                    $x\in\downward{\snd{q}}{b}$ by \auto{dom_intro,downward_closure}.
                \caseOf{$\fst{q}$ is a common initial segment of $\snd{p}$ on $\fst{p}$ and $\snd{q}$ on $\fst{q}$.}
                    $\snd{q}\subseteq\snd{p}$ by \obv{common_initial_segment_symmetric,common_initial_segment_relation_subseteq}.
                    Thus $(x,b)\in\snd{p}$ by \obv{subseteq}.
                    Follows by \obv{dom_intro,downward_closure}.
                \end{byCase}
            \end{subproof}
            Show $\downward{\snd{p}}{b}\subseteq\downward{S}{b}$.
            \begin{subproof}
                It suffices to show that every element of $\downward{\snd{p}}{b}$ is an element of $\downward{S}{b}$
                    by \obv{subseteq}.
                Fix $x\in\downward{\snd{p}}{b}$.
                $(x,b)\in\snd{p}$ by \obv{downward_closure}.
                $\snd{p}\in V$ by \auto.
                Thus $(x,b)\in S$ by \auto{unions_iff}.
                Follows by \obv{dom_intro,downward_closure}.
            \end{subproof}
            Follows by \obv{subseteq_antisymmetric}.
        \end{subproof}
        Follows by \auto{generated_well_ordering_entry}.
    \end{subproof}

    Follows by \obv{generated_well_ordering}.
\end{proof}
