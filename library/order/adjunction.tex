\section{Order adjunctions}

\import{order/morphism.tex}

\begin{definition}\label{order_adjunction}
    Let $P,Q$ be quasiordered sets.
    $\orderAdjunction{l}{P}{Q}{r}$ iff
        $l\in\funs{\carrier[P]}{\carrier[Q]}$ and
        $r\in\funs{\carrier[Q]}{\carrier[P]}$ and
        for all $p,q$ such that $p\in\carrier[P]$ and
            $q\in\carrier[Q]$ we have
            $l(p)\mathrel{\leq[Q]}q$ iff
            $p\mathrel{\leq[P]}r(q)$.
\end{definition}

\begin{proposition}\label{order_adjunction_left_type}
    Let $P,Q$ be quasiordered sets.
    Suppose $\orderAdjunction{l}{P}{Q}{r}$.
    Then $l\in\funs{\carrier[P]}{\carrier[Q]}$.
\end{proposition}
\begin{proof}
    Follows by \obv{order_adjunction}.
\end{proof}

\begin{proposition}\label{order_adjunction_right_type}
    Let $P,Q$ be quasiordered sets.
    Suppose $\orderAdjunction{l}{P}{Q}{r}$.
    Then $r\in\funs{\carrier[Q]}{\carrier[P]}$.
\end{proposition}
\begin{proof}
    Follows by \obv{order_adjunction}.
\end{proof}

\begin{proposition}\label{order_adjunction_iff}
    Let $P,Q$ be quasiordered sets.
    Suppose $\orderAdjunction{l}{P}{Q}{r}$.
    Let $p\in\carrier[P]$ and $q\in\carrier[Q]$.
    Then $l(p)\mathrel{\leq[Q]}q$ iff
        $p\mathrel{\leq[P]}r(q)$.
\end{proposition}
\begin{proof}
    Follows by \obv{order_adjunction}.
\end{proof}

\begin{proposition}\label{order_adjunction_unit}
    Let $P,Q$ be quasiordered sets.
    Suppose $\orderAdjunction{l}{P}{Q}{r}$.
    Let $p\in\carrier[P]$.
    Then $p\mathrel{\leq[P]}r(l(p))$.
\end{proposition}
\begin{proof}
    $l(p)\in\carrier[Q]$ by
        \auto{order_adjunction_left_type,funs_type_apply}.
    $\leq[Q]$ is reflexive on $\carrier[Q]$ by
        \obv{quasiorder_refl}.
    Hence $l(p)\mathrel{\leq[Q]}l(p)$ by \obv{reflexive_on}.
    Follows by \obv{order_adjunction_iff}.
\end{proof}

\begin{proposition}\label{order_adjunction_counit}
    Let $P,Q$ be quasiordered sets.
    Suppose $\orderAdjunction{l}{P}{Q}{r}$.
    Let $q\in\carrier[Q]$.
    Then $l(r(q))\mathrel{\leq[Q]}q$.
\end{proposition}
\begin{proof}
    $r(q)\in\carrier[P]$ by
        \auto{order_adjunction_right_type,funs_type_apply}.
    $\leq[P]$ is reflexive on $\carrier[P]$ by
        \obv{quasiorder_refl}.
    Hence $r(q)\mathrel{\leq[P]}r(q)$ by \obv{reflexive_on}.
    Follows by \obv{order_adjunction_iff}.
\end{proof}

\begin{proposition}\label{order_adjunction_left_order_preserving}
    Let $P,Q$ be quasiordered sets.
    Suppose $\orderAdjunction{l}{P}{Q}{r}$.
    Then $\orderPreserving{l}{\carrier[P]}{\leq[P]}
        {\carrier[Q]}{\leq[Q]}$.
\end{proposition}
\begin{proof}
    $\leq[P]$ is a binary relation on $\carrier[P]$ and
        $\leq[Q]$ is a binary relation on $\carrier[Q]$ by
        \obv{quasiorder_type}.
    $l\in\funs{\carrier[P]}{\carrier[Q]}$ by
        \obv{order_adjunction_left_type}.
    Show for all $p_1,p_2\in\carrier[P]$ such that
        $p_1\mathrel{\leq[P]}p_2$ we have
        $l(p_1)\mathrel{\leq[Q]}l(p_2)$.
    \begin{subproof}
        Fix $p_1,p_2\in\carrier[P]$.
        Assume $p_1\mathrel{\leq[P]}p_2$.
        $p_2\mathrel{\leq[P]}r(l(p_2))$ by
            \obv{order_adjunction_unit}.
        $\leq[P]$ is transitive by \obv{quasiorder_tran}.
        Hence $p_1\mathrel{\leq[P]}r(l(p_2))$ by
            \obv{transitive}.
        $l(p_2)\in\carrier[Q]$ by \obv{funs_type_apply}.
        Follows by \obv{order_adjunction_iff}.
    \end{subproof}
    Follows by \obv{order_preserving}.
\end{proof}

\begin{proposition}\label{order_adjunction_right_order_preserving}
    Let $P,Q$ be quasiordered sets.
    Suppose $\orderAdjunction{l}{P}{Q}{r}$.
    Then $\orderPreserving{r}{\carrier[Q]}{\leq[Q]}
        {\carrier[P]}{\leq[P]}$.
\end{proposition}
\begin{proof}
    $\leq[P]$ is a binary relation on $\carrier[P]$ and
        $\leq[Q]$ is a binary relation on $\carrier[Q]$ by
        \obv{quasiorder_type}.
    $r\in\funs{\carrier[Q]}{\carrier[P]}$ by
        \obv{order_adjunction_right_type}.
    Show for all $q_1,q_2\in\carrier[Q]$ such that
        $q_1\mathrel{\leq[Q]}q_2$ we have
        $r(q_1)\mathrel{\leq[P]}r(q_2)$.
    \begin{subproof}
        Fix $q_1,q_2\in\carrier[Q]$.
        Assume $q_1\mathrel{\leq[Q]}q_2$.
        $l(r(q_1))\mathrel{\leq[Q]}q_1$ by
            \obv{order_adjunction_counit}.
        $\leq[Q]$ is transitive by \obv{quasiorder_tran}.
        Hence $l(r(q_1))\mathrel{\leq[Q]}q_2$ by
            \obv{transitive}.
        $r(q_1)\in\carrier[P]$ by \obv{funs_type_apply}.
        Follows by \obv{order_adjunction_iff}.
    \end{subproof}
    Follows by \obv{order_preserving}.
\end{proof}

\begin{proposition}\label{order_adjunction_counit_identity_leftinverse}
    Let $P,Q$ be quasiordered sets.
    Suppose $\orderAdjunction{l}{P}{Q}{r}$.
    Suppose for all $q\in\carrier[Q]$ we have $l(r(q))=q$.
    Then $l$ is a left inverse of $r$.
\end{proposition}
\begin{proof}
    $r\in\funs{\carrier[Q]}{\carrier[P]}$ and
        $l\in\funs{\carrier[P]}{\carrier[Q]}$ by
        \obv{order_adjunction_left_type,order_adjunction_right_type}.
    Follows by \obv{leftinverse_from_to_iff}.
\end{proof}

\begin{proposition}\label{order_adjunction_unit_identity_rightinverse}
    Let $P,Q$ be quasiordered sets.
    Suppose $\orderAdjunction{l}{P}{Q}{r}$.
    Suppose for all $p\in\carrier[P]$ we have $r(l(p))=p$.
    Then $l$ is a right inverse of $r$.
\end{proposition}
\begin{proof}
    $r\in\funs{\carrier[Q]}{\carrier[P]}$ and
        $l\in\funs{\carrier[P]}{\carrier[Q]}$ by
        \obv{order_adjunction_left_type,order_adjunction_right_type}.
    Follows by \obv{rightinverse_from_to_iff}.
\end{proof}

\begin{proposition}\label{order_adjunction_right_embedding_of_counit_identity}
    Let $P,Q$ be quasiordered sets.
    Suppose $\orderAdjunction{l}{P}{Q}{r}$.
    Suppose for all $q\in\carrier[Q]$ we have $l(r(q))=q$.
    Then $\orderEmbedding{r}{\carrier[Q]}{\leq[Q]}
        {\carrier[P]}{\leq[P]}$.
\end{proposition}
\begin{proof}
    $r\in\funs{\carrier[Q]}{\carrier[P]}$ by
        \obv{order_adjunction_right_type}.
    Hence $r$ is a function and $\dom{r}=\carrier[Q]$ by
        \obv{funs_is_function,funs}.
    $l$ is a left inverse of $r$ by
        \obv{order_adjunction_counit_identity_leftinverse}.
    Hence $r$ is an injection by \obv{leftinverse_implies_injection}.
    Hence $r\in\inj{\carrier[Q]}{\carrier[P]}$ by
        \auto{inj,injective_function}.
    $\orderPreserving{r}{\carrier[Q]}{\leq[Q]}
        {\carrier[P]}{\leq[P]}$ by
        \obv{order_adjunction_right_order_preserving}.
    Show that $\orderReflecting{r}{\carrier[Q]}{\leq[Q]}
        {\carrier[P]}{\leq[P]}$.
    \begin{subproof}
        $\leq[P]$ is a binary relation on $\carrier[P]$ and
            $\leq[Q]$ is a binary relation on $\carrier[Q]$ by
            \obv{quasiorder_type}.
        Show for all $q_1,q_2\in\carrier[Q]$ such that
            $r(q_1)\mathrel{\leq[P]}r(q_2)$ we have
            $q_1\mathrel{\leq[Q]}q_2$.
        \begin{subproof}
            Fix $q_1,q_2\in\carrier[Q]$.
            Assume $r(q_1)\mathrel{\leq[P]}r(q_2)$.
            $r(q_1)\in\carrier[P]$ by \obv{funs_type_apply}.
            Hence $l(r(q_1))\mathrel{\leq[Q]}q_2$ by
                \obv{order_adjunction_iff}.
            $l(r(q_1))=q_1$.
        \end{subproof}
        Follows by \obv{order_reflecting}.
    \end{subproof}
    Follows by
        \obv{order_embedding_of_injective_preserving_reflecting}.
\end{proof}

\begin{proposition}\label{order_adjunction_left_embedding_of_unit_identity}
    Let $P,Q$ be quasiordered sets.
    Suppose $\orderAdjunction{l}{P}{Q}{r}$.
    Suppose for all $p\in\carrier[P]$ we have $r(l(p))=p$.
    Then $\orderEmbedding{l}{\carrier[P]}{\leq[P]}
        {\carrier[Q]}{\leq[Q]}$.
\end{proposition}
\begin{proof}
    $l\in\funs{\carrier[P]}{\carrier[Q]}$ and
        $r\in\funs{\carrier[Q]}{\carrier[P]}$ by
        \obv{order_adjunction_left_type,order_adjunction_right_type}.
    Hence $l$ is a function and $\dom{l}=\carrier[P]$ by
        \obv{funs_is_function,funs}.
    $r$ is a left inverse of $l$ by
        \obv{leftinverse_from_to_iff}.
    Hence $l$ is an injection by \obv{leftinverse_implies_injection}.
    Hence $l\in\inj{\carrier[P]}{\carrier[Q]}$ by
        \auto{inj,injective_function}.
    $\orderPreserving{l}{\carrier[P]}{\leq[P]}
        {\carrier[Q]}{\leq[Q]}$ by
        \obv{order_adjunction_left_order_preserving}.
    Show that $\orderReflecting{l}{\carrier[P]}{\leq[P]}
        {\carrier[Q]}{\leq[Q]}$.
    \begin{subproof}
        $\leq[P]$ is a binary relation on $\carrier[P]$ and
            $\leq[Q]$ is a binary relation on $\carrier[Q]$ by
            \obv{quasiorder_type}.
        Show for all $p_1,p_2\in\carrier[P]$ such that
            $l(p_1)\mathrel{\leq[Q]}l(p_2)$ we have
            $p_1\mathrel{\leq[P]}p_2$.
        \begin{subproof}
            Fix $p_1,p_2\in\carrier[P]$.
            Assume $l(p_1)\mathrel{\leq[Q]}l(p_2)$.
            $l(p_2)\in\carrier[Q]$ by \obv{funs_type_apply}.
            Hence $p_1\mathrel{\leq[P]}r(l(p_2))$ by
                \obv{order_adjunction_iff}.
            $r(l(p_2))=p_2$.
        \end{subproof}
        Follows by \obv{order_reflecting}.
    \end{subproof}
    Follows by
        \obv{order_embedding_of_injective_preserving_reflecting}.
\end{proof}

\begin{proposition}\label{order_adjunction_order_isomorphism_of_unit_counit_identity}
    Let $P,Q$ be quasiordered sets.
    Suppose $\orderAdjunction{l}{P}{Q}{r}$.
    Suppose for all $p\in\carrier[P]$ we have $r(l(p))=p$.
    Suppose for all $q\in\carrier[Q]$ we have $l(r(q))=q$.
    Then $\orderIsomorphism{r}{\carrier[Q]}{\leq[Q]}
        {\carrier[P]}{\leq[P]}$.
\end{proposition}
\begin{proof}
    $\orderEmbedding{r}{\carrier[Q]}{\leq[Q]}
        {\carrier[P]}{\leq[P]}$ by
        \obv{order_adjunction_right_embedding_of_counit_identity}.
    Hence $r\in\inj{\carrier[Q]}{\carrier[P]}$ by
        \obv{order_embedding_injective}.
    $r\in\funs{\carrier[Q]}{\carrier[P]}$ and
        $l\in\funs{\carrier[P]}{\carrier[Q]}$ by
        \obv{order_adjunction_left_type,order_adjunction_right_type}.
    $l$ is a right inverse of $r$ by
        \obv{order_adjunction_unit_identity_rightinverse}.
    Hence $r\in\surj{\carrier[Q]}{\carrier[P]}$ by
        \obv{rightinverse_implies_surjection}.
    $r$ is a function and $\dom{r}=\carrier[Q]$ by
        \obv{funs_is_function,funs}.
    Hence $r$ is injective by \auto{inj,injective_function}.
    Hence $r\in\bijections{\carrier[Q]}{\carrier[P]}$ by
        \obv{bijections}.
    Follows by \obv{order_isomorphism,order_embedding}.
\end{proof}
