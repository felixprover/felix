\section{Downsets}

\import{order/quasiorder.tex}
\import{set/powerset.tex}

\begin{definition}\label{downset}
    Let $P$ be a quasiordered set.
    $D$ is a downset in $P$ iff
        $D\subseteq\carrier[P]$ and
        for all $x,y$ such that $x\in D$ and
            $y\in\carrier[P]$ and $y\mathrel{\leq[P]}x$
            we have $y\in D$.
\end{definition}

\begin{definition}\label{downsets}
    Let $P$ be a quasiordered set.
    $\downsets{P}=\{D\in\pow{\carrier[P]}\mid
        \text{$D$ is a downset in $P$}\}$.
\end{definition}

\begin{proposition}\label{downset_subseteq_carrier}
    Let $P$ be a quasiordered set.
    Suppose $D$ is a downset in $P$.
    Then $D\subseteq\carrier[P]$.
\end{proposition}
\begin{proof}
    Follows by \obv{downset}.
\end{proof}

\begin{proposition}\label{downset_closed_downward}
    Let $P$ be a quasiordered set.
    Suppose $D$ is a downset in $P$.
    Suppose $x\in D$ and $y\in\carrier[P]$ and
        $y\mathrel{\leq[P]}x$.
    Then $y\in D$.
\end{proposition}
\begin{proof}
    Follows by \obv{downset}.
\end{proof}

\begin{proposition}\label{downsets_iff}
    Let $P$ be a quasiordered set.
    Then $D\in\downsets{P}$ iff $D$ is a downset in $P$.
\end{proposition}
\begin{proof}
    Follows by \auto{downsets,downset,pow_iff,subseteq}.
\end{proof}

\begin{proposition}\label{downsets_subseteq_powerset}
    Let $P$ be a quasiordered set.
    Then $\downsets{P}\subseteq\pow{\carrier[P]}$.
\end{proposition}
\begin{proof}
    Follows by \obv{downsets,pow_iff,subseteq}.
\end{proof}

\begin{proposition}\label{emptyset_is_downset}
    Let $P$ be a quasiordered set.
    Then $\emptyset$ is a downset in $P$.
\end{proposition}
\begin{proof}
    $\emptyset\subseteq\carrier[P]$ by \obv{emptyset_subseteq}.
    Follows by \obv{downset,emptyset}.
\end{proof}

\begin{proposition}\label{carrier_is_downset}
    Let $P$ be a quasiordered set.
    Then $\carrier[P]$ is a downset in $P$.
\end{proposition}
\begin{proof}
    Follows by \obv{downset,subseteq}.
\end{proof}

\begin{proposition}\label{downset_inter}
    Let $P$ be a quasiordered set.
    Suppose $D$ and $E$ are downsets in $P$.
    Then $D\inter E$ is a downset in $P$.
\end{proposition}
\begin{proof}
    $D,E\subseteq\carrier[P]$ by \obv{downset_subseteq_carrier}.
    $D\inter E\subseteq\carrier[P]$ by
        \obv{inter_lower_left,subseteq_transitive}.
    Show for all $x,y$ such that $x\in D\inter E$ and
        $y\in\carrier[P]$ and $y\mathrel{\leq[P]}x$ we have
        $y\in D\inter E$.
    \begin{subproof}
        Fix $x,y$.
        Assume $x\in D\inter E$ and $y\in\carrier[P]$ and
            $y\mathrel{\leq[P]}x$.
        $x\in D$ and $x\in E$ by
            \obv{inter_elim_left,inter_elim_right}.
        Hence $y\in D$ and $y\in E$ by
            \obv{downset_closed_downward}.
        Follows by \obv{inter_intro}.
    \end{subproof}
    Follows by \obv{downset}.
\end{proof}

\begin{proposition}\label{downset_union}
    Let $P$ be a quasiordered set.
    Suppose $D$ and $E$ are downsets in $P$.
    Then $D\union E$ is a downset in $P$.
\end{proposition}
\begin{proof}
    $D,E\subseteq\carrier[P]$ by \obv{downset_subseteq_carrier}.
    $D\union E\subseteq\carrier[P]$ by
        \auto{subseteq,union_iff}.
    Show for all $x,y$ such that $x\in D\union E$ and
        $y\in\carrier[P]$ and $y\mathrel{\leq[P]}x$ we have
        $y\in D\union E$.
    \begin{subproof}
        Fix $x,y$.
        Assume $x\in D\union E$ and $y\in\carrier[P]$ and
            $y\mathrel{\leq[P]}x$.
        $x\in D$ or $x\in E$ by \obv{union_iff}.
        Follows by \auto{downset_closed_downward,union_iff}.
    \end{subproof}
    Follows by \obv{downset}.
\end{proof}

\begin{proposition}\label{downset_unions}
    Let $P$ be a quasiordered set.
    Suppose for all $D\in F$ we have $D$ is a downset in $P$.
    Then $\unions{F}$ is a downset in $P$.
\end{proposition}
\begin{proof}
    Show for all $x\in\unions{F}$ we have $x\in\carrier[P]$.
    \begin{subproof}
        Fix $x\in\unions{F}$.
        Take $D\in F$ such that $x\in D$ by \obv{unions_iff}.
        $D\subseteq\carrier[P]$ by \obv{downset_subseteq_carrier}.
        Follows by \obv{elem_subseteq}.
    \end{subproof}
    Hence $\unions{F}\subseteq\carrier[P]$ by \obv{subseteq}.
    Show for all $x,y$ such that $x\in\unions{F}$ and
        $y\in\carrier[P]$ and $y\mathrel{\leq[P]}x$ we have
        $y\in\unions{F}$.
    \begin{subproof}
        Fix $x,y$.
        Assume $x\in\unions{F}$ and $y\in\carrier[P]$ and
            $y\mathrel{\leq[P]}x$.
        Take $D\in F$ such that $x\in D$ by \obv{unions_iff}.
        $D$ is a downset in $P$.
        Hence $y\in D$ by \obv{downset_closed_downward}.
        Follows by \auto{unions_iff}.
    \end{subproof}
    Follows by \obv{downset}.
\end{proof}

\begin{theorem}\label{relation_downward_is_downset}
    Let $P$ be a quasiordered set.
    Let $x\in\carrier[P]$.
    Then $\downward{\leq[P]}{x}\in\downsets{P}$.
\end{theorem}
\begin{proof}
    $\leq[P]$ is a binary relation on $\carrier[P]$ by
        \obv{quasiorder_type}.
    Show for all $y\in\downward{\leq[P]}{x}$ we have
        $y\in\carrier[P]$.
    \begin{subproof}
        Fix $y\in\downward{\leq[P]}{x}$.
        $y\mathrel{\leq[P]}x$ by \obv{downward_closure_iff}.
        Follows by \obv{quasiorder_type,subseteq,times_tuple_elim}.
    \end{subproof}
    Hence $\downward{\leq[P]}{x}\subseteq\carrier[P]$ by
        \obv{subseteq}.
    $\leq[P]$ is transitive by \obv{quasiorder_tran}.
    Show for all $y,z$ such that
        $y\in\downward{\leq[P]}{x}$ and $z\in\carrier[P]$ and
        $z\mathrel{\leq[P]}y$ we have
        $z\in\downward{\leq[P]}{x}$.
    \begin{subproof}
        Fix $y,z$.
        Assume $y\in\downward{\leq[P]}{x}$ and
            $z\in\carrier[P]$ and $z\mathrel{\leq[P]}y$.
        $y\mathrel{\leq[P]}x$ by \obv{downward_closure_iff}.
        Hence $z\mathrel{\leq[P]}x$ by \obv{transitive}.
        Follows by \obv{downward_closure_iff}.
    \end{subproof}
    Hence $\downward{\leq[P]}{x}$ is a downset in $P$ by
        \obv{downset}.
    Follows by \auto{downsets_iff}.
\end{proof}
