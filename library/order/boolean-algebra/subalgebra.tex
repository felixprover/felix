\begin{definition}\label{boolean_subalgebra}
    $\booleanSubalgebra{A}{B}$ iff
        $B$ is a Boolean algebra and
        $A\subseteq\carrier[B]$ and
        $\latticeBottom[B],\latticeTop[B]\in A$ and
        (for every $x\in A$ we have
            $\booleanComplement[B](x)\in A$) and
        (for all $x,y\in A$ we have $\meet[B](x,y)\in A$) and
        (for all $x,y\in A$ we have $\join[B](x,y)\in A$).
\end{definition}

\begin{proposition}\label{boolean_subalgebra_ambient}
    Suppose $\booleanSubalgebra{A}{B}$.
    Then $B$ is a Boolean algebra.
\end{proposition}
\begin{proof}
    Follows by \obv{boolean_subalgebra}.
\end{proof}

\end{proof}

\begin{proposition}\label{full_carrier_boolean_subalgebra}
    Let $B$ be a Boolean algebra.
    Then $\booleanSubalgebra{\carrier[B]}{B}$.
\end{proposition}
\begin{proof}
    $\carrier[B]\subseteq\carrier[B]$ by \obv{subseteq}.
    $\latticeBottom[B],\latticeTop[B]\in\carrier[B]$ by
        \obv{lattice_bottom_type,lattice_top_type}.
    For every $x\in\carrier[B]$ we have
        $\booleanComplement[B](x)\in\carrier[B]$ by
        \obv{boolean_complement_apply_type}.
    For all $x,y\in\carrier[B]$ we have
        $\meet[B](x,y),\join[B](x,y)\in\carrier[B]$ by
        \obv{meet_type,join_type}.
    Follows by \auto{boolean_subalgebra}.
\end{proof}

\begin{theorem}\label{boolean_subalgebra_intersection}
    Suppose $\booleanSubalgebra{A}{B}$ and
        $\booleanSubalgebra{C}{B}$.
    Then $\booleanSubalgebra{A\inter C}{B}$.
\end{theorem}
\begin{proof}
    $A,C\subseteq\carrier[B]$ by \obv{boolean_subalgebra_subseteq}.
    Hence $A\inter C\subseteq\carrier[B]$ by \obv{inter_subseteq}.
    $\latticeBottom[B],\latticeTop[B]\in A\inter C$ by
        \obv{boolean_subalgebra_bottom,boolean_subalgebra_top,
            inter_intro}.
    Show that for every $x\in A\inter C$ we have
        $\booleanComplement[B](x)\in A\inter C$.
    \begin{subproof}
        Follows by
            \obv{inter_elim_left,inter_elim_right,
                boolean_subalgebra_complement,inter_intro}.
    \end{subproof}
    Show that for all $x,y\in A\inter C$ we have
        $\meet[B](x,y)\in A\inter C$.
    \begin{subproof}
        Follows by
            \obv{inter_elim_left,inter_elim_right,
                boolean_subalgebra_meet,inter_intro}.
    \end{subproof}
    Show that for all $x,y\in A\inter C$ we have
        $\join[B](x,y)\in A\inter C$.
    \begin{subproof}
        Follows by
            \obv{inter_elim_left,inter_elim_right,
                boolean_subalgebra_join,inter_intro}.
    \end{subproof}
    Follows by \obv{boolean_subalgebra}.
\end{proof}

\begin{theorem}\label{boolean_subalgebra_indexed_intersection}
    Let $B$ be a Boolean algebra.
    Let $F\in\funs{I}{\pow{\carrier[B]}}$.
    Suppose for every $i\in I$ we have
        $\booleanSubalgebra{F(i)}{B}$.
    Then $\booleanSubalgebra{
        \indexedIntersection{I}{F}{\carrier[B]}}{B}$.
\end{theorem}
\begin{proof}
    Let $C=\indexedIntersection{I}{F}{\carrier[B]}$.
    $C\subseteq\carrier[B]$ by \auto{indexed_intersection_subseteq}.
    $\latticeBottom[B],\latticeTop[B]\in\carrier[B]$ by
        \obv{lattice_bottom_type,lattice_top_type}.
    For every $i\in I$ we have
        $\latticeBottom[B],\latticeTop[B]\in F(i)$ by
        \obv{boolean_subalgebra_bottom,boolean_subalgebra_top}.
    Hence $\latticeBottom[B]\in C$ by
        \obv{indexed_intersection_iff}.
    $\latticeTop[B]\in C$ by
        \obv{indexed_intersection_iff}.
    Show that for every $x\in C$ we have
        $\booleanComplement[B](x)\in C$.
    \begin{subproof}
        Fix $x\in C$.
        Hence $x\in\carrier[B]$ by \obv{indexed_intersection}.
        $\booleanComplement[B](x)\in\carrier[B]$ by
            \obv{boolean_complement_apply_type}.
        Show that for every $i\in I$ we have
            $\booleanComplement[B](x)\in F(i)$.
        \begin{subproof}
            Fix $i\in I$.
            $x\in F(i)$ by \auto{indexed_intersection_iff}.
            Follows by \obv{boolean_subalgebra_complement}.
        \end{subproof}
        Follows by \obv{indexed_intersection}.
    \end{subproof}
    Show that for all $x,y\in C$ we have $\meet[B](x,y)\in C$.
    \begin{subproof}
        Fix $x,y\in C$.
        Hence $x,y\in\carrier[B]$ by \obv{indexed_intersection}.
        $\meet[B](x,y)\in\carrier[B]$ by
            \obv{boolean_algebra_is_lattice,
                lattice_is_meet_semilattice,meet_type}.
        Show that for every $i\in I$ we have $\meet[B](x,y)\in F(i)$.
        \begin{subproof}
            Fix $i\in I$.
            $x,y\in F(i)$ by \auto{indexed_intersection_iff}.
            Follows by \obv{boolean_subalgebra_meet}.
        \end{subproof}
        Follows by \obv{indexed_intersection}.
    \end{subproof}
    Show that for all $x,y\in C$ we have $\join[B](x,y)\in C$.
    \begin{subproof}
        Fix $x,y\in C$.
        Hence $x,y\in\carrier[B]$ by \obv{indexed_intersection}.
        $\join[B](x,y)\in\carrier[B]$ by
            \obv{boolean_algebra_is_lattice,
                lattice_is_join_semilattice,join_type}.
        Show that for every $i\in I$ we have $\join[B](x,y)\in F(i)$.
        \begin{subproof}
            Fix $i\in I$.
            $x,y\in F(i)$ by \auto{indexed_intersection_iff}.
            Follows by \obv{boolean_subalgebra_join}.
        \end{subproof}
        Follows by \obv{indexed_intersection}.
    \end{subproof}
    Follows by \auto{boolean_subalgebra}.
\end{proof}

\begin{theorem}\label{boolean_subalgebra_family_intersection}
    Let $B$ be a Boolean algebra.
    Suppose $C$ is inhabited.
    Suppose for every $A\in C$ we have $\booleanSubalgebra{A}{B}$.
    Then $\booleanSubalgebra{\inters{C}}{B}$.
\end{theorem}
\begin{proof}
    Let $D=\inters{C}$.
    Show that $D\subseteq\carrier[B]$.
    \begin{subproof}
        Take $A$ such that $A\in C$.
        $\booleanSubalgebra{A}{B}$.
        Hence $A\subseteq\carrier[B]$ by
            \obv{boolean_subalgebra_subseteq}.
        $D\subseteq A$ by \auto{inters_subseteq_elem}.
        Follows by \obv{subseteq_transitive}.
    \end{subproof}
    Show that $\latticeBottom[B]\in D$.
    \begin{subproof}
        Show that for every $A\in C$ we have
            $\latticeBottom[B]\in A$.
        \begin{subproof}
            Follows by \obv{boolean_subalgebra_bottom}.
        \end{subproof}
        Follows by \obv{inters_intro}.
    \end{subproof}
    Show that $\latticeTop[B]\in D$.
    \begin{subproof}
        Show that for every $A\in C$ we have $\latticeTop[B]\in A$.
        \begin{subproof}
            Follows by \obv{boolean_subalgebra_top}.
        \end{subproof}
        Follows by \obv{inters_intro}.
    \end{subproof}
    Show that for every $x\in D$ we have
        $\booleanComplement[B](x)\in D$.
    \begin{subproof}
        Fix $x\in D$.
        Show that for every $A\in C$ we have
            $\booleanComplement[B](x)\in A$.
        \begin{subproof}
            Fix $A\in C$.
            $x\in A$ by \obv{inters_destr}.
            Follows by \obv{boolean_subalgebra_complement}.
        \end{subproof}
        Follows by \obv{inters_intro}.
    \end{subproof}
    Show that for all $x,y\in D$ we have $\meet[B](x,y)\in D$.
    \begin{subproof}
        Fix $x,y\in D$.
        Show that for every $A\in C$ we have $\meet[B](x,y)\in A$.
        \begin{subproof}
            Fix $A\in C$.
            $x,y\in A$ by \obv{inters_destr}.
            Follows by \obv{boolean_subalgebra_meet}.
        \end{subproof}
        Follows by \obv{inters_intro}.
    \end{subproof}
    Show that for all $x,y\in D$ we have $\join[B](x,y)\in D$.
    \begin{subproof}
        Fix $x,y\in D$.
        Show that for every $A\in C$ we have $\join[B](x,y)\in A$.

\begin{proposition}\label{boolean_subalgebra_meet_restriction_type}
    Suppose $\booleanSubalgebra{A}{B}$.
    Then $\restrl{\meet[B]}{A\times A}\in\funs{A\times A}{A}$.
\end{proposition}
\begin{proof}
    Hence $B$ is a lattice by \obv{boolean_algebra_is_lattice}.
    Let $f=\meet[B]$.
    Let $m=\restrl{f}{A\times A}$.
    $f\in\funs{\carrier[B]\times\carrier[B]}{\carrier[B]}$ by
        \obv{meet_graph}.
    $A\subseteq\carrier[B]$ by \obv{boolean_subalgebra_subseteq}.
    Hence $A\times A\subseteq\carrier[B]\times\carrier[B]$ by
        \auto{times_subseteq_left,times_subseteq_right,
            subseteq_transitive}.
    Hence $A\times A\subseteq\dom{f}$ by \auto{funs}.
    $m\in\funs{A\times A}{\carrier[B]}$ by \auto{funs_restrl}.
    Hence $m$ is a function and $A\times A=\dom{m}$ by
        \auto{funs_elim}.
    Show that $m$ is a function to $A$.
    \begin{subproof}
        Show that for every $p\in\dom{m}$ we have $m(p)\in A$.
        \begin{subproof}
            Fix $p\in\dom{m}$.
            Hence $p\in A\times A$.
            Take $x,y$ such that $x,y\in A$ and $p=(x,y)$ by
                \obv{times_elem_is_tuple}.
            $f$ is a function by \obv{funs_is_function}.
            $m(p)=f(p)$ by \auto{restrl_of_function_apply}.
            Hence $m(p)=\meet[B](x,y)$.
            Follows by \auto{boolean_subalgebra_meet}.
        \end{subproof}
    \end{subproof}
    Follows by \auto{funs_intro}.
\end{proof}

\begin{theorem}\label{boolean_subalgebra_structure_is_exact_boolean_algebra}
    Suppose $\booleanSubalgebra{A}{B}$.
    Then $\booleanSubalgebraStructure{B}{A}$
        is an exact Boolean algebra.
\end{theorem}
\begin{proof}
    Let $K=\booleanSubalgebraStructure{B}{A}$.
    $B$ is a Boolean algebra by \obv{boolean_subalgebra_ambient}.
    Hence $B$ is a bounded distributive lattice by
        \obv{boolean_algebra_is_bounded_distributive_lattice}.
    Hence $B$ is a bounded lattice and $B$ is a distributive lattice by
        \obv{bounded_distributive_lattice_is_bounded_lattice,
            bounded_distributive_lattice_is_distributive_lattice}.
    Hence $B$ is a lattice by \obv{bounded_lattice_is_lattice}.
    Hence $B$ is an ordered set and $B$ is a meet semilattice and
        $B$ is a join semilattice by
        \obv{lattice_is_ordered_set,lattice_is_meet_semilattice,
            lattice_is_join_semilattice}.
    $A\subseteq\carrier[B]$ by \obv{boolean_subalgebra_subseteq}.
    $\carrier[K]=A$ by \auto{boolean_subalgebra_structure_carrier}.
    $\leq[K]=\relationTrace{\leq[B]}{A}$ by
        \auto{boolean_subalgebra_structure_order}.
    $\meet[K]=\restrl{\meet[B]}{A\times A}$ by
        \auto{boolean_subalgebra_structure_meet}.
    $\join[K]=\restrl{\join[B]}{A\times A}$ by
        \auto{boolean_subalgebra_structure_join}.
    $\latticeBottom[K]=\latticeBottom[B]$ by
        \auto{boolean_subalgebra_structure_bottom}.
    $\latticeTop[K]=\latticeTop[B]$ by
        \auto{boolean_subalgebra_structure_top}.
    $\booleanComplement[K]=\restrl{\booleanComplement[B]}{A}$ by
        \auto{boolean_subalgebra_structure_complement}.

    $\leq[B]$ is an order on $\carrier[B]$ by
        \obv{ordered_set_order_on_carrier}.
    Hence $\relationTrace{\leq[B]}{A}$ is an order on $A$ by
        \obv{relation_trace_order_on}.
    Hence $\leq[K]$ is an order on $\carrier[K]$.
    Hence $K$ is a quasiordered set by
        \auto{quasiordered_set,boolean_subalgebra_structure,
            boolean_algebraaggregateexact,
            boolean_algebraaggregateprojection_carrier,
            boolean_algebraaggregateprojection_leq}.
    $\leq[K]$ is antisymmetric.
    Hence $K$ is an ordered set by
        \auto{orderedset,boolean_subalgebra_structure,
            boolean_algebraaggregateexact,
            boolean_algebraaggregateprojection_carrier,
            boolean_algebraaggregateprojection_leq}.

    $\meet[K]\in
        \funs{\carrier[K]\times\carrier[K]}{\carrier[K]}$ by
        \auto{boolean_subalgebra_meet_restriction_type}.
    Show that for all $x,y\in\carrier[K]$ we have
        $\meet[K](x,y)\mathrel{\leq[K]}x$ and
        $\meet[K](x,y)\mathrel{\leq[K]}y$.
    \begin{subproof}
        Fix $x,y\in\carrier[K]$.
        Hence $x,y\in A$.
        Hence $x,y\in\carrier[B]$ by \obv{subseteq}.
        $\meet[B](x,y)\in A$ by \obv{boolean_subalgebra_meet}.
        $\meet[K](x,y)=\meet[B](x,y)$ by
            \auto{boolean_subalgebra_structure_meet_apply}.
        $\meet[B](x,y)\mathrel{\leq[B]}x$ and
            $\meet[B](x,y)\mathrel{\leq[B]}y$ by
            \obv{meet_lb}.
        Follows by
            \auto{boolean_subalgebra_structure_order_iff}.
    \end{subproof}
    Show that for all $a,x,y\in\carrier[K]$ such that
        $a\mathrel{\leq[K]}x$ and $a\mathrel{\leq[K]}y$ we have
        $a\mathrel{\leq[K]}\meet[K](x,y)$.
    \begin{subproof}
        Fix $a,x,y\in\carrier[K]$.
        Assume $a\mathrel{\leq[K]}x$ and $a\mathrel{\leq[K]}y$.
        Hence $a,x,y\in A$.
        Hence $a,x,y\in\carrier[B]$ by \obv{subseteq}.
        Hence $a\mathrel{\leq[B]}x$ and $a\mathrel{\leq[B]}y$ by
            \auto{boolean_subalgebra_structure_order_iff}.
        Hence $a\mathrel{\leq[B]}\meet[B](x,y)$ by \auto{meet_glb}.
        $\meet[B](x,y)\in A$ by \obv{boolean_subalgebra_meet}.
        $\meet[K](x,y)=\meet[B](x,y)$ by
            \auto{boolean_subalgebra_structure_meet_apply}.
        Follows by
            \auto{boolean_subalgebra_structure_order_iff}.
    \end{subproof}
    Hence $K$ is a meet semilattice by
        \auto{meet_semilattice,boolean_subalgebra_structure,
            boolean_algebraaggregateexact,
            boolean_algebraaggregateprojection_carrier,
            boolean_algebraaggregateprojection_leq,
            boolean_algebraaggregateprojection_meet}.

    $\join[K]\in
        \funs{\carrier[K]\times\carrier[K]}{\carrier[K]}$ by
        \auto{boolean_subalgebra_join_restriction_type}.
    Show that for all $x,y\in\carrier[K]$ we have
        $x\mathrel{\leq[K]}\join[K](x,y)$ and
        $y\mathrel{\leq[K]}\join[K](x,y)$.
    \begin{subproof}
        Fix $x,y\in\carrier[K]$.
        Hence $x,y\in A$ by \auto.
        Hence $x,y\in\carrier[B]$ by \auto{subseteq}.
        $\join[B](x,y)\in A$ by \obv{boolean_subalgebra_join}.
        $\join[K](x,y)=\join[B](x,y)$ by
            \auto{boolean_subalgebra_structure_join_apply}.
        $x\mathrel{\leq[B]}\join[B](x,y)$ and
            $y\mathrel{\leq[B]}\join[B](x,y)$ by \auto{join_ub}.
        Follows by
            \auto{boolean_subalgebra_structure_order_iff}.
    \end{subproof}
    Show that for all $a,x,y\in\carrier[K]$ such that
        $x\mathrel{\leq[K]}a$ and $y\mathrel{\leq[K]}a$ we have
        $\join[K](x,y)\mathrel{\leq[K]}a$.
    \begin{subproof}
        Fix $a,x,y\in\carrier[K]$.
        Assume $x\mathrel{\leq[K]}a$ and $y\mathrel{\leq[K]}a$.
        Hence $a,x,y\in A$ by \auto.
        Hence $a,x,y\in\carrier[B]$ by \auto{subseteq}.
        Hence $x\mathrel{\leq[B]}a$ and $y\mathrel{\leq[B]}a$ by
            \auto{boolean_subalgebra_structure_order_iff}.
        Hence $\join[B](x,y)\mathrel{\leq[B]}a$ by \auto{join_lub}.
        $\join[B](x,y)\in A$ by \obv{boolean_subalgebra_join}.
        $\join[K](x,y)=\join[B](x,y)$ by
            \auto{boolean_subalgebra_structure_join_apply}.
        Follows by
            \auto{boolean_subalgebra_structure_order_iff}.
    \end{subproof}
    Hence $K$ is a join semilattice by
        \auto{join_semilattice,boolean_subalgebra_structure,
            boolean_algebraaggregateexact,
            boolean_algebraaggregateprojection_carrier,
            boolean_algebraaggregateprojection_leq,
            boolean_algebraaggregateprojection_join}.
    Hence $K$ is a lattice by
        \auto{lattice,boolean_subalgebra_structure,
            boolean_algebraaggregateexact,
            boolean_algebraaggregateprojection_carrier,
            boolean_algebraaggregateprojection_leq,
            boolean_algebraaggregateprojection_meet,
            boolean_algebraaggregateprojection_join}.

    $\latticeBottom[B],\latticeTop[B]\in A$ by
        \obv{boolean_subalgebra_bottom,boolean_subalgebra_top}.
    Hence $\latticeBottom[K],\latticeTop[K]\in\carrier[K]$.
    Show that for every $x\in\carrier[K]$ we have
        $\latticeBottom[K]\mathrel{\leq[K]}x$.
    \begin{subproof}
        Fix $x\in\carrier[K]$.
        Hence $x\in A$ by \auto.
        Hence $x\in\carrier[B]$ by \auto{subseteq}.
        $\latticeBottom[B]\mathrel{\leq[B]}x$ by
            \obv{lattice_bottom_leq}.
        Follows by
            \auto{boolean_subalgebra_structure_order_iff}.
    \end{subproof}
    Show that for every $x\in\carrier[K]$ we have
        $x\mathrel{\leq[K]}\latticeTop[K]$.
    \begin{subproof}
        Fix $x\in\carrier[K]$.
        Hence $x\in A$ by \auto.
        Hence $x\in\carrier[B]$ by \auto{subseteq}.
        $x\mathrel{\leq[B]}\latticeTop[B]$ by
            \obv{lattice_leq_top}.
        Follows by
            \auto{boolean_subalgebra_structure_order_iff}.
    \end{subproof}
    Hence $K$ is a bounded lattice by
        \auto{bounded_lattice,boolean_subalgebra_structure,
            boolean_algebraaggregateexact,
            boolean_algebraaggregateprojection_carrier,
            boolean_algebraaggregateprojection_leq,
            boolean_algebraaggregateprojection_meet,
            boolean_algebraaggregateprojection_join,
            boolean_algebraaggregateprojection_latticeBottom,
            boolean_algebraaggregateprojection_latticeTop}.

    Show that for all $x,y,z\in\carrier[K]$ we have
        $\meet[K](x,\join[K](y,z))
            =\join[K](\meet[K](x,y),\meet[K](x,z))$.
    \begin{subproof}
        Fix $x,y,z\in\carrier[K]$.
        Hence $x,y,z\in A$ by \auto.
        Hence $x,y,z\in\carrier[B]$ by \auto{subseteq}.
        $\join[B](y,z),\meet[B](x,y),\meet[B](x,z)\in A$ by
            \obv{boolean_subalgebra_join,boolean_subalgebra_meet}.
        $\join[K](y,z)=\join[B](y,z)$ and
            $\meet[K](x,y)=\meet[B](x,y)$ and
            $\meet[K](x,z)=\meet[B](x,z)$ by
            \auto{boolean_subalgebra_structure_meet_apply,
                boolean_subalgebra_structure_join_apply}.
        $\meet[K](x,\join[B](y,z))
            =\meet[B](x,\join[B](y,z))$ by
            \auto{boolean_subalgebra_structure_meet_apply}.
        $\join[K](\meet[B](x,y),\meet[B](x,z))
            =\join[B](\meet[B](x,y),\meet[B](x,z))$ by
            \auto{boolean_subalgebra_structure_join_apply}.
        Follows by \auto{meet_distributes_over_join}.
    \end{subproof}
    Hence $K$ is a distributive lattice by
        \auto{distributive_lattice,boolean_subalgebra_structure,
            boolean_algebraaggregateexact,
            boolean_algebraaggregateprojection_carrier,
            boolean_algebraaggregateprojection_leq,
            boolean_algebraaggregateprojection_meet,
            boolean_algebraaggregateprojection_join}.
    Hence $K$ is a bounded distributive lattice by
        \auto{bounded_distributive_lattice,
            boolean_subalgebra_structure,
            boolean_algebraaggregateexact}.

    $\booleanComplement[K]\in
        \funs{\carrier[K]}{\carrier[K]}$ by
        \auto{boolean_subalgebra_complement_restriction_type}.
    Show that for every $x\in\carrier[K]$ we have
        $\meet[K](x,\booleanComplement[K](x))
            =\latticeBottom[K]$.
    \begin{subproof}
        Fix $x\in\carrier[K]$.
        Hence $x\in A$ by \auto.
        Hence $x\in\carrier[B]$ by \auto{subseteq}.
        $\booleanComplement[B](x)\in A$ by
            \obv{boolean_subalgebra_complement}.
        $\booleanComplement[K](x)=\booleanComplement[B](x)$ by
            \auto{boolean_subalgebra_structure_complement_apply}.
        $\meet[K](x,\booleanComplement[B](x))
            =\meet[B](x,\booleanComplement[B](x))$ by
            \auto{boolean_subalgebra_structure_meet_apply}.
        Follows by \auto{boolean_meet_complement}.
    \end{subproof}
    Show that for every $x\in\carrier[K]$ we have
        $\join[K](x,\booleanComplement[K](x))
            =\latticeTop[K]$.
    \begin{subproof}
        Fix $x\in\carrier[K]$.
        Hence $x\in A$ by \auto.
        Hence $x\in\carrier[B]$ by \auto{subseteq}.
        $\booleanComplement[B](x)\in A$ by
            \obv{boolean_subalgebra_complement}.
        $\booleanComplement[K](x)=\booleanComplement[B](x)$ by
            \auto{boolean_subalgebra_structure_complement_apply}.
        $\join[K](x,\booleanComplement[B](x))
            =\join[B](x,\booleanComplement[B](x))$ by
            \auto{boolean_subalgebra_structure_join_apply}.
        Follows by \auto{boolean_join_complement}.
    \end{subproof}
    Hence $K$ is a Boolean algebra by
        \auto{boolean_algebra,boolean_subalgebra_structure,
            boolean_algebraaggregateexact,
            boolean_algebraaggregateprojection_carrier,
            boolean_algebraaggregateprojection_leq,
            boolean_algebraaggregateprojection_meet,
            boolean_algebraaggregateprojection_join,
            boolean_algebraaggregateprojection_latticeBottom,
            boolean_algebraaggregateprojection_latticeTop,
            boolean_algebraaggregateprojection_booleanComplement}.
    Follows by
        \auto{boolean_subalgebra_structure,
            boolean_algebraaggregateexact}.
\end{proof}
