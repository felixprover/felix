\begin{definition}\label{field_of_sets}
    $F$ is a field of sets on $X$ iff
        $F$ is a bounded set-lattice family on $X$ and
        for every $A\in F$ we have $X\setminus A\in F$.
\end{definition}

\begin{proposition}\label{field_of_sets_is_bounded_set_lattice_family}
    Suppose $F$ is a field of sets on $X$.
    Then $F$ is a bounded set-lattice family on $X$.
\end{proposition}
\begin{proof}
    Follows by \obv{field_of_sets}.
\end{proof}

\begin{proposition}\label{field_of_sets_subseteq_powerset}
    Suppose $F$ is a field of sets on $X$.
    Then $F\subseteq\pow{X}$.
\end{proposition}
\begin{proof}
    Follows by
        \obv{field_of_sets_is_bounded_set_lattice_family,
            bounded_set_lattice_family_subsets}.
\end{proof}

\begin{proposition}\label{field_of_sets_emptyset}
    Suppose $F$ is a field of sets on $X$.
    Then $\emptyset\in F$.
\end{proposition}
\begin{proof}
    Follows by
        \obv{field_of_sets_is_bounded_set_lattice_family,
            bounded_set_lattice_family_emptyset}.
\end{proof}

\begin{proposition}\label{field_of_sets_whole}
    Suppose $F$ is a field of sets on $X$.
    Then $X\in F$.
\end{proposition}
\begin{proof}
    Follows by
        \obv{field_of_sets_is_bounded_set_lattice_family,
            bounded_set_lattice_family_whole}.
\end{proof}

\begin{proposition}\label{field_of_sets_inter}
    Suppose $F$ is a field of sets on $X$.
    Let $A,C\in F$.

\end{proposition}
\begin{proof}

        \obv{field_of_sets_is_bounded_set_lattice_family,

\end{proof}

\begin{proposition}\label{field_of_sets_union}

\end{proposition}
\begin{proof}

\begin{proposition}\label{field_of_sets_complement}
    Suppose $F$ is a field of sets on $X$.
    Let $A\in F$.
    Then $X\setminus A\in F$.
\end{proposition}
\begin{proof}

\end{proof}

\begin{theorem}\label{field_of_sets_iff_boolean_subalgebra_of_powerset}
    $F$ is a field of sets on $X$ iff
        $\booleanSubalgebra{F}{\powersetBooleanAlgebra{X}}$.
\end{theorem}
\begin{proof}
    Let $P=\powersetBooleanAlgebra{X}$.
    $P$ is a Boolean algebra by
        \obv{powerset_boolean_algebra_is_boolean_algebra}.

    Show that if $F$ is a field of sets on $X$, then
        $\booleanSubalgebra{F}{P}$.
    \begin{subproof}
        Assume $F$ is a field of sets on $X$.
        $F\subseteq\pow{X}$ by \obv{field_of_sets_subseteq_powerset}.
        Hence $F\subseteq\carrier[P]$ by
            \auto{powerset_boolean_algebra_carrier}.
        $\emptyset,X\in F$ by
            \obv{field_of_sets_emptyset,field_of_sets_whole}.
        Hence $\latticeBottom[P],\latticeTop[P]\in F$ by
            \auto{powerset_boolean_algebra_bottom,
                powerset_boolean_algebra_top}.
        Show that for every $A\in F$ we have
            $\booleanComplement[P](A)\in F$.
        \begin{subproof}
            Fix $A\in F$.
            Hence $A\in\pow{X}$ by \obv{subseteq}.
            $\booleanComplement[P](A)=X\setminus A$ by
                \auto{powerset_boolean_algebra_complement_apply}.
            Follows by \auto{field_of_sets_complement}.
        \end{subproof}
        Show that for all $A,C\in F$ we have $\meet[P](A,C)\in F$.
        \begin{subproof}
            Fix $A,C\in F$.
            Hence $A,C\in\pow{X}$ by \obv{subseteq}.
            $\meet[P](A,C)=A\inter C$ by
                \auto{powerset_boolean_algebra_meet_apply}.
            Follows by \auto{field_of_sets_inter}.
        \end{subproof}
        Show that for all $A,C\in F$ we have $\join[P](A,C)\in F$.
        \begin{subproof}
            Fix $A,C\in F$.
            Hence $A,C\in\pow{X}$ by \obv{subseteq}.
            $\join[P](A,C)=A\union C$ by
                \auto{powerset_boolean_algebra_join_apply}.
            Follows by \auto{field_of_sets_union}.
        \end{subproof}
        Hence $\booleanSubalgebra{F}{P}$ by
            \obv{boolean_subalgebra}.
    \end{subproof}

    Show that if $\booleanSubalgebra{F}{P}$, then
        $F$ is a field of sets on $X$.
    \begin{subproof}
        Assume $\booleanSubalgebra{F}{P}$.
        $F\subseteq\carrier[P]$ by
            \obv{boolean_subalgebra_subseteq}.
        Hence $F\subseteq\pow{X}$ by
            \auto{powerset_boolean_algebra_carrier}.
        $\latticeBottom[P],\latticeTop[P]\in F$ by
            \obv{boolean_subalgebra_bottom,boolean_subalgebra_top}.
        Hence $\emptyset,X\in F$ by
            \auto{powerset_boolean_algebra_bottom,
                powerset_boolean_algebra_top}.
        Show that for all $A,C\in F$ we have $A\inter C\in F$.
        \begin{subproof}
            Fix $A,C\in F$.
            Hence $A,C\in\pow{X}$ by \obv{subseteq}.
            $\meet[P](A,C)\in F$ by \obv{boolean_subalgebra_meet}.
            Follows by
                \auto{powerset_boolean_algebra_meet_apply}.
        \end{subproof}
        Show that for all $A,C\in F$ we have $A\union C\in F$.
        \begin{subproof}
            Fix $A,C\in F$.
            Hence $A,C\in\pow{X}$ by \obv{subseteq}.
            $\join[P](A,C)\in F$ by \obv{boolean_subalgebra_join}.
            Follows by
                \auto{powerset_boolean_algebra_join_apply}.
        \end{subproof}
        Hence $F$ is a bounded set-lattice family on $X$ by
            \obv{bounded_set_lattice_family}.
        Show that for every $A\in F$ we have $X\setminus A\in F$.
        \begin{subproof}
            Fix $A\in F$.
            Hence $A\in\pow{X}$ by \obv{subseteq}.
            $\booleanComplement[P](A)\in F$ by
                \obv{boolean_subalgebra_complement}.
            Follows by
                \auto{powerset_boolean_algebra_complement_apply}.
        \end{subproof}
        Follows by \obv{field_of_sets}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{definition}\label{field_of_sets_algebra}
    $\fieldOfSetsAlgebra{F}{X}=
        \booleanSubalgebraStructure{\powersetBooleanAlgebra{X}}{F}$.
\end{definition}

\begin{proposition}\label{field_of_sets_algebra_carrier}
    $\carrier[\fieldOfSetsAlgebra{F}{X}]=F$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{field_of_sets_algebra,
            boolean_subalgebra_structure_carrier}.
\end{proof}

\begin{proposition}\label{field_of_sets_algebra_order_iff}
    Suppose $F$ is a field of sets on $X$.
    Let $A,C\in F$.
    Then $A\mathrel{\leq[\fieldOfSetsAlgebra{F}{X}]}C$ iff
        $A\subseteq C$.
\end{proposition}
\begin{proof}
    $\booleanSubalgebra{F}{\powersetBooleanAlgebra{X}}$ by
        \obv{field_of_sets_iff_boolean_subalgebra_of_powerset}.
    $A,C\in\pow{X}$ by
        \auto{field_of_sets_subseteq_powerset,subseteq}.
    $A\mathrel{\leq[\fieldOfSetsAlgebra{F}{X}]}C$ iff
        $A\mathrel{\leq[\powersetBooleanAlgebra{X}]}C$ by
        \auto{field_of_sets_algebra,
            boolean_subalgebra_structure_order_iff}.
    $A\mathrel{\leq[\powersetBooleanAlgebra{X}]}C$ iff
        $A\subseteq C$ by
        \obv{powerset_boolean_algebra_order_iff}.
\end{proof}

\begin{proposition}\label{field_of_sets_algebra_meet_apply}
    Suppose $F$ is a field of sets on $X$.
    Let $A,C\in F$.
    Then $\meet[\fieldOfSetsAlgebra{F}{X}](A,C)=A\inter C$.
\end{proposition}
\begin{proof}
    $\booleanSubalgebra{F}{\powersetBooleanAlgebra{X}}$ by
        \obv{field_of_sets_iff_boolean_subalgebra_of_powerset}.
    $A,C\in\pow{X}$ by
        \auto{field_of_sets_subseteq_powerset,subseteq}.
    $\meet[\fieldOfSetsAlgebra{F}{X}](A,C)
        =\meet[\powersetBooleanAlgebra{X}](A,C)$ by
        \auto{field_of_sets_algebra,
            boolean_subalgebra_structure_meet_apply}.
    $\meet[\powersetBooleanAlgebra{X}](A,C)=A\inter C$ by
        \obv{powerset_boolean_algebra_meet_apply}.
\end{proof}

\begin{proposition}\label{field_of_sets_algebra_join_apply}

\begin{proposition}\label{field_of_sets_algebra_complement_apply}

\begin{theorem}\label{field_of_sets_algebra_is_exact_boolean_algebra}
    Suppose $F$ is a field of sets on $X$.
    Then $\fieldOfSetsAlgebra{F}{X}$ is an exact Boolean algebra.
\end{theorem}
\begin{proof}
    $\booleanSubalgebra{F}{\powersetBooleanAlgebra{X}}$ by
        \obv{field_of_sets_iff_boolean_subalgebra_of_powerset}.
    Follows by
        \auto{field_of_sets_algebra,
            boolean_subalgebra_structure_is_exact_boolean_algebra}.
\end{proof}

\begin{classification}\label{field_of_sets_algebra_is_boolean_algebra}

    Then $\fieldOfSetsAlgebra{F}{X}$ is a Boolean algebra.

\begin{theorem}\label{field_of_sets_algebra_bounded_lattice_reduct}
    Suppose $F$ is a field of sets on $X$.
    Then
        $\reduct[bounded lattice]{\fieldOfSetsAlgebra{F}{X}}
            =\reduct[bounded lattice]{\setFamilyLattice{X}{F}}$.
\end{theorem}
\begin{proof}
    $\fieldOfSetsAlgebra{F}{X}$ is a Boolean algebra by
        \obv{field_of_sets_algebra_is_boolean_algebra}.
    Hence $\fieldOfSetsAlgebra{F}{X}$ is a bounded lattice by
        \obv{boolean_algebra_is_bounded_lattice}.
    Hence $\fieldOfSetsAlgebra{F}{X}$ is a lattice by
        \obv{bounded_lattice_is_lattice}.
    Hence $\fieldOfSetsAlgebra{F}{X}$ is a meet semilattice and
        $\fieldOfSetsAlgebra{F}{X}$ is a join semilattice by
        \obv{lattice_is_meet_semilattice,lattice_is_join_semilattice}.
    $F$ is a bounded set-lattice family on $X$ by
        \obv{field_of_sets_is_bounded_set_lattice_family}.
    $\setFamilyLattice{X}{F}$ is an exact bounded distributive lattice by
        \obv{set_family_lattice_is_exact_bounded_distributive_lattice}.
    Hence $\setFamilyLattice{X}{F}$ is a bounded distributive lattice.
    Hence $\setFamilyLattice{X}{F}$ is a bounded lattice by
        \obv{bounded_distributive_lattice_is_bounded_lattice}.
    $\reduct[bounded lattice]{\fieldOfSetsAlgebra{F}{X}}$
        is an exact bounded lattice by
        \obv{bounded_lattice_reduct_is_exact_bounded_lattice}.
    $\reduct[bounded lattice]{\setFamilyLattice{X}{F}}$
        is an exact bounded lattice by
        \obv{bounded_lattice_reduct_is_exact_bounded_lattice}.

    Show that
        $\leq[\fieldOfSetsAlgebra{F}{X}]=\subseteqrel{F}$.
    \begin{subproof}
        $F\subseteq\pow{X}$ by \obv{field_of_sets_subseteq_powerset}.
        $\leq[\fieldOfSetsAlgebra{F}{X}]
            =\relationTrace{
                \leq[\powersetBooleanAlgebra{X}]}{F}$ by
            \auto{field_of_sets_algebra,
                boolean_subalgebra_structure_order}.
        $\leq[\powersetBooleanAlgebra{X}]
            =\subseteqrel{\pow{X}}$ by
            \obv{powerset_boolean_algebra_order}.
        $F\times F$ is a relation by
            \auto{relation,times_elem_is_tuple}.
        $\relationTrace{\subseteqrel{\pow{X}}}{F}$ is a relation by
            \auto{relation_trace,subseteqrel_is_relation,
                inter_relations_is_relation}.
        $\subseteqrel{F}$ is a relation by \obv{subseteqrel_is_relation}.
        For all $A,C$ we have
            $A\mathrel{\relationTrace{
                \subseteqrel{\pow{X}}}{F}}C$ iff
            $A\mathrel{\subseteqrel{F}}C$ by
            \auto{relation_trace_iff,subseteqrel,pair_eq_iff,
                subseteq}.
        Hence $\relationTrace{\subseteqrel{\pow{X}}}{F}
            =\subseteqrel{F}$ by \obv{relext}.
    \end{subproof}

    Show that
        $\meet[\fieldOfSetsAlgebra{F}{X}]
            =\setFamilyMeetGraph{F}$.
    \begin{subproof}
        $\meet[\fieldOfSetsAlgebra{F}{X}]
            \in\funs{
                \carrier[\fieldOfSetsAlgebra{F}{X}]
                    \times\carrier[\fieldOfSetsAlgebra{F}{X}]}
                {\carrier[\fieldOfSetsAlgebra{F}{X}]}$ by
            \obv{meet_type}.
        $\meet[\fieldOfSetsAlgebra{F}{X}]
            \in\funs{F\times F}{F}$ by
            \obv{field_of_sets_algebra_carrier}.
        $\setFamilyMeetGraph{F}\in\funs{F\times F}{F}$ by
            \obv{set_family_meet_graph_type}.
        Hence $\meet[\fieldOfSetsAlgebra{F}{X}]$ and
            $\setFamilyMeetGraph{F}$ are functions and
            $\dom{\meet[\fieldOfSetsAlgebra{F}{X}]}
                =\dom{\setFamilyMeetGraph{F}}$ by
            \auto{funs_is_function,funs}.
        Show that for every $p\in
            \dom{\meet[\fieldOfSetsAlgebra{F}{X}]}$ we have
            $\apply{\meet[\fieldOfSetsAlgebra{F}{X}]}{p}
                =\apply{\setFamilyMeetGraph{F}}{p}$.
        \begin{subproof}
            Fix $p\in\dom{\meet[\fieldOfSetsAlgebra{F}{X}]}$.
            Hence $p\in F\times F$ by \auto{funs}.
            Take $A,C$ such that $p=(A,C)$ by
                \auto{times_elem_is_tuple}.
            Hence $A,C\in F$ by \auto{times_tuple_elim}.
            $\meet[\fieldOfSetsAlgebra{F}{X}](A,C)=A\inter C$ by
                \obv{field_of_sets_algebra_meet_apply}.
            $\apply{\setFamilyMeetGraph{F}}{(A,C)}=A\inter C$ by
                \obv{set_family_meet_graph_apply}.
        \end{subproof}
        Follows by \obv{funext_on_domain}.
    \end{subproof}

    Show that
        $\join[\fieldOfSetsAlgebra{F}{X}]
            =\setFamilyJoinGraph{F}$.
    \begin{subproof}
        $\join[\fieldOfSetsAlgebra{F}{X}]
            \in\funs{
                \carrier[\fieldOfSetsAlgebra{F}{X}]
                    \times\carrier[\fieldOfSetsAlgebra{F}{X}]}
                {\carrier[\fieldOfSetsAlgebra{F}{X}]}$ by
            \obv{join_type}.
        $\join[\fieldOfSetsAlgebra{F}{X}]
            \in\funs{F\times F}{F}$ by
            \obv{field_of_sets_algebra_carrier}.
        $\setFamilyJoinGraph{F}\in\funs{F\times F}{F}$ by
            \obv{set_family_join_graph_type}.
        Hence $\join[\fieldOfSetsAlgebra{F}{X}]$ and
            $\setFamilyJoinGraph{F}$ are functions and
            $\dom{\join[\fieldOfSetsAlgebra{F}{X}]}
                =\dom{\setFamilyJoinGraph{F}}$ by
            \auto{funs_is_function,funs}.
        Show that for every $p\in
            \dom{\join[\fieldOfSetsAlgebra{F}{X}]}$ we have
            $\apply{\join[\fieldOfSetsAlgebra{F}{X}]}{p}
                =\apply{\setFamilyJoinGraph{F}}{p}$.
        \begin{subproof}
            Fix $p\in\dom{\join[\fieldOfSetsAlgebra{F}{X}]}$.
            Hence $p\in F\times F$ by \auto{funs}.
            Take $A,C$ such that $p=(A,C)$ by
                \auto{times_elem_is_tuple}.
            Hence $A,C\in F$ by \auto{times_tuple_elim}.
            $\join[\fieldOfSetsAlgebra{F}{X}](A,C)=A\union C$ by
                \obv{field_of_sets_algebra_join_apply}.
            $\apply{\setFamilyJoinGraph{F}}{(A,C)}=A\union C$ by
                \obv{set_family_join_graph_apply}.
        \end{subproof}
        Follows by \obv{funext_on_domain}.
    \end{subproof}

    $\carrier[\fieldOfSetsAlgebra{F}{X}]=F$ and
        $\latticeBottom[\fieldOfSetsAlgebra{F}{X}]=\emptyset$ and
        $\latticeTop[\fieldOfSetsAlgebra{F}{X}]=X$ by
        \obv{field_of_sets_algebra_carrier,
            field_of_sets_algebra_bottom,field_of_sets_algebra_top}.
    $\carrier[\setFamilyLattice{X}{F}]=F$ and
        $\leq[\setFamilyLattice{X}{F}]=\subseteqrel{F}$ and
        $\meet[\setFamilyLattice{X}{F}]=\setFamilyMeetGraph{F}$ and
        $\join[\setFamilyLattice{X}{F}]=\setFamilyJoinGraph{F}$ and
        $\latticeBottom[\setFamilyLattice{X}{F}]=\emptyset$ and
        $\latticeTop[\setFamilyLattice{X}{F}]=X$ by
        \obv{set_family_lattice_carrier,set_family_lattice_order,
            set_family_lattice_meet,set_family_lattice_join,
            set_family_lattice_bottom,set_family_lattice_top}.
    Hence
        $\carrier[\reduct[bounded lattice]{
            \fieldOfSetsAlgebra{F}{X}}]
            =\carrier[\reduct[bounded lattice]{
                \setFamilyLattice{X}{F}}]$ and
        $\leq[\reduct[bounded lattice]{
            \fieldOfSetsAlgebra{F}{X}}]
            =\leq[\reduct[bounded lattice]{
                \setFamilyLattice{X}{F}}]$ and
        $\meet[\reduct[bounded lattice]{
            \fieldOfSetsAlgebra{F}{X}}]
            =\meet[\reduct[bounded lattice]{
                \setFamilyLattice{X}{F}}]$ and
        $\join[\reduct[bounded lattice]{
            \fieldOfSetsAlgebra{F}{X}}]
            =\join[\reduct[bounded lattice]{
                \setFamilyLattice{X}{F}}]$ and
        $\latticeBottom[\reduct[bounded lattice]{
            \fieldOfSetsAlgebra{F}{X}}]
            =\latticeBottom[\reduct[bounded lattice]{
                \setFamilyLattice{X}{F}}]$ and
        $\latticeTop[\reduct[bounded lattice]{
            \fieldOfSetsAlgebra{F}{X}}]
            =\latticeTop[\reduct[bounded lattice]{
                \setFamilyLattice{X}{F}}]$ by
        \auto{bounded_latticereductprojection_carrier,
            bounded_latticereductprojection_leq,
            bounded_latticereductprojection_meet,
            bounded_latticereductprojection_join,
            bounded_latticereductprojection_latticeBottom,
            bounded_latticereductprojection_latticeTop}.
    Follows by structure extensionality.
\end{proof}
