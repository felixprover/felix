\section{Pullback of Boolean ultrafilters}

\import{order/boolean-algebra/ultrafilter.tex}
\import{order/boolean-algebra/homomorphism.tex}
\import{order/lattice-filter-morphism.tex}

\begin{theorem}\label{boolean_hom_preimage_boolean_ultrafilter}
    Suppose $\booleanHom{f}{B}{C}$ and
        $\booleanUltrafilter{U}{C}$.
    Then $\booleanUltrafilter{\preimg{f}{U}}{B}$.
\end{theorem}
\begin{proof}
    $\boundedLatticeHom{f}{B}{C}$ by
        \obv{boolean_hom_is_bounded_lattice_hom}.
    $\latticeFilter{U}{C}$ by
        \obv{boolean_ultrafilter_is_lattice_filter}.
    Hence $\latticeFilter{\preimg{f}{U}}{B}$ by
        \obv{bounded_lattice_hom_preimage_lattice_filter}.
    Show that for every $a\in\carrier[B]$ we have
        $a\in\preimg{f}{U}$ or
            $\booleanComplement[B](a)\in\preimg{f}{U}$.
    \begin{subproof}
        Fix $a\in\carrier[B]$.
        $f(a)\in\carrier[C]$ by \obv{boolean_hom_apply_type}.
        $f(a)\in U$ or $\booleanComplement[C](f(a))\in U$ by
            \obv{boolean_ultrafilter_decides}.
        $\booleanComplement[B](a)\in\carrier[B]$ by
            \obv{boolean_complement_apply_type}.
        $f(\booleanComplement[B](a))
            =\booleanComplement[C](f(a))$ by
            \obv{boolean_hom_complement}.
        Follows by \auto{boolean_hom_preimage_iff}.
    \end{subproof}
    Follows by
        \obv{filter_and_decision_implies_boolean_ultrafilter}.
\end{proof}
