\begin{definition}\label{powerset_complement_graph}
    $\powersetComplementGraph{X}=
        \Apply{\graphOnDomainOp}
            {\pow{X},\fun{A:\Set}{X\setminus A}}$.
\end{definition}

\begin{proposition}\label{powerset_complement_graph_type}
    $\powersetComplementGraph{X}\in\funs{\pow{X}}{\pow{X}}$.
\end{proposition}
\begin{proof}
    Show that for every $A\in\pow{X}$ we have
        $X\setminus A\in\pow{X}$.
    \begin{subproof}
        Fix $A\in\pow{X}$.
        $X\setminus A\subseteq X$ by \obv{setminus_subseteq}.
        Follows by \obv{powerset_intro}.
    \end{subproof}
    Follows by
        \auto{powerset_complement_graph,logical_graph_on_domain_type}.
\end{proof}

\begin{proposition}\label{powerset_complement_graph_apply}
    Let $A\in\pow{X}$.
    Then $\apply{\powersetComplementGraph{X}}{A}=X\setminus A$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{powerset_complement_graph,logical_graph_on_domain_apply}.
\end{proof}

\begin{definition}\label{powerset_boolean_algebra}
    $\powersetBooleanAlgebra{X}=
        \structure[boolean algebra]{
            \pow{X},
            \subseteqrel{\pow{X}},
            \setFamilyMeetGraph{\pow{X}},
            \setFamilyJoinGraph{\pow{X}},
            \emptyset,
            X,
            \powersetComplementGraph{X}
        }$.
\end{definition}

            boolean_algebraaggregateprojection_carrier}.
\end{proof}

\begin{proposition}\label{powerset_boolean_algebra_order_iff}

\begin{theorem}\label{powerset_boolean_algebra_is_exact_boolean_algebra}
    $\powersetBooleanAlgebra{X}$ is an exact Boolean algebra.
\end{theorem}
\begin{proof}
    Let $P=\powersetBooleanAlgebra{X}$.
    $\carrier[P]=\pow{X}$ by \auto{powerset_boolean_algebra_carrier}.
    $\leq[P]=\subseteqrel{\pow{X}}$ by
        \auto{powerset_boolean_algebra_order}.
    $\meet[P]=\setFamilyMeetGraph{\pow{X}}$ by
        \auto{powerset_boolean_algebra_meet}.
    $\join[P]=\setFamilyJoinGraph{\pow{X}}$ by
        \auto{powerset_boolean_algebra_join}.
    $\latticeBottom[P]=\emptyset$ by
        \auto{powerset_boolean_algebra_bottom}.
    $\latticeTop[P]=X$ by \auto{powerset_boolean_algebra_top}.
    $\booleanComplement[P]=\powersetComplementGraph{X}$ by
        \auto{powerset_boolean_algebra_complement}.

    $\pow{X}$ is a bounded set-lattice family on $X$ by
        \obv{powerset_is_bounded_set_lattice_family}.
    $\subseteqrel{\pow{X}}$ is an order on $\pow{X}$ by
        \obv{subseteqrel_is_order}.
    Hence $\leq[P]$ is an order on $\carrier[P]$.
    $\leq[P]$ is antisymmetric.

    $\meet[P]\in
        \funs{\carrier[P]\times\carrier[P]}{\carrier[P]}$ by
        \auto{set_family_meet_graph_type}.
    Show that for all $A,C\in\carrier[P]$ we have
        $\meet[P](A,C)\mathrel{\leq[P]}A$ and
        $\meet[P](A,C)\mathrel{\leq[P]}C$.
    \begin{subproof}
        Fix $A,C\in\carrier[P]$.
        Hence $A,C\in\pow{X}$.
        $A\inter C\in\pow{X}$ by \obv{inter_powerset}.
        $\meet[P](A,C)=A\inter C$ by
            \auto{powerset_boolean_algebra_meet_apply}.
        $A\inter C\subseteq A$ and $A\inter C\subseteq C$ by
            \obv{inter_lower_left,inter_lower_right}.
        Follows by
            \auto{powerset_boolean_algebra_order_iff}.
    \end{subproof}
    Show that for all $D,A,C\in\carrier[P]$ such that
        $D\mathrel{\leq[P]}A$ and $D\mathrel{\leq[P]}C$ we have
        $D\mathrel{\leq[P]}\meet[P](A,C)$.
    \begin{subproof}
        Fix $D,A,C\in\carrier[P]$.
        Assume $D\mathrel{\leq[P]}A$ and $D\mathrel{\leq[P]}C$.
        Hence $D,A,C\in\pow{X}$.
        Hence $D\subseteq A$ and $D\subseteq C$ by
            \auto{powerset_boolean_algebra_order_iff}.
        Hence $D\subseteq A\inter C$ by \obv{subseteq_inter_iff}.
        $A\inter C\in\pow{X}$ by \obv{inter_powerset}.
        $\meet[P](A,C)=A\inter C$ by
            \auto{powerset_boolean_algebra_meet_apply}.
        Follows by
            \auto{powerset_boolean_algebra_order_iff}.
    \end{subproof}

    $\join[P]\in
        \funs{\carrier[P]\times\carrier[P]}{\carrier[P]}$ by
        \auto{set_family_join_graph_type}.
    Show that for all $A,C\in\carrier[P]$ we have
        $A\mathrel{\leq[P]}\join[P](A,C)$ and
        $C\mathrel{\leq[P]}\join[P](A,C)$.
    \begin{subproof}
        Fix $A,C\in\carrier[P]$.
        Hence $A,C\in\pow{X}$.
        $A\union C\in\pow{X}$ by
            \obv{bounded_set_lattice_family_union}.
        $\join[P](A,C)=A\union C$ by
            \auto{powerset_boolean_algebra_join_apply}.
        $A\subseteq A\union C$ and $C\subseteq A\union C$ by
            \obv{union_upper_left,union_upper_right}.
        Follows by
            \auto{powerset_boolean_algebra_order_iff}.
    \end{subproof}
    Show that for all $A,C,D\in\carrier[P]$ such that
        $A\mathrel{\leq[P]}D$ and $C\mathrel{\leq[P]}D$ we have
        $\join[P](A,C)\mathrel{\leq[P]}D$.
    \begin{subproof}
        Fix $A,C,D\in\carrier[P]$.
        Assume $A\mathrel{\leq[P]}D$ and $C\mathrel{\leq[P]}D$.
        Hence $A,C,D\in\pow{X}$.
        Hence $A\subseteq D$ and $C\subseteq D$ by
            \auto{powerset_boolean_algebra_order_iff}.
        Hence $A\union C\subseteq D$ by \auto{subseteq_union_iff}.
        $A\union C\in\pow{X}$ by
            \obv{bounded_set_lattice_family_union}.
        $\join[P](A,C)=A\union C$ by
            \auto{powerset_boolean_algebra_join_apply}.
        Follows by
            \auto{powerset_boolean_algebra_order_iff}.
    \end{subproof}

    $\emptyset\in\pow{X}$ and $X\in\pow{X}$ by
        \obv{powerset_bottom,powerset_top}.
    Hence $\latticeBottom[P],\latticeTop[P]\in\carrier[P]$.
    For every $A\in\pow{X}$ we have $\emptyset\subseteq A$ by
        \obv{emptyset_subseteq}.
    Hence for every $A\in\carrier[P]$ we have
        $\latticeBottom[P]\mathrel{\leq[P]}A$ by
        \auto{subseteqrel}.
    For every $A\in\pow{X}$ we have $A\subseteq X$ by
        \obv{powerset_elim}.
    Hence for every $A\in\carrier[P]$ we have
        $A\mathrel{\leq[P]}\latticeTop[P]$ by
        \auto{subseteqrel}.

    For all $A,C,D\in\carrier[P]$ we have
        $\meet[P](A,\join[P](C,D))
            =\join[P](\meet[P](A,C),\meet[P](A,D))$ by
        \auto{powerset_boolean_algebra_meet_apply,
            powerset_boolean_algebra_join_apply,
            inter_distrib_union,inter_powerset,
            bounded_set_lattice_family_union}.

    $\powersetComplementGraph{X}\in\funs{\pow{X}}{\pow{X}}$ by
        \obv{powerset_complement_graph_type}.
    Hence $\booleanComplement[P]\in
        \funs{\carrier[P]}{\carrier[P]}$ by \auto.
    Show that for every $A\in\carrier[P]$ we have
        $\meet[P](A,\booleanComplement[P](A))
            =\latticeBottom[P]$.
    \begin{subproof}
        Fix $A\in\carrier[P]$.
        Hence $A\in\pow{X}$ by \auto.
        $\booleanComplement[P](A)=X\setminus A$ by
            \auto{powerset_boolean_algebra_complement_apply}.
        $X\setminus A\in\pow{X}$ by
            \auto{setminus_subseteq,powerset_intro}.
        $\meet[P](A,X\setminus A)=A\inter(X\setminus A)$ by
            \auto{powerset_boolean_algebra_meet_apply}.
        $A\inter(X\setminus A)=\emptyset$ by
            \obv{setminus_disjoint}.
        Follows by \auto.
    \end{subproof}
    Show that for every $A\in\carrier[P]$ we have
        $\join[P](A,\booleanComplement[P](A))
            =\latticeTop[P]$.
    \begin{subproof}
        Fix $A\in\carrier[P]$.
        Hence $A\in\pow{X}$ by \auto.
        Hence $A\subseteq X$ by \obv{powerset_elim}.
        $\booleanComplement[P](A)=X\setminus A$ by
            \auto{powerset_boolean_algebra_complement_apply}.
        $X\setminus A\in\pow{X}$ by
            \auto{setminus_subseteq,powerset_intro}.
        $\join[P](A,X\setminus A)=A\union(X\setminus A)$ by
            \auto{powerset_boolean_algebra_join_apply}.
        $A\union(X\setminus A)=X$ by \obv{setminus_partition}.
        Follows by \auto.
    \end{subproof}
    Hence $P$ is a Boolean algebra by
        \auto{boolean_algebra,powerset_boolean_algebra,
            boolean_algebraaggregateexact,
            boolean_algebraaggregateprojection_carrier,
            boolean_algebraaggregateprojection_leq,
            boolean_algebraaggregateprojection_meet,
            boolean_algebraaggregateprojection_join,
            boolean_algebraaggregateprojection_latticeBottom,
            boolean_algebraaggregateprojection_latticeTop,
            boolean_algebraaggregateprojection_booleanComplement}.
    Follows by
        \auto{powerset_boolean_algebra,boolean_algebraaggregateexact}.
\end{proof}

\begin{theorem}\label{powerset_boolean_algebra_bounded_lattice_reduct}
    $\reduct[bounded lattice]{\powersetBooleanAlgebra{X}}
        =\reduct[bounded lattice]{\powersetLattice{X}}$.
\end{theorem}
\begin{proof}
    $\powersetBooleanAlgebra{X}$ is a Boolean algebra by
        \auto{powerset_boolean_algebra_is_exact_boolean_algebra}.
    Hence $\powersetBooleanAlgebra{X}$ is a bounded lattice by
        \obv{boolean_algebra_is_bounded_lattice}.
    $\powersetLattice{X}$ is an exact bounded distributive lattice by
        \obv{powerset_lattice_is_exact_bounded_distributive_lattice}.
    Hence $\powersetLattice{X}$ is a bounded distributive lattice.
    Hence $\powersetLattice{X}$ is a bounded lattice by
        \obv{bounded_distributive_lattice_is_bounded_lattice}.
    $\reduct[bounded lattice]{\powersetBooleanAlgebra{X}}$
        is an exact bounded lattice by
        \obv{bounded_lattice_reduct_is_exact_bounded_lattice}.
    $\reduct[bounded lattice]{\powersetLattice{X}}$
        is an exact bounded lattice by
        \obv{bounded_lattice_reduct_is_exact_bounded_lattice}.

    $\carrier[\powersetBooleanAlgebra{X}]=\pow{X}$ and
        $\leq[\powersetBooleanAlgebra{X}]=\subseteqrel{\pow{X}}$ and
        $\meet[\powersetBooleanAlgebra{X}]
            =\setFamilyMeetGraph{\pow{X}}$ and
        $\join[\powersetBooleanAlgebra{X}]
            =\setFamilyJoinGraph{\pow{X}}$ and
        $\latticeBottom[\powersetBooleanAlgebra{X}]=\emptyset$ and
        $\latticeTop[\powersetBooleanAlgebra{X}]=X$ by
        \auto{powerset_boolean_algebra_carrier,
            powerset_boolean_algebra_order,
            powerset_boolean_algebra_meet,
            powerset_boolean_algebra_join,
            powerset_boolean_algebra_bottom,
            powerset_boolean_algebra_top}.
    $\carrier[\powersetLattice{X}]=\pow{X}$ and
        $\leq[\powersetLattice{X}]=\subseteqrel{\pow{X}}$ and
        $\meet[\powersetLattice{X}]
            =\setFamilyMeetGraph{\pow{X}}$ and
        $\join[\powersetLattice{X}]
            =\setFamilyJoinGraph{\pow{X}}$ and
        $\latticeBottom[\powersetLattice{X}]=\emptyset$ and
        $\latticeTop[\powersetLattice{X}]=X$ by
        \auto{powerset_lattice,set_family_lattice_carrier,
            set_family_lattice_order,set_family_lattice_meet,
            set_family_lattice_join,set_family_lattice_bottom,
            set_family_lattice_top}.
    Hence
        $\carrier[\reduct[bounded lattice]{\powersetBooleanAlgebra{X}}]
            =\carrier[\reduct[bounded lattice]{\powersetLattice{X}}]$ and
        $\leq[\reduct[bounded lattice]{\powersetBooleanAlgebra{X}}]
            =\leq[\reduct[bounded lattice]{\powersetLattice{X}}]$ and
        $\meet[\reduct[bounded lattice]{\powersetBooleanAlgebra{X}}]
            =\meet[\reduct[bounded lattice]{\powersetLattice{X}}]$ and
        $\join[\reduct[bounded lattice]{\powersetBooleanAlgebra{X}}]
            =\join[\reduct[bounded lattice]{\powersetLattice{X}}]$ and
        $\latticeBottom[
            \reduct[bounded lattice]{\powersetBooleanAlgebra{X}}]
            =\latticeBottom[
                \reduct[bounded lattice]{\powersetLattice{X}}]$ and
        $\latticeTop[
            \reduct[bounded lattice]{\powersetBooleanAlgebra{X}}]
            =\latticeTop[
                \reduct[bounded lattice]{\powersetLattice{X}}]$ by
        \auto{bounded_latticereductprojection_carrier,
            bounded_latticereductprojection_leq,
            bounded_latticereductprojection_meet,
            bounded_latticereductprojection_join,
            bounded_latticereductprojection_latticeBottom,
            bounded_latticereductprojection_latticeTop}.
    Follows by structure extensionality.
\end{proof}

\begin{classification}\label{powerset_boolean_algebra_is_boolean_algebra}
    $\powersetBooleanAlgebra{X}$ is a Boolean algebra.
\end{classification}
\begin{proof}
    Follows by
        \auto{powerset_boolean_algebra_is_exact_boolean_algebra}.
\end{proof}
