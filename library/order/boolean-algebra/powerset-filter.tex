\section{Filters on powerset Boolean algebras}

\import{order/boolean-algebra/powerset.tex}
\import{order/boolean-algebra/ultrafilter.tex}
\import{set/ultrafilter.tex}

\begin{proposition}\label{proper_filter_is_lattice_filter_on_powerset_boolean_algebra}
    Let $F$ be a proper filter on $X$. Then
        $\latticeFilter{F}{\powersetBooleanAlgebra{X}}$.
\end{proposition}
\begin{proof}
    Let $P=\powersetBooleanAlgebra{X}$.
    $P$ is a Boolean algebra by
        \obv{powerset_boolean_algebra_is_boolean_algebra}.
    Hence $P$ is a bounded lattice by
        \obv{boolean_algebra_is_bounded_lattice}.
    $\carrier[P]=\pow{X}$ by
        \auto{powerset_boolean_algebra_carrier}.
    $\latticeTop[P]=X$ by
        \auto{powerset_boolean_algebra_top}.
    $\latticeBottom[P]=\emptyset$ by
        \auto{powerset_boolean_algebra_bottom}.

    $F$ is a filter on $X$ by \obv{proper_filter}.
    $F\subseteq\pow{X}$ by
        \auto{filter,pow_iff,subseteq}.
    Hence $F\subseteq\carrier[P]$.
    $\latticeTop[P]\in F$ by \obv{filter}.
    $\latticeBottom[P]\notin F$ by \obv{proper_filter}.

    Show that for all $A,C\in F$ we have
        $\meet[P](A,C)\in F$.
    \begin{subproof}
        Fix $A,C\in F$.
        Hence $A,C\in\pow{X}$ by \obv{subseteq}.
        $A\inter C\in F$ by \obv{filter}.
        $\meet[P](A,C)=A\inter C$ by
            \auto{powerset_boolean_algebra_meet_apply}.
    \end{subproof}

    Show that for every $A\in F$ we have
        for every $C\in\carrier[P]$ such that
            $A\mathrel{\leq[P]}C$ we have $C\in F$.
    \begin{subproof}
        Fix $A\in F$ and $C\in\carrier[P]$.
        Assume $A\mathrel{\leq[P]}C$.
        Hence $A,C\in\pow{X}$ by \obv{subseteq}.
        Hence $A\subseteq C$ by
            \auto{powerset_boolean_algebra_order_iff}.
        $A\subseteq X$ by \obv{powerset_elim}.
        $C\subseteq X$ by \obv{powerset_elim}.
        Follows by \obv{filter}.
    \end{subproof}
    Hence $\latticeFilter{F}{P}$ by \obv{lattice_filter}.
\end{proof}

\begin{proposition}\label{lattice_filter_on_powerset_boolean_algebra_is_proper_filter}
    Assume $\latticeFilter{F}{\powersetBooleanAlgebra{X}}$.
    Then $F$ is a proper filter on $X$.
\end{proposition}
\begin{proof}
    Let $P=\powersetBooleanAlgebra{X}$.
    $\carrier[P]=\pow{X}$ by
        \auto{powerset_boolean_algebra_carrier}.
    $\latticeTop[P]=X$ by
        \auto{powerset_boolean_algebra_top}.
    $\latticeBottom[P]=\emptyset$ by
        \auto{powerset_boolean_algebra_bottom}.
    $\latticeFilter{F}{P}$.

    $F\subseteq\carrier[P]$ by
        \obv{lattice_filter_subseteq_carrier}.
    Hence $F\subseteq\pow{X}$.
    Thus $F$ is a family of subsets of $X$ by
        \auto{pow_iff,subseteq}.
    $X\in F$ and $\emptyset\notin F$ by
        \obv{lattice_filter_top,lattice_filter_bottom_notin}.

    Show that $F$ is closed under binary intersections.
    \begin{subproof}
        Fix $A,C\in F$.
        Hence $A,C\in\pow{X}$ by \obv{subseteq}.
        $\meet[P](A,C)\in F$ by \obv{lattice_filter_meet}.
        $\meet[P](A,C)=A\inter C$ by
            \auto{powerset_boolean_algebra_meet_apply}.
    \end{subproof}

    Show that $F$ is upward-closed in $X$.
    \begin{subproof}
        Fix $A,C$.
        Assume $A\subseteq C\subseteq X$ and $A\in F$.
        $C\in\pow{X}$ by \obv{pow_iff,subseteq}.
        Hence $C\in\carrier[P]$.
        $A\in\pow{X}$ by \obv{subseteq}.
        Hence $A\mathrel{\leq[P]}C$ by
            \obv{powerset_boolean_algebra_order_iff}.
        Follows by \obv{lattice_filter_upward}.
    \end{subproof}
    Hence $F$ is a filter on $X$ by \obv{filter}.
    Follows by \obv{proper_filter}.
\end{proof}

\begin{theorem}\label{proper_filter_iff_lattice_filter_on_powerset_boolean_algebra}
    $F$ is a proper filter on $X$ iff
        $\latticeFilter{F}{\powersetBooleanAlgebra{X}}$.
\end{theorem}
\begin{proof}
    Follows by
        \auto{proper_filter_is_lattice_filter_on_powerset_boolean_algebra,
              lattice_filter_on_powerset_boolean_algebra_is_proper_filter}.
\end{proof}

\begin{theorem}\label{ultrafilter_iff_boolean_ultrafilter_on_powerset_boolean_algebra}
    $U$ is an ultrafilter on $X$ iff
        $\booleanUltrafilter{U}{\powersetBooleanAlgebra{X}}$.
\end{theorem}
\begin{proof}
    Let $P=\powersetBooleanAlgebra{X}$.
    $P$ is a Boolean algebra by
        \obv{powerset_boolean_algebra_is_boolean_algebra}.
    $\carrier[P]=\pow{X}$ by
        \auto{powerset_boolean_algebra_carrier}.

    Show that if $U$ is an ultrafilter on $X$, then
        $\booleanUltrafilter{U}{P}$.
    \begin{subproof}
        Assume $U$ is an ultrafilter on $X$.
        Hence $U$ is a proper filter on $X$ by
            \auto{ultrafilter,maximalfilter}.
        Thus $\latticeFilter{U}{P}$ by
            \obv{proper_filter_iff_lattice_filter_on_powerset_boolean_algebra}.
        Show that for every $A\in\carrier[P]$ we have
            $A\in U$ or $\booleanComplement[P](A)\in U$.
        \begin{subproof}
            Fix $A\in\carrier[P]$.
            Hence $A\in\pow{X}$.
            Thus $A\subseteq X$ by \obv{powerset_elim}.
            $A\in U$ or $X\setminus A\in U$ by
                \obv{ultrafilter_decides_subset}.
            $\booleanComplement[P](A)=X\setminus A$ by
                \auto{powerset_boolean_algebra_complement_apply}.
        \end{subproof}
        Follows by
            \obv{filter_and_decision_implies_boolean_ultrafilter}.
    \end{subproof}

    Show that if $\booleanUltrafilter{U}{P}$, then
        $U$ is an ultrafilter on $X$.
    \begin{subproof}
        Assume $\booleanUltrafilter{U}{P}$.
        Hence $\latticeFilter{U}{P}$ by
            \obv{boolean_ultrafilter_is_lattice_filter}.
        Thus $U$ is a proper filter on $X$ by
            \obv{proper_filter_iff_lattice_filter_on_powerset_boolean_algebra}.
        Show that for every $A\subseteq X$ we have
            $A\in U$ or $X\setminus A\in U$.
        \begin{subproof}
            Fix $A\subseteq X$.
            Hence $A\in\pow{X}$ by \obv{pow_iff,subseteq}.
            Thus $A\in\carrier[P]$.
            $A\in U$ or $\booleanComplement[P](A)\in U$ by
                \obv{boolean_ultrafilter_decides}.
            $\booleanComplement[P](A)=X\setminus A$ by
                \auto{powerset_boolean_algebra_complement_apply}.
        \end{subproof}
        Follows by \obv{filter_decision_implies_ultrafilter}.
    \end{subproof}
    Follows by \auto.
\end{proof}
