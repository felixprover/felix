\section{Boolean algebra homomorphisms}

\import{order/boolean-algebra/subalgebra.tex}
\import{order/lattice-morphism.tex}

\begin{definition}\label{boolean_hom}
    $\booleanHom{f}{B}{C}$ iff
        $B$ is a Boolean algebra and $C$ is a Boolean algebra and
        $\boundedLatticeHom{f}{B}{C}$ and
        for every $x\in\carrier[B]$ we have
            $f(\booleanComplement[B](x))
                =\booleanComplement[C](f(x))$.
\end{definition}

\begin{proposition}\label{boolean_hom_source_boolean_algebra}
    Suppose $\booleanHom{f}{B}{C}$.
    Then $B$ is a Boolean algebra.
\end{proposition}
\begin{proof}
    Follows by \obv{boolean_hom}.
\end{proof}

\begin{proposition}\label{boolean_hom_target_boolean_algebra}
    Suppose $\booleanHom{f}{B}{C}$.
    Then $C$ is a Boolean algebra.
\end{proposition}
\begin{proof}
    Follows by \obv{boolean_hom}.
\end{proof}

\begin{proposition}\label{boolean_hom_is_bounded_lattice_hom}
    Suppose $\booleanHom{f}{B}{C}$.
    Then $\boundedLatticeHom{f}{B}{C}$.
\end{proposition}
\begin{proof}
    Follows by \obv{boolean_hom}.
\end{proof}

\begin{proposition}\label{boolean_hom_is_lattice_hom}
    Suppose $\booleanHom{f}{B}{C}$.
    Then $\latticeHom{f}{B}{C}$.
\end{proposition}
\begin{proof}
    Follows by
        \obv{boolean_hom_is_bounded_lattice_hom,
            bounded_lattice_hom_is_lattice_hom}.
\end{proof}

\begin{proposition}\label{boolean_hom_type}
    Suppose $\booleanHom{f}{B}{C}$.
    Then $f\in\funs{\carrier[B]}{\carrier[C]}$.
\end{proposition}
\begin{proof}
    Follows by
        \obv{boolean_hom_is_bounded_lattice_hom,
            bounded_lattice_hom_type}.
\end{proof}

\begin{proposition}\label{boolean_hom_is_function}
    Suppose $\booleanHom{f}{B}{C}$.
    Then $f$ is a function.
\end{proposition}
\begin{proof}
    Follows by \auto{boolean_hom_type,funs_is_function}.
\end{proof}

\begin{proposition}\label{boolean_hom_apply_type}
    Suppose $\booleanHom{f}{B}{C}$.
    Let $x\in\carrier[B]$.
    Then $f(x)\in\carrier[C]$.
\end{proposition}
\begin{proof}
    Follows by \auto{boolean_hom_type,funs_type_apply}.
\end{proof}

\begin{proposition}\label{boolean_hom_bottom}

\end{proposition}
\begin{proof}

\begin{proposition}\label{boolean_hom_top}

\begin{proposition}\label{boolean_hom_meet}
    Suppose $\booleanHom{f}{B}{C}$.

\begin{proposition}\label{boolean_hom_complement}

\end{proof}

\begin{theorem}\label{boolean_hom_order_preserving}
    Suppose $\booleanHom{f}{B}{C}$.
    Then
        $\orderPreserving

\end{theorem}
\begin{proof}

            lattice_hom_order_preserving}.
\end{proof}

\begin{proposition}\label{identity_boolean_hom}
    Let $B$ be a Boolean algebra.
    Then $\booleanHom{\identity{\carrier[B]}}{B}{B}$.
\end{proposition}
\begin{proof}
    $B$ is a bounded lattice by
        \obv{boolean_algebra_is_bounded_lattice}.
    Hence $\boundedLatticeHom{\identity{\carrier[B]}}{B}{B}$ by
        \obv{identity_bounded_lattice_hom}.
    Show that for every $x\in\carrier[B]$ we have
        $(\identity{\carrier[B]})(\booleanComplement[B](x))
            =\booleanComplement[B]
                ((\identity{\carrier[B]})(x))$.
    \begin{subproof}
        Fix $x\in\carrier[B]$.
        $\booleanComplement[B](x)\in\carrier[B]$ by
            \obv{boolean_complement_apply_type}.
        Follows by \obv{id_apply}.
    \end{subproof}
    Follows by \obv{boolean_hom}.
\end{proof}

\begin{theorem}\label{boolean_hom_composition}
    Suppose $\booleanHom{f}{B}{C}$ and
        $\booleanHom{g}{C}{D}$.
    Then $\booleanHom{g\circ f}{B}{D}$.
\end{theorem}
\begin{proof}
    $\boundedLatticeHom{f}{B}{C}$ and
        $\boundedLatticeHom{g}{C}{D}$ by
        \obv{boolean_hom_is_bounded_lattice_hom}.
    Hence $\boundedLatticeHom{g\circ f}{B}{D}$ by
        \obv{bounded_lattice_hom_composition}.
    $f\in\funs{\carrier[B]}{\carrier[C]}$ and
        $g\in\funs{\carrier[C]}{\carrier[D]}$ by
        \obv{boolean_hom_type}.
    Hence $f$ and $g$ are functions by \obv{funs_is_function}.
    $g$ is composable with $f$ by \auto{funs_ran,funs}.
    Show that for every $x\in\carrier[B]$ we have
        $(g\circ f)(\booleanComplement[B](x))
            =\booleanComplement[D]((g\circ f)(x))$.
    \begin{subproof}
        Fix $x\in\carrier[B]$.
        $\booleanComplement[B](x)\in\carrier[B]$ by
            \obv{boolean_complement_apply_type}.
        $f(x)\in\carrier[C]$ by \obv{boolean_hom_apply_type}.
        $(g\circ f)(\booleanComplement[B](x))
            =g(f(\booleanComplement[B](x)))$ by
            \obv{circ_apply,funs}.
        Hence $(g\circ f)(\booleanComplement[B](x))
            =g(\booleanComplement[C](f(x)))$ by
            \obv{boolean_hom_complement}.
        Hence $(g\circ f)(\booleanComplement[B](x))
            =\booleanComplement[D](g(f(x)))$ by
            \obv{boolean_hom_complement}.
        $(g\circ f)(x)=g(f(x))$ by \auto{circ_apply,funs}.
    \end{subproof}
    Follows by \obv{boolean_hom}.
\end{proof}

\begin{theorem}\label{boolean_hom_extensionality}
    Suppose $\booleanHom{f}{B}{C}$ and
        $\booleanHom{g}{B}{C}$.
    Suppose for every $x\in\carrier[B]$ we have $f(x)=g(x)$.
    Then $f=g$.
\end{theorem}
\begin{proof}
    $\latticeHom{f}{B}{C}$ and $\latticeHom{g}{B}{C}$ by
        \obv{boolean_hom_is_lattice_hom}.
    Follows by \obv{lattice_hom_extensionality}.
\end{proof}

\subsection{Images and preimages}

\begin{proposition}\label{boolean_hom_image_iff}
    Suppose $\booleanHom{f}{B}{C}$ and $A\subseteq\carrier[B]$.
    Then $y\in\img{f}{A}$ iff
        there exists $x\in A$ such that $f(x)=y$.
\end{proposition}
\begin{proof}
    $f$ is a function and $\dom{f}=\carrier[B]$ by
        \auto{boolean_hom_type,funs_elim}.
    Show that if $y\in\img{f}{A}$, then there exists $x\in A$
        such that $f(x)=y$.
    \begin{subproof}
        Assume $y\in\img{f}{A}$.
        Take $x\in A$ such that $(x,y)\in f$ by \obv{img_iff}.
        $f(x)=y$ by \obv{function_apply_intro}.
    \end{subproof}
    Show that if there exists $x\in A$ such that $f(x)=y$, then
        $y\in\img{f}{A}$.
    \begin{subproof}
        Assume that there exists $x\in A$ such that $f(x)=y$.
        Take $x\in A$ such that $f(x)=y$.
        Hence $x\in\dom{f}$ by \obv{subseteq}.
        Hence $(x,y)\in f$ by \auto{function_apply_elim}.
        Follows by \auto{img_iff}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{proposition}\label{boolean_hom_preimage_iff}
    Suppose $\booleanHom{f}{B}{C}$.
    Then $x\in\preimg{f}{A}$ iff
        $x\in\carrier[B]$ and $f(x)\in A$.
\end{proposition}
\begin{proof}
    $f$ is a function and $\dom{f}=\carrier[B]$ by
        \auto{boolean_hom_type,funs_elim}.
    Show that if $x\in\preimg{f}{A}$, then
        $x\in\carrier[B]$ and $f(x)\in A$.
    \begin{subproof}
        Assume $x\in\preimg{f}{A}$.
        Take $y\in A$ such that $(x,y)\in f$ by
            \obv{preimg_iff}.
        $x\in\dom{f}$ and $f(x)=y$ by
            \obv{dom_iff,function_apply_intro}.
    \end{subproof}
    Show that if $x\in\carrier[B]$ and $f(x)\in A$, then
        $x\in\preimg{f}{A}$.
    \begin{subproof}
        Assume $x\in\carrier[B]$ and $f(x)\in A$.
        Hence $(x,f(x))\in f$ by \obv{function_apply_elim}.
        Follows by \obv{preimg_iff}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{theorem}\label{boolean_hom_subalgebra_image}
    Suppose $\booleanHom{f}{B}{C}$ and
        $\booleanSubalgebra{A}{B}$.
    Then $\booleanSubalgebra{\img{f}{A}}{C}$.
\end{theorem}
\begin{proof}
    Let $J=\img{f}{A}$.
    $C$ is a Boolean algebra by
        \obv{boolean_hom_target_boolean_algebra}.
    $A\subseteq\carrier[B]$ by \obv{boolean_subalgebra_subseteq}.
    Show that $J\subseteq\carrier[C]$.
    \begin{subproof}
        Show that for every $z\in J$ we have $z\in\carrier[C]$.
        \begin{subproof}
            Fix $z\in J$.
            Take $x\in A$ such that $f(x)=z$ by
                \auto{boolean_hom_image_iff}.
            Hence $x\in\carrier[B]$ by \obv{subseteq}.
            Follows by \auto{boolean_hom_apply_type}.
        \end{subproof}
        Follows by \obv{subseteq}.
    \end{subproof}
    $\latticeBottom[B],\latticeTop[B]\in A$ by
        \obv{boolean_subalgebra_bottom,boolean_subalgebra_top}.
    $f(\latticeBottom[B])=\latticeBottom[C]$ and
        $f(\latticeTop[B])=\latticeTop[C]$ by
        \obv{boolean_hom_bottom,boolean_hom_top}.
    Hence $\latticeBottom[C],\latticeTop[C]\in J$ by
        \auto{boolean_hom_image_iff}.
    Show that for every $z\in J$ we have
        $\booleanComplement[C](z)\in J$.
    \begin{subproof}
        Fix $z\in J$.
        Take $x\in A$ such that $f(x)=z$ by
            \auto{boolean_hom_image_iff}.
        $\booleanComplement[B](x)\in A$ by
            \obv{boolean_subalgebra_complement}.
        $x\in\carrier[B]$ by \obv{subseteq}.
        $f(\booleanComplement[B](x))
            =\booleanComplement[C](f(x))$ by
            \obv{boolean_hom_complement}.
        Hence $f(\booleanComplement[B](x))
            =\booleanComplement[C](z)$.
        Follows by \auto{boolean_hom_image_iff}.
    \end{subproof}
    Show that for all $z,w\in J$ we have $\meet[C](z,w)\in J$.
    \begin{subproof}
        Fix $z,w\in J$.
        Take $x\in A$ such that $f(x)=z$ by
            \auto{boolean_hom_image_iff}.
        Take $y\in A$ such that $f(y)=w$ by
            \auto{boolean_hom_image_iff}.
        $\meet[B](x,y)\in A$ by \obv{boolean_subalgebra_meet}.
        $x,y\in\carrier[B]$ by \obv{subseteq}.
        $f(\meet[B](x,y))=\meet[C](f(x),f(y))$ by
            \obv{boolean_hom_meet}.
        Hence $f(\meet[B](x,y))=\meet[C](z,w)$.
        Follows by \auto{boolean_hom_image_iff}.
    \end{subproof}
    Show that for all $z,w\in J$ we have $\join[C](z,w)\in J$.
    \begin{subproof}
        Fix $z,w\in J$.
        Take $x\in A$ such that $f(x)=z$ by
            \auto{boolean_hom_image_iff}.
        Take $y\in A$ such that $f(y)=w$ by
            \auto{boolean_hom_image_iff}.
        $\join[B](x,y)\in A$ by \obv{boolean_subalgebra_join}.
        $x,y\in\carrier[B]$ by \obv{subseteq}.
        $f(\join[B](x,y))=\join[C](f(x),f(y))$ by
            \obv{boolean_hom_join}.
        Hence $f(\join[B](x,y))=\join[C](z,w)$.
        Follows by \auto{boolean_hom_image_iff}.
    \end{subproof}
    Follows by \auto{boolean_subalgebra}.
\end{proof}

\begin{corollary}\label{boolean_hom_image_boolean_subalgebra}
    Suppose $\booleanHom{f}{B}{C}$.
    Then $\booleanSubalgebra{\img{f}{\carrier[B]}}{C}$.
\end{corollary}
\begin{proof}
    Hence $\booleanSubalgebra{\carrier[B]}{B}$ by
        \obv{full_carrier_boolean_subalgebra}.
    Follows by \obv{boolean_hom_subalgebra_image}.
\end{proof}

\begin{theorem}\label{boolean_hom_subalgebra_preimage}
    Suppose $\booleanHom{f}{B}{C}$ and
        $\booleanSubalgebra{A}{C}$.
    Then $\booleanSubalgebra{\preimg{f}{A}}{B}$.
\end{theorem}
\begin{proof}
    Let $P=\preimg{f}{A}$.
    $B$ is a Boolean algebra by
        \obv{boolean_hom_source_boolean_algebra}.
    $P\subseteq\carrier[B]$ by
        \auto{subseteq,boolean_hom_preimage_iff}.
    $f(\latticeBottom[B])=\latticeBottom[C]$ and
        $f(\latticeTop[B])=\latticeTop[C]$ by
        \obv{boolean_hom_bottom,boolean_hom_top}.
    $\latticeBottom[B],\latticeTop[B]\in\carrier[B]$ by
        \obv{boolean_algebra_is_bounded_lattice,
            lattice_bottom_type,lattice_top_type}.
    $\latticeBottom[C],\latticeTop[C]\in A$ by
        \obv{boolean_subalgebra_bottom,boolean_subalgebra_top}.
    Hence $\latticeBottom[B],\latticeTop[B]\in P$ by
        \auto{boolean_hom_preimage_iff}.
    Show that for every $x\in P$ we have
        $\booleanComplement[B](x)\in P$.
    \begin{subproof}
        Fix $x\in P$.
        Hence $x\in\carrier[B]$ and $f(x)\in A$ by
            \auto{boolean_hom_preimage_iff}.
        $\booleanComplement[B](x)\in\carrier[B]$ by
            \obv{boolean_complement_apply_type}.
        $f(\booleanComplement[B](x))
            =\booleanComplement[C](f(x))$ by
            \obv{boolean_hom_complement}.
        $\booleanComplement[C](f(x))\in A$ by
            \obv{boolean_subalgebra_complement}.
        Follows by \auto{boolean_hom_preimage_iff}.
    \end{subproof}
    Show that for all $x,y\in P$ we have $\meet[B](x,y)\in P$.
    \begin{subproof}
        Fix $x,y\in P$.
        Hence $x,y\in\carrier[B]$ and $f(x),f(y)\in A$ by
            \auto{boolean_hom_preimage_iff}.
        $\meet[B](x,y)\in\carrier[B]$ by
            \obv{boolean_algebra_is_lattice,
                lattice_is_meet_semilattice,meet_type}.
        $f(\meet[B](x,y))=\meet[C](f(x),f(y))$ by
            \obv{boolean_hom_meet}.
        $\meet[C](f(x),f(y))\in A$ by
            \obv{boolean_subalgebra_meet}.
        Follows by \auto{boolean_hom_preimage_iff}.
    \end{subproof}
    Show that for all $x,y\in P$ we have $\join[B](x,y)\in P$.
    \begin{subproof}
        Fix $x,y\in P$.
        Hence $x,y\in\carrier[B]$ and $f(x),f(y)\in A$ by
            \auto{boolean_hom_preimage_iff}.
        $\join[B](x,y)\in\carrier[B]$ by
            \obv{boolean_algebra_is_lattice,
                lattice_is_join_semilattice,join_type}.
        $f(\join[B](x,y))=\join[C](f(x),f(y))$ by
            \obv{boolean_hom_join}.
        $\join[C](f(x),f(y))\in A$ by
            \obv{boolean_subalgebra_join}.
        Follows by \auto{boolean_hom_preimage_iff}.
    \end{subproof}
    Follows by \auto{boolean_subalgebra}.
\end{proof}

\begin{theorem}\label{boolean_hom_codomain_corestriction}
    Suppose $\booleanHom{f}{B}{C}$ and
        $\booleanSubalgebra{A}{C}$.
    Suppose $\img{f}{\carrier[B]}\subseteq A$.
    Then
        $\booleanHom{f}{B}{\booleanSubalgebraStructure{C}{A}}$.
\end{theorem}
\begin{proof}
    Let $T=\booleanSubalgebraStructure{C}{A}$.
    $T$ is a Boolean algebra by notion.
    Hence $B$ is a bounded lattice and $T$ is a bounded lattice by
        \obv{boolean_algebra_is_bounded_lattice}.
    Hence $B$ is a lattice and $T$ is a lattice by
        \obv{bounded_lattice_is_lattice}.
    $\carrier[T]=A$ by
        \auto{boolean_subalgebra_structure_carrier}.
    $f\in\funs{\carrier[B]}{\carrier[C]}$ by
        \obv{boolean_hom_type}.
    Hence $f$ is a function and $\dom{f}=\carrier[B]$ by
        \auto{funs_elim}.
    Show that $f$ is a function to $A$.
    \begin{subproof}
        Show that for every $x\in\dom{f}$ we have $f(x)\in A$.
        \begin{subproof}
            Fix $x\in\dom{f}$.
            Hence $x\in\carrier[B]$.
            $(x,f(x))\in f$ by \obv{function_apply_elim}.
            Hence $f(x)\in\img{f}{\carrier[B]}$ by
                \auto{img_iff}.
            Follows by \obv{subseteq}.
        \end{subproof}
    \end{subproof}
    Hence $f\in\funs{\carrier[B]}{A}$ by \obv{funs_intro}.
    Hence $f\in\funs{\carrier[B]}{\carrier[T]}$.
    Show that for all $x,y\in\carrier[B]$ we have
        $f(\meet[B](x,y))=\meet[T](f(x),f(y))$ and
        $f(\join[B](x,y))=\join[T](f(x),f(y))$.
    \begin{subproof}
        Fix $x,y\in\carrier[B]$.
        $f(x),f(y)\in A$ by \obv{funs_type_apply}.
        $f(\meet[B](x,y))=\meet[C](f(x),f(y))$ by
            \obv{boolean_hom_meet}.
        $\meet[T](f(x),f(y))=\meet[C](f(x),f(y))$ by
            \auto{boolean_subalgebra_structure_meet_apply}.
        $f(\join[B](x,y))=\join[C](f(x),f(y))$ by
            \obv{boolean_hom_join}.
        $\join[T](f(x),f(y))=\join[C](f(x),f(y))$ by
            \auto{boolean_subalgebra_structure_join_apply}.
    \end{subproof}
    Hence $\latticeHom{f}{B}{T}$ by \obv{lattice_hom}.
    $f(\latticeBottom[B])=\latticeBottom[C]$ and
        $f(\latticeTop[B])=\latticeTop[C]$ by
        \obv{boolean_hom_bottom,boolean_hom_top}.
    $\latticeBottom[T]=\latticeBottom[C]$ and
        $\latticeTop[T]=\latticeTop[C]$ by
        \auto{boolean_subalgebra_structure_bottom,
            boolean_subalgebra_structure_top}.
    Hence $\boundedLatticeHom{f}{B}{T}$ by
        \obv{bounded_lattice_hom}.
    Show that for every $x\in\carrier[B]$ we have
        $f(\booleanComplement[B](x))
            =\booleanComplement[T](f(x))$.
    \begin{subproof}
        Fix $x\in\carrier[B]$.
        $f(x)\in A$ by \obv{funs_type_apply}.
        $f(\booleanComplement[B](x))
            =\booleanComplement[C](f(x))$ by
            \obv{boolean_hom_complement}.
        $\booleanComplement[T](f(x))
            =\booleanComplement[C](f(x))$ by
            \auto{boolean_subalgebra_structure_complement_apply}.
    \end{subproof}
    Follows by \auto{boolean_hom}.
\end{proof}

\subsection{Boolean embeddings}

\begin{definition}\label{boolean_embedding}
    $\booleanEmbedding{f}{B}{C}$ iff
        $\booleanHom{f}{B}{C}$ and
        $f\in\inj{\carrier[B]}{\carrier[C]}$.
\end{definition}

\begin{proposition}\label{boolean_embedding_is_hom}
    Suppose $\booleanEmbedding{f}{B}{C}$.
    Then $\booleanHom{f}{B}{C}$.
\end{proposition}
\begin{proof}
    Follows by \obv{boolean_embedding}.
\end{proof}

\begin{proposition}\label{boolean_embedding_injective}
    Suppose $\booleanEmbedding{f}{B}{C}$.
    Then $f\in\inj{\carrier[B]}{\carrier[C]}$.
\end{proposition}
\begin{proof}
    Follows by \obv{boolean_embedding}.
\end{proof}

\begin{proposition}\label{identity_boolean_embedding}
    Let $B$ be a Boolean algebra.
    Then $\booleanEmbedding{\identity{\carrier[B]}}{B}{B}$.
\end{proposition}
\begin{proof}
    $\booleanHom{\identity{\carrier[B]}}{B}{B}$ by
        \obv{identity_boolean_hom}.
    $\identity{\carrier[B]}\in
        \funs{\carrier[B]}{\carrier[B]}$ by \obv{id_elem_funs}.
    Hence $\identity{\carrier[B]}$ is a function and
        $\dom{\identity{\carrier[B]}}=\carrier[B]$ by
        \obv{funs_is_function,funs}.
    $\identity{\carrier[B]}$ is injective by
        \obv{identity_injective}.
    $\identity{\carrier[B]}\in
        \inj{\carrier[B]}{\carrier[B]}$ by
        \auto{inj,injective_function}.
    Follows by \obv{boolean_embedding}.
\end{proof}

\begin{theorem}\label{boolean_embedding_composition}
    Suppose $\booleanEmbedding{f}{B}{C}$ and
        $\booleanEmbedding{g}{C}{D}$.
    Then $\booleanEmbedding{g\circ f}{B}{D}$.
\end{theorem}
\begin{proof}
    $\booleanHom{f}{B}{C}$ and $\booleanHom{g}{C}{D}$ by
        \obv{boolean_embedding_is_hom}.
    Hence $\booleanHom{g\circ f}{B}{D}$ by
        \obv{boolean_hom_composition}.
    $f\in\inj{\carrier[B]}{\carrier[C]}$ and
        $g\in\inj{\carrier[C]}{\carrier[D]}$ by
        \obv{boolean_embedding_injective}.
    Hence $g\circ f\in\inj{\carrier[B]}{\carrier[D]}$ by
        \obv{inj_circ}.
    Follows by \obv{boolean_embedding}.
\end{proof}

\begin{theorem}\label{boolean_embedding_order_reflecting}
    Suppose $\booleanEmbedding{f}{B}{C}$.
    Then
        $\orderReflecting
            {f}{\carrier[B]}{\leq[B]}
            {\carrier[C]}{\leq[C]}$.
\end{theorem}
\begin{proof}
    Follows by
        \obv{boolean_embedding_is_hom,boolean_hom_is_lattice_hom,
            boolean_embedding_injective,
            injective_lattice_hom_order_reflecting}.
\end{proof}

\begin{proposition}\label{boolean_embedding_order_iff}
    Suppose $\booleanEmbedding{f}{B}{C}$.
    Let $x,y\in\carrier[B]$.
    Then $x\mathrel{\leq[B]}y$ iff
        $f(x)\mathrel{\leq[C]}f(y)$.
\end{proposition}
\begin{proof}
    Follows by
        \obv{boolean_embedding_is_hom,boolean_hom_is_lattice_hom,
            boolean_embedding_injective,
