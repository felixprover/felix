\section{Boolean algebra isomorphisms}

\import{order/boolean-algebra/homomorphism.tex}
\import{function.tex}

\begin{definition}\label{boolean_isomorphism}
    $\booleanIsomorphism{f}{B}{C}$ iff
        $\booleanHom{f}{B}{C}$ and
        $f\in\bijections{\carrier[B]}{\carrier[C]}$.
\end{definition}

\begin{proposition}\label{boolean_isomorphism_is_hom}
    Suppose $\booleanIsomorphism{f}{B}{C}$.
    Then $\booleanHom{f}{B}{C}$.
\end{proposition}
\begin{proof}
    Follows by \obv{boolean_isomorphism}.
\end{proof}

\begin{proposition}\label{boolean_isomorphism_is_bijection}
    Suppose $\booleanIsomorphism{f}{B}{C}$.
    Then $f\in\bijections{\carrier[B]}{\carrier[C]}$.
\end{proposition}
\begin{proof}
    Follows by \obv{boolean_isomorphism}.
\end{proof}

\begin{corollary}\label{boolean_isomorphism_is_embedding}
    Suppose $\booleanIsomorphism{f}{B}{C}$.
    Then $\booleanEmbedding{f}{B}{C}$.
\end{corollary}
\begin{proof}
    $f$ is injective by
        \auto{boolean_isomorphism_is_bijection,bijections}.
    $f\in\funs{\carrier[B]}{\carrier[C]}$ by
        \obv{boolean_isomorphism_is_hom,boolean_hom_type}.
    Hence $f$ is a function and $\dom{f}=\carrier[B]$ by
        \obv{funs_is_function,funs}.
    Hence $f\in\inj{\carrier[B]}{\carrier[C]}$ by
        \auto{inj,injective_function}.
    Follows by
        \obv{boolean_embedding,boolean_isomorphism_is_hom}.
\end{proof}

\begin{proposition}\label{identity_boolean_isomorphism}
    Let $B$ be a Boolean algebra.
    Then
        $\booleanIsomorphism{\identity{\carrier[B]}}{B}{B}$.
\end{proposition}
\begin{proof}
    $\booleanHom{\identity{\carrier[B]}}{B}{B}$ by
        \obv{identity_boolean_hom}.
    $\identity{\carrier[B]}\in
        \bijections{\carrier[B]}{\carrier[B]}$ by
        \obv{id_elem_bijections}.
    Follows by \obv{boolean_isomorphism}.
\end{proof}

\begin{theorem}\label{boolean_isomorphism_composition}
    Suppose $\booleanIsomorphism{f}{B}{C}$ and
        $\booleanIsomorphism{g}{C}{D}$.
    Then $\booleanIsomorphism{g\circ f}{B}{D}$.
\end{theorem}
\begin{proof}
    $\booleanHom{f}{B}{C}$ and $\booleanHom{g}{C}{D}$ by
        \obv{boolean_isomorphism_is_hom}.
    Hence $\booleanHom{g\circ f}{B}{D}$ by
        \obv{boolean_hom_composition}.
    $f\in\bijections{\carrier[B]}{\carrier[C]}$ and
        $g\in\bijections{\carrier[C]}{\carrier[D]}$ by
        \obv{boolean_isomorphism_is_bijection}.
    Hence $g\circ f\in
        \bijections{\carrier[B]}{\carrier[D]}$ by
        \obv{bijection_circ}.
    Follows by \obv{boolean_isomorphism}.
\end{proof}

\begin{theorem}\label{boolean_isomorphism_converse_hom}
    Suppose $\booleanIsomorphism{f}{B}{C}$.
    Then $\booleanHom{\converse{f}}{C}{B}$.
\end{theorem}
\begin{proof}
    $\booleanHom{f}{B}{C}$ by
        \obv{boolean_isomorphism_is_hom}.
    $\boundedLatticeHom{f}{B}{C}$ by
        \obv{boolean_hom_is_bounded_lattice_hom}.
    $f\in\bijections{\carrier[B]}{\carrier[C]}$ by
        \obv{boolean_isomorphism_is_bijection}.
    Hence $\boundedLatticeIsomorphism{f}{B}{C}$ by
        \obv{bounded_lattice_isomorphism}.
    Hence $\boundedLatticeIsomorphism{\converse{f}}{C}{B}$ by
        \obv{bounded_lattice_isomorphism_converse}.
    Hence $\boundedLatticeHom{\converse{f}}{C}{B}$ by
        \obv{bounded_lattice_isomorphism_is_hom}.
    $\converse{f}\in\funs{\carrier[C]}{\carrier[B]}$ by
        \obv{bijective_converse_are_funs}.
    Hence $\converse{f}$ is a function and
        $\dom{\converse{f}}=\carrier[C]$ by
        \auto{funs_elim}.
    $f$ is a function and $\dom{f}=\carrier[B]$ by
        \auto{boolean_hom_type,funs_elim}.
    Show that for every $a\in\carrier[C]$ we have
        $(\converse{f})(\booleanComplement[C](a))
            =\booleanComplement[B]((\converse{f})(a))$.
    \begin{subproof}
        Fix $a\in\carrier[C]$.
        Let $x=(\converse{f})(a)$.
        $x\in\carrier[B]$ by \auto{funs_type_apply}.
        $(a,x)\in\converse{f}$ by
            \auto{function_apply_elim}.
        Hence $(x,a)\in f$ by \obv{converse_iff}.
        Hence $f(x)=a$ by \obv{function_apply_intro}.
        $\booleanComplement[B](x)\in\carrier[B]$ by
            \obv{boolean_complement_apply_type}.
        $f(\booleanComplement[B](x))
            =\booleanComplement[C](f(x))$ by
            \obv{boolean_hom_complement}.
        Hence $f(\booleanComplement[B](x))
            =\booleanComplement[C](a)$.
        Hence $(\booleanComplement[B](x),
            \booleanComplement[C](a))\in f$ by
            \auto{function_apply_elim}.
        Hence $(\booleanComplement[C](a),
            \booleanComplement[B](x))\in\converse{f}$ by
            \obv{converse_iff}.
        Hence $(\converse{f})(\booleanComplement[C](a))
            =\booleanComplement[B](x)$ by
            \obv{function_apply_intro}.
    \end{subproof}
    Follows by \obv{boolean_hom}.
\end{proof}

\begin{theorem}\label{boolean_isomorphism_converse}
    Suppose $\booleanIsomorphism{f}{B}{C}$.
    Then $\booleanIsomorphism{\converse{f}}{C}{B}$.
\end{theorem}
\begin{proof}
    $\booleanHom{\converse{f}}{C}{B}$ by
        \obv{boolean_isomorphism_converse_hom}.
    $f\in\bijections{\carrier[B]}{\carrier[C]}$ by
        \obv{boolean_isomorphism_is_bijection}.
    Hence $\converse{f}\in
        \bijections{\carrier[C]}{\carrier[B]}$ by
        \obv{bijection_converse_is_bijection}.
    Follows by \obv{boolean_isomorphism}.
\end{proof}

\begin{theorem}\label{boolean_embedding_image_isomorphism}
    Suppose $\booleanEmbedding{f}{B}{C}$.
    Then
        $\booleanIsomorphism
            {f}
            {B}
            {\booleanSubalgebraStructure
                {C}{\img{f}{\carrier[B]}}}$.
\end{theorem}
\begin{proof}
    Let $A=\img{f}{\carrier[B]}$.
    Let $T=\booleanSubalgebraStructure{C}{A}$.
    $\booleanHom{f}{B}{C}$ by \obv{boolean_embedding_is_hom}.
    Hence $\booleanSubalgebra{A}{C}$ by
        \auto{boolean_hom_image_boolean_subalgebra}.
    $A\subseteq A$ by \obv{subseteq_refl}.
    Hence $\booleanHom{f}{B}{T}$ by
        \obv{boolean_hom_codomain_corestriction}.
    Hence $f\in\funs{\carrier[B]}{\carrier[T]}$ by
        \obv{boolean_hom_type}.
    $\carrier[T]=A$ by
        \auto{boolean_subalgebra_structure_carrier}.
    $\dom{f}=\carrier[B]$ by \obv{boolean_hom_type,funs}.
    Hence $\ran{f}=A$ by \auto{img_dom}.
    Hence $f\in\surj{\carrier[B]}{\carrier[T]}$ by
        \auto{funs_surj_iff}.
    $f\in\inj{\carrier[B]}{\carrier[C]}$ by
        \obv{boolean_embedding_injective}.
    $f$ is a function and $\dom{f}=\carrier[B]$ by
        \obv{funs_is_function,funs}.
    For all $x,y\in\dom{f}$ such that $f(x)=f(y)$ we have
        $x=y$ by \obv{inj}.
    Hence $f$ is injective by \obv{injective_function}.
    Hence $f\in\bijections{\carrier[B]}{\carrier[T]}$ by
        \obv{bijections}.
    Follows by \obv{boolean_isomorphism}.
\end{proof}
