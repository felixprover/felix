\section{Order filters}

\import{order/directed.tex}
\import{order/morphism.tex}
\import{set/filter-tendsto.tex}

For a quasiorder, the filter at the top is generated by its final segments.
The construction is total: directedness is needed precisely for properness,
not for the existence of the filter.

\subsection{Final segments and the filter at the top}

\begin{definition}\label{order_final_segment}
    $\orderFinalSegment{I}{i}
        =\{j\in\carrier[I]\mid i\mathrel{\leq[I]}j\}$.
\end{definition}

\begin{proposition}\label{order_final_segment_iff}
    $j\in\orderFinalSegment{I}{i}$ iff
        $j\in\carrier[I]$ and $i\mathrel{\leq[I]}j$.
\end{proposition}
\begin{proof}
    Follows by \obv{order_final_segment}.
\end{proof}

\begin{proposition}\label{order_final_segment_type}
    $\orderFinalSegment{I}{i}\subseteq\carrier[I]$.
\end{proposition}
\begin{proof}
    Follows by \obv{order_final_segment_iff,subseteq}.
\end{proof}

\begin{definition}\label{order_final_segment_family}
    $\orderFinalSegmentFamily{I}
        =\{\orderFinalSegment{I}{i}\mid i\in\carrier[I]\}$.
\end{definition}

\begin{proposition}\label{order_final_segment_family_iff}
    $A\in\orderFinalSegmentFamily{I}$ iff there exists
        $i\in\carrier[I]$ such that
        $A=\orderFinalSegment{I}{i}$.
\end{proposition}
\begin{proof}
    Follows by \obv{order_final_segment_family}.
\end{proof}

\begin{proposition}\label{order_final_segment_family_type}
    $\orderFinalSegmentFamily{I}$ is a family of subsets of
        $\carrier[I]$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{order_final_segment_family_iff,
            order_final_segment_type,subseteq}.
\end{proof}

\begin{definition}\label{at_top_filter}
    $\atTop{I}
        =\generatedFilter{\carrier[I]}{\orderFinalSegmentFamily{I}}$.
\end{definition}

\begin{classification}\label{at_top_is_filter}
    Let $I$ be a quasiordered set.
    Then $\atTop{I}$ is a filter on $\carrier[I]$.
\end{classification}
\begin{proof}
    $\orderFinalSegmentFamily{I}$ is a family of subsets of
        $\carrier[I]$ by \obv{order_final_segment_family_type}.
    Follows by \auto{at_top_filter,generated_filter_is_filter}.
\end{proof}

\begin{proposition}\label{order_final_segment_in_at_top}
    Let $I$ be a quasiordered set and $i\in\carrier[I]$.
    Then $\orderFinalSegment{I}{i}\in\atTop{I}$.
\end{proposition}
\begin{proof}
    $\orderFinalSegment{I}{i}\in\orderFinalSegmentFamily{I}$ by
        \obv{order_final_segment_family_iff}.
    $\orderFinalSegmentFamily{I}\subseteq\atTop{I}$ by
        \auto{at_top_filter,order_final_segment_family_type,
            generated_filter_contains}.
    Follows by \obv{subseteq}.
\end{proof}

\begin{definition}\label{upward_directed_quasiorder}
    Let $I$ be a quasiordered set.
    $I$ is upward directed iff $\carrier[I]$ is inhabited and
        for all $i,j\in\carrier[I]$ there exists $k\in\carrier[I]$
        such that $i\mathrel{\leq[I]}k$ and
        $j\mathrel{\leq[I]}k$.
\end{definition}

\begin{proposition}\label{directed_quasiorder_is_upward_directed}
    Let $I$ be a directed quasiordered set.
    Then $I$ is upward directed.
\end{proposition}
\begin{proof}
    $I$ is a quasiordered set by
        \obv{directed_quasiordered_set_is_quasiordered}.
    Follows by
        \obv{upward_directed_quasiorder,
            directed_quasiordered_set_carrier_inhabited,
            directed_common_upper_bound}.
\end{proof}

\begin{theorem}\label{upward_directed_final_segments_are_proper_filter_basis}
    Let $I$ be a quasiordered set.
    Suppose $I$ is upward directed.
    Then $\orderFinalSegmentFamily{I}$ is a proper filter basis on
        $\carrier[I]$.
\end{theorem}
\begin{proof}
    $\carrier[I]$ is inhabited and for all
        $i,j\in\carrier[I]$ there exists $k\in\carrier[I]$ such that
        $i\mathrel{\leq[I]}k$ and $j\mathrel{\leq[I]}k$ by
        \obv{upward_directed_quasiorder}.
    $\orderFinalSegmentFamily{I}$ is a family of subsets of
        $\carrier[I]$ by \obv{order_final_segment_family_type}.
    Take $i$ such that $i\in\carrier[I]$.
    $\orderFinalSegment{I}{i}\in\orderFinalSegmentFamily{I}$ by
        \obv{order_final_segment_family_iff}.
    Hence $\orderFinalSegmentFamily{I}$ is inhabited.

    Show that for all $A,C$ such that
        $A,C\in\orderFinalSegmentFamily{I}$ there exists
        $D\in\orderFinalSegmentFamily{I}$ such that
        $D\subseteq A\inter C$.
    \begin{subproof}
        Fix $A,C$.
        Assume $A,C\in\orderFinalSegmentFamily{I}$.
        Take $a\in\carrier[I]$ such that
            $A=\orderFinalSegment{I}{a}$ by
            \obv{order_final_segment_family_iff}.
        Take $c\in\carrier[I]$ such that
            $C=\orderFinalSegment{I}{c}$ by
            \obv{order_final_segment_family_iff}.
        Take $d\in\carrier[I]$ such that
            $a\mathrel{\leq[I]}d$ and $c\mathrel{\leq[I]}d$.
        $\orderFinalSegment{I}{d}\in\orderFinalSegmentFamily{I}$ by
            \obv{order_final_segment_family_iff}.
        Show that $\orderFinalSegment{I}{d}\subseteq A\inter C$.
        \begin{subproof}
            It suffices to show that every element of
                $\orderFinalSegment{I}{d}$ is an element of $A\inter C$
                by \obv{subseteq}.
            Fix $x\in\orderFinalSegment{I}{d}$.
            Hence $x\in\carrier[I]$ and $d\mathrel{\leq[I]}x$ by
                \obv{order_final_segment}.
            $a\mathrel{\leq[I]}x$ and $c\mathrel{\leq[I]}x$ by
                \auto{quasiorder_tran,transitive}.
            Hence $x\in\orderFinalSegment{I}{a}$ and
                $x\in\orderFinalSegment{I}{c}$ by
                \obv{order_final_segment_iff}.
            Follows by \obv{inter_intro}.
        \end{subproof}
    \end{subproof}
    Hence $\orderFinalSegmentFamily{I}$ is a filter basis on
        $\carrier[I]$ by \obv{filter_basis}.

    Show that for every $A\in\orderFinalSegmentFamily{I}$ we have
        $A$ is inhabited.
    \begin{subproof}
        Fix $A\in\orderFinalSegmentFamily{I}$.
        Take $a\in\carrier[I]$ such that
            $A=\orderFinalSegment{I}{a}$ by
            \obv{order_final_segment_family_iff}.
        $a\mathrel{\leq[I]}a$ by
            \auto{quasiorder_refl,reflexive_on}.
        Hence $a\in\orderFinalSegment{I}{a}$ by
            \obv{order_final_segment}.
        Hence $a\in A$.
    \end{subproof}
    Follows by \obv{proper_filter_basis}.
\end{proof}

\begin{corollary}\label{directed_final_segments_are_proper_filter_basis}
    Let $I$ be a directed quasiordered set.
    Then $\orderFinalSegmentFamily{I}$ is a proper filter basis on
        $\carrier[I]$.
\end{corollary}
\begin{proof}
    $I$ is upward directed by
        \obv{directed_quasiorder_is_upward_directed}.
    Follows by
        \obv{upward_directed_final_segments_are_proper_filter_basis}.
\end{proof}

\begin{theorem}\label{at_top_iff_contains_final_segment}
    Let $I$ be a directed quasiordered set.
    Then $A\in\atTop{I}$ iff $A\subseteq\carrier[I]$ and there exists
        $i\in\carrier[I]$ such that
        $\orderFinalSegment{I}{i}\subseteq A$.
\end{theorem}
\begin{proof}
    $\orderFinalSegmentFamily{I}$ is a proper filter basis on
        $\carrier[I]$ by
        \obv{directed_final_segments_are_proper_filter_basis}.
    Hence $\orderFinalSegmentFamily{I}$ is a filter basis on
        $\carrier[I]$ by
        \obv{proper_filter_basis_is_filter_basis}.
    Follows by
        \auto{at_top_filter,generated_filter_iff,
            order_final_segment_family_iff}.
\end{proof}

\begin{theorem}\label{at_top_is_proper_iff_directed}
    Let $I$ be a quasiordered set.
    Then $\atTop{I}$ is a proper filter on $\carrier[I]$ iff
        $I$ is upward directed.
\end{theorem}
\begin{proof}
    Show that if $\atTop{I}$ is a proper filter on $\carrier[I]$, then
        $I$ is upward directed.
    \begin{subproof}
        Assume $\atTop{I}$ is a proper filter on $\carrier[I]$.
        $\atTop{I}$ is a filter on $\carrier[I]$ by
            \obv{proper_filter}.
        $\carrier[I]$ is inhabited by
            \obv{proper_filter_carrier_inhabited}.
        Show that for all $i,j\in\carrier[I]$ there exists
            $k\in\carrier[I]$ such that
            $i\mathrel{\leq[I]}k$ and $j\mathrel{\leq[I]}k$.
        \begin{subproof}
            Fix $i,j\in\carrier[I]$.
            $\orderFinalSegment{I}{i}\in\atTop{I}$ by
                \obv{order_final_segment_in_at_top}.
            $\orderFinalSegment{I}{j}\in\atTop{I}$ by
                \obv{order_final_segment_in_at_top}.
            Hence $\orderFinalSegment{I}{i}\inter
                \orderFinalSegment{I}{j}\in\atTop{I}$ by
                \obv{filter}.
            $\emptyset\notin\atTop{I}$ by \obv{proper_filter}.
            Hence $\orderFinalSegment{I}{i}\inter
                \orderFinalSegment{I}{j}\neq\emptyset$.
            Thus $\orderFinalSegment{I}{i}\inter
                \orderFinalSegment{I}{j}$ is inhabited by
                \obv{nonempty_inhabited}.
            Take $k$ such that $k\in\orderFinalSegment{I}{i}\inter
                \orderFinalSegment{I}{j}$.
            $k\in\orderFinalSegment{I}{i}$ and
                $k\in\orderFinalSegment{I}{j}$ by \obv{inter}.
            Hence $k\in\carrier[I]$ and
                $i\mathrel{\leq[I]}k$ and
                $j\mathrel{\leq[I]}k$ by
                \obv{order_final_segment_iff}.
            Follows by \auto.
        \end{subproof}
        Follows by \obv{upward_directed_quasiorder}.
    \end{subproof}

    Show that if $I$ is upward directed, then
        $\atTop{I}$ is a proper filter on $\carrier[I]$.
    \begin{subproof}
        Assume $I$ is upward directed.
        $\orderFinalSegmentFamily{I}$ is a proper filter basis on
            $\carrier[I]$ by
            \obv{upward_directed_final_segments_are_proper_filter_basis}.
        Follows by
            \auto{at_top_filter,generated_filter_is_proper_filter}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{corollary}\label{final_segments_have_fip_iff_directed}
    Let $I$ be a quasiordered set.
    Then $\orderFinalSegmentFamily{I}$ has the finite intersection
        property on $\carrier[I]$ iff
        $I$ is upward directed.
\end{corollary}
\begin{proof}
    $\orderFinalSegmentFamily{I}$ is a family of subsets of
        $\carrier[I]$ by \obv{order_final_segment_family_type}.
    $\atTop{I}$ is a proper filter on $\carrier[I]$ iff
        $\orderFinalSegmentFamily{I}$ has the finite intersection
            property on $\carrier[I]$ by
        \auto{at_top_filter,generated_filter_is_proper_iff_hosted_fip}.
    $\atTop{I}$ is a proper filter on $\carrier[I]$ iff
        $I$ is upward directed by
        \obv{at_top_is_proper_iff_directed}.
\end{proof}

\subsection{The opposite quasiorder and the filter at the bottom}

\begin{definition}\label{opposite_quasiorder}
    $\oppositeQuasiorder{I}
        =\structure[quasiordered set]{
            \carrier=\carrier[I],
            \leq=\converse{\leq[I]}
        }$.
\end{definition}

\begin{proposition}\label{opposite_quasiorder_carrier}
    $\carrier[\oppositeQuasiorder{I}]=\carrier[I]$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{opposite_quasiorder,quasiordered_setaggregateprojection_carrier}.
\end{proof}

\begin{proposition}\label{opposite_quasiorder_relation}
    $\leq[\oppositeQuasiorder{I}]=\converse{\leq[I]}$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{opposite_quasiorder,quasiordered_setaggregateprojection_leq}.
\end{proof}

\begin{theorem}\label{opposite_quasiorder_is_exact}
    Let $I$ be a quasiordered set.
    Then $\oppositeQuasiorder{I}$ is an exact quasiordered set.
\end{theorem}
\begin{proof}
    Let $J=\oppositeQuasiorder{I}$.
    $\leq[I]$ is a binary relation on $\carrier[I]$ by
        \obv{quasiorder_type}.
    $\leq[I]$ is reflexive on $\carrier[I]$ and
        $\leq[I]$ is transitive by
        \obv{quasiorder_refl,quasiorder_tran}.
    Hence $\leq[I]$ is a quasiorder on $\carrier[I]$ by definition.
    $\converse{\leq[I]}$ is a quasiorder on $\carrier[I]$ by
        \obv{converse_quasiorder_on}.
    $\carrier[J]=\carrier[I]$ by \auto{opposite_quasiorder_carrier}.
    $\leq[J]=\converse{\leq[I]}$ by
        \auto{opposite_quasiorder_relation}.
    Hence $\leq[J]$ is a binary relation on $\carrier[J]$ and
        $\leq[J]$ is reflexive on $\carrier[J]$ and
        $\leq[J]$ is transitive.
    $J$ is an exact quasiordered set by
        \auto{opposite_quasiorder,quasiordered_setaggregateexact}.
    Follows by \auto.
\end{proof}

\begin{classification}\label{opposite_quasiorder_is_quasiordered}
    Let $I$ be a quasiordered set.
    Then $\oppositeQuasiorder{I}$ is a quasiordered set.
\end{classification}
\begin{proof}
    Follows by \obv{opposite_quasiorder_is_exact}.
\end{proof}

\begin{proposition}\label{opposite_quasiorder_order_iff}
    $i\mathrel{\leq[\oppositeQuasiorder{I}]}j$ iff
        $j\mathrel{\leq[I]}i$.
\end{proposition}
\begin{proof}
    Follows by \auto{opposite_quasiorder_relation,converse_iff}.
\end{proof}

\begin{definition}\label{downward_directed_quasiorder}
    Let $I$ be a quasiordered set.
    $I$ is downward directed iff $\carrier[I]$ is inhabited and
        for all $i,j\in\carrier[I]$ there exists $k\in\carrier[I]$
        such that $k\mathrel{\leq[I]}i$ and
        $k\mathrel{\leq[I]}j$.
\end{definition}

\begin{proposition}\label{opposite_upward_directed_iff_downward_directed}
    Let $I$ be a quasiordered set.
    Then $\oppositeQuasiorder{I}$ is upward directed iff
        $I$ is downward directed.
\end{proposition}
\begin{proof}
    $\oppositeQuasiorder{I}$ is a quasiordered set by
        \obv{opposite_quasiorder_is_quasiordered}.
    Follows by
        \auto{upward_directed_quasiorder,
            downward_directed_quasiorder,
            opposite_quasiorder_carrier,
            opposite_quasiorder_order_iff}.
\end{proof}

\begin{definition}\label{order_initial_segment}
    $\orderInitialSegment{I}{i}
        =\{j\in\carrier[I]\mid j\mathrel{\leq[I]}i\}$.
\end{definition}

\begin{proposition}\label{order_initial_segment_iff}
    $j\in\orderInitialSegment{I}{i}$ iff
        $j\in\carrier[I]$ and $j\mathrel{\leq[I]}i$.
\end{proposition}
\begin{proof}
    Follows by \obv{order_initial_segment}.
\end{proof}

\begin{proposition}\label{opposite_final_segment_is_initial_segment}
    $\orderFinalSegment{\oppositeQuasiorder{I}}{i}
        =\orderInitialSegment{I}{i}$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{setext,order_final_segment_iff,order_initial_segment_iff,
            opposite_quasiorder_carrier,
            opposite_quasiorder_order_iff}.
\end{proof}

\begin{definition}\label{at_bottom_filter}
    $\atBottom{I}=\atTop{\oppositeQuasiorder{I}}$.
\end{definition}

\begin{classification}\label{at_bottom_is_filter}
    Let $I$ be a quasiordered set.
    Then $\atBottom{I}$ is a filter on $\carrier[I]$.
\end{classification}
\begin{proof}
    $\oppositeQuasiorder{I}$ is a quasiordered set by
        \obv{opposite_quasiorder_is_quasiordered}.
    $\atTop{\oppositeQuasiorder{I}}$ is a filter on
        $\carrier[\oppositeQuasiorder{I}]$ by
        \obv{at_top_is_filter}.
    Follows by
        \obv{at_bottom_filter,opposite_quasiorder_carrier}.
\end{proof}

\begin{theorem}\label{at_bottom_is_proper_iff_downward_directed}
    Let $I$ be a quasiordered set.
    Then $\atBottom{I}$ is a proper filter on $\carrier[I]$ iff
        $I$ is downward directed.
\end{theorem}
\begin{proof}
    $\oppositeQuasiorder{I}$ is a quasiordered set by
        \obv{opposite_quasiorder_is_quasiordered}.
    $\atTop{\oppositeQuasiorder{I}}$ is a proper filter on
        $\carrier[\oppositeQuasiorder{I}]$ iff
        $\oppositeQuasiorder{I}$ is upward directed by
        \obv{at_top_is_proper_iff_directed}.
    Follows by
        \auto{at_bottom_filter,opposite_quasiorder_carrier,
            opposite_upward_directed_iff_downward_directed}.
\end{proof}

\subsection{Nets and cofinal reindexing}

\begin{definition}\label{net}
    Let $I$ be a directed quasiordered set.
    $s$ is a net in $S$ indexed by $I$ iff
        $s\in\funs{\carrier[I]}{S}$.
\end{definition}

\begin{definition}\label{net_filter}
    $\netFilter{I}{S}{s}
        =\filterPushforward{\carrier[I]}{S}{s}{\atTop{I}}$.
\end{definition}

\begin{proposition}\label{net_filter_is_filter}
    Let $I$ be a directed quasiordered set and
        $s\in\funs{\carrier[I]}{S}$.
    Then $\netFilter{I}{S}{s}$ is a filter on $S$.
\end{proposition}
\begin{proof}
    $\atTop{I}$ is a filter on $\carrier[I]$ by
        \obv{at_top_is_filter}.
    Follows by
        \auto{net_filter,filter_pushforward_is_filter}.
\end{proof}

\begin{theorem}\label{net_filter_is_proper}
    Let $I$ be a directed quasiordered set and
        $s\in\funs{\carrier[I]}{S}$.
    Then $\netFilter{I}{S}{s}$ is a proper filter on $S$.
\end{theorem}
\begin{proof}
    $I$ is upward directed by
        \obv{directed_quasiorder_is_upward_directed}.
    $\atTop{I}$ is a proper filter on $\carrier[I]$ by
        \obv{at_top_is_proper_iff_directed}.
    Follows by
        \auto{net_filter,filter_pushforward_preserves_properness}.
\end{proof}

\begin{definition}\label{at_top_cofinal}
    Let $J,I$ be directed quasiordered sets.
    $e$ is cofinal at the top from $J$ to $I$ iff
        $e$ tends from $\atTop{J}$ on $\carrier[J]$ to
            $\atTop{I}$ on $\carrier[I]$.
\end{definition}

\begin{proposition}\label{at_top_cofinal_type}
    Let $J,I$ be directed quasiordered sets.
    Suppose $e$ is cofinal at the top from $J$ to $I$.
    Then $e\in\funs{\carrier[J]}{\carrier[I]}$.
\end{proposition}
\begin{proof}
    Follows by \auto{at_top_cofinal,filter_tendsto}.
\end{proof}

\begin{theorem}\label{at_top_cofinal_iff_eventually_above}
    Let $J,I$ be directed quasiordered sets.
    Let $e\in\funs{\carrier[J]}{\carrier[I]}$.
    Then $e$ is cofinal at the top from $J$ to $I$ iff
        for every $i\in\carrier[I]$ there exists $j\in\carrier[J]$
        such that for every $k\in\carrier[J]$ such that
        $j\mathrel{\leq[J]}k$ we have
        $i\mathrel{\leq[I]}e(k)$.
\end{theorem}
\begin{proof}
    $\atTop{J}$ is a filter on $\carrier[J]$ by
        \obv{at_top_is_filter}.
    $\atTop{I}$ is a filter on $\carrier[I]$ by
        \obv{at_top_is_filter}.

    Show that if $e$ is cofinal at the top from $J$ to $I$, then
        for every $i\in\carrier[I]$ there exists $j\in\carrier[J]$
        such that for every $k\in\carrier[J]$ such that
        $j\mathrel{\leq[J]}k$ we have
        $i\mathrel{\leq[I]}e(k)$.
    \begin{subproof}
        Assume $e$ is cofinal at the top from $J$ to $I$.
        $e$ tends from $\atTop{J}$ on $\carrier[J]$ to
            $\atTop{I}$ on $\carrier[I]$ by \obv{at_top_cofinal}.
        Hence for every $B\in\atTop{I}$ we have
            $\preimg{e}{B}\in\atTop{J}$ by
            \obv{filter_tendsto_iff_preimage}.
        Fix $i\in\carrier[I]$.
        $\orderFinalSegment{I}{i}\in\atTop{I}$ by
            \obv{order_final_segment_in_at_top}.
        Hence $\preimg{e}{\orderFinalSegment{I}{i}}\in\atTop{J}$.
        Take $j\in\carrier[J]$ such that
            $\orderFinalSegment{J}{j}\subseteq
                \preimg{e}{\orderFinalSegment{I}{i}}$ by
            \auto{at_top_iff_contains_final_segment}.
        Show that for every $k\in\carrier[J]$ such that
            $j\mathrel{\leq[J]}k$ we have
            $i\mathrel{\leq[I]}e(k)$.
        \begin{subproof}
            Fix $k\in\carrier[J]$.
            Assume $j\mathrel{\leq[J]}k$.
            Hence $k\in\orderFinalSegment{J}{j}$ by
                \obv{order_final_segment}.
            Hence $k\in\preimg{e}{\orderFinalSegment{I}{i}}$ by
                \obv{subseteq}.
            Take $y\in\orderFinalSegment{I}{i}$ such that
                $(k,y)\in e$ by \obv{preimg_iff}.
            $e(k)=y$ by \obv{funs_is_function,function_apply_intro}.
            $i\mathrel{\leq[I]}y$ by
                \obv{order_final_segment}.
        \end{subproof}
        Follows by \auto.
    \end{subproof}

    Show that if for every $i\in\carrier[I]$ there exists
        $j\in\carrier[J]$ such that for every $k\in\carrier[J]$
        such that $j\mathrel{\leq[J]}k$ we have
        $i\mathrel{\leq[I]}e(k)$, then
        $e$ is cofinal at the top from $J$ to $I$.
    \begin{subproof}
        Assume for every $i\in\carrier[I]$ there exists
            $j\in\carrier[J]$ such that for every $k\in\carrier[J]$
            such that $j\mathrel{\leq[J]}k$ we have
            $i\mathrel{\leq[I]}e(k)$.
        Show that for every $B\in\atTop{I}$ we have
            $\preimg{e}{B}\in\atTop{J}$.
        \begin{subproof}
            Fix $B\in\atTop{I}$.
            Take $i\in\carrier[I]$ such that
                $\orderFinalSegment{I}{i}\subseteq B$ by
                \auto{at_top_iff_contains_final_segment}.
            Take $j\in\carrier[J]$ such that for every
                $k\in\carrier[J]$ such that
                $j\mathrel{\leq[J]}k$ we have
                $i\mathrel{\leq[I]}e(k)$.
            Show that $\orderFinalSegment{J}{j}\subseteq
                \preimg{e}{B}$.
            \begin{subproof}
                It suffices to show that every element of
                    $\orderFinalSegment{J}{j}$ is an element of
                    $\preimg{e}{B}$ by \obv{subseteq}.
                Fix $k\in\orderFinalSegment{J}{j}$.
                Hence $k\in\carrier[J]$ and
                    $j\mathrel{\leq[J]}k$ by
                    \obv{order_final_segment}.
                Hence $i\mathrel{\leq[I]}e(k)$.
                $e(k)\in\carrier[I]$ by \obv{funs_type_apply}.
                Hence $e(k)\in\orderFinalSegment{I}{i}$ by
                    \obv{order_final_segment}.
                Hence $e(k)\in B$ by \obv{subseteq}.
                $(k,e(k))\in e$ by
                    \auto{funs_elim,function_apply_iff}.
                Follows by \obv{preimg_iff}.
            \end{subproof}
            $\preimg{e}{B}\subseteq\carrier[J]$ by
                \auto{preimg_subseteq_dom,funs}.
            Follows by \auto{at_top_iff_contains_final_segment}.
        \end{subproof}
        $e$ tends from $\atTop{J}$ on $\carrier[J]$ to
            $\atTop{I}$ on $\carrier[I]$ by
            \obv{filter_tendsto_iff_preimage}.
        Follows by \obv{at_top_cofinal}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{definition}\label{cofinal_range_between_quasiorders}
    Let $J,I$ be quasiordered sets.
    $e$ has cofinal range from $J$ to $I$ iff
        $e\in\funs{\carrier[J]}{\carrier[I]}$ and
        for every $i\in\carrier[I]$ there exists $j\in\carrier[J]$
        such that $i\mathrel{\leq[I]}e(j)$.
\end{definition}

\begin{theorem}\label{at_top_cofinal_implies_cofinal_range}
    Let $J,I$ be directed quasiordered sets.
    Suppose $e$ is cofinal at the top from $J$ to $I$.
    Then $e$ has cofinal range from $J$ to $I$.
\end{theorem}
\begin{proof}
    $e\in\funs{\carrier[J]}{\carrier[I]}$ by
        \obv{at_top_cofinal_type}.
    For every $i\in\carrier[I]$ there exists $j\in\carrier[J]$
        such that for every $k\in\carrier[J]$ such that
        $j\mathrel{\leq[J]}k$ we have
        $i\mathrel{\leq[I]}e(k)$ by
        \obv{at_top_cofinal_iff_eventually_above}.
    Show that for every $i\in\carrier[I]$ there exists
        $j\in\carrier[J]$ such that $i\mathrel{\leq[I]}e(j)$.
    \begin{subproof}
        Fix $i\in\carrier[I]$.
        Take $j\in\carrier[J]$ such that for every
            $k\in\carrier[J]$ such that $j\mathrel{\leq[J]}k$
            we have $i\mathrel{\leq[I]}e(k)$.
        $j\mathrel{\leq[J]}j$ by \obv{directed_order_reflexive}.
        Hence $i\mathrel{\leq[I]}e(j)$.
    \end{subproof}
    Follows by \obv{cofinal_range_between_quasiorders}.
\end{proof}

\begin{theorem}\label{monotone_cofinal_range_implies_at_top_cofinal}
    Let $J,I$ be directed quasiordered sets.
    Suppose $\orderPreserving{e}{\carrier[J]}{\leq[J]}
        {\carrier[I]}{\leq[I]}$.
    Suppose $e$ has cofinal range from $J$ to $I$.
    Then $e$ is cofinal at the top from $J$ to $I$.
\end{theorem}
\begin{proof}
    $e\in\funs{\carrier[J]}{\carrier[I]}$ by
        \obv{order_preserving_type}.
    Show that for every $i\in\carrier[I]$ there exists
        $j\in\carrier[J]$ such that for every $k\in\carrier[J]$
        such that $j\mathrel{\leq[J]}k$ we have
        $i\mathrel{\leq[I]}e(k)$.
    \begin{subproof}
        Fix $i\in\carrier[I]$.
        Take $j\in\carrier[J]$ such that
            $i\mathrel{\leq[I]}e(j)$ by
            \obv{cofinal_range_between_quasiorders}.
        Show that for every $k\in\carrier[J]$ such that
            $j\mathrel{\leq[J]}k$ we have
            $i\mathrel{\leq[I]}e(k)$.
        \begin{subproof}
            Fix $k\in\carrier[J]$.
            Assume $j\mathrel{\leq[J]}k$.
            \begin{align*}
                i
                    &\mathrel{\leq[I]} e(j)
                        \explanation{by assumption}
                        \\
                    &\mathrel{\leq[I]} e(k)
                        \explanation{by \obv{order_preserving}}
            \end{align*}
        \end{subproof}
        Follows by \auto.
    \end{subproof}
    Follows by \obv{at_top_cofinal_iff_eventually_above}.
\end{proof}

\begin{theorem}\label{at_top_cofinal_reindex}
    Let $J,I$ be directed quasiordered sets.
    Suppose $e$ is cofinal at the top from $J$ to $I$.
    Let $s\in\funs{\carrier[I]}{S}$ and $F$ be a filter on $S$.
    Suppose $s$ tends from $\atTop{I}$ on $\carrier[I]$ to $F$ on $S$.
    Then $s\circ e$ tends from $\atTop{J}$ on $\carrier[J]$ to
        $F$ on $S$.
\end{theorem}
\begin{proof}
    $e\in\funs{\carrier[J]}{\carrier[I]}$ by
        \obv{at_top_cofinal_type}.
    $\atTop{J}$ is a filter on $\carrier[J]$ by
        \obv{at_top_is_filter}.
    $\atTop{I}$ is a filter on $\carrier[I]$ by
        \obv{at_top_is_filter}.
    $e$ tends from $\atTop{J}$ on $\carrier[J]$ to
        $\atTop{I}$ on $\carrier[I]$ by \obv{at_top_cofinal}.
    Follows by \obv{composition_filter_tendsto}.
\end{proof}
