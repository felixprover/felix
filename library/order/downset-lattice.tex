\section{The downset lattice}

\import{order/downset.tex}
\import{order/set-family-lattice.tex}

\begin{theorem}\label{downsets_are_bounded_set_lattice_family}
    Let $P$ be a quasiordered set.
    Then $\downsets{P}$ is a bounded set-lattice family on $\carrier[P]$.
\end{theorem}
\begin{proof}
    $\downsets{P}\subseteq\pow{\carrier[P]}$ by
        \obv{downsets_subseteq_powerset}.
    $\emptyset$ is a downset in $P$ by \obv{emptyset_is_downset}.
    Hence $\emptyset\in\downsets{P}$ by \auto{downsets_iff}.
    $\carrier[P]$ is a downset in $P$ by \obv{carrier_is_downset}.
    Hence $\carrier[P]\in\downsets{P}$ by \auto{downsets_iff}.
    Show for all $A,B\in\downsets{P}$ we have
        $A\inter B\in\downsets{P}$.
    \begin{subproof}
        Fix $A,B\in\downsets{P}$.
        $A$ and $B$ are downsets in $P$ by \obv{downsets_iff}.
        Hence $A\inter B$ is a downset in $P$ by \obv{downset_inter}.
        Follows by \auto{downsets_iff}.
    \end{subproof}
    Show for all $A,B\in\downsets{P}$ we have
        $A\union B\in\downsets{P}$.
    \begin{subproof}
        Fix $A,B\in\downsets{P}$.
        $A$ and $B$ are downsets in $P$ by \obv{downsets_iff}.
        Hence $A\union B$ is a downset in $P$ by \obv{downset_union}.
        Follows by \auto{downsets_iff}.
    \end{subproof}
    Follows by \obv{bounded_set_lattice_family}.
\end{proof}

\begin{definition}\label{downset_lattice}
    Let $P$ be a quasiordered set.
    $\downsetLattice{P}=
        \setFamilyLattice{\carrier[P]}{\downsets{P}}$.
\end{definition}

\begin{theorem}\label{downset_lattice_is_exact_bounded_distributive_lattice}
    Let $P$ be a quasiordered set.
    Then $\downsetLattice{P}$ is an exact bounded distributive lattice.
\end{theorem}
\begin{proof}
    $\downsets{P}$ is a bounded set-lattice family on $\carrier[P]$ by
        \obv{downsets_are_bounded_set_lattice_family}.
    Follows by
        \auto{downset_lattice,
            set_family_lattice_is_exact_bounded_distributive_lattice}.
\end{proof}

\begin{classification}\label{downset_lattice_is_bounded_distributive_lattice}
    Let $P$ be a quasiordered set.
    Then $\downsetLattice{P}$ is a bounded distributive lattice.
\end{classification}
\begin{proof}
    Follows by
        \obv{downset_lattice_is_exact_bounded_distributive_lattice}.
\end{proof}

\begin{proposition}\label{downset_lattice_carrier}

\begin{proposition}\label{downset_lattice_meet}
    Let $P$ be a quasiordered set.
    Let $A,B\in\downsets{P}$.
    Then $\meet[\downsetLattice{P}](A,B)=A\inter B$.
\end{proposition}
\begin{proof}
    $\meet[\downsetLattice{P}]
        =\setFamilyMeetGraph{\downsets{P}}$ by
        \auto{downset_lattice,set_family_lattice_meet}.
    Follows by \auto{set_family_meet_graph_apply}.
\end{proof}
