\section{Induced ordered subsets}

\import{relation/trace.tex}
\import{order/order.tex}

For an arbitrary set $A$, the induced carrier is the part of $A$ that lies in
the ambient ordered carrier.  This keeps the construction total without
silently coercing a set to a substructure.

\begin{definition}\label{ordered_subset}
    $\orderedSubset{P}{A}=
        \structure[ordered set]{
            \carrier=A\inter\carrier[P],
            \leq=\relationTrace{\leq[P]}
                {A\inter\carrier[P]}
        }$.
\end{definition}

\begin{proposition}\label{ordered_subset_carrier}
    $\carrier[\orderedSubset{P}{A}]=A\inter\carrier[P]$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{ordered_subset,quasiordered_setaggregateprojection_carrier}.
\end{proof}

\begin{proposition}\label{ordered_subset_order}
    $\leq[\orderedSubset{P}{A}]=
        \relationTrace{\leq[P]}{A\inter\carrier[P]}$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{ordered_subset,quasiordered_setaggregateprojection_leq}.
\end{proof}

\begin{proposition}\label{relation_trace_quasiorder_on}
    Suppose $R$ is a quasiorder on $X$ and $A\subseteq X$.
    Then $\relationTrace{R}{A}$ is a quasiorder on $A$.
\end{proposition}
\begin{proof}
    $R$ is reflexive on $X$ and $R$ is transitive by
        definition.
    $\relationTrace{R}{A}$ is a binary relation on $A$ by
        \obv{relation_trace_binary_on}.
    $\relationTrace{R}{A}$ is reflexive on $A$ by
        \obv{relation_trace_reflexive_on}.
    $\relationTrace{R}{A}$ is transitive by
        \obv{relation_trace_transitive}.
\end{proof}

\begin{proposition}\label{relation_trace_order_on}
    Suppose $R$ is an order on $X$ and $A\subseteq X$.
    Then $\relationTrace{R}{A}$ is an order on $A$.
\end{proposition}
\begin{proof}
    $R$ is a quasiorder on $X$ and $R$ is antisymmetric by
        definition.
    $\relationTrace{R}{A}$ is a quasiorder on $A$ by
        \obv{relation_trace_quasiorder_on}.
    $\relationTrace{R}{A}$ is antisymmetric by
        \obv{relation_trace_antisymmetric}.
    Follows by definition.
\end{proof}

\begin{theorem}\label{ordered_subset_is_exact_ordered_set}
    Let $P$ be an ordered set.
    Then $\orderedSubset{P}{A}$ is an exact ordered set.
\end{theorem}
\begin{proof}
    Let $Q=\orderedSubset{P}{A}$.
    Let $C=A\inter\carrier[P]$.
    $C\subseteq\carrier[P]$ by \auto{inter_lower_right}.
    $\leq[P]$ is a binary relation on $\carrier[P]$ by
        \obv{quasiorder_type}.
    $\leq[P]$ is reflexive on $\carrier[P]$ and
        $\leq[P]$ is transitive by
        \obv{quasiorder_refl,quasiorder_tran}.
    Hence $\leq[P]$ is a quasiorder on $\carrier[P]$ by
        definition.
    $\leq[P]$ is antisymmetric by \obv{orderedset_antisym}.
    Hence $\leq[P]$ is an order on $\carrier[P]$ by definition.
    $\relationTrace{\leq[P]}{C}$ is an order on $C$ by
        \obv{relation_trace_order_on}.
    $\carrier[Q]=C$ by \auto{ordered_subset_carrier}.
    $\leq[Q]=\relationTrace{\leq[P]}{C}$ by
        \auto{ordered_subset_order}.
    Hence $\leq[Q]$ is a binary relation on $\carrier[Q]$ and
        $\leq[Q]$ is reflexive on $\carrier[Q]$ and
        $\leq[Q]$ is transitive.
    $\leq[Q]$ is antisymmetric.
    Hence $Q$ is a quasiordered set by
        \auto{ordered_subset,quasiordered_set,
            quasiordered_setaggregateexact,
            quasiordered_setaggregateprojection_carrier,
            quasiordered_setaggregateprojection_leq}.
    $Q$ is an exact quasiordered set by
        \auto{ordered_subset,quasiordered_setaggregateexact}.
    Hence $Q$ is an ordered set by \obv{orderedset}.
    Follows by
        \auto{ordered_subset,quasiordered_setaggregateexact}.
\end{proof}

\begin{classification}\label{ordered_subset_is_ordered_set}
    Let $P$ be an ordered set.
    Then $\orderedSubset{P}{A}$ is an ordered set.
\end{classification}
\begin{proof}
    Follows by \auto{ordered_subset_is_exact_ordered_set}.
\end{proof}

\begin{corollary}\label{ordered_subset_carrier_of_subset}
    Suppose $A\subseteq\carrier[P]$.
    Then $\carrier[\orderedSubset{P}{A}]=A$.
\end{corollary}
\begin{proof}
    $A\inter\carrier[P]=A$ by
        \obv{inter_absorb_supseteq_right}.
    Follows by \obv{ordered_subset_carrier}.
\end{proof}

\begin{proposition}\label{ordered_subset_order_iff}
    $x\mathrel{\leq[\orderedSubset{P}{A}]}y$ iff
        $x\in A\inter\carrier[P]$ and
        $y\in A\inter\carrier[P]$ and
        $x\mathrel{\leq[P]}y$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{ordered_subset_order,relation_trace_iff}.
\end{proof}

\begin{proposition}\label{ordered_subset_order_iff_of_membership}
    Let $P$ be an ordered set.
    Let $x,y\in\carrier[\orderedSubset{P}{A}]$.
    Then $x\mathrel{\leq[\orderedSubset{P}{A}]}y$ iff
        $x\mathrel{\leq[P]}y$.
\end{proposition}
\begin{proof}
    $x,y\in A\inter\carrier[P]$ by
        \obv{ordered_subset_carrier}.
    Follows by \auto{ordered_subset_order_iff}.
\end{proof}
