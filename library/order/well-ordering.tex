\section{The well-ordering theorem}

\import{order/generated-well-order.tex}
\import{set/choice.tex}

The construction uses the same generated-order machinery as Hausdorff's
principle, but is independent of Zorn's lemma.

\begin{theorem}\label{well_ordering_theorem}
    There exists $R$ such that $R$ is a strict well order on $A$.
\end{theorem}
\begin{proof}
    Let $H=\{B\in\pow{A}\mid B\neq A\}$.
    Show that for all $B\in H$ there exists $a\in A\setminus B$.
    \begin{subproof}
        Fix $B\in H$.
        $B\in\pow{A}$ and $B\neq A$.
        Hence $B\subseteq A$ by \obv{powerset_elim}.
        Thus $B\subset A$ by \obv{subset}.
        Take $a\in A$ such that $a\notin B$ by \obv{subset_witness}.
        Hence $a\in A\setminus B$ by \obv{setminus}.
    \end{subproof}

    Let $T=\{w\in H\times A\mid \snd{w}\notin\fst{w}\}$.
    Show $T\subseteq H\times A$.
    \begin{subproof}
        It suffices to show that every element of $T$ is an element of $H\times A$
            by \obv{subseteq}.
        Fix $w\in T$.
    \end{subproof}
    Hence $T$ is a relation by \auto{relation,subseteq,times_elem_is_tuple}.
    Show that for all $B\in H$ there exists $a$ such that $B\mathrel{T}a$.
    \begin{subproof}
        Fix $B\in H$.
        Take $a$ such that $a\in A\setminus B$ by \auto.
        $(B,a)\in H\times A$ by \obv{setminus,times_tuple_intro}.
        $\fst{(B,a)}=B$ and $\snd{(B,a)}=a$ by \obv{fst_eq,snd_eq}.
        Hence $(B,a)\in T$ by \obv{setminus}.
    \end{subproof}
    Take $f$ such that $f$ is a function on $H$ and for all $B\in H$ we have $B\mathrel{T}f(B)$
        by \obv{axiom_of_choice}.
    Show that for all $B\in H$ we have $f(B)\in A\setminus B$.
    \begin{subproof}
        Fix $B\in H$.
        $(B,f(B))\in T$.
        Hence $(B,f(B))\in H\times A$.
        Thus $f(B)\in A$ by \obv{times_tuple_elim}.
        $\fst{(B,f(B))}=B$ and $\snd{(B,f(B))}=f(B)$ by \obv{fst_eq,snd_eq}.
        Hence $f(B)\notin B$.
        Follows by \obv{setminus}.
    \end{subproof}

    Let $G=\{p\in\pow{A}\times\pow{A\times A}\mid
        \text{$\snd{p}$ is a well ordering generated by $f$ on $\fst{p}$}\}$.
    $(\emptyset,\emptyset)\in G$
        by \auto{times_tuple_intro,powerset_bottom,empty_generated_well_ordering,fst_eq,snd_eq}.
    Thus $G$ is inhabited.
    For all $p\in G$ we have $\snd{p}$ is a well ordering generated by $f$ on $\fst{p}$.

    Let $U=\{\fst{p}\mid p\in G\}$.
    Let $V=\{\snd{p}\mid p\in G\}$.
    Let $B=\unions{U}$.
    Let $S=\unions{V}$.
    $S$ is a well ordering generated by $f$ on $B$ by \obv{union_generated_well_orderings}.

    Show $B\subseteq A$.
    \begin{subproof}
        Show that for all $C\in U$ we have $C\subseteq A$.
        \begin{subproof}
            Fix $C\in U$.
            Take $p\in G$ such that $C=\fst{p}$.
            $p\in\pow{A}\times\pow{A\times A}$.
            Hence $\fst{p}\in\pow{A}$ by \obv{fst_type}.
            Thus $C\subseteq A$ by \obv{powerset_elim}.
        \end{subproof}
        Follows by \auto{unions_family}.
    \end{subproof}

    Show $B=A$.
    \begin{subproof}
        Suppose not.
        $B\in\pow{A}$ by \obv{powerset_intro}.
        Hence $B\in H$.
        Let $c=f(B)$.
        $c\in A\setminus B$ by \auto.
        Hence $c\in A$ and $c\notin B$ by \obv{setminus}.
        Let $C=\cons{c}{B}$.
        $C\subseteq A$ by \auto{cons_subseteq_intro}.
        Hence $C\in\pow{A}$ by \obv{powerset_intro}.
        Let $R=S\union(B\times\{c\})$.
        $(B,c)\in f$ by \auto{function_apply_elim}.
        $R$ is a well ordering generated by $f$ on $C$
            by \obv{generated_well_ordering_adjoin}.
        $R\subseteq C\times C$ by \auto{generated_well_ordering,strict_well_order_iff}.
        $C\times C\subseteq A\times A$
            by \auto{times_subseteq_left,times_subseteq_right,subseteq_transitive}.
        Hence $R\subseteq A\times A$ by \obv{subseteq_transitive}.
        Thus $R\in\pow{A\times A}$ by \obv{powerset_intro}.
        $(C,R)\in\pow{A}\times\pow{A\times A}$ by \obv{times_tuple_intro}.
        $\fst{(C,R)}=C$ and $\snd{(C,R)}=R$ by \obv{fst_eq,snd_eq}.
        Hence $(C,R)\in G$ by \auto.
        Thus $C\in U$ by \auto.
        Hence $C\subseteq B$ by \auto{subseteq,unions_iff}.
        Therefore $c\in B$ by \obv{cons_left,subseteq}.
        Contradiction.
    \end{subproof}

    Hence $S$ is a strict well order on $A$
        by \obv{generated_well_ordering}.
\end{proof}

\begin{theorem}\label{well_order_exists}
    There exists $R$ such that $R$ is a well order on $A$.
\end{theorem}
\begin{proof}
    Take $S$ such that $S$ is a strict well order on $A$
        by \obv{well_ordering_theorem}.
    $\toorder{A}{S}$ is a well order on $A$
        by \obv{well_order_from_strict_well_order}.
\end{proof}
