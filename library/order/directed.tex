\section{Directed quasiorders}

Directedness here is inhabited and pairwise.  Antisymmetry is deliberately
absent: direct systems only need a common upper index and transitivity.

\begin{struct}\label{directed_quasiordered_set}
    A directed quasiordered set $I$ is a quasiordered set such that
    \begin{enumerate}
        \item\label{directed_quasiorder_inhabited}
            $\carrier[I]$ is inhabited.
        \item\label{directed_quasiorder_upper_bound}
            for all $i,j\in\carrier[I]$ there exists $k\in\carrier[I]$
            such that $i\mathrel{\leq[I]}k$ and
                $j\mathrel{\leq[I]}k$.
    \end{enumerate}
\end{struct}

\begin{proposition}\label{directed_quasiordered_set_intro}
    Let $I$ be a quasiordered set.
    Suppose $\carrier[I]$ is inhabited.
    Suppose for all $i,j\in\carrier[I]$ there exists $k\in\carrier[I]$
        such that $i\mathrel{\leq[I]}k$ and $j\mathrel{\leq[I]}k$.
    Then $I$ is a directed quasiordered set.
\end{proposition}
\begin{proof}
    Follows by definition.
\end{proof}

\begin{proposition}\label{directed_quasiordered_set_is_quasiordered}
    Let $I$ be a directed quasiordered set.
    Then $I$ is a quasiordered set.
\end{proposition}
\begin{proof}
    Follows by \obv{directed_quasiordered_set}.
\end{proof}

\begin{proposition}\label{directed_quasiordered_set_carrier_inhabited}
    Let $I$ be a directed quasiordered set.
    Then $\carrier[I]$ is inhabited.
\end{proposition}
\begin{proof}
    Follows by \obv{directed_quasiorder_inhabited}.
\end{proof}

\begin{theorem}\label{directed_common_upper_bound}
    Let $I$ be a directed quasiordered set.
    Let $i,j\in\carrier[I]$.
    Then there exists $k\in\carrier[I]$ such that
        $i\mathrel{\leq[I]}k$ and $j\mathrel{\leq[I]}k$.
\end{theorem}
\begin{proof}
    Follows by \obv{directed_quasiorder_upper_bound}.
\end{proof}

\begin{proposition}\label{directed_order_reflexive}
    Let $I$ be a directed quasiordered set.
    Let $i\in\carrier[I]$.
    Then $i\mathrel{\leq[I]}i$.
\end{proposition}
\begin{proof}
    $I$ is a quasiordered set by
        \obv{directed_quasiordered_set_is_quasiordered}.
    Follows by \auto{quasiorder_refl,reflexive_on}.
\end{proof}

\begin{proposition}\label{directed_order_transitive}
    Let $I$ be a directed quasiordered set.
    Let $i,j,k\in\carrier[I]$.
    Suppose $i\mathrel{\leq[I]}j$ and $j\mathrel{\leq[I]}k$.
    Then $i\mathrel{\leq[I]}k$.
\end{proposition}
\begin{proof}
    $I$ is a quasiordered set by
        \obv{directed_quasiordered_set_is_quasiordered}.
    Follows by \auto{quasiorder_tran,transitive}.
\end{proof}

\begin{definition}\label{directed_quasiorder}
    $\directedQuasiorder{A}{R}=
        \structure[directed quasiordered set]{
            \carrier=A,
            \leq=R
        }$.
\end{definition}

\begin{proposition}\label{directed_quasiorder_carrier}
    $\carrier[\directedQuasiorder{A}{R}]=A$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{directed_quasiorder,quasiordered_setaggregateprojection_carrier}.
\end{proof}

\begin{proposition}\label{directed_quasiorder_relation}
    $\leq[\directedQuasiorder{A}{R}]=R$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{directed_quasiorder,quasiordered_setaggregateprojection_leq}.
\end{proof}

\begin{theorem}\label{directed_quasiorder_is_directed}
    Suppose $R$ is a quasiorder on $A$ and $A$ is inhabited.
    Suppose for all $i,j\in A$ there exists $k\in A$ such that
        $i\mathrel{R}k$ and $j\mathrel{R}k$.
    Then $\directedQuasiorder{A}{R}$ is a directed quasiordered set.
\end{theorem}
\begin{proof}
    Let $I=\directedQuasiorder{A}{R}$.
    $\carrier[I]=A$ by \auto{directed_quasiorder_carrier}.
    $\leq[I]=R$ by \auto{directed_quasiorder_relation}.
    $\leq[I]\subseteq\carrier[I]\times\carrier[I]$.
    $\leq[I]$ is reflexive on $\carrier[I]$ and
        $\leq[I]$ is transitive.
    $I$ is an exact quasiordered set by
        \auto{directed_quasiorder,quasiordered_setaggregateexact}.
    $\carrier[I]$ is inhabited.
    For all $i,j\in\carrier[I]$ there exists $k\in\carrier[I]$
        such that $i\mathrel{\leq[I]}k$ and
            $j\mathrel{\leq[I]}k$ by \auto.
    $I$ is a quasiordered set and $\carrier[I]$ is inhabited and
        for all $i,j\in\carrier[I]$ there exists $k\in\carrier[I]$
        such that $i\mathrel{\leq[I]}k$ and
            $j\mathrel{\leq[I]}k$.
    Follows by \auto{directed_quasiordered_set}.
\end{proof}

\begin{theorem}\label{directed_quasiorder_is_exact}
    Suppose $R$ is a quasiorder on $A$ and $A$ is inhabited.
    Suppose for all $i,j\in A$ there exists $k\in A$ such that
        $i\mathrel{R}k$ and $j\mathrel{R}k$.
    Then $\directedQuasiorder{A}{R}$ is an exact
        directed quasiordered set.
\end{theorem}
\begin{proof}
    $\directedQuasiorder{A}{R}$ is a directed quasiordered set by
        \obv{directed_quasiorder_is_directed}.
    Follows by
        \auto{directed_quasiorder,quasiordered_setaggregateexact}.
\end{proof}

\begin{definition}\label{ordinal_directed_quasiorder}
    $\ordinalDirectedQuasiorder{\lambda}=
        \directedQuasiorder{\lambda}{\subseteqrel{\lambda}}$.
\end{definition}

\begin{proposition}\label{ordinal_directed_quasiorder_carrier}
    $\carrier[\ordinalDirectedQuasiorder{\lambda}]=\lambda$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{ordinal_directed_quasiorder,directed_quasiorder_carrier}.
\end{proof}

\begin{proposition}\label{ordinal_directed_quasiorder_relation}
    $\leq[\ordinalDirectedQuasiorder{\lambda}]
        =\subseteqrel{\lambda}$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{ordinal_directed_quasiorder,directed_quasiorder_relation}.
\end{proof}

\begin{theorem}\label{ordinal_directed_quasiorder_is_exact}
    Let $\lambda$ be an ordinal.
    Suppose $\lambda\neq\emptyset$.
    Then $\ordinalDirectedQuasiorder{\lambda}$ is an exact
        directed quasiordered set.
\end{theorem}
\begin{proof}
    $\lambda$ is inhabited by \obv{nonempty_inhabited}.
    $\subseteqrel{\lambda}$ is a quasiorder on $\lambda$ by
        \obv{subseteqrel_is_quasiorder}.
    Show that for all $\alpha,\beta\in\lambda$ there exists
        $\gamma\in\lambda$ such that
        $\alpha\mathrel{\subseteqrel{\lambda}}\gamma$ and
        $\beta\mathrel{\subseteqrel{\lambda}}\gamma$.
    \begin{subproof}
        Fix $\alpha,\beta\in\lambda$.
        $\alpha$ and $\beta$ are ordinals by
            \obv{elem_of_ordinal_is_ordinal}.
        $\alpha\subseteq\beta$ or $\beta\subseteq\alpha$ by
            \obv{ordinal_subseteq_or_subseteq}.
        If $\alpha\subseteq\beta$, then
            $\beta\in\lambda$ and
            $\alpha\mathrel{\subseteqrel{\lambda}}\beta$ and
            $\beta\mathrel{\subseteqrel{\lambda}}\beta$ by
            \obv{subseteqrel,subseteq_refl}.
        If $\beta\subseteq\alpha$, then
            $\alpha\in\lambda$ and
            $\alpha\mathrel{\subseteqrel{\lambda}}\alpha$ and
            $\beta\mathrel{\subseteqrel{\lambda}}\alpha$ by
            \auto{subseteqrel,subseteq_refl}.
        Follows by \auto.
    \end{subproof}
    $\directedQuasiorder{\lambda}{\subseteqrel{\lambda}}$ is an exact
        directed quasiordered set by
        \obv{directed_quasiorder_is_exact}.
    Follows by \auto{ordinal_directed_quasiorder}.
\end{proof}

\begin{classification}\label{ordinal_directed_quasiorder_is_directed}
    Let $\lambda$ be an inhabited ordinal.
    Then $\ordinalDirectedQuasiorder{\lambda}$ is a directed
        quasiordered set.
\end{classification}
\begin{proof}
    $\lambda\neq\emptyset$ by \auto{emptyset}.
    Follows by \obv{ordinal_directed_quasiorder_is_exact}.
\end{proof}

\subsection{Directed index substructures}

An index substructure is a literal reduct on a directed subset.  Directedness
of the smaller carrier is part of the contract; an arbitrary inhabited subset
of a directed quasiorder need not be directed.

\begin{definition}\label{index_substructure}
    $\indexSubstructure{J}{I}$ iff
        $I$ is a directed quasiordered set and
        $J$ is a directed quasiordered set and
        $\carrier[J]\subseteq\carrier[I]$ and
        $\leq[J]=\leq[I]\inter
            (\carrier[J]\times\carrier[J])$.
\end{definition}

\begin{proposition}\label{index_substructure_ambient}
    Suppose $\indexSubstructure{J}{I}$.
    Then $I$ is a directed quasiordered set.
\end{proposition}
\begin{proof}
    Follows by \obv{index_substructure}.
\end{proof}

\begin{proposition}\label{index_substructure_is_directed}
    Suppose $\indexSubstructure{J}{I}$.
    Then $J$ is a directed quasiordered set.
\end{proposition}
\begin{proof}
    Follows by \obv{index_substructure}.
\end{proof}

\begin{proposition}\label{index_substructure_carrier_subseteq}
    Suppose $\indexSubstructure{J}{I}$.
    Then $\carrier[J]\subseteq\carrier[I]$.
\end{proposition}
\begin{proof}
    Follows by \obv{index_substructure}.
\end{proof}

\begin{proposition}\label{index_substructure_order_reduct}
    Suppose $\indexSubstructure{J}{I}$.
    Then $\leq[J]=\leq[I]\inter
        (\carrier[J]\times\carrier[J])$.
\end{proposition}
\begin{proof}
    Follows by \obv{index_substructure}.
\end{proof}

\begin{corollary}\label{index_substructure_order_subseteq}
    Suppose $\indexSubstructure{J}{I}$.
    Then $\leq[J]\subseteq\leq[I]$.
\end{corollary}
\begin{proof}
    Follows by
        \auto{index_substructure_order_reduct,inter_lower_left}.
\end{proof}

\begin{proposition}\label{index_substructure_order_iff}
    Suppose $\indexSubstructure{J}{I}$.
    Let $i,j\in\carrier[J]$.
    Then $i\mathrel{\leq[J]}j$ iff $i\mathrel{\leq[I]}j$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{index_substructure_order_reduct,inter,times_tuple_intro}.
\end{proof}

\begin{theorem}\label{ordinal_index_substructure}
    Let $\lambda,\theta$ be ordinals.
    Suppose $\lambda\neq\emptyset$ and $\theta\neq\emptyset$ and
        $\lambda\subseteq\theta$.
    Then $\indexSubstructure
        {\ordinalDirectedQuasiorder{\lambda}}
        {\ordinalDirectedQuasiorder{\theta}}$.
\end{theorem}
\begin{proof}
    Let $J=\ordinalDirectedQuasiorder{\lambda}$.
    Let $I=\ordinalDirectedQuasiorder{\theta}$.
    $J$ is a directed quasiordered set and
        $I$ is a directed quasiordered set by
        \obv{ordinal_directed_quasiorder_is_exact}.
    $\carrier[J]=\lambda$ and $\carrier[I]=\theta$ by
        \auto{ordinal_directed_quasiorder_carrier}.
    $\leq[J]=\subseteqrel{\lambda}$ and
        $\leq[I]=\subseteqrel{\theta}$ by
        \auto{ordinal_directed_quasiorder_relation}.
    Show that $\subseteqrel{\lambda}
        =\subseteqrel{\theta}\inter(\lambda\times\lambda)$.
    \begin{subproof}
        Show that for every $p\in\subseteqrel{\lambda}$ we have
            $p\in\subseteqrel{\theta}\inter
                (\lambda\times\lambda)$.
        \begin{subproof}
            Fix $p\in\subseteqrel{\lambda}$.
            Take $i,j$ such that $p=(i,j)$ and $i,j\in\lambda$ and
                $i\subseteq j$ by \auto{subseteqrel}.
            Hence $i,j\in\theta$ by \obv{subseteq}.
            Hence $p\in\subseteqrel{\theta}$ by
                \auto{subseteqrel}.
            $p\in\lambda\times\lambda$ by
                \auto{times_tuple_intro}.
            Follows by \obv{inter}.
        \end{subproof}
        Show that for every
            $p\in\subseteqrel{\theta}\inter(\lambda\times\lambda)$
            we have $p\in\subseteqrel{\lambda}$.
        \begin{subproof}
            Fix $p\in\subseteqrel{\theta}\inter
                (\lambda\times\lambda)$.
            Hence $p\in\subseteqrel{\theta}$ and
                $p\in\lambda\times\lambda$ by \obv{inter}.
            Take $i,j\in\lambda$ such that $p=(i,j)$ by
                \obv{times_elem_is_tuple}.
            Take $a,b$ such that $p=(a,b)$ and $a,b\in\theta$ and
                $a\subseteq b$ by \auto{subseteqrel}.
            Hence $i=a$ and $j=b$ by \auto{pair_eq_iff}.
            Hence $i\subseteq j$.
            Follows by \auto{subseteqrel}.
        \end{subproof}
        Follows by \obv{setext}.
    \end{subproof}
    Follows by \obv{index_substructure}.
\end{proof}
