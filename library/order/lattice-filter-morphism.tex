\section{Pullback of lattice filters}

\import{order/lattice-filter.tex}
\import{order/lattice-morphism.tex}

\begin{proposition}\label{bounded_lattice_hom_preimage_iff}
    Suppose $\boundedLatticeHom{f}{L}{M}$.
    Then $a\in\preimg{f}{F}$ iff
        $a\in\carrier[L]$ and $f(a)\in F$.
\end{proposition}
\begin{proof}
    $f$ is a function and $\dom{f}=\carrier[L]$ by
        \auto{bounded_lattice_hom_type,funs_elim}.
    Follows by \auto{preimg_iff,function_apply_iff}.
\end{proof}

\begin{theorem}\label{bounded_lattice_hom_preimage_lattice_filter}
    Suppose $\boundedLatticeHom{f}{L}{M}$ and
        $\latticeFilter{F}{M}$.
    Then $\latticeFilter{\preimg{f}{F}}{L}$.
\end{theorem}
\begin{proof}
    $\latticeHom{f}{L}{M}$ by
        \obv{bounded_lattice_hom_is_lattice_hom}.
    $f\in\funs{\carrier[L]}{\carrier[M]}$ by
        \obv{bounded_lattice_hom_type}.
    Show that $\preimg{f}{F}\subseteq\carrier[L]$.
    \begin{subproof}
        Follows by
            \auto{subseteq,bounded_lattice_hom_preimage_iff}.
    \end{subproof}
    $f(\latticeTop[L])=\latticeTop[M]$ by
        \obv{bounded_lattice_hom_top}.
    $\latticeTop[M]\in F$ by \obv{lattice_filter_top}.
    $\latticeTop[L]\in\carrier[L]$ by \obv{lattice_top_type}.
    Hence $\latticeTop[L]\in\preimg{f}{F}$ by
        \auto{bounded_lattice_hom_preimage_iff}.
    Show that $\latticeBottom[L]\notin\preimg{f}{F}$.
    \begin{subproof}
        Assume $\latticeBottom[L]\in\preimg{f}{F}$.
        Hence $f(\latticeBottom[L])\in F$ by
            \auto{bounded_lattice_hom_preimage_iff}.
        $f(\latticeBottom[L])=\latticeBottom[M]$ by
            \obv{bounded_lattice_hom_bottom}.
        Hence $\latticeBottom[M]\in F$.
        $\latticeBottom[M]\notin F$ by
            \obv{lattice_filter_bottom_notin}.
        Contradiction.
    \end{subproof}
    Show that for all $a,b\in\preimg{f}{F}$ we have
        $\meet[L](a,b)\in\preimg{f}{F}$.
    \begin{subproof}
        Fix $a,b\in\preimg{f}{F}$.
        Hence $a,b\in\carrier[L]$ and $f(a),f(b)\in F$ by
            \auto{bounded_lattice_hom_preimage_iff}.
        $\meet[L](a,b)\in\carrier[L]$ by
            \obv{bounded_lattice_is_lattice,
                lattice_is_meet_semilattice,meet_type}.
        $f(\meet[L](a,b))=\meet[M](f(a),f(b))$ by
            \obv{lattice_hom_meet}.
        $\meet[M](f(a),f(b))\in F$ by
            \obv{lattice_filter_meet}.
        Follows by
            \auto{bounded_lattice_hom_preimage_iff}.
    \end{subproof}
    Show that for every $a\in\preimg{f}{F}$ we have
        for every $b\in\carrier[L]$ such that
            $a\mathrel{\leq[L]}b$ we have $b\in\preimg{f}{F}$.
    \begin{subproof}
        Fix $a\in\preimg{f}{F}$ and $b\in\carrier[L]$.
        Assume $a\mathrel{\leq[L]}b$.
        Hence $a\in\carrier[L]$ and $f(a)\in F$ by
            \auto{bounded_lattice_hom_preimage_iff}.
        $f(b)\in\carrier[M]$ by \obv{lattice_hom_apply_type}.
        $\orderPreserving
            {f}{\carrier[L]}{\leq[L]}
            {\carrier[M]}{\leq[M]}$ by
            \obv{lattice_hom_order_preserving}.
        Hence $f(a)\mathrel{\leq[M]}f(b)$ by
            \obv{order_preserving}.
        Hence $f(b)\in F$ by \obv{lattice_filter_upward}.
        Follows by
            \auto{bounded_lattice_hom_preimage_iff}.
    \end{subproof}
    Follows by \obv{lattice_filter}.
\end{proof}
