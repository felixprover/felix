\section{Suprema and infima}

\import{order/bounds.tex}
\import{set/choice.tex}

The chosen supremum and infimum below are total set terms.  Their
specification laws are deliberately guarded by the corresponding existence
predicates.

\begin{definition}\label{supremum_element}
    Let $P$ be an ordered set.
    $u$ is a supremum of $S$ in $P$ iff
        $u$ is a least upper bound of $S$ in $\carrier[P]$
            with respect to $\leq[P]$.
\end{definition}

\begin{definition}\label{infimum_element}
    Let $P$ be an ordered set.
    $u$ is an infimum of $S$ in $P$ iff
        $u$ is a greatest lower bound of $S$ in $\carrier[P]$
            with respect to $\leq[P]$.
\end{definition}

\begin{definition}\label{has_supremum}
    Let $P$ be an ordered set.
    $\hasSupremum{P}{S}$ iff
        there exists $u\in\carrier[P]$ such that
            $u$ is a supremum of $S$ in $P$.
\end{definition}

\begin{definition}\label{has_infimum}
    Let $P$ be an ordered set.
    $\hasInfimum{P}{S}$ iff
        there exists $u\in\carrier[P]$ such that
            $u$ is an infimum of $S$ in $P$.
\end{definition}

\begin{definition}\label{bounded_above_in_ordered_set}
    Let $P$ be an ordered set.
    $S$ is bounded above in $P$ iff
        $S\subseteq\carrier[P]$ and there exists $u\in\carrier[P]$
            such that for every $s\in S$ we have
                $s\mathrel{\leq[P]}u$.
\end{definition}

\begin{definition}\label{bounded_below_in_ordered_set}
    Let $P$ be an ordered set.
    $S$ is bounded below in $P$ iff
        $S\subseteq\carrier[P]$ and there exists $l\in\carrier[P]$
            such that for every $s\in S$ we have
                $l\mathrel{\leq[P]}s$.
\end{definition}

For linear orders, the one-sided least-upper-bound property already contains
its order dual.  We therefore take it as the definition of conditional
completeness and derive the infimum property below.

\begin{definition}\label{conditionally_complete_linear_order}
    Let $P$ be a linearly ordered set.
    $P$ is conditionally complete iff for every
        $S\subseteq\carrier[P]$ such that $S$ is inhabited and
        $S$ is bounded above in $P$ we have $\hasSupremum{P}{S}$.
\end{definition}

\begin{definition}\label{supremum_candidates}
    Let $P$ be an ordered set.
    $\supremumCandidates{P}{S}
        =\{u\in\carrier[P]\mid
            \text{$u$ is a supremum of $S$ in $P$}\}$.
\end{definition}

\begin{definition}\label{infimum_candidates}
    Let $P$ be an ordered set.
    $\infimumCandidates{P}{S}
        =\{u\in\carrier[P]\mid
            \text{$u$ is an infimum of $S$ in $P$}\}$.
\end{definition}

\begin{definition}\label{supremum}
    Let $P$ be an ordered set.
    $\supremum{P}{S}=\choice{\supremumCandidates{P}{S}}$.
\end{definition}

\begin{definition}\label{infimum}
    Let $P$ be an ordered set.
    $\infimum{P}{S}=\choice{\infimumCandidates{P}{S}}$.
\end{definition}

\begin{proposition}\label{has_supremum_iff}
    Let $P$ be an ordered set.
    Then $\hasSupremum{P}{S}$ iff
        there exists $u\in\carrier[P]$ such that
            $u$ is a supremum of $S$ in $P$.
\end{proposition}
\begin{proof}
    Follows by \auto{has_supremum}.
\end{proof}

\begin{proposition}\label{has_infimum_iff}
    Let $P$ be an ordered set.
    Then $\hasInfimum{P}{S}$ iff
        there exists $u\in\carrier[P]$ such that
            $u$ is an infimum of $S$ in $P$.
\end{proposition}
\begin{proof}
    Follows by \auto{has_infimum}.
\end{proof}

\begin{proposition}\label{supremum_candidate_iff}
    Let $P$ be an ordered set.
    Then $u\in\supremumCandidates{P}{S}$ iff
        $u\in\carrier[P]$ and $u$ is a supremum of $S$ in $P$.
\end{proposition}
\begin{proof}
    Follows by \obv{supremum_candidates}.
\end{proof}

\begin{proposition}\label{infimum_candidate_iff}
    Let $P$ be an ordered set.
    Then $u\in\infimumCandidates{P}{S}$ iff
        $u\in\carrier[P]$ and $u$ is an infimum of $S$ in $P$.
\end{proposition}
\begin{proof}
    Follows by \obv{infimum_candidates}.
\end{proof}

\begin{proposition}\label{has_supremum_iff_candidates_inhabited}
    Let $P$ be an ordered set.
    Then $\hasSupremum{P}{S}$ iff
        $\supremumCandidates{P}{S}$ is inhabited.
\end{proposition}
\begin{proof}
    Follows by \auto{has_supremum_iff,supremum_candidate_iff}.
\end{proof}

\begin{proposition}\label{has_infimum_iff_candidates_inhabited}
    Let $P$ be an ordered set.
    Then $\hasInfimum{P}{S}$ iff
        $\infimumCandidates{P}{S}$ is inhabited.
\end{proposition}
\begin{proof}
    Follows by \auto{has_infimum_iff,infimum_candidate_iff}.
\end{proof}

\begin{proposition}\label{supremum_spec}
    Let $P$ be an ordered set.
    Suppose $\hasSupremum{P}{S}$.
    Then $\supremum{P}{S}$ is a supremum of $S$ in $P$.
\end{proposition}
\begin{proof}
    $\supremumCandidates{P}{S}$ is inhabited by
        \auto{has_supremum_iff_candidates_inhabited}.
    Hence $\choice{\supremumCandidates{P}{S}}
        \in\supremumCandidates{P}{S}$ by \obv{choice_in}.
    Follows by \obv{supremum,supremum_candidate_iff}.
\end{proof}

\begin{proposition}\label{infimum_spec}
    Let $P$ be an ordered set.
    Suppose $\hasInfimum{P}{S}$.
    Then $\infimum{P}{S}$ is an infimum of $S$ in $P$.
\end{proposition}
\begin{proof}
    $\infimumCandidates{P}{S}$ is inhabited by
        \auto{has_infimum_iff_candidates_inhabited}.
    Hence $\choice{\infimumCandidates{P}{S}}
        \in\infimumCandidates{P}{S}$ by \obv{choice_in}.
    Follows by \obv{infimum,infimum_candidate_iff}.
\end{proof}

\begin{proposition}\label{supremum_type}
    Let $P$ be an ordered set.
    Suppose $\hasSupremum{P}{S}$.
    Then $\supremum{P}{S}\in\carrier[P]$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{supremum_spec,supremum_element,
            least_upper_bound_is_upper_bound,upper_bound}.
\end{proof}

\begin{proposition}\label{infimum_type}
    Let $P$ be an ordered set.
    Suppose $\hasInfimum{P}{S}$.
    Then $\infimum{P}{S}\in\carrier[P]$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{infimum_spec,infimum_element,
            greatest_lower_bound_is_lower_bound,lower_bound}.
\end{proof}

\begin{proposition}\label{supremum_upper}
    Let $P$ be an ordered set.
    Suppose $\hasSupremum{P}{S}$ and $s\in S$.
    Then $s\mathrel{\leq[P]}\supremum{P}{S}$.
\end{proposition}
\begin{proof}
    $\leq[P]$ is reflexive on $\carrier[P]$ by
        \obv{ordered_set_order_reflexive}.
    Follows by
        \auto{supremum_spec,supremum_element,
            least_upper_bound_is_upper_bound,
            upper_bound_iff_reflexive_on}.
\end{proof}

\begin{proposition}\label{infimum_lower}
    Let $P$ be an ordered set.
    Suppose $\hasInfimum{P}{S}$ and $s\in S$.
    Then $\infimum{P}{S}\mathrel{\leq[P]}s$.
\end{proposition}
\begin{proof}
    $\leq[P]$ is reflexive on $\carrier[P]$ by
        \obv{ordered_set_order_reflexive}.
    Follows by
        \auto{infimum_spec,infimum_element,
            greatest_lower_bound_is_lower_bound,
            lower_bound_iff_reflexive_on}.
\end{proof}

\begin{proposition}\label{supremum_least}
    Let $P$ be an ordered set.
    Suppose $\hasSupremum{P}{S}$ and $u\in\carrier[P]$.
    Suppose for all $s\in S$ we have $s\mathrel{\leq[P]}u$.
    Then $\supremum{P}{S}\mathrel{\leq[P]}u$.
\end{proposition}
\begin{proof}
    $\leq[P]$ is an order on $\carrier[P]$ by
        \obv{ordered_set_order_on_carrier}.
    Follows by
        \auto{supremum_spec,supremum_element,
            least_upper_bound_iff_order_on}.
\end{proof}

\begin{proposition}\label{infimum_greatest}
    Let $P$ be an ordered set.
    Suppose $\hasInfimum{P}{S}$ and $u\in\carrier[P]$.
    Suppose for all $s\in S$ we have $u\mathrel{\leq[P]}s$.
    Then $u\mathrel{\leq[P]}\infimum{P}{S}$.
\end{proposition}
\begin{proof}
    $\leq[P]$ is an order on $\carrier[P]$ by
        \obv{ordered_set_order_on_carrier}.
    Follows by
        \auto{infimum_spec,infimum_element,
            greatest_lower_bound_iff_order_on}.
\end{proof}

\begin{proposition}\label{supremum_eq_of_is_supremum}
    Let $P$ be an ordered set.
    Suppose $u$ is a supremum of $S$ in $P$.
    Then $\supremum{P}{S}=u$.
\end{proposition}
\begin{proof}
    $u\in\carrier[P]$ by
        \auto{supremum_element,least_upper_bound_is_upper_bound,upper_bound}.
    Hence $\hasSupremum{P}{S}$ by \obv{has_supremum_iff}.
    $\supremum{P}{S}$ is a supremum of $S$ in $P$ by
        \obv{supremum_spec}.
    $\leq[P]$ is antisymmetric by
        \obv{ordered_set_order_antisymmetric}.
    Follows by \auto{supremum_element,least_upper_bound_unique}.
\end{proof}

\begin{proposition}\label{infimum_eq_of_is_infimum}
    Let $P$ be an ordered set.
    Suppose $u$ is an infimum of $S$ in $P$.
    Then $\infimum{P}{S}=u$.
\end{proposition}
\begin{proof}
    $u\in\carrier[P]$ by
        \auto{infimum_element,greatest_lower_bound_is_lower_bound,lower_bound}.
    Hence $\hasInfimum{P}{S}$ by \obv{has_infimum_iff}.
    $\infimum{P}{S}$ is an infimum of $S$ in $P$ by
        \obv{infimum_spec}.
    $\leq[P]$ is antisymmetric by
        \obv{ordered_set_order_antisymmetric}.
    Follows by \auto{infimum_element,greatest_lower_bound_unique}.
\end{proof}

\begin{theorem}\label{conditionally_complete_has_infimum}
    Let $P$ be a conditionally complete linearly ordered set.
    Suppose $S$ is inhabited and $S$ is bounded below in $P$.
    Then $\hasInfimum{P}{S}$.
\end{theorem}
\begin{proof}
    $P$ is an ordered set by
        \obv{linearly_ordered_set_is_ordered_set}.
    $S\subseteq\carrier[P]$ by
        \obv{bounded_below_in_ordered_set}.
    Take $a$ such that $a\in S$.
    Let $B=\{b\in\carrier[P]\mid
        \text{for every $s\in S$ we have $b\mathrel{\leq[P]}s$}\}$.

    Take $l\in\carrier[P]$ such that for every $s\in S$ we have
        $l\mathrel{\leq[P]}s$ by
        \obv{bounded_below_in_ordered_set}.
    Hence $l\in B$.
    Thus $B$ is inhabited.
    $B\subseteq\carrier[P]$ by \obv{subseteq}.
    $a\in\carrier[P]$ by \obv{subseteq}.
    For every $b\in B$ we have $b\mathrel{\leq[P]}a$.
    Hence $B$ is bounded above in $P$ by
        \obv{bounded_above_in_ordered_set}.
    $\hasSupremum{P}{B}$ by
        \obv{conditionally_complete_linear_order}.
    Let $u=\supremum{P}{B}$.
    $u\in\carrier[P]$ by \auto{supremum_type}.

    Show that for every $s\in S$ we have $u\mathrel{\leq[P]}s$.
    \begin{subproof}
        Fix $s\in S$.
        $s\in\carrier[P]$ by \obv{subseteq}.
        For every $b\in B$ we have $b\mathrel{\leq[P]}s$.
        $\supremum{P}{B}\mathrel{\leq[P]}s$ by
            \obv{supremum_least}.
    \end{subproof}
    Show that for every $v\in\carrier[P]$ such that
        for every $s\in S$ we have $v\mathrel{\leq[P]}s$ we have
        $v\mathrel{\leq[P]}u$.
    \begin{subproof}
        Fix $v\in\carrier[P]$.
        Assume for every $s\in S$ we have $v\mathrel{\leq[P]}s$.
        Hence $v\in B$.
        $v\mathrel{\leq[P]}\supremum{P}{B}$ by
            \obv{supremum_upper}.
    \end{subproof}
    $\leq[P]$ is an order on $\carrier[P]$ by
        \obv{ordered_set_order_on_carrier}.
    Hence $u$ is a greatest lower bound of $S$ in $\carrier[P]$
        with respect to $\leq[P]$ by
        \auto{greatest_lower_bound_iff_order_on}.
    Thus $u$ is an infimum of $S$ in $P$ by
        \obv{infimum_element}.
    Follows by \obv{has_infimum_iff}.
\end{proof}
