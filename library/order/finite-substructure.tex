\section{Finite induced ordered subsets}

\import{order/finite.tex}
\import{order/substructure.tex}
\import{structure/finite.tex}

This small companion module contains the structure-level finiteness theorem.
Keeping the structure adjective out of the general finite-order module avoids
making ordinary set-finiteness phrases ambiguous throughout its large import
cone.

\begin{theorem}\label{finite_ordered_subset}
    Let $P$ be a finite ordered set.
    Then $\orderedSubset{P}{A}$ is a finite ordered set.
\end{theorem}
\begin{proof}
    $\orderedSubset{P}{A}$ is an ordered set by
        \obv{ordered_subset_is_ordered_set}.
    Hence $\orderedSubset{P}{A}$ is a onesorted structure by
        \obv{quasiordered_set}.
    $\carrier[P]$ is a finite set by \obv{finite_structure}.
    $\carrier[\orderedSubset{P}{A}]\subseteq\carrier[P]$ by
        \auto{ordered_subset_carrier,inter_lower_right}.
    Hence $\carrier[\orderedSubset{P}{A}]$ is a finite set by
        \obv{subseteq_finite_is_finite}.
    Follows by \obv{finite_structure}.
\end{proof}
