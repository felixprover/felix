\subsection{The finite join map}

\begin{proposition}\label{join_irreducible_poset_is_finite}
    Let $L$ be a finite bounded lattice.
    Then $\joinIrreduciblePoset{L}$ is a finite ordered set.
\end{proposition}
\begin{proof}
    Hence $L$ is a finite ordered set.
    $\joinIrreduciblePoset{L}
        =\orderedSubset{L}{\joinIrreducibles{L}}$ by
        \obv{join_irreducible_poset}.
    Follows by \auto{finite_ordered_subset}.
\end{proof}

\begin{theorem}\label{finite_birkhoff_downset_lattice}
    Let $L$ be a finite bounded lattice.
    Then $\downsetLattice{\joinIrreduciblePoset{L}}$ is a finite
        bounded distributive lattice.
\end{theorem}
\begin{proof}
    $\joinIrreduciblePoset{L}$ is a finite ordered set by
        \obv{join_irreducible_poset_is_finite}.
    Hence $\joinIrreduciblePoset{L}$ is a finite quasiordered set by
        \obv{orderedset}.
    Follows by \obv{finite_downset_lattice}.
\end{proof}

\begin{proposition}
    \label{downset_of_join_irreducibles_is_finite_carrier_subset}
    Let $L$ be a finite bounded lattice.
    Let $D\in\downsets{\joinIrreduciblePoset{L}}$.
    Then $D\subseteq\carrier[L]$ and $D$ is a finite set.
\end{proposition}
\begin{proof}
    $D$ is a downset in $\joinIrreduciblePoset{L}$ by
        \obv{downsets_iff}.
    Hence $D\subseteq\carrier[\joinIrreduciblePoset{L}]$ by
        \obv{downset_subseteq_carrier}.
    $\carrier[\joinIrreduciblePoset{L}]=\joinIrreducibles{L}$ by
        \obv{join_irreducible_poset_carrier}.
    $\joinIrreducibles{L}\subseteq\carrier[L]$ by
        \obv{join_irreducibles_subseteq_carrier}.
    Hence $D\subseteq\carrier[L]$ by \obv{subseteq_transitive}.
    $\carrier[L]$ is a finite set by \obv{finite_structure}.
    Follows by \obv{subseteq_finite_is_finite}.
\end{proof}

\begin{definition}\label{birkhoff_join_map}
    Let $L$ be a finite bounded lattice.
    $\birkhoffJoinMap{L}
        =\Apply{\graphOnDomainOp}
            {\downsets{\joinIrreduciblePoset{L}},
             \fun{D:\Set}{\finiteJoin{L}{D}}}$.
\end{definition}

\begin{theorem}\label{birkhoff_join_map_type}
    Let $L$ be a finite bounded lattice.
    Then $\birkhoffJoinMap{L}\in
        \funs{\downsets{\joinIrreduciblePoset{L}}}{\carrier[L]}$.
\end{theorem}
\begin{proof}
    For every $D\in\downsets{\joinIrreduciblePoset{L}}$ we have
        $D\subseteq\carrier[L]$ and $D$ is a finite set by
        \obv{downset_of_join_irreducibles_is_finite_carrier_subset}.
    Hence for every $D\in\downsets{\joinIrreduciblePoset{L}}$ we have
        $\finiteJoin{L}{D}\in\carrier[L]$ by
        \auto{finite_join_type}.
    Follows by
        \auto{birkhoff_join_map,logical_graph_on_domain_type}.
\end{proof}

\begin{proposition}\label{birkhoff_join_map_is_function}
    Let $L$ be a finite bounded lattice.
    Then $\birkhoffJoinMap{L}$ is a function.
\end{proposition}
\begin{proof}
    Follows by \auto{birkhoff_join_map_type,funs_is_function}.
\end{proof}

\begin{proposition}\label{birkhoff_join_map_apply}
    Let $L$ be a finite bounded lattice.
    Let $D\in\downsets{\joinIrreduciblePoset{L}}$.
    Then $\apply{\birkhoffJoinMap{L}}{D}=\finiteJoin{L}{D}$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{birkhoff_join_map,logical_graph_on_domain_apply}.
\end{proof}

\begin{theorem}\label{birkhoff_join_leq_iff_subseteq_map}
    Let $L$ be a finite bounded lattice.
    Let $D\in\downsets{\joinIrreduciblePoset{L}}$.
    Let $x\in\carrier[L]$.
    Then $\apply{\birkhoffJoinMap{L}}{D}\mathrel{\leq[L]}x$ iff
        $D\subseteq\apply{\birkhoffMap{L}}{x}$.
\end{theorem}
\begin{proof}
    $D\subseteq\carrier[L]$ and $D$ is a finite set by
        \obv{downset_of_join_irreducibles_is_finite_carrier_subset}.
    $\apply{\birkhoffJoinMap{L}}{D}=\finiteJoin{L}{D}$ and
        $\apply{\birkhoffMap{L}}{x}=\joinIrreduciblesBelow{L}{x}$ by
        \obv{birkhoff_join_map_apply,birkhoff_map_apply}.
    $\finiteJoin{L}{D}\mathrel{\leq[L]}x$ iff
        for all $d\in D$ we have $d\mathrel{\leq[L]}x$ by
        \obv{finite_join_leq_iff}.
    $D$ is a downset in $\joinIrreduciblePoset{L}$ by
        \obv{downsets_iff}.
    Hence $D\subseteq\carrier[\joinIrreduciblePoset{L}]$ by
        \obv{downset_subseteq_carrier}.
    $\carrier[\joinIrreduciblePoset{L}]=\joinIrreducibles{L}$ by
        \obv{join_irreducible_poset_carrier}.
    For every $d\in D$ we have $d\in\joinIrreducibles{L}$ by
        \obv{elem_subseteq}.
    Hence for every $d\in D$ we have
        $d\in\joinIrreduciblesBelow{L}{x}$ iff
            $d\mathrel{\leq[L]}x$ by
        \auto{join_irreducibles_iff,join_irreducibles_below_iff}.
    Follows by \auto{subseteq}.
\end{proof}

\subsection{The Birkhoff adjunction and retract}

\begin{theorem}\label{finite_birkhoff_order_adjunction}
    Let $L$ be a finite bounded lattice.
    Then
        $\orderAdjunction
            {\birkhoffJoinMap{L}}
            {\downsetLattice{\joinIrreduciblePoset{L}}}
            {L}
            {\birkhoffMap{L}}$.
\end{theorem}
\begin{proof}
    Let $P=\joinIrreduciblePoset{L}$.
    Let $Q=\downsetLattice{P}$.
    $\joinIrreduciblePoset{L}$ is an ordered set by
        \obv{join_irreducible_poset_is_ordered_set}.
    Hence $\joinIrreduciblePoset{L}$ is a quasiordered set by
        \obv{orderedset}.
    $\downsetLattice{\joinIrreduciblePoset{L}}$ is a bounded
        distributive lattice by
        \obv{downset_lattice_is_bounded_distributive_lattice}.
    Hence $\downsetLattice{\joinIrreduciblePoset{L}}$ is a bounded
        lattice by
        \obv{bounded_distributive_lattice_is_bounded_lattice}.
    Hence $\downsetLattice{\joinIrreduciblePoset{L}}$ is a lattice by
        \obv{bounded_lattice_is_lattice}.
    Hence $\downsetLattice{\joinIrreduciblePoset{L}}$ is an ordered
        set by \obv{lattice_is_ordered_set}.
    Hence $\downsetLattice{\joinIrreduciblePoset{L}}$ is a
        quasiordered set by \obv{ordered_set_is_quasiordered_set}.
    Hence $P$ is a quasiordered set and $Q$ is a quasiordered set.
    $\carrier[Q]=\downsets{P}$ by
        \auto{downset_lattice_carrier}.
    $\birkhoffJoinMap{L}\in\funs{\carrier[Q]}{\carrier[L]}$ by
        \auto{birkhoff_join_map_type}.
    $\birkhoffMap{L}\in\funs{\carrier[L]}{\carrier[Q]}$ by
        \auto{birkhoff_map_type}.
    Show for all $D,x$ such that $D\in\carrier[Q]$ and
        $x\in\carrier[L]$ we have
        $\apply{\birkhoffJoinMap{L}}{D}\mathrel{\leq[L]}x$ iff
        $D\mathrel{\leq[Q]}\apply{\birkhoffMap{L}}{x}$.
    \begin{subproof}
        Fix $D,x$.
        Assume $D\in\carrier[Q]$ and $x\in\carrier[L]$.
        Hence $D\in\downsets{P}$.
        $\leq[Q]=\subseteqrel{\downsets{P}}$ by
            \auto{downset_lattice_order}.
        $\apply{\birkhoffMap{L}}{x}\in\downsets{P}$ by
            \obv{birkhoff_map_type,funs_type_apply}.
        $D\mathrel{\leq[Q]}\apply{\birkhoffMap{L}}{x}$ iff
            $D\subseteq\apply{\birkhoffMap{L}}{x}$ by
            \auto{subseteqrel,pair_eq_iff}.
        Follows by \auto{birkhoff_join_leq_iff_subseteq_map}.
    \end{subproof}
    Follows by \obv{order_adjunction}.
\end{proof}

\begin{proposition}\label{birkhoff_unit_subseteq}
    Let $L$ be a finite bounded lattice.
    Let $D\in\downsets{\joinIrreduciblePoset{L}}$.
    Then $D\subseteq
        \apply{\birkhoffMap{L}}{\apply{\birkhoffJoinMap{L}}{D}}$.
\end{proposition}
\begin{proof}
    $\apply{\birkhoffJoinMap{L}}{D}\in\carrier[L]$ by
        \auto{birkhoff_join_map_type,funs_type_apply}.
    $\leq[L]$ is reflexive on $\carrier[L]$ by
        \obv{ordered_set_order_reflexive}.
    Hence $\apply{\birkhoffJoinMap{L}}{D}
        \mathrel{\leq[L]}\apply{\birkhoffJoinMap{L}}{D}$ by
        \obv{reflexive_on}.
    Follows by \obv{birkhoff_join_leq_iff_subseteq_map}.
\end{proof}

\begin{proposition}\label{birkhoff_counit_leq}
    Let $L$ be a finite bounded lattice.
    Let $x\in\carrier[L]$.
    Then $\apply{\birkhoffJoinMap{L}}{\apply{\birkhoffMap{L}}{x}}
        \mathrel{\leq[L]}x$.
\end{proposition}
\begin{proof}
    $\apply{\birkhoffMap{L}}{x}\in
        \downsets{\joinIrreduciblePoset{L}}$ by
        \auto{birkhoff_map_type,funs_type_apply}.
    $\apply{\birkhoffMap{L}}{x}
        \subseteq\apply{\birkhoffMap{L}}{x}$ by
        \obv{subseteq_refl}.
    Follows by \obv{birkhoff_join_leq_iff_subseteq_map}.
\end{proof}

\begin{theorem}\label{birkhoff_join_after_map}
    Let $L$ be a finite bounded lattice.
    Let $x\in\carrier[L]$.
    Then $\apply{\birkhoffJoinMap{L}}{\apply{\birkhoffMap{L}}{x}}=x$.
\end{theorem}
\begin{proof}
    $\apply{\birkhoffMap{L}}{x}\in
        \downsets{\joinIrreduciblePoset{L}}$ by
        \auto{birkhoff_map_type,funs_type_apply}.
    $\apply{\birkhoffMap{L}}{x}=\joinIrreduciblesBelow{L}{x}$ by
        \obv{birkhoff_map_apply}.
    $\apply{\birkhoffJoinMap{L}}{\apply{\birkhoffMap{L}}{x}}
        =\finiteJoin{L}{\apply{\birkhoffMap{L}}{x}}$ by
        \obv{birkhoff_join_map_apply}.
    $\joinIrreducibleReconstruction{L}{x}
        =\finiteJoin{L}{\joinIrreduciblesBelow{L}{x}}$ by
        \obv{join_irreducible_reconstruction_apply}.
    Follows by \obv{finite_join_irreducible_decomposition}.
\end{proof}

\begin{proposition}\label{birkhoff_join_map_is_left_inverse}
    Let $L$ be a finite bounded lattice.
    Then $\birkhoffJoinMap{L}$ is a left inverse of $\birkhoffMap{L}$.
\end{proposition}
\begin{proof}
    $\birkhoffMap{L}\in
        \funs{\carrier[L]}{\downsets{\joinIrreduciblePoset{L}}}$ by
        \obv{birkhoff_map_type}.
    $\birkhoffJoinMap{L}\in
        \funs{\downsets{\joinIrreduciblePoset{L}}}{\carrier[L]}$ by
        \obv{birkhoff_join_map_type}.
    Follows by \auto{leftinverse_from_to_iff,birkhoff_join_after_map}.
\end{proof}

\begin{proposition}\label{birkhoff_map_is_right_inverse_of_join_map}
    Let $L$ be a finite bounded lattice.
    Then $\birkhoffMap{L}$ is a right inverse of
        $\birkhoffJoinMap{L}$.
\end{proposition}
\begin{proof}
    $\birkhoffJoinMap{L}\in
        \funs{\downsets{\joinIrreduciblePoset{L}}}{\carrier[L]}$ and
        $\birkhoffMap{L}\in
        \funs{\carrier[L]}{\downsets{\joinIrreduciblePoset{L}}}$ by
        \obv{birkhoff_join_map_type,birkhoff_map_type}.

\end{proof}

\begin{theorem}\label{finite_birkhoff_order_embedding}
    Let $L$ be a finite bounded lattice.
    Then
        $\orderEmbedding
            {\birkhoffMap{L}}
            {\carrier[L]}{\leq[L]}
            {\carrier[\downsetLattice{\joinIrreduciblePoset{L}}]}
            {\leq[\downsetLattice{\joinIrreduciblePoset{L}}]}$.
\end{theorem}
\begin{proof}
    Let $P=\downsetLattice{\joinIrreduciblePoset{L}}$.
    $\downsetLattice{\joinIrreduciblePoset{L}}$ is a finite bounded
        distributive lattice by
        \obv{finite_birkhoff_downset_lattice}.
    Hence $\downsetLattice{\joinIrreduciblePoset{L}}$ is a bounded
        lattice by
        \obv{bounded_distributive_lattice_is_bounded_lattice}.
    Hence $\downsetLattice{\joinIrreduciblePoset{L}}$ is an ordered
        set by \obv{lattice_is_ordered_set}.
    Hence $\downsetLattice{\joinIrreduciblePoset{L}}$ is a
        quasiordered set by \obv{ordered_set_is_quasiordered_set}.
    Hence $L$ is a quasiordered set by \obv{orderedset}.
    $\orderAdjunction{\birkhoffJoinMap{L}}{P}{L}{\birkhoffMap{L}}$ by
        \auto{finite_birkhoff_order_adjunction}.
    For all $x\in\carrier[L]$ we have
        $\apply{\birkhoffJoinMap{L}}{\apply{\birkhoffMap{L}}{x}}=x$ by
        \obv{birkhoff_join_after_map}.
    Follows by
        \auto{order_adjunction_right_embedding_of_counit_identity}.
\end{proof}

\begin{proposition}\label{birkhoff_map_order_iff}
    Let $L$ be a finite bounded lattice.
    Let $x,y\in\carrier[L]$.
    Then $x\mathrel{\leq[L]}y$ iff
        $\apply{\birkhoffMap{L}}{x}
            \subseteq\apply{\birkhoffMap{L}}{y}$.
\end{proposition}
\begin{proof}
    $\orderEmbedding
        {\birkhoffMap{L}}
        {\carrier[L]}{\leq[L]}
        {\carrier[\downsetLattice{\joinIrreduciblePoset{L}}]}
        {\leq[\downsetLattice{\joinIrreduciblePoset{L}}]}$ by
        \obv{finite_birkhoff_order_embedding}.
    Hence $x\mathrel{\leq[L]}y$ iff
        $\apply{\birkhoffMap{L}}{x}
            \mathrel{\leq[\downsetLattice{\joinIrreduciblePoset{L}}]}
                \apply{\birkhoffMap{L}}{y}$ by
        \obv{order_embedding_order_iff}.
    $\joinIrreduciblePoset{L}$ is an ordered set by
        \obv{join_irreducible_poset_is_ordered_set}.
    $\leq[\downsetLattice{\joinIrreduciblePoset{L}}]
        =\subseteqrel{\downsets{\joinIrreduciblePoset{L}}}$ by
        \obv{downset_lattice_order}.
    $\apply{\birkhoffMap{L}}{x},\apply{\birkhoffMap{L}}{y}
        \in\downsets{\joinIrreduciblePoset{L}}$ by
        \auto{birkhoff_map_type,funs_type_apply}.
    Follows by \auto{subseteqrel,pair_eq_iff}.
\end{proof}

\begin{theorem}\label{birkhoff_join_map_join}

    $D\union E$ is a finite set by
        \obv{union_of_finite_is_finite}.
    $\join[\downsetLattice{P}](D,E)=D\union E$ by

    $\apply{\birkhoffJoinMap{L}}{D}=\finiteJoin{L}{D}$ and
        $\apply{\birkhoffJoinMap{L}}{E}=\finiteJoin{L}{E}$ and
        $\apply{\birkhoffJoinMap{L}}{D\union E}

\subsection{The distributive representation}

\begin{theorem}\label{birkhoff_map_after_join}
    Let $L$ be a finite bounded lattice.
    Suppose $L$ is a distributive lattice.
    Let $D\in\downsets{\joinIrreduciblePoset{L}}$.
    Then $\apply{\birkhoffMap{L}}{\apply{\birkhoffJoinMap{L}}{D}}=D$.
\end{theorem}
\begin{proof}
    $D$ is a downset in $\joinIrreduciblePoset{L}$ by
        \obv{downsets_iff}.
    Hence $D\subseteq\carrier[\joinIrreduciblePoset{L}]$ by
        \obv{downset_subseteq_carrier}.
    $\carrier[\joinIrreduciblePoset{L}]=\joinIrreducibles{L}$ by
        \obv{join_irreducible_poset_carrier}.
    Let $x=\apply{\birkhoffJoinMap{L}}{D}$.
    $x\in\carrier[L]$ by
        \auto{birkhoff_join_map_type,funs_type_apply}.
    $D\subseteq\apply{\birkhoffMap{L}}{x}$ by
        \auto{birkhoff_unit_subseteq}.
    Show for every $j\in\apply{\birkhoffMap{L}}{x}$ we have $j\in D$.
    \begin{subproof}
        Fix $j\in\apply{\birkhoffMap{L}}{x}$.
        Hence $j\in\joinIrreduciblesBelow{L}{x}$ by
            \obv{birkhoff_map_apply}.
        Hence $j$ is join-irreducible in $L$ and
            $j\mathrel{\leq[L]}x$ by
            \obv{join_irreducibles_below_iff}.
        Hence $j$ is join-prime in $L$ by
            \obv{join_irreducible_implies_join_prime}.
        $D\subseteq\carrier[L]$ and $D$ is a finite set by
            \obv{downset_of_join_irreducibles_is_finite_carrier_subset}.
        $x=\finiteJoin{L}{D}$ by \obv{birkhoff_join_map_apply}.
        Hence $j\mathrel{\leq[L]}\finiteJoin{L}{D}$.
        Take $d\in D$ such that $j\mathrel{\leq[L]}d$ by
            \obv{join_prime_below_finite_join_iff}.
        $j\in\joinIrreducibles{L}$ by
            \obv{join_irreducibles_iff}.
        $d\in\joinIrreducibles{L}$ by \obv{elem_subseteq}.
        Hence $j\mathrel{\leq[\joinIrreduciblePoset{L}]}d$ by
            \obv{join_irreducible_poset_order_iff}.
        Follows by \obv{downset_closed_downward}.
    \end{subproof}
    Follows by \obv{subseteq_antisymmetric,subseteq}.
\end{proof}

\begin{proposition}\label{birkhoff_join_map_is_right_inverse}

\begin{proof}

        $\birkhoffJoinMap{L}\in
        \funs{\downsets{\joinIrreduciblePoset{L}}}{\carrier[L]}$ by
        \obv{birkhoff_map_type,birkhoff_join_map_type}.
    Follows by \auto{rightinverse_from_to_iff,birkhoff_map_after_join}.
\end{proof}

\subsection{Birkhoff closure and fixed points}

\begin{definition}\label{birkhoff_closure}
    Let $L$ be a finite bounded lattice.
    $\birkhoffClosure{L}=\birkhoffMap{L}\circ\birkhoffJoinMap{L}$.
\end{definition}

\begin{proposition}\label{birkhoff_closure_type}
    Let $L$ be a finite bounded lattice.
    Then $\birkhoffClosure{L}\in
        \funs{\downsets{\joinIrreduciblePoset{L}}}
            {\downsets{\joinIrreduciblePoset{L}}}$.
\end{proposition}
\begin{proof}
    $\birkhoffJoinMap{L}\in
        \funs{\downsets{\joinIrreduciblePoset{L}}}{\carrier[L]}$ and
        $\birkhoffMap{L}\in
        \funs{\carrier[L]}{\downsets{\joinIrreduciblePoset{L}}}$ by
        \obv{birkhoff_join_map_type,birkhoff_map_type}.
    Hence $\birkhoffMap{L}\circ\birkhoffJoinMap{L}\in
        \funs{\downsets{\joinIrreduciblePoset{L}}}
            {\downsets{\joinIrreduciblePoset{L}}}$ by
        \obv{funs_circ}.
    Follows by \obv{birkhoff_closure}.
\end{proof}

\begin{proposition}\label{birkhoff_closure_apply}
    Let $L$ be a finite bounded lattice.
    Let $D\in\downsets{\joinIrreduciblePoset{L}}$.
    Then $\apply{\birkhoffClosure{L}}{D}
        =\apply{\birkhoffMap{L}}{\apply{\birkhoffJoinMap{L}}{D}}$.
\end{proposition}
\begin{proof}
    $\birkhoffJoinMap{L}\in
        \funs{\downsets{\joinIrreduciblePoset{L}}}{\carrier[L]}$ and
        $\birkhoffMap{L}\in
        \funs{\carrier[L]}{\downsets{\joinIrreduciblePoset{L}}}$ by
        \obv{birkhoff_join_map_type,birkhoff_map_type}.
    Hence $\birkhoffJoinMap{L}$ and $\birkhoffMap{L}$ are functions by
        \obv{funs_is_function}.
    $\dom{\birkhoffJoinMap{L}}
        =\downsets{\joinIrreduciblePoset{L}}$ by \obv{funs}.
    Hence $D\in\dom{\birkhoffJoinMap{L}}$.
    $\birkhoffMap{L}$ is composable with $\birkhoffJoinMap{L}$ by
        \auto{funs_ran,funs}.
    $(\birkhoffMap{L}\circ\birkhoffJoinMap{L})(D)
        =\apply{\birkhoffMap{L}}{\apply{\birkhoffJoinMap{L}}{D}}$ by
        \obv{circ_apply}.
    Follows by \auto{birkhoff_closure}.
\end{proof}

\begin{proposition}\label{birkhoff_closure_extensive}
    Let $L$ be a finite bounded lattice.
    Let $D\in\downsets{\joinIrreduciblePoset{L}}$.
    Then $D\subseteq\apply{\birkhoffClosure{L}}{D}$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{birkhoff_closure_apply,birkhoff_unit_subseteq}.
\end{proof}

\begin{proposition}\label{birkhoff_closure_monotone}
    Let $L$ be a finite bounded lattice.
    Let $D,E\in\downsets{\joinIrreduciblePoset{L}}$.
    Suppose $D\subseteq E$.
    Then $\apply{\birkhoffClosure{L}}{D}
        \subseteq\apply{\birkhoffClosure{L}}{E}$.
\end{proposition}
\begin{proof}
    $D,E\subseteq\carrier[L]$ and $D$ and $E$ are finite sets by
        \obv{downset_of_join_irreducibles_is_finite_carrier_subset}.
    Hence $\finiteJoin{L}{D}\mathrel{\leq[L]}\finiteJoin{L}{E}$ by
        \obv{finite_join_monotone}.
    $\apply{\birkhoffJoinMap{L}}{D}=\finiteJoin{L}{D}$ and
        $\apply{\birkhoffJoinMap{L}}{E}=\finiteJoin{L}{E}$ by
        \obv{birkhoff_join_map_apply}.
    $\apply{\birkhoffJoinMap{L}}{D},
        \apply{\birkhoffJoinMap{L}}{E}\in\carrier[L]$ by
        \auto{birkhoff_join_map_type,funs_type_apply}.
    Hence $\apply{\birkhoffJoinMap{L}}{D}
        \mathrel{\leq[L]}\apply{\birkhoffJoinMap{L}}{E}$.
    Hence $\apply{\birkhoffMap{L}}{\apply{\birkhoffJoinMap{L}}{D}}
        \subseteq
        \apply{\birkhoffMap{L}}{\apply{\birkhoffJoinMap{L}}{E}}$ by
        \obv{birkhoff_map_order_iff}.
    Follows by \obv{birkhoff_closure_apply}.
\end{proof}

\begin{proposition}\label{birkhoff_closure_idempotent}
    Let $L$ be a finite bounded lattice.
    Let $D\in\downsets{\joinIrreduciblePoset{L}}$.
    Then $\apply{\birkhoffClosure{L}}{\apply{\birkhoffClosure{L}}{D}}
        =\apply{\birkhoffClosure{L}}{D}$.
\end{proposition}
\begin{proof}
    Let $x=\apply{\birkhoffJoinMap{L}}{D}$.
    $x\in\carrier[L]$ by
        \auto{birkhoff_join_map_type,funs_type_apply}.
    $\apply{\birkhoffClosure{L}}{D}=\apply{\birkhoffMap{L}}{x}$ by
        \auto{birkhoff_closure_apply}.
    Hence $\apply{\birkhoffClosure{L}}{D}
        \in\downsets{\joinIrreduciblePoset{L}}$ by
        \auto{birkhoff_map_type,funs_type_apply}.
    $\apply{\birkhoffJoinMap{L}}{\apply{\birkhoffMap{L}}{x}}=x$ by
        \obv{birkhoff_join_after_map}.
    $\apply{\birkhoffClosure{L}}{\apply{\birkhoffClosure{L}}{D}}
        =\apply{\birkhoffMap{L}}
            {\apply{\birkhoffJoinMap{L}}
                {\apply{\birkhoffClosure{L}}{D}}}$ by
        \obv{birkhoff_closure_apply}.
\end{proof}

\begin{theorem}\label{finite_birkhoff_closure_laws}
    Let $L$ be a finite bounded lattice.
    Then $\birkhoffClosure{L}\in
        \funs{\downsets{\joinIrreduciblePoset{L}}}
            {\downsets{\joinIrreduciblePoset{L}}}$ and
        for all $D,E\in\downsets{\joinIrreduciblePoset{L}}$ such that
            $D\subseteq E$ we have
            $D\subseteq\apply{\birkhoffClosure{L}}{D}$ and
            $\apply{\birkhoffClosure{L}}{D}
                \subseteq\apply{\birkhoffClosure{L}}{E}$ and
            $\apply{\birkhoffClosure{L}}
                {\apply{\birkhoffClosure{L}}{D}}
                =\apply{\birkhoffClosure{L}}{D}$.
\end{theorem}
\begin{proof}
    Follows by
        \obv{birkhoff_closure_type,birkhoff_closure_extensive,
            birkhoff_closure_monotone,birkhoff_closure_idempotent}.
\end{proof}

\begin{definition}\label{birkhoff_closed_downsets}
    Let $L$ be a finite bounded lattice.
    $\birkhoffClosedDownsets{L}
        =\{D\in\downsets{\joinIrreduciblePoset{L}}\mid
            \apply{\birkhoffClosure{L}}{D}=D\}$.
\end{definition}

\begin{proposition}\label{birkhoff_closed_downsets_iff}
    Let $L$ be a finite bounded lattice.
    Then $D\in\birkhoffClosedDownsets{L}$ iff
        $D\in\downsets{\joinIrreduciblePoset{L}}$ and
        $\apply{\birkhoffClosure{L}}{D}=D$.
\end{proposition}
\begin{proof}
    Follows by \obv{birkhoff_closed_downsets}.
\end{proof}

\begin{theorem}\label{birkhoff_map_range_eq_closed_downsets}
    Let $L$ be a finite bounded lattice.
    Then $\ran{\birkhoffMap{L}}=\birkhoffClosedDownsets{L}$.
\end{theorem}
\begin{proof}
    $\birkhoffMap{L}\in
        \funs{\carrier[L]}{\downsets{\joinIrreduciblePoset{L}}}$ by
        \obv{birkhoff_map_type}.
    Hence $\birkhoffMap{L}$ is a function and
        $\dom{\birkhoffMap{L}}=\carrier[L]$ by
        \obv{funs_is_function,funs}.
    Show for every $D\in\ran{\birkhoffMap{L}}$ we have
        $D\in\birkhoffClosedDownsets{L}$.
    \begin{subproof}
        Fix $D\in\ran{\birkhoffMap{L}}$.
        Take $x\in\dom{\birkhoffMap{L}}$ such that
            $\apply{\birkhoffMap{L}}{x}=D$ by \obv{fun_ran_iff}.
        Hence $x\in\carrier[L]$.
        $D\in\downsets{\joinIrreduciblePoset{L}}$ by
            \auto{birkhoff_map_type,funs_type_apply}.
        $\apply{\birkhoffJoinMap{L}}
            {\apply{\birkhoffMap{L}}{x}}=x$ by
            \obv{birkhoff_join_after_map}.
        $\apply{\birkhoffClosure{L}}{D}
            =\apply{\birkhoffMap{L}}
                {\apply{\birkhoffJoinMap{L}}{D}}$ by
            \obv{birkhoff_closure_apply}.
        Hence $\apply{\birkhoffClosure{L}}{D}=D$.
        Follows by \obv{birkhoff_closed_downsets_iff}.
    \end{subproof}
    Show for every $D\in\birkhoffClosedDownsets{L}$ we have
        $D\in\ran{\birkhoffMap{L}}$.
    \begin{subproof}
        Fix $D\in\birkhoffClosedDownsets{L}$.
        Hence $D\in\downsets{\joinIrreduciblePoset{L}}$ and
            $\apply{\birkhoffClosure{L}}{D}=D$ by
            \obv{birkhoff_closed_downsets_iff}.
        $\apply{\birkhoffJoinMap{L}}{D}\in\carrier[L]$ by
            \auto{birkhoff_join_map_type,funs_type_apply}.
        $\apply{\birkhoffClosure{L}}{D}
            =\apply{\birkhoffMap{L}}
                {\apply{\birkhoffJoinMap{L}}{D}}$ by
            \obv{birkhoff_closure_apply}.
        Hence $D=\apply{\birkhoffMap{L}}
            {\apply{\birkhoffJoinMap{L}}{D}}$.
        Follows by \auto{fun_ran_iff}.
    \end{subproof}
    Follows by \obv{setext}.
\end{proof}

\begin{theorem}\label{finite_birkhoff_closed_downsets_representation}
    Let $L$ be a finite bounded lattice.
    Then
        $\orderIsomorphism
            {\birkhoffMap{L}}
            {\carrier[L]}{\leq[L]}
            {\carrier[\orderedSubset
                {\downsetLattice{\joinIrreduciblePoset{L}}}
                {\birkhoffClosedDownsets{L}}]}
            {\leq[\orderedSubset
                {\downsetLattice{\joinIrreduciblePoset{L}}}
                {\birkhoffClosedDownsets{L}}]}$.
\end{theorem}
\begin{proof}
    Let $P=\joinIrreduciblePoset{L}$.
    Let $Q=\downsetLattice{P}$.
    Let $C=\birkhoffClosedDownsets{L}$.
    Let $R=\orderedSubset{Q}{C}$.
    $P$ is an ordered set by
        \obv{join_irreducible_poset_is_ordered_set}.
    Hence $P$ is a quasiordered set by
        \obv{ordered_set_is_quasiordered_set}.
    $Q$ is a bounded distributive lattice by
        \obv{downset_lattice_is_bounded_distributive_lattice}.
    Hence $Q$ is a bounded lattice by
        \obv{bounded_distributive_lattice_is_bounded_lattice}.
    Hence $Q$ is a lattice by \obv{bounded_lattice_is_lattice}.
    Hence $Q$ is an ordered set by \obv{lattice_is_ordered_set}.
    $R$ is an ordered set by \obv{ordered_subset_is_ordered_set}.
    $C\subseteq\downsets{P}$ by
        \auto{birkhoff_closed_downsets_iff,subseteq}.
    $\carrier[Q]=\downsets{P}$ by
        \auto{downset_lattice_carrier}.
    Hence $C\subseteq\carrier[Q]$.
    $\carrier[R]=C$ by
        \auto{ordered_subset_carrier_of_subset}.
    $\ran{\birkhoffMap{L}}=C$ by
        \obv{birkhoff_map_range_eq_closed_downsets}.
    $\birkhoffMap{L}\in\funs{\carrier[L]}{\downsets{P}}$ by
        \auto{birkhoff_map_type}.
    Hence $\birkhoffMap{L}$ is a function and
        $\dom{\birkhoffMap{L}}=\carrier[L]$ by
        \obv{funs_is_function,funs}.
    $\birkhoffMap{L}$ is injective by \obv{birkhoff_map_injective}.
    $\birkhoffMap{L}\in
        \funs{\dom{\birkhoffMap{L}}}{\ran{\birkhoffMap{L}}}$ by
        \obv{function_exact_type_to_range}.
    Hence $\birkhoffMap{L}\in\funs{\carrier[L]}{C}$.
    Hence $\birkhoffMap{L}\in
        \surj{\carrier[L]}{\ran{\birkhoffMap{L}}}$ by
        \obv{fun_to_surj}.
    Hence $\birkhoffMap{L}\in\surj{\carrier[L]}{C}$.
    Hence $\birkhoffMap{L}\in\bijections{\carrier[L]}{C}$ by
        \obv{bijections}.
    Hence $\birkhoffMap{L}\in
        \bijections{\carrier[L]}{\carrier[R]}$.
    Show for all $x,y\in\carrier[L]$ we have
        $x\mathrel{\leq[L]}y$ iff
            $\apply{\birkhoffMap{L}}{x}
                \mathrel{\leq[R]}\apply{\birkhoffMap{L}}{y}$.
    \begin{subproof}
        Fix $x,y\in\carrier[L]$.
        $\apply{\birkhoffMap{L}}{x},\apply{\birkhoffMap{L}}{y}
            \in\ran{\birkhoffMap{L}}$ by \auto{fun_ran_iff}.
        Hence $\apply{\birkhoffMap{L}}{x},
            \apply{\birkhoffMap{L}}{y}\in C$.
        Hence $\apply{\birkhoffMap{L}}{x},
            \apply{\birkhoffMap{L}}{y}\in\carrier[R]$.
        $\apply{\birkhoffMap{L}}{x}
            \mathrel{\leq[R]}\apply{\birkhoffMap{L}}{y}$ iff
            $\apply{\birkhoffMap{L}}{x}
                \mathrel{\leq[Q]}\apply{\birkhoffMap{L}}{y}$ by
            \auto{ordered_subset_order_iff_of_membership}.
        $x\mathrel{\leq[L]}y$ iff
            $\apply{\birkhoffMap{L}}{x}
                \mathrel{\leq[Q]}\apply{\birkhoffMap{L}}{y}$ by
            \auto{finite_birkhoff_order_embedding,
                order_embedding_order_iff}.
    \end{subproof}
    $L$ is a bounded lattice by \obv{bounded_lattice}.
    Hence $L$ is a lattice by \obv{bounded_lattice_is_lattice}.
    Hence $L$ is an ordered set by \obv{lattice_is_ordered_set}.
    Hence $L$ is a quasiordered set and $R$ is a quasiordered set by
        \obv{ordered_set_is_quasiordered_set}.
    $\leq[L]$ is a binary relation on $\carrier[L]$ and
        $\leq[R]$ is a binary relation on $\carrier[R]$ by
        \obv{quasiorder_type}.
    Follows by \auto{order_isomorphism}.
\end{proof}

\begin{proposition}\label{birkhoff_closure_identity_of_distributive}

    Let $D\in\downsets{\joinIrreduciblePoset{L}}$.

\end{proposition}

\subsection{Distributivity characterizations}

\begin{theorem}\label{birkhoff_map_surjective_iff_distributive}
    Let $L$ be a finite bounded lattice.
    Then $\birkhoffMap{L}\in
        \surj{\carrier[L]}{\downsets{\joinIrreduciblePoset{L}}}$ iff
        $L$ is a distributive lattice.
\end{theorem}
\begin{proof}
    Show that if $L$ is a distributive lattice, then
        $\birkhoffMap{L}\in
            \surj{\carrier[L]}{\downsets{\joinIrreduciblePoset{L}}}$.
    \begin{subproof}
        Assume $L$ is a distributive lattice.
        $\birkhoffMap{L}\in
            \funs{\carrier[L]}{\downsets{\joinIrreduciblePoset{L}}}$ by
            \obv{birkhoff_map_type}.
        Show for every $D\in\downsets{\joinIrreduciblePoset{L}}$
            there exists $x\in\carrier[L]$ such that
                $\apply{\birkhoffMap{L}}{x}=D$.
        \begin{subproof}
            Fix $D\in\downsets{\joinIrreduciblePoset{L}}$.
            Let $x=\apply{\birkhoffJoinMap{L}}{D}$.
            $x\in\carrier[L]$ by
                \auto{birkhoff_join_map_type,funs_type_apply}.
            $\apply{\birkhoffMap{L}}{x}=D$ by
                \auto{birkhoff_map_after_join}.
        \end{subproof}
        Follows by \obv{surj}.
    \end{subproof}
    Show that if $\birkhoffMap{L}\in
        \surj{\carrier[L]}{\downsets{\joinIrreduciblePoset{L}}}$, then
        $L$ is a distributive lattice.
    \begin{subproof}
        Assume $\birkhoffMap{L}\in
            \surj{\carrier[L]}{\downsets{\joinIrreduciblePoset{L}}}$.
        Let $P=\joinIrreduciblePoset{L}$.
        Let $Q=\downsetLattice{P}$.
        $P$ is an ordered set by
            \obv{join_irreducible_poset_is_ordered_set}.
        Hence $P$ is a quasiordered set by
            \obv{ordered_set_is_quasiordered_set}.
        $Q$ is a bounded distributive lattice by
            \obv{downset_lattice_is_bounded_distributive_lattice}.
        Hence $Q$ is a bounded lattice by
            \obv{bounded_distributive_lattice_is_bounded_lattice}.
        Hence $Q$ is a lattice by \obv{bounded_lattice_is_lattice}.
        $\carrier[Q]=\downsets{P}$ by
            \auto{downset_lattice_carrier}.
        Hence $\birkhoffMap{L}\in
            \surj{\carrier[L]}{\carrier[Q]}$.
        $\orderEmbedding
            {\birkhoffMap{L}}
            {\carrier[L]}{\leq[L]}
            {\carrier[Q]}{\leq[Q]}$ by
            \auto{finite_birkhoff_order_embedding}.
        Hence $\birkhoffMap{L}\in\inj{\carrier[L]}{\carrier[Q]}$ by
            \obv{order_embedding_injective}.
        $\birkhoffMap{L}\in\funs{\carrier[L]}{\carrier[Q]}$ by
            \obv{order_embedding_type}.
        Hence $\birkhoffMap{L}$ is a function and
            $\dom{\birkhoffMap{L}}=\carrier[L]$ and
            $\birkhoffMap{L}$ is injective by
            \auto{funs_is_function,funs,inj,injective_function}.
        Hence $\birkhoffMap{L}\in
            \bijections{\carrier[L]}{\carrier[Q]}$ by
            \obv{bijections}.
        Hence $\orderIsomorphism
            {\birkhoffMap{L}}
            {\carrier[L]}{\leq[L]}
            {\carrier[Q]}{\leq[Q]}$ by
            \obv{order_isomorphism,order_embedding}.
        $L$ is a bounded lattice by \obv{bounded_lattice}.
        Hence $L$ is a lattice by \obv{bounded_lattice_is_lattice}.
        Hence $\latticeIsomorphism{\birkhoffMap{L}}{L}{Q}$ by
            \obv{order_isomorphism_between_lattices_is_lattice_isomorphism}.
        Hence $L$ is a distributive lattice iff
            $Q$ is a distributive lattice by
            \obv{lattice_isomorphism_distributive_iff}.
        $Q$ is a distributive lattice by
            \obv{bounded_distributive_lattice_is_distributive_lattice}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{theorem}\label{birkhoff_closure_identity_iff_distributive}
    Let $L$ be a finite bounded lattice.
    Then
        (for every $D\in\downsets{\joinIrreduciblePoset{L}}$ we have
            $\apply{\birkhoffClosure{L}}{D}=D$) iff
        $L$ is a distributive lattice.
\end{theorem}
\begin{proof}
    Show that if for every
        $D\in\downsets{\joinIrreduciblePoset{L}}$ we have
        $\apply{\birkhoffClosure{L}}{D}=D$, then
        $L$ is a distributive lattice.
    \begin{subproof}
        Assume for every $D\in\downsets{\joinIrreduciblePoset{L}}$ we
            have $\apply{\birkhoffClosure{L}}{D}=D$.
        Show that $\birkhoffMap{L}\in
            \surj{\carrier[L]}{\downsets{\joinIrreduciblePoset{L}}}$.
        \begin{subproof}
            $\birkhoffMap{L}\in
                \funs{\carrier[L]}
                    {\downsets{\joinIrreduciblePoset{L}}}$ by
                \obv{birkhoff_map_type}.
            Show for every $D\in\downsets{\joinIrreduciblePoset{L}}$
                there exists $x\in\carrier[L]$ such that
                    $\apply{\birkhoffMap{L}}{x}=D$.
            \begin{subproof}
                Fix $D\in\downsets{\joinIrreduciblePoset{L}}$.
                Let $x=\apply{\birkhoffJoinMap{L}}{D}$.
                $x\in\carrier[L]$ by
                    \auto{birkhoff_join_map_type,funs_type_apply}.
                $\apply{\birkhoffClosure{L}}{D}
                    =\apply{\birkhoffMap{L}}{x}$ by
                    \auto{birkhoff_closure_apply}.
                $\apply{\birkhoffClosure{L}}{D}=D$.
            \end{subproof}
            Follows by \obv{surj}.
        \end{subproof}
        Follows by \obv{birkhoff_map_surjective_iff_distributive}.
    \end{subproof}
    Show that if $L$ is a distributive lattice, then for every
        $D\in\downsets{\joinIrreduciblePoset{L}}$ we have
        $\apply{\birkhoffClosure{L}}{D}=D$.
    \begin{subproof}
        Assume $L$ is a distributive lattice.
        Follows by \obv{birkhoff_closure_identity_of_distributive}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{theorem}\label{birkhoff_unit_identity_iff_distributive}
    Let $L$ be a finite bounded lattice.
    Then
        (for every $D\in\downsets{\joinIrreduciblePoset{L}}$ we have
            $\apply{\birkhoffMap{L}}
                {\apply{\birkhoffJoinMap{L}}{D}}=D$) iff
        $L$ is a distributive lattice.
\end{theorem}
\begin{proof}
    For every $D\in\downsets{\joinIrreduciblePoset{L}}$ we have
        $\apply{\birkhoffClosure{L}}{D}
            =\apply{\birkhoffMap{L}}
                {\apply{\birkhoffJoinMap{L}}{D}}$ by
        \obv{birkhoff_closure_apply}.
    Follows by \auto{birkhoff_closure_identity_iff_distributive}.
\end{proof}
