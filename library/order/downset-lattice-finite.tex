\section{Finite downset lattices}

\import{order/downset-lattice.tex}
\import{set/finite.tex}
\import{structure/finite.tex}

\begin{proposition}\label{finite_downsets}
    Let $P$ be a finite quasiordered set.
    Then $\downsets{P}$ is a finite set.
\end{proposition}
\begin{proof}
    $\carrier[P]$ is a finite set by \obv{finite_structure}.
    Hence $\pow{\carrier[P]}$ is a finite set by \obv{finite_powerset}.
    $\downsets{P}\subseteq\pow{\carrier[P]}$ by
        \obv{downsets_subseteq_powerset}.
    Follows by \obv{subseteq_finite_is_finite}.
\end{proof}

\begin{theorem}\label{finite_downset_lattice}
    Let $P$ be a finite quasiordered set.
    Then $\downsetLattice{P}$ is a finite bounded distributive lattice.
\end{theorem}
\begin{proof}
    $\downsetLattice{P}$ is a bounded distributive lattice by
        \obv{downset_lattice_is_bounded_distributive_lattice}.
    Hence $\downsetLattice{P}$ is a bounded lattice by
        \obv{bounded_distributive_lattice}.
    Hence $\downsetLattice{P}$ is a lattice by \obv{bounded_lattice}.
    Hence $\downsetLattice{P}$ is an ordered set by
        \obv{lattice_is_ordered_set}.
    Hence $\downsetLattice{P}$ is a quasiordered set by
        \obv{orderedset}.
    Hence $\downsetLattice{P}$ is a onesorted structure by
        \obv{quasiordered_set}.
    $\downsets{P}$ is a finite set by \obv{finite_downsets}.
    $\carrier[\downsetLattice{P}]=\downsets{P}$ by
        \obv{downset_lattice_carrier}.
    Hence $\downsetLattice{P}$ is a finite onesorted structure by
        \obv{finite_structure}.
\end{proof}
