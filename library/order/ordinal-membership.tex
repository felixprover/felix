\section{Membership below an ordinal}

\begin{proposition}\label{memrel_is_binary_relation_on}
    $\memrel{A}$ is a binary relation on $A$.
\end{proposition}
\begin{proof}
    Show that for every $w\in\memrel{A}$ we have $w\in A\times A$.
    \begin{subproof}
        Fix $w\in\memrel{A}$.
        Take $a,b\in A$ such that $w=(a,b)$
            by \auto{memrel_elem_inspect}.
        Follows by \auto{times_tuple_intro}.
    \end{subproof}
    Follows by \obv{subseteq}.
\end{proof}

\begin{lemma}\label{ordinal_memrel_downward}
    Let $\alpha$ be an ordinal.
    Let $\beta\in\alpha$.
    Then $\downward{\memrel{\alpha}}{\beta}=\beta$.
\end{lemma}
\begin{proof}
    Show that $\downward{\memrel{\alpha}}{\beta}\subseteq\beta$.
    \begin{subproof}
        Show that for every $x\in\downward{\memrel{\alpha}}{\beta}$
            we have $x\in\beta$.
        \begin{subproof}
            Fix $x\in\downward{\memrel{\alpha}}{\beta}$.
            $(x,\beta)\in\memrel{\alpha}$
                by \obv{downward_closure_iff}.
            Follows by \auto{memrel_elem_inspect,pair_eq_iff}.
        \end{subproof}
        Follows by \obv{subseteq}.
    \end{subproof}
    Show that $\beta\subseteq\downward{\memrel{\alpha}}{\beta}$.
    \begin{subproof}
        Show that for every $x\in\beta$ we have
            $x\in\downward{\memrel{\alpha}}{\beta}$.
        \begin{subproof}
            Fix $x\in\beta$.
            $x\in\alpha$ by \obv{ordinal_transitivity}.
            Thus $(x,\beta)\in\memrel{\alpha}$
                by \obv{memrel_elem_intro}.
            Follows by \obv{downward_closure_iff}.
        \end{subproof}
        Follows by \obv{subseteq}.
    \end{subproof}
    Follows by \obv{subseteq_antisymmetric}.
\end{proof}
