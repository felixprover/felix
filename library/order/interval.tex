\section{Intervals in ordered sets}

\import{order/order.tex}

The strict relation of an ordered set is obtained by removing the diagonal
from its non-strict order.  All intervals below are total set-valued
constructions, while their familiar order laws require the indicated
orderedness hypotheses.

\begin{definition}\label{ordered_set_strict_relation}
    $\orderStrictRelation{L}=\tostrictorder{\leq[L]}$.
\end{definition}

\begin{proposition}\label{ordered_set_strict_relation_iff}
    $x\mathrel{\orderStrictRelation{L}}y$ iff
        $x\mathrel{\leq[L]}y$ and $x\neq y$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{ordered_set_strict_relation,tostrictorder_iff}.
\end{proof}

\begin{proposition}\label{linearly_ordered_set_strict_trichotomy}
    Let $L$ be a linearly ordered set.
    Let $x,y\in\carrier[L]$.
    Then $x\mathrel{\orderStrictRelation{L}}y$ or $x=y$ or
        $y\mathrel{\orderStrictRelation{L}}x$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{linearly_ordered_set_total,
            ordered_set_strict_relation_iff}.
\end{proof}

\begin{proposition}\label{ordered_set_strict_le_transitive}
    Let $L$ be an ordered set.
    Let $a,b,c\in\carrier[L]$.
    Suppose $a\mathrel{\orderStrictRelation{L}}b$ and
        $b\mathrel{\leq[L]}c$.
    Then $a\mathrel{\orderStrictRelation{L}}c$.
\end{proposition}
\begin{proof}
    $a\mathrel{\leq[L]}b$ and $a\neq b$ by
        \auto{ordered_set_strict_relation_iff}.
    Hence $a\mathrel{\leq[L]}c$ by
        \auto{ordered_set_order_transitive,transitive}.
    Show that $a\neq c$.
    \begin{subproof}
        Assume $a=c$.
        Hence $b\mathrel{\leq[L]}a$.
        Thus $a=b$ by
            \auto{ordered_set_order_antisymmetric,antisymmetric}.
        Contradiction.
    \end{subproof}
    Follows by \auto{ordered_set_strict_relation_iff}.
\end{proof}

\begin{proposition}\label{ordered_set_le_strict_transitive}
    Let $L$ be an ordered set.
    Let $a,b,c\in\carrier[L]$.

\end{proposition}

\begin{definition}\label{order_closed_interval}
    $\orderClosedInterval{L}{a}{b}
        =\{x\in\carrier[L]\mid
            a\mathrel{\leq[L]}x\land x\mathrel{\leq[L]}b\}$.
\end{definition}

\begin{definition}\label{order_open_interval}
    $\orderOpenInterval{L}{a}{b}
        =\{x\in\carrier[L]\mid
            a\mathrel{\orderStrictRelation{L}}x\land
            x\mathrel{\orderStrictRelation{L}}b\}$.
\end{definition}

\begin{definition}\label{order_closed_lower_ray}
    $\orderClosedLowerRay{L}{b}
        =\{x\in\carrier[L]\mid x\mathrel{\leq[L]}b\}$.
\end{definition}

\begin{definition}\label{order_closed_upper_ray}
    $\orderClosedUpperRay{L}{a}
        =\{x\in\carrier[L]\mid a\mathrel{\leq[L]}x\}$.
\end{definition}

\begin{definition}\label{order_open_lower_ray}
    $\orderOpenLowerRay{L}{b}
        =\{x\in\carrier[L]\mid
            x\mathrel{\orderStrictRelation{L}}b\}$.
\end{definition}

\begin{definition}\label{order_open_upper_ray}
    $\orderOpenUpperRay{L}{a}
        =\{x\in\carrier[L]\mid
            a\mathrel{\orderStrictRelation{L}}x\}$.
\end{definition}

\begin{proposition}\label{order_closed_interval_iff}
    $x\in\orderClosedInterval{L}{a}{b}$ iff
        $x\in\carrier[L]$ and $a\mathrel{\leq[L]}x$ and
        $x\mathrel{\leq[L]}b$.
\end{proposition}
\begin{proof}
    Follows by \auto{order_closed_interval}.
\end{proof}

\begin{proposition}\label{order_open_interval_iff}
    $x\in\orderOpenInterval{L}{a}{b}$ iff
        $x\in\carrier[L]$ and
        $a\mathrel{\orderStrictRelation{L}}x$ and
        $x\mathrel{\orderStrictRelation{L}}b$.
\end{proposition}
\begin{proof}
    Follows by \auto{order_open_interval}.
\end{proof}

\begin{proposition}\label{order_closed_lower_ray_iff}
    $x\in\orderClosedLowerRay{L}{b}$ iff
        $x\in\carrier[L]$ and $x\mathrel{\leq[L]}b$.
\end{proposition}
\begin{proof}
    Follows by \obv{order_closed_lower_ray}.
\end{proof}

\begin{proposition}\label{order_closed_upper_ray_iff}
    $x\in\orderClosedUpperRay{L}{a}$ iff
        $x\in\carrier[L]$ and $a\mathrel{\leq[L]}x$.
\end{proposition}
\begin{proof}
    Follows by \obv{order_closed_upper_ray}.
\end{proof}

\begin{proposition}\label{order_open_lower_ray_iff}
    $x\in\orderOpenLowerRay{L}{b}$ iff
        $x\in\carrier[L]$ and
        $x\mathrel{\orderStrictRelation{L}}b$.
\end{proposition}
\begin{proof}
    Follows by \obv{order_open_lower_ray}.
\end{proof}

\begin{proposition}\label{order_open_upper_ray_iff}
    $x\in\orderOpenUpperRay{L}{a}$ iff
        $x\in\carrier[L]$ and
        $a\mathrel{\orderStrictRelation{L}}x$.
\end{proposition}
\begin{proof}
    Follows by \obv{order_open_upper_ray}.
\end{proof}

\begin{proposition}\label{order_interval_and_ray_types}
    $\orderClosedInterval{L}{a}{b},\orderOpenInterval{L}{a}{b},
        \orderClosedLowerRay{L}{b},\orderClosedUpperRay{L}{a},
        \orderOpenLowerRay{L}{b},\orderOpenUpperRay{L}{a}
        \subseteq\carrier[L]$.
\end{proposition}
\begin{proof}
    Follows by
        \obv{order_closed_interval_iff,order_open_interval_iff,
            order_closed_lower_ray_iff,order_closed_upper_ray_iff,
            order_open_lower_ray_iff,order_open_upper_ray_iff,subseteq}.
\end{proof}

\begin{proposition}\label{order_closed_interval_eq_intersection_rays}
    $\orderClosedInterval{L}{a}{b}
        =\orderClosedUpperRay{L}{a}\inter
            \orderClosedLowerRay{L}{b}$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{setext,inter,order_closed_interval_iff,
            order_closed_upper_ray_iff,order_closed_lower_ray_iff}.
\end{proof}

\begin{proposition}\label{order_open_interval_eq_intersection_rays}
    $\orderOpenInterval{L}{a}{b}
        =\orderOpenUpperRay{L}{a}\inter\orderOpenLowerRay{L}{b}$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{setext,inter,order_open_interval_iff,
            order_open_upper_ray_iff,order_open_lower_ray_iff}.
\end{proof}

\begin{proposition}\label{order_complement_closed_lower_ray}
    Let $L$ be a linearly ordered set and $b\in\carrier[L]$.
    Then $\carrier[L]\setminus\orderClosedLowerRay{L}{b}
        =\orderOpenUpperRay{L}{b}$.
\end{proposition}
\begin{proof}
    Show that every element of
        $\carrier[L]\setminus\orderClosedLowerRay{L}{b}$ is an element
        of $\orderOpenUpperRay{L}{b}$.
    \begin{subproof}
        Fix $x\in\carrier[L]\setminus\orderClosedLowerRay{L}{b}$.
        Hence $x\in\carrier[L]$ and
            $x\notin\orderClosedLowerRay{L}{b}$ by \obv{setminus}.
        Hence not $x\mathrel{\leq[L]}b$ by
            \auto{order_closed_lower_ray_iff}.
        $b\mathrel{\leq[L]}x$ by
            \obv{linearly_ordered_set_total}.
        $b\neq x$ by
            \obv{ordered_set_order_reflexive,reflexive_on,
                linearly_ordered_set_is_ordered_set}.
        Hence $b\mathrel{\orderStrictRelation{L}}x$ by
            \auto{ordered_set_strict_relation_iff}.
        Follows by \obv{order_open_upper_ray}.
    \end{subproof}
    Show that every element of $\orderOpenUpperRay{L}{b}$ is an element
        of $\carrier[L]\setminus\orderClosedLowerRay{L}{b}$.
    \begin{subproof}
        Fix $x\in\orderOpenUpperRay{L}{b}$.
        Hence $x\in\carrier[L]$ and
            $b\mathrel{\orderStrictRelation{L}}x$ by
            \obv{order_open_upper_ray}.
        Hence $b\mathrel{\leq[L]}x$ and $b\neq x$ by
            \auto{ordered_set_strict_relation_iff}.
        Show that $x\notin\orderClosedLowerRay{L}{b}$.
        \begin{subproof}
            Assume $x\in\orderClosedLowerRay{L}{b}$.
            Hence $x\mathrel{\leq[L]}b$ by
                \obv{order_closed_lower_ray}.
            Hence $b=x$ by
                \auto{ordered_set_order_antisymmetric,antisymmetric,
                    linearly_ordered_set_is_ordered_set}.
            Contradiction.
        \end{subproof}
        Follows by \obv{setminus}.
    \end{subproof}
    Follows by set extensionality.
\end{proof}

\begin{proposition}\label{order_complement_open_lower_ray}
    Let $L$ be a linearly ordered set and $b\in\carrier[L]$.
    Then $\carrier[L]\setminus\orderOpenLowerRay{L}{b}
        =\orderClosedUpperRay{L}{b}$.
\end{proposition}
\begin{proof}
    $\orderOpenLowerRay{L}{b}
        =\carrier[L]\setminus\orderClosedUpperRay{L}{b}$ by
        \obv{order_complement_closed_upper_ray}.
    Hence $\carrier[L]\setminus\orderOpenLowerRay{L}{b}
        =\carrier[L]\setminus
            (\carrier[L]\setminus\orderClosedUpperRay{L}{b})$.
    $\orderClosedUpperRay{L}{b}\subseteq\carrier[L]$ by
        \obv{order_closed_upper_ray,subseteq}.
    Follows by \obv{double_relative_complement}.
\end{proof}

\begin{proposition}\label{order_complement_open_upper_ray}
    Let $L$ be a linearly ordered set and $a\in\carrier[L]$.
    Then $\carrier[L]\setminus\orderOpenUpperRay{L}{a}
        =\orderClosedLowerRay{L}{a}$.
\end{proposition}
\begin{proof}
    $\orderOpenUpperRay{L}{a}
        =\carrier[L]\setminus\orderClosedLowerRay{L}{a}$ by
        \obv{order_complement_closed_lower_ray}.
    Hence $\carrier[L]\setminus\orderOpenUpperRay{L}{a}
        =\carrier[L]\setminus
            (\carrier[L]\setminus\orderClosedLowerRay{L}{a})$.
    $\orderClosedLowerRay{L}{a}\subseteq\carrier[L]$ by
        \obv{order_closed_lower_ray,subseteq}.
    Follows by \obv{double_relative_complement}.
\end{proof}
