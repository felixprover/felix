\section{Order morphisms}

\import{function.tex}
\import{relation/trace.tex}
\import{order/bounds.tex}

\begin{definition}\label{order_preserving}
    $\orderPreserving{f}{A}{R}{B}{S}$ iff
        $R$ is a binary relation on $A$ and
        $S$ is a binary relation on $B$ and
        $f\in\funs{A}{B}$ and
        for all $x,y\in A$ such that $x\mathrel{R}y$ we have
            $f(x)\mathrel{S}f(y)$.
\end{definition}

\begin{definition}\label{order_reflecting}
    $\orderReflecting{f}{A}{R}{B}{S}$ iff
        $R$ is a binary relation on $A$ and
        $S$ is a binary relation on $B$ and
        $f\in\funs{A}{B}$ and
        for all $x,y\in A$ such that $f(x)\mathrel{S}f(y)$ we have
            $x\mathrel{R}y$.
\end{definition}

\begin{definition}\label{order_embedding}
    $\orderEmbedding{f}{A}{R}{B}{S}$ iff
        $R$ is a binary relation on $A$ and
        $S$ is a binary relation on $B$ and
        $f\in\inj{A}{B}$ and
        for all $x,y\in A$ we have
            $x\mathrel{R}y$ iff $f(x)\mathrel{S}f(y)$.
\end{definition}

\begin{definition}\label{order_isomorphism}
    $\orderIsomorphism{f}{A}{R}{B}{S}$ iff
        $R$ is a binary relation on $A$ and
        $S$ is a binary relation on $B$ and
        $f\in\bijections{A}{B}$ and
        for all $x,y\in A$ we have
            $x\mathrel{R}y$ iff $f(x)\mathrel{S}f(y)$.
\end{definition}

\begin{definition}\label{order_isomorphic}
    $\orderIsomorphic{A}{R}{B}{S}$ iff there exists $f$ such that
        $\orderIsomorphism{f}{A}{R}{B}{S}$.
\end{definition}

\begin{proposition}\label{order_preserving_type}
    Suppose $\orderPreserving{f}{A}{R}{B}{S}$.
    Then $f\in\funs{A}{B}$.
\end{proposition}
\begin{proof}
    Follows by \obv{order_preserving}.
\end{proof}

\begin{proposition}\label{order_reflecting_type}
    Suppose $\orderReflecting{f}{A}{R}{B}{S}$.
    Then $f\in\funs{A}{B}$.
\end{proposition}
\begin{proof}
    Follows by \obv{order_reflecting}.
\end{proof}

\begin{proposition}\label{order_embedding_type}
    Suppose $\orderEmbedding{f}{A}{R}{B}{S}$.
    Then $f\in\funs{A}{B}$.
\end{proposition}
\begin{proof}
    Follows by \auto{order_embedding,inj}.
\end{proof}

\begin{proposition}\label{order_embedding_injective}
    Suppose $\orderEmbedding{f}{A}{R}{B}{S}$.
    Then $f\in\inj{A}{B}$.
\end{proposition}
\begin{proof}
    Follows by \obv{order_embedding}.
\end{proof}

\begin{proposition}\label{order_embedding_order_iff}
    Suppose $\orderEmbedding{f}{A}{R}{B}{S}$.
    Let $x,y\in A$.
    Then $x\mathrel{R}y$ iff $f(x)\mathrel{S}f(y)$.
\end{proposition}
\begin{proof}
    Follows by \obv{order_embedding}.
\end{proof}

\begin{proposition}\label{order_embedding_is_order_preserving}
    Suppose $\orderEmbedding{f}{A}{R}{B}{S}$.
    Then $\orderPreserving{f}{A}{R}{B}{S}$.
\end{proposition}
\begin{proof}
    $f\in\funs{A}{B}$ by \obv{order_embedding_type}.
    Follows by \auto{order_embedding,order_preserving}.
\end{proof}

\begin{proposition}\label{order_embedding_is_order_reflecting}

    Then $\orderReflecting{f}{A}{R}{B}{S}$.
\end{proposition}
\begin{proof}

    Follows by \auto{order_embedding,order_reflecting}.
\end{proof}

\begin{proposition}\label{order_isomorphism_is_embedding}
    Suppose $\orderIsomorphism{f}{A}{R}{B}{S}$.
    Then $\orderEmbedding{f}{A}{R}{B}{S}$.
\end{proposition}
\begin{proof}
    $f\in\bijections{A}{B}$ by \obv{order_isomorphism}.
    Hence $f\in\funs{A}{B}$ by \obv{bijections_to_funs}.
    $f$ is a function and $f$ is injective by
        \obv{bijection_is_function,bijections}.
    $\dom{f}=A$ by \obv{bijections_dom}.
    Hence $f\in\inj{A}{B}$ by \auto{inj,injective_function}.
    Follows by \obv{order_isomorphism,order_embedding}.
\end{proof}

\begin{proposition}\label{order_isomorphism_type}
    Suppose $\orderIsomorphism{f}{A}{R}{B}{S}$.
    Then $f\in\bijections{A}{B}$.
\end{proposition}
\begin{proof}
    Follows by \obv{order_isomorphism}.
\end{proof}

\begin{proposition}\label{order_isomorphism_restriction_to_image}
    Suppose $\orderIsomorphism{f}{A}{R}{B}{S}$ and $C\subseteq A$.
    Then
        $\orderIsomorphism{\restrl{f}{C}}
            {C}{\relationTrace{R}{C}}
            {\img{f}{C}}{\relationTrace{S}{\img{f}{C}}}$.
\end{proposition}
\begin{proof}
    $f\in\bijections{A}{B}$ by \obv{order_isomorphism_type}.
    Hence $\restrl{f}{C}\in\bijections{C}{\img{f}{C}}$ by
        \obv{bijection_restriction_to_image}.
    $\relationTrace{R}{C}$ is a binary relation on $C$ and
        $\relationTrace{S}{\img{f}{C}}$ is a binary relation on
            $\img{f}{C}$ by
        \obv{relation_trace_binary_on}.
    $f$ is a function and $\dom{f}=A$ by
        \obv{bijection_is_function,bijections_dom}.
    Hence $C\subseteq\dom{f}$.
    For every $x\in C$ we have
        $(\restrl{f}{C})(x)=f(x)$ by
        \obv{restrl_of_function_apply}.
    For every $x\in C$ we have $x\in\dom{f}\inter C$ by
        \obv{inter,elem_subseteq}.
    Hence for every $x\in C$ we have $f(x)\in\img{f}{C}$ by
        \obv{img_of_function_intro}.
    For every $x\in C$ we have $x\in A$ by \obv{elem_subseteq}.
    For all $x,y\in C$ we have
        $x\mathrel{\relationTrace{R}{C}}y$ iff
        $x\mathrel{R}y$ by \auto{relation_trace_iff}.
    For all $x,y\in C$ we have
        $x\mathrel{R}y$ iff $f(x)\mathrel{S}f(y)$ by
        \obv{order_isomorphism}.
    For all $x,y\in C$ we have
        $f(x)\mathrel{S}f(y)$ iff
        $f(x)\mathrel{\relationTrace{S}{\img{f}{C}}}f(y)$ by
        \auto{relation_trace_iff}.
    For all $x,y\in C$ we have
        $x\mathrel{\relationTrace{R}{C}}y$ iff
        $(\restrl{f}{C})(x)
            \mathrel{\relationTrace{S}{\img{f}{C}}}
            (\restrl{f}{C})(y)$ by
        \auto.
    Follows by \obv{order_isomorphism}.
\end{proof}

\begin{proposition}\label{identity_order_embedding}
    Suppose $R$ is a binary relation on $A$.
    Then $\orderEmbedding{\identity{A}}{A}{R}{A}{R}$.
\end{proposition}
\begin{proof}
    $\identity{A}\in\funs{A}{A}$ by \obv{id_elem_funs}.
    $\identity{A}$ is a function and $\dom{\identity{A}}=A$ by
        \obv{funs_is_function,funs}.
    $\identity{A}$ is injective by \obv{identity_injective}.
    Hence $\identity{A}\in\inj{A}{A}$ by
        \auto{inj,injective_function}.
    Follows by \obv{order_embedding,id_apply}.
\end{proof}

\begin{proposition}\label{identity_order_isomorphism}
    Suppose $R$ is a binary relation on $A$.
    Then $\orderIsomorphism{\identity{A}}{A}{R}{A}{R}$.
\end{proposition}
\begin{proof}
    $\identity{A}\in\bijections{A}{A}$ by
        \obv{id_elem_bijections}.
    Follows by \obv{order_isomorphism,id_apply}.
\end{proof}

\begin{proposition}\label{order_embedding_of_injective_preserving_reflecting}
    Suppose $f\in\inj{A}{B}$.
    Suppose $\orderPreserving{f}{A}{R}{B}{S}$ and
        $\orderReflecting{f}{A}{R}{B}{S}$.
    Then $\orderEmbedding{f}{A}{R}{B}{S}$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{order_embedding,order_preserving,order_reflecting}.
\end{proof}

\begin{proposition}\label{order_embedding_composition}
    Suppose $\orderEmbedding{f}{A}{R}{B}{S}$ and
        $\orderEmbedding{g}{B}{S}{C}{T}$.
    Then $\orderEmbedding{g\circ f}{A}{R}{C}{T}$.
\end{proposition}
\begin{proof}
    $f\in\inj{A}{B}$ and $g\in\inj{B}{C}$ by
        \obv{order_embedding_injective}.
    Hence $g\circ f\in\inj{A}{C}$ by \obv{inj_circ}.
    $f\in\funs{A}{B}$ and $g\in\funs{B}{C}$ by
        \obv{order_embedding_type}.
    Hence $f$ and $g$ are functions and $\dom{f}=A$ and $\dom{g}=B$
        by \obv{funs_is_function,funs}.
    $g$ is composable with $f$ by \obv{funs_ran}.
    Show that for all $x\in A$ we have $(g\circ f)(x)=g(f(x))$.
    \begin{subproof}
        Fix $x\in A$.
        $f(x)\in B$ by \obv{funs_type_apply}.
        Hence $f(x)\in\dom{g}$.
        Follows by \obv{circ_apply}.
    \end{subproof}
    For all $x,y\in A$ we have $f(x),f(y)\in B$ by
        \obv{funs_type_apply}.
    For all $x,y\in A$ we have
        $x\mathrel{R}y$ iff $f(x)\mathrel{S}f(y)$ by
        \obv{order_embedding}.
    For all $x,y\in A$ we have
        $f(x)\mathrel{S}f(y)$ iff $g(f(x))\mathrel{T}g(f(y))$ by
        \auto{order_embedding}.
    Follows by \auto{order_embedding}.
\end{proof}

\begin{proposition}\label{order_isomorphism_inverse}
    Suppose $\orderIsomorphism{f}{A}{R}{B}{S}$.
    Then $\orderIsomorphism{\converse{f}}{B}{S}{A}{R}$.
\end{proposition}
\begin{proof}
    $f\in\bijections{A}{B}$ by \obv{order_isomorphism}.
    Hence $f$ is a function and $\dom{f}=A$ and $\ran{f}=B$
        by
        \auto{bijection_is_function,bijections_dom,bijections,ran_of_surj}.
    $\converse{f}\in\bijections{B}{A}$
        by \obv{bijection_converse_is_bijection}.
    Hence $\converse{f}$ is a function
        by \obv{bijection_is_function}.
    Show that for all $u,v\in B$ we have
        $u\mathrel{S}v$ iff
        $(\converse{f})(u)\mathrel{R}(\converse{f})(v)$.
    \begin{subproof}
        Fix $u,v\in B$.
        Take $x\in A$ such that $f(x)=u$
            by \auto{fun_ran_iff}.
        Take $y\in A$ such that $f(y)=v$
            by \auto{fun_ran_iff}.
        $(x,u),(y,v)\in f$ by \auto{function_apply_elim}.
        Hence $(u,x),(v,y)\in\converse{f}$ by \obv{converse_iff}.
        Hence $(\converse{f})(u)=x$ and
            $(\converse{f})(v)=y$ by \obv{function_apply_intro}.
        $x\mathrel{R}y$ iff $f(x)\mathrel{S}f(y)$
            by \obv{order_isomorphism}.
    \end{subproof}
    Follows by \obv{order_isomorphism}.
\end{proof}

\begin{proposition}\label{order_isomorphism_composition}
    Suppose $\orderIsomorphism{f}{A}{R}{B}{S}$ and
        $\orderIsomorphism{g}{B}{S}{C}{T}$.
    Then $\orderIsomorphism{g\circ f}{A}{R}{C}{T}$.
\end{proposition}

        by \obv{order_isomorphism}.

        by \obv{bijections_to_funs}.
    Hence $f$ and $g$ are functions and $\dom{f}=A$ and $\dom{g}=B$
        by \obv{funs_is_function,funs}.
    Hence $g\circ f\in\bijections{A}{C}$
        by \obv{bijection_circ}.
    For all $x,y\in A$ we have $f(x),f(y)\in B$
        by \obv{bijections_to_funs,funs_type_apply}.
    $g$ is composable with $f$ by \obv{funs_ran}.
    Show that for every $x\in A$ we have $(g\circ f)(x)=g(f(x))$.
    \begin{subproof}
        Fix $x\in A$.
        $x\in\dom{f}$.
        $f(x)\in B$ by \obv{funs_type_apply}.
        Hence $f(x)\in\dom{g}$.
        Follows by \obv{circ_apply}.
    \end{subproof}
    For all $x,y\in A$ we have
        $x\mathrel{R}y$ iff $f(x)\mathrel{S}f(y)$
        by \obv{order_isomorphism}.
    For all $x,y\in A$ we have
        $f(x)\mathrel{S}f(y)$ iff $g(f(x))\mathrel{T}g(f(y))$
        by \auto{order_isomorphism}.
    Follows by \auto{order_isomorphism}.
\end{proof}

\begin{proposition}\label{order_isomorphism_preserves_least}
    Suppose $\orderIsomorphism{f}{A}{R}{B}{S}$ and
        $a$ is a least element of $A$ with respect to $R$.
    Then $f(a)$ is a least element of $B$ with respect to $S$.
\end{proposition}
\begin{proof}
    $f\in\bijections{A}{B}$ by \obv{order_isomorphism_type}.
    Hence $f\in\funs{A}{B}$ and $\ran{f}=B$ by
        \auto{bijections_to_funs,bijections,ran_of_surj}.
    Hence $f$ is a function and $\dom{f}=A$ by
        \obv{funs_is_function,funs}.
    $a\in A$ by \obv{least_element}.
    Hence $f(a)\in B$ by \obv{funs_type_apply}.
    Show for all $y\in B$ we have $f(a)=y$ or $f(a)\mathrel{S}y$.
    \begin{subproof}
        Fix $y\in B$.
        Hence $y\in\ran{f}$.
        Take $x\in A$ such that $f(x)=y$ by \auto{fun_ran_iff}.
        We have $a=x$ or $a\mathrel{R}x$ by \obv{least_element}.
        Follows by \auto{order_isomorphism}.
    \end{subproof}
    Follows by \obv{least_element}.
\end{proof}
