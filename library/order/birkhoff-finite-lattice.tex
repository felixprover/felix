\section{Birkhoff representation for finite distributive lattices}

\import{order/birkhoff-finite.tex}
\import{order/finite-lattice.tex}

The bounded expansion below equips an inhabited finite lattice with its
unique least and greatest elements.  Join-irreducibility in the representation
is therefore computed relative to that unique least element, not to a bound
field already present in the original lattice record.

\begin{theorem}\label{finite_distributive_lattice_has_bounded_expansion}
    Let $L$ be a finite distributive lattice.
    Suppose $\carrier[L]$ is inhabited.
    Then there exists $M$ such that
        $M=\boundedExpansion{L}$ and
        $M$ is a bounded distributive lattice and
        $M$ is a finite bounded lattice.
\end{theorem}
\begin{proof}
    $L$ is a finite lattice by \obv{distributive_lattice}.
    Let $M=\boundedExpansion{L}$.
    $M$ is a finite bounded lattice by
        \obv{bounded_expansion_finite}.
    $M$ is a bounded distributive lattice by
        \obv{bounded_expansion_preserves_distributivity}.
\end{proof}

The following witnessed form is the representation theorem.  The preceding
theorem supplies its canonical witness; keeping the witness explicit makes
the hosted Birkhoff terms well formed in Felix's current source language.

\begin{theorem}\label{finite_distributive_lattice_birkhoff_representation}
    Let $L$ be a finite distributive lattice.
    Let $M$ be a finite bounded lattice.
    Suppose $\carrier[L]$ is inhabited and
        $M=\boundedExpansion{L}$ and
        $M$ is a bounded distributive lattice.
    Then $\reduct[lattice]{M}=\reduct[lattice]{L}$ and
        $\boundedLatticeIsomorphism
            {\birkhoffMap{M}}{M}
            {\downsetLattice{\joinIrreduciblePoset{M}}}$ and
        $\birkhoffJoinMap{M}$ is a left inverse of $\birkhoffMap{M}$ and
        $\birkhoffJoinMap{M}$ is a right inverse of $\birkhoffMap{M}$.
\end{theorem}
\begin{proof}
    $L$ is a finite lattice by \obv{distributive_lattice}.
    $\reduct[lattice]{\boundedExpansion{L}}
        =\reduct[lattice]{L}$ by
        \obv{bounded_expansion_lattice_reduct}.
    Hence $\reduct[lattice]{M}=\reduct[lattice]{L}$.
    $\boundedLatticeIsomorphism
        {\birkhoffMap{M}}{M}
        {\downsetLattice{\joinIrreduciblePoset{M}}}$ and
        $\birkhoffJoinMap{M}$ is a left inverse of $\birkhoffMap{M}$ and
        $\birkhoffJoinMap{M}$ is a right inverse of $\birkhoffMap{M}$ by
        \obv{finite_birkhoff_bounded_lattice_isomorphism}.
\end{proof}

\begin{theorem}
    \label{finite_distributive_lattice_isomorphic_to_set_family_lattice}
    Let $L$ be a finite distributive lattice.
    Let $M$ be a finite bounded lattice.
    Suppose $\carrier[L]$ is inhabited and
        $M=\boundedExpansion{L}$ and
        $M$ is a bounded distributive lattice.
    Then there exists a finite set $X$ such that
        there exists a bounded set-lattice family $C$ on $X$ such that
            $C$ is a finite set and
            $\reduct[lattice]{M}=\reduct[lattice]{L}$ and
            $X=\carrier[\joinIrreduciblePoset{M}]$ and
            $C=\downsets{\joinIrreduciblePoset{M}}$ and
            $\latticeIsomorphism
                {\birkhoffMap{M}}{L}{\setFamilyLattice{X}{C}}$.
\end{theorem}
\begin{proof}
    $L$ is a finite lattice by \obv{distributive_lattice}.
    $\reduct[lattice]{\boundedExpansion{L}}
        =\reduct[lattice]{L}$ by
        \obv{bounded_expansion_lattice_reduct}.
    Hence $\reduct[lattice]{M}=\reduct[lattice]{L}$.

    Let $P=\joinIrreduciblePoset{M}$.
    $P$ is a finite ordered set by
        \obv{join_irreducible_poset_is_finite}.
    Hence $P$ is a finite quasiordered set by \obv{orderedset}.
    Let $X=\carrier[P]$.
    $\carrier[P]=\joinIrreducibles{M}$ by
        \auto{join_irreducible_poset_carrier}.
    $\joinIrreducibles{M}$ is a finite set by
        \obv{finite_join_irreducibles}.
    Hence $X$ is a finite set.
    Let $C=\downsets{P}$.
    $C$ is a bounded set-lattice family on $X$ by
        \auto{downsets_are_bounded_set_lattice_family}.
    $C$ is a finite set by \obv{finite_downsets}.
    $\setFamilyLattice{X}{C}=\downsetLattice{P}$ by
        \obv{downset_lattice}.
    Hence $\setFamilyLattice{X}{C}$ is a bounded distributive lattice by
        \obv{downset_lattice_is_bounded_distributive_lattice}.
    Hence $\setFamilyLattice{X}{C}$ is a bounded lattice by
        \obv{bounded_distributive_lattice_is_bounded_lattice}.
    Hence $\setFamilyLattice{X}{C}$ is a lattice by
        \obv{bounded_lattice_is_lattice}.

    $\orderIsomorphism
        {\birkhoffMap{M}}
        {\carrier[M]}{\leq[M]}
        {\carrier[\downsetLattice{P}]}{\leq[\downsetLattice{P}]}$ by
        \auto{finite_birkhoff_order_isomorphism}.
    $\carrier[M]=\carrier[L]$ and $\leq[M]=\leq[L]$ by
        \auto{bounded_expansion_carrier,bounded_expansion_order}.
    $\carrier[\setFamilyLattice{X}{C}]
        =\carrier[\downsetLattice{P}]$ and
        $\leq[\setFamilyLattice{X}{C}]
        =\leq[\downsetLattice{P}]$.
    Hence $\orderIsomorphism
        {\birkhoffMap{M}}
        {\carrier[L]}{\leq[L]}
        {\carrier[\setFamilyLattice{X}{C}]}
        {\leq[\setFamilyLattice{X}{C}]}$.
    Hence $\latticeIsomorphism
        {\birkhoffMap{M}}{L}{\setFamilyLattice{X}{C}}$ by
        \obv{order_isomorphism_between_lattices_is_lattice_isomorphism}.
\end{proof}
