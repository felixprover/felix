\section{Linear orders, well-orders, and initial segments}

\import{relation/trace.tex}
\import{order/order.tex}
\import{order/bounds.tex}

\begin{definition}\label{strict_linear_order}
    $R$ is a strict linear order on $A$ iff
        $R$ is a binary relation on $A$ and
        $R$ is a strict order and $R$ is connex on $A$.
\end{definition}

\begin{definition}\label{linear_order}
    $R$ is a linear order on $A$ iff
        $R$ is an order on $A$ and $R$ is connex on $A$.
\end{definition}

\begin{abbreviation}\label{simple_order_relation}
    $R$ is a simple order relation on $A$ iff
        $R$ is a strict linear order on $A$.
\end{abbreviation}

\begin{definition}\label{strict_well_order}
    $R$ is a strict well order on $A$ iff
        $R$ is a strict linear order on $A$ and
        for all $X$ such that $X$ is inhabited and $X\subseteq A$ there exists $a\in X$ such that
            $a$ is a least element of $X$ with respect to $R$.
\end{definition}

\begin{definition}\label{well_order}
    $R$ is a well order on $A$ iff
        $R$ is a linear order on $A$ and
        for all $X$ such that $X$ is inhabited and $X\subseteq A$ there exists $a\in X$ such that
            $a$ is a least element of $X$ with respect to $R$.
\end{definition}

\begin{abbreviation}\label{well_order_relation}
    $R$ is a well order relation on $A$ iff
        $R$ is a strict well order on $A$.
\end{abbreviation}

\begin{definition}\label{initial_segment}
    $I$ is an initial segment of $A$ with respect to $R$ iff
        $I\subseteq A$ and for all $x\in I$ we have for all $y\in A$
            such that $y\mathrel{R}x$ we have $y\in I$.
\end{definition}

\begin{definition}\label{proper_initial_segment}
    $I$ is a proper initial segment of $A$ with respect to $R$ iff
        $I$ is an initial segment of $A$ with respect to $R$ and $I\neq A$.
\end{definition}

\begin{proposition}\label{strict_well_order_iff}
    $R$ is a strict well order on $A$ iff
        $R$ is a binary relation on $A$ and
        $R$ is transitive and $R$ is asymmetric and $R$ is connex on $A$ and
        for all $X$ such that $X\subseteq A$ and $X\neq\emptyset$
            there exists $a\in X$ such that for all $x\in X$ we have
                $a\mathrel{R}x$ or $a=x$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{strict_well_order,strict_linear_order,least_element,
            nonempty_inhabited,relation,subseteq,times_elem_is_tuple}.
\end{proof}

\begin{proposition}\label{relation_trace_strict_linear_order}
    Suppose $R$ is a strict linear order on $X$ and $A\subseteq X$.
    Then $\relationTrace{R}{A}$ is a strict linear order on $A$.
\end{proposition}
\begin{proof}
    $R$ is a strict order by \obv{strict_linear_order}.
    Hence $R$ is a relation and $R$ is transitive and
        $R$ is asymmetric by definition.
    $\relationTrace{R}{A}$ is a binary relation on $A$ by
        \obv{relation_trace_binary_on}.
    $\relationTrace{R}{A}$ is a relation by
        \auto{relation,relation_trace_subseteq,subseteq}.
    $\relationTrace{R}{A}$ is transitive by
        \obv{relation_trace_transitive}.
    $\relationTrace{R}{A}$ is asymmetric by
        \obv{relation_trace_asymmetric}.
    Hence $\relationTrace{R}{A}$ is a strict order by definition.
    $R$ is connex on $X$ by \obv{strict_linear_order}.
    Hence $\relationTrace{R}{A}$ is connex on $A$ by
        \obv{relation_trace_connex_on}.
    Follows by definition.
\end{proof}

\begin{proposition}\label{relation_trace_strict_well_order}
    Suppose $R$ is a strict well order on $X$ and $A\subseteq X$.
    Then $\relationTrace{R}{A}$ is a strict well order on $A$.
\end{proposition}
\begin{proof}
    $R$ is a strict linear order on $X$ by \obv{strict_well_order}.
    Hence $\relationTrace{R}{A}$ is a strict linear order on $A$ by
        \obv{relation_trace_strict_linear_order}.
    Show that for every $Y$ such that $Y$ is inhabited and $Y\subseteq A$
        there exists $a\in Y$ such that
            $a$ is a least element of $Y$ with respect to
                $\relationTrace{R}{A}$.
    \begin{subproof}
        Fix $Y$.
        Assume $Y$ is inhabited and $Y\subseteq A$.
        Then $Y\subseteq X$ by \obv{subseteq_transitive}.
        Take $a\in Y$ such that
            $a$ is a least element of $Y$ with respect to $R$ by
            \obv{strict_well_order}.
        Then $a$ is a least element of $Y$ with respect to
            $\relationTrace{R}{A}$ by
            \auto{least_element,relation_trace_iff,elem_subseteq}.
    \end{subproof}
    Follows by definition.
\end{proof}

\begin{proposition}\label{emptyset_initial_segment}
    $\emptyset$ is an initial segment of $A$ with respect to $R$.
\end{proposition}
\begin{proof}
    Follows by \auto{initial_segment,emptyset_subseteq,emptyset}.
\end{proof}

\begin{proposition}\label{carrier_initial_segment}
    $A$ is an initial segment of $A$ with respect to $R$.
\end{proposition}
\begin{proof}
    Follows by \obv{initial_segment,subseteq_refl}.
\end{proof}

\begin{proposition}\label{inter_initial_segments}
    Suppose $I$ and $J$ are initial segments of $A$ with respect to $R$.
    Then $I\inter J$ is an initial segment of $A$ with respect to $R$.
\end{proposition}
\begin{proof}
    Follows by \auto{initial_segment,inter,subseteq}.
\end{proof}

\begin{proposition}\label{union_initial_segments}
    Suppose every element of $F$ is an initial segment of $A$ with respect to $R$.
    Then $\unions{F}$ is an initial segment of $A$ with respect to $R$.
\end{proposition}
\begin{proof}
    Follows by \auto{initial_segment,unions_iff,subseteq,subseteq_transitive}.
\end{proof}

\begin{proposition}\label{principal_initial_segment}
    Suppose $R$ is a binary relation on $A$ and $R$ is transitive.
    Suppose $a\in A$.
    Then $\downward{R}{a}$ is an initial segment of $A$ with respect to $R$.
\end{proposition}
\begin{proof}
    Follows by \auto{initial_segment,downward_closure_iff,transitive,subseteq,times_tuple_elim}.
\end{proof}

\begin{proposition}\label{least_element_toorder_iff}
    $a$ is a least element of $X$ with respect to $R$ iff
        $a$ is a least element of $X$ with respect to $\toorder{A}{R}$.
\end{proposition}
\begin{proof}
    Follows by \auto{least_element,toorder_iff}.
\end{proof}

\begin{proposition}\label{least_element_tostrictorder_iff}
    $a$ is a least element of $X$ with respect to $R$ iff
        $a$ is a least element of $X$ with respect to $\tostrictorder{R}$.
\end{proposition}
\begin{proof}
    Follows by \auto{least_element,tostrictorder_iff}.
\end{proof}

\begin{proposition}\label{initial_segment_toorder_iff}
    $I$ is an initial segment of $A$ with respect to $R$ iff
        $I$ is an initial segment of $A$ with respect to $\toorder{A}{R}$.
\end{proposition}
\begin{proof}
    Follows by \auto{initial_segment,toorder_iff}.
\end{proof}

\begin{proposition}\label{initial_segment_tostrictorder_iff}
    $I$ is an initial segment of $A$ with respect to $R$ iff
        $I$ is an initial segment of $A$ with respect to $\tostrictorder{R}$.
\end{proposition}
\begin{proof}
    Follows by \auto{initial_segment,tostrictorder_iff}.
\end{proof}

\begin{proposition}\label{linear_order_from_strict_linear_order}
    Suppose $R$ is a strict linear order on $A$.
    Then $\toorder{A}{R}$ is a linear order on $A$.
\end{proposition}
\begin{proof}
    $\toorder{A}{R}$ is an order on $A$
        by \obv{strict_linear_order,order_from_strictorder}.
    $\toorder{A}{R}$ is connex on $A$
        by \auto{strict_linear_order,connex,toorder_intro}.
    Follows by \obv{linear_order}.
\end{proof}

\begin{proposition}\label{strict_linear_order_from_linear_order}
    Suppose $R$ is a linear order on $A$.
    Then $\tostrictorder{R}$ is a strict linear order on $A$.
\end{proposition}
\begin{proof}
    $\tostrictorder{R}$ is a strict order
        by \obv{linear_order,order_on_implies_order,strictorder_from_order}.
    $\tostrictorder{R}$ is a binary relation on $A$
        by \auto{linear_order,tostrictorder,subseteq,times_tuple_elim}.
    $\tostrictorder{R}$ is connex on $A$
        by \auto{linear_order,connex,tostrictorder_iff}.
    Follows by \obv{strict_linear_order}.
\end{proof}

\begin{theorem}\label{well_order_from_strict_well_order}
    Suppose $R$ is a strict well order on $A$.
    Then $\toorder{A}{R}$ is a well order on $A$.
\end{theorem}
\begin{proof}
    $\toorder{A}{R}$ is a linear order on $A$
        by \obv{strict_well_order,linear_order_from_strict_linear_order}.
    Follows by \auto{well_order,strict_well_order,least_element_toorder_iff}.
\end{proof}

\begin{theorem}\label{strict_well_order_from_well_order}
    Suppose $R$ is a well order on $A$.
    Then $\tostrictorder{R}$ is a strict well order on $A$.
\end{theorem}
\begin{proof}
    $\tostrictorder{R}$ is a strict linear order on $A$
        by \obv{well_order,strict_linear_order_from_linear_order}.
    Follows by \auto{well_order,strict_well_order,least_element_tostrictorder_iff}.
\end{proof}
