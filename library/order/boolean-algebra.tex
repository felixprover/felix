\section{Boolean algebras}

\import{order/lattice-complement.tex}

\begin{struct}\label{boolean_algebra}
    A Boolean algebra $B$ is a bounded distributive lattice equipped with
    \begin{enumerate}
        \item\label{boolean_complement_graph}
            a complement operation
            $\booleanComplement[B]:\carrier[B]\to\carrier[B]$;
    \end{enumerate}
    such that
    \begin{enumerate}
        \item\label{boolean_meet_complement_law}
            for every $x\in\carrier[B]$ we have
                $\meet[B](x,\booleanComplement[B](x))
                    =\latticeBottom[B]$.
        \item\label{boolean_join_complement_law}
            for every $x\in\carrier[B]$ we have
                $\join[B](x,\booleanComplement[B](x))
                    =\latticeTop[B]$.
    \end{enumerate}
\end{struct}

\begin{proposition}\label{boolean_algebra_is_bounded_distributive_lattice}
    Let $B$ be a Boolean algebra.
    Then $B$ is a bounded distributive lattice.
\end{proposition}
\begin{proof}
    Follows by \obv{boolean_algebra}.
\end{proof}

\begin{theorem}\label{boolean_complement_involutive}
    Let $B$ be a Boolean algebra.
    Let $x\in\carrier[B]$.
    Then $\booleanComplement[B](\booleanComplement[B](x))=x$.
\end{theorem}
\begin{proof}
    $B$ is a bounded distributive lattice by
        \obv{boolean_algebra_is_bounded_distributive_lattice}.
    $\booleanComplement[B](x)\in\carrier[B]$ by
        \obv{boolean_complement_apply_type}.
    $\areLatticeComplements{B}{x}{\booleanComplement[B](x)}$ by
        \obv{boolean_are_lattice_complements}.
    Hence $\areLatticeComplements{B}{\booleanComplement[B](x)}{x}$ by
        \obv{are_lattice_complements_symmetric}.
    $\areLatticeComplements
        {B}{\booleanComplement[B](x)}
        {\booleanComplement[B](\booleanComplement[B](x))}$ by
        \obv{boolean_are_lattice_complements}.
    Hence $x=\booleanComplement[B](\booleanComplement[B](x))$ by
        \obv{lattice_complement_unique}.
\end{proof}

\begin{proposition}\label{boolean_complement_lattice_bottom}
    Let $B$ be a Boolean algebra.
    Then $\booleanComplement[B](\latticeBottom[B])
        =\latticeTop[B]$.
\end{proposition}
\begin{proof}

        \obv{boolean_algebra_is_bounded_distributive_lattice}.
    Hence $B$ is a bounded lattice by
        \obv{bounded_distributive_lattice_is_bounded_lattice}.
    $\latticeBottom[B],\latticeTop[B]\in\carrier[B]$ by
        \obv{lattice_bottom_type,lattice_top_type}.
    $\meet[B](\latticeBottom[B],\latticeTop[B])
        =\latticeBottom[B]$ by \obv{meet_lattice_bottom}.
    $\join[B](\latticeBottom[B],\latticeTop[B])
        =\latticeTop[B]$ by \obv{join_lattice_bottom}.
    Hence $\areLatticeComplements
        {B}{\latticeBottom[B]}{\latticeTop[B]}$ by
        \obv{are_lattice_complements}.
    $\areLatticeComplements
        {B}{\latticeBottom[B]}
        {\booleanComplement[B](\latticeBottom[B])}$ by
        \obv{boolean_are_lattice_complements}.
    Follows by \obv{lattice_complement_unique}.
\end{proof}

\begin{theorem}\label{meet_eq_lattice_bottom_iff_leq_boolean_complement}
    Let $B$ be a Boolean algebra.
    Let $x,y\in\carrier[B]$.
    Then $\meet[B](x,y)=\latticeBottom[B]$ iff
        $x\mathrel{\leq[B]}\booleanComplement[B](y)$.
\end{theorem}
\begin{proof}
    $B$ is a bounded lattice and $B$ is a distributive lattice by
        \obv{boolean_algebra_is_bounded_lattice,
            boolean_algebra_is_distributive_lattice}.
    Hence $B$ is a meet semilattice and $B$ is a join semilattice by
        \obv{lattice_is_meet_semilattice,lattice_is_join_semilattice}.
    $\booleanComplement[B](y)\in\carrier[B]$ by
        \obv{boolean_complement_apply_type}.
    $\meet[B](x,y),
        \meet[B](x,\booleanComplement[B](y))\in\carrier[B]$ by
        \obv{meet_type}.

    Show that if $\meet[B](x,y)=\latticeBottom[B]$, then
        $x\mathrel{\leq[B]}\booleanComplement[B](y)$.
    \begin{subproof}
        Assume $\meet[B](x,y)=\latticeBottom[B]$.
        $\join[B](y,\booleanComplement[B](y))
            =\latticeTop[B]$ by \obv{boolean_join_complement}.
        $x=\meet[B](x,\latticeTop[B])$ by \obv{meet_lattice_top}.
        Hence $x=\meet[B]
            (x,\join[B](y,\booleanComplement[B](y)))$.
        $\meet[B](x,\join[B](y,\booleanComplement[B](y)))
            =\join[B]
                (\meet[B](x,y),
                 \meet[B](x,\booleanComplement[B](y)))$ by
            \obv{meet_distributes_over_join}.
        Hence $x=\join[B]
            (\latticeBottom[B],
             \meet[B](x,\booleanComplement[B](y)))$.
        $\join[B]
            (\latticeBottom[B],
             \meet[B](x,\booleanComplement[B](y)))
            =\meet[B](x,\booleanComplement[B](y))$ by
            \obv{join_lattice_bottom}.
        Hence $\meet[B](x,\booleanComplement[B](y))=x$.
        Follows by \obv{leq_iff_meet_eq_left}.
    \end{subproof}

    Show that if $x\mathrel{\leq[B]}\booleanComplement[B](y)$,
        then $\meet[B](x,y)=\latticeBottom[B]$.
    \begin{subproof}
        Assume $x\mathrel{\leq[B]}\booleanComplement[B](y)$.
        $\meet[B](x,y)\mathrel{\leq[B]}
            \meet[B](\booleanComplement[B](y),y)$ by
            \obv{meet_monotone_left}.
        $\meet[B](\booleanComplement[B](y),y)
            =\meet[B](y,\booleanComplement[B](y))$ by
            \obv{meet_commutative}.
        Hence $\meet[B](\booleanComplement[B](y),y)
            =\latticeBottom[B]$ by \obv{boolean_meet_complement}.
        Hence $\meet[B](x,y)\mathrel{\leq[B]}
            \latticeBottom[B]$.
        $\latticeBottom[B]\mathrel{\leq[B]}\meet[B](x,y)$ by
            \obv{lattice_bottom_leq}.
        $\leq[B]$ is antisymmetric by
            \obv{lattice_is_ordered_set,
                ordered_set_order_antisymmetric}.
        Follows by \obv{antisymmetric}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{theorem}\label{boolean_complement_antitone}
    Let $B$ be a Boolean algebra.
    Let $x,y\in\carrier[B]$.
    Suppose $x\mathrel{\leq[B]}y$.
    Then $\booleanComplement[B](y)
        \mathrel{\leq[B]}\booleanComplement[B](x)$.
\end{theorem}
\begin{proof}
    $\booleanComplement[B](x),\booleanComplement[B](y)
        \in\carrier[B]$ by \obv{boolean_complement_apply_type}.
    $\meet[B](\booleanComplement[B](y),x)
        \mathrel{\leq[B]}
        \meet[B](\booleanComplement[B](y),y)$ by
        \obv{meet_monotone_right}.
    $\meet[B](\booleanComplement[B](y),y)
        =\meet[B](y,\booleanComplement[B](y))$ by
        \obv{meet_commutative}.
    Hence $\meet[B](\booleanComplement[B](y),y)
        =\latticeBottom[B]$ by \obv{boolean_meet_complement}.
    Hence $\meet[B](\booleanComplement[B](y),x)
        \mathrel{\leq[B]}\latticeBottom[B]$.
    $\meet[B](\booleanComplement[B](y),x)\in\carrier[B]$ by
        \obv{meet_type}.
    $\latticeBottom[B]\mathrel{\leq[B]}
        \meet[B](\booleanComplement[B](y),x)$ by
        \obv{lattice_bottom_leq}.
    $\leq[B]$ is antisymmetric by
        \obv{lattice_is_ordered_set,
            ordered_set_order_antisymmetric}.
    Hence $\meet[B](\booleanComplement[B](y),x)
        =\latticeBottom[B]$ by \obv{antisymmetric}.
    Follows by
        \obv{meet_eq_lattice_bottom_iff_leq_boolean_complement}.
\end{proof}

\begin{theorem}\label{boolean_complement_meet}
    Let $B$ be a Boolean algebra.
    Let $x,y\in\carrier[B]$.
    Then $\booleanComplement[B](\meet[B](x,y))
        =\join[B]
            (\booleanComplement[B](x),\booleanComplement[B](y))$.
\end{theorem}
\begin{proof}
    $B$ is a bounded distributive lattice by
        \obv{boolean_algebra_is_bounded_distributive_lattice}.
    Hence $B$ is a bounded lattice and $B$ is a distributive lattice by
        \obv{bounded_distributive_lattice_is_bounded_lattice,
            bounded_distributive_lattice_is_distributive_lattice}.
    Hence $B$ is a lattice by \obv{bounded_lattice_is_lattice}.
    Hence $B$ is a meet semilattice and $B$ is a join semilattice by
        \obv{lattice_is_meet_semilattice,lattice_is_join_semilattice}.
    Let $u=\booleanComplement[B](x)$.
    Let $v=\booleanComplement[B](y)$.
    Let $m=\meet[B](x,y)$.
    Let $c=\join[B](u,v)$.
    $u,v,m,c\in\carrier[B]$ by
        \auto{boolean_complement_apply_type,meet_type,join_type}.
    $\booleanComplement[B](u)=x$ and
        $\booleanComplement[B](v)=y$ by
        \auto{boolean_complement_involutive}.

    $m\mathrel{\leq[B]}x$ by \auto{meet_lower_left}.
    Hence $m\mathrel{\leq[B]}\booleanComplement[B](u)$.
    Hence $\meet[B](m,u)=\latticeBottom[B]$ by
        \obv{meet_eq_lattice_bottom_iff_leq_boolean_complement}.
    $m\mathrel{\leq[B]}y$ by \auto{meet_lower_right}.
    Hence $m\mathrel{\leq[B]}\booleanComplement[B](v)$.
    Hence $\meet[B](m,v)=\latticeBottom[B]$ by
        \obv{meet_eq_lattice_bottom_iff_leq_boolean_complement}.
    $\meet[B](m,c)=\join[B](\meet[B](m,u),\meet[B](m,v))$ by
        \auto{meet_distributes_over_join}.
    Hence $\meet[B](m,c)=\join[B]
        (\latticeBottom[B],\latticeBottom[B])$.
    $\join[B](\latticeBottom[B],\latticeBottom[B])
        =\latticeBottom[B]$ by \obv{join_lattice_bottom}.
    Hence $\meet[B](m,c)=\latticeBottom[B]$.

    $u\mathrel{\leq[B]}c$ and $v\mathrel{\leq[B]}c$ by
        \auto{join_upper_left,join_upper_right}.
    $\join[B](u,x)=\join[B](x,u)$ by \obv{join_commutative}.
    Hence $\join[B](u,x)=\latticeTop[B]$ by
        \auto{boolean_join_complement}.
    $\join[B](u,x)\mathrel{\leq[B]}\join[B](c,x)$ by
        \obv{join_monotone_left}.
    Hence $\latticeTop[B]\mathrel{\leq[B]}\join[B](c,x)$.
    $\join[B](c,x)\in\carrier[B]$ by \obv{join_type}.
    Hence $\join[B](c,x)\mathrel{\leq[B]}\latticeTop[B]$ by
        \obv{lattice_leq_top}.
    $\leq[B]$ is antisymmetric by
        \obv{lattice_is_ordered_set,
            ordered_set_order_antisymmetric}.
    Hence $\join[B](c,x)=\latticeTop[B]$ by \obv{antisymmetric}.

    $\join[B](v,y)=\join[B](y,v)$ by \obv{join_commutative}.
    Hence $\join[B](v,y)=\latticeTop[B]$ by
        \auto{boolean_join_complement}.
    $\join[B](v,y)\mathrel{\leq[B]}\join[B](c,y)$ by
        \obv{join_monotone_left}.
    Hence $\latticeTop[B]\mathrel{\leq[B]}\join[B](c,y)$.
    $\join[B](c,y)\in\carrier[B]$ by \obv{join_type}.
    Hence $\join[B](c,y)\mathrel{\leq[B]}\latticeTop[B]$ by
        \obv{lattice_leq_top}.
    Hence $\join[B](c,y)=\latticeTop[B]$ by \obv{antisymmetric}.

    $\join[B](m,c)=\join[B](c,m)$ by \obv{join_commutative}.
    $\join[B](c,m)
        =\meet[B](\join[B](c,x),\join[B](c,y))$ by
        \auto{join_distributes_over_meet}.
    Hence $\join[B](m,c)=\meet[B]
        (\latticeTop[B],\latticeTop[B])$.
    $\meet[B](\latticeTop[B],\latticeTop[B])
        =\latticeTop[B]$ by \obv{meet_lattice_top}.
    Hence $\join[B](m,c)=\latticeTop[B]$.
    Hence $\areLatticeComplements{B}{m}{c}$ by
        \obv{are_lattice_complements}.
    $\areLatticeComplements
        {B}{m}{\booleanComplement[B](m)}$ by
        \obv{boolean_are_lattice_complements}.
    Follows by \obv{lattice_complement_unique}.
\end{proof}

\begin{theorem}\label{boolean_complement_join}
    Let $B$ be a Boolean algebra.
    Let $x,y\in\carrier[B]$.
    Then $\booleanComplement[B](\join[B](x,y))
        =\meet[B]
            (\booleanComplement[B](x),\booleanComplement[B](y))$.
\end{theorem}
\begin{proof}
    Hence $B$ is a meet semilattice and $B$ is a join semilattice by
        \obv{lattice_is_meet_semilattice,lattice_is_join_semilattice}.
    Let $u=\booleanComplement[B](x)$.
    Let $v=\booleanComplement[B](y)$.
    $u,v\in\carrier[B]$ by \auto{boolean_complement_apply_type}.
    Let $m=\meet[B](u,v)$.
    $m\in\carrier[B]$ by \auto{meet_type}.
    $\booleanComplement[B](m)
        =\join[B]
            (\booleanComplement[B](u),\booleanComplement[B](v))$ by
        \auto{boolean_complement_meet}.
    $\booleanComplement[B](u)=x$ and
        $\booleanComplement[B](v)=y$ by
        \auto{boolean_complement_involutive}.
    Hence $\booleanComplement[B](m)=\join[B](x,y)$.
    Hence $\booleanComplement[B](\booleanComplement[B](m))
        =\booleanComplement[B](\join[B](x,y))$.
    $\booleanComplement[B](\booleanComplement[B](m))=m$ by
        \obv{boolean_complement_involutive}.
\end{proof}

\begin{definition}\label{boolean_difference}
    $\booleanDifference{B}{a}{b}
        =\meet[B](a,\booleanComplement[B](b))$.
\end{definition}

\begin{proposition}\label{boolean_difference_apply}
    Let $B$ be a Boolean algebra.
    Let $a,b\in\carrier[B]$.
    Then $\booleanDifference{B}{a}{b}
        =\meet[B](a,\booleanComplement[B](b))$.
\end{proposition}
\begin{proof}
    Follows by \obv{boolean_difference}.
\end{proof}

\begin{proposition}\label{boolean_difference_type}
    Let $B$ be a Boolean algebra.
    Let $a,b\in\carrier[B]$.
    Then $\booleanDifference{B}{a}{b}\in\carrier[B]$.
\end{proposition}
\begin{proof}

        \obv{lattice_is_meet_semilattice}.
    $\booleanComplement[B](b)\in\carrier[B]$ by
        \obv{boolean_complement_apply_type}.
    Follows by \auto{boolean_difference_apply,meet_type}.
\end{proof}

\begin{proposition}\label{boolean_difference_leq_left}

\begin{proposition}\label{boolean_difference_leq_complement_right}

\begin{theorem}\label{leq_iff_boolean_difference_lattice_bottom}
    Let $B$ be a Boolean algebra.
    Let $a,b\in\carrier[B]$.
    Then $a\mathrel{\leq[B]}b$ iff
        $\booleanDifference{B}{a}{b}=\latticeBottom[B]$.
\end{theorem}
\begin{proof}
    $\booleanComplement[B](b)\in\carrier[B]$ by
        \obv{boolean_complement_apply_type}.
    $\booleanComplement[B](\booleanComplement[B](b))=b$ by
        \obv{boolean_complement_involutive}.
    Show that if $a\mathrel{\leq[B]}b$, then
        $\booleanDifference{B}{a}{b}=\latticeBottom[B]$.
    \begin{subproof}
        Assume $a\mathrel{\leq[B]}b$.
        Hence $a\mathrel{\leq[B]}
            \booleanComplement[B](\booleanComplement[B](b))$.
        Hence $\meet[B](a,\booleanComplement[B](b))
            =\latticeBottom[B]$ by
            \obv{meet_eq_lattice_bottom_iff_leq_boolean_complement}.
        Follows by \obv{boolean_difference_apply}.
    \end{subproof}
    Show that if $\booleanDifference{B}{a}{b}=\latticeBottom[B]$,
        then $a\mathrel{\leq[B]}b$.
    \begin{subproof}
        Assume $\booleanDifference{B}{a}{b}=\latticeBottom[B]$.
        Hence $\meet[B](a,\booleanComplement[B](b))
            =\latticeBottom[B]$ by \obv{boolean_difference_apply}.
        Hence $a\mathrel{\leq[B]}
            \booleanComplement[B](\booleanComplement[B](b))$ by
            \obv{meet_eq_lattice_bottom_iff_leq_boolean_complement}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{theorem}\label{not_leq_iff_boolean_difference_nonbottom}
    Let $B$ be a Boolean algebra.
    Let $a,b\in\carrier[B]$.
    Then $a\not\mathrel{\leq[B]}b$ iff
        $\booleanDifference{B}{a}{b}\neq\latticeBottom[B]$.
\end{theorem}
\begin{proof}
    Follows by
        \auto{leq_iff_boolean_difference_lattice_bottom}.
\end{proof}

\begin{theorem}\label{exact_boolean_algebra_extensionality}
    Let $A,B$ be sets.
    Suppose $A$ is an exact Boolean algebra and
        $B$ is an exact Boolean algebra.
    Suppose $\carrier[A]=\carrier[B]$.
    Suppose $\leq[A]=\leq[B]$.
    Suppose $\meet[A]=\meet[B]$.
    Suppose $\join[A]=\join[B]$.
    Suppose $\latticeBottom[A]=\latticeBottom[B]$.
    Suppose $\latticeTop[A]=\latticeTop[B]$.
    Suppose $\booleanComplement[A]=\booleanComplement[B]$.
    Then $A=B$.
\end{theorem}
\begin{proof}
    Follows by structure extensionality.
\end{proof}
