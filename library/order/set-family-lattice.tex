\section{Bounded distributive lattices of sets}

\import{order/lattice.tex}
\import{set/logical-graph.tex}
\import{set/powerset.tex}

\begin{definition}\label{bounded_set_lattice_family}
    $C$ is a bounded set-lattice family on $X$ iff
        $C\subseteq\pow{X}$ and
        $\emptyset\in C$ and
        $X\in C$ and
        (for all $A,B\in C$ we have $A\inter B\in C$) and
        (for all $A,B\in C$ we have $A\union B\in C$).
\end{definition}

\begin{proposition}\label{bounded_set_lattice_family_subsets}
    Suppose $C$ is a bounded set-lattice family on $X$.
    Then $C\subseteq\pow{X}$.
\end{proposition}
\begin{proof}
    Follows by \obv{bounded_set_lattice_family}.
\end{proof}

\begin{proposition}\label{bounded_set_lattice_family_emptyset}
    Suppose $C$ is a bounded set-lattice family on $X$.
    Then $\emptyset\in C$.
\end{proposition}
\begin{proof}
    Follows by \obv{bounded_set_lattice_family}.
\end{proof}

\begin{proposition}\label{bounded_set_lattice_family_whole}
    Suppose $C$ is a bounded set-lattice family on $X$.
    Then $X\in C$.
\end{proposition}
\begin{proof}
    Follows by \obv{bounded_set_lattice_family}.
\end{proof}

\begin{proposition}\label{bounded_set_lattice_family_inter}
    Suppose $C$ is a bounded set-lattice family on $X$.
    Let $A,B\in C$.
    Then $A\inter B\in C$.
\end{proposition}
\begin{proof}
    Follows by \obv{bounded_set_lattice_family}.
\end{proof}

\begin{proposition}\label{bounded_set_lattice_family_union}
    Suppose $C$ is a bounded set-lattice family on $X$.
    Let $A,B\in C$.
    Then $A\union B\in C$.
\end{proposition}
\begin{proof}
    Follows by \obv{bounded_set_lattice_family}.
\end{proof}

\begin{definition}\label{set_family_meet_graph}
    $\setFamilyMeetGraph{C}=
        \Apply{\binaryGraphOnOp}
            {C,C,\fun{A,B:\Set}{A\inter B}}$.
\end{definition}

\begin{definition}\label{set_family_join_graph}
    $\setFamilyJoinGraph{C}=
        \Apply{\binaryGraphOnOp}
            {C,C,\fun{A,B:\Set}{A\union B}}$.
\end{definition}

\begin{proposition}\label{set_family_meet_graph_type}
    Suppose $C$ is a bounded set-lattice family on $X$.
    Then $\setFamilyMeetGraph{C}\in\funs{C\times C}{C}$.
\end{proposition}
\begin{proof}
    For all $A,B\in C$ we have $A\inter B\in C$ by
        \obv{bounded_set_lattice_family_inter}.
    Follows by
        \auto{set_family_meet_graph,logical_binary_graph_on_function}.
\end{proof}

\begin{proposition}\label{set_family_join_graph_type}
    Suppose $C$ is a bounded set-lattice family on $X$.
    Then $\setFamilyJoinGraph{C}\in\funs{C\times C}{C}$.
\end{proposition}
\begin{proof}
    For all $A,B\in C$ we have $A\union B\in C$ by
        \obv{bounded_set_lattice_family_union}.
    Follows by
        \auto{set_family_join_graph,logical_binary_graph_on_function}.
\end{proof}

\begin{proposition}\label{set_family_meet_graph_apply}
    Let $A,B\in C$.
    Then $\apply{\setFamilyMeetGraph{C}}{(A,B)}=A\inter B$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{set_family_meet_graph,logical_binary_graph_on_apply}.
\end{proof}

\begin{proposition}\label{set_family_join_graph_apply}
    Let $A,B\in C$.
    Then $\apply{\setFamilyJoinGraph{C}}{(A,B)}=A\union B$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{set_family_join_graph,logical_binary_graph_on_apply}.
\end{proof}

\begin{definition}\label{set_family_lattice}
    $\setFamilyLattice{X}{C}=
        \structure[bounded lattice]{
            C,
            \subseteqrel{C},
            \setFamilyMeetGraph{C},
            \setFamilyJoinGraph{C},
            \emptyset,
            X
        }$.
\end{definition}

\begin{proposition}\label{set_family_lattice_carrier}
    $\carrier[\setFamilyLattice{X}{C}]=C$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{set_family_lattice,bounded_latticeaggregateprojection_carrier}.
\end{proof}

\begin{proposition}\label{set_family_lattice_order}
    $\leq[\setFamilyLattice{X}{C}]=\subseteqrel{C}$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{set_family_lattice,bounded_latticeaggregateprojection_leq}.
\end{proof}

\begin{proposition}\label{set_family_lattice_meet}
    $\meet[\setFamilyLattice{X}{C}]=\setFamilyMeetGraph{C}$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{set_family_lattice,bounded_latticeaggregateprojection_meet}.
\end{proof}

\begin{proposition}\label{set_family_lattice_join}
    $\join[\setFamilyLattice{X}{C}]=\setFamilyJoinGraph{C}$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{set_family_lattice,bounded_latticeaggregateprojection_join}.
\end{proof}

\begin{proposition}\label{set_family_lattice_bottom}
    $\latticeBottom[\setFamilyLattice{X}{C}]=\emptyset$.

\begin{proof}

\begin{proposition}\label{set_family_lattice_top}

\begin{theorem}\label{set_family_lattice_is_exact_bounded_distributive_lattice}
    Suppose $C$ is a bounded set-lattice family on $X$.
    Then $\setFamilyLattice{X}{C}$ is an exact bounded distributive lattice.
\end{theorem}
\begin{proof}
    Let $L=\setFamilyLattice{X}{C}$.
    $\carrier[L]=C$ by \auto{set_family_lattice_carrier}.
    $\leq[L]=\subseteqrel{C}$ by \auto{set_family_lattice_order}.
    $\meet[L]=\setFamilyMeetGraph{C}$ by
        \auto{set_family_lattice_meet}.
    $\join[L]=\setFamilyJoinGraph{C}$ by
        \auto{set_family_lattice_join}.
    $\latticeBottom[L]=\emptyset$ by
        \auto{set_family_lattice_bottom}.
    $\latticeTop[L]=X$ by \auto{set_family_lattice_top}.
    $\subseteqrel{C}$ is an order on $C$ by
        \obv{subseteqrel_is_order}.
    Hence $\leq[L]$ is an order on $\carrier[L]$.
    $\leq[L]$ is antisymmetric.

    $\meet[L]\in\funs{\carrier[L]\times\carrier[L]}{\carrier[L]}$ by
        \auto{set_family_meet_graph_type}.
    For all $A,B\in\carrier[L]$ we have
        $\meet[L](A,B)\in\carrier[L]$ by
        \obv{times_tuple_intro,funs_type_apply}.
    For all $A,B\in\carrier[L]$ we have
        $\meet[L](A,B)\mathrel{\leq[L]}A$ and
        $\meet[L](A,B)\mathrel{\leq[L]}B$ by
        \auto{set_family_meet_graph_apply,subseteqrel,
            inter_lower_left,inter_lower_right}.
    For all $D,A,B\in\carrier[L]$ such that
        $D\mathrel{\leq[L]}A$ and $D\mathrel{\leq[L]}B$ we have
        $D\mathrel{\leq[L]}\meet[L](A,B)$ by
        \auto{set_family_meet_graph_apply,subseteqrel,
            subseteq_inter_iff}.

    $\join[L]\in\funs{\carrier[L]\times\carrier[L]}{\carrier[L]}$ by
        \auto{set_family_join_graph_type}.
    For all $A,B\in\carrier[L]$ we have
        $\join[L](A,B)\in\carrier[L]$ by
        \obv{times_tuple_intro,funs_type_apply}.
    For all $A,B\in\carrier[L]$ we have
        $A\mathrel{\leq[L]}\join[L](A,B)$ and
        $B\mathrel{\leq[L]}\join[L](A,B)$ by
        \auto{set_family_join_graph_apply,subseteqrel,
            union_upper_left,union_upper_right}.
    For all $A,B,D\in\carrier[L]$ such that
        $A\mathrel{\leq[L]}D$ and $B\mathrel{\leq[L]}D$ we have
        $\join[L](A,B)\mathrel{\leq[L]}D$ by
        \auto{set_family_join_graph_apply,subseteqrel,
            subseteq_union_iff}.

    $\emptyset\in C$ and $X\in C$ by
        \obv{bounded_set_lattice_family_emptyset,
            bounded_set_lattice_family_whole}.
    For all $A\in C$ we have $\emptyset\subseteq A$ by
        \obv{emptyset_subseteq}.
    Hence for all $A\in\carrier[L]$ we have
        $\latticeBottom[L]\mathrel{\leq[L]}A$ by
        \auto{subseteqrel}.
    $C\subseteq\pow{X}$ by
        \obv{bounded_set_lattice_family_subsets}.
    For all $A\in C$ we have $A\subseteq X$ by
        \auto{elem_subseteq,pow_iff,subseteq}.
    Hence for all $A\in\carrier[L]$ we have
        $A\mathrel{\leq[L]}\latticeTop[L]$ by
        \auto{subseteqrel}.

    For all $A,B,D\in\carrier[L]$ we have
        $\meet[L](A,\join[L](B,D))
            =\join[L](\meet[L](A,B),\meet[L](A,D))$ by
        \auto{set_family_meet_graph_apply,
            set_family_join_graph_apply,inter_distrib_union,
            bounded_set_lattice_family_inter,
            bounded_set_lattice_family_union}.
    Hence $L$ is a bounded distributive lattice by
        \auto{bounded_distributive_lattice,set_family_lattice,
            bounded_latticeaggregateexact}.
    Follows by
        \auto{set_family_lattice,bounded_latticeaggregateexact}.
\end{proof}

\begin{classification}\label{set_family_lattice_is_bounded_distributive_lattice}
    Suppose $C$ is a bounded set-lattice family on $X$.
    Then $\setFamilyLattice{X}{C}$ is a bounded distributive lattice.
\end{classification}
\begin{proof}
    Follows by
        \obv{set_family_lattice_is_exact_bounded_distributive_lattice}.
\end{proof}

\begin{proposition}\label{powerset_is_bounded_set_lattice_family}
    $\pow{X}$ is a bounded set-lattice family on $X$.
\end{proposition}
\begin{proof}
    $\pow{X}\subseteq\pow{X}$ by \obv{subseteq}.
    $\emptyset\in\pow{X}$ by \obv{powerset_bottom}.
    $X\in\pow{X}$ by \auto{pow_iff,subseteq}.
    For all $A,B\in\pow{X}$ we have $A\inter B\in\pow{X}$ by
        \auto{pow_iff,subseteq,inter_lower_left,subseteq_transitive}.
    For all $A,B\in\pow{X}$ we have $A\union B\in\pow{X}$ by
        \auto{pow_iff,subseteq,union_iff}.
    Follows by \obv{bounded_set_lattice_family}.
\end{proof}

\begin{definition}\label{powerset_lattice}
    $\powersetLattice{X}=\setFamilyLattice{X}{\pow{X}}$.
\end{definition}

\begin{theorem}\label{powerset_lattice_is_exact_bounded_distributive_lattice}
    $\powersetLattice{X}$ is an exact bounded distributive lattice.
\end{theorem}
\begin{proof}
    Follows by
        \auto{powerset_lattice,powerset_is_bounded_set_lattice_family,
            set_family_lattice_is_exact_bounded_distributive_lattice}.
\end{proof}
