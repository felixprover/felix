\begin{proposition}\label{lattice_minimal_element_is_least}
    Let $L$ be a lattice.
    Suppose $m$ is a minimal element of $\carrier[L]$ with respect to
        $\leq[L]$.
    Then $m$ is a least element of $\carrier[L]$ with respect to $\leq[L]$.
\end{proposition}
\begin{proof}
    $L$ is a meet semilattice by \obv{lattice_is_meet_semilattice}.
    $m\in\carrier[L]$ by \obv{minimal_element}.
    Show for all $x\in\carrier[L]$ we have
        $x=m$ or $m\mathrel{\leq[L]}x$.
    \begin{subproof}
        Fix $x\in\carrier[L]$.
        $\meet[L](m,x)\in\carrier[L]$ by \obv{meet_type}.
        $\meet[L](m,x)\mathrel{\leq[L]}m$ by \auto{meet_lb}.
        Hence $\meet[L](m,x)=m$ by \obv{minimal_element}.
        Thus $m\mathrel{\leq[L]}x$ by \obv{leq_iff_meet_eq_left}.
    \end{subproof}
    Follows by \obv{least_element}.
\end{proof}

\begin{proposition}\label{lattice_maximal_element_is_greatest}
    Let $L$ be a lattice.
    Suppose $m$ is a maximal element of $\carrier[L]$ with respect to
        $\leq[L]$.
    Then $m$ is a greatest element of $\carrier[L]$ with respect to $\leq[L]$.
\end{proposition}
\begin{proof}
    $L$ is a join semilattice by \obv{lattice_is_join_semilattice}.
    $m\in\carrier[L]$ by \obv{maximal_element}.
    Show for all $x\in\carrier[L]$ we have
        $x=m$ or $x\mathrel{\leq[L]}m$.
    \begin{subproof}
        Fix $x\in\carrier[L]$.
        $\join[L](m,x)\in\carrier[L]$ by \obv{join_type}.
        $m\mathrel{\leq[L]}\join[L](m,x)$ by \auto{join_ub}.
        Hence $\join[L](m,x)=m$ by \obv{maximal_element}.
        Thus $\join[L](x,m)=m$ by \obv{join_commutative}.
        Hence $x\mathrel{\leq[L]}m$ by \obv{leq_iff_join_eq_right}.
    \end{subproof}
    Follows by \obv{greatest_element}.
\end{proof}

\begin{theorem}\label{finite_inhabited_lattice_has_least_element}
    Let $L$ be a finite lattice.
    Suppose $\carrier[L]$ is inhabited.
    Then there exists $m\in\carrier[L]$ such that
        $m$ is a least element of $\carrier[L]$ with respect to $\leq[L]$.
\end{theorem}
\begin{proof}
    $\leq[L]$ is an order on $\carrier[L]$ by
        \obv{ordered_set_order_on_carrier}.
    $\carrier[L]$ is a finite set by \obv{finite_structure}.
    $\carrier[L]\subseteq\carrier[L]$ by \obv{subseteq}.
    Take $m\in\carrier[L]$ such that
        $m$ is a minimal element of $\carrier[L]$ with respect to $\leq[L]$
        by \obv{finite_minimal_element}.
    $m$ is a least element of $\carrier[L]$ with respect to $\leq[L]$ by
        \obv{lattice_minimal_element_is_least}.
\end{proof}

\begin{definition}\label{least_lattice_elements}
    Let $L$ be an ordered set.
    $\leastLatticeElements{L}
        =\{b\in\carrier[L]\mid
            \text{$b$ is a least element of $\carrier[L]$
                with respect to $\leq[L]$}\}$.
\end{definition}

\begin{definition}\label{greatest_lattice_elements}
    Let $L$ be an ordered set.
    $\greatestLatticeElements{L}
        =\{t\in\carrier[L]\mid
            \text{$t$ is a greatest element of $\carrier[L]$
                with respect to $\leq[L]$}\}$.
\end{definition}

\begin{definition}\label{canonical_lattice_bottom}
    Let $L$ be an ordered set.
    $\canonicalLatticeBottom{L}=\choice{\leastLatticeElements{L}}$.
\end{definition}

\begin{definition}\label{canonical_lattice_top}
    Let $L$ be an ordered set.
    $\canonicalLatticeTop{L}=\choice{\greatestLatticeElements{L}}$.
\end{definition}

\begin{theorem}\label{canonical_lattice_bottom_is_least}
    Let $L$ be a finite lattice.
    Suppose $\carrier[L]$ is inhabited.
    Then $\canonicalLatticeBottom{L}$ is a least element of $\carrier[L]$
        with respect to $\leq[L]$.
\end{theorem}
\begin{proof}
    Take $b\in\carrier[L]$ such that
        $b$ is a least element of $\carrier[L]$ with respect to $\leq[L]$
        by \obv{finite_inhabited_lattice_has_least_element}.
    $b\in\leastLatticeElements{L}$ by \obv{least_lattice_elements}.
    Hence $\choice{\leastLatticeElements{L}}\in\leastLatticeElements{L}$ by
        \obv{choice_member}.
    Follows by \obv{canonical_lattice_bottom,least_lattice_elements}.
\end{proof}

\begin{theorem}\label{canonical_lattice_top_is_greatest}
    Let $L$ be a finite lattice.

\begin{proposition}\label{canonical_lattice_bottom_unique}
    Let $L$ be a finite lattice.
    Suppose $\carrier[L]$ is inhabited.
    Suppose $b$ is a least element of $\carrier[L]$ with respect to $\leq[L]$.
    Then $b=\canonicalLatticeBottom{L}$.

\begin{proof}

        \obv{ordered_set_order_antisymmetric}.
    $\canonicalLatticeBottom{L}$ is a least element of $\carrier[L]$
        with respect to $\leq[L]$ by

\end{definition}

\begin{theorem}\label{bounded_expansion_is_exact_bounded_lattice}
    Let $L$ be a finite lattice.
    Suppose $\carrier[L]$ is inhabited.
    Then $\boundedExpansion{L}$ is an exact bounded lattice.
\end{theorem}
\begin{proof}
    $L$ is a lattice by \obv{lattice}.
    Hence $L$ is an ordered set by \obv{lattice_is_ordered_set}.
    Let $E=\boundedExpansion{L}$.
    $\carrier[E]=\carrier[L]$ by \auto{bounded_expansion_carrier}.
    $\leq[E]=\leq[L]$ by \auto{bounded_expansion_order}.
    $\meet[E]=\meet[L]$ by \auto{bounded_expansion_meet}.
    $\join[E]=\join[L]$ by \auto{bounded_expansion_join}.
    $\latticeBottom[E]=\canonicalLatticeBottom{L}$ by
        \auto{bounded_expansion_bottom}.
    $\latticeTop[E]=\canonicalLatticeTop{L}$ by
        \auto{bounded_expansion_top}.
    $\leq[E]$ is an order on $\carrier[E]$ by
        \obv{ordered_set_order_on_carrier}.
    Hence $E$ is a quasiordered set by
        \auto{quasiordered_set,bounded_expansion,
            bounded_latticeaggregateexact,
            bounded_latticeaggregateprojection_carrier,
            bounded_latticeaggregateprojection_leq}.
    $\leq[E]$ is antisymmetric.
    Hence $E$ is an ordered set by
        \auto{orderedset,bounded_expansion,
            bounded_latticeaggregateexact,
            bounded_latticeaggregateprojection_carrier,
            bounded_latticeaggregateprojection_leq}.
    $L$ is a meet semilattice by \obv{lattice_is_meet_semilattice}.
    $\meet[E]\in\funs{\carrier[E]\times\carrier[E]}{\carrier[E]}$ by
        \obv{meet_graph}.
    For all $x,y\in\carrier[E]$ we have
        $\meet[E](x,y)\mathrel{\leq[E]}x$ and
        $\meet[E](x,y)\mathrel{\leq[E]}y$ by \auto{meet_lb}.
    For all $a,x,y\in\carrier[E]$ such that
        $a\mathrel{\leq[E]}x$ and $a\mathrel{\leq[E]}y$ we have
        $a\mathrel{\leq[E]}\meet[E](x,y)$ by \auto{meet_glb}.
    Hence $E$ is a meet semilattice by
        \auto{meet_semilattice,bounded_expansion,
            bounded_latticeaggregateexact,
            bounded_latticeaggregateprojection_carrier,
            bounded_latticeaggregateprojection_leq,
            bounded_latticeaggregateprojection_meet}.
    $L$ is a join semilattice by \obv{lattice_is_join_semilattice}.
    $\join[E]\in\funs{\carrier[E]\times\carrier[E]}{\carrier[E]}$ by
        \auto{join_graph}.
    For all $x,y\in\carrier[E]$ we have
        $x\mathrel{\leq[E]}\join[E](x,y)$ and
        $y\mathrel{\leq[E]}\join[E](x,y)$ by \auto{join_ub}.
    For all $x,y,a\in\carrier[E]$ such that
        $x\mathrel{\leq[E]}a$ and $y\mathrel{\leq[E]}a$ we have
        $\join[E](x,y)\mathrel{\leq[E]}a$ by \auto{join_lub}.
    Hence $E$ is a join semilattice by
        \auto{join_semilattice,bounded_expansion,
            bounded_latticeaggregateexact,
            bounded_latticeaggregateprojection_carrier,
            bounded_latticeaggregateprojection_leq,
            bounded_latticeaggregateprojection_join}.
    Hence $E$ is a lattice by
        \auto{lattice,bounded_expansion,
            bounded_latticeaggregateexact,
            bounded_latticeaggregateprojection_carrier,
            bounded_latticeaggregateprojection_leq,
            bounded_latticeaggregateprojection_meet,
            bounded_latticeaggregateprojection_join}.
    $\canonicalLatticeBottom{L}\in\carrier[L]$ and
        $\canonicalLatticeTop{L}\in\carrier[L]$ by
        \obv{canonical_lattice_bottom_type,canonical_lattice_top_type}.
    $\leq[L]$ is reflexive on $\carrier[L]$ by
        \obv{ordered_set_order_reflexive}.
    For every $x\in\carrier[L]$ we have
        $\canonicalLatticeBottom{L}\mathrel{\leq[L]}x$ by
        \auto{canonical_lattice_bottom_is_least,least_element,
            reflexive_on}.
    For every $x\in\carrier[L]$ we have
        $x\mathrel{\leq[L]}\canonicalLatticeTop{L}$ by
        \auto{canonical_lattice_top_is_greatest,greatest_element,
            reflexive_on}.
    Hence $E$ is a bounded lattice by
        \auto{bounded_lattice,bounded_expansion,
            bounded_latticeaggregateexact,
            bounded_latticeaggregateprojection_carrier,
            bounded_latticeaggregateprojection_leq,
            bounded_latticeaggregateprojection_meet,
            bounded_latticeaggregateprojection_join,
            bounded_latticeaggregateprojection_latticeBottom,
            bounded_latticeaggregateprojection_latticeTop}.
    Follows by
        \auto{bounded_expansion,bounded_latticeaggregateexact}.
\end{proof}

\begin{theorem}\label{bounded_expansion_finite}
    Let $L$ be a finite lattice.
    Suppose $\carrier[L]$ is inhabited.
    Then $\boundedExpansion{L}$ is a finite bounded lattice.
\end{theorem}
\begin{proof}
    $\boundedExpansion{L}$ is a bounded lattice by
        \auto{bounded_expansion_is_exact_bounded_lattice}.
    Hence $\boundedExpansion{L}$ is a onesorted structure by
        \obv{quasiordered_set}.
    $\carrier[L]$ is a finite set by \obv{finite_structure}.
    $\carrier[\boundedExpansion{L}]=\carrier[L]$ by
        \obv{bounded_expansion_carrier}.
    Follows by \auto{finite_structure}.
\end{proof}

\begin{theorem}\label{bounded_expansion_preserves_distributivity}
    Let $L$ be a finite distributive lattice.
    Suppose $\carrier[L]$ is inhabited.
    Then $\boundedExpansion{L}$ is a bounded distributive lattice.
\end{theorem}
\begin{proof}
    Hence $L$ is a lattice by \obv{distributive_lattice}.
    $\boundedExpansion{L}$ is a bounded lattice by
        \auto{bounded_expansion_is_exact_bounded_lattice}.
    For all $x,y,z\in\carrier[\boundedExpansion{L}]$ we have
        $\meet[\boundedExpansion{L}]
            (x,\join[\boundedExpansion{L}](y,z))
        =\join[\boundedExpansion{L}]
            (\meet[\boundedExpansion{L}](x,y),
             \meet[\boundedExpansion{L}](x,z))$ by
        \auto{bounded_expansion_carrier,bounded_expansion_meet,
            bounded_expansion_join,meet_distributes_over_join}.
    Hence $\boundedExpansion{L}$ is a distributive lattice by definition.
    Follows by definition.
\end{proof}

\begin{theorem}\label{bounded_expansion_lattice_reduct}
    Let $L$ be a finite lattice.
    Suppose $\carrier[L]$ is inhabited.
    Then
        $\reduct[lattice]{\boundedExpansion{L}}
            =\reduct[lattice]{L}$.
\end{theorem}
\begin{proof}
    $\boundedExpansion{L}$ is a bounded lattice by
        \auto{bounded_expansion_is_exact_bounded_lattice}.
    $\reduct[lattice]{\boundedExpansion{L}}$ is an exact lattice by
        \obv{lattice_reduct_is_exact_lattice}.
    $\reduct[lattice]{L}$ is an exact lattice by
        \obv{lattice_reduct_is_exact_lattice}.
    $\carrier[\reduct[lattice]{\boundedExpansion{L}}]
        =\carrier[\reduct[lattice]{L}]$ by
        \auto{latticereductprojection_carrier,bounded_expansion_carrier}.
    $\leq[\reduct[lattice]{\boundedExpansion{L}}]
        =\leq[\reduct[lattice]{L}]$ by
        \auto{latticereductprojection_leq,bounded_expansion_order}.
    $\meet[\reduct[lattice]{\boundedExpansion{L}}]
        =\meet[\reduct[lattice]{L}]$ by
        \auto{latticereductprojection_meet,bounded_expansion_meet}.
    $\join[\reduct[lattice]{\boundedExpansion{L}}]
        =\join[\reduct[lattice]{L}]$ by
        \auto{latticereductprojection_join,bounded_expansion_join}.
    Follows by structure extensionality.
\end{proof}
