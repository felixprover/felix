\section{Maps of Cartesian products}

\import{function.tex}
\import{set/product.tex}
\import{set/logical-graph.tex}

The coordinate projections and the pairing of two maps are ordinary set-coded
graphs. Their domain and inverse-projection laws do not require a topology.

\subsection{Coordinate projections}

\begin{definition}\label{first_product_projection}
    $\firstProductProjection{S}{T}
        =\{(p,\fst{p})\mid p\in S\times T\}$.
\end{definition}

\begin{proposition}\label{first_product_projection_iff}
    $(p,x)\in\firstProductProjection{S}{T}$ iff
        $p\in S\times T$ and $x=\fst{p}$.
\end{proposition}
\begin{proof}
    Follows by \auto{first_product_projection,pair_eq_iff}.
\end{proof}

\begin{proposition}\label{first_product_projection_type}
    $\firstProductProjection{S}{T}\in\funs{S\times T}{S}$.
\end{proposition}
\begin{proof}
    Let $p_1=\firstProductProjection{S}{T}$.
    $p_1$ is a relation by \auto{first_product_projection,relation}.
    $p_1$ is right-unique by
        \auto{first_product_projection_iff,rightunique}.
    $\dom{p_1}=S\times T$ by
        \auto{first_product_projection_iff,dom_iff,setext}.
    Show that $p_1$ is a function to $S$.
    \begin{subproof}
        Show that for every $p\in\dom{p_1}$ we have $p_1(p)\in S$.
        \begin{subproof}
            Fix $p\in\dom{p_1}$.
            Hence $p\in S\times T$.
            $(p,\fst{p})\in p_1$ by
                \obv{first_product_projection_iff}.
            Hence $p_1(p)=\fst{p}$ by \obv{function_apply_intro}.
            $\fst{p}\in S$ by \obv{fst_type}.
        \end{subproof}
    \end{subproof}
    Hence $p_1$ is a function to $S$.
    $S\times T=\dom{p_1}$.
    Hence $p_1\in\funs{S\times T}{S}$ by \obv{funs_intro}.
\end{proof}

\begin{proposition}\label{first_product_projection_apply}
    Let $p\in S\times T$.
    Then $(\firstProductProjection{S}{T})(p)=\fst{p}$.
\end{proposition}
\begin{proof}
    $\firstProductProjection{S}{T}$ is a function by
        \auto{first_product_projection_type,funs_is_function}.
    $(p,\fst{p})\in\firstProductProjection{S}{T}$ by
        \obv{first_product_projection_iff}.
    Follows by \obv{function_apply_intro}.
\end{proof}

\begin{proposition}\label{first_product_projection_preimage}
    Suppose $A\subseteq S$.
    Then $\preimg{\firstProductProjection{S}{T}}{A}=A\times T$.
\end{proposition}
\begin{proof}
    Show that every element of
        $\preimg{\firstProductProjection{S}{T}}{A}$ is an element of
        $A\times T$.
    \begin{subproof}
        Fix $p\in\preimg{\firstProductProjection{S}{T}}{A}$.
        Take $x\in A$ such that
            $(p,x)\in\firstProductProjection{S}{T}$ by \obv{preimg_iff}.
        $(p,x)\in\firstProductProjection{S}{T}$.
        $(p,x)\in\firstProductProjection{S}{T}$ iff
            $p\in S\times T$ and $x=\fst{p}$ by
            \obv{first_product_projection_iff}.
        Hence $p\in S\times T$ and $x=\fst{p}$.
        Take $s,t$ such that $s\in S$ and $t\in T$ and $p=(s,t)$ by
            \obv{times_elem_is_tuple}.
        $\fst{p}=s$ by \auto{fst_eq}.
        Hence $s\in A$.
        Follows by \auto{times_tuple_intro}.
    \end{subproof}
    Show that every element of $A\times T$ is an element of
        $\preimg{\firstProductProjection{S}{T}}{A}$.
    \begin{subproof}
        Fix $p\in A\times T$.
        Take $a,t$ such that $a\in A$ and $t\in T$ and $p=(a,t)$ by
            \obv{times_elem_is_tuple}.
        Hence $a\in S$ by \obv{subseteq}.
        Hence $p\in S\times T$ by \auto{times_tuple_intro}.
        $\fst{p}=a$ by \auto{fst_eq}.
        Hence $\fst{p}\in A$.
        $(p,\fst{p})\in\firstProductProjection{S}{T}$ by
            \obv{first_product_projection_iff}.
        Follows by \auto{preimg_iff}.
    \end{subproof}
    Follows by set extensionality.
\end{proof}

\begin{definition}\label{second_product_projection}
    $\secondProductProjection{S}{T}
        =\{(p,\snd{p})\mid p\in S\times T\}$.
\end{definition}

\begin{proposition}\label{second_product_projection_iff}
    $(p,y)\in\secondProductProjection{S}{T}$ iff
        $p\in S\times T$ and $y=\snd{p}$.
\end{proposition}
\begin{proof}
    Follows by \auto{second_product_projection,pair_eq_iff}.
\end{proof}

\begin{proposition}\label{second_product_projection_type}
    $\secondProductProjection{S}{T}\in\funs{S\times T}{T}$.
\end{proposition}
\begin{proof}
    Let $p_2=\secondProductProjection{S}{T}$.
    $p_2$ is a relation by \auto{second_product_projection,relation}.
    $p_2$ is right-unique by
        \auto{second_product_projection_iff,rightunique}.
    $\dom{p_2}=S\times T$ by
        \auto{second_product_projection_iff,dom_iff,setext}.
    Show that $p_2$ is a function to $T$.
    \begin{subproof}
        Show that for every $p\in\dom{p_2}$ we have $p_2(p)\in T$.
        \begin{subproof}
            Fix $p\in\dom{p_2}$.
            Hence $p\in S\times T$.
            $(p,\snd{p})\in p_2$ by
                \obv{second_product_projection_iff}.
            Hence $p_2(p)=\snd{p}$ by \obv{function_apply_intro}.
            $\snd{p}\in T$ by \obv{snd_type}.
        \end{subproof}
    \end{subproof}
    Hence $p_2$ is a function to $T$.
    $S\times T=\dom{p_2}$.
    Hence $p_2\in\funs{S\times T}{T}$ by \obv{funs_intro}.
\end{proof}

\begin{proposition}\label{second_product_projection_apply}
    Let $p\in S\times T$.
    Then $(\secondProductProjection{S}{T})(p)=\snd{p}$.
\end{proposition}
\begin{proof}
    $\secondProductProjection{S}{T}$ is a function by
        \auto{second_product_projection_type,funs_is_function}.
    $(p,\snd{p})\in\secondProductProjection{S}{T}$ by
        \obv{second_product_projection_iff}.
    Follows by \obv{function_apply_intro}.
\end{proof}

\begin{proposition}\label{second_product_projection_preimage}
    Suppose $B\subseteq T$.
    Then $\preimg{\secondProductProjection{S}{T}}{B}=S\times B$.
\end{proposition}
\begin{proof}
    Show that every element of
        $\preimg{\secondProductProjection{S}{T}}{B}$ is an element of
        $S\times B$.
    \begin{subproof}
        Fix $p\in\preimg{\secondProductProjection{S}{T}}{B}$.
        Take $y\in B$ such that
            $(p,y)\in\secondProductProjection{S}{T}$ by \obv{preimg_iff}.
        $(p,y)\in\secondProductProjection{S}{T}$.
        $(p,y)\in\secondProductProjection{S}{T}$ iff
            $p\in S\times T$ and $y=\snd{p}$ by
            \obv{second_product_projection_iff}.
        Hence $p\in S\times T$ and $y=\snd{p}$.
        Take $s,t$ such that $s\in S$ and $t\in T$ and $p=(s,t)$ by
            \obv{times_elem_is_tuple}.
        $\snd{p}=t$ by \auto{snd_eq}.
        Hence $t\in B$.
        Follows by \auto{times_tuple_intro}.
    \end{subproof}
    Show that every element of $S\times B$ is an element of
        $\preimg{\secondProductProjection{S}{T}}{B}$.
    \begin{subproof}
        Fix $p\in S\times B$.
        Take $s,b$ such that $s\in S$ and $b\in B$ and $p=(s,b)$ by
            \obv{times_elem_is_tuple}.
        Hence $b\in T$ by \obv{subseteq}.
        Hence $p\in S\times T$ by \auto{times_tuple_intro}.
        $\snd{p}=b$ by \auto{snd_eq}.
        Hence $\snd{p}\in B$.
        $(p,\snd{p})\in\secondProductProjection{S}{T}$ by
            \obv{second_product_projection_iff}.
        Follows by \auto{preimg_iff}.
    \end{subproof}
    Follows by set extensionality.
\end{proof}

\subsection{Pairing maps}

\begin{definition}\label{product_pairing}
    $\productPairing{R}{f}{g}
        =\Apply{\graphOnDomainOp}{
            R,\fun{r:\Set}{(\apply{f}{r},\apply{g}{r})}}$.
\end{definition}

\begin{proposition}\label{product_pairing_type}
    Let $f\in\funs{R}{S}$ and $g\in\funs{R}{T}$.
    Then $\productPairing{R}{f}{g}\in\funs{R}{S\times T}$.
\end{proposition}
\begin{proof}
    Show that for every $r\in R$ we have $(f(r),g(r))\in S\times T$.
    \begin{subproof}
        Fix $r\in R$.
        $f(r)\in S$ and $g(r)\in T$ by \obv{funs_type_apply}.
        Follows by \obv{times_tuple_intro}.
    \end{subproof}
    Follows by
        \auto{product_pairing,logical_graph_on_domain_type}.
\end{proof}

\begin{proposition}\label{product_pairing_apply}
    Let $r\in R$.
    Then $(\productPairing{R}{f}{g})(r)=(f(r),g(r))$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{product_pairing,logical_graph_on_domain_apply}.
\end{proof}

\begin{theorem}\label{first_projection_of_product_pairing}
    Let $f\in\funs{R}{S}$ and $g\in\funs{R}{T}$.
    Then $\firstProductProjection{S}{T}\circ
        \productPairing{R}{f}{g}=f$.
\end{theorem}
\begin{proof}
    $\productPairing{R}{f}{g}\in\funs{R}{S\times T}$ by
        \obv{product_pairing_type}.
    $\firstProductProjection{S}{T}\in\funs{S\times T}{S}$ by
        \obv{first_product_projection_type}.
    Hence $\firstProductProjection{S}{T}\circ
        \productPairing{R}{f}{g}\in\funs{R}{S}$ by
        \obv{funs_circ}.
    Show that for every $r\in R$ we have
        $(\firstProductProjection{S}{T}\circ
            \productPairing{R}{f}{g})(r)=f(r)$.
    \begin{subproof}
        Fix $r\in R$.
        $(\productPairing{R}{f}{g})(r)=(f(r),g(r))$ by
            \obv{product_pairing_apply}.
        $(f(r),g(r))\in S\times T$ by
            \obv{funs_type_apply,times_tuple_intro}.
        $(r,(\productPairing{R}{f}{g})(r))
            \in\productPairing{R}{f}{g}$ by
            \obv{function_apply_elim,funs_is_function,funs}.
        $((f(r),g(r)),f(r))\in
            \firstProductProjection{S}{T}$ by
            \obv{first_product_projection_iff,fst_eq}.
        Hence $((\productPairing{R}{f}{g})(r),f(r))\in
            \firstProductProjection{S}{T}$.
        Thus $(r,f(r))\in\firstProductProjection{S}{T}\circ
            \productPairing{R}{f}{g}$ by \auto{circ_iff}.
        $\firstProductProjection{S}{T}\circ
            \productPairing{R}{f}{g}$ is a function by
            \obv{funs_is_function}.
        Follows by \obv{function_apply_intro}.
    \end{subproof}
    $\dom{\firstProductProjection{S}{T}\circ
        \productPairing{R}{f}{g}}=R$ by \obv{funs}.
    $\dom{f}=R$ by \obv{funs}.
    Hence $\dom{\firstProductProjection{S}{T}\circ
        \productPairing{R}{f}{g}}=\dom{f}$.
    For every $r\in\dom{\firstProductProjection{S}{T}\circ
        \productPairing{R}{f}{g}}$ we have
        $(\firstProductProjection{S}{T}\circ
            \productPairing{R}{f}{g})(r)=f(r)$.
    Follows by \obv{funext_on_domain,funs_is_function}.
\end{proof}


\begin{theorem}\label{second_projection_of_product_pairing}
    Let $f\in\funs{R}{S}$ and $g\in\funs{R}{T}$.
    Then $\secondProductProjection{S}{T}\circ

            \productPairing{R}{f}{g})(r)=g(r)$.

        $(r,(\productPairing{R}{f}{g})(r))

        $((f(r),g(r)),g(r))\in

        Hence $((\productPairing{R}{f}{g})(r),g(r))\in

\end{proof}

\subsection{Products of maps}

\begin{definition}\label{product_map}
    $\productMap{S}{T}{f}{g}
        =\productPairing{S\times T}{
            f\circ\firstProductProjection{S}{T}}{
            g\circ\secondProductProjection{S}{T}}$.
\end{definition}

\begin{proposition}\label{product_map_type}
    Let $f\in\funs{S}{U}$ and $g\in\funs{T}{V}$.
    Then $\productMap{S}{T}{f}{g}\in
        \funs{S\times T}{U\times V}$.
\end{proposition}
\begin{proof}
    $\firstProductProjection{S}{T}\in\funs{S\times T}{S}$ and
        $\secondProductProjection{S}{T}\in\funs{S\times T}{T}$ by
        \obv{first_product_projection_type,
            second_product_projection_type}.
    Hence $f\circ\firstProductProjection{S}{T}
        \in\funs{S\times T}{U}$ and
        $g\circ\secondProductProjection{S}{T}
        \in\funs{S\times T}{V}$ by \obv{funs_circ}.
    Follows by \auto{product_map,product_pairing_type}.
\end{proof}

\begin{proposition}\label{product_map_apply}
    Let $f\in\funs{S}{U}$ and $g\in\funs{T}{V}$.
    Let $p\in S\times T$.
    Then $(\productMap{S}{T}{f}{g})(p)
        =(f(\fst{p}),g(\snd{p}))$.
\end{proposition}
\begin{proof}
    $\firstProductProjection{S}{T}\in\funs{S\times T}{S}$ and
        $\secondProductProjection{S}{T}\in\funs{S\times T}{T}$ by
        \obv{first_product_projection_type,
            second_product_projection_type}.
    Hence $f\circ\firstProductProjection{S}{T}
        \in\funs{S\times T}{U}$ and
        $g\circ\secondProductProjection{S}{T}
        \in\funs{S\times T}{V}$ by \obv{funs_circ}.
    $\firstProductProjection{S}{T}$ is a function by
        \obv{funs_is_function}.
    $f$ is a function by \obv{funs_is_function}.
    $f$ is composable with $\firstProductProjection{S}{T}$ by
        \auto{funs_ran,funs}.
    $p\in\dom{\firstProductProjection{S}{T}}$ by \auto{funs}.
    $(f\circ\firstProductProjection{S}{T})(p)
        =f((\firstProductProjection{S}{T})(p))$ by
        \obv{circ_apply}.
    $\secondProductProjection{S}{T}$ is a function by
        \obv{funs_is_function}.
    $g$ is a function by \obv{funs_is_function}.
    $g$ is composable with $\secondProductProjection{S}{T}$ by
        \auto{funs_ran,funs}.
    $p\in\dom{\secondProductProjection{S}{T}}$ by \auto{funs}.
    $(g\circ\secondProductProjection{S}{T})(p)
        =g((\secondProductProjection{S}{T})(p))$ by
        \obv{circ_apply}.
    $(\firstProductProjection{S}{T})(p)=\fst{p}$ and
        $(\secondProductProjection{S}{T})(p)=\snd{p}$ by
        \obv{first_product_projection_apply,
            second_product_projection_apply}.
    Follows by
        \auto{product_map,product_pairing_apply}.
\end{proof}

\begin{theorem}\label{identity_product_map}
    $\productMap{S}{T}{\identity{S}}{\identity{T}}
        =\identity{S\times T}$.
\end{theorem}
\begin{proof}
    $\identity{S}\in\funs{S}{S}$ and
        $\identity{T}\in\funs{T}{T}$ by \obv{id_elem_funs}.
    Hence $\productMap{S}{T}{\identity{S}}{\identity{T}}
        \in\funs{S\times T}{S\times T}$ by \obv{product_map_type}.
    $\identity{S\times T}\in\funs{S\times T}{S\times T}$ by
        \obv{id_elem_funs}.
    Show that for every $p\in S\times T$ we have
        $(\productMap{S}{T}{\identity{S}}{\identity{T}})(p)
        =(\identity{S\times T})(p)$.
    \begin{subproof}
        Fix $p\in S\times T$.
        $(\productMap{S}{T}{\identity{S}}{\identity{T}})(p)
            =((\identity{S})(\fst{p}),(\identity{T})(\snd{p}))$ by
            \obv{product_map_apply}.
        $\fst{p}\in S$ and $\snd{p}\in T$ by \obv{fst_type,snd_type}.
        Hence $(\identity{S})(\fst{p})=\fst{p}$ and
            $(\identity{T})(\snd{p})=\snd{p}$ by \obv{id_apply}.
        $(\fst{p},\snd{p})=p$ by
            \auto{times_elem_is_tuple,pair_eq_pair_of_proj}.
        $(\identity{S\times T})(p)=p$ by \obv{id_apply}.
    \end{subproof}
    Follows by \obv{typed_functions_equal_from_agreement}.
\end{proof}

\begin{theorem}\label{composition_product_map}
    Let $f\in\funs{S}{U}$ and $g\in\funs{T}{V}$.
    Let $h\in\funs{U}{W}$ and $k\in\funs{V}{Z}$.
    Then
        $\productMap{U}{V}{h}{k}\circ\productMap{S}{T}{f}{g}
        =\productMap{S}{T}{h\circ f}{k\circ g}$.
\end{theorem}
\begin{proof}
    Let $P=\productMap{S}{T}{f}{g}$ and
        $Q=\productMap{U}{V}{h}{k}$.
    Let $R=\productMap{S}{T}{h\circ f}{k\circ g}$.
    $\productMap{S}{T}{f}{g}\in\funs{S\times T}{U\times V}$ by
        \obv{product_map_type}.
    Hence $P\in\funs{S\times T}{U\times V}$.
    $\productMap{U}{V}{h}{k}\in\funs{U\times V}{W\times Z}$ by
        \obv{product_map_type}.
    Hence $Q\in\funs{U\times V}{W\times Z}$.
    Hence $Q\circ P\in\funs{S\times T}{W\times Z}$ by
        \obv{funs_circ}.
    $h\circ f\in\funs{S}{W}$ and
        $k\circ g\in\funs{T}{Z}$ by \obv{funs_circ}.
    Hence $\productMap{S}{T}{h\circ f}{k\circ g}
        \in\funs{S\times T}{W\times Z}$ by \obv{product_map_type}.
    Hence $R\in\funs{S\times T}{W\times Z}$.
    Show that for every $p\in S\times T$ we have
        $(Q\circ P)(p)=R(p)$.
    \begin{subproof}
        Fix $p\in S\times T$.
        $P(p)=(f(\fst{p}),g(\snd{p}))$ by
            \auto{product_map_apply}.
        $f(\fst{p})\in U$ and $g(\snd{p})\in V$ by
            \obv{fst_type,snd_type,funs_type_apply}.
        Hence $P(p)\in U\times V$ by \auto{times_tuple_intro}.
        $(Q\circ P)(p)=Q(P(p))$ by
            \auto{circ_apply,funs_is_function,funs_ran,funs}.
        $Q(P(p))=(h(f(\fst{p})),k(g(\snd{p})))$ by
            \auto{product_map_apply,fst_eq,snd_eq}.
        $(h\circ f)(\fst{p})=h(f(\fst{p}))$ and
            $(k\circ g)(\snd{p})=k(g(\snd{p}))$ by
            \auto{circ_apply,funs_is_function,funs_ran,funs,
                fst_type,snd_type}.
        $(\productMap{S}{T}{h\circ f}{k\circ g})(p)
            =((h\circ f)(\fst{p}),(k\circ g)(\snd{p}))$ by
            \obv{product_map_apply}.
        Hence $R(p)=((h\circ f)(\fst{p}),(k\circ g)(\snd{p}))$.
    \end{subproof}
    Hence $Q\circ P=R$ by
        \obv{typed_functions_equal_from_agreement}.
    $\productMap{U}{V}{h}{k}\circ\productMap{S}{T}{f}{g}
        =Q\circ P$.
    $R=\productMap{S}{T}{h\circ f}{k\circ g}$.
\end{proof}

\begin{theorem}\label{first_projection_of_product_map}
    Let $f\in\funs{S}{U}$ and $g\in\funs{T}{V}$.
    Then $\firstProductProjection{U}{V}\circ
        \productMap{S}{T}{f}{g}
        =f\circ\firstProductProjection{S}{T}$.
\end{theorem}
\begin{proof}
    $\firstProductProjection{S}{T}\in\funs{S\times T}{S}$ and
        $\secondProductProjection{S}{T}\in\funs{S\times T}{T}$ by
        \obv{first_product_projection_type,
            second_product_projection_type}.
    Hence $f\circ\firstProductProjection{S}{T}
        \in\funs{S\times T}{U}$ and
        $g\circ\secondProductProjection{S}{T}
        \in\funs{S\times T}{V}$ by \obv{funs_circ}.
    $\firstProductProjection{U}{V}\circ
        \productPairing{S\times T}{
            f\circ\firstProductProjection{S}{T}}{
            g\circ\secondProductProjection{S}{T}}
        =f\circ\firstProductProjection{S}{T}$ by
        \obv{first_projection_of_product_pairing}.
    Follows by \auto{product_map}.
\end{proof}

\begin{theorem}\label{second_projection_of_product_map}
    Let $f\in\funs{S}{U}$ and $g\in\funs{T}{V}$.
    Then $\secondProductProjection{U}{V}\circ
        \productMap{S}{T}{f}{g}
        =g\circ\secondProductProjection{S}{T}$.
\end{theorem}
\begin{proof}
    $\firstProductProjection{S}{T}\in\funs{S\times T}{S}$ and
        $\secondProductProjection{S}{T}\in\funs{S\times T}{T}$ by
        \obv{first_product_projection_type,
            second_product_projection_type}.
    Hence $f\circ\firstProductProjection{S}{T}
        \in\funs{S\times T}{U}$ and
        $g\circ\secondProductProjection{S}{T}
        \in\funs{S\times T}{V}$ by \obv{funs_circ}.
    $\secondProductProjection{U}{V}\circ
        \productPairing{S\times T}{
            f\circ\firstProductProjection{S}{T}}{
            g\circ\secondProductProjection{S}{T}}
        =g\circ\secondProductProjection{S}{T}$ by
        \obv{second_projection_of_product_pairing}.
    Follows by \auto{product_map}.
\end{proof}

\begin{theorem}\label{product_map_bijection}
    Let $f\in\bijections{S}{U}$ and $g\in\bijections{T}{V}$.
    Then $\productMap{S}{T}{f}{g}\in\bijections{S\times T}{U\times V}$.
\end{theorem}
\begin{proof}
    Let $P=\productMap{S}{T}{f}{g}$ and
        $Q=\productMap{U}{V}{\converse{f}}{\converse{g}}$.
    $f\in\funs{S}{U}$ and $g\in\funs{T}{V}$ by \obv{bijections_to_funs}.
    $\converse{f}\in\funs{U}{S}$ and $\converse{g}\in\funs{V}{T}$ by
        \obv{bijection_converse_is_bijection,bijections_to_funs}.
    $P\in\funs{S\times T}{U\times V}$ and $Q\in\funs{U\times V}{S\times T}$ by
        \auto{product_map_type}.
    $Q\circ P=\identity{S\times T}$ by
        \auto{composition_product_map,bijection_converse_circ,identity_product_map}.
    For every $p\in S\times T$ we have $Q(P(p))=p$ by
        \auto{circ_apply,id_apply,funs_is_function,funs,funs_ran}.
    Hence $Q$ is a left inverse of $P$ by \obv{leftinverse_from_to_iff}.
    Hence $P$ is an injection by \obv{leftinverse_implies_injection}.
    $P\circ Q=\identity{U\times V}$ by
        \auto{composition_product_map,bijection_circ_converse,identity_product_map}.
    For every $q\in U\times V$ we have $P(Q(q))=q$ by
        \auto{circ_apply,id_apply,funs_is_function,funs,funs_ran}.
    Hence $Q$ is a right inverse of $P$ by \obv{rightinverse_from_to_iff}.
    Hence $P\in\surj{S\times T}{U\times V}$ by \obv{rightinverse_implies_surjection}.
    Follows by \auto{bijections}.
\end{proof}

\begin{theorem}\label{product_map_converse}
    Let $f\in\bijections{S}{U}$ and $g\in\bijections{T}{V}$.
    Then $\converse{\productMap{S}{T}{f}{g}}
        =\productMap{U}{V}{\converse{f}}{\converse{g}}$.
\end{theorem}
\begin{proof}
    Let $P=\productMap{S}{T}{f}{g}$ and
        $Q=\productMap{U}{V}{\converse{f}}{\converse{g}}$.
    $P\in\bijections{S\times T}{U\times V}$ by \auto{product_map_bijection}.
    Hence $P$ is an injection by \auto{bijections,surj,funs_is_function}.
    Hence $\converse{P}$ is a left inverse of $P$ by
        \obv{injective_converse_is_leftinverse}.
    $\converse{P}\in\funs{U\times V}{S\times T}$ by
        \obv{bijection_converse_is_bijection,bijections_to_funs}.
    $f\in\funs{S}{U}$ and $g\in\funs{T}{V}$ by \obv{bijections_to_funs}.
    $\converse{f}\in\funs{U}{S}$ and $\converse{g}\in\funs{V}{T}$ by
        \obv{bijection_converse_is_bijection,bijections_to_funs}.
    $Q\in\funs{U\times V}{S\times T}$ by \auto{product_map_type}.
    $P\circ Q=\identity{U\times V}$ by
        \auto{composition_product_map,bijection_circ_converse,identity_product_map}.
    Show that for every $q\in U\times V$ we have $(\converse{P})(q)=Q(q)$.
    \begin{subproof}
        Fix $q\in U\times V$.
        $Q(q)\in S\times T$ by \obv{funs_type_apply}.
        $P(Q(q))=q$ by
            \auto{circ_apply,id_apply,bijections_to_funs,funs_is_function,funs,funs_ran}.
        Follows by \auto{leftinverse,bijections_dom}.
    \end{subproof}
    Follows by \auto{typed_functions_equal_from_agreement}.
\end{proof}
