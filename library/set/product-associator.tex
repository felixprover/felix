\section{Associativity of Cartesian products}

\import{set/product-map.tex}

Different parenthesizations have distinct Cartesian carriers. Projections
and pairing give the canonical mutually inverse reassociation maps.

\begin{definition}\label{product_associator}
    $\productAssociator{S}{T}{U}
        =\productPairing{(S\times T)\times U}{
            \firstProductProjection{S}{T}\circ
                \firstProductProjection{S\times T}{U}}{
            \productPairing{(S\times T)\times U}{
                \secondProductProjection{S}{T}\circ
                    \firstProductProjection{S\times T}{U}}{
                \secondProductProjection{S\times T}{U}}}$.
\end{definition}

\begin{definition}\label{inverse_product_associator}
    $\inverseProductAssociator{S}{T}{U}
        =\productPairing{S\times(T\times U)}{
            \productPairing{S\times(T\times U)}{
                \firstProductProjection{S}{T\times U}}{
                \firstProductProjection{T}{U}\circ
                    \secondProductProjection{S}{T\times U}}}{
            \secondProductProjection{T}{U}\circ
                \secondProductProjection{S}{T\times U}}$.
\end{definition}

\begin{proposition}\label{product_associator_type}
    $\productAssociator{S}{T}{U}\in
        \funs{(S\times T)\times U}{S\times(T\times U)}$.
\end{proposition}
\begin{proof}
    $\firstProductProjection{S\times T}{U}\in
        \funs{(S\times T)\times U}{S\times T}$ by
        \obv{first_product_projection_type}.
    $\firstProductProjection{S}{T}\in\funs{S\times T}{S}$ and
        $\secondProductProjection{S}{T}\in\funs{S\times T}{T}$ by
        \obv{first_product_projection_type,
            second_product_projection_type}.
    Hence $\firstProductProjection{S}{T}\circ
        \firstProductProjection{S\times T}{U}
        \in\funs{(S\times T)\times U}{S}$ and
        $\secondProductProjection{S}{T}\circ
        \firstProductProjection{S\times T}{U}
        \in\funs{(S\times T)\times U}{T}$ by \obv{funs_circ}.
    $\secondProductProjection{S\times T}{U}\in
        \funs{(S\times T)\times U}{U}$ by
        \obv{second_product_projection_type}.
    Hence $\productPairing{(S\times T)\times U}{
        \secondProductProjection{S}{T}\circ
            \firstProductProjection{S\times T}{U}}{
        \secondProductProjection{S\times T}{U}}
        \in\funs{(S\times T)\times U}{T\times U}$ by
        \obv{product_pairing_type}.
    Follows by \auto{product_associator,product_pairing_type}.
\end{proof}

\begin{proposition}\label{inverse_product_associator_type}
    $\inverseProductAssociator{S}{T}{U}\in
        \funs{S\times(T\times U)}{(S\times T)\times U}$.
\end{proposition}
\begin{proof}
    $\firstProductProjection{S}{T\times U}\in
        \funs{S\times(T\times U)}{S}$ and
        $\secondProductProjection{S}{T\times U}\in
        \funs{S\times(T\times U)}{T\times U}$ by
        \obv{first_product_projection_type,
            second_product_projection_type}.
    $\firstProductProjection{T}{U}\in\funs{T\times U}{T}$ and
        $\secondProductProjection{T}{U}\in\funs{T\times U}{U}$ by
        \obv{first_product_projection_type,
            second_product_projection_type}.
    Hence $\firstProductProjection{T}{U}\circ
        \secondProductProjection{S}{T\times U}
        \in\funs{S\times(T\times U)}{T}$ and
        $\secondProductProjection{T}{U}\circ
        \secondProductProjection{S}{T\times U}
        \in\funs{S\times(T\times U)}{U}$ by \obv{funs_circ}.
    Hence $\productPairing{S\times(T\times U)}{
        \firstProductProjection{S}{T\times U}}{
        \firstProductProjection{T}{U}\circ
            \secondProductProjection{S}{T\times U}}
        \in\funs{S\times(T\times U)}{S\times T}$ by
        \obv{product_pairing_type}.
    Follows by
        \auto{inverse_product_associator,product_pairing_type}.
\end{proof}

\begin{theorem}\label{first_projection_of_product_associator}
    $\firstProductProjection{S}{T\times U}\circ
        \productAssociator{S}{T}{U}
        =\firstProductProjection{S}{T}\circ
            \firstProductProjection{S\times T}{U}$.
\end{theorem}
\begin{proof}
    $\firstProductProjection{S\times T}{U}\in
        \funs{(S\times T)\times U}{S\times T}$ by
        \obv{first_product_projection_type}.
    $\firstProductProjection{S}{T}\in\funs{S\times T}{S}$ and
        $\secondProductProjection{S}{T}\in\funs{S\times T}{T}$ by
        \obv{first_product_projection_type,
            second_product_projection_type}.
    Hence $\firstProductProjection{S}{T}\circ
        \firstProductProjection{S\times T}{U}
        \in\funs{(S\times T)\times U}{S}$ and
        $\secondProductProjection{S}{T}\circ
        \firstProductProjection{S\times T}{U}
        \in\funs{(S\times T)\times U}{T}$ by \obv{funs_circ}.
    $\secondProductProjection{S\times T}{U}\in
        \funs{(S\times T)\times U}{U}$ by
        \obv{second_product_projection_type}.
    Hence $\productPairing{(S\times T)\times U}{
        \secondProductProjection{S}{T}\circ
            \firstProductProjection{S\times T}{U}}{
        \secondProductProjection{S\times T}{U}}
        \in\funs{(S\times T)\times U}{T\times U}$ by
        \obv{product_pairing_type}.
    Hence $\firstProductProjection{S}{T\times U}\circ
        \productPairing{(S\times T)\times U}{
            \firstProductProjection{S}{T}\circ
                \firstProductProjection{S\times T}{U}}{
            \productPairing{(S\times T)\times U}{
                \secondProductProjection{S}{T}\circ
                    \firstProductProjection{S\times T}{U}}{
                \secondProductProjection{S\times T}{U}}}
        =\firstProductProjection{S}{T}\circ
            \firstProductProjection{S\times T}{U}$ by
        \obv{first_projection_of_product_pairing}.
    Follows by \auto{product_associator}.
\end{proof}

\begin{theorem}\label{second_projection_of_product_associator}
    $\secondProductProjection{S}{T\times U}\circ
        \productAssociator{S}{T}{U}
        =\productPairing{(S\times T)\times U}{
            \secondProductProjection{S}{T}\circ
                \firstProductProjection{S\times T}{U}}{
            \secondProductProjection{S\times T}{U}}$.
\end{theorem}
\begin{proof}
    $\firstProductProjection{S\times T}{U}\in
        \funs{(S\times T)\times U}{S\times T}$ by
        \obv{first_product_projection_type}.
    $\firstProductProjection{S}{T}\in\funs{S\times T}{S}$ and
        $\secondProductProjection{S}{T}\in\funs{S\times T}{T}$ by
        \obv{first_product_projection_type,
            second_product_projection_type}.
    Hence $\firstProductProjection{S}{T}\circ
        \firstProductProjection{S\times T}{U}
        \in\funs{(S\times T)\times U}{S}$ and
        $\secondProductProjection{S}{T}\circ
        \firstProductProjection{S\times T}{U}
        \in\funs{(S\times T)\times U}{T}$ by \obv{funs_circ}.
    $\secondProductProjection{S\times T}{U}\in
        \funs{(S\times T)\times U}{U}$ by
        \obv{second_product_projection_type}.
    Hence $\productPairing{(S\times T)\times U}{
        \secondProductProjection{S}{T}\circ
            \firstProductProjection{S\times T}{U}}{
        \secondProductProjection{S\times T}{U}}
        \in\funs{(S\times T)\times U}{T\times U}$ by
        \obv{product_pairing_type}.
    Hence $\secondProductProjection{S}{T\times U}\circ
        \productPairing{(S\times T)\times U}{
            \firstProductProjection{S}{T}\circ
                \firstProductProjection{S\times T}{U}}{
            \productPairing{(S\times T)\times U}{
                \secondProductProjection{S}{T}\circ
                    \firstProductProjection{S\times T}{U}}{
                \secondProductProjection{S\times T}{U}}}
        =\productPairing{(S\times T)\times U}{
            \secondProductProjection{S}{T}\circ
                \firstProductProjection{S\times T}{U}}{
            \secondProductProjection{S\times T}{U}}$ by
        \obv{second_projection_of_product_pairing}.
    Follows by \auto{product_associator}.
\end{proof}

\begin{theorem}\label{first_projection_of_inverse_product_associator}
    $\firstProductProjection{S\times T}{U}\circ
        \inverseProductAssociator{S}{T}{U}
        =\productPairing{S\times(T\times U)}{
            \firstProductProjection{S}{T\times U}}{
            \firstProductProjection{T}{U}\circ
                \secondProductProjection{S}{T\times U}}$.
\end{theorem}
\begin{proof}
    $\firstProductProjection{S}{T\times U}\in
        \funs{S\times(T\times U)}{S}$ and
        $\secondProductProjection{S}{T\times U}\in
        \funs{S\times(T\times U)}{T\times U}$ by
        \obv{first_product_projection_type,
            second_product_projection_type}.
    $\firstProductProjection{T}{U}\in\funs{T\times U}{T}$ and
        $\secondProductProjection{T}{U}\in\funs{T\times U}{U}$ by
        \obv{first_product_projection_type,
            second_product_projection_type}.
    Hence $\firstProductProjection{T}{U}\circ
        \secondProductProjection{S}{T\times U}
        \in\funs{S\times(T\times U)}{T}$ and
        $\secondProductProjection{T}{U}\circ
        \secondProductProjection{S}{T\times U}
        \in\funs{S\times(T\times U)}{U}$ by \obv{funs_circ}.
    Hence $\productPairing{S\times(T\times U)}{
        \firstProductProjection{S}{T\times U}}{
        \firstProductProjection{T}{U}\circ
            \secondProductProjection{S}{T\times U}}
        \in\funs{S\times(T\times U)}{S\times T}$ by
        \obv{product_pairing_type}.
    Hence $\firstProductProjection{S\times T}{U}\circ
        \productPairing{S\times(T\times U)}{
            \productPairing{S\times(T\times U)}{
                \firstProductProjection{S}{T\times U}}{
                \firstProductProjection{T}{U}\circ
                    \secondProductProjection{S}{T\times U}}}{
            \secondProductProjection{T}{U}\circ
                \secondProductProjection{S}{T\times U}}
        =\productPairing{S\times(T\times U)}{
            \firstProductProjection{S}{T\times U}}{
            \firstProductProjection{T}{U}\circ
                \secondProductProjection{S}{T\times U}}$ by
        \obv{first_projection_of_product_pairing}.
    Follows by \auto{inverse_product_associator}.
\end{proof}

\begin{theorem}\label{second_projection_of_inverse_product_associator}
    $\secondProductProjection{S\times T}{U}\circ
        \inverseProductAssociator{S}{T}{U}
        =\secondProductProjection{T}{U}\circ
            \secondProductProjection{S}{T\times U}$.
\end{theorem}
\begin{proof}
    $\firstProductProjection{S}{T\times U}\in
        \funs{S\times(T\times U)}{S}$ and
        $\secondProductProjection{S}{T\times U}\in
        \funs{S\times(T\times U)}{T\times U}$ by
        \obv{first_product_projection_type,
            second_product_projection_type}.
    $\firstProductProjection{T}{U}\in\funs{T\times U}{T}$ and
        $\secondProductProjection{T}{U}\in\funs{T\times U}{U}$ by
        \obv{first_product_projection_type,
            second_product_projection_type}.
    Hence $\firstProductProjection{T}{U}\circ
        \secondProductProjection{S}{T\times U}
        \in\funs{S\times(T\times U)}{T}$ and
        $\secondProductProjection{T}{U}\circ
        \secondProductProjection{S}{T\times U}
        \in\funs{S\times(T\times U)}{U}$ by \obv{funs_circ}.
    Hence $\productPairing{S\times(T\times U)}{
        \firstProductProjection{S}{T\times U}}{
        \firstProductProjection{T}{U}\circ
            \secondProductProjection{S}{T\times U}}
        \in\funs{S\times(T\times U)}{S\times T}$ by
        \obv{product_pairing_type}.
    Hence $\secondProductProjection{S\times T}{U}\circ
        \productPairing{S\times(T\times U)}{
            \productPairing{S\times(T\times U)}{
                \firstProductProjection{S}{T\times U}}{
                \firstProductProjection{T}{U}\circ
                    \secondProductProjection{S}{T\times U}}}{
            \secondProductProjection{T}{U}\circ
                \secondProductProjection{S}{T\times U}}
        =\secondProductProjection{T}{U}\circ
            \secondProductProjection{S}{T\times U}$ by
        \obv{second_projection_of_product_pairing}.
    Follows by \auto{inverse_product_associator}.
\end{proof}

\begin{proposition}\label{product_associator_apply}
    Let $p\in(S\times T)\times U$.
    Then $(\productAssociator{S}{T}{U})(p)
        =(\fst{\fst{p}},(\snd{\fst{p}},\snd{p}))$.
\end{proposition}
\begin{proof}
    $\firstProductProjection{S\times T}{U}$ is a function and
        $\firstProductProjection{S}{T}$ is a function and
        $\secondProductProjection{S}{T}$ is a function by
        \auto{first_product_projection_type,
            second_product_projection_type,funs_is_function}.
    $\firstProductProjection{S}{T}$ is composable with
        $\firstProductProjection{S\times T}{U}$ and
        $\secondProductProjection{S}{T}$ is composable with
        $\firstProductProjection{S\times T}{U}$ by
        \auto{first_product_projection_type,
            second_product_projection_type,funs_ran,funs}.
    $p\in\dom{\firstProductProjection{S\times T}{U}}$ by
        \auto{first_product_projection_type,funs}.
    Hence $(\firstProductProjection{S}{T}\circ
        \firstProductProjection{S\times T}{U})(p)
        =(\firstProductProjection{S}{T})
            ((\firstProductProjection{S\times T}{U})(p))$ and
        $(\secondProductProjection{S}{T}\circ
        \firstProductProjection{S\times T}{U})(p)
        =(\secondProductProjection{S}{T})
            ((\firstProductProjection{S\times T}{U})(p))$ by
        \obv{circ_apply}.
    $(\firstProductProjection{S\times T}{U})(p)=\fst{p}$ and
        $(\secondProductProjection{S\times T}{U})(p)=\snd{p}$ by
        \obv{first_product_projection_apply,
            second_product_projection_apply}.
    $\fst{p}\in S\times T$ by \obv{fst_type}.
    Hence $(\firstProductProjection{S}{T})(\fst{p})
        =\fst{\fst{p}}$ and
        $(\secondProductProjection{S}{T})(\fst{p})
        =\snd{\fst{p}}$ by
        \obv{first_product_projection_apply,
            second_product_projection_apply}.
    Follows by
        \auto{product_associator,product_pairing_apply}.
\end{proof}

\begin{proposition}\label{inverse_product_associator_apply}
    Let $p\in S\times(T\times U)$.
    Then $(\inverseProductAssociator{S}{T}{U})(p)
        =((\fst{p},\fst{\snd{p}}),\snd{\snd{p}})$.
\end{proposition}
\begin{proof}
    $\secondProductProjection{S}{T\times U}$ is a function and
        $\firstProductProjection{T}{U}$ is a function and
        $\secondProductProjection{T}{U}$ is a function by
        \auto{first_product_projection_type,
            second_product_projection_type,funs_is_function}.
    $\firstProductProjection{T}{U}$ is composable with
        $\secondProductProjection{S}{T\times U}$ and
        $\secondProductProjection{T}{U}$ is composable with
        $\secondProductProjection{S}{T\times U}$ by
        \auto{first_product_projection_type,
            second_product_projection_type,funs_ran,funs}.
    $p\in\dom{\secondProductProjection{S}{T\times U}}$ by
        \auto{second_product_projection_type,funs}.
    Hence $(\firstProductProjection{T}{U}\circ
        \secondProductProjection{S}{T\times U})(p)
        =(\firstProductProjection{T}{U})
            ((\secondProductProjection{S}{T\times U})(p))$ and
        $(\secondProductProjection{T}{U}\circ
        \secondProductProjection{S}{T\times U})(p)
        =(\secondProductProjection{T}{U})
            ((\secondProductProjection{S}{T\times U})(p))$ by
        \obv{circ_apply}.
    $(\firstProductProjection{S}{T\times U})(p)=\fst{p}$ and
        $(\secondProductProjection{S}{T\times U})(p)=\snd{p}$ by
        \obv{first_product_projection_apply,
            second_product_projection_apply}.
    $\snd{p}\in T\times U$ by \obv{snd_type}.
    Hence $(\firstProductProjection{T}{U})(\snd{p})
        =\fst{\snd{p}}$ and
        $(\secondProductProjection{T}{U})(\snd{p})
        =\snd{\snd{p}}$ by
        \obv{first_product_projection_apply,
            second_product_projection_apply}.
    Follows by
        \auto{inverse_product_associator,product_pairing_apply}.
\end{proof}

\begin{proposition}\label{inverse_associator_after_associator}
    Let $p\in(S\times T)\times U$.
    Then $(\inverseProductAssociator{S}{T}{U})
        ((\productAssociator{S}{T}{U})(p))=p$.
\end{proposition}
\begin{proof}
    $(\productAssociator{S}{T}{U})(p)
        =(\fst{\fst{p}},(\snd{\fst{p}},\snd{p}))$ by
        \obv{product_associator_apply}.
    $\fst{p}\in S\times T$ and $\snd{p}\in U$ by
        \obv{fst_type,snd_type}.
    Hence $\fst{\fst{p}}\in S$ and $\snd{\fst{p}}\in T$ by
        \obv{fst_type,snd_type}.
    Hence $(\snd{\fst{p}},\snd{p})\in T\times U$ by
        \obv{times_tuple_intro}.
    Hence $(\fst{\fst{p}},(\snd{\fst{p}},\snd{p}))
        \in S\times(T\times U)$ by \obv{times_tuple_intro}.
    Hence $(\inverseProductAssociator{S}{T}{U})
        ((\productAssociator{S}{T}{U})(p))
        =((\fst{\fst{p}},\snd{\fst{p}}),\snd{p})$ by
        \auto{inverse_product_associator_apply,fst_eq,snd_eq}.
    $\fst{p}=(\fst{\fst{p}},\snd{\fst{p}})$ by
        \auto{times_elem_is_tuple,pair_eq_pair_of_proj}.
    $p=(\fst{p},\snd{p})$ by
        \auto{times_elem_is_tuple,pair_eq_pair_of_proj}.
\end{proof}

\begin{proposition}\label{associator_after_inverse_associator}
    Let $p\in S\times(T\times U)$.
    Then $(\productAssociator{S}{T}{U})
        ((\inverseProductAssociator{S}{T}{U})(p))=p$.
\end{proposition}
\begin{proof}
    $(\inverseProductAssociator{S}{T}{U})(p)
        =((\fst{p},\fst{\snd{p}}),\snd{\snd{p}})$ by
        \obv{inverse_product_associator_apply}.
    $\fst{p}\in S$ and $\snd{p}\in T\times U$ by
        \obv{fst_type,snd_type}.
    Hence $\fst{\snd{p}}\in T$ and $\snd{\snd{p}}\in U$ by
        \obv{fst_type,snd_type}.
    Hence $(\fst{p},\fst{\snd{p}})\in S\times T$ by
        \obv{times_tuple_intro}.
    Hence $((\fst{p},\fst{\snd{p}}),\snd{\snd{p}})
        \in(S\times T)\times U$ by \obv{times_tuple_intro}.
    Hence $(\productAssociator{S}{T}{U})
        ((\inverseProductAssociator{S}{T}{U})(p))
        =(\fst{p},(\fst{\snd{p}},\snd{\snd{p}}))$ by
        \auto{product_associator_apply,fst_eq,snd_eq}.
    $\snd{p}=(\fst{\snd{p}},\snd{\snd{p}})$ by
        \auto{times_elem_is_tuple,pair_eq_pair_of_proj}.
    $p=(\fst{p},\snd{p})$ by
        \auto{times_elem_is_tuple,pair_eq_pair_of_proj}.
\end{proof}

\begin{theorem}\label{product_associator_bijection}
    $\productAssociator{S}{T}{U}$ is a bijection from
        $(S\times T)\times U$ to $S\times(T\times U)$.
\end{theorem}
\begin{proof}
    $\productAssociator{S}{T}{U}\in
        \funs{(S\times T)\times U}{S\times(T\times U)}$ and
        $\inverseProductAssociator{S}{T}{U}\in
        \funs{S\times(T\times U)}{(S\times T)\times U}$ by
        \obv{product_associator_type,inverse_product_associator_type}.
    For every $p\in(S\times T)\times U$ we have
        $(\inverseProductAssociator{S}{T}{U})
            ((\productAssociator{S}{T}{U})(p))=p$ by
        \obv{inverse_associator_after_associator}.
    Hence $\inverseProductAssociator{S}{T}{U}$ is a left inverse of
        $\productAssociator{S}{T}{U}$ by
        \obv{leftinverse_from_to_iff}.
    For every $q\in S\times(T\times U)$ we have
        $(\productAssociator{S}{T}{U})
            ((\inverseProductAssociator{S}{T}{U})(q))=q$ by
        \obv{associator_after_inverse_associator}.
    Hence $\inverseProductAssociator{S}{T}{U}$ is a right inverse of
        $\productAssociator{S}{T}{U}$ by
        \obv{rightinverse_from_to_iff}.
    Hence $\productAssociator{S}{T}{U}$ is an injection by
        \obv{leftinverse_implies_injection}.
    Hence $\productAssociator{S}{T}{U}$ is a surjection from
        $(S\times T)\times U$ to $S\times(T\times U)$ by
        \obv{rightinverse_implies_surjection}.
    Follows by \obv{bijections}.
\end{proof}

\begin{theorem}\label{product_associator_converse}
    $\converse{\productAssociator{S}{T}{U}}
        =\inverseProductAssociator{S}{T}{U}$.
\end{theorem}
\begin{proof}
    $\productAssociator{S}{T}{U}\in
        \bijections{(S\times T)\times U}{S\times(T\times U)}$ by
        \obv{product_associator_bijection}.
    $\productAssociator{S}{T}{U}\in
        \funs{(S\times T)\times U}{S\times(T\times U)}$ by
        \obv{product_associator_type}.
    Hence $\converse{\productAssociator{S}{T}{U}}\in
        \funs{S\times(T\times U)}{(S\times T)\times U}$ by
        \obv{bijective_converse_are_funs}.
    $\inverseProductAssociator{S}{T}{U}\in
        \funs{S\times(T\times U)}{(S\times T)\times U}$ by
        \obv{inverse_product_associator_type}.
    Show that for every $q\in S\times(T\times U)$ we have
        $(\converse{\productAssociator{S}{T}{U}})(q)
        =(\inverseProductAssociator{S}{T}{U})(q)$.
    \begin{subproof}
        Fix $q\in S\times(T\times U)$.
        $(\inverseProductAssociator{S}{T}{U})(q)
            \in(S\times T)\times U$ by \obv{funs_type_apply}.
        $(\productAssociator{S}{T}{U})
            ((\inverseProductAssociator{S}{T}{U})(q))=q$ by
            \obv{associator_after_inverse_associator}.
        Hence $((\inverseProductAssociator{S}{T}{U})(q),q)
            \in\productAssociator{S}{T}{U}$ by
            \auto{function_apply_elim,funs_is_function,funs}.
        Thus $(q,(\inverseProductAssociator{S}{T}{U})(q))
            \in\converse{\productAssociator{S}{T}{U}}$ by
            \obv{converse_iff}.
        $\converse{\productAssociator{S}{T}{U}}$ is a function by
            \obv{funs_is_function}.
        Follows by \obv{function_apply_intro}.
    \end{subproof}
    Follows by \obv{typed_functions_equal_from_agreement}.
\end{proof}
