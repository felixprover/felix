\import{set.tex}

\section{Logical separation and bounded inductive definitions}

\begin{proposition}\label{logical_separation_iff}
    $\forall A:\Set.\ \forall P:\Set\to\Prop.\ \forall x:\Set.
        x\in\sep{A}{P}\iff(x\in A\land\Apply{P}{x})$.
\end{proposition}
\begin{proof}
    Follows by \auto{setext}.
\end{proof}

\begin{proposition}\label{logical_separation_subseteq}
    $\forall A:\Set.\ \forall P:\Set\to\Prop.\ \forall x:\Set.
        x\in\sep{A}{P}\implies x\in A$.
\end{proposition}
\begin{proof}
    Follows by \auto{logical_separation_iff}.
\end{proof}
