\section{Gluing functions}

\import{function.tex}

Functions which agree on the overlap of their domains can be joined by union.
In particular, functions on disjoint domains require no compatibility check.

\begin{theorem}\label{compatible_function_union_type}
    Let $f\in\funs{I}{A}$ and $g\in\funs{J}{A}$.
    Suppose for every $x\in I\inter J$ we have $f(x)=g(x)$.
    Then $f\union g\in\funs{I\union J}{A}$.
\end{theorem}
\begin{proof}
    $f$ and $g$ are functions by \obv{funs_is_function}.
    $f\union g$ is a relation by \obv{union_relations_is_relation}.
    Show that $f\union g$ is right-unique.
    \begin{subproof}
        Show that for all $x,y,z$ such that $(x,y),(x,z)\in f\union g$ we have $y=z$.
        \begin{subproof}
            Fix $x,y,z$.
            Assume $(x,y),(x,z)\in f\union g$.
            $(x,y)\in f$ or $(x,y)\in g$ by \obv{union_iff}.
            $(x,z)\in f$ or $(x,z)\in g$ by \obv{union_iff}.
            Follows by \auto{function_rightunique,function_apply_intro,dom_iff,funs,inter}.
        \end{subproof}
        Follows by \obv{rightunique}.
    \end{subproof}
    $\dom{f\union g}=I\union J$ by \auto{dom_union,funs}.
    $\ran{f\union g}\subseteq A$ by \auto{ran_union,funs_ran,union_iff,subseteq}.
    $f\union g\in\funs{\dom{f\union g}}{\ran{f\union g}}$ by
        \obv{function_exact_type_to_range}.
    Follows by \auto{funs_weaken_codom}.
\end{proof}

\begin{corollary}\label{disjoint_function_union_type}
    Let $f\in\funs{I}{A}$ and $g\in\funs{J}{A}$.
    Suppose $I$ is disjoint from $J$.
    Then $f\union g\in\funs{I\union J}{A}$.
\end{corollary}
\begin{proof}
    For every $x\in I\inter J$ we have $f(x)=g(x)$ by \auto{disjoint,inter}.
    Follows by \obv{compatible_function_union_type}.
\end{proof}

\begin{proposition}\label{function_union_restrictions}
    Let $f\in\funs{I\union J}{A}$.
    Then $\restrl{f}{I}\union\restrl{f}{J}=f$.
\end{proposition}
\begin{proof}
    $f$ is a relation by \obv{funs_is_relation}.
    Hence $\restrl{f}{I}\union\restrl{f}{J}$ is a relation by
        \obv{union_relations_is_relation}.
    For all $x,y$ we have $(x,y)\in\restrl{f}{I}\union\restrl{f}{J}$ iff
        $(x,y)\in f$ by \auto{union_iff,restrl_iff,dom_iff,funs}.
    Follows by \obv{relext}.
\end{proof}
