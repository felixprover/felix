\section{Generated filters}

\import{set/filter-finite.tex}

The filters on a fixed carrier form a closure system inside its powerset.
This gives a canonical generated filter for every well-typed family of
generators; concrete finite intersections provide its normal proof interface.

\begin{definition}\label{filters_on}
    $\filtersOn{S}=\{F\in\pow{\pow{S}}\mid
        \text{$F$ is a filter on $S$}\}$.
\end{definition}

\begin{proposition}\label{filters_on_iff}
    $F\in\filtersOn{S}$ iff $F$ is a filter on $S$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{filters_on,filter,pow_iff,subseteq}.
\end{proof}

\begin{classification}\label{filters_on_member_is_filter}
    Let $F\in\filtersOn{S}$.
    Then $F$ is a filter on $S$.
\end{classification}
\begin{proof}
    Follows by \obv{filters_on}.
\end{proof}

\begin{proposition}\label{filter_in_filters_on}
    Let $F$ be a filter on $S$.
    Then $F\in\filtersOn{S}$.
\end{proposition}
\begin{proof}
    Follows by \auto{filters_on,filter,pow_iff,subseteq}.
\end{proof}

\begin{theorem}\label{filters_on_is_closure_system}
    $\filtersOn{S}$ is a closure system on $\pow{S}$.
\end{theorem}
\begin{proof}
    $\filtersOn{S}\subseteq\pow{\pow{S}}$ by
        \obv{filters_on,subseteq}.
    Show that for every $C\subseteq\filtersOn{S}$ we have
        $\boundedIntersection{\pow{S}}{C}\in\filtersOn{S}$.
    \begin{subproof}
        Fix $C\subseteq\filtersOn{S}$.
        Let $I=\boundedIntersection{\pow{S}}{C}$.
        $\boundedIntersection{\pow{S}}{C}\subseteq\pow{S}$ by
            \obv{bounded_intersection_subseteq}.
        Hence $I\subseteq\pow{S}$.
        Hence $I$ is a family of subsets of $S$ by
            \auto{pow_iff,subseteq}.

        Show that $S\in I$.
        \begin{subproof}
            $S\in\pow{S}$ by \obv{powerset_top}.
            Show that for every $F\in C$ we have $S\in F$.
            \begin{subproof}
                Fix $F\in C$.
                $F\in\filtersOn{S}$ by \obv{subseteq}.
                Hence $F$ is a filter on $S$ by
                    \obv{filters_on_member_is_filter}.
                Follows by \obv{filter}.
            \end{subproof}
            Follows by \obv{bounded_intersection_iff}.
        \end{subproof}

        Show that $I$ is closed under binary intersections.
        \begin{subproof}
            Fix $A,D\in I$.
            $A,D\subseteq S$.
            Hence $A\inter D\subseteq S$ by \obv{inter_subseteq}.
            Thus $A\inter D\in\pow{S}$ by \obv{powerset_intro}.
            Show that for every $F\in C$ we have $A\inter D\in F$.
            \begin{subproof}
                Fix $F\in C$.
                $F\in\filtersOn{S}$ by \obv{subseteq}.
                Hence $F$ is a filter on $S$ by
                    \obv{filters_on_member_is_filter}.
                $A,D\in F$ by \auto{bounded_intersection_iff}.
                Follows by \obv{filter}.
            \end{subproof}
            Follows by \obv{bounded_intersection_iff}.
        \end{subproof}

        Show that $I$ is upward-closed in $S$.
        \begin{subproof}
            Fix $A,D$.
            Assume $A\subseteq D\subseteq S$ and $A\in I$.
            $D\in\pow{S}$ by \obv{powerset_intro}.
            Show that for every $F\in C$ we have $D\in F$.
            \begin{subproof}
                Fix $F\in C$.
                $F\in\filtersOn{S}$ by \obv{subseteq}.
                Hence $F$ is a filter on $S$ by
                    \obv{filters_on_member_is_filter}.
                $A\in F$ by \auto{bounded_intersection_iff}.
                Follows by \obv{filter}.
            \end{subproof}
            Follows by \obv{bounded_intersection_iff}.
        \end{subproof}
        Hence $I$ is a filter on $S$ by \obv{filter}.
        Thus $\boundedIntersection{\pow{S}}{C}$ is a filter on $S$.
        Follows by \obv{filter_in_filters_on}.
    \end{subproof}
    Follows by \obv{closure_system}.
\end{proof}

\begin{proposition}\label{bounded_intersection_of_filters_is_filter}
    Suppose $C\subseteq\filtersOn{S}$.
    Then $\boundedIntersection{\pow{S}}{C}$ is a filter on $S$.
\end{proposition}
\begin{proof}
    $\boundedIntersection{\pow{S}}{C}\in\filtersOn{S}$ by
        \obv{filters_on_is_closure_system,closure_system_intersection}.
    Follows by \obv{filters_on_member_is_filter}.
\end{proof}

\begin{definition}\label{generated_filter}
    $\generatedFilter{S}{B}
        =\closureHull{\pow{S}}{\filtersOn{S}}{B}$.
\end{definition}

\begin{proposition}\label{generated_filter_type}
    $\generatedFilter{S}{B}$ is a family of subsets of $S$.
\end{proposition}
\begin{proof}
    $\generatedFilter{S}{B}\subseteq\pow{S}$ by
        \auto{generated_filter,closure_hull_subseteq}.
    Follows by \auto{pow_iff,subseteq}.
\end{proof}

\begin{proposition}\label{generated_filter_contains}
    Suppose $B$ is a family of subsets of $S$.
    Then $B\subseteq\generatedFilter{S}{B}$.
\end{proposition}
\begin{proof}
    $\filtersOn{S}$ is a closure system on $\pow{S}$ by
        \obv{filters_on_is_closure_system}.
    $B\subseteq\pow{S}$ by \auto{pow_iff,subseteq}.
    Follows by \auto{generated_filter,closure_hull_extensive}.
\end{proof}

\begin{classification}\label{generated_filter_is_filter}
    Suppose $B$ is a family of subsets of $S$.
    Then $\generatedFilter{S}{B}$ is a filter on $S$.
\end{classification}
\begin{proof}
    $\filtersOn{S}$ is a closure system on $\pow{S}$ by
        \obv{filters_on_is_closure_system}.
    $B\subseteq\pow{S}$ by \auto{pow_iff,subseteq}.
    $\closureHull{\pow{S}}{\filtersOn{S}}{B}
        \in\filtersOn{S}$ by \obv{closure_hull_closed}.
    Hence $\generatedFilter{S}{B}\in\filtersOn{S}$ by
        \obv{generated_filter}.
    Follows by \obv{filters_on_member_is_filter}.
\end{proof}

\begin{theorem}\label{generated_filter_least}
    Suppose $B$ is a family of subsets of $S$.
    Let $F$ be a filter on $S$.
    Suppose $B\subseteq F$.
    Then $\generatedFilter{S}{B}\subseteq F$.
\end{theorem}
\begin{proof}
    $\filtersOn{S}$ is a closure system on $\pow{S}$ by
        \obv{filters_on_is_closure_system}.
    $B\subseteq\pow{S}$ by \auto{pow_iff,subseteq}.
    $F\in\filtersOn{S}$ by \obv{filter_in_filters_on}.
    Follows by
        \auto{generated_filter,closure_hull_least}.
\end{proof}

\begin{proposition}\label{generated_filter_monotone}
    Suppose $B$ is a family of subsets of $S$ and
        $C$ is a family of subsets of $S$ and $B\subseteq C$.
    Then $\generatedFilter{S}{B}\subseteq\generatedFilter{S}{C}$.
\end{proposition}
\begin{proof}
    $\filtersOn{S}$ is a closure system on $\pow{S}$ by
        \obv{filters_on_is_closure_system}.
    $C\subseteq\pow{S}$ by \auto{pow_iff,subseteq}.
    Follows by
        \auto{generated_filter,closure_hull_monotone}.
\end{proof}

\begin{proposition}\label{generated_filter_idempotent}
    Suppose $B$ is a family of subsets of $S$.
    Then $\generatedFilter{S}{\generatedFilter{S}{B}}
        =\generatedFilter{S}{B}$.
\end{proposition}
\begin{proof}
    $\filtersOn{S}$ is a closure system on $\pow{S}$ by
        \obv{filters_on_is_closure_system}.
    $B\subseteq\pow{S}$ by \auto{pow_iff,subseteq}.
    Follows by
        \auto{generated_filter,closure_hull_idempotent}.
\end{proof}

\begin{theorem}\label{filter_generates_itself}
    Let $F$ be a filter on $S$.
    Then $\generatedFilter{S}{F}=F$.
\end{theorem}
\begin{proof}
    $F\in\filtersOn{S}$ by \obv{filter_in_filters_on}.
    Follows by
        \obv{generated_filter,filters_on_is_closure_system,
            closure_hull_already_closed}.
\end{proof}

\begin{proposition}\label{finite_intersection_supersets_are_filter}
    Suppose $B$ is a family of subsets of $S$.
    Then
        $\{D\in\pow{S}\mid
            \text{there exists $C\in\finiteIntersectionsOn{S}{B}$ such that
                $C\subseteq D$}\}$
        is a filter on $S$.
\end{proposition}
\begin{proof}
    Let $G=\{D\in\pow{S}\mid
        \text{there exists $C\in\finiteIntersectionsOn{S}{B}$ such that
            $C\subseteq D$}\}$.
    $G$ is a family of subsets of $S$ by
        \obv{pow_iff,subseteq}.
    $S\in\finiteIntersectionsOn{S}{B}$ by
        \obv{carrier_in_finite_intersections_on}.
    Hence $S\in G$ by \obv{subseteq_refl}.
    Show that $G$ is closed under binary intersections.
    \begin{subproof}
        Fix $D,E\in G$.
        $D,E\subseteq S$.
        Take $C\in\finiteIntersectionsOn{S}{B}$ such that $C\subseteq D$.
        Take $H\in\finiteIntersectionsOn{S}{B}$ such that $H\subseteq E$.
        $C\inter H\in\finiteIntersectionsOn{S}{B}$ by
            \obv{finite_intersections_on_intersection}.
        $C\inter H\subseteq D\inter E$ by \obv{inter,subseteq}.
        $D\inter E\subseteq S$ by \obv{inter_subseteq}.
        Hence $D\inter E\in\pow{S}$ by \obv{powerset_intro}.
        There exists $K\in\finiteIntersectionsOn{S}{B}$ such that
            $K\subseteq D\inter E$.
    \end{subproof}
    Show that $G$ is upward-closed in $S$.
    \begin{subproof}
        Fix $D,E$.
        Assume $D\subseteq E\subseteq S$ and $D\in G$.
        Take $C\in\finiteIntersectionsOn{S}{B}$ such that $C\subseteq D$.
        Hence $C\subseteq E$ by \obv{subseteq_transitive}.
        $E\in\pow{S}$ by \obv{powerset_intro}.
        There exists $K\in\finiteIntersectionsOn{S}{B}$ such that
            $K\subseteq E$.
    \end{subproof}
    Hence $G$ is a filter on $S$ by \obv{filter}.
\end{proof}

\begin{proposition}\label{generators_in_finite_intersection_supersets}
    Suppose $B$ is a family of subsets of $S$.
    Then $B\subseteq
        \{D\in\pow{S}\mid
            \text{there exists $C\in\finiteIntersectionsOn{S}{B}$ such that
                $C\subseteq D$}\}$.
\end{proposition}
\begin{proof}
    $B\subseteq\finiteIntersectionsOn{S}{B}$ by
        \obv{generators_in_finite_intersections_on}.
    It suffices to show that every element of $B$ is an element of
        $\{D\in\pow{S}\mid
            \text{there exists $C\in\finiteIntersectionsOn{S}{B}$ such that
                $C\subseteq D$}\}$
        by \obv{subseteq}.
    Fix $C\in B$.
    Hence $C\in\finiteIntersectionsOn{S}{B}$ by \obv{subseteq}.
    $C\subseteq S$.
    Hence $C\in\pow{S}$ by \obv{powerset_intro}.
    There exists $D\in\finiteIntersectionsOn{S}{B}$ such that
        $D\subseteq C$ by \obv{subseteq_refl}.
    Follows by \auto.
\end{proof}

\begin{theorem}\label{generated_filter_iff_finite_intersection_member}
    Suppose $B$ is a family of subsets of $S$.
    Then $A\in\generatedFilter{S}{B}$ iff
        $A\subseteq S$ and there exists
            $C\in\finiteIntersectionsOn{S}{B}$ such that $C\subseteq A$.
\end{theorem}
\begin{proof}
    $\{D\in\pow{S}\mid
        \text{there exists $C\in\finiteIntersectionsOn{S}{B}$ such that
            $C\subseteq D$}\}$ is a filter on $S$ by
        \obv{finite_intersection_supersets_are_filter}.
    $B\subseteq\{D\in\pow{S}\mid
        \text{there exists $C\in\finiteIntersectionsOn{S}{B}$ such that
            $C\subseteq D$}\}$ by
        \obv{generators_in_finite_intersection_supersets}.
    Hence $\generatedFilter{S}{B}\subseteq
        \{D\in\pow{S}\mid
            \text{there exists $C\in\finiteIntersectionsOn{S}{B}$ such that
                $C\subseteq D$}\}$ by \obv{generated_filter_least}.

    Show that if $A\in\generatedFilter{S}{B}$, then
        $A\subseteq S$ and there exists
            $C\in\finiteIntersectionsOn{S}{B}$ such that $C\subseteq A$.
    \begin{subproof}
        Assume $A\in\generatedFilter{S}{B}$.
        Hence $A\in\{D\in\pow{S}\mid
            \text{there exists $C\in\finiteIntersectionsOn{S}{B}$ such that
                $C\subseteq D$}\}$ by \obv{subseteq}.
        $A\in\pow{S}$ and there exists
            $C\in\finiteIntersectionsOn{S}{B}$ such that $C\subseteq A$
            by \auto.
        Hence $A\subseteq S$ by \obv{powerset_elim}.
    \end{subproof}

    Show that if $A\subseteq S$ and there exists
        $C\in\finiteIntersectionsOn{S}{B}$ such that $C\subseteq A$, then
        $A\in\generatedFilter{S}{B}$.
    \begin{subproof}
        Assume $A\subseteq S$ and there exists
            $C\in\finiteIntersectionsOn{S}{B}$ such that $C\subseteq A$.
        Take $C\in\finiteIntersectionsOn{S}{B}$ such that $C\subseteq A$.
        Take $D$ such that $D\subseteq B$ and $D$ is finite and
            $C=\boundedIntersection{S}{D}$
            by \obv{finite_intersections_on}.
        $B\subseteq\generatedFilter{S}{B}$ by
            \obv{generated_filter_contains}.
        Hence $D\subseteq\generatedFilter{S}{B}$ by
            \obv{subseteq_transitive}.
        $\generatedFilter{S}{B}$ is a filter on $S$ by
            \obv{generated_filter_is_filter}.
        $\boundedIntersection{S}{D}\in\generatedFilter{S}{B}$ by
            \obv{filter_bounded_finite_intersection_in}.
        Hence $C\in\generatedFilter{S}{B}$.
        Follows by \obv{filter}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{theorem}\label{generated_filter_of_finite_intersections_on}
    Suppose $B$ is a family of subsets of $S$.
    Then
        $\generatedFilter{S}{\finiteIntersectionsOn{S}{B}}
        =\generatedFilter{S}{B}$.
\end{theorem}
\begin{proof}
    $\finiteIntersectionsOn{S}{B}$ is a family of subsets of $S$ by
        \obv{finite_intersections_on_type}.
    $B\subseteq\finiteIntersectionsOn{S}{B}$ by
        \obv{generators_in_finite_intersections_on}.
    Hence $\generatedFilter{S}{B}\subseteq
        \generatedFilter{S}{\finiteIntersectionsOn{S}{B}}$ by
        \obv{generated_filter_monotone}.

    Show that $\finiteIntersectionsOn{S}{B}
        \subseteq\generatedFilter{S}{B}$.
    \begin{subproof}
        It suffices to show that every element of
            $\finiteIntersectionsOn{S}{B}$ is an element of
            $\generatedFilter{S}{B}$ by \obv{subseteq}.
        Fix $C\in\finiteIntersectionsOn{S}{B}$.
        $C\subseteq S$ by \obv{finite_intersections_on}.
        There exists $D\in\finiteIntersectionsOn{S}{B}$ such that
            $D\subseteq C$ by \obv{subseteq_refl}.
        Follows by \auto{generated_filter_iff_finite_intersection_member}.
    \end{subproof}
    $\generatedFilter{S}{B}$ is a filter on $S$ by
        \obv{generated_filter_is_filter}.
    Hence $\generatedFilter{S}{\finiteIntersectionsOn{S}{B}}
        \subseteq\generatedFilter{S}{B}$ by
        \obv{generated_filter_least}.
    Follows by \obv{subseteq_antisymmetric}.
\end{proof}

\begin{theorem}\label{generated_filter_iff_finite_intersections}
    Suppose $B$ is a family of subsets of $S$.
    Then $A\in\generatedFilter{S}{B}$ iff $A\subseteq S$ and
        there exists $D$ such that $D\subseteq B$ and $D$ is finite and
            $\boundedIntersection{S}{D}\subseteq A$.
\end{theorem}
\begin{proof}
    Show that if $A\in\generatedFilter{S}{B}$, then $A\subseteq S$ and
        there exists $D$ such that $D\subseteq B$ and $D$ is finite and
            $\boundedIntersection{S}{D}\subseteq A$.
    \begin{subproof}
        Assume $A\in\generatedFilter{S}{B}$.
        $A\subseteq S$ and there exists
            $C\in\finiteIntersectionsOn{S}{B}$ such that $C\subseteq A$
            by \auto{generated_filter_iff_finite_intersection_member}.
        Take $C\in\finiteIntersectionsOn{S}{B}$ such that $C\subseteq A$.
        Take $D$ such that $D\subseteq B$ and $D$ is finite and
            $C=\boundedIntersection{S}{D}$
            by \obv{finite_intersections_on}.
        Hence $\boundedIntersection{S}{D}\subseteq A$.
    \end{subproof}

    Show that if $A\subseteq S$ and there exists $D$ such that
        $D\subseteq B$ and $D$ is finite and
        $\boundedIntersection{S}{D}\subseteq A$, then
        $A\in\generatedFilter{S}{B}$.
    \begin{subproof}
        Assume $A\subseteq S$ and there exists $D$ such that
            $D\subseteq B$ and $D$ is finite and
            $\boundedIntersection{S}{D}\subseteq A$.
        Take $D$ such that $D\subseteq B$ and $D$ is finite and
            $\boundedIntersection{S}{D}\subseteq A$.
        Let $C=\boundedIntersection{S}{D}$.
        $C\subseteq S$ by \obv{bounded_intersection_subseteq}.
        $C\in\finiteIntersectionsOn{S}{B}$ by
            \auto{finite_intersections_on_iff}.
        Follows by
            \auto{generated_filter_iff_finite_intersection_member}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{theorem}\label{generated_filter_is_proper_iff_hosted_fip}
    Suppose $B$ is a family of subsets of $S$.
    Then $\generatedFilter{S}{B}$ is a proper filter on $S$ iff
        $B$ has the finite intersection property on $S$.
\end{theorem}
\begin{proof}
    $\generatedFilter{S}{B}$ is a filter on $S$ by
        \obv{generated_filter_is_filter}.
    Show that if $\generatedFilter{S}{B}$ is a proper filter on $S$, then
        $B$ has the finite intersection property on $S$.
    \begin{subproof}
        Assume $\generatedFilter{S}{B}$ is a proper filter on $S$.
        Show that for every $D$ such that $D\subseteq B$ and $D$ is finite
            we have $\boundedIntersection{S}{D}$ is inhabited.
        \begin{subproof}
            Fix $D$.
            Assume $D\subseteq B$ and $D$ is finite.
            $B\subseteq\generatedFilter{S}{B}$ by
                \obv{generated_filter_contains}.
            Hence $D\subseteq\generatedFilter{S}{B}$ by
                \obv{subseteq_transitive}.
            $\boundedIntersection{S}{D}\in\generatedFilter{S}{B}$ by
                \obv{filter_bounded_finite_intersection_in}.
            Hence $\boundedIntersection{S}{D}\neq\emptyset$ by
                \obv{proper_filter}.
            Follows by \obv{nonempty_inhabited}.
        \end{subproof}
        Follows by \obv{finite_intersection_property_on}.
    \end{subproof}

    Show that if $B$ has the finite intersection property on $S$, then
        $\generatedFilter{S}{B}$ is a proper filter on $S$.
    \begin{subproof}
        Assume $B$ has the finite intersection property on $S$.
        Show that $\emptyset\notin\generatedFilter{S}{B}$.
        \begin{subproof}
            Suppose not.
            Take $D$ such that $D\subseteq B$ and $D$ is finite and
                $\boundedIntersection{S}{D}\subseteq\emptyset$
                by \auto{generated_filter_iff_finite_intersections}.
            $\boundedIntersection{S}{D}$ is inhabited by
                \obv{finite_intersection_property_on}.
            Take $x$ such that $x\in\boundedIntersection{S}{D}$.
            Hence $x\in\emptyset$ by \obv{subseteq}.
            Contradiction by \obv{emptyset}.
        \end{subproof}
        Follows by \obv{proper_filter}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{theorem}\label{principalfilter_eq_generated_singleton}
    Suppose $A\subseteq S$.
    Then $\principalfilter{S}{A}=\generatedFilter{S}{\{A\}}$.
\end{theorem}
\begin{proof}
    $\{A\}$ is a family of subsets of $S$ by
        \auto{singleton_iff}.
    $\generatedFilter{S}{\{A\}}$ is a filter on $S$ by
        \obv{generated_filter_is_filter}.
    $\{A\}\subseteq\generatedFilter{S}{\{A\}}$ by
        \obv{generated_filter_contains}.
    Hence $A\in\generatedFilter{S}{\{A\}}$ by
        \auto{singleton_iff,subseteq}.
    $\principalfilter{S}{A}$ is a filter on $S$ by
        \obv{principalfilter_is_filter}.
    Show that $\principalfilter{S}{A}\subseteq
        \generatedFilter{S}{\{A\}}$.
    \begin{subproof}
        It suffices to show that every element of $\principalfilter{S}{A}$
            is an element of $\generatedFilter{S}{\{A\}}$
            by \obv{subseteq}.
        Fix $D\in\principalfilter{S}{A}$.
        $D\subseteq S$ by \obv{filter}.
        $A\subseteq D$ by \obv{principalfilter_iff}.
        Follows by \obv{filter}.
    \end{subproof}
    $\{A\}\subseteq\principalfilter{S}{A}$ by
        \auto{singleton_subset_intro,principalfilter_iff,
            subseteq_refl}.
    Hence $\generatedFilter{S}{\{A\}}\subseteq
        \principalfilter{S}{A}$ by \obv{generated_filter_least}.
    Follows by \obv{subseteq_antisymmetric}.
\end{proof}
