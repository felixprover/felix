\section{Choice}

\import{function.tex}

The set-level operator $\choice{A}$ is a total term.  Its public witness law
asserts membership only when $A$ is inhabited; no particular value is claimed
for the empty set.  This global choice operator is distinct from typed logical
selection in the logical framework.

\begin{classification}\label{choice_in}
    Let $A$ be an inhabited set.
    Then $\choice{A}\in A$.
\end{classification}
\begin{proof}
    Follows by \obv{choice_member}.
\end{proof}


\begin{definition}\label{choice_function}
    $f$ is a choice function for $F$ iff
        $f$ is a function on $F$ and
        for all $A\in F$ we have $f(A)\in A$.
\end{definition}


\begin{theorem}\label{choice_function_exists}
    Suppose every element of $F$ is inhabited.
    Then there exists $f$ such that $f$ is a choice function for $F$.
\end{theorem}
\begin{proof}
    Let $f=\{(A,\choice{A})\mid A\in F\}$.
    Show that $f$ is a relation.
    \begin{subproof}
        Follows by \auto{relation}.
    \end{subproof}
    Show that $f$ is right-unique.
    \begin{subproof}
        Follows by \auto{rightunique,pair_eq_iff}.
    \end{subproof}
    Show that $\dom{f}=F$.
    \begin{subproof}
        Follows by \auto{setext,dom_iff,pair_eq_iff}.
    \end{subproof}
    Hence $f$ is a function on $F$.
    For all $A\in F$ we have $\choice{A}\in A$
        by \obv{choice_in}.
    For all $A\in F$ we have $f(A)=\choice{A}$
        by \obv{function_apply_intro}.
    Thus for all $A\in F$ we have $f(A)\in A$ by \auto.
    Hence $f$ is a choice function for $F$ by \obv{choice_function}.
\end{proof}


\begin{theorem}\label{choice_function_exists_iff}
    Every element of $F$ is inhabited iff
        there exists $f$ such that $f$ is a choice function for $F$.
\end{theorem}
\begin{proof}
    Show that if every element of $F$ is inhabited, then there exists $f$
        such that $f$ is a choice function for $F$.
    \begin{subproof}
        Assume every element of $F$ is inhabited.
        Follows by \obv{choice_function_exists}.
    \end{subproof}
    Show that if there exists $f$ such that $f$ is a choice function for
        $F$, then every element of $F$ is inhabited.
    \begin{subproof}
        Assume there exists $f$ such that $f$ is a choice function for $F$.
        Take $f$ such that $f$ is a choice function for $F$.
        For every $A\in F$ we have $f(A)\in A$ by
            \obv{choice_function}.
        Follows by \auto.
    \end{subproof}
    Follows by \auto.
\end{proof}


\begin{theorem}\label{axiom_of_choice}
    Suppose $R$ is a relation.
    Suppose for all $x\in A$ there exists $y$ such that $x\mathrel{R} y$.
    Then there exists $f$ such that $f$ is a function on $A$ and
        for all $x\in A$ we have $x\mathrel{R} f(x)$.
\end{theorem}
\begin{proof}
    Let $f=\{(x,\choice{\img{R}{\{x\}}})\mid x\in A\}$.
    Show that $f$ is a relation.
    \begin{subproof}
        Follows by \auto{relation}.
    \end{subproof}
    Show that $f$ is right-unique.
    \begin{subproof}
        Follows by \auto{rightunique,pair_eq_iff}.
    \end{subproof}
    Show that $\dom{f}=A$.
    \begin{subproof}
        Follows by \auto{setext,dom_iff,pair_eq_iff}.
    \end{subproof}
    Hence $f$ is a function on $A$.
    Show that for all $x\in A$ we have
        $\choice{\img{R}{\{x\}}}\in\img{R}{\{x\}}$.
    \begin{subproof}
        Fix $x\in A$.
        Take $y$ such that $x\mathrel{R}y$.
        Then $y\in\img{R}{\{x\}}$ by \auto{img_iff,singleton_iff}.
        Thus $\img{R}{\{x\}}$ is inhabited.
        Follows by \obv{choice_in}.
    \end{subproof}
    For all $x\in A$ we have
        $x\mathrel{R}\choice{\img{R}{\{x\}}}$
        by \auto{img_iff,singleton_iff}.
    For all $x\in A$ we have
        $f(x)=\choice{\img{R}{\{x\}}}$
        by \obv{function_apply_intro}.
    Thus for all $x\in A$ we have $x\mathrel{R}f(x)$ by \auto.
\end{proof}
