\section{Almost-disjoint families}

\import{set/finite.tex}
\import{set/powerset.tex}

An almost-disjoint family may be empty or finite, although each of its members
is infinite. An infinite almost-disjoint family has infinitely many members.

\begin{definition}\label{almost_disjoint_family_on}
    $F$ is an almost-disjoint family on $X$ iff
        $F\subseteq\pow{X}$ and
        (for every $A\in F$ we have $A$ is infinite) and
        (for all $A,B\in F$ such that $A\neq B$
            we have $A\inter B$ is finite).
\end{definition}

\begin{proposition}\label{empty_is_almost_disjoint_family}
    $\emptyset$ is an almost-disjoint family on $X$.
\end{proposition}
\begin{proof}
    $\emptyset\subseteq\pow{X}$ by \obv{emptyset_subseteq}.
    For every $A\in\emptyset$ we have $A$ is infinite by
        \obv{emptyset}.
    For all $A,B\in\emptyset$ such that $A\neq B$ we have
        $A\inter B$ is finite by \auto{emptyset}.
    Follows by \obv{almost_disjoint_family_on}.
\end{proof}

\begin{proposition}\label{almost_disjoint_members_subset}
    Suppose $F$ is an almost-disjoint family on $X$.
    Let $A\in F$.
    Then $A\subseteq X$.
\end{proposition}
\begin{proof}
    $A\in\pow{X}$
        by \auto{almost_disjoint_family_on,subseteq}.
    Follows by \obv{pow_iff,subseteq}.
\end{proof}

\begin{proposition}\label{almost_disjoint_members_infinite}
    Suppose $F$ is an almost-disjoint family on $X$.
    Let $A\in F$.
    Then $A$ is infinite.
\end{proposition}
\begin{proof}
    Follows by \obv{almost_disjoint_family_on}.
\end{proof}

\begin{proposition}\label{almost_disjoint_distinct_intersection_finite}
    Suppose $F$ is an almost-disjoint family on $X$.
    Let $A,B\in F$.
    Suppose $A\neq B$.
    Then $A\inter B$ is finite.
\end{proposition}
\begin{proof}
    Follows by \obv{almost_disjoint_family_on}.
\end{proof}

\begin{definition}\label{almost_disjoint_from_family_on}
    $B$ is almost disjoint from $F$ on $X$ iff
        $B\subseteq X$ and $B$ is infinite and
        for every $A\in F$ we have $A\inter B$ is finite.
\end{definition}

\begin{proposition}\label{almost_disjoint_from_family_not_member}
    Suppose $F$ is an almost-disjoint family on $X$.
    Suppose $B$ is almost disjoint from $F$ on $X$.
    Then $B\notin F$.
\end{proposition}
\begin{proof}
    Suppose $B\in F$.
    $B\inter B$ is finite by \obv{almost_disjoint_from_family_on}.
    Hence $B$ is finite by \obv{inter_idempotent}.
    $B$ is infinite by \obv{almost_disjoint_from_family_on}.
    Contradiction.
\end{proof}

\begin{proposition}\label{almost_disjoint_cons_of_member}
    Suppose $F$ is an almost-disjoint family on $X$.
    Let $B\in F$.
    Then $\cons{B}{F}$ is an almost-disjoint family on $X$.
\end{proposition}
\begin{proof}
    $\cons{B}{F}=F$ by \obv{cons_absorb}.
\end{proof}

\begin{proposition}\label{almost_disjoint_cons_of_compatible}
    Suppose $F$ is an almost-disjoint family on $X$.
    Suppose $B$ is almost disjoint from $F$ on $X$.
    Then $\cons{B}{F}$ is an almost-disjoint family on $X$.
\end{proposition}
\begin{proof}
    $B\subseteq X$ by \obv{almost_disjoint_from_family_on}.
    Hence $B\in\pow{X}$ by \obv{pow_iff,subseteq}.
    $F\subseteq\pow{X}$
        by \obv{almost_disjoint_family_on}.
    Hence $\cons{B}{F}\subseteq\pow{X}$
        by \obv{cons_subseteq_intro}.
    Show that for every $C\in\cons{B}{F}$ we have $C$ is infinite.
    \begin{subproof}
        Fix $C\in\cons{B}{F}$.
        $C=B$ or $C\in F$ by \auto{cons_iff}.
        Follows by
            \obv{almost_disjoint_from_family_on,
                almost_disjoint_members_infinite}.
    \end{subproof}
    Show that for all $C,D\in\cons{B}{F}$ such that $C\neq D$
        we have $C\inter D$ is finite.
    \begin{subproof}
        Fix $C,D\in\cons{B}{F}$.
        Assume $C\neq D$.
        $C=B$ or $C\in F$ by \auto{cons_iff}.
        $D=B$ or $D\in F$ by \auto{cons_iff}.
        \begin{byCase}
            \caseOf{$C,D\in F$.}
                Follows by
                    \obv{almost_disjoint_distinct_intersection_finite}.
            \caseOf{$C=B$ and $D\in F$.}
                $D\inter B$ is finite
                    by \obv{almost_disjoint_from_family_on}.
                Follows by \auto{inter_comm}.
            \caseOf{$C\in F$ and $D=B$.}
                Follows by \auto{almost_disjoint_from_family_on}.
        \end{byCase}
    \end{subproof}
    Follows by \obv{almost_disjoint_family_on}.
\end{proof}

\begin{proposition}\label{almost_disjoint_cons_forward_compatible}
    Suppose $F$ is an almost-disjoint family on $X$.
    Suppose $\cons{B}{F}$ is an almost-disjoint family on $X$.
    Suppose $B\notin F$.
    Then $B$ is almost disjoint from $F$ on $X$.
\end{proposition}
\begin{proof}
    $B\in\cons{B}{F}$ by \obv{cons_left}.
    Hence $B\subseteq X$
        by \obv{almost_disjoint_members_subset}.
    $B$ is infinite by \obv{almost_disjoint_members_infinite}.
    Show that for every $A\in F$ we have $A\inter B$ is finite.
    \begin{subproof}
        Fix $A\in F$.
        $A\in\cons{B}{F}$ by \auto{cons_iff}.
        $A\neq B$.
        Follows by
            \obv{almost_disjoint_distinct_intersection_finite}.
    \end{subproof}
    Follows by \obv{almost_disjoint_from_family_on}.
\end{proof}

\begin{theorem}\label{almost_disjoint_cons_iff}
    Suppose $F$ is an almost-disjoint family on $X$.
    Then $\cons{B}{F}$ is an almost-disjoint family on $X$ iff
        $B\in F$ or
        $B$ is almost disjoint from $F$ on $X$.
\end{theorem}
\begin{proof}
    Show that if $\cons{B}{F}$ is an almost-disjoint family
        on $X$, then $B\in F$ or
        $B$ is almost disjoint from $F$ on $X$.
    \begin{subproof}
        Assume $\cons{B}{F}$ is an almost-disjoint family
            on $X$.
        \begin{byCase}
            \caseOf{$B\in F$.}
                Follows.
            \caseOf{$B\notin F$.}
                $B$ is almost disjoint from $F$ on $X$
                    by \obv{almost_disjoint_cons_forward_compatible}.
        \end{byCase}
    \end{subproof}
    Show that if $B\in F$ or
        $B$ is almost disjoint from $F$ on $X$, then
        $\cons{B}{F}$ is an almost-disjoint family on $X$.
    \begin{subproof}
        Assume $B\in F$ or
            $B$ is almost disjoint from $F$ on $X$.
        Follows by
            \obv{almost_disjoint_cons_of_member,
                almost_disjoint_cons_of_compatible}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{definition}\label{blocks_infinite_subsets_of}
    $F$ blocks the infinite subsets of $X$ iff
        for every $B\subseteq X$ such that $B$ is infinite
        there exists $A\in F$ such that $A\inter B$ is infinite.
\end{definition}

\begin{theorem}\label{not_blocking_iff_almost_disjoint_witness}
    Suppose $F$ is an almost-disjoint family on $X$.
    Then $F$ does not block the infinite subsets of $X$ iff
        there exists $B$ such that
            $B$ is almost disjoint from $F$ on $X$.
\end{theorem}
\begin{proof}
    Follows by
        \auto{blocks_infinite_subsets_of,almost_disjoint_from_family_on}.
\end{proof}
