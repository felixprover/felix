\section{Empty and singleton indexed sums}

\import{set/indexed-family.tex}

The tagged sum over no indices is empty.  Over one index, the canonical
injection is a bijection; keeping this bijection explicit avoids identifying
the tagged carrier with its sole fibre.

\begin{proposition}\label{empty_indexed_sum}
    $\indexedSum{\emptyset}{A}=\emptyset$.
\end{proposition}
\begin{proof}
    Show that every element of $\indexedSum{\emptyset}{A}$ is an element of
        $\emptyset$.
    \begin{subproof}
        Fix $p\in\indexedSum{\emptyset}{A}$.
        Hence $p\in\emptyset\times\unions{\ran{A}}$ by
            \obv{indexed_sum}.
        $\emptyset\times\unions{\ran{A}}=\emptyset$ by
            \obv{times_empty_left}.
        Contradiction by \obv{emptyset}.
    \end{subproof}
    Hence $\indexedSum{\emptyset}{A}\subseteq\emptyset$ by
        \obv{subseteq}.
    Follows by \obv{subseteq_emptyset_iff}.
\end{proof}

\begin{theorem}\label{singleton_indexed_sum_injection_is_bijection}
    Suppose $A$ is a function on $\{i\}$.
    Then $\indexedSumInjection{\{i\}}{A}{i}$ is a bijection from
        $A(i)$ to $\indexedSum{\{i\}}{A}$.
\end{theorem}
\begin{proof}
    Let $f=\indexedSumInjection{\{i\}}{A}{i}$.
    $i\in\{i\}$ by \auto{singleton_iff}.
    Hence $f\in\inj{A(i)}{\indexedSum{\{i\}}{A}}$ by
        \auto{indexed_sum_injection_type}.
    Show that for all $x,y,p$ such that $x,y\mathrel{f}p$ we have
        $x=y$.
    \begin{subproof}
        Fix $x,y,p$.
        Assume $x,y\mathrel{f}p$.
        Hence $x,y\mathrel{\indexedSumInjection{\{i\}}{A}{i}}p$.
        Follows by \auto{indexed_sum_injection,pair_eq_iff}.
    \end{subproof}
    Hence $f$ is injective by \obv{injective}.
    Show that for every $p\in\indexedSum{\{i\}}{A}$ there exists
        $x\in A(i)$ such that
        $f(x)=p$.
    \begin{subproof}
        Fix $p\in\indexedSum{\{i\}}{A}$.
        Hence $p\in\{i\}\times\unions{\ran{A}}$ by
            \obv{indexed_sum}.
        Take $j,x$ such that $p=(j,x)$ by
            \auto{times_elem_is_tuple}.
        Hence $(j,x)\in\indexedSum{\{i\}}{A}$.
        Hence $j\in\{i\}$ and $x\in A(j)$ by
            \obv{indexed_sum_iff}.
        Hence $j=i$ by \obv{singleton_iff}.
        Hence $x\in A(i)$.
        $(\indexedSumInjection{\{i\}}{A}{i})(x)=(i,x)$ by
            \obv{indexed_sum_injection_apply}.
        Hence $f(x)=p$.
    \end{subproof}
    Hence $f\in\surj{A(i)}{\indexedSum{\{i\}}{A}}$ by \obv{surj}.
    Hence $f\in\bijections{A(i)}{\indexedSum{\{i\}}{A}}$ by
        \obv{bijections}.
\end{proof}
