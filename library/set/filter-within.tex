\section{Filters within subsets}

\import{set/filter-tendsto.tex}

A filter within a subset keeps the ambient carrier fixed while requiring the
subset to be large.

\begin{definition}\label{filter_within}
    $\filterWithin{S}{F}{A}
        =\binaryCommonFilterRefinement{S}{F}{\principalfilter{S}{A}}$.
\end{definition}

\begin{proposition}\label{filter_within_is_filter}
    Let $F$ be a filter on $S$.
    Suppose $A\subseteq S$.
    Then $\filterWithin{S}{F}{A}$ is a filter on $S$.
\end{proposition}
\begin{proof}
    $\principalfilter{S}{A}$ is a filter on $S$ by
        \obv{principalfilter_is_filter}.
    $\binaryCommonFilterRefinement{S}{F}{\principalfilter{S}{A}}$ is a
        filter on $S$ by \obv{binary_common_filter_refinement_is_filter}.
    $\filterWithin{S}{F}{A}
        =\binaryCommonFilterRefinement{S}{F}{\principalfilter{S}{A}}$ by
        \obv{filter_within}.
\end{proof}

\begin{theorem}\label{filter_within_iff}
    Let $F$ be a filter on $S$.
    Suppose $A\subseteq S$.
    Then $B\in\filterWithin{S}{F}{A}$ iff
        $B\subseteq S$ and there exists $C\in F$ such that
        $C\inter A\subseteq B$.
\end{theorem}
\begin{proof}
    $\principalfilter{S}{A}$ is a filter on $S$ by
        \obv{principalfilter_is_filter}.
    $B\in\binaryCommonFilterRefinement{S}{F}{\principalfilter{S}{A}}$ iff
        $B\subseteq S$ and there exist $C,D$ such that $C\in F$ and
        $D\in\principalfilter{S}{A}$ and $C\inter D\subseteq B$ by
        \obv{binary_common_filter_refinement_iff}.
    Show that if $B\in\filterWithin{S}{F}{A}$, then
        $B\subseteq S$ and there exists $C\in F$ such that
        $C\inter A\subseteq B$.
    \begin{subproof}
        Assume $B\in\filterWithin{S}{F}{A}$.
        Hence $B\in\binaryCommonFilterRefinement{S}{F}{
            \principalfilter{S}{A}}$ by \obv{filter_within}.
        Hence $B\subseteq S$ and there exist $C,D$ such that $C\in F$ and
            $D\in\principalfilter{S}{A}$ and $C\inter D\subseteq B$.
        Take $C,D$ such that $C\in F$ and
            $D\in\principalfilter{S}{A}$ and $C\inter D\subseteq B$.
        $A\subseteq D$ by \obv{principalfilter_iff}.
        Hence $C\inter A\subseteq C\inter D$ by \obv{inter,subseteq}.
        Hence $C\inter A\subseteq B$ by \obv{subseteq_transitive}.
    \end{subproof}
    Show that if $B\subseteq S$ and there exists $C\in F$ such that
        $C\inter A\subseteq B$, then $B\in\filterWithin{S}{F}{A}$.
    \begin{subproof}
        Assume $B\subseteq S$ and there exists $C\in F$ such that
            $C\inter A\subseteq B$.
        Take $C\in F$ such that $C\inter A\subseteq B$.
        $A\in\principalfilter{S}{A}$ by \obv{principalfilter_bottom}.
        Hence $B\in\binaryCommonFilterRefinement{S}{F}{
            \principalfilter{S}{A}}$ by \auto.
        Follows by \obv{filter_within}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{proposition}\label{filter_within_carrier}
    Let $F$ be a filter on $S$.
    Then $\filterWithin{S}{F}{S}=F$.
\end{proposition}
\begin{proof}
    Show that for every $B$ we have $B\in\filterWithin{S}{F}{S}$ iff
        $B\in F$.
    \begin{subproof}
        Fix $B$.
        $B\in\filterWithin{S}{F}{S}$ iff
            $B\subseteq S$ and there exists $C\in F$ such that
            $C\inter S\subseteq B$ by \obv{filter_within_iff}.
        Show that if $B\in\filterWithin{S}{F}{S}$, then $B\in F$.
        \begin{subproof}
            Assume $B\in\filterWithin{S}{F}{S}$.
            Hence $B\subseteq S$ and there exists $C\in F$ such that
                $C\inter S\subseteq B$.
            Take $C\in F$ such that $C\inter S\subseteq B$.
            $C\subseteq S$ by \obv{filter}.
            Hence $C\inter S=C$ by \obv{inter_absorb_supseteq_right}.
            Hence $C\subseteq B$.
            Follows by \obv{filter}.
        \end{subproof}
        Show that if $B\in F$, then $B\in\filterWithin{S}{F}{S}$.
        \begin{subproof}
            Assume $B\in F$.
            Hence $B\subseteq S$ by \obv{filter}.
            $B\inter S=B$ by \obv{inter_absorb_supseteq_right}.
            Hence $B\inter S\subseteq B$ by \obv{subseteq_refl}.
        \end{subproof}
        Follows by \auto.
    \end{subproof}
    Follows by set extensionality.
\end{proof}

\begin{proposition}\label{filter_within_eventual_set}
    Let $F$ be a filter on $S$.
    Suppose $A\in F$.
    Then $\filterWithin{S}{F}{A}=F$.
\end{proposition}
\begin{proof}
    $A\subseteq S$ by \obv{filter}.
    $\filterWithin{S}{F}{A}$ is a filter on $S$ by
        \obv{filter_within_is_filter}.
    Show that $\filterWithin{S}{F}{A}\subseteq F$.
    \begin{subproof}
        It suffices to show that every element of
            $\filterWithin{S}{F}{A}$ is an element of $F$ by
            \obv{subseteq}.
        Fix $B\in\filterWithin{S}{F}{A}$.
        $B\in\filterWithin{S}{F}{A}$ iff
            $B\subseteq S$ and there exists $C\in F$ such that
            $C\inter A\subseteq B$ by \obv{filter_within_iff}.
        Hence $B\subseteq S$ and there exists $C\in F$ such that
            $C\inter A\subseteq B$.
        Take $C\in F$ such that $C\inter A\subseteq B$.
        $C\inter A\in F$ by \obv{filter}.
        Follows by \obv{filter}.
    \end{subproof}
    Show that $F\subseteq\filterWithin{S}{F}{A}$.
    \begin{subproof}
        It suffices to show that every element of $F$ is an element of
            $\filterWithin{S}{F}{A}$ by \obv{subseteq}.
        Fix $B\in F$.
        $B\subseteq S$ by \obv{filter}.
        $B\inter A\subseteq B$ by \obv{inter_lower_left}.
        $B\in\filterWithin{S}{F}{A}$ iff
            $B\subseteq S$ and there exists $C\in F$ such that
            $C\inter A\subseteq B$ by \obv{filter_within_iff}.
    \end{subproof}
    Follows by \obv{subseteq_antisymmetric}.
\end{proof}

\begin{theorem}\label{filter_within_intersection}
    Let $F$ be a filter on $S$.
    Suppose $A,B\subseteq S$.
    Then $\filterWithin{S}{\filterWithin{S}{F}{A}}{B}
        =\filterWithin{S}{F}{A\inter B}$.
\end{theorem}
\begin{proof}
    $\filterWithin{S}{F}{A}$ is a filter on $S$ by
        \obv{filter_within_is_filter}.
    $A\inter B\subseteq S$ by \obv{inter_subseteq}.
    Show that for every $C$ we have
        $C\in\filterWithin{S}{\filterWithin{S}{F}{A}}{B}$ iff
        $C\in\filterWithin{S}{F}{A\inter B}$.
    \begin{subproof}
        Fix $C$.
        $C\in\filterWithin{S}{\filterWithin{S}{F}{A}}{B}$ iff
            $C\subseteq S$ and there exists
            $D\in\filterWithin{S}{F}{A}$ such that
            $D\inter B\subseteq C$ by \obv{filter_within_iff}.
        $C\in\filterWithin{S}{F}{A\inter B}$ iff
            $C\subseteq S$ and there exists $E\in F$ such that
            $E\inter(A\inter B)\subseteq C$ by \obv{filter_within_iff}.
        Show that if
            $C\in\filterWithin{S}{\filterWithin{S}{F}{A}}{B}$, then
            $C\in\filterWithin{S}{F}{A\inter B}$.
        \begin{subproof}
            Assume $C\in\filterWithin{S}{\filterWithin{S}{F}{A}}{B}$.
            Hence $C\subseteq S$ and there exists
                $D\in\filterWithin{S}{F}{A}$ such that
                $D\inter B\subseteq C$.
            Take $D\in\filterWithin{S}{F}{A}$ such that
                $D\inter B\subseteq C$.
            $D\in\filterWithin{S}{F}{A}$ iff
                $D\subseteq S$ and there exists $E\in F$ such that
                $E\inter A\subseteq D$ by \obv{filter_within_iff}.
            Hence $D\subseteq S$ and there exists $E\in F$ such that
                $E\inter A\subseteq D$.
            Take $E\in F$ such that $E\inter A\subseteq D$.
            $E\inter(A\inter B)\subseteq D\inter B$ by
                \auto{inter,subseteq}.
            Hence $E\inter(A\inter B)\subseteq C$ by
                \obv{subseteq_transitive}.
        \end{subproof}
        Show that if $C\in\filterWithin{S}{F}{A\inter B}$, then
            $C\in\filterWithin{S}{\filterWithin{S}{F}{A}}{B}$.
        \begin{subproof}
            Assume $C\in\filterWithin{S}{F}{A\inter B}$.
            Hence $C\subseteq S$ and there exists $E\in F$ such that
                $E\inter(A\inter B)\subseteq C$.
            Take $E\in F$ such that $E\inter(A\inter B)\subseteq C$.
            Let $D=E\inter A$.
            $E\subseteq S$ by \obv{filter}.
            Hence $D\subseteq S$ by \auto{inter_subseteq}.
            $E\inter A\subseteq D$ by \obv{subseteq_refl}.
            $D\in\filterWithin{S}{F}{A}$ iff
                $D\subseteq S$ and there exists $K\in F$ such that
                $K\inter A\subseteq D$ by \obv{filter_within_iff}.
            Hence $D\in\filterWithin{S}{F}{A}$.
            $D\inter B\subseteq C$ by \auto{inter,subseteq}.
        \end{subproof}
        Follows by \auto.
    \end{subproof}
    Follows by set extensionality.
\end{proof}

\begin{proposition}\label{filter_within_emptyset}
    Let $F$ be a filter on $S$.
    Then $\filterWithin{S}{F}{\emptyset}=\pow{S}$.
\end{proposition}
\begin{proof}
    $\emptyset\subseteq S$ by \obv{emptyset_subseteq}.
    Show that for every $B$ we have
        $B\in\filterWithin{S}{F}{\emptyset}$ iff $B\in\pow{S}$.
    \begin{subproof}
        Fix $B$.
        $B\in\filterWithin{S}{F}{\emptyset}$ iff
            $B\subseteq S$ and there exists $C\in F$ such that
            $C\inter\emptyset\subseteq B$ by \obv{filter_within_iff}.
        If $B\in\filterWithin{S}{F}{\emptyset}$, then $B\subseteq S$ by
            \auto.
        If $B\subseteq S$, then $B\in\pow{S}$ by \obv{powerset_intro}.
        If $B\in\pow{S}$, then $B\subseteq S$ by \obv{powerset_elim}.
        $S\in F$ by \obv{filter}.
        $S\inter\emptyset=\emptyset$ by \obv{inter_emptyset}.
        $\emptyset\subseteq B$ by \obv{emptyset_subseteq}.
        Hence $S\inter\emptyset\subseteq B$.
        If $B\subseteq S$, then $B\in\filterWithin{S}{F}{\emptyset}$.
        Follows by \auto.
    \end{subproof}
    Follows by set extensionality.
\end{proof}

\begin{theorem}\label{filter_within_is_proper_iff_frequent}
    Let $F$ be a filter on $S$.
    Suppose $A\subseteq S$.
    Then $\filterWithin{S}{F}{A}$ is a proper filter on $S$ iff
        $A$ is frequent along $F$ on $S$.
\end{theorem}
\begin{proof}
    $\principalfilter{S}{A}$ is a filter on $S$ by
        \obv{principalfilter_is_filter}.
    $A$ is frequent along $F$ on $S$ iff
        $F$ meshes with $\principalfilter{S}{A}$ on $S$ by
        \obv{filter_frequent_iff_meshes_principal}.
    $F$ meshes with $\principalfilter{S}{A}$ on $S$ iff
        $\binaryCommonFilterRefinement{S}{F}{\principalfilter{S}{A}}$
        is a proper filter on $S$ by
        \obv{filters_mesh_iff_common_refinement_proper}.
    $\filterWithin{S}{F}{A}
        =\binaryCommonFilterRefinement{S}{F}{\principalfilter{S}{A}}$ by
        \obv{filter_within}.
\end{proof}

\subsection{Changing the carrier by trace}

\begin{definition}\label{subset_inclusion}
    $\subsetInclusion{S}{A}=\identity{A}$.
\end{definition}

\begin{proposition}\label{subset_inclusion_type}
    Suppose $A\subseteq S$.
    Then $\subsetInclusion{S}{A}\in\funs{A}{S}$.
\end{proposition}
\begin{proof}
    $\identity{A}\in\funs{A}{A}$ by \obv{id_elem_funs}.
    Hence $\identity{A}\in\funs{A}{S}$ by \obv{funs_weaken_codom}.
    $\subsetInclusion{S}{A}=\identity{A}$ by \obv{subset_inclusion}.
\end{proof}

\begin{proposition}\label{subset_inclusion_apply}
    Suppose $A\subseteq S$ and $x\in A$.
    Then $(\subsetInclusion{S}{A})(x)=x$.
\end{proposition}
\begin{proof}
    $\subsetInclusion{S}{A}=\identity{A}$ by \obv{subset_inclusion}.
    $(\identity{A})(x)=x$ by \obv{id_apply}.
\end{proof}

\begin{proposition}\label{subset_inclusion_range}
    Suppose $A\subseteq S$.
    Then $\ran{\subsetInclusion{S}{A}}=A$.
\end{proposition}
\begin{proof}
    $\subsetInclusion{S}{A}=\identity{A}$ by \obv{subset_inclusion}.
    $\ran{\identity{A}}=A$ by \obv{id_ran}.
\end{proof}

\begin{proposition}\label{subset_inclusion_preimage}
    Suppose $A\subseteq S$.
    Then $\preimg{\subsetInclusion{S}{A}}{B}=A\inter B$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{subset_inclusion,preimg_iff,id_iff,inter,setext}.
\end{proof}

\begin{definition}\label{filter_trace}
    $\filterTrace{S}{F}{A}
        =\filterPullback{A}{S}{\subsetInclusion{S}{A}}{F}$.
\end{definition}

\begin{proposition}\label{filter_trace_is_filter}
    Let $F$ be a filter on $S$.
    Suppose $A\subseteq S$.
    Then $\filterTrace{S}{F}{A}$ is a filter on $A$.
\end{proposition}
\begin{proof}
    $\subsetInclusion{S}{A}\in\funs{A}{S}$ by
        \obv{subset_inclusion_type}.
    $\filterPullback{A}{S}{\subsetInclusion{S}{A}}{F}$ is a filter on $A$
        by \obv{filter_pullback_is_filter}.
    $\filterTrace{S}{F}{A}
        =\filterPullback{A}{S}{\subsetInclusion{S}{A}}{F}$ by
        \obv{filter_trace}.
\end{proof}

\begin{theorem}\label{filter_trace_iff}
    Let $F$ be a filter on $S$.
    Suppose $A\subseteq S$.
    Then $D\in\filterTrace{S}{F}{A}$ iff
        $D\subseteq A$ and there exists $B\in F$ such that
        $A\inter B\subseteq D$.
\end{theorem}
\begin{proof}
    $\subsetInclusion{S}{A}\in\funs{A}{S}$ by
        \obv{subset_inclusion_type}.
    $D\in\filterPullback{A}{S}{\subsetInclusion{S}{A}}{F}$ iff
        $D\subseteq A$ and there exists $B\in F$ such that
        $\preimg{\subsetInclusion{S}{A}}{B}\subseteq D$ by
        \obv{filter_pullback_iff}.
    $\filterTrace{S}{F}{A}
        =\filterPullback{A}{S}{\subsetInclusion{S}{A}}{F}$ by
        \obv{filter_trace}.
    Show that if $D\in\filterTrace{S}{F}{A}$, then
        $D\subseteq A$ and there exists $B\in F$ such that
        $A\inter B\subseteq D$.
    \begin{subproof}
        Assume $D\in\filterTrace{S}{F}{A}$.
        Hence $D\in\filterPullback{A}{S}{\subsetInclusion{S}{A}}{F}$.
        Hence $D\subseteq A$ and there exists $B\in F$ such that
            $\preimg{\subsetInclusion{S}{A}}{B}\subseteq D$.
        Take $B\in F$ such that
            $\preimg{\subsetInclusion{S}{A}}{B}\subseteq D$.
        $\preimg{\subsetInclusion{S}{A}}{B}=A\inter B$ by
            \obv{subset_inclusion_preimage}.
        Hence $A\inter B\subseteq D$.
    \end{subproof}
    Show that if $D\subseteq A$ and there exists $B\in F$ such that
        $A\inter B\subseteq D$, then $D\in\filterTrace{S}{F}{A}$.
    \begin{subproof}
        Assume $D\subseteq A$ and there exists $B\in F$ such that
            $A\inter B\subseteq D$.
        Take $B\in F$ such that $A\inter B\subseteq D$.
        $\preimg{\subsetInclusion{S}{A}}{B}=A\inter B$ by
            \obv{subset_inclusion_preimage}.
        Hence $\preimg{\subsetInclusion{S}{A}}{B}\subseteq D$.
        Hence $D\in\filterPullback{A}{S}{\subsetInclusion{S}{A}}{F}$.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{theorem}\label{filter_trace_pushforward_eq_within}
    Let $F$ be a filter on $S$.
    Suppose $A\subseteq S$.
    Then $\filterPushforward{A}{S}{\subsetInclusion{S}{A}}{
        \filterTrace{S}{F}{A}}=\filterWithin{S}{F}{A}$.
\end{theorem}
\begin{proof}
    $\subsetInclusion{S}{A}\in\funs{A}{S}$ by
        \obv{subset_inclusion_type}.
    $\filterTrace{S}{F}{A}$ is a filter on $A$ by
        \obv{filter_trace_is_filter}.
    Show that for every $C$ we have
        $C\in\filterPushforward{A}{S}{\subsetInclusion{S}{A}}{
            \filterTrace{S}{F}{A}}$ iff
        $C\in\filterWithin{S}{F}{A}$.
    \begin{subproof}
        Fix $C$.
        $C\in\filterPushforward{A}{S}{\subsetInclusion{S}{A}}{
            \filterTrace{S}{F}{A}}$ iff
            $C\subseteq S$ and
            $\preimg{\subsetInclusion{S}{A}}{C}
                \in\filterTrace{S}{F}{A}$ by
            \obv{filter_pushforward_iff}.
        $C\in\filterWithin{S}{F}{A}$ iff
            $C\subseteq S$ and there exists $E\in F$ such that
            $E\inter A\subseteq C$ by \obv{filter_within_iff}.
        $\preimg{\subsetInclusion{S}{A}}{C}=A\inter C$ by
            \obv{subset_inclusion_preimage}.
        $\preimg{\subsetInclusion{S}{A}}{C}
            \in\filterTrace{S}{F}{A}$ iff
            $\preimg{\subsetInclusion{S}{A}}{C}\subseteq A$ and
            there exists $D\in F$ such that
            $A\inter D\subseteq\preimg{\subsetInclusion{S}{A}}{C}$ by
            \obv{filter_trace_iff}.
        Show that if
            $C\in\filterPushforward{A}{S}{\subsetInclusion{S}{A}}{
                \filterTrace{S}{F}{A}}$, then
            $C\in\filterWithin{S}{F}{A}$.
        \begin{subproof}
            Assume $C\in\filterPushforward{A}{S}{
                \subsetInclusion{S}{A}}{\filterTrace{S}{F}{A}}$.
            Hence $C\subseteq S$ and
                $\preimg{\subsetInclusion{S}{A}}{C}
                    \in\filterTrace{S}{F}{A}$.
            Hence there exists $D\in F$ such that
                $A\inter D\subseteq
                    \preimg{\subsetInclusion{S}{A}}{C}$ by
                \auto.
            Take $K$ such that $K\in F$ and $A\inter K\subseteq
                \preimg{\subsetInclusion{S}{A}}{C}$ by \auto.
            Hence $K\inter A\subseteq C$ by
                \auto{inter,subseteq}.
        \end{subproof}
        Show that if $C\in\filterWithin{S}{F}{A}$, then
            $C\in\filterPushforward{A}{S}{\subsetInclusion{S}{A}}{
                \filterTrace{S}{F}{A}}$.
        \begin{subproof}
            Assume $C\in\filterWithin{S}{F}{A}$.
            Hence $C\subseteq S$ and there exists $E\in F$ such that
                $E\inter A\subseteq C$.
            Take $K$ such that $K\in F$ and $K\inter A\subseteq C$.
            $A\inter C\subseteq A$ by \obv{inter_lower_left}.
            $A\inter K\subseteq A\inter C$ by
                \obv{inter,subseteq}.
            Hence $\preimg{\subsetInclusion{S}{A}}{C}\subseteq A$ and
                $A\inter K\subseteq
                    \preimg{\subsetInclusion{S}{A}}{C}$.
            Hence $\preimg{\subsetInclusion{S}{A}}{C}
                \in\filterTrace{S}{F}{A}$ by \auto.
        \end{subproof}
        Follows by \auto.
    \end{subproof}
    Follows by set extensionality.
\end{proof}

\begin{theorem}\label{filter_trace_is_proper_iff_frequent}
    Let $F$ be a filter on $S$.
    Suppose $A\subseteq S$.
    Then $\filterTrace{S}{F}{A}$ is a proper filter on $A$ iff
        $A$ is frequent along $F$ on $S$.
\end{theorem}
\begin{proof}
    $\subsetInclusion{S}{A}\in\funs{A}{S}$ by
        \obv{subset_inclusion_type}.
    $\ran{\subsetInclusion{S}{A}}=A$ by
        \obv{subset_inclusion_range}.
    $\filterPullback{A}{S}{\subsetInclusion{S}{A}}{F}$ is a proper filter
        on $A$ iff
        $F$ meshes with
            $\principalfilter{S}{\ran{\subsetInclusion{S}{A}}}$ on $S$ by
        \obv{filter_pullback_is_proper_iff_meshes_range}.
    $A$ is frequent along $F$ on $S$ iff
        $F$ meshes with $\principalfilter{S}{A}$ on $S$ by
        \obv{filter_frequent_iff_meshes_principal}.
    $\filterTrace{S}{F}{A}
        =\filterPullback{A}{S}{\subsetInclusion{S}{A}}{F}$ by
        \obv{filter_trace}.
\end{proof}
