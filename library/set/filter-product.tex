\subsection{Projection pullbacks}

\begin{theorem}\label{first_projection_filter_pullback_iff}
    Let $F$ be a filter on $S$.
    Then $W\in\filterPullback{S\times T}{S}{
        \firstProductProjection{S}{T}}{F}$ iff
        $W\subseteq S\times T$ and there exists $A\in F$ such that
        $A\times T\subseteq W$.
\end{theorem}
\begin{proof}
    $\firstProductProjection{S}{T}\in\funs{S\times T}{S}$ by
        \obv{first_product_projection_type}.
    $W\in\filterPullback{S\times T}{S}{
        \firstProductProjection{S}{T}}{F}$ iff
        $W\subseteq S\times T$ and there exists $A\in F$ such that
        $\preimg{\firstProductProjection{S}{T}}{A}\subseteq W$ by
        \obv{filter_pullback_iff}.
    Show that if $W\in\filterPullback{S\times T}{S}{
        \firstProductProjection{S}{T}}{F}$, then
        $W\subseteq S\times T$ and there exists $A\in F$ such that
        $A\times T\subseteq W$.
    \begin{subproof}
        Assume $W\in\filterPullback{S\times T}{S}{
            \firstProductProjection{S}{T}}{F}$.
        Hence $W\subseteq S\times T$ and there exists $A\in F$ such that
            $\preimg{\firstProductProjection{S}{T}}{A}\subseteq W$.
        Take $A\in F$ such that
            $\preimg{\firstProductProjection{S}{T}}{A}\subseteq W$.
        $A\subseteq S$ by \obv{filter}.
        Hence $\preimg{\firstProductProjection{S}{T}}{A}=A\times T$ by
            \obv{first_product_projection_preimage}.
        Hence $A\times T\subseteq W$.
    \end{subproof}
    Show that if $W\subseteq S\times T$ and there exists $A\in F$ such
        that $A\times T\subseteq W$, then
        $W\in\filterPullback{S\times T}{S}{
            \firstProductProjection{S}{T}}{F}$.
    \begin{subproof}
        Assume $W\subseteq S\times T$ and there exists $A\in F$ such that
            $A\times T\subseteq W$.
        Take $A\in F$ such that $A\times T\subseteq W$.
        $A\subseteq S$ by \obv{filter}.
        Hence $\preimg{\firstProductProjection{S}{T}}{A}=A\times T$ by
            \obv{first_product_projection_preimage}.
        Hence $\preimg{\firstProductProjection{S}{T}}{A}\subseteq W$.
    \end{subproof}
    Follows by \auto.
\end{proof}

\subsection{The product filter}

\begin{definition}\label{product_filter}
    $\productFilter{S}{T}{F}{G}
        =\binaryCommonFilterRefinement{S\times T}{
            \filterPullback{S\times T}{S}{
                \firstProductProjection{S}{T}}{F}}{
            \filterPullback{S\times T}{T}{
                \secondProductProjection{S}{T}}{G}}$.
\end{definition}

\begin{proposition}\label{product_filter_is_filter}
    Let $F$ be a filter on $S$ and $G$ be a filter on $T$.
    Then $\productFilter{S}{T}{F}{G}$ is a filter on $S\times T$.
\end{proposition}
\begin{proof}
    $\firstProductProjection{S}{T}\in\funs{S\times T}{S}$ and
        $\secondProductProjection{S}{T}\in\funs{S\times T}{T}$ by
        \obv{first_product_projection_type,second_product_projection_type}.
    $\filterPullback{S\times T}{S}{
            \firstProductProjection{S}{T}}{F}$ and
        $\filterPullback{S\times T}{T}{
            \secondProductProjection{S}{T}}{G}$ are filters on
        $S\times T$ by \obv{filter_pullback_is_filter}.
    Hence $\binaryCommonFilterRefinement{S\times T}{
            \filterPullback{S\times T}{S}{
                \firstProductProjection{S}{T}}{F}}{
            \filterPullback{S\times T}{T}{
                \secondProductProjection{S}{T}}{G}}$ is a filter on
        $S\times T$ by \obv{binary_common_filter_refinement_is_filter}.
    $\productFilter{S}{T}{F}{G}
        =\binaryCommonFilterRefinement{S\times T}{
            \filterPullback{S\times T}{S}{
                \firstProductProjection{S}{T}}{F}}{
            \filterPullback{S\times T}{T}{
                \secondProductProjection{S}{T}}{G}}$ by
        \obv{product_filter}.
\end{proof}

\begin{theorem}\label{product_filter_iff}
    Let $F$ be a filter on $S$ and $G$ be a filter on $T$.
    Then $W\in\productFilter{S}{T}{F}{G}$ iff
        $W\subseteq S\times T$ and there exist $A,B$ such that
        $A\in F$ and $B\in G$ and $A\times B\subseteq W$.
\end{theorem}
\begin{proof}
    Let $P=\filterPullback{S\times T}{S}{
        \firstProductProjection{S}{T}}{F}$.
    Let $Q=\filterPullback{S\times T}{T}{
        \secondProductProjection{S}{T}}{G}$.
    $\firstProductProjection{S}{T}\in\funs{S\times T}{S}$ and
        $\secondProductProjection{S}{T}\in\funs{S\times T}{T}$ by
        \obv{first_product_projection_type,second_product_projection_type}.
    Hence $P$ is a filter on $S\times T$ and $Q$ is a filter on
        $S\times T$ by
        \auto{filter_pullback_is_filter}.
    $W\in\binaryCommonFilterRefinement{S\times T}{P}{Q}$ iff
        $W\subseteq S\times T$ and there exist $C,D$ such that
        $C\in P$ and $D\in Q$ and $C\inter D\subseteq W$ by
        \obv{binary_common_filter_refinement_iff}.
    $\productFilter{S}{T}{F}{G}
        =\binaryCommonFilterRefinement{S\times T}{P}{Q}$ by
        \obv{product_filter}.
    Show that if $W\in\productFilter{S}{T}{F}{G}$, then
        $W\subseteq S\times T$ and there exist $A,B$ such that
        $A\in F$ and $B\in G$ and $A\times B\subseteq W$.
    \begin{subproof}
        Assume $W\in\productFilter{S}{T}{F}{G}$.
        Hence $W\in\binaryCommonFilterRefinement{S\times T}{P}{Q}$.
        Hence $W\subseteq S\times T$ and there exist $C,D$ such that
            $C\in P$ and $D\in Q$ and $C\inter D\subseteq W$.
        Take $C,D$ such that $C\in P$ and $D\in Q$ and
            $C\inter D\subseteq W$.
        $C\in\filterPullback{S\times T}{S}{
            \firstProductProjection{S}{T}}{F}$.
        $C\in\filterPullback{S\times T}{S}{
            \firstProductProjection{S}{T}}{F}$ iff
            $C\subseteq S\times T$ and there exists $X\in F$ such that
            $X\times T\subseteq C$ by
            \obv{first_projection_filter_pullback_iff}.
        Hence $C\subseteq S\times T$ and there exists $X\in F$ such that
            $X\times T\subseteq C$.
        Take $E$ such that $E\in F$ and $E\times T\subseteq C$.
        $D\in\filterPullback{S\times T}{T}{
            \secondProductProjection{S}{T}}{G}$.
        $D\in\filterPullback{S\times T}{T}{
            \secondProductProjection{S}{T}}{G}$ iff
            $D\subseteq S\times T$ and there exists $Y\in G$ such that
            $S\times Y\subseteq D$ by
            \obv{second_projection_filter_pullback_iff}.
        Hence $D\subseteq S\times T$ and there exists $Y\in G$ such that
            $S\times Y\subseteq D$.
        Take $K$ such that $K\in G$ and $S\times K\subseteq D$.
        $E\subseteq S$ and $K\subseteq T$ by \obv{filter}.
        $E\times K\subseteq(E\times T)\inter(S\times K)$ by
            \auto{times,inter,subseteq}.
        Hence $E\times K\subseteq C\inter D$ by
            \auto{inter,subseteq}.
        Hence $E\times K\subseteq W$ by \obv{subseteq_transitive}.
        Follows by \auto.
    \end{subproof}
    Show that if $W\subseteq S\times T$ and there exist $A,B$ such that
        $A\in F$ and $B\in G$ and $A\times B\subseteq W$, then
        $W\in\productFilter{S}{T}{F}{G}$.
    \begin{subproof}
        Assume $W\subseteq S\times T$ and there exist $A,B$ such that
            $A\in F$ and $B\in G$ and $A\times B\subseteq W$.
        Take $A,B$ such that $A\in F$ and $B\in G$ and
            $A\times B\subseteq W$.
        Let $C=A\times T$.
        Let $D=S\times B$.
        $A\subseteq S$ and $B\subseteq T$ by \obv{filter}.
        $C\subseteq S\times T$ and $D\subseteq S\times T$ by
            \auto{times,subseteq}.
        $C\in\filterPullback{S\times T}{S}{
            \firstProductProjection{S}{T}}{F}$ iff
            $C\subseteq S\times T$ and there exists $E\in F$ such that
            $E\times T\subseteq C$ by
            \obv{first_projection_filter_pullback_iff}.
        Hence $C\in\filterPullback{S\times T}{S}{
            \firstProductProjection{S}{T}}{F}$ by
            \auto{subseteq_refl}.
        Hence $C\in P$.
        $D\in\filterPullback{S\times T}{T}{
            \secondProductProjection{S}{T}}{G}$ iff
            $D\subseteq S\times T$ and there exists $K\in G$ such that
            $S\times K\subseteq D$ by
            \obv{second_projection_filter_pullback_iff}.
        Hence $D\in\filterPullback{S\times T}{T}{
            \secondProductProjection{S}{T}}{G}$ by
            \auto{subseteq_refl}.
        Hence $D\in Q$.
        $A\inter S=A$ by \obv{inter_absorb_supseteq_right}.
        $T\inter B=B$ by \obv{inter_absorb_supseteq_left}.
        $(A\inter S)\times(T\inter B)
            =(A\times T)\inter(S\times B)$ by \obv{inter_times}.
        Hence $C\inter D=A\times B$.
        Hence $C\inter D\subseteq W$.
        Hence $W\in\binaryCommonFilterRefinement{S\times T}{P}{Q}$ by
            \auto.
    \end{subproof}
    Follows by \auto.
\end{proof}

\subsection{Structural laws}

\begin{theorem}\label{product_filter_is_proper_iff}
    Let $F$ be a filter on $S$ and $G$ be a filter on $T$.
    Then $\productFilter{S}{T}{F}{G}$ is a proper filter on $S\times T$
        iff $F$ is a proper filter on $S$ and
        $G$ is a proper filter on $T$.
\end{theorem}
\begin{proof}
    $\productFilter{S}{T}{F}{G}$ is a filter on $S\times T$ by
        \obv{product_filter_is_filter}.
    $\emptyset\in\productFilter{S}{T}{F}{G}$ iff
        $\emptyset\subseteq S\times T$ and there exist $A,B$ such that
        $A\in F$ and $B\in G$ and $A\times B\subseteq\emptyset$ by
        \obv{product_filter_iff}.
    Show that if $\productFilter{S}{T}{F}{G}$ is a proper filter on
        $S\times T$, then $F$ is a proper filter on $S$ and
        $G$ is a proper filter on $T$.
    \begin{subproof}
        Assume $\productFilter{S}{T}{F}{G}$ is a proper filter on
            $S\times T$.
        Show that $\emptyset\notin F$.
        \begin{subproof}
            Suppose not.
            Hence $\emptyset\in F$.
            $T\in G$ by \obv{filter}.
            $\emptyset\times T=\emptyset$ by \obv{times_empty_left}.
            Hence $\emptyset\times T\subseteq\emptyset$ by
                \obv{subseteq_refl}.
            Hence $\emptyset\in\productFilter{S}{T}{F}{G}$ by
                \auto{emptyset_subseteq}.
            Contradiction by \obv{proper_filter}.
        \end{subproof}
        Hence $F$ is a proper filter on $S$ by \obv{proper_filter}.
        Show that $\emptyset\notin G$.
        \begin{subproof}
            Suppose not.
            Hence $\emptyset\in G$.
            $S\in F$ by \obv{filter}.
            $S\times\emptyset=\emptyset$ by \obv{times_empty_right}.
            Hence $S\times\emptyset\subseteq\emptyset$ by
                \obv{subseteq_refl}.
            Hence $\emptyset\in\productFilter{S}{T}{F}{G}$ by
                \auto{emptyset_subseteq}.
            Contradiction by \obv{proper_filter}.
        \end{subproof}
        Hence $G$ is a proper filter on $T$ by \obv{proper_filter}.
    \end{subproof}
    Show that if $F$ is a proper filter on $S$ and
        $G$ is a proper filter on $T$, then
        $\productFilter{S}{T}{F}{G}$ is a proper filter on $S\times T$.
    \begin{subproof}
        Assume $F$ is a proper filter on $S$ and
            $G$ is a proper filter on $T$.
        Show that $\emptyset\notin\productFilter{S}{T}{F}{G}$.
        \begin{subproof}
            Suppose not.
            Hence $\emptyset\in\productFilter{S}{T}{F}{G}$.
            Hence there exist $A,B$ such that $A\in F$ and $B\in G$ and
                $A\times B\subseteq\emptyset$ by \auto.
            Take $K,L$ such that $K\in F$ and $L\in G$ and
                $K\times L\subseteq\emptyset$ by \auto.
            $\emptyset\subseteq K\times L$ by \obv{emptyset_subseteq}.
            Hence $K\times L=\emptyset$ by \obv{subseteq_antisymmetric}.
            $K\neq\emptyset$ and $L\neq\emptyset$ by
                \obv{proper_filter}.
            Hence $K$ and $L$ are inhabited by \obv{nonempty_inhabited}.
            Take $x$ such that $x\in K$.
            Take $y$ such that $y\in L$.
            Hence $(x,y)\in K\times L$ by \obv{times_tuple_intro}.
            Hence $(x,y)\in\emptyset$.
            Contradiction by \obv{emptyset}.
        \end{subproof}
        Follows by \obv{proper_filter}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{theorem}\label{self_product_filter_refines_iff_squares}
    Let $F$ be a filter on $S$ and $G$ be a filter on $S\times S$.
    Then $\productFilter{S}{S}{F}{F}$ refines $G$ on $S\times S$ iff
        for every $U\in G$ there exists $A\in F$ such that
            $A\times A\subseteq U$.
\end{theorem}
\begin{proof}
    $\productFilter{S}{S}{F}{F}$ is a filter on $S\times S$ by
        \obv{product_filter_is_filter}.

    Show that if $\productFilter{S}{S}{F}{F}$ refines $G$ on
        $S\times S$, then for every $U\in G$ there exists $A\in F$
        such that $A\times A\subseteq U$.
    \begin{subproof}
        Assume $\productFilter{S}{S}{F}{F}$ refines $G$ on
            $S\times S$.
        $G\subseteq\productFilter{S}{S}{F}{F}$ by
            \obv{filter_refines}.
        Fix $U\in G$.
        Hence $U\in\productFilter{S}{S}{F}{F}$ by \obv{subseteq}.
        $U\in\productFilter{S}{S}{F}{F}$ iff
            $U\subseteq S\times S$ and there exist $X,Y$ such that
            $X\in F$ and $Y\in F$ and $X\times Y\subseteq U$ by
            \obv{product_filter_iff}.
        Take $E,K$ such that $E\in F$ and $K\in F$ and
            $E\times K\subseteq U$ by \auto.
        $E\inter K\in F$ by \obv{filter}.
        $E\inter K\subseteq E$ and $E\inter K\subseteq K$ by
            \obv{inter_subseteq}.
        Hence $(E\inter K)\times(E\inter K)\subseteq E\times K$ by
            \auto{times,subseteq}.
        Hence $(E\inter K)\times(E\inter K)\subseteq U$ by
            \obv{subseteq_transitive}.
    \end{subproof}

    Show that if for every $U\in G$ there exists $A\in F$ such that
        $A\times A\subseteq U$, then
        $\productFilter{S}{S}{F}{F}$ refines $G$ on $S\times S$.
    \begin{subproof}
        Assume for every $U\in G$ there exists $A\in F$ such that
            $A\times A\subseteq U$.
        Show that $G\subseteq\productFilter{S}{S}{F}{F}$.
        \begin{subproof}
            It suffices to show that every element of $G$ is an element of
                $\productFilter{S}{S}{F}{F}$ by \obv{subseteq}.
            Fix $U\in G$.
            $U\subseteq S\times S$ by \obv{filter}.
            Take $A\in F$ such that $A\times A\subseteq U$.
            Follows by \auto{product_filter_iff}.
        \end{subproof}
        Follows by \obv{filter_refines}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{theorem}\label{filter_tendsto_product_iff}
    Let $H$ be a filter on $R$, $F$ be a filter on $S$, and
        $G$ be a filter on $T$.
    Let $h\in\funs{R}{S\times T}$.
    Then $h$ tends from $H$ on $R$ to $\productFilter{S}{T}{F}{G}$ on
        $S\times T$ iff
        $\firstProductProjection{S}{T}\circ h$ tends from $H$ on $R$ to
            $F$ on $S$ and
        $\secondProductProjection{S}{T}\circ h$ tends from $H$ on $R$ to
            $G$ on $T$.
\end{theorem}
\begin{proof}
    Let $p_1=\firstProductProjection{S}{T}$.
    Let $p_2=\secondProductProjection{S}{T}$.
    Let $K=\filterPushforward{R}{S\times T}{h}{H}$.
    Let $P=\filterPullback{S\times T}{S}{p_1}{F}$.
    Let $Q=\filterPullback{S\times T}{T}{p_2}{G}$.
    $p_1\in\funs{S\times T}{S}$ and $p_2\in\funs{S\times T}{T}$ by
        \auto{first_product_projection_type,second_product_projection_type}.
    Hence $p_1\circ h\in\funs{R}{S}$ and
        $p_2\circ h\in\funs{R}{T}$ by \obv{funs_circ}.
    $K$ is a filter on $S\times T$ by
        \auto{filter_pushforward_is_filter}.
    $P$ and $Q$ are filters on $S\times T$ by
        \auto{filter_pullback_is_filter}.
    $\productFilter{S}{T}{F}{G}$ is a filter on $S\times T$ by
        \obv{product_filter_is_filter}.
    $\productFilter{S}{T}{F}{G}
        =\binaryCommonFilterRefinement{S\times T}{P}{Q}$ by
        \auto{product_filter}.

    $h$ tends from $H$ on $R$ to $\productFilter{S}{T}{F}{G}$ on
        $S\times T$ iff
        $h\in\funs{R}{S\times T}$ and $H$ is a filter on $R$ and
        $\productFilter{S}{T}{F}{G}$ is a filter on $S\times T$ and
        $\filterPushforward{R}{S\times T}{h}{H}$ refines
            $\productFilter{S}{T}{F}{G}$ on $S\times T$ by
        \obv{filter_tendsto}.
    Hence $h$ tends from $H$ on $R$ to $\productFilter{S}{T}{F}{G}$ on
        $S\times T$ iff
        $K$ refines $\productFilter{S}{T}{F}{G}$ on $S\times T$ by
        \auto.
    $K$ refines $\binaryCommonFilterRefinement{S\times T}{P}{Q}$ on
        $S\times T$ iff
        $K$ refines $P$ on $S\times T$ and
        $K$ refines $Q$ on $S\times T$ by
        \obv{binary_common_filter_refinement_universal}.
    Hence $K$ refines $\productFilter{S}{T}{F}{G}$ on $S\times T$ iff
        $K$ refines $P$ on $S\times T$ and
        $K$ refines $Q$ on $S\times T$.

    $\filterPushforward{S\times T}{S}{p_1}{K}$ refines $F$ on $S$ iff
        $K$ refines $P$ on $S\times T$ by
        \auto{filter_pushforward_pullback_adjunction}.
    $\filterPushforward{S\times T}{T}{p_2}{K}$ refines $G$ on $T$ iff
        $K$ refines $Q$ on $S\times T$ by
        \auto{filter_pushforward_pullback_adjunction}.
    $\filterPushforward{R}{S}{p_1\circ h}{H}
        =\filterPushforward{S\times T}{S}{p_1}{K}$ by
        \auto{composition_filter_pushforward}.
    $\filterPushforward{R}{T}{p_2\circ h}{H}
        =\filterPushforward{S\times T}{T}{p_2}{K}$ by
        \auto{composition_filter_pushforward}.
    $p_1\circ h$ tends from $H$ on $R$ to $F$ on $S$ iff
        $p_1\circ h\in\funs{R}{S}$ and $H$ is a filter on $R$ and
        $F$ is a filter on $S$ and
        $\filterPushforward{R}{S}{p_1\circ h}{H}$ refines $F$ on $S$ by
        \obv{filter_tendsto}.
    Hence $p_1\circ h$ tends from $H$ on $R$ to $F$ on $S$ iff
        $\filterPushforward{R}{S}{p_1\circ h}{H}$ refines $F$ on $S$ by
        \auto.
    $p_2\circ h$ tends from $H$ on $R$ to $G$ on $T$ iff
        $p_2\circ h\in\funs{R}{T}$ and $H$ is a filter on $R$ and
        $G$ is a filter on $T$ and
        $\filterPushforward{R}{T}{p_2\circ h}{H}$ refines $G$ on $T$ by
        \obv{filter_tendsto}.
    Hence $p_2\circ h$ tends from $H$ on $R$ to $G$ on $T$ iff
        $\filterPushforward{R}{T}{p_2\circ h}{H}$ refines $G$ on $T$ by
        \auto.
    Follows by \auto.
\end{proof}

\begin{theorem}\label{principal_product_filter}
    Suppose $A\subseteq S$ and $B\subseteq T$.
    Then $\productFilter{S}{T}{\principalfilter{S}{A}}{
        \principalfilter{T}{B}}
        =\principalfilter{S\times T}{A\times B}$.
\end{theorem}
\begin{proof}
    $\principalfilter{S}{A}$ is a filter on $S$ and
        $\principalfilter{T}{B}$ is a filter on $T$ by
        \obv{principalfilter_is_filter}.
    $A\times B\subseteq S\times T$ by \auto{times,subseteq}.
    $\principalfilter{S\times T}{A\times B}$ is a filter on $S\times T$
        by \obv{principalfilter_is_filter}.
    Show that for every $W$ we have
        $W\in\productFilter{S}{T}{\principalfilter{S}{A}}{
            \principalfilter{T}{B}}$ iff
        $W\in\principalfilter{S\times T}{A\times B}$.
    \begin{subproof}
        Fix $W$.
        $W\in\productFilter{S}{T}{\principalfilter{S}{A}}{
            \principalfilter{T}{B}}$ iff
            $W\subseteq S\times T$ and there exist $C,D$ such that
            $C\in\principalfilter{S}{A}$ and
            $D\in\principalfilter{T}{B}$ and $C\times D\subseteq W$ by
            \obv{product_filter_iff}.
        Show that if $W\in\productFilter{S}{T}{\principalfilter{S}{A}}{
            \principalfilter{T}{B}}$, then
            $W\in\principalfilter{S\times T}{A\times B}$.
        \begin{subproof}
            Assume $W\in\productFilter{S}{T}{\principalfilter{S}{A}}{
                \principalfilter{T}{B}}$.
            Hence $W\subseteq S\times T$ and there exist $C,D$ such that
                $C\in\principalfilter{S}{A}$ and
                $D\in\principalfilter{T}{B}$ and
                $C\times D\subseteq W$.
            Take $C,D$ such that $C\in\principalfilter{S}{A}$ and
                $D\in\principalfilter{T}{B}$ and $C\times D\subseteq W$.
            $C\subseteq S$ and $D\subseteq T$ by \obv{filter}.
            Hence $A\subseteq C$ and $B\subseteq D$ by
                \obv{principalfilter_iff}.
            Hence $A\times B\subseteq C\times D$ by
                \auto{times,subseteq}.
            Hence $A\times B\subseteq W$ by \obv{subseteq_transitive}.
            $W\in\principalfilter{S\times T}{A\times B}$ iff
                $A\times B\subseteq W$ by \obv{principalfilter_iff}.
        \end{subproof}
        Show that if $W\in\principalfilter{S\times T}{A\times B}$, then
            $W\in\productFilter{S}{T}{\principalfilter{S}{A}}{
                \principalfilter{T}{B}}$.
        \begin{subproof}
            Assume $W\in\principalfilter{S\times T}{A\times B}$.
            Hence $W\subseteq S\times T$ by \obv{filter}.
            $W\in\principalfilter{S\times T}{A\times B}$ iff
                $A\times B\subseteq W$ by \obv{principalfilter_iff}.
            Hence $A\times B\subseteq W$.
            $A\in\principalfilter{S}{A}$ and
                $B\in\principalfilter{T}{B}$ by
                \obv{principalfilter_bottom}.
            Follows by \auto.
        \end{subproof}
        Follows by \auto.
    \end{subproof}
    Follows by set extensionality.
\end{proof}
