\import{set.tex}

\section{Partitions}

\begin{definition}\label{partition}
    $P$ is a partition iff
    $\emptyset\notin P$ and
    for all $B, C\in P$ such that $B\neq C$ we have $B$ is disjoint from $C$.
\end{definition}

\begin{abbreviation}\label{partition_of}
    $P$ is a partition of $A$ iff
    $P$ is a partition and
    $\unions{P} = A$.
\end{abbreviation}

\begin{proposition}\label{partition_emptyset}
    $\emptyset$ is a partition of $\emptyset$.
\end{proposition}
\begin{proof}
    Follows by \auto{emptyset,partition,setext,unions_iff}.
\end{proof}

\begin{proposition}\label{partition_block_subseteq}
    Suppose $P$ is a partition of $A$ and $B\in P$.
    Then $B\subseteq A$.
\end{proposition}
\begin{proof}
    Every element of $B$ is an element of $\unions{P}$
        by \auto{unions_iff}.
    $\unions{P}=A$.
    Follows by \obv{subseteq}.
\end{proof}

\begin{proposition}\label{partition_block_exists}
    Suppose $P$ is a partition of $A$ and $x\in A$.
    Then there exists $B\in P$ such that $x\in B$.
\end{proposition}
\begin{proof}
    $\unions{P}=A$.
    Hence $x\in\unions{P}$.
    Follows by \obv{unions_iff}.
\end{proof}

\begin{proposition}\label{partition_block_unique}
    Suppose $P$ is a partition.
    Suppose $B,C\in P$ and $x\in B,C$.
    Then $B=C$.
\end{proposition}
\begin{proof}
    Suppose not.
    Then $B$ is disjoint from $C$ by \obv{partition}.
    Contradiction by \auto{disjoint}.
\end{proof}

\begin{proposition}\label{partition_remove_block}
    Suppose $P$ is a partition of $X$ and $B\in P$.
    Then $P\setminus\{B\}$ is a partition of $X\setminus B$.
\end{proposition}
\begin{proof}
    Let $P'=P\setminus\{B\}$.
    $\emptyset\notin P'$ by \auto{partition,setminus}.
    Show that for all $C,D\in P'$ such that $C\neq D$ we have
        $C$ is disjoint from $D$.
    \begin{subproof}
        Fix $C,D\in P'$.
        Assume $C\neq D$.
        $C,D\in P$ by \auto{setminus}.
        Follows by \obv{partition}.
    \end{subproof}
    Hence $P'$ is a partition by \obv{partition}.

    Show that $\unions{P'}\subseteq X\setminus B$.
    \begin{subproof}
        Show that every element of $\unions{P'}$ is an element of
            $X\setminus B$.
        \begin{subproof}
            Fix $x\in\unions{P'}$.
            Take $C\in P'$ such that $x\in C$ by \obv{unions_iff}.
            $C\in P$ and $C\neq B$ by
                \auto{setminus,singleton_iff}.
            $C\subseteq X$ by \obv{partition_block_subseteq}.
            Hence $x\in X$ by \obv{subseteq}.
            $C$ is disjoint from $B$ by \obv{partition}.
            Hence $x\notin B$ by \auto{disjoint}.
            Follows by \obv{setminus}.
        \end{subproof}
        Follows by \obv{subseteq}.
    \end{subproof}
    Show that $X\setminus B\subseteq\unions{P'}$.
    \begin{subproof}
        Show that every element of $X\setminus B$ is an element of
            $\unions{P'}$.
        \begin{subproof}
            Fix $x\in X\setminus B$.
            Hence $x\in X$ and $x\notin B$ by \obv{setminus}.
            Take $C\in P$ such that $x\in C$ by
                \obv{partition_block_exists}.
            $C\neq B$.
            Hence $C\in P'$ by \auto{setminus,singleton_iff}.
            Follows by \auto{unions_iff}.
        \end{subproof}
        Follows by \obv{subseteq}.
    \end{subproof}
    Hence $\unions{P'}=X\setminus B$ by
        \obv{subseteq_antisymmetric}.
\end{proof}

\begin{definition}\label{partition_refinement}
    $P'$ is a refinement of $P$ iff
    for every $A'\in P'$ there exists
    $A\in P$ such that $A'\subseteq A$.
\end{definition}

\begin{abbreviation}\label{partition_refines}
    $P'\refines P$ iff
    $P'$ is a refinement of $P$.
\end{abbreviation}

\begin{composition}\label{partition_refinement_transitive}
    Suppose $P''\refines P'\refines P$.
    Then $P''\refines P$.
\end{composition}
\begin{proof}
    Follows by \auto{partition_refinement,subseteq_transitive}.
\end{proof}

\begin{proposition}\label{partition_refinement_reflexive}
    $P\refines P$.
\end{proposition}
\begin{proof}
    Follows by \obv{partition_refinement,subseteq_refl}.
\end{proof}
