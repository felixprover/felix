\section{Finite-intersection families}

\import{order/zorn.tex}
\import{order/finite.tex}
\import{set/finite-intersection-family.tex}

\begin{definition}\label{maximal_proper_finite_intersection_property_on}
    $D$ is maximal with respect to the proper finite intersection property on $X$ iff
        $D\subseteq\pow{X}$ and
        $D$ has the proper finite intersection property and
        for every $E$ such that $D\subseteq E\subseteq\pow{X}$ and
            $E$ has the proper finite intersection property we have $E=D$.
\end{definition}

\begin{theorem}\label{maximal_proper_fip_extension}
    Suppose $A\subseteq\pow{X}$.
    Suppose $A$ has the proper finite intersection property.
    Then there exists $D$ such that $A\subseteq D$ and
        $D$ is maximal with respect to the proper finite intersection property on $X$.
\end{theorem}
\begin{proof}
    Let $P=\{E\in\pow{\pow{X}}\mid
        \text{$A\subseteq E$ and $E$ has the proper finite intersection property}\}$.
    $A\in\pow{\pow{X}}$ by \obv{powerset_intro}.
    Thus $A\in P$ by \obv{subseteq_refl}.
    Hence $P$ is inhabited.

    Show that for every $C$ such that $C\subseteq P$ and $C$ is inhabited
        and $C$ is an inclusion chain we have $\unions{C}\in P$.
    \begin{subproof}
        Fix $C$.
        Assume $C\subseteq P$ and $C$ is inhabited and
            $C$ is an inclusion chain.
        Let $D=\unions{C}$.
        Take $E_0$ such that $E_0\in C$.
        $E_0\in P$ by \obv{subseteq}.
        \begin{align*}
            A
                &\subseteq E_0
                \\
                &\subseteq D
                    \explanation{by \auto{subseteq,unions_iff}}
        \end{align*}

        Show $D\subseteq\pow{X}$.
        \begin{subproof}
            For every $E\in C$ we have $E\subseteq\pow{X}$
                by \auto{subseteq,powerset_elim}.
            Follows by \auto{unions_family}.
        \end{subproof}

        Show $D$ has the proper finite intersection property.
        \begin{subproof}
            $A$ is inhabited by \obv{proper_finite_intersection_property}.
            Thus $D$ is inhabited by \auto{subseteq}.
            Show that for every $F$ such that $F\subseteq D$ and $F$ is finite
                and $F$ is inhabited we have $\inters{F}\neq\emptyset$.
            \begin{subproof}
                Fix $F$.
                Assume $F\subseteq D$ and $F$ is finite and $F$ is inhabited.
                Take $E\in C$ such that $F\subseteq E$
                    by \obv{finite_subset_of_chain_union}.
                $E\in P$ by \obv{subseteq}.
                Hence $E$ has the proper finite intersection property.
                Follows by \auto{proper_finite_intersection_property,finite_intersection_property}.
            \end{subproof}
            Follows by \obv{proper_finite_intersection_property,finite_intersection_property}.
        \end{subproof}

        $D\in\pow{\pow{X}}$ by \obv{powerset_intro}.
    \end{subproof}

    Take $D\in P$ such that for every $E\in P$ such that $D\subseteq E$
        we have $E=D$ by \obv{zorn_subset_family}.
    $A\subseteq D$ and $D\subseteq\pow{X}$ and
        $D$ has the proper finite intersection property by \obv{powerset_elim}.
    Show that for every $E$ such that $D\subseteq E\subseteq\pow{X}$ and
        $E$ has the proper finite intersection property we have $E=D$.
    \begin{subproof}
        Fix $E$.
        Assume $D\subseteq E\subseteq\pow{X}$ and
            $E$ has the proper finite intersection property.
        $A\subseteq E$ by \obv{subseteq_transitive}.
        $E\in\pow{\pow{X}}$ by \obv{powerset_intro}.
        Thus $E\in P$.
    \end{subproof}
    Thus $D$ is maximal with respect to the proper finite intersection property on $X$
        by \obv{maximal_proper_finite_intersection_property_on}.
\end{proof}
