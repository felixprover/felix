\section{Bounded closure systems and hulls}

The unrestricted intersection of the empty family is empty in this library.
A closure system on an ambient set therefore uses intersection bounded by that
set.  In particular, the empty bounded intersection is the whole ambient set.

\begin{definition}\label{bounded_intersection}
    $\boundedIntersection{X}{D}
        =\{x\in X\mid
            \text{for every $B\in D$ we have $x\in B$}\}$.
\end{definition}

\begin{proposition}\label{bounded_intersection_iff}
    $x\in\boundedIntersection{X}{D}$ iff
        $x\in X$ and for every $B\in D$ we have $x\in B$.
\end{proposition}
\begin{proof}
    Follows by \obv{bounded_intersection}.
\end{proof}

\begin{proposition}\label{bounded_intersection_subseteq}
    $\boundedIntersection{X}{D}\subseteq X$.
\end{proposition}
\begin{proof}
    Follows by \obv{subseteq,bounded_intersection_iff}.
\end{proof}

\begin{proposition}\label{bounded_intersection_empty}
    $\boundedIntersection{X}{\emptyset}=X$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{setext,bounded_intersection_iff,emptyset}.
\end{proof}

\begin{proposition}\label{bounded_intersection_union}
    $\boundedIntersection{X}{D\union E}
        =\boundedIntersection{X}{D}\inter\boundedIntersection{X}{E}$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{setext,bounded_intersection_iff,union_iff,inter}.
\end{proof}

\begin{proposition}\label{bounded_intersection_as_indexed_intersection}
    Suppose $D\subseteq\pow{X}$.
    Then
        $\boundedIntersection{X}{D}
            =\indexedIntersection{D}{\identity{D}}{X}$.
\end{proposition}
\begin{proof}
    Show that for every $x$ we have
        $x\in\boundedIntersection{X}{D}$ iff
        $x\in\indexedIntersection{D}{\identity{D}}{X}$.
    \begin{subproof}
        Fix $x$.
        For every $B\in D$ we have $(\identity{D})(B)=B$ by
            \obv{id_apply}.
        Follows by
            \auto{bounded_intersection_iff,indexed_intersection_iff}.
    \end{subproof}
    Follows by \obv{setext}.
\end{proof}

\begin{proposition}\label{bounded_intersection_eq_inters}
    Suppose $D$ is inhabited and $D\subseteq\pow{X}$.
    Then $\boundedIntersection{X}{D}=\inters{D}$.
\end{proposition}
\begin{proof}
    Show that every element of $\boundedIntersection{X}{D}$ is an element of
        $\inters{D}$.
    \begin{subproof}
        Fix $x\in\boundedIntersection{X}{D}$.
        Follows by
            \auto{bounded_intersection_iff,inters_iff_forall}.
    \end{subproof}
    Show that every element of $\inters{D}$ is an element of
        $\boundedIntersection{X}{D}$.
    \begin{subproof}
        Fix $x\in\inters{D}$.
        Take $B$ such that $B\in D$.
        $B\subseteq X$ by \obv{subseteq,powerset_elim}.
        $x\in B$ by \obv{inters_destr}.
        Hence $x\in X$ by \obv{subseteq}.
        For every $A\in D$ we have $x\in A$ by
            \obv{inters_destr}.
        Follows by \obv{bounded_intersection_iff}.
    \end{subproof}
    Follows by \obv{setext}.
\end{proof}

\begin{definition}\label{closure_system}
    $C$ is a closure system on $X$ iff
        $C\subseteq\pow{X}$ and
        for every $D\subseteq C$ we have
            $\boundedIntersection{X}{D}\in C$.
\end{definition}

\begin{proposition}\label{closure_system_bounded}
    Suppose $C$ is a closure system on $X$.
    Then $C\subseteq\pow{X}$.
\end{proposition}
\begin{proof}
    Follows by \obv{closure_system}.
\end{proof}

\begin{proposition}\label{closure_system_intersection}
    Suppose $C$ is a closure system on $X$ and $D\subseteq C$.
    Then $\boundedIntersection{X}{D}\in C$.
\end{proposition}
\begin{proof}
    Follows by \obv{closure_system}.
\end{proof}

\begin{proposition}\label{closure_system_top}
    Suppose $C$ is a closure system on $X$.
    Then $X\in C$.
\end{proposition}
\begin{proof}
    $\emptyset\subseteq C$ by \obv{emptyset_subseteq}.
    Hence $\boundedIntersection{X}{\emptyset}\in C$ by
        \obv{closure_system_intersection}.
    Follows by \obv{bounded_intersection_empty}.
\end{proof}

\begin{definition}\label{closure_supersets}
    $\closureSupersets{C}{A}=\{B\in C\mid A\subseteq B\}$.
\end{definition}

\begin{proposition}\label{closure_supersets_iff}
    $B\in\closureSupersets{C}{A}$ iff
        $B\in C$ and $A\subseteq B$.
\end{proposition}
\begin{proof}
    Follows by \obv{closure_supersets}.
\end{proof}

\begin{proposition}\label{closure_supersets_subseteq}
    $\closureSupersets{C}{A}\subseteq C$.
\end{proposition}
\begin{proof}
    Follows by \obv{subseteq,closure_supersets_iff}.
\end{proof}

\begin{definition}\label{closure_hull}
    $\closureHull{X}{C}{A}
        =\boundedIntersection{X}{\closureSupersets{C}{A}}$.
\end{definition}

\begin{proposition}\label{closure_hull_subseteq}
    $\closureHull{X}{C}{A}\subseteq X$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{closure_hull,bounded_intersection_subseteq}.
\end{proof}

\begin{proposition}\label{closure_supersets_inhabited}
    Suppose $C$ is a closure system on $X$ and $A\subseteq X$.
    Then $\closureSupersets{C}{A}$ is inhabited.
\end{proposition}
\begin{proof}
    $X\in C$ by \obv{closure_system_top}.
    Hence $X\in\closureSupersets{C}{A}$ by
        \obv{closure_supersets_iff}.
\end{proof}

\begin{proposition}\label{closure_hull_extensive}
    Suppose $C$ is a closure system on $X$ and $A\subseteq X$.
    Then $A\subseteq\closureHull{X}{C}{A}$.
\end{proposition}
\begin{proof}
    Show that every element of $A$ is an element of
        $\closureHull{X}{C}{A}$.
    \begin{subproof}
        Fix $x\in A$.
        Hence $x\in X$ by \obv{subseteq}.
        Show that for every $B\in\closureSupersets{C}{A}$ we have
            $x\in B$.
        \begin{subproof}
            Fix $B\in\closureSupersets{C}{A}$.
            $A\subseteq B$ by \obv{closure_supersets_iff}.
            Follows by \obv{subseteq}.
        \end{subproof}
        Follows by
            \obv{closure_hull,bounded_intersection_iff}.
    \end{subproof}
    Follows by \obv{subseteq}.
\end{proof}

\begin{proposition}\label{closure_hull_closed}
    Suppose $C$ is a closure system on $X$ and $A\subseteq X$.
    Then $\closureHull{X}{C}{A}\in C$.
\end{proposition}
\begin{proof}
    $\closureSupersets{C}{A}\subseteq C$ by
        \obv{closure_supersets_subseteq}.
    $\boundedIntersection{X}{\closureSupersets{C}{A}}\in C$ by
        \obv{closure_system_intersection}.
    Follows by \obv{closure_hull}.
\end{proof}

\begin{proposition}\label{closure_hull_least}
    Suppose $C$ is a closure system on $X$ and $A\subseteq X$.
    Let $B\in C$.
    Suppose $A\subseteq B$.
    Then $\closureHull{X}{C}{A}\subseteq B$.
\end{proposition}
\begin{proof}
    $B\in\closureSupersets{C}{A}$ by
        \obv{closure_supersets_iff}.
    Follows by
        \auto{subseteq,closure_hull,bounded_intersection_iff}.
\end{proof}

\begin{proposition}\label{closure_hull_monotone}
    Suppose $C$ is a closure system on $X$.
    Suppose $A\subseteq B$ and $B\subseteq X$.
    Then $\closureHull{X}{C}{A}\subseteq\closureHull{X}{C}{B}$.
\end{proposition}
\begin{proof}
    $A\subseteq X$ by \obv{subseteq_transitive}.
    $B\subseteq\closureHull{X}{C}{B}$ by
        \obv{closure_hull_extensive}.
    Hence $A\subseteq\closureHull{X}{C}{B}$ by
        \obv{subseteq_transitive}.
    $\closureHull{X}{C}{B}\in C$ by
        \obv{closure_hull_closed}.
    Follows by \obv{closure_hull_least}.
\end{proof}

\begin{proposition}\label{closure_hull_idempotent}
    Suppose $C$ is a closure system on $X$ and $A\subseteq X$.
    Then
        $\closureHull{X}{C}{\closureHull{X}{C}{A}}
            =\closureHull{X}{C}{A}$.
\end{proposition}
\begin{proof}
    Let $H=\closureHull{X}{C}{A}$.
    $H\subseteq X$ by \auto{closure_hull_subseteq}.
    $H\subseteq\closureHull{X}{C}{H}$ by
        \obv{closure_hull_extensive}.
    $H\in C$ by \auto{closure_hull_closed}.
    $\closureHull{X}{C}{H}\subseteq H$ by
        \obv{closure_hull_least,subseteq_refl}.
    Follows by \obv{subseteq_antisymmetric}.
\end{proof}

\begin{proposition}\label{closure_hull_already_closed}
    Suppose $C$ is a closure system on $X$ and $A\in C$.
    Then $\closureHull{X}{C}{A}=A$.
\end{proposition}
\begin{proof}
    $C\subseteq\pow{X}$ by \obv{closure_system_bounded}.
    Hence $A\subseteq X$ by \obv{subseteq,powerset_elim}.
    $A\subseteq\closureHull{X}{C}{A}$ by
        \obv{closure_hull_extensive}.
    $\closureHull{X}{C}{A}\subseteq A$ by
        \obv{closure_hull_least,subseteq_refl}.
    Follows by \obv{subseteq_antisymmetric}.
\end{proof}

\begin{proposition}\label{empty_closure_supersets}
    $\closureSupersets{C}{\emptyset}=C$.
\end{proposition}
\begin{proof}
    Follows by
        \obv{setext,closure_supersets_iff,emptyset_subseteq}.
\end{proof}

\begin{proposition}\label{closure_hull_empty}
    Suppose $C$ is a closure system on $X$.
    Then
        $\closureHull{X}{C}{\emptyset}
            =\boundedIntersection{X}{C}$.
\end{proposition}
\begin{proof}
    Follows by
        \obv{closure_hull,empty_closure_supersets}.
\end{proof}
