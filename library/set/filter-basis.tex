\section{Filter bases and generated filters}

\import{set/filter-generation.tex}

\subsection{Filter bases}

\begin{definition}\label{filter_basis}
    $B$ is a filter basis on $S$ iff
        $B$ is a family of subsets of $S$ and
        $B$ is inhabited and
        for all $A,C$ such that $A\in B$ and $C\in B$
            there exists $D\in B$ such that $D\subseteq A\inter C$.
\end{definition}

\begin{definition}\label{proper_filter_basis}
    $B$ is a proper filter basis on $S$ iff
        $B$ is a filter basis on $S$ and
        for every $A\in B$ we have $A$ is inhabited.
\end{definition}

\begin{classification}\label{proper_filter_basis_is_filter_basis}
    Suppose $B$ is a proper filter basis on $S$.
    Then $B$ is a filter basis on $S$.
\end{classification}
\begin{proof}
    Follows by \obv{proper_filter_basis}.
\end{proof}

\begin{proposition}\label{proper_filter_basis_excludes_emptyset}
    Suppose $B$ is a proper filter basis on $S$.
    Then $\emptyset\notin B$.
\end{proposition}
\begin{proof}
    Suppose not.
    Hence $\emptyset\in B$.
    Hence $\emptyset$ is inhabited by \obv{proper_filter_basis}.
    Take $x$ such that $x\in\emptyset$.
    Contradiction by \obv{emptyset}.
\end{proof}

\begin{theorem}\label{proper_filter_basis_iff_emptyset_notin}
    Let $B$ be a filter basis on $S$.
    Then $B$ is a proper filter basis on $S$ iff $\emptyset\notin B$.
\end{theorem}
\begin{proof}
    If $B$ is a proper filter basis on $S$, then $\emptyset\notin B$
        by \obv{proper_filter_basis_excludes_emptyset}.
    Show that if $\emptyset\notin B$, then
        $B$ is a proper filter basis on $S$.
    \begin{subproof}
        Assume $\emptyset\notin B$.
        Show that for every $A\in B$ we have $A$ is inhabited.
        \begin{subproof}
            Fix $A\in B$.
            $A\neq\emptyset$.
            Follows by \obv{nonempty_inhabited}.
        \end{subproof}
        Follows by \obv{proper_filter_basis}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{definition}\label{filter_family_refinement}
    $B$ refines $C$ as filter families on $S$ iff
        $B$ is a family of subsets of $S$ and
        $C$ is a family of subsets of $S$ and
        for every $A\in C$ there exists $D\in B$ such that $D\subseteq A$.
\end{definition}

\begin{definition}\label{filter_basis_of}
    Let $F$ be a filter on $S$.
    $B$ is a basis of $F$ on $S$ iff
        $B\subseteq F$ and
        $B$ refines $F$ as filter families on $S$.
\end{definition}

\begin{proposition}\label{proper_filter_basis_member_properties}
    Suppose $B$ is a proper filter basis on $S$.
    Let $A\in B$.
    Then $A\subseteq S$ and $A$ is inhabited.
\end{proposition}
\begin{proof}
    $B$ is a filter basis on $S$ by
        \obv{proper_filter_basis_is_filter_basis}.
    $A\subseteq S$ by \obv{filter_basis}.
    $A$ is inhabited by \obv{proper_filter_basis}.
\end{proof}

\begin{proposition}\label{filter_basis_member_subseteq}
    Suppose $B$ is a filter basis on $S$.
    Let $A\in B$.
    Then $A\subseteq S$.
\end{proposition}
\begin{proof}
    Follows by \obv{filter_basis}.
\end{proof}

\begin{proposition}\label{filter_basis_supersets_are_filter}
    Suppose $B$ is a filter basis on $S$.
    Then
        $\{A\in\pow{S}\mid
            \text{there exists $C\in B$ such that $C\subseteq A$}\}$
        is a filter on $S$.
\end{proposition}
\begin{proof}
    Let $G=\{A\in\pow{S}\mid
        \text{there exists $C\in B$ such that $C\subseteq A$}\}$.
    $G$ is a family of subsets of $S$
        by \obv{pow_iff,subseteq}.

    Show that $S\in G$.
    \begin{subproof}
        Take $C$ such that $C\in B$ by \obv{filter_basis}.
        $C\subseteq S$ by \obv{filter_basis_member_subseteq}.
        $S\in\pow{S}$ by \obv{powerset_top}.
    \end{subproof}

    Show that $G$ is closed under binary intersections.
    \begin{subproof}
        Fix $A,C\in G$.
        Take $U\in B$ such that $U\subseteq A$.
        Take $V\in B$ such that $V\subseteq C$.
        Take $D\in B$ such that $D\subseteq U\inter V$
            by \obv{filter_basis}.
        $D\subseteq U$ and $D\subseteq V$
            by \obv{subseteq_inter_iff}.
        Hence $D\subseteq A$ and $D\subseteq C$
            by \obv{subseteq_transitive}.
        Hence $D\subseteq A\inter C$ by \obv{subseteq_inter_iff}.
        $A\subseteq S$ and $C\subseteq S$.
        Hence $A\inter C\subseteq S$ by \obv{inter_subseteq}.
        Hence $A\inter C\in\pow{S}$ by \obv{powerset_intro}.
        There exists $E\in B$ such that $E\subseteq A\inter C$.
    \end{subproof}

    Show that $G$ is upward-closed in $S$.
    \begin{subproof}
        Fix $A,C$.
        Assume $A\subseteq C\subseteq S$ and $A\in G$.
        Take $D\in B$ such that $D\subseteq A$.
        Hence $D\subseteq C$ by \obv{subseteq_transitive}.
        $C\in\pow{S}$ by \obv{powerset_intro}.
    \end{subproof}
    Follows by \obv{filter}.
\end{proof}

\begin{theorem}\label{filter_basis_generated_filter_eq_supersets}
    Suppose $B$ is a filter basis on $S$.
    Then $\generatedFilter{S}{B}=
        \{A\in\pow{S}\mid
            \text{there exists $C\in B$ such that $C\subseteq A$}\}$.
\end{theorem}
\begin{proof}
    $B$ is a family of subsets of $S$ by \obv{filter_basis}.
    $\{A\in\pow{S}\mid
        \text{there exists $C\in B$ such that $C\subseteq A$}\}$
        is a filter on $S$ by \obv{filter_basis_supersets_are_filter}.
    Show that $B\subseteq
        \{A\in\pow{S}\mid
            \text{there exists $C\in B$ such that $C\subseteq A$}\}$.
    \begin{subproof}
        It suffices to show that every element of $B$ is an element of
            $\{A\in\pow{S}\mid
                \text{there exists $C\in B$ such that $C\subseteq A$}\}$
            by \obv{subseteq}.
        Fix $C\in B$.
        $C\subseteq S$ by \obv{filter_basis_member_subseteq}.
        Hence $C\in\pow{S}$ by \obv{powerset_intro}.
        There exists $D\in B$ such that $D\subseteq C$
            by \obv{subseteq_refl}.
        Follows by \auto.
    \end{subproof}
    Hence $\generatedFilter{S}{B}\subseteq
        \{A\in\pow{S}\mid
            \text{there exists $C\in B$ such that $C\subseteq A$}\}$
        by \obv{generated_filter_least}.

    Show that
        $\{A\in\pow{S}\mid
            \text{there exists $C\in B$ such that $C\subseteq A$}\}
        \subseteq\generatedFilter{S}{B}$.
    \begin{subproof}
        It suffices to show that every element of
            $\{A\in\pow{S}\mid
                \text{there exists $C\in B$ such that $C\subseteq A$}\}$
            is an element of $\generatedFilter{S}{B}$ by \obv{subseteq}.
        Fix $A\in\{D\in\pow{S}\mid
            \text{there exists $C\in B$ such that $C\subseteq D$}\}$.
        Take $C\in B$ such that $C\subseteq A$ by \auto.
        $B\subseteq\generatedFilter{S}{B}$ by
            \obv{generated_filter_contains}.
        Hence $C\in\generatedFilter{S}{B}$ by \obv{subseteq}.
        $A\in\pow{S}$ by \auto.
        Hence $A\subseteq S$ by \obv{powerset_elim}.
        $\generatedFilter{S}{B}$ is a filter on $S$ by
            \obv{generated_filter_is_filter}.
        Follows by \obv{filter}.
    \end{subproof}
    Follows by \obv{subseteq_antisymmetric}.
\end{proof}

\begin{theorem}\label{generated_filter_iff}
    Suppose $B$ is a filter basis on $S$.
    Then $A\in\generatedFilter{S}{B}$ iff
        $A\subseteq S$ and
        there exists $C\in B$ such that $C\subseteq A$.
\end{theorem}
\begin{proof}
    $\generatedFilter{S}{B}=
        \{D\in\pow{S}\mid
            \text{there exists $C\in B$ such that $C\subseteq D$}\}$
        by \obv{filter_basis_generated_filter_eq_supersets}.
    Show that if $A\in\generatedFilter{S}{B}$, then
        $A\subseteq S$ and
        there exists $C\in B$ such that $C\subseteq A$.
    \begin{subproof}
        Assume $A\in\generatedFilter{S}{B}$.
        Hence $A\in\{D\in\pow{S}\mid
            \text{there exists $C\in B$ such that $C\subseteq D$}\}$.
        $A\in\pow{S}$ and
            there exists $C\in B$ such that $C\subseteq A$
            by \auto.
        Hence $A\subseteq S$ by \obv{powerset_elim}.
    \end{subproof}
    Show that if $A\subseteq S$ and
        there exists $C\in B$ such that $C\subseteq A$, then
        $A\in\generatedFilter{S}{B}$.
    \begin{subproof}
        Assume $A\subseteq S$ and
            there exists $C\in B$ such that $C\subseteq A$.
        $A\in\pow{S}$ by \obv{powerset_intro}.
        Hence $A\in\{D\in\pow{S}\mid
            \text{there exists $C\in B$ such that $C\subseteq D$}\}$
            by \auto.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{theorem}\label{generated_filter_is_proper_filter}
    Suppose $B$ is a proper filter basis on $S$.
    Then $\generatedFilter{S}{B}$ is a proper filter on $S$.
\end{theorem}
\begin{proof}
    $B$ is a filter basis on $S$ by
        \obv{proper_filter_basis_is_filter_basis}.
    $B$ is a family of subsets of $S$ by \obv{filter_basis}.
    $\generatedFilter{S}{B}$ is a filter on $S$
        by \obv{generated_filter_is_filter}.
    Show that $\emptyset\notin\generatedFilter{S}{B}$.
    \begin{subproof}
        Suppose not.
        Take $C\in B$ such that $C\subseteq\emptyset$
            by \auto{generated_filter_iff}.
        Hence $C=\emptyset$ by \obv{subseteq_emptyset_iff}.
        $\emptyset\notin B$ by
            \obv{proper_filter_basis_excludes_emptyset}.
        Contradiction.
    \end{subproof}
    Follows by \obv{proper_filter}.
\end{proof}

\begin{proposition}\label{proper_filter_basis_in_generated_filter}
    Suppose $B$ is a proper filter basis on $S$.
    Then $B\subseteq\generatedFilter{S}{B}$.
\end{proposition}
\begin{proof}
    $B$ is a filter basis on $S$ by
        \obv{proper_filter_basis_is_filter_basis}.
    $B$ is a family of subsets of $S$ by \obv{filter_basis}.
    Follows by \obv{generated_filter_contains}.
\end{proof}

\begin{proposition}\label{proper_filter_basis_has_hosted_fip}
    Suppose $B$ is a proper filter basis on $S$.
    Then $B$ has the finite intersection property on $S$.
\end{proposition}
\begin{proof}
    $B$ is a filter basis on $S$ by
        \obv{proper_filter_basis_is_filter_basis}.
    $B$ is a family of subsets of $S$ by \obv{filter_basis}.
    $\generatedFilter{S}{B}$ is a proper filter on $S$ by
        \obv{generated_filter_is_proper_filter}.
    $\generatedFilter{S}{B}$ is a proper filter on $S$ iff
        $B$ has the finite intersection property on $S$ by
        \obv{generated_filter_is_proper_iff_hosted_fip}.
\end{proof}

\begin{proposition}\label{finite_intersections_on_is_filter_basis}
    $\finiteIntersectionsOn{S}{B}$ is a filter basis on $S$.
\end{proposition}
\begin{proof}
    $\finiteIntersectionsOn{S}{B}$ is a family of subsets of $S$ by
        \obv{finite_intersections_on_type}.
    $S\in\finiteIntersectionsOn{S}{B}$ by
        \obv{carrier_in_finite_intersections_on}.
    Hence $\finiteIntersectionsOn{S}{B}$ is inhabited.
    Show that for all $A,C$ such that
        $A\in\finiteIntersectionsOn{S}{B}$ and
        $C\in\finiteIntersectionsOn{S}{B}$ there exists
        $D\in\finiteIntersectionsOn{S}{B}$ such that
        $D\subseteq A\inter C$.
    \begin{subproof}
        Fix $A,C$.
        Assume $A\in\finiteIntersectionsOn{S}{B}$ and
            $C\in\finiteIntersectionsOn{S}{B}$.
        $A\inter C\in\finiteIntersectionsOn{S}{B}$ by
            \obv{finite_intersections_on_intersection}.
        Follows by \obv{subseteq_refl}.
    \end{subproof}
    Follows by \obv{filter_basis}.
\end{proof}

\subsection{Refinement and bases of filters}

\begin{proposition}\label{filter_family_refinement_reflexive}
    Suppose $B$ is a family of subsets of $S$.
    Then $B$ refines $B$ as filter families on $S$.
\end{proposition}
\begin{proof}
    For every $A\in B$ there exists $D\in B$ such that $D\subseteq A$
        by \obv{subseteq_refl}.
    Follows by \obv{filter_family_refinement}.
\end{proof}

\begin{proposition}\label{filter_family_refinement_transitive}
    Suppose $B$ refines $C$ as filter families on $S$.
    Suppose $C$ refines $D$ as filter families on $S$.
    Then $B$ refines $D$ as filter families on $S$.
\end{proposition}
\begin{proof}
    $B$ is a family of subsets of $S$ and
        $D$ is a family of subsets of $S$
        by \obv{filter_family_refinement}.
    Show that for every $A\in D$ there exists $E\in B$ such that
        $E\subseteq A$.
    \begin{subproof}
        Fix $A\in D$.
        Take $C'\in C$ such that $C'\subseteq A$
            by \obv{filter_family_refinement}.
        Take $E\in B$ such that $E\subseteq C'$
            by \obv{filter_family_refinement}.
        Hence $E\subseteq A$ by \obv{subseteq_transitive}.
    \end{subproof}
    Follows by \obv{filter_family_refinement}.
\end{proof}

\begin{proposition}\label{mutual_refinement_preserves_filter_basis}
    Suppose $B$ is a filter basis on $S$.
    Suppose $B$ refines $C$ as filter families on $S$.
    Suppose $C$ refines $B$ as filter families on $S$.
    Then $C$ is a filter basis on $S$.
\end{proposition}
\begin{proof}
    $C$ is a family of subsets of $S$ by
        \obv{filter_family_refinement}.
    Show that $C$ is inhabited.
    \begin{subproof}
        Take $A$ such that $A\in B$ by \obv{filter_basis}.
        Take $D\in C$ such that $D\subseteq A$ by
            \obv{filter_family_refinement}.
        Follows by \auto.
    \end{subproof}

    Show that for all $U,V$ such that $U\in C$ and $V\in C$
        there exists $W\in C$ such that $W\subseteq U\inter V$.
    \begin{subproof}
        Fix $U,V$.
        Assume $U\in C$ and $V\in C$.
        Take $A\in B$ such that $A\subseteq U$ by
            \obv{filter_family_refinement}.
        Take $D\in B$ such that $D\subseteq V$ by
            \obv{filter_family_refinement}.
        Take $E\in B$ such that $E\subseteq A\inter D$ by
            \obv{filter_basis}.
        $E\subseteq A$ and $E\subseteq D$ by
            \auto{subseteq_inter_iff}.
        Hence $E\subseteq U$ and $E\subseteq V$ by
            \obv{subseteq_transitive}.
        Hence $E\subseteq U\inter V$ by \obv{subseteq_inter_iff}.
        Take $W\in C$ such that $W\subseteq E$ by
            \obv{filter_family_refinement}.
        Hence $W\subseteq U\inter V$ by \obv{subseteq_transitive}.
    \end{subproof}
    Follows by \obv{filter_basis}.
\end{proof}

\begin{theorem}\label{filter_family_refinement_generated_filter_inclusion}
    Suppose $B$ refines $C$ as filter families on $S$.
    Then $\generatedFilter{S}{C}\subseteq\generatedFilter{S}{B}$.
\end{theorem}
\begin{proof}
    $B$ is a family of subsets of $S$ and
        $C$ is a family of subsets of $S$ by
        \obv{filter_family_refinement}.
    $B\subseteq\generatedFilter{S}{B}$ by
        \obv{generated_filter_contains}.
    $\generatedFilter{S}{B}$ is a filter on $S$ by
        \obv{generated_filter_is_filter}.
    Show that $C\subseteq\generatedFilter{S}{B}$.
    \begin{subproof}
        It suffices to show that every element of $C$ is an element of
            $\generatedFilter{S}{B}$ by \obv{subseteq}.
        Fix $A\in C$.
        Take $D\in B$ such that $D\subseteq A$ by
            \obv{filter_family_refinement}.
        Hence $D\in\generatedFilter{S}{B}$ by \obv{subseteq}.
        $A\subseteq S$.
        Follows by \obv{filter}.
    \end{subproof}
    Follows by \obv{generated_filter_least}.
\end{proof}

\begin{theorem}\label{mutual_filter_family_refinement_generated_filters_equal}
    Suppose $B$ refines $C$ as filter families on $S$.
    Suppose $C$ refines $B$ as filter families on $S$.
    Then $\generatedFilter{S}{B}=\generatedFilter{S}{C}$.
\end{theorem}
\begin{proof}
    $\generatedFilter{S}{B}\subseteq\generatedFilter{S}{C}$
        by \obv{filter_family_refinement_generated_filter_inclusion}.
    $\generatedFilter{S}{C}\subseteq\generatedFilter{S}{B}$
        by \obv{filter_family_refinement_generated_filter_inclusion}.
    Follows by \obv{subseteq_antisymmetric}.
\end{proof}

\begin{corollary}\label{mutual_proper_filter_basis_generated_filter}
    Suppose $B$ is a proper filter basis on $S$.
    Suppose $C$ is a proper filter basis on $S$.
    Suppose $B$ refines $C$ as filter families on $S$.
    Suppose $C$ refines $B$ as filter families on $S$.
    Then $\generatedFilter{S}{B}=\generatedFilter{S}{C}$ and
        $\generatedFilter{S}{B}$ is a filter on $S$ and
        $\generatedFilter{S}{C}$ is a filter on $S$.
\end{corollary}
\begin{proof}
    $B$ is a filter basis on $S$ by
        \obv{proper_filter_basis_is_filter_basis}.
    $C$ is a filter basis on $S$ by
        \obv{proper_filter_basis_is_filter_basis}.
    $\generatedFilter{S}{B}=\generatedFilter{S}{C}$
        by \obv{mutual_filter_family_refinement_generated_filters_equal}.
    $\generatedFilter{S}{B}$ is a filter on $S$
        by \obv{generated_filter_is_filter}.
    $\generatedFilter{S}{C}$ is a filter on $S$
        by \obv{generated_filter_is_filter}.
\end{proof}

\begin{proposition}\label{filter_is_filter_basis}
    Let $F$ be a filter on $S$.
    Then $F$ is a filter basis on $S$.
\end{proposition}
\begin{proof}
    $F$ is a family of subsets of $S$ by \obv{filter}.
    $F$ is inhabited by \obv{filter}.
    Show that for all $A,C$ such that $A\in F$ and $C\in F$
        there exists $D\in F$ such that $D\subseteq A\inter C$.
    \begin{subproof}
        Fix $A,C$.
        Assume $A\in F$ and $C\in F$.
        $A\inter C\in F$ by \obv{filter}.
        Follows by \obv{subseteq_refl}.
    \end{subproof}
    Follows by \obv{filter_basis}.
\end{proof}

\begin{proposition}\label{filter_is_proper_filter_basis}
    Let $F$ be a proper filter on $S$.
    Then $F$ is a proper filter basis on $S$.
\end{proposition}
\begin{proof}
    $F$ is a filter on $S$ and $\emptyset\notin F$
        by \obv{proper_filter}.
    $F$ is a filter basis on $S$ by \obv{filter_is_filter_basis}.
    Show that for every $A\in F$ we have $A$ is inhabited.
    \begin{subproof}
        Fix $A\in F$.
        $A\neq\emptyset$.
        Follows by \obv{nonempty_inhabited}.
    \end{subproof}
    Follows by \obv{proper_filter_basis}.
\end{proof}

\begin{proposition}\label{filter_is_basis_of_itself}
    Let $F$ be a filter on $S$.
    Then $F$ is a basis of $F$ on $S$.
\end{proposition}
\begin{proof}
    $F\subseteq F$ by \obv{subseteq_refl}.
    $F$ is a family of subsets of $S$ by \obv{filter}.
    Hence $F$ refines $F$ as filter families on $S$
        by \obv{filter_family_refinement_reflexive}.
    Follows by \obv{filter_basis_of}.
\end{proof}

\begin{theorem}\label{filter_basis_of_iff_generated_filter}
    Let $F$ be a filter on $S$.
    Then $B$ is a basis of $F$ on $S$ iff
        $B$ is a filter basis on $S$ and
        $\generatedFilter{S}{B}=F$.
\end{theorem}
\begin{proof}
    Show that if $B$ is a basis of $F$ on $S$, then
        $B$ is a filter basis on $S$ and
        $\generatedFilter{S}{B}=F$.
    \begin{subproof}
        Assume $B$ is a basis of $F$ on $S$.
        $B\subseteq F$ and
            $B$ refines $F$ as filter families on $S$
            by \obv{filter_basis_of}.
        $B$ is a family of subsets of $S$
            by \obv{filter_family_refinement}.
        Take $C\in B$ such that $C\subseteq S$
            by \obv{filter,filter_family_refinement}.
        Hence $B$ is inhabited.
        Show that for all $A,D$ such that $A\in B$ and $D\in B$
            there exists $E\in B$ such that $E\subseteq A\inter D$.
        \begin{subproof}
            Fix $A,D$.
            Assume $A\in B$ and $D\in B$.
            $A,D\in F$ by \obv{subseteq}.
            Hence $A\inter D\in F$ by \obv{filter}.
            Follows by \obv{filter_family_refinement}.
        \end{subproof}
        Hence $B$ is a filter basis on $S$ by \obv{filter_basis}.

        $\generatedFilter{S}{B}\subseteq F$
            by \obv{generated_filter_least}.
        Show that $F\subseteq\generatedFilter{S}{B}$.
        \begin{subproof}
            It suffices to show that every element of $F$ is an element of
                $\generatedFilter{S}{B}$ by \obv{subseteq}.
            Fix $A\in F$.
            Take $D\in B$ such that $D\subseteq A$
                by \obv{filter_family_refinement}.
            $B\subseteq\generatedFilter{S}{B}$ by
                \obv{generated_filter_contains}.
            Hence $D\in\generatedFilter{S}{B}$ by \obv{subseteq}.
            $\generatedFilter{S}{B}$ is a filter on $S$ by
                \obv{generated_filter_is_filter}.
            $A\subseteq S$ by \obv{filter}.
            Follows by \obv{filter}.
        \end{subproof}
        Hence $\generatedFilter{S}{B}=F$
            by \obv{subseteq_antisymmetric}.
    \end{subproof}

    Show that if $B$ is a filter basis on $S$ and
        $\generatedFilter{S}{B}=F$, then
        $B$ is a basis of $F$ on $S$.
    \begin{subproof}
        Assume $B$ is a filter basis on $S$ and
            $\generatedFilter{S}{B}=F$.
        $B\subseteq\generatedFilter{S}{B}$
            by \obv{generated_filter_contains}.
        Hence $B\subseteq F$.
        Show that $B$ refines $F$ as filter families on $S$.
        \begin{subproof}
            $B$ is a family of subsets of $S$
                by \obv{filter_basis}.
            $F$ is a family of subsets of $S$ by \obv{filter}.
            Show that for every $A\in F$ there exists $D\in B$ such that
                $D\subseteq A$.
            \begin{subproof}
                Fix $A\in F$.
                Hence $A\in\generatedFilter{S}{B}$.
                Follows by \auto{generated_filter_iff}.
            \end{subproof}
            Follows by \obv{filter_family_refinement}.
        \end{subproof}
        Follows by \obv{filter_basis_of}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{corollary}\label{finite_intersections_on_is_basis_of_generated_filter}
    Suppose $B$ is a family of subsets of $S$.
    Then $\finiteIntersectionsOn{S}{B}$ is a basis of
        $\generatedFilter{S}{B}$ on $S$.
\end{corollary}
\begin{proof}
    $\generatedFilter{S}{B}$ is a filter on $S$ by
        \obv{generated_filter_is_filter}.
    $\finiteIntersectionsOn{S}{B}$ is a filter basis on $S$ by
        \obv{finite_intersections_on_is_filter_basis}.
    $\generatedFilter{S}{\finiteIntersectionsOn{S}{B}}
        =\generatedFilter{S}{B}$ by
        \obv{generated_filter_of_finite_intersections_on}.
    $\finiteIntersectionsOn{S}{B}$ is a basis of
        $\generatedFilter{S}{B}$ on $S$ iff
        $\finiteIntersectionsOn{S}{B}$ is a filter basis on $S$ and
        $\generatedFilter{S}{\finiteIntersectionsOn{S}{B}}
            =\generatedFilter{S}{B}$ by
        \obv{filter_basis_of_iff_generated_filter}.
\end{proof}

\begin{theorem}\label{refining_subfamily_is_filter_basis}
    Let $F$ be a filter on $S$.
    Suppose $B$ is a basis of $F$ on $S$.
    Suppose $C\subseteq B$ and
        $C$ refines $B$ as filter families on $S$.
    Then $C$ is a basis of $F$ on $S$.
\end{theorem}
\begin{proof}
    $B\subseteq F$ and $B$ refines $F$ as filter families on $S$
        by \obv{filter_basis_of}.
    Hence $C\subseteq F$ by \obv{subseteq_transitive}.
    $C$ refines $F$ as filter families on $S$
        by \obv{filter_family_refinement_transitive}.
    Follows by \obv{filter_basis_of}.
\end{proof}
