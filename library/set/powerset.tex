\import{set.tex}

\section{Powerset}

\begin{abbreviation}\label{powerset_of}
    The powerset of $X$ denotes $\pow{X}$.
\end{abbreviation}

\begin{proposition}\label{powerset_intro}
    Suppose $A\subseteq B$.
    Then $A\in\pow{B}$.
\end{proposition}
\begin{proof}
    Follows by \obv{pow_iff,subseteq}.
\end{proof}

\begin{proposition}\label{powerset_elim}
    Let $A\in\pow{B}$.
    Then $A\subseteq B$.
\end{proposition}
\begin{proof}
    Follows by \obv{pow_iff,subseteq}.
\end{proof}

\begin{proposition}\label{powerset_bottom}
    $\emptyset\in\pow{A}$.
\end{proposition}
\begin{proof}
    Follows by \auto{pow_iff,subseteq,emptyset_subseteq}.
\end{proof}

\begin{proposition}\label{powerset_top}
    $A\in\pow{A}$.
\end{proposition}
\begin{proof}
    Follows by \auto{pow_iff,subseteq,subseteq_refl}.
\end{proof}

\begin{proposition}\label{unions_subseteq_of_powerset_is_subseteq}
    Let $A$ be a set.
    Let $B$ be a subset of $\pow{A}$.
    Then $\unions{B}\subseteq A$.
\end{proposition}
\begin{proof}
    Follows by \auto{subseteq,powerset_elim,unions_iff}.
\end{proof}

\begin{corollary}\label{powerset_closed_unions}
    Let $A$ be a set.
    Let $B$ be a subset of $\pow{A}$.
    Then $\unions{B}\in\pow{A}$.
\end{corollary}
\begin{proof}
    Follows by \obv{powerset_intro,unions_subseteq_of_powerset_is_subseteq}.
\end{proof}


\begin{proposition}\label{inter_powerset}
    Let $A,B\in\pow{C}$.
    Then $A\inter B\in\pow{C}$.
\end{proposition}
\begin{proof}
    We have $A,B\subseteq C$ by \obv{powerset_elim}.
    $A\inter B\subseteq C$ by \obv{inter_subseteq}.
    Follows by \obv{powerset_intro}.
\end{proof}

\begin{proposition}\label{unions_powerset}
    $\unions{\pow{A}} = A$.
\end{proposition}
\begin{proof}
    We have for all $x\in\unions{\pow{A}}$
        we have $x\in A$ by \auto{pow_iff,unions_iff}.
    We have for all $x\in A$
        we have $x\in\unions{\pow{A}}$ by \auto{pow_iff,unions_iff}.
    Follows by set extensionality.
\end{proof}

\begin{proposition}\label{inters_powerset}
    $\inters{\pow{A}} = \emptyset$.
\end{proposition}
\begin{proof}
    We have for all $x\in\inters{\pow{A}}$
        we have $x\in\emptyset$
        by \auto{inters_iff_forall,pow_iff}.
    We have for all $x\in\emptyset$
        we have $x\in\inters{\pow{A}}$ by \auto{emptyset}.
    Follows by set extensionality.
\end{proof}

\begin{proposition}\label{union_powersets_subseteq}
    $\pow{A}\union\pow{B} \subseteq \pow{A\union B}$.
\end{proposition}
\begin{proof}
    $\pow{A}\subseteq \pow{A}\union\pow{B}$ by \obv{union_upper_left}.
    $\pow{B}\subseteq \pow{A}\union\pow{B}$ by \obv{union_upper_right}.
    Follows by \auto{subseteq,pow_iff,powerset_elim,union_iff,subseteq_transitive}.
\end{proof}

\begin{proposition}\label{powerset_emptyset}
    $\pow{\emptyset} = \{\emptyset\}$.
\end{proposition}
\begin{proof}
    Follows by
    \auto{pow_iff,subseteq,emptyset_subseteq,singleton_iff,setext}.
\end{proof}

\begin{proposition}\label{powerset_union_subseteq}
    $\pow{A}\union\pow{B}\subseteq \pow{A\union B}$.
\end{proposition}
\begin{proof}
    Follows by \obv{union_powersets_subseteq}.
\end{proof}

\begin{proposition}\label{subseteq_pow_unions}
    $A\subseteq\pow{\unions{A}}$.
\end{proposition}
\begin{proof}
    Follows by \auto{subseteq,pow_iff,unions_intro}.
\end{proof}

\begin{proposition}\label{unions_pow}
    $\unions{\pow{A}} = A$.
\end{proposition}
\begin{proof}
    Follows by \obv{unions_powerset}.
\end{proof}


\begin{proposition}\label{unions_elem_pow_iff}
    $\unions{A}\in\pow{B}$ iff $A\in\pow{\pow{B}}$.
\end{proposition}
\begin{proof}
    Follows by \auto{pow_iff,unions_iff,subseteq}.
\end{proof}

\begin{proposition}\label{pow_inter}
    $\pow{A\inter B} = \pow{A}\inter\pow{B}$.
\end{proposition}
\begin{proof}
    We have for all $X\in\pow{A\inter B}$
        we have $X\in\pow{A}\inter\pow{B}$
        by \auto{inter,powerset_elim,powerset_intro,subseteq_inter_iff}.
    We have for all $X\in\pow{A}\inter\pow{B}$
        we have $X\in\pow{A\inter B}$
        by \auto{inter,powerset_elim,powerset_intro,subseteq_inter_iff}.
    Follows by set extensionality.
\end{proof}
