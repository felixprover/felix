\section{Finite intersections in filters}

\import{set/filter.tex}
\import{set/finite-intersection-family.tex}

\begin{proposition}\label{filter_inhabited_finite_intersection_in}
    Suppose $F$ is a filter on $S$.
    Suppose $G\subseteq F$ and $G$ is finite and $G$ is inhabited.
    Then $\inters{G}\in F$.
\end{proposition}
\begin{proof}
    Let $C=\{H\in\pow{G}\mid
        \text{$H=\emptyset$ or $\inters{H}\in F$}\}$.
    $\emptyset\in C$ by \obv{powerset_bottom}.
    Show that for all $H,h$ such that $H\in C$ and $h\in G$ we have
        $\cons{h}{H}\in C$.
    \begin{subproof}
        Fix $H$.
        Fix $h$.
        Assume $H\in C$ and $h\in G$.
        $H\subseteq G$ by \obv{powerset_elim}.
        Thus $\cons{h}{H}\subseteq G$ by \obv{cons_subseteq_intro}.
        Hence $\cons{h}{H}\in\pow{G}$ by \obv{powerset_intro}.
        $h\in F$ by \obv{subseteq}.
        \begin{byCase}
        \caseOf{$H=\emptyset$.}
            Every element of $\cons{h}{H}$ is an element of $\{h\}$
                by \obv{cons_iff,singleton_iff}.
            Every element of $\{h\}$ is an element of $\cons{h}{H}$
                by \obv{cons_iff,singleton_iff}.
            Hence $\cons{h}{H}=\{h\}$ by set extensionality.
            Hence $\inters{\cons{h}{H}}=h$
                by \auto{inters_singleton}.
            Thus $\inters{\cons{h}{H}}\in F$.
        \caseOf{$H\neq\emptyset$.}
            $H$ is inhabited by \obv{nonempty_inhabited}.
            $\inters{H}\in F$.
            Hence $h\inter\inters{H}\in F$ by \obv{filter}.
            $\inters{\cons{h}{H}}=h\inter\inters{H}$
                by \obv{inters_cons}.
            Thus $\inters{\cons{h}{H}}\in F$.
        \end{byCase}
    \end{subproof}
    $G\in C$ by \obv{finite_induction}.
    $G\neq\emptyset$ by \obv{emptyset}.
\end{proof}

\begin{theorem}\label{filter_bounded_finite_intersection_in}
    Suppose $F$ is a filter on $S$.
    Suppose $G\subseteq F$ and $G$ is finite.
    Then $\boundedIntersection{S}{G}\in F$.
\end{theorem}
\begin{proof}
    \begin{byCase}
    \caseOf{$G=\emptyset$.}
        $\boundedIntersection{S}{G}=S$ by
            \auto{bounded_intersection_empty}.
        $S\in F$ by \obv{filter}.
    \caseOf{$G\neq\emptyset$.}
        $G$ is inhabited by \obv{nonempty_inhabited}.
        $\inters{G}\in F$ by
            \obv{filter_inhabited_finite_intersection_in}.
        $G\subseteq\pow{S}$ by
            \auto{filter,pow_iff,subseteq}.
        $\boundedIntersection{S}{G}=\inters{G}$ by
            \obv{bounded_intersection_eq_inters}.
    \end{byCase}
\end{proof}

\begin{proposition}\label{filter_finite_intersection_in}
    Suppose $F$ is a filter on $S$.
    Suppose $G\subseteq F$ and $G$ is finite and $G$ is inhabited.
    Then $\inters{G}\in F$.
\end{proposition}
\begin{proof}
    $\boundedIntersection{S}{G}\in F$ by
        \obv{filter_bounded_finite_intersection_in}.
    $G\subseteq\pow{S}$ by
        \auto{filter,pow_iff,subseteq}.
    $\boundedIntersection{S}{G}=\inters{G}$ by
        \obv{bounded_intersection_eq_inters}.
\end{proof}

\begin{theorem}\label{filter_finite_family_intersection}
    Let $F$ be a filter on $S$ and $G$ be a finite set.
    Suppose $G\subseteq F$ and $G$ is inhabited.
    Then $\inters{G}\in F$.
\end{theorem}
\begin{proof}
    Follows by \obv{filter_finite_intersection_in}.
\end{proof}
