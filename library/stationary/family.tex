\section{Stationary families}

\import{cardinal/regularity.tex}
\import{stationary/club.tex}
\import{set/indexed-family.tex}

Stationary subpartitions retain their index set and ambient ordinal explicitly.
The carrier $S$ bounds every cell; exact coverage is a separate refinement.

\begin{definition}\label{stationary_subpartition}
    $\stationarySubpartition{P}{S}{\kappa}$ iff
        $S\subseteq\kappa$ and
        $P\in\funs{\kappa}{\pow{S}}$ and
        $P$ is pairwise disjoint on $\kappa$ and
        for every $i\in\kappa$ we have
            $P(i)$ is stationary in $\kappa$.
\end{definition}

\begin{abbreviation}\label{stationary_subpartition_phrase}
    $P$ is a stationary subpartition of $S$ in $\kappa$ iff
        $\stationarySubpartition{P}{S}{\kappa}$.
\end{abbreviation}

\begin{definition}\label{stationary_partition}
    $\stationaryPartition{P}{S}{\kappa}$ iff
        $P$ is a stationary subpartition of $S$ in $\kappa$ and
        $\unions{\ran{P}}=S$.
\end{definition}

\begin{proposition}\label{stationary_subpartition_elim}
    Suppose $P$ is a stationary subpartition of $S$ in $\kappa$.
    Then $S\subseteq\kappa$ and
        $P\in\funs{\kappa}{\pow{S}}$ and
        $P$ is pairwise disjoint on $\kappa$ and
        for every $i\in\kappa$ we have
            $P(i)$ is stationary in $\kappa$.
\end{proposition}
\begin{proof}
    Follows by \obv{stationary_subpartition}.
\end{proof}

\begin{proposition}\label{stationary_partition_elim}
    Suppose $\stationaryPartition{P}{S}{\kappa}$.
    Then $P$ is a stationary subpartition of $S$ in $\kappa$ and
        $\unions{\ran{P}}=S$.
\end{proposition}
\begin{proof}
    Follows by \obv{stationary_partition}.
\end{proof}

\begin{proposition}\label{stationary_partition_is_subpartition}
    Suppose $\stationaryPartition{P}{S}{\kappa}$.
    Then $P$ is a stationary subpartition of $S$ in $\kappa$.
\end{proposition}
\begin{proof}
    Follows by \obv{stationary_partition_elim}.
\end{proof}

\begin{proposition}\label{stationary_partition_cell}
    Suppose $\stationaryPartition{P}{S}{\kappa}$.
    Let $i\in\kappa$.
    Then $P(i)\subseteq S$ and $P(i)$ is stationary in $\kappa$.
\end{proposition}
\begin{proof}
    $P\in\funs{\kappa}{\pow{S}}$ and
        for every $j\in\kappa$ we have $P(j)$ is stationary in $\kappa$
        by \auto{stationary_partition_is_subpartition,
            stationary_subpartition_elim}.
    $P(i)\in\pow{S}$ by \obv{funs_type_apply}.
    Follows by \obv{powerset_elim}.
\end{proof}

\begin{proposition}\label{stationary_partition_cover}
    Suppose $\stationaryPartition{P}{S}{\kappa}$.
    Then $\unions{\ran{P}}=S$.
\end{proposition}
\begin{proof}
    Follows by \obv{stationary_partition_elim}.
\end{proof}

\begin{proposition}\label{stationary_subpartition_superset}
    Suppose $P$ is a stationary subpartition of $R$ in $\kappa$.
    Suppose $R\subseteq S\subseteq\kappa$.
    Then $P$ is a stationary subpartition of $S$ in $\kappa$.
\end{proposition}
\begin{proof}
    $P\in\funs{\kappa}{\pow{R}}$ and
        $P$ is pairwise disjoint on $\kappa$ and
        for every $i\in\kappa$ we have $P(i)$ is stationary in $\kappa$
        by \obv{stationary_subpartition_elim}.
    $\pow{R}\subseteq\pow{S}$ by \auto{subseteq,pow_iff}.
    Hence $P\in\funs{\kappa}{\pow{S}}$ by \obv{funs_weaken_codom}.
    Follows by \obv{stationary_subpartition}.
\end{proof}

\begin{theorem}\label{stationary_subpartition_reindex_bijection}
    Let $e\in\bijections{\kappa}{J}$.
    Let $Q\in\funs{J}{\pow{S}}$.
    Suppose $Q$ is pairwise disjoint on $J$.
    Suppose for every $j\in J$ we have $Q(j)$ is stationary in $\kappa$.
    Suppose $S\subseteq\kappa$.
    Then $Q\circ e$ is a stationary subpartition of $S$ in $\kappa$.
\end{theorem}
\begin{proof}
    $e\in\funs{\kappa}{J}$ by \obv{bijections_to_funs}.
    Hence $e$ is a function and $\dom{e}=\kappa$
        by \obv{funs_is_function,funs}.
    $e$ is injective by \obv{bijections}.
    Hence $e\in\inj{\kappa}{J}$
        by \auto{inj,injective_function}.
    $Q\circ e\in\funs{\kappa}{\pow{S}}$ by \obv{funs_circ}.
    $Q$ is a function and $\dom{Q}=J$
        by \obv{funs_is_function,funs}.
    $Q$ is composable with $e$ by \obv{funs_ran,subseteq}.
    $Q\circ e$ is pairwise disjoint on $\kappa$
        by \obv{pairwise_disjoint_on_reindex_injection}.
    Show that for every $i\in\kappa$ we have
        $(Q\circ e)(i)$ is stationary in $\kappa$.
    \begin{subproof}
        Fix $i\in\kappa$.
        $e(i)\in J$ by \obv{funs_type_apply}.
        Hence $Q(e(i))$ is stationary in $\kappa$.
        $(Q\circ e)(i)=Q(e(i))$ by \obv{circ_apply}.
    \end{subproof}
    Follows by \obv{stationary_subpartition}.
\end{proof}

\begin{theorem}\label{cofinal_stationary_cells_to_subpartition}
    Let $\kappa$ be a regular cardinal.
    Suppose $\cofinalIn{H}{\kappa}$.
    Let $Q\in\funs{H}{\pow{S}}$.
    Suppose $Q$ is pairwise disjoint on $H$.
    Suppose for every $\eta\in H$ we have
        $Q(\eta)$ is stationary in $\kappa$.
    Suppose $S\subseteq\kappa$.
    Then there exists $P$ such that
        $P$ is a stationary subpartition of $S$ in $\kappa$.
\end{theorem}
\begin{proof}
    $H\approx\kappa$
        by \obv{cofinal_subset_equinumerous_regular_cardinal}.
    Hence $\kappa\approx H$ by \obv{equinum_sym}.
    Take $e$ such that $e\in\bijections{\kappa}{H}$
        by \obv{equinum}.
    Let $P=Q\circ e$.
    $P$ is a stationary subpartition of $S$ in $\kappa$
        by \auto{stationary_subpartition_reindex_bijection}.
\end{proof}

\begin{theorem}\label{stationary_subpartition_completion}
    Let $\kappa$ be an infinite cardinal.
    Suppose $P$ is a stationary subpartition of $S$ in $\kappa$.
    Let $Q=\completeDisjointFamilyAt{P}{S}{\kappa}{\emptyset}$.
    Then $\stationaryPartition{Q}{S}{\kappa}$.
\end{theorem}
\begin{proof}
    $S\subseteq\kappa$ and $P\in\funs{\kappa}{\pow{S}}$ and
        $P$ is pairwise disjoint on $\kappa$ and
        for every $i\in\kappa$ we have $P(i)$ is stationary in $\kappa$
        by \obv{stationary_subpartition_elim}.
    $\emptyset\in\kappa$ by \obv{zero_one_in_infinite_cardinal}.
    $Q\in\funs{\kappa}{\pow{S}}$
        by \auto{complete_disjoint_family_at_type}.
    $Q$ is pairwise disjoint on $\kappa$
        by \auto{complete_disjoint_family_at_pairwise}.
    Show that for every $i\in\kappa$ we have
        $Q(i)$ is stationary in $\kappa$.
    \begin{subproof}
        Fix $i\in\kappa$.
        \begin{byCase}
        \caseOf{$i=\emptyset$.}
            $Q(i)=P(\emptyset)\union\familyRemainder{P}{S}$
                by \auto{complete_disjoint_family_at_selected}.
            $P(\emptyset)\subseteq Q(i)$ by \auto{union_iff,subseteq}.
            $Q(i)\in\pow{S}$ by \obv{funs_type_apply}.
            Hence $Q(i)\subseteq S$ by \obv{powerset_elim}.
            Hence $Q(i)\subseteq\kappa$ by \obv{subseteq_transitive}.
            $P(\emptyset)$ is stationary in $\kappa$.
            Follows by \obv{stationary_superset}.
        \caseOf{$i\neq\emptyset$.}
            $Q(i)=P(i)$ by \auto{complete_disjoint_family_at_other}.
            Follows by \auto.
        \end{byCase}
    \end{subproof}
    Hence $Q$ is a stationary subpartition of $S$ in $\kappa$
        by \obv{stationary_subpartition}.
    $\unions{\ran{Q}}=S$ by \auto{complete_disjoint_family_at_cover}.
    Follows by \obv{stationary_partition}.
\end{proof}
