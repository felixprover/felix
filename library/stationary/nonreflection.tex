\section{Self-nonreflection and canonical club gaps}

\import{stationary/family.tex}
\import{stationary/fodor.tex}

The supplied-sequence interface separates the mathematical gap construction
from the one set-indexed choice used to obtain local avoiding clubs.

\begin{definition}\label{self_nonreflecting_in}
    $\selfNonreflectingIn{T}{\kappa}$ iff
        $T\subseteq\kappa$ and
        for every $\beta\in T$ we have
            $\nonstationaryIn{T\inter\beta}{\beta}$.
\end{definition}

\begin{definition}\label{avoiding_club_sequence_on}
    $\avoidingClubSequenceOn{C}{T}{\kappa}$ iff
        $\kappa$ is an ordinal and $T\subseteq\kappa$ and
        $C\in\funs{T}{\pow{\kappa}}$ and
        for every $\beta\in T$ we have
            $\clubIn{C(\beta)}{\beta}$ and $C(\beta)\notmeets T$.
\end{definition}

\begin{definition}\label{avoiding_club_candidates}
    $\avoidingClubCandidates{T}{\beta}
        =\{C\in\pow{\beta}\mid
            \clubIn{C}{\beta}\land C\notmeets(T\inter\beta)\}$.
\end{definition}

\begin{definition}\label{chosen_avoiding_club_sequence}
    $\chosenAvoidingClubSequence{T}
        =\{(\beta,\choice{\avoidingClubCandidates{T}{\beta}})
            \mid \beta\in T\}$.
\end{definition}

\begin{definition}\label{gap_cell}
    $\gapCell{C}{T}{\xi}{\eta}
        =\{\beta\in T\mid \xi\in\beta\land
            \leastMemberAbove{C(\beta)}{\xi}{\beta}=\eta\}$.
\end{definition}

\begin{definition}\label{lower_gap_map}
    $\lowerGapMap{C}{T}{\alpha}
        =\{(\beta,\unions{C(\beta)\inter\alpha})
            \mid \beta\in T\setminus\suc{\alpha}\}$.
\end{definition}

\begin{definition}\label{lower_gap_value}
    $\lowerGapValue{C}{T}{\kappa}{\alpha}
        =\fodorValue{\lowerGapMap{C}{T}{\alpha}}{\kappa}$.
\end{definition}

\begin{definition}\label{lower_gap_fibre}
    $\lowerGapFibre{C}{T}{\kappa}{\alpha}
        =\fodorFibre{\lowerGapMap{C}{T}{\alpha}}{\kappa}$.
\end{definition}

\begin{definition}\label{upper_gap_map}
    $\upperGapMap{C}{T}{\kappa}{\alpha}
        =\{(\beta,\leastMemberAbove{C(\beta)}
                {\lowerGapValue{C}{T}{\kappa}{\alpha}}{\beta})
            \mid \beta\in\lowerGapFibre{C}{T}{\kappa}{\alpha}\}$.
\end{definition}

\begin{definition}\label{upper_gap_value}
    $\upperGapValue{C}{T}{\kappa}{\alpha}
        =\fodorValue{\upperGapMap{C}{T}{\kappa}{\alpha}}{\kappa}$.
\end{definition}

\begin{definition}\label{upper_gap_fibre}
    $\upperGapFibre{C}{T}{\kappa}{\alpha}
        =\fodorFibre{\upperGapMap{C}{T}{\kappa}{\alpha}}{\kappa}$.
\end{definition}

\begin{definition}\label{lower_endpoint_map}
    $\lowerEndpointMap{C}{T}{\kappa}
        =\{(\alpha,\lowerGapValue{C}{T}{\kappa}{\alpha})
            \mid \alpha\in T\}$.
\end{definition}

\begin{proposition}\label{self_nonreflecting_points_are_limits}
    Suppose $\selfNonreflectingIn{T}{\kappa}$.
    Let $\beta\in T$.
    Then $\beta$ is a limit ordinal.
\end{proposition}
\begin{proof}
    $\nonstationaryIn{T\inter\beta}{\beta}$
        by \obv{self_nonreflecting_in}.
    Follows by \obv{nonstationary_in}.
\end{proof}

\begin{proposition}\label{avoiding_club_sequence_on_type}
    Suppose $\avoidingClubSequenceOn{C}{T}{\kappa}$.
    Then $\kappa$ is an ordinal and $T\subseteq\kappa$ and
        $C\in\funs{T}{\pow{\kappa}}$.
\end{proposition}
\begin{proof}
    Follows by \obv{avoiding_club_sequence_on}.
\end{proof}

\begin{proposition}\label{avoiding_club_sequence_on_club}
    Suppose $\avoidingClubSequenceOn{C}{T}{\kappa}$.
    Let $\beta\in T$.
    Then $\clubIn{C(\beta)}{\beta}$.
\end{proposition}
\begin{proof}
    Follows by \obv{avoiding_club_sequence_on}.
\end{proof}

\begin{proposition}\label{avoiding_club_sequence_on_avoids}
    Suppose $\avoidingClubSequenceOn{C}{T}{\kappa}$.
    Let $\beta\in T$.
    Then $C(\beta)\notmeets T$.
\end{proposition}
\begin{proof}
    Follows by \obv{avoiding_club_sequence_on}.
\end{proof}

\begin{proposition}\label{avoiding_club_sequence_on_implies_self_nonreflecting}
    Suppose $\avoidingClubSequenceOn{C}{T}{\kappa}$.
    Then $\selfNonreflectingIn{T}{\kappa}$.
\end{proposition}
\begin{proof}
    $T\subseteq\kappa$ by \obv{avoiding_club_sequence_on_type}.
    Show that for every $\beta\in T$ we have
        $\nonstationaryIn{T\inter\beta}{\beta}$.
    \begin{subproof}
        Fix $\beta\in T$.
        $\clubIn{C(\beta)}{\beta}$
            by \obv{avoiding_club_sequence_on_club}.
        $T\inter\beta\subseteq\beta$ by \obv{subseteq,inter}.
        $C(\beta)\notmeets T$
            by \obv{avoiding_club_sequence_on_avoids}.
        Hence $C(\beta)\notmeets(T\inter\beta)$
            by \auto{disjoint,inter_elim_left}.
        Hence $(T\inter\beta)\notmeets C(\beta)$ by \auto{disjoint}.
        Follows by \obv{nonstationary_in}.
    \end{subproof}
    Follows by \obv{self_nonreflecting_in}.
\end{proof}

\begin{proposition}\label{avoiding_club_candidates_inhabited}
    Suppose $\selfNonreflectingIn{T}{\kappa}$.
    Let $\beta\in T$.
    Then $\avoidingClubCandidates{T}{\beta}$ is inhabited.
\end{proposition}
\begin{proof}
    $\nonstationaryIn{T\inter\beta}{\beta}$
        by \obv{self_nonreflecting_in}.
    Take $C$ such that $\clubIn{C}{\beta}$ and
        $(T\inter\beta)\notmeets C$ by \obv{nonstationary_in}.
    $C\subseteq\beta$ by \obv{club_in_elim}.
    Hence $C\in\pow{\beta}$ by \obv{powerset_intro}.
    $C\notmeets(T\inter\beta)$ by \auto{disjoint}.
    Hence $C\in\avoidingClubCandidates{T}{\beta}$
        by \obv{avoiding_club_candidates}.
\end{proof}

\begin{theorem}\label{chosen_avoiding_club_sequence_on}
    Let $\kappa$ be an ordinal.
    Suppose $\selfNonreflectingIn{T}{\kappa}$.
    Then $\avoidingClubSequenceOn{\chosenAvoidingClubSequence{T}}{T}{\kappa}$.
\end{theorem}
\begin{proof}
    Let $F=\chosenAvoidingClubSequence{T}$.
    $T\subseteq\kappa$ by \obv{self_nonreflecting_in}.
    $F$ is a relation by \auto{relation,chosen_avoiding_club_sequence}.
    $F$ is right-unique
        by \auto{rightunique,chosen_avoiding_club_sequence,pair_eq_iff}.
    $\dom{F}=T$
        by \auto{setext,dom_iff,chosen_avoiding_club_sequence,pair_eq_iff}.
    Show that for every $\beta\in\dom{F}$ we have $F(\beta)\in\pow{\kappa}$.
    \begin{subproof}
        Fix $\beta\in\dom{F}$.
        Hence $\beta\in T$.
        $\avoidingClubCandidates{T}{\beta}$ is inhabited
            by \obv{avoiding_club_candidates_inhabited}.
        Hence $\choice{\avoidingClubCandidates{T}{\beta}}
            \in\avoidingClubCandidates{T}{\beta}$ by \obv{choice_in}.
        Hence $\choice{\avoidingClubCandidates{T}{\beta}}\in\pow{\beta}$
            by \obv{avoiding_club_candidates}.
        Hence $\choice{\avoidingClubCandidates{T}{\beta}}\subseteq\beta$
            by \obv{powerset_elim}.
        $\beta\in\kappa$ by \obv{subseteq}.
        Hence $\beta$ is an ordinal by \obv{elem_of_ordinal_is_ordinal}.
        Hence $\beta\subseteq\kappa$ by \obv{ordinal_elem_implies_subseteq}.
        Hence $\choice{\avoidingClubCandidates{T}{\beta}}\subseteq\kappa$
            by \obv{subseteq_transitive}.
        Hence $\choice{\avoidingClubCandidates{T}{\beta}}\in\pow{\kappa}$
            by \obv{powerset_intro}.
        $(\beta,\choice{\avoidingClubCandidates{T}{\beta}})\in F$
            by \obv{chosen_avoiding_club_sequence}.
        Hence $F(\beta)=\choice{\avoidingClubCandidates{T}{\beta}}$
            by \obv{function_apply_intro}.
    \end{subproof}
    Hence $F\in\funs{T}{\pow{\kappa}}$ by \obv{funs_intro}.
    Show that for every $\beta\in T$ we have
        $\clubIn{F(\beta)}{\beta}$ and $F(\beta)\notmeets T$.
    \begin{subproof}
        Fix $\beta\in T$.
        $\avoidingClubCandidates{T}{\beta}$ is inhabited
            by \obv{avoiding_club_candidates_inhabited}.
        Hence $\choice{\avoidingClubCandidates{T}{\beta}}
            \in\avoidingClubCandidates{T}{\beta}$ by \obv{choice_in}.
        Hence $\choice{\avoidingClubCandidates{T}{\beta}}\subseteq\beta$ and
            $\clubIn{\choice{\avoidingClubCandidates{T}{\beta}}}{\beta}$ and
            $\choice{\avoidingClubCandidates{T}{\beta}}
                \notmeets(T\inter\beta)$
            by \obv{avoiding_club_candidates,powerset_elim}.
        $(\beta,\choice{\avoidingClubCandidates{T}{\beta}})\in F$
            by \obv{chosen_avoiding_club_sequence}.
        Hence $F(\beta)=\choice{\avoidingClubCandidates{T}{\beta}}$
            by \obv{function_apply_intro}.
        Show that $F(\beta)\notmeets T$.
        \begin{subproof}
            Suppose not.
            Take $x$ such that $x\in F(\beta),T$ by \auto{disjoint}.
            Hence $x\in\beta$ by \obv{subseteq}.
            Hence $x\in T\inter\beta$ by \obv{inter_intro}.
            Contradiction by \auto{disjoint}.
        \end{subproof}
    \end{subproof}
    Follows by \auto{avoiding_club_sequence_on}.
\end{proof}

\begin{proposition}\label{gap_cell_subset}
    $\gapCell{C}{T}{\xi}{\eta}\subseteq T$.
\end{proposition}
\begin{proof}
    Follows by \obv{gap_cell,subseteq}.
\end{proof}

\begin{proposition}\label{gap_cell_value_bounds}
    Let $\beta\in\gapCell{C}{T}{\xi}{\eta}$.
    Suppose $\clubIn{C(\beta)}{\beta}$.
    Then $\xi\in\eta$ and $\eta\in\beta$.
\end{proposition}
\begin{proof}
    $\xi\in\beta$ and
        $\leastMemberAbove{C(\beta)}{\xi}{\beta}=\eta$
        by \obv{gap_cell}.
    $\beta$ is a limit ordinal and
        $\ordinalUnbounded{C(\beta)}{\beta}$ by \obv{club_in_elim}.
    $\xi\in\leastMemberAbove{C(\beta)}{\xi}{\beta}$ and
        $\leastMemberAbove{C(\beta)}{\xi}{\beta}\in\beta$
        by \obv{least_member_above_strict,
            least_member_above_below_ambient}.
\end{proof}

\begin{proposition}\label{gap_cells_pairwise_disjoint}
    Suppose $\eta\neq\zeta$.
    Then $\gapCell{C}{T}{\xi}{\eta}$ is disjoint from
        $\gapCell{C}{T}{\xi}{\zeta}$.
\end{proposition}
\begin{proof}
    Suppose not.
    Take $\beta$ such that
        $\beta\in\gapCell{C}{T}{\xi}{\eta},
            \gapCell{C}{T}{\xi}{\zeta}$ by \auto{disjoint}.
    $\leastMemberAbove{C(\beta)}{\xi}{\beta}=\eta$ and
        $\leastMemberAbove{C(\beta)}{\xi}{\beta}=\zeta$
        by \obv{gap_cell}.
    Contradiction.
\end{proof}

\begin{proposition}\label{lower_gap_map_value_properties}
    Suppose $\avoidingClubSequenceOn{C}{T}{\kappa}$.
    Let $\alpha\in T$.
    Let $\beta\in T\setminus\suc{\alpha}$.
    Let $f=\lowerGapMap{C}{T}{\alpha}$.
    Then $\alpha\in\beta$ and
        $f(\beta)=\unions{C(\beta)\inter\alpha}$ and
        $f(\beta)\in\alpha$ and
        $\alpha\in\leastMemberAbove{C(\beta)}{f(\beta)}{\beta}$ and
        $\leastMemberAbove{C(\beta)}{f(\beta)}{\beta}\in\beta$.
\end{proposition}
\begin{proof}
    $\kappa$ is an ordinal and $T\subseteq\kappa$
        by \obv{avoiding_club_sequence_on_type}.
    $\alpha,\beta\in\kappa$ by \obv{subseteq,setminus_elim_left}.
    Hence $\alpha$ and $\beta$ are ordinals
        by \obv{elem_of_ordinal_is_ordinal}.
    $\beta\notin\suc{\alpha}$ by \obv{setminus_elim_right}.
    Hence $\beta\notin\alpha$ and $\beta\neq\alpha$ by \auto{suc_iff}.
    Hence $\alpha\in\beta$ by \obv{ordinal_elem_connex}.
    $\clubIn{C(\alpha)}{\alpha}$ and $\clubIn{C(\beta)}{\beta}$
        by \obv{avoiding_club_sequence_on_club,setminus_elim_left}.
    Hence $\alpha$ is a limit ordinal by \obv{club_in_elim}.
    $C(\beta)\notmeets T$
        by \obv{avoiding_club_sequence_on_avoids,setminus_elim_left}.
    Hence $\alpha\notin C(\beta)$ by \auto{disjoint}.
    Let $\xi=\unions{C(\beta)\inter\alpha}$.
    $\xi\in\alpha$ and
        $\alpha\in\leastMemberAbove{C(\beta)}{\xi}{\beta}$ and
        $\leastMemberAbove{C(\beta)}{\xi}{\beta}\in\beta$
        by \obv{closed_club_gap_straddles}.
    $f$ is a relation by \auto{relation,lower_gap_map}.
    $f$ is right-unique
        by \auto{rightunique,lower_gap_map,pair_eq_iff}.
    $(\beta,\xi)\in f$ by \obv{lower_gap_map}.
    Hence $f(\beta)=\xi$ by \obv{function_apply_intro}.
\end{proof}

\begin{theorem}\label{lower_gap_map_type_and_regressive}
    Let $\kappa$ be a regular cardinal.
    Suppose $\naturals\in\kappa$.
    Suppose $T$ is stationary in $\kappa$.
    Suppose $\avoidingClubSequenceOn{C}{T}{\kappa}$.
    Let $\alpha\in T$.
    Then $\lowerGapMap{C}{T}{\alpha}
            \in\funs{T\setminus\suc{\alpha}}{\kappa}$ and
        $\lowerGapMap{C}{T}{\alpha}$ is regressive on
            $T\setminus\suc{\alpha}$.
\end{theorem}
\begin{proof}
    Let $D=T\setminus\suc{\alpha}$.
    Let $f=\lowerGapMap{C}{T}{\alpha}$.
    $\kappa$ is an ordinal by \obv{avoiding_club_sequence_on_type}.
    $T\subseteq\kappa$ by \obv{stationary_in_elim}.
    $f$ is a relation by \auto{relation,lower_gap_map}.
    $f$ is right-unique
        by \auto{rightunique,lower_gap_map,pair_eq_iff}.
    $\dom{f}=D$ by \auto{setext,dom_iff,lower_gap_map,pair_eq_iff}.
    Show that for every $\beta\in\dom{f}$ we have $f(\beta)\in\kappa$.
    \begin{subproof}
        Fix $\beta\in\dom{f}$.
        Hence $\beta\in D$.
        $f(\beta)\in\alpha$ and $\alpha\in\beta$
            by \auto{lower_gap_map_value_properties}.
        $\beta\in T$ by \obv{setminus_elim_left}.
        Hence $\beta\in\kappa$ by \obv{subseteq}.
        Follows by \obv{ordinal_transitivity}.
    \end{subproof}
    Hence $f\in\funs{D}{\kappa}$ by \obv{funs_intro}.
    Show that for every $\beta\in D\setminus\{\emptyset\}$ we have
        $f(\beta)\in\beta$.
    \begin{subproof}
        Fix $\beta\in D\setminus\{\emptyset\}$.
        Hence $\beta\in D$ by \obv{setminus_elim_left}.
        $f(\beta)\in\alpha$ and $\alpha\in\beta$
            by \auto{lower_gap_map_value_properties}.
        $\beta\in T$ by \obv{setminus_elim_left}.
        Hence $\beta\in\kappa$ by \obv{subseteq}.
        Hence $\beta$ is an ordinal
            by \obv{elem_of_ordinal_is_ordinal,
                avoiding_club_sequence_on_type}.
        Follows by \obv{ordinal_transitivity}.
    \end{subproof}
    Hence $f$ is regressive on $D$ by \obv{regressive_on}.
\end{proof}

\begin{theorem}\label{lower_gap_value_and_fibre_properties}
    Let $\kappa$ be a regular cardinal.
    Suppose $\naturals\in\kappa$.
    Suppose $T$ is stationary in $\kappa$.
    Suppose $\avoidingClubSequenceOn{C}{T}{\kappa}$.
    Let $\alpha\in T$.
    Then $\lowerGapValue{C}{T}{\kappa}{\alpha}\in\alpha$ and
        $\lowerGapFibre{C}{T}{\kappa}{\alpha}$ is stationary in $\kappa$ and
        $\lowerGapFibre{C}{T}{\kappa}{\alpha}
            \subseteq T\setminus\suc{\alpha}$ and
        for every $\beta\in\lowerGapFibre{C}{T}{\kappa}{\alpha}$ we have
            $(\lowerGapMap{C}{T}{\alpha})(\beta)
                =\lowerGapValue{C}{T}{\kappa}{\alpha}$.
\end{theorem}
\begin{proof}
    Let $D=T\setminus\suc{\alpha}$.
    Let $f=\lowerGapMap{C}{T}{\alpha}$.
    $\kappa$ is an infinite cardinal and $\cofinality{\kappa}=\kappa$
        by \obv{regular_cardinal}.
    $\kappa$ is a limit ordinal and $T\subseteq\kappa$
        by \obv{stationary_in_elim}.
    Hence $\alpha\in\kappa$ by \obv{subseteq}.
    Hence $\naturals\in\cofinality{\kappa}$.
    Hence $D$ is stationary in $\kappa$
        by \auto{stationary_strict_tail}.
    $f\in\funs{D}{\kappa}$ and $f$ is regressive on $D$
        by \auto{lower_gap_map_type_and_regressive}.
    $\fodorValue{f}{\kappa}\in\kappa$ and
        $\fodorFibre{f}{\kappa}$ is stationary in $\kappa$ and
        $\fodorFibre{f}{\kappa}\subseteq D$ and
        for every $\beta\in\fodorFibre{f}{\kappa}$ we have
            $f(\beta)=\fodorValue{f}{\kappa}$
        by \obv{fodor_value_and_fibre_properties}.
    Let $B=\lowerGapFibre{C}{T}{\kappa}{\alpha}$.
    Hence $B=\fodorFibre{f}{\kappa}$
        by \obv{lower_gap_fibre}.
    Hence $B\subseteq D$.
    $\emptyset\in\kappa$ by \obv{zero_one_in_infinite_cardinal}.
    $\ordinalUnbounded{B}{\kappa}$ by \auto{stationary_is_unbounded}.
    Take $\beta\in B$ such that $\emptyset\in\beta$
        by \obv{ordinal_unbounded}.
    Hence $\beta\in D$ by \obv{subseteq}.
    $f(\beta)\in\alpha$ by \auto{lower_gap_map_value_properties}.
    $f(\beta)=\fodorValue{f}{\kappa}$.
    $\lowerGapValue{C}{T}{\kappa}{\alpha}=\fodorValue{f}{\kappa}$
        by \obv{lower_gap_value}.
    Hence $\lowerGapValue{C}{T}{\kappa}{\alpha}\in\alpha$.
    Follows by \auto{lower_gap_fibre,lower_gap_value}.
\end{proof}

\begin{proposition}\label{upper_gap_map_value_properties}
    Let $\kappa$ be a regular cardinal.
    Suppose $\naturals\in\kappa$.
    Suppose $T$ is stationary in $\kappa$.
    Suppose $\avoidingClubSequenceOn{C}{T}{\kappa}$.
    Let $\alpha\in T$.
    Let $\beta\in\lowerGapFibre{C}{T}{\kappa}{\alpha}$.
    Let $g=\upperGapMap{C}{T}{\kappa}{\alpha}$.
    Then $\alpha\in\beta$ and
        $g(\beta)=\leastMemberAbove{C(\beta)}
            {\lowerGapValue{C}{T}{\kappa}{\alpha}}{\beta}$ and
        $\alpha\in g(\beta)$ and $g(\beta)\in\beta$.
\end{proposition}
\begin{proof}
    Let $B=\lowerGapFibre{C}{T}{\kappa}{\alpha}$.
    Let $f=\lowerGapMap{C}{T}{\alpha}$.
    $B\subseteq T\setminus\suc{\alpha}$ and
        for every $\gamma\in B$ we have
            $f(\gamma)=\lowerGapValue{C}{T}{\kappa}{\alpha}$
        by \auto{lower_gap_value_and_fibre_properties}.
    Hence $\beta\in T\setminus\suc{\alpha}$ by \obv{subseteq}.
    $\alpha\in\beta$ and
        $f(\beta)=\unions{C(\beta)\inter\alpha}$ and
        $\alpha\in\leastMemberAbove{C(\beta)}{f(\beta)}{\beta}$ and
        $\leastMemberAbove{C(\beta)}{f(\beta)}{\beta}\in\beta$
        by \auto{lower_gap_map_value_properties}.
    Hence $\alpha\in\leastMemberAbove{C(\beta)}
            {\lowerGapValue{C}{T}{\kappa}{\alpha}}{\beta}$ and
        $\leastMemberAbove{C(\beta)}
            {\lowerGapValue{C}{T}{\kappa}{\alpha}}{\beta}\in\beta$.
    $g$ is a relation by \auto{relation,upper_gap_map}.
    $g$ is right-unique
        by \auto{rightunique,upper_gap_map,pair_eq_iff}.
    $(\beta,\leastMemberAbove{C(\beta)}
        {\lowerGapValue{C}{T}{\kappa}{\alpha}}{\beta})\in g$
        by \obv{upper_gap_map}.
    Hence $g(\beta)=\leastMemberAbove{C(\beta)}
        {\lowerGapValue{C}{T}{\kappa}{\alpha}}{\beta}$
        by \obv{function_apply_intro}.
\end{proof}

\begin{theorem}\label{upper_gap_map_type_and_regressive}
    Let $\kappa$ be a regular cardinal.
    Suppose $\naturals\in\kappa$.
    Suppose $T$ is stationary in $\kappa$.
    Suppose $\avoidingClubSequenceOn{C}{T}{\kappa}$.
    Let $\alpha\in T$.
    Let $B=\lowerGapFibre{C}{T}{\kappa}{\alpha}$.
    Then $\upperGapMap{C}{T}{\kappa}{\alpha}\in\funs{B}{\kappa}$ and
        $\upperGapMap{C}{T}{\kappa}{\alpha}$ is regressive on $B$.
\end{theorem}
\begin{proof}
    Let $g=\upperGapMap{C}{T}{\kappa}{\alpha}$.
    $\kappa$ is an ordinal and $T\subseteq\kappa$
        by \obv{avoiding_club_sequence_on_type}.
    $B\subseteq T\setminus\suc{\alpha}$
        by \auto{lower_gap_value_and_fibre_properties}.
    $g$ is a relation by \auto{relation,upper_gap_map}.
    $g$ is right-unique
        by \auto{rightunique,upper_gap_map,pair_eq_iff}.
    $\dom{g}=B$ by \auto{setext,dom_iff,upper_gap_map,pair_eq_iff}.
    Show that for every $\beta\in\dom{g}$ we have $g(\beta)\in\kappa$.
    \begin{subproof}
        Fix $\beta\in\dom{g}$.
        Hence $\beta\in B$.
        $g(\beta)\in\beta$ by \auto{upper_gap_map_value_properties}.
        $\beta\in T$ by \obv{subseteq,setminus_elim_left}.
        Hence $\beta\in\kappa$ by \obv{subseteq}.
        Follows by \obv{ordinal_transitivity}.
    \end{subproof}
    Hence $g\in\funs{B}{\kappa}$ by \obv{funs_intro}.
    Show that for every $\beta\in B\setminus\{\emptyset\}$ we have
        $g(\beta)\in\beta$.
    \begin{subproof}
        Fix $\beta\in B\setminus\{\emptyset\}$.
        Hence $\beta\in B$ by \obv{setminus_elim_left}.
        Follows by \auto{upper_gap_map_value_properties}.
    \end{subproof}
    Hence $g$ is regressive on $B$ by \obv{regressive_on}.
\end{proof}

\begin{theorem}\label{upper_gap_value_and_fibre_properties}
    Let $\kappa$ be a regular cardinal.
    Suppose $\naturals\in\kappa$.
    Suppose $T$ is stationary in $\kappa$.
    Suppose $\avoidingClubSequenceOn{C}{T}{\kappa}$.
    Let $\alpha\in T$.
    Then $\upperGapValue{C}{T}{\kappa}{\alpha}\in\kappa$ and
        $\upperGapFibre{C}{T}{\kappa}{\alpha}$ is stationary in $\kappa$ and
        $\alpha\in\upperGapValue{C}{T}{\kappa}{\alpha}$ and
        $\upperGapFibre{C}{T}{\kappa}{\alpha}\subseteq
            \gapCell{C}{T}{\lowerGapValue{C}{T}{\kappa}{\alpha}}
                {\upperGapValue{C}{T}{\kappa}{\alpha}}$.
\end{theorem}
\begin{proof}
    Let $B=\lowerGapFibre{C}{T}{\kappa}{\alpha}$.
    Let $g=\upperGapMap{C}{T}{\kappa}{\alpha}$.
    $\kappa$ is an ordinal and $T\subseteq\kappa$
        by \obv{avoiding_club_sequence_on_type}.
    $B$ is stationary in $\kappa$ and
        $B\subseteq T\setminus\suc{\alpha}$ and
        $\lowerGapValue{C}{T}{\kappa}{\alpha}\in\alpha$
        by \auto{lower_gap_value_and_fibre_properties}.
    $g\in\funs{B}{\kappa}$ and $g$ is regressive on $B$
        by \auto{upper_gap_map_type_and_regressive}.
    $\fodorValue{g}{\kappa}\in\kappa$ and
        $\fodorFibre{g}{\kappa}$ is stationary in $\kappa$ and
        $\fodorFibre{g}{\kappa}\subseteq B$ and
        for every $\beta\in\fodorFibre{g}{\kappa}$ we have
            $g(\beta)=\fodorValue{g}{\kappa}$
        by \obv{fodor_value_and_fibre_properties}.
    Let $U=\upperGapFibre{C}{T}{\kappa}{\alpha}$.
    Hence $U=\fodorFibre{g}{\kappa}$ by \obv{upper_gap_fibre}.
    Hence $U\subseteq B$.
    $\emptyset\in\kappa$ by \obv{zero_one_in_infinite_cardinal}.
    $\ordinalUnbounded{U}{\kappa}$ by \obv{stationary_is_unbounded}.
    Take $\beta\in U$ such that $\emptyset\in\beta$
        by \obv{ordinal_unbounded}.
    Hence $\beta\in B$ by \obv{subseteq}.
    $\alpha\in g(\beta)$ by \auto{upper_gap_map_value_properties}.
    $g(\beta)=\fodorValue{g}{\kappa}$.
    $\upperGapValue{C}{T}{\kappa}{\alpha}=\fodorValue{g}{\kappa}$
        by \obv{upper_gap_value}.
    Hence $\alpha\in\upperGapValue{C}{T}{\kappa}{\alpha}$.
    Show that every element of $U$ is an element of
        $\gapCell{C}{T}{\lowerGapValue{C}{T}{\kappa}{\alpha}}
            {\upperGapValue{C}{T}{\kappa}{\alpha}}$.
    \begin{subproof}
        Fix $\gamma\in U$.
        Hence $\gamma\in B$ by \obv{subseteq}.
        Hence $\gamma\in T\setminus\suc{\alpha}$ by \obv{subseteq}.
        Hence $\gamma\in T$ by \obv{setminus_elim_left}.
        Hence $\gamma\in\kappa$ by \obv{subseteq}.
        Hence $\gamma$ is an ordinal by \obv{elem_of_ordinal_is_ordinal}.
        $\alpha\in\gamma$ and
            $g(\gamma)=\leastMemberAbove{C(\gamma)}
                {\lowerGapValue{C}{T}{\kappa}{\alpha}}{\gamma}$
            by \auto{upper_gap_map_value_properties}.
        Hence $\lowerGapValue{C}{T}{\kappa}{\alpha}\in\gamma$
            by \obv{ordinal_transitivity}.
        $g(\gamma)=\fodorValue{g}{\kappa}$.
        Hence $\leastMemberAbove{C(\gamma)}
            {\lowerGapValue{C}{T}{\kappa}{\alpha}}{\gamma}
                =\upperGapValue{C}{T}{\kappa}{\alpha}$.
        Follows by \obv{gap_cell}.
    \end{subproof}
    Hence $U\subseteq\gapCell{C}{T}
        {\lowerGapValue{C}{T}{\kappa}{\alpha}}
        {\upperGapValue{C}{T}{\kappa}{\alpha}}$ by \obv{subseteq}.
    Follows by \obv{upper_gap_fibre,upper_gap_value}.
\end{proof}

\begin{theorem}\label{lower_endpoint_map_type_and_regressive}
    Let $\kappa$ be a regular cardinal.
    Suppose $\naturals\in\kappa$.
    Suppose $T$ is stationary in $\kappa$.
    Suppose $\avoidingClubSequenceOn{C}{T}{\kappa}$.
    Then $\lowerEndpointMap{C}{T}{\kappa}\in\funs{T}{\kappa}$ and
        $\lowerEndpointMap{C}{T}{\kappa}$ is regressive on $T$.
\end{theorem}
\begin{proof}
    Let $h=\lowerEndpointMap{C}{T}{\kappa}$.
    $h$ is a relation by \auto{relation,lower_endpoint_map}.
    $h$ is right-unique
        by \auto{rightunique,lower_endpoint_map,pair_eq_iff}.
    $\dom{h}=T$
        by \auto{setext,dom_iff,lower_endpoint_map,pair_eq_iff}.
    Show that for every $\alpha\in\dom{h}$ we have $h(\alpha)\in\kappa$.
    \begin{subproof}
        Fix $\alpha\in\dom{h}$.
        Hence $\alpha\in T$.
        $\lowerGapValue{C}{T}{\kappa}{\alpha}\in\alpha$
            by \auto{lower_gap_value_and_fibre_properties}.
        $T\subseteq\kappa$ by \obv{stationary_in_elim}.
        Hence $\alpha\in\kappa$ by \obv{subseteq}.
        $\kappa$ is an ordinal by \obv{avoiding_club_sequence_on_type}.
        Hence $\lowerGapValue{C}{T}{\kappa}{\alpha}\in\kappa$
            by \obv{ordinal_transitivity}.
        $(\alpha,\lowerGapValue{C}{T}{\kappa}{\alpha})\in h$
            by \obv{lower_endpoint_map}.
        Hence $h(\alpha)=\lowerGapValue{C}{T}{\kappa}{\alpha}$
            by \obv{function_apply_intro}.
    \end{subproof}
    Hence $h\in\funs{T}{\kappa}$ by \obv{funs_intro}.
    Show that for every $\alpha\in T\setminus\{\emptyset\}$ we have
        $h(\alpha)\in\alpha$.
    \begin{subproof}
        Fix $\alpha\in T\setminus\{\emptyset\}$.
        Hence $\alpha\in T$ by \obv{setminus_elim_left}.
        $\lowerGapValue{C}{T}{\kappa}{\alpha}\in\alpha$
            by \auto{lower_gap_value_and_fibre_properties}.
        $(\alpha,\lowerGapValue{C}{T}{\kappa}{\alpha})\in h$
            by \obv{lower_endpoint_map}.
        Hence $h(\alpha)=\lowerGapValue{C}{T}{\kappa}{\alpha}$
            by \obv{function_apply_intro}.
    \end{subproof}
    Hence $h$ is regressive on $T$ by \obv{regressive_on}.
\end{proof}

\begin{theorem}\label{avoiding_club_sequence_stationary_subpartition}
    Let $\kappa$ be a regular cardinal.
    Suppose $\naturals\in\kappa$.
    Suppose $T$ is stationary in $\kappa$.
    Suppose $\avoidingClubSequenceOn{C}{T}{\kappa}$.
    Then there exists $P$ such that
        $P$ is a stationary subpartition of $T$ in $\kappa$.
\end{theorem}
\begin{proof}
    Let $h=\lowerEndpointMap{C}{T}{\kappa}$.
    $h\in\funs{T}{\kappa}$ and $h$ is regressive on $T$
        by \auto{lower_endpoint_map_type_and_regressive}.
    Let $\xi=\fodorValue{h}{\kappa}$.
    Let $A=\fodorFibre{h}{\kappa}$.
    $\xi\in\kappa$ and $A$ is stationary in $\kappa$ and
        $A\subseteq T$ and
        for every $\alpha\in A$ we have $h(\alpha)=\xi$
        by \auto{fodor_value_and_fibre_properties}.
    $\kappa$ is an ordinal and $T\subseteq\kappa$
        by \obv{avoiding_club_sequence_on_type}.
    Show that for every $\alpha\in A$ we have
        $\lowerGapValue{C}{T}{\kappa}{\alpha}=\xi$.
    \begin{subproof}
        Fix $\alpha\in A$.
        Hence $\alpha\in T$ by \obv{subseteq}.
        $(\alpha,\lowerGapValue{C}{T}{\kappa}{\alpha})\in h$
            by \obv{lower_endpoint_map}.
        Hence $h(\alpha)=\lowerGapValue{C}{T}{\kappa}{\alpha}$
            by \obv{function_apply_intro}.
    \end{subproof}
    Let $H=\{\eta\in\kappa\mid
        \text{$\gapCell{C}{T}{\xi}{\eta}$ is stationary in $\kappa$}\}$.
    Show that $\cofinalIn{H}{\kappa}$.
    \begin{subproof}
        $H\subseteq\kappa$ by \obv{subseteq}.
        Show that for every $\delta\in\kappa$ there exists $\eta\in H$
            such that $\delta\subseteq\eta$.
        \begin{subproof}
            Fix $\delta\in\kappa$.
            $\ordinalUnbounded{A}{\kappa}$
                by \obv{stationary_is_unbounded}.
            Take $\alpha\in A$ such that $\delta\in\alpha$
                by \obv{ordinal_unbounded}.
            Hence $\alpha\in T$ by \obv{subseteq}.
            Let $\eta=\upperGapValue{C}{T}{\kappa}{\alpha}$.
            Let $U=\upperGapFibre{C}{T}{\kappa}{\alpha}$.
            $\eta\in\kappa$ and $U$ is stationary in $\kappa$ and
                $\alpha\in\eta$ and
                $U\subseteq\gapCell{C}{T}
                    {\lowerGapValue{C}{T}{\kappa}{\alpha}}{\eta}$
                by \auto{upper_gap_value_and_fibre_properties}.
            $\lowerGapValue{C}{T}{\kappa}{\alpha}=\xi$.
            Hence $U\subseteq\gapCell{C}{T}{\xi}{\eta}$.
            $\gapCell{C}{T}{\xi}{\eta}\subseteq T$
                by \obv{gap_cell_subset}.
            Hence $\gapCell{C}{T}{\xi}{\eta}\subseteq\kappa$
                by \obv{subseteq_transitive}.
            Hence $\gapCell{C}{T}{\xi}{\eta}$ is stationary in $\kappa$
                by \obv{stationary_superset}.
            Hence $\eta\in H$.
            $\eta$ is an ordinal by \obv{elem_of_ordinal_is_ordinal}.
            $\delta\in\eta$ by \obv{ordinal_transitivity}.
            Hence $\delta\subseteq\eta$
                by \obv{ordinal_elem_implies_subseteq,
                    elem_of_ordinal_is_ordinal}.
        \end{subproof}
        Follows by \obv{cofinal_in}.
    \end{subproof}
    Let $Q=\{(\eta,\gapCell{C}{T}{\xi}{\eta})\mid\eta\in H\}$.
    $Q$ is a relation by \auto{relation,pair_eq_iff}.
    $Q$ is right-unique by \auto{rightunique,pair_eq_iff}.
    Hence $Q$ is a function.
    $\dom{Q}=H$ by \auto{setext,dom_iff,pair_eq_iff}.
    Show that for every $\eta\in\dom{Q}$ we have $Q(\eta)\in\pow{T}$.
    \begin{subproof}
        Fix $\eta\in\dom{Q}$.
        Hence $\eta\in H$.
        $\gapCell{C}{T}{\xi}{\eta}\subseteq T$
            by \obv{gap_cell_subset}.
        Hence $\gapCell{C}{T}{\xi}{\eta}\in\pow{T}$
            by \obv{powerset_intro}.
        $(\eta,\gapCell{C}{T}{\xi}{\eta})\in Q$.
        Hence $Q(\eta)=\gapCell{C}{T}{\xi}{\eta}$
            by \obv{function_apply_intro}.
    \end{subproof}
    Hence $Q\in\funs{H}{\pow{T}}$ by \obv{funs_intro}.
    Show that $Q$ is pairwise disjoint on $H$.
    \begin{subproof}
        Show that for all $\eta,\zeta\in H$ such that $\eta\neq\zeta$
            we have $Q(\eta)$ is disjoint from $Q(\zeta)$.
        \begin{subproof}
            Fix $\eta,\zeta\in H$.
            Assume $\eta\neq\zeta$.
            $(\eta,\gapCell{C}{T}{\xi}{\eta})\in Q$ and
                $(\zeta,\gapCell{C}{T}{\xi}{\zeta})\in Q$.
            Hence $Q(\eta)=\gapCell{C}{T}{\xi}{\eta}$ and
                $Q(\zeta)=\gapCell{C}{T}{\xi}{\zeta}$
                by \obv{function_apply_intro}.
            Follows by \auto{gap_cells_pairwise_disjoint}.
        \end{subproof}
        Follows by \obv{pairwise_disjoint_on}.
    \end{subproof}
    Show that for every $\eta\in H$ we have
        $Q(\eta)$ is stationary in $\kappa$.
    \begin{subproof}
        Fix $\eta\in H$.
        $(\eta,\gapCell{C}{T}{\xi}{\eta})\in Q$.
        Hence $Q(\eta)=\gapCell{C}{T}{\xi}{\eta}$
            by \obv{function_apply_intro}.
        $\gapCell{C}{T}{\xi}{\eta}$ is stationary in $\kappa$.
    \end{subproof}
    Follows by \obv{cofinal_stationary_cells_to_subpartition}.
\end{proof}

\begin{theorem}\label{self_nonreflecting_stationary_subpartition}
    Let $\kappa$ be a regular cardinal.
    Suppose $\naturals\in\kappa$.
    Suppose $T$ is stationary in $\kappa$.
    Suppose $\selfNonreflectingIn{T}{\kappa}$.
    Then there exists $P$ such that
        $P$ is a stationary subpartition of $T$ in $\kappa$.
\end{theorem}
\begin{proof}
    Hence $\kappa$ is a limit ordinal
        by \obv{infinite_cardinal_is_limit_ordinal}.
    $\avoidingClubSequenceOn{\chosenAvoidingClubSequence{T}}{T}{\kappa}$
        by \obv{chosen_avoiding_club_sequence_on}.
    Follows by \obv{avoiding_club_sequence_stationary_subpartition}.
\end{proof}
