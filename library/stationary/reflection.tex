\section{Club limit points and stationary reflection}

\import{stationary/club.tex}

For a club in an ambient ordinal, its limit-point set retains both the
ambient bound and the local unboundedness needed to restrict that club.  This
keeps the local and global stationarity ambients explicit.

\begin{definition}\label{club_limit_points}
    $\clubLimitPoints{C}{\kappa}
        =\{\beta\in\kappa\mid
            \text{$\beta$ is a limit ordinal and
                $\ordinalUnbounded{C\inter\beta}{\beta}$}\}$.
\end{definition}

\begin{proposition}\label{club_limit_points_subset}
    $\clubLimitPoints{C}{\kappa}\subseteq\kappa$.
\end{proposition}
\begin{proof}
    Follows by \obv{club_limit_points,subseteq}.
\end{proof}

\begin{proposition}\label{club_limit_point_is_limit}
    Let $\beta\in\clubLimitPoints{C}{\kappa}$.
    Then $\beta$ is a limit ordinal.
\end{proposition}
\begin{proof}
    Follows by \obv{club_limit_points}.
\end{proof}

\begin{proposition}\label{club_limit_point_unbounded_restriction}
    Let $\beta\in\clubLimitPoints{C}{\kappa}$.
    Then $\ordinalUnbounded{C\inter\beta}{\beta}$.
\end{proposition}
\begin{proof}
    Follows by \obv{club_limit_points}.
\end{proof}

\begin{proposition}\label{club_limit_point_of_closed_set_mem}
    Suppose $\ordinalClosed{C}{\kappa}$.
    Let $\beta\in\clubLimitPoints{C}{\kappa}$.
    Then $\beta\in C$.
\end{proposition}
\begin{proof}
    $\beta\in\kappa$ and $\beta$ is a limit ordinal and
        $\ordinalUnbounded{C\inter\beta}{\beta}$
        by \obv{club_limit_points}.
    Follows by \obv{ordinal_closed}.
\end{proof}

\begin{theorem}\label{club_limit_points_is_club}
    Let $\kappa$ be a limit ordinal.
    Suppose $\naturals\in\cofinality{\kappa}$.
    Suppose $\clubIn{C}{\kappa}$.
    Then $\clubIn{\clubLimitPoints{C}{\kappa}}{\kappa}$.
\end{theorem}
\begin{proof}
    Let $L=\clubLimitPoints{C}{\kappa}$.
    $L\subseteq\kappa$ by \auto{club_limit_points_subset}.
    $\clubIn{\kappa}{\kappa}$
        by \obv{limit_ordinal_is_club_in_itself}.
    Show that $\ordinalUnbounded{L}{\kappa}$.
    \begin{subproof}
        Show that for every $\alpha\in\kappa$ there exists $\beta\in L$
            such that $\alpha\in\beta$.
        \begin{subproof}
            Fix $\alpha\in\kappa$.
            Take $\beta\in\kappa$ such that $\alpha\in\beta$ and
                $\beta$ is a limit ordinal and
                $\ordinalUnbounded{C\inter\beta}{\beta}$ and
                $\ordinalUnbounded{\kappa\inter\beta}{\beta}$
                by \obv{two_clubs_have_common_limit_point_above}.
            Hence $\beta\in L$ by \obv{club_limit_points}.
        \end{subproof}
        Follows by \obv{ordinal_unbounded}.
    \end{subproof}
    Show that $\ordinalClosed{L}{\kappa}$.
    \begin{subproof}
        $\ordinalClosed{C}{\kappa}$ by \obv{club_in_elim}.
        Show that for every limit ordinal $\gamma\in\kappa$ such that
            $\ordinalUnbounded{L\inter\gamma}{\gamma}$
            we have $\gamma\in L$.
        \begin{subproof}
            Fix $\gamma$.
            Assume $\gamma\in\kappa$.
            Assume $\gamma$ is a limit ordinal.
            Assume $\ordinalUnbounded{L\inter\gamma}{\gamma}$.
            Show that $L\inter\gamma\subseteq C\inter\gamma$.
            \begin{subproof}
                It suffices to show that every element of $L\inter\gamma$
                    is an element of $C\inter\gamma$ by \obv{subseteq}.
                Fix $\delta\in L\inter\gamma$.
                Hence $\delta\in L$ and $\delta\in\gamma$ by \obv{inter}.
                Hence $\delta\in C$
                    by \obv{club_limit_point_of_closed_set_mem}.
                Follows by \obv{inter_intro}.
            \end{subproof}
            $C\inter\gamma\subseteq\gamma$ by \obv{subseteq,inter}.
            Hence $\ordinalUnbounded{C\inter\gamma}{\gamma}$
                by \obv{ordinal_unbounded_superset}.
            Follows by \obv{club_limit_points}.
        \end{subproof}
        Follows by \obv{ordinal_closed}.
    \end{subproof}
    Follows by \obv{club_in}.
\end{proof}

\begin{theorem}\label{club_restrict_at_limit_point}
    Suppose $\clubIn{C}{\kappa}$.
    Let $\beta\in\clubLimitPoints{C}{\kappa}$.
    Then $\clubIn{C\inter\beta}{\beta}$.
\end{theorem}
\begin{proof}
    $\kappa$ is a limit ordinal and $C\subseteq\kappa$ and
        $\ordinalClosed{C}{\kappa}$ by \obv{club_in_elim}.
    $\beta\in\kappa$ and $\beta$ is a limit ordinal and
        $\ordinalUnbounded{C\inter\beta}{\beta}$
        by \obv{club_limit_points}.
    Show that $\ordinalClosed{C\inter\beta}{\beta}$.
    \begin{subproof}
        $C\inter\beta\subseteq\beta$ by \obv{subseteq,inter}.
        Show that for every limit ordinal $\gamma\in\beta$ such that
            $\ordinalUnbounded{(C\inter\beta)\inter\gamma}{\gamma}$
            we have $\gamma\in C\inter\beta$.
        \begin{subproof}
            Fix $\gamma$.
            Assume $\gamma\in\beta$.
            Assume $\gamma$ is a limit ordinal.
            Assume $\ordinalUnbounded{(C\inter\beta)\inter\gamma}{\gamma}$.
            $(C\inter\beta)\inter\gamma\subseteq C\inter\gamma$
                by \auto{subseteq,inter}.
            $C\inter\gamma\subseteq\gamma$ by \obv{subseteq,inter}.
            Hence $\ordinalUnbounded{C\inter\gamma}{\gamma}$
                by \obv{ordinal_unbounded_superset}.
            Hence $\gamma\in\kappa$ by \obv{ordinal_transitivity}.
            Hence $\gamma\in C$ by \obv{ordinal_closed}.
            Follows by \obv{inter_intro}.
        \end{subproof}
        Follows by \obv{ordinal_closed}.
    \end{subproof}
    Follows by \obv{club_in}.
\end{proof}

\begin{theorem}\label{stationary_from_stationarily_many_local_stationary}
    Let $\kappa$ be a regular cardinal.
    Suppose $\naturals\in\kappa$ and $A\subseteq\kappa$.
    Suppose $R$ is stationary in $\kappa$.
    Suppose for every $\beta\in R$ we have
        $A\inter\beta$ is stationary in $\beta$.
    Then $A$ is stationary in $\kappa$.
\end{theorem}
\begin{proof}
    $\kappa$ is an infinite cardinal and $\cofinality{\kappa}=\kappa$
        by \obv{regular_cardinal}.
    Hence $\naturals\in\cofinality{\kappa}$.
    Show that for every $D$ such that $\clubIn{D}{\kappa}$
        we have $A\meets D$.
    \begin{subproof}
        Fix $D$.
        Assume $\clubIn{D}{\kappa}$.
        $\clubIn{\clubLimitPoints{D}{\kappa}}{\kappa}$
            by \obv{club_limit_points_is_club}.
        Hence $R\inter\clubLimitPoints{D}{\kappa}$ is stationary in $\kappa$
            by \obv{stationary_intersection_club}.
        $\clubIn{\kappa}{\kappa}$
            by \obv{limit_ordinal_is_club_in_itself}.
        Hence $(R\inter\clubLimitPoints{D}{\kappa})\meets\kappa$
            by \obv{stationary_in}.
        Take $\beta$ such that
            $\beta\in R\inter\clubLimitPoints{D}{\kappa}$ and
            $\beta\in\kappa$ by \auto{disjoint}.
        Hence $\beta\in R$ and
            $\beta\in\clubLimitPoints{D}{\kappa}$ by \obv{inter}.
        $\clubIn{D\inter\beta}{\beta}$
            by \obv{club_restrict_at_limit_point}.
        $A\inter\beta$ is stationary in $\beta$.
        Hence $(A\inter\beta)\meets(D\inter\beta)$
            by \obv{stationary_in}.
        Take $x$ such that $x\in A\inter\beta$ and
            $x\in D\inter\beta$ by \auto{disjoint}.
        Hence $x\in A$ and $x\in D$
            by \obv{inter_elim_left}.
        Follows by \auto{disjoint}.
    \end{subproof}
    Follows by \obv{stationary_in}.
\end{proof}
