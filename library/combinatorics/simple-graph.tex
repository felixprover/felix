\section{Simple graphs}

\import{relation/properties.tex}
\import{relation/trace.tex}

A simple graph is presented by its symmetric, irreflexive edge relation.  The
carrier is kept explicit, so the same relation can be restricted without
changing representations.

\begin{definition}\label{simple_graph}
    $E$ is a simple graph on $X$ iff
        $E$ is a binary relation on $X$ and
        $E$ is symmetric and
        $E$ is irreflexive.
\end{definition}

\begin{proposition}\label{simple_graph_relation}
    Suppose $E$ is a simple graph on $X$.
    Then $E$ is a binary relation on $X$.
\end{proposition}
\begin{proof}
    Follows by definition.
\end{proof}

\begin{proposition}\label{simple_graph_symmetric}
    Suppose $E$ is a simple graph on $X$.
    Then $E$ is symmetric.
\end{proposition}
\begin{proof}
    Follows by definition.
\end{proof}

\begin{proposition}\label{simple_graph_irreflexive}
    Suppose $E$ is a simple graph on $X$.
    Then $E$ is irreflexive.
\end{proposition}
\begin{proof}
    Follows by definition.
\end{proof}

\subsection{Neighbourhoods}

The neighbourhood is relative to a set of possible neighbours; that set need
not be the carrier of a graph.

\begin{definition}\label{graph_neighbourhood}
    $\graphNeighbourhood{E}{x}{Y}=Y\inter\upward{E}{x}$.
\end{definition}

\begin{proposition}\label{graph_neighbourhood_iff}
    $y\in\graphNeighbourhood{E}{x}{Y}$ iff
        $y\in Y$ and $x\mathrel{E}y$.
\end{proposition}
\begin{proof}
    Follows by \auto{graph_neighbourhood,inter,upward_closure_iff}.
\end{proof}

\begin{proposition}\label{graph_neighbourhood_subseteq}
    $\graphNeighbourhood{E}{x}{Y}\subseteq Y$.
\end{proposition}
\begin{proof}
    Follows by \auto{graph_neighbourhood,inter_lower_left}.
\end{proof}

\begin{proposition}\label{graph_neighbourhood_edge}
    Suppose $y\in\graphNeighbourhood{E}{x}{Y}$.
    Then $x\mathrel{E}y$.
\end{proposition}
\begin{proof}
    Follows by \auto{graph_neighbourhood_iff}.
\end{proof}

\begin{proposition}\label{graph_neighbourhood_monotone}
    Suppose $Y\subseteq Z$.
    Then $\graphNeighbourhood{E}{x}{Y}\subseteq
        \graphNeighbourhood{E}{x}{Z}$.
\end{proposition}
\begin{proof}
    Follows by \auto{subseteq,graph_neighbourhood_iff}.
\end{proof}

\begin{proposition}\label{graph_neighbourhood_no_loop}
    Suppose $E$ is irreflexive.
    Then $x\notin\graphNeighbourhood{E}{x}{Y}$.
\end{proposition}
\begin{proof}
    Follows by \auto{graph_neighbourhood_iff,irreflexive}.
\end{proof}

\begin{proposition}\label{graph_neighbourhood_distinct}
    Suppose $E$ is irreflexive and
        $y\in\graphNeighbourhood{E}{x}{Y}$.
    Then $x\neq y$.
\end{proposition}
\begin{proof}
    Follows by \auto{graph_neighbourhood_iff,irreflexive}.
\end{proof}

\begin{proposition}\label{simple_graph_neighbourhood_symmetric}
    Suppose $E$ is a simple graph on $X$ and $x,y\in X$.
    Then $y\in\graphNeighbourhood{E}{x}{X}$ iff
        $x\in\graphNeighbourhood{E}{y}{X}$.
\end{proposition}
\begin{proof}
    $E$ is symmetric by \obv{simple_graph_symmetric}.
    Follows by \auto{graph_neighbourhood_iff,symmetric}.
\end{proof}

\begin{proposition}\label{simple_graph_edge_symmetric}
    Suppose $E$ is a simple graph on $X$ and $x\mathrel{E}y$.
    Then $y\mathrel{E}x$.
\end{proposition}
\begin{proof}
    $E$ is symmetric by \obv{simple_graph_symmetric}.
    Follows by \obv{symmetric}.
\end{proof}

\subsection{Cliques and independent sets}

\begin{definition}\label{clique_in_graph}
    $H$ is a clique in $X$ with respect to $E$ iff
        $H\subseteq X$ and for all $x,y\in H$ such that $x\neq y$
            we have $x\mathrel{E}y$.
\end{definition}

\begin{definition}\label{independent_in_graph}
    $H$ is an independent set in $X$ with respect to $E$ iff
        $H\subseteq X$ and for all $x,y\in H$ such that $x\neq y$
            we have $x\not\mathrel{E}y$.
\end{definition}

\begin{proposition}\label{clique_subseteq}
    Suppose $H$ is a clique in $X$ with respect to $E$.
    Then $H\subseteq X$.
\end{proposition}
\begin{proof}
    Follows by definition.
\end{proof}

\begin{proposition}\label{clique_edge}
    Suppose $H$ is a clique in $X$ with respect to $E$.
    Suppose $x,y\in H$ and $x\neq y$.
    Then $x\mathrel{E}y$.
\end{proposition}
\begin{proof}
    Follows by definition.
\end{proof}

\begin{proposition}\label{independent_subseteq}
    Suppose $H$ is an independent set in $X$ with respect to $E$.
    Then $H\subseteq X$.
\end{proposition}
\begin{proof}
    Follows by definition.
\end{proof}

\begin{proposition}\label{independent_nonedge}
    Suppose $H$ is an independent set in $X$ with respect to $E$.
    Suppose $x,y\in H$ and $x\neq y$.
    Then $x\not\mathrel{E}y$.
\end{proposition}
\begin{proof}
    Follows by definition.
\end{proof}

\begin{proposition}\label{subset_of_clique_is_clique}
    Suppose $K\subseteq H$.
    Suppose $H$ is a clique in $X$ with respect to $E$.
    Then $K$ is a clique in $X$ with respect to $E$.
\end{proposition}
\begin{proof}
    Follows by \auto{clique_in_graph,subseteq,subseteq_transitive}.
\end{proof}

\begin{proposition}\label{subset_of_independent_is_independent}
    Suppose $K\subseteq H$.
    Suppose $H$ is an independent set in $X$ with respect to $E$.
    Then $K$ is an independent set in $X$ with respect to $E$.
\end{proposition}
\begin{proof}
    Follows by \auto{independent_in_graph,subseteq,subseteq_transitive}.
\end{proof}

\subsection{Induced graphs}

\begin{proposition}\label{induced_simple_graph}
    Suppose $E$ is a simple graph on $X$ and $Y\subseteq X$.
    Then $\relationTrace{E}{Y}$ is a simple graph on $Y$.
\end{proposition}
\begin{proof}
    $\relationTrace{E}{Y}$ is a binary relation on $Y$ by
        \obv{relation_trace_binary_on}.
    $\relationTrace{E}{Y}$ is symmetric by
        \obv{simple_graph_symmetric,relation_trace_symmetric}.
    $\relationTrace{E}{Y}$ is irreflexive by
        \obv{simple_graph_irreflexive,relation_trace_irreflexive}.
    Follows by definition.
\end{proof}

\begin{proposition}\label{induced_graph_neighbourhood}
    Suppose $x\in Y$.
    Then $\graphNeighbourhood{\relationTrace{E}{Y}}{x}{Y}=
        \graphNeighbourhood{E}{x}{Y}$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{setext,graph_neighbourhood_iff,relation_trace_iff}.
\end{proof}

\begin{proposition}\label{induced_clique_iff}
    Suppose $H\subseteq Y$ and $Y\subseteq X$.
    Then $H$ is a clique in $Y$ with respect to $\relationTrace{E}{Y}$ iff
        $H$ is a clique in $X$ with respect to $E$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{clique_in_graph,relation_trace_iff,subseteq,
            subseteq_transitive}.
\end{proof}

\begin{proposition}\label{induced_independent_iff}
    Suppose $H\subseteq Y$ and $Y\subseteq X$.
    Then $H$ is an independent set in $Y$ with respect to
            $\relationTrace{E}{Y}$ iff
        $H$ is an independent set in $X$ with respect to $E$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{independent_in_graph,relation_trace_iff,subseteq,
            subseteq_transitive}.
\end{proof}

\subsection{Complements}

The complement is relative to the displayed carrier and omits the diagonal.
This makes it a graph operation even when the input relation is arbitrary.

\begin{definition}\label{graph_complement}
    $\graphComplement{E}{X}=
        ((X\times X)\setminus\identity{X})\setminus E$.
\end{definition}

\begin{proposition}\label{graph_complement_iff}
    $x\mathrel{\graphComplement{E}{X}}y$ iff
        $x\in X$ and $y\in X$ and $x\neq y$ and $x\not\mathrel{E}y$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{graph_complement,setminus,times_tuple_elim,
            times_tuple_intro,id_iff}.
\end{proof}

\begin{proposition}\label{graph_complement_relation}
    $\graphComplement{E}{X}$ is a binary relation on $X$.
\end{proposition}
\begin{proof}
    $\graphComplement{E}{X}\subseteq
        (X\times X)\setminus\identity{X}$ by
        \auto{graph_complement,setminus_subseteq}.
    $(X\times X)\setminus\identity{X}\subseteq X\times X$ by
        \obv{setminus_subseteq}.
    Hence $\graphComplement{E}{X}\subseteq X\times X$ by
        \obv{subseteq_transitive}.
    Follows by definition.
\end{proof}

\begin{proposition}\label{simple_graph_off_diagonal}
    Suppose $E$ is a simple graph on $X$.
    Then $E\subseteq (X\times X)\setminus\identity{X}$.
\end{proposition}
\begin{proof}
    $E$ is a binary relation on $X$ by \obv{simple_graph_relation}.
    $E$ is irreflexive by \obv{simple_graph_irreflexive}.
    $E\subseteq X\times X$ by definition.
    $E$ is a relation by \obv{binary_relation_is_relation}.
    Show that every element of $E$ is an element of
        $(X\times X)\setminus\identity{X}$.
    \begin{subproof}
        Fix $w\in E$.
        Take $x,y$ such that $w=(x,y)$ by \obv{relation}.
        $(x,y)\in E$.
        $(x,y)\in X\times X$ by \obv{subseteq}.
        $x\neq y$ by \auto{irreflexive}.
        $(x,y)\notin\identity{X}$ by \auto{id_iff}.
        $(x,y)\in (X\times X)\setminus\identity{X}$ by \obv{setminus}.
    \end{subproof}
    Follows by \obv{subseteq}.
\end{proof}

\begin{proposition}\label{graph_complement_simple}
    Suppose $E$ is a simple graph on $X$.
    Then $\graphComplement{E}{X}$ is a simple graph on $X$.
\end{proposition}
\begin{proof}
    $\graphComplement{E}{X}$ is a binary relation on $X$ by
        \obv{graph_complement_relation}.
    $\graphComplement{E}{X}$ is symmetric by
        \auto{graph_complement_iff,simple_graph_symmetric,symmetric}.
    $\graphComplement{E}{X}$ is irreflexive by
        \auto{graph_complement_iff,irreflexive}.
    Follows by definition.
\end{proof}

\begin{proposition}\label{graph_complement_involutive}
    Suppose $E\subseteq (X\times X)\setminus\identity{X}$.
    Then $\graphComplement{\graphComplement{E}{X}}{X}=E$.
\end{proposition}
\begin{proof}
    $(X\times X)\setminus\identity{X}\subseteq X\times X$ by
        \obv{setminus_subseteq}.
    Hence $E\subseteq X\times X$ by \obv{subseteq_transitive}.
    Hence $E$ is a binary relation on $X$ by definition.
    Hence $E$ is a relation by \obv{binary_relation_is_relation}.
    $\graphComplement{\graphComplement{E}{X}}{X}$ is a binary relation
        on $X$ by \obv{graph_complement_relation}.
    $\graphComplement{\graphComplement{E}{X}}{X}$ is a relation by
        \obv{binary_relation_is_relation}.
    Show that for all $x,y$ we have
        $x\mathrel{\graphComplement{\graphComplement{E}{X}}{X}}y$ iff
        $x\mathrel{E}y$.
    \begin{subproof}
        Fix $x,y$.
        Show that if
                $x\mathrel{\graphComplement{\graphComplement{E}{X}}{X}}y$,
            then $x\mathrel{E}y$.
        \begin{subproof}
            Assume
                $x\mathrel{\graphComplement{\graphComplement{E}{X}}{X}}y$.
            Then $x,y\in X$ and $x\neq y$ and
                $x\not\mathrel{\graphComplement{E}{X}}y$ by
                \auto{graph_complement_iff}.
            Follows by \auto{graph_complement_iff}.
        \end{subproof}
        Show that if $x\mathrel{E}y$, then
            $x\mathrel{\graphComplement{\graphComplement{E}{X}}{X}}y$.
        \begin{subproof}
            Assume $x\mathrel{E}y$.
            $(x,y)\in (X\times X)\setminus\identity{X}$ by
                \obv{subseteq}.
            Hence $x,y\in X$ and $x\neq y$ by
                \auto{setminus,times_tuple_elim,id_iff}.
            $x\not\mathrel{\graphComplement{E}{X}}y$ by
                \auto{graph_complement_iff}.
            Follows by \obv{graph_complement_iff}.
        \end{subproof}
        Follows by \auto.
    \end{subproof}
    Follows by \obv{relext}.
\end{proof}

\begin{corollary}\label{simple_graph_complement_involutive}
    Suppose $E$ is a simple graph on $X$.
    Then $\graphComplement{\graphComplement{E}{X}}{X}=E$.
\end{corollary}
\begin{proof}
    $E\subseteq (X\times X)\setminus\identity{X}$ by
        \obv{simple_graph_off_diagonal}.
    Follows by \obv{graph_complement_involutive}.
\end{proof}

\begin{proposition}\label{graph_complement_clique_iff}
    $H$ is a clique in $X$ with respect to $\graphComplement{E}{X}$ iff
        $H$ is an independent set in $X$ with respect to $E$.
\end{proposition}
\begin{proof}
    Show that if $H$ is a clique in $X$ with respect to
            $\graphComplement{E}{X}$, then
        $H$ is an independent set in $X$ with respect to $E$.
    \begin{subproof}
        Assume $H$ is a clique in $X$ with respect to
            $\graphComplement{E}{X}$.
        $H\subseteq X$ by \obv{clique_subseteq}.
        For all $x,y\in H$ such that $x\neq y$ we have
            $x\not\mathrel{E}y$ by
            \auto{clique_edge,graph_complement_iff}.
        Follows by \obv{independent_in_graph}.
    \end{subproof}
    Show that if $H$ is an independent set in $X$ with respect to $E$, then
        $H$ is a clique in $X$ with respect to
            $\graphComplement{E}{X}$.
    \begin{subproof}
        Assume $H$ is an independent set in $X$ with respect to $E$.
        $H\subseteq X$ by \obv{independent_subseteq}.
        For all $x,y\in H$ such that $x\neq y$ we have
            $x\mathrel{\graphComplement{E}{X}}y$ by
            \auto{independent_nonedge,graph_complement_iff,subseteq}.
        Follows by \obv{clique_in_graph}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{proposition}\label{graph_complement_independent_iff}
    $H$ is an independent set in $X$ with respect to
            $\graphComplement{E}{X}$ iff
        $H$ is a clique in $X$ with respect to $E$.
\end{proposition}
\begin{proof}
    Show that if $H$ is an independent set in $X$ with respect to
            $\graphComplement{E}{X}$, then
        $H$ is a clique in $X$ with respect to $E$.
    \begin{subproof}
        Assume $H$ is an independent set in $X$ with respect to
            $\graphComplement{E}{X}$.
        $H\subseteq X$ by \obv{independent_subseteq}.
        For all $x,y\in H$ such that $x\neq y$ we have
            $x\mathrel{E}y$ by
            \auto{independent_nonedge,graph_complement_iff,subseteq}.
        Follows by \obv{clique_in_graph}.
    \end{subproof}
    Show that if $H$ is a clique in $X$ with respect to $E$, then
        $H$ is an independent set in $X$ with respect to
            $\graphComplement{E}{X}$.
    \begin{subproof}
        Assume $H$ is a clique in $X$ with respect to $E$.
        $H\subseteq X$ by \obv{clique_subseteq}.
        For all $x,y\in H$ such that $x\neq y$ we have
            $x\not\mathrel{\graphComplement{E}{X}}y$ by
            \auto{clique_edge,graph_complement_iff}.
        Follows by \obv{independent_in_graph}.
    \end{subproof}
    Follows by \auto.
\end{proof}
