\section{Subsets of a fixed cardinality}

\import{set/finite.tex}
\import{set/powerset.tex}
\import{set/equinumerosity.tex}
\import{cardinal/exponentiation.tex}
\import{cardinal/reconciliation.tex}

For arbitrary sets $X$ and $\kappa$, the fixed-cardinality subset family
consists of the subsets of $X$ equinumerous with $\kappa$.

\begin{definition}\label{fixed_cardinality_subsets}
    $\fixedCardinalitySubsets{X}{\kappa}
        =\{A\in\pow{X}\mid A\approx\kappa\}$.
\end{definition}

\begin{proposition}\label{fixed_cardinality_subsets_iff}
    $A\in\fixedCardinalitySubsets{X}{\kappa}$ iff
        $A\subseteq X$ and $A\approx\kappa$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{fixed_cardinality_subsets,pow_iff,subseteq}.
\end{proof}

\begin{proposition}\label{fixed_cardinality_subsets_monotone}
    Suppose $Y\subseteq X$.
    Then $\fixedCardinalitySubsets{Y}{\kappa}
        \subseteq\fixedCardinalitySubsets{X}{\kappa}$.
\end{proposition}
\begin{proof}
    Show that every element of $\fixedCardinalitySubsets{Y}{\kappa}$ is an
        element of $\fixedCardinalitySubsets{X}{\kappa}$.
    \begin{subproof}
        Fix $A\in\fixedCardinalitySubsets{Y}{\kappa}$.
        Hence $A\subseteq Y$ and $A\approx\kappa$ by
            \auto{fixed_cardinality_subsets_iff}.
        Hence $A\subseteq X$ by \obv{subseteq_transitive}.
        Follows by \auto{fixed_cardinality_subsets_iff}.
    \end{subproof}
    Follows by \obv{subseteq}.
\end{proof}

\begin{proposition}\label{two_subset_characterization}
    $p\in\fixedCardinalitySubsets{X}{\cardTwo}$ iff
        there exist $x,y\in X$ such that $x\neq y$ and $p=\{x,y\}$.
\end{proposition}
\begin{proof}
    Show that if $p\in\fixedCardinalitySubsets{X}{\cardTwo}$, then
        there exist $x,y\in X$ such that $x\neq y$ and $p=\{x,y\}$.
    \begin{subproof}
        Assume $p\in\fixedCardinalitySubsets{X}{\cardTwo}$.
        Hence $p\subseteq X$ and $p\approx\cardTwo$ by
            \auto{fixed_cardinality_subsets_iff}.
        Take $x,y$ such that $x\neq y$ and $p=\{x,y\}$ by
            \obv{two_element_cardinality_characterization}.
        $x,y\in p$ by \auto{upair_iff}.
        Hence $x,y\in X$ by \obv{subseteq}.
        Follows by \auto.
    \end{subproof}
    Show that if there exist $x,y\in X$ such that $x\neq y$ and
            $p=\{x,y\}$, then
        $p\in\fixedCardinalitySubsets{X}{\cardTwo}$.
    \begin{subproof}
        Assume there exist $x,y\in X$ such that $x\neq y$ and
            $p=\{x,y\}$.
        Take $x,y\in X$ such that $x\neq y$ and $p=\{x,y\}$.
        Show that $p\subseteq X$.
        \begin{subproof}
            Follows by \auto{subseteq,upair_iff}.
        \end{subproof}
        $p\approx\cardTwo$ by
            \auto{two_element_cardinality_characterization}.
        Follows by \auto{fixed_cardinality_subsets_iff}.
    \end{subproof}
    Follows by \auto.
\end{proof}

\begin{proposition}\label{countably_infinite_subsets_are_infinite}
    $\fixedCardinalitySubsets{X}{\naturals}
        \subseteq\infiniteSubsets{X}$.
\end{proposition}
\begin{proof}
    Show that every element of $\fixedCardinalitySubsets{X}{\naturals}$ is
        an element of $\infiniteSubsets{X}$.
    \begin{subproof}
        Fix $A\in\fixedCardinalitySubsets{X}{\naturals}$.
        Hence $A\subseteq X$ and $A\approx\naturals$ by
            \auto{fixed_cardinality_subsets_iff}.
        Show that $A$ is infinite.
        \begin{subproof}
            Suppose not.
            Hence $A$ is finite.
            Hence $\naturals$ is finite by \obv{finite_equinumerous}.
            Contradiction by \obv{naturals_is_infinite}.
        \end{subproof}
        Follows by \obv{infinite_subsets_iff}.
    \end{subproof}
    Follows by \obv{subseteq}.
\end{proof}

\begin{proposition}\label{finite_subset_fixed_cardinality_characterization}
    $F\in\finiteSubsets{X}$ iff there exists $n\in\naturals$ such that
        $F\in\fixedCardinalitySubsets{X}{n}$.
\end{proposition}
\begin{proof}
    Show that if $F\in\finiteSubsets{X}$, then there exists
        $n\in\naturals$ such that
            $F\in\fixedCardinalitySubsets{X}{n}$.
    \begin{subproof}
        Assume $F\in\finiteSubsets{X}$.
        Hence $F\subseteq X$ and $F$ is finite by
            \obv{finite_subsets_iff}.
        Take $n\in\naturals$ such that there exists
            $f\in\bijections{n}{F}$ by \auto{finite}.
        Take $f$ such that $f\in\bijections{n}{F}$ by \auto.
        Hence $n\approx F$ by \obv{equinum}.
        Hence $F\approx n$ by \obv{equinum_sym}.
        Hence $F\in\fixedCardinalitySubsets{X}{n}$ by
            \obv{fixed_cardinality_subsets_iff}.
    \end{subproof}
    Show that if there exists $n\in\naturals$ such that
            $F\in\fixedCardinalitySubsets{X}{n}$, then
        $F\in\finiteSubsets{X}$.
    \begin{subproof}
        Assume there exists $n\in\naturals$ such that
            $F\in\fixedCardinalitySubsets{X}{n}$.
        Take $n\in\naturals$ such that
            $F\in\fixedCardinalitySubsets{X}{n}$.
        Hence $F\subseteq X$ and $F\approx n$ by
            \auto{fixed_cardinality_subsets_iff}.
        $n$ is finite by \obv{natural_number_is_finite}.
        $n\approx F$ by \obv{equinum_sym}.
        Hence $F$ is finite by \obv{finite_equinumerous}.
        Follows by \obv{finite_subsets_iff}.
    \end{subproof}
    Follows by \auto.
\end{proof}
