\section{Two-colour pair arrows}

\import{function.tex}
\import{combinatorics/fixed-cardinality-subsets.tex}
\import{combinatorics/pair-colouring.tex}
\import{order/order-type.tex}
\import{cardinal/exponentiation.tex}

The arrow predicate includes its ordinal formation conditions.  Its first
target is the colour-$\cardZero$ order type and its second target is the
colour-$\cardOne$ order type.

\begin{definition}\label{two_colour_pair_arrow}
    $\twoColourPairArrow{\kappa}{\alpha}{\beta}$ iff
        $\kappa$ is an ordinal and $\alpha$ is an ordinal and
        $\beta$ is an ordinal and
        for every
            $c\in\funs{\fixedCardinalitySubsets{\kappa}{\cardTwo}}
                {\cardTwo}$ we have
            (there exists $H$ such that
                $\orderType{H}{\memrel{H}}=\alpha$ and
                $\pairHomogeneous{H}{c}{\kappa}{\cardZero}$) or
            (there exists $K$ such that
                $\orderType{K}{\memrel{K}}=\beta$ and
                $\pairHomogeneous{K}{c}{\kappa}{\cardOne}$).
\end{definition}

\begin{proposition}\label{two_colour_pair_arrow_ordinals}
    Suppose $\twoColourPairArrow{\kappa}{\alpha}{\beta}$.
    Then $\kappa$ is an ordinal and $\alpha$ is an ordinal and
        $\beta$ is an ordinal.
\end{proposition}
\begin{proof}
    Follows by definition.
\end{proof}

\begin{proposition}\label{two_colour_pair_arrow_elim}
    Suppose $\twoColourPairArrow{\kappa}{\alpha}{\beta}$.
    Let $c\in\funs{\fixedCardinalitySubsets{\kappa}{\cardTwo}}
        {\cardTwo}$.
    Then (there exists $H$ such that
            $\orderType{H}{\memrel{H}}=\alpha$ and
            $\pairHomogeneous{H}{c}{\kappa}{\cardZero}$) or
        (there exists $K$ such that
            $\orderType{K}{\memrel{K}}=\beta$ and
            $\pairHomogeneous{K}{c}{\kappa}{\cardOne}$).
\end{proposition}
\begin{proof}
    Follows by definition.
\end{proof}

\begin{proposition}\label{two_colour_pair_arrow_intro}
    Let $\kappa,\alpha,\beta$ be ordinals.
    Suppose for every
        $c\in\funs{\fixedCardinalitySubsets{\kappa}{\cardTwo}}
            {\cardTwo}$ we have
        (there exists $H$ such that
            $\orderType{H}{\memrel{H}}=\alpha$ and
            $\pairHomogeneous{H}{c}{\kappa}{\cardZero}$) or
        (there exists $K$ such that
            $\orderType{K}{\memrel{K}}=\beta$ and
            $\pairHomogeneous{K}{c}{\kappa}{\cardOne}$).
    Then $\twoColourPairArrow{\kappa}{\alpha}{\beta}$.
\end{proposition}
\begin{proof}
    Follows by definition.
\end{proof}
