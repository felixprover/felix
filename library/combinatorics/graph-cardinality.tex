\section{Cliques and independent sets of prescribed size}

\import{combinatorics/simple-graph.tex}
\import{cardinal.tex}

\begin{abbreviation}\label{has_equinumerous_independent_set}
    $E$ has an independent set in $X$ equinumerous with $Y$ iff
        some independent set in $X$ with respect to $E$
            is equinumerous with $Y$.
\end{abbreviation}

\begin{abbreviation}\label{has_equinumerous_clique}
    $E$ has a clique in $X$ equinumerous with $Y$ iff
        some clique in $X$ with respect to $E$ is equinumerous with $Y$.
\end{abbreviation}

\begin{abbreviation}\label{has_countably_infinite_independent_set}
    $E$ has a countably infinite independent set in $X$ iff
        some independent set in $X$ with respect to $E$ is countably infinite.
\end{abbreviation}

\begin{abbreviation}\label{has_countably_infinite_clique}
    $E$ has a countably infinite clique in $X$ iff
        some clique in $X$ with respect to $E$ is countably infinite.
\end{abbreviation}
