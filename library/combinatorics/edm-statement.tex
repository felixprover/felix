\section{The Erdős--Dushnik--Miller statement}

\import{combinatorics/graph-cardinality.tex}
\import{cardinal.tex}
\import{set/equinumerosity.tex}

The induction statement is formulated for every carrier equinumerous with the
parameter.  This lets an induction hypothesis apply directly to induced
subgraphs, without first replacing their vertices by ordinals.

\begin{definition}\label{edm_at}
    $\edmAt{\kappa}$ iff for every $X,E$ such that
        $X\approx\kappa$ and $E$ is a simple graph on $X$ we have
        $E$ has an independent set in $X$ equinumerous with $\kappa$ or
        $E$ has a countably infinite clique in $X$.
\end{definition}

\begin{proposition}\label{edm_at_elim}
    Suppose $\edmAt{\kappa}$ and $X\approx\kappa$.
    Suppose $E$ is a simple graph on $X$.
    Then $E$ has an independent set in $X$ equinumerous with $\kappa$ or
        $E$ has a countably infinite clique in $X$.
\end{proposition}
\begin{proof}
    Follows by definition.
\end{proof}

\begin{proposition}\label{edm_at_intro}
    Suppose for every $X,E$ such that
        $X\approx\kappa$ and $E$ is a simple graph on $X$ we have
        $E$ has an independent set in $X$ equinumerous with $\kappa$ or
        $E$ has a countably infinite clique in $X$.
    Then $\edmAt{\kappa}$.
\end{proposition}
\begin{proof}
    Follows by definition.
\end{proof}
