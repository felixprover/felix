\section{Pair colourings}

\import{function.tex}
\import{combinatorics/fixed-cardinality-subsets.tex}

\begin{definition}\label{pair_homogeneous}
    $\pairHomogeneous{H}{c}{X}{i}$ iff $H\subseteq X$ and
        for every $p\in\fixedCardinalitySubsets{H}{\cardTwo}$ we have
            $c(p)=i$.
\end{definition}

\begin{proposition}\label{pair_homogeneous_subseteq}
    Suppose $\pairHomogeneous{H}{c}{X}{i}$.
    Then $H\subseteq X$.
\end{proposition}
\begin{proof}
    Follows by definition.
\end{proof}

\begin{proposition}\label{pair_homogeneous_colour}
    Suppose $\pairHomogeneous{H}{c}{X}{i}$.
    Let $p\in\fixedCardinalitySubsets{H}{\cardTwo}$.
    Then $c(p)=i$.
\end{proposition}
\begin{proof}
    Follows by definition.
\end{proof}

\begin{proposition}\label{pair_homogeneous_hereditary}
    Suppose $\pairHomogeneous{H}{c}{X}{i}$ and $L\subseteq H$.
    Then $\pairHomogeneous{L}{c}{X}{i}$.
\end{proposition}
\begin{proof}
    $H\subseteq X$ by \obv{pair_homogeneous_subseteq}.
    Hence $L\subseteq X$ by \obv{subseteq_transitive}.
    $\fixedCardinalitySubsets{L}{\cardTwo}
        \subseteq\fixedCardinalitySubsets{H}{\cardTwo}$ by
        \obv{fixed_cardinality_subsets_monotone}.
    Show that for every $p\in\fixedCardinalitySubsets{L}{\cardTwo}$
        we have $c(p)=i$.
    \begin{subproof}
        Fix $p\in\fixedCardinalitySubsets{L}{\cardTwo}$.
        Hence $p\in\fixedCardinalitySubsets{H}{\cardTwo}$ by
            \obv{subseteq}.
        Follows by \obv{pair_homogeneous_colour}.
    \end{subproof}
    Follows by \obv{pair_homogeneous}.
\end{proof}
