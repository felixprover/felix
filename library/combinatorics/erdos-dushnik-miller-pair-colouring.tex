\subsection{The cardinality form}

\begin{theorem}\label{erdos_dushnik_miller_pair_cardinality}
    Let $\kappa$ be an infinite cardinal.
    Let $c\in\funs{\fixedCardinalitySubsets{\kappa}{\cardTwo}}
        {\cardTwo}$.
    Then (there exists $H$ such that $H\approx\kappa$ and
            $\pairHomogeneous{H}{c}{\kappa}{\cardZero}$) or
        (there exists $K$ such that $K$ is countably infinite and
            $\pairHomogeneous{K}{c}{\kappa}{\cardOne}$).
\end{theorem}
\begin{proof}
    $\kappa$ is a cardinal and $\kappa$ is infinite by definition.
    $\pairColourGraph{c}{\kappa}{\cardOne}$ is a simple graph on $\kappa$ by
        \obv{pair_colour_graph_simple}.
    Hence $\pairColourGraph{c}{\kappa}{\cardOne}$ has an independent set
            in $\kappa$ equinumerous with $\kappa$ or
        $\pairColourGraph{c}{\kappa}{\cardOne}$ has a countably infinite clique
            in $\kappa$ by
        \obv{erdos_dushnik_miller_graph}.
    \begin{byCase}
        \caseOf{$\pairColourGraph{c}{\kappa}{\cardOne}$ has an independent set
            in $\kappa$ equinumerous with $\kappa$.}
            Take an independent set $H$ in $\kappa$ with respect to
                $\pairColourGraph{c}{\kappa}{\cardOne}$ such that $H\approx\kappa$.
            Hence $\pairHomogeneous{H}{c}{\kappa}{\cardZero}$ by
                \obv{two_colour_pair_graph_independent_iff}.
        \caseOf{$\pairColourGraph{c}{\kappa}{\cardOne}$ has a countably infinite
            clique in $\kappa$.}
            Take a countably infinite clique $K$ in $\kappa$ with respect to
                $\pairColourGraph{c}{\kappa}{\cardOne}$.
            Hence $\pairHomogeneous{K}{c}{\kappa}{\cardOne}$ by
                \obv{pair_colour_graph_clique_iff}.
    \end{byCase}
\end{proof}

\subsection{The order-type form}

\begin{theorem}\label{erdos_dushnik_miller_pair_order_type}
    Let $\kappa$ be an infinite cardinal.
    Let $c\in\funs{\fixedCardinalitySubsets{\kappa}{\cardTwo}}
        {\cardTwo}$.
    Then (there exists $H$ such that
            $\orderType{H}{\memrel{H}}=\kappa$ and
            $\pairHomogeneous{H}{c}{\kappa}{\cardZero}$) or
        (there exists $L$ such that
            $\orderType{L}{\memrel{L}}=\naturals$ and
            $\pairHomogeneous{L}{c}{\kappa}{\cardOne}$).
\end{theorem}
\begin{proof}
    Hence (there exists $H$ such that $H\approx\kappa$ and
            $\pairHomogeneous{H}{c}{\kappa}{\cardZero}$) or
        (there exists $K$ such that $K$ is countably infinite and
            $\pairHomogeneous{K}{c}{\kappa}{\cardOne}$) by
        \obv{erdos_dushnik_miller_pair_cardinality}.
    \begin{byCase}
        \caseOf{There exists $H$ such that $H\approx\kappa$ and
            $\pairHomogeneous{H}{c}{\kappa}{\cardZero}$.}
            Take $H$ such that $H\approx\kappa$ and
                $\pairHomogeneous{H}{c}{\kappa}{\cardZero}$.
            $H\subseteq\kappa$ by \obv{pair_homogeneous_subseteq}.
            Hence $\orderType{H}{\memrel{H}}=\kappa$ by
                \obv{full_cardinality_subset_order_type}.
        \caseOf{There exists no $H$ such that $H\approx\kappa$ and
            $\pairHomogeneous{H}{c}{\kappa}{\cardZero}$.}
            Take $K$ such that $K$ is countably infinite and
                $\pairHomogeneous{K}{c}{\kappa}{\cardOne}$.
            $K\subseteq\kappa$ by \obv{pair_homogeneous_subseteq}.
            $K$ is infinite by \obv{countably_infinite}.
            $\kappa$ is an ordinal by
                \auto{cardinal,initial_ordinal}.
            Take $L\subseteq K$ such that
                $\orderType{L}{\memrel{L}}=\naturals$ by
                \obv{infinite_ordinal_subset_has_omega_order_type}.
            Hence $\pairHomogeneous{L}{c}{\kappa}{\cardOne}$ by
                \obv{pair_homogeneous_hereditary}.
    \end{byCase}
\end{proof}

\subsection{The partition relation}

\begin{theorem}\label{erdos_dushnik_miller_pair_arrow}
    Let $\kappa$ be an infinite cardinal.
    Then $\twoColourPairArrow{\kappa}{\kappa}{\naturals}$.
\end{theorem}
\begin{proof}
    $\kappa$ is an ordinal by \auto{cardinal,initial_ordinal}.
    $\naturals$ is an ordinal by \obv{omega_is_ordinal}.
    Show that for every
        $c\in\funs{\fixedCardinalitySubsets{\kappa}{\cardTwo}}
            {\cardTwo}$ we have
        (there exists $H$ such that
            $\orderType{H}{\memrel{H}}=\kappa$ and
            $\pairHomogeneous{H}{c}{\kappa}{\cardZero}$) or
        (there exists $K$ such that
            $\orderType{K}{\memrel{K}}=\naturals$ and
            $\pairHomogeneous{K}{c}{\kappa}{\cardOne}$).
    \begin{subproof}
        Fix $c\in\funs{\fixedCardinalitySubsets{\kappa}{\cardTwo}}
            {\cardTwo}$.
        Follows by \obv{erdos_dushnik_miller_pair_order_type}.
    \end{subproof}
    Follows by \obv{two_colour_pair_arrow_intro}.
\end{proof}
