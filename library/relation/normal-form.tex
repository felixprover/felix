\section{Irreducibility and normal forms}

\begin{definition}\label{irreducible_with_respect_to_on}
    Let $R$ be a binary relation on $X$.
    Let $y\in X$.
    $y$ is irreducible in $X$ with respect to $R$ iff
        there exists no $z\in X$ such that $y\mathrel{R}z$.
\end{definition}

\begin{definition}\label{normal_form_with_respect_to_on}
    Let $R$ be a binary relation on $X$.
    Let $x,y\in X$.
    $y$ is a normal form of $x$ in $X$ with respect to $R$ iff
        $x\mathrel{\reflexiveTransitiveClosure{X}{R}}y$ and
        $y$ is irreducible in $X$ with respect to $R$.
\end{definition}

\begin{definition}\label{normalizing_on}
    Let $R$ be a binary relation on $X$.
    $R$ is normalizing on $X$ iff for every $x\in X$ there exists $y\in X$
        such that $y$ is a normal form of $x$ in $X$ with respect to $R$.
\end{definition}

\begin{proposition}\label{irreducible_reflexive_transitive_elim}
    Let $R$ be a binary relation on $X$.
    Let $y,z\in X$.
    Suppose $y$ is irreducible in $X$ with respect to $R$.
    Suppose $y\mathrel{\reflexiveTransitiveClosure{X}{R}}z$.
    Then $y=z$.
\end{proposition}
\begin{proof}
    Suppose not.
    Take $u\in X$ such that $y\mathrel{R}u$ and
        $u\mathrel{\reflexiveTransitiveClosure{X}{R}}z$
        by \auto{reflexive_transitive_closure_first_step}.
    There exists no $v\in X$ such that $y\mathrel{R}v$
        by \obv{irreducible_with_respect_to_on}.
    Contradiction by \auto.
\end{proof}
