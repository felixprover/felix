\section{Normal forms under Newman's hypotheses}

\begin{corollary}\label{newman_relation_unique_normal_forms}
    Let $R$ be a binary relation on $X$.
    Suppose $R$ is Noetherian on $X$.
    Suppose $R$ is locally confluent on $X$.
    Then for every $x\in X$ there exists $y\in X$ such that
        $y$ is a normal form of $x$ in $X$ with respect to $R$ and
        for every $z\in X$ we have if
            $z$ is a normal form of $x$ in $X$ with respect to $R$,
            then $z=y$.
\end{corollary}
\begin{proof}
    $R$ is convergent on $X$ by \obv{newman_relation_convergent}.
    Follows by \obv{convergent_on_unique_normal_form}.
\end{proof}
