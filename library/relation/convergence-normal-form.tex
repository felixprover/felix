\section{Normal forms of convergent relations}

\begin{corollary}\label{convergent_on_unique_normal_form}
    Let $R$ be a binary relation on $X$.
    Suppose $R$ is convergent on $X$.
    Then for every $x\in X$ there exists $y\in X$ such that
        $y$ is a normal form of $x$ in $X$ with respect to $R$ and
        for every $z\in X$ we have if $z$ is a normal form of $x$ in $X$
            with respect to $R$, then $z=y$.
\end{corollary}
\begin{proof}
    $R$ is Noetherian on $X$ and $R$ is confluent on $X$
        by \obv{convergent_on}.
    $R$ is normalizing on $X$
        by \obv{noetherian_on_implies_normalizing_on}.
    Fix $x\in X$.
    Take $y\in X$ such that
        $y$ is a normal form of $x$ in $X$ with respect to $R$
        by \obv{normalizing_on}.
    For every $z\in X$ we have if $z$ is a normal form of $x$ in $X$
        with respect to $R$, then $z=y$
        by \obv{confluent_on_normal_forms_unique}.
    Follows by \auto.
\end{proof}
