\section{Noetherian relations}

\import{relation/well-founded.tex}
\import{relation/trace.tex}

\begin{definition}\label{noetherian}
    Let $R$ be a relation.
    $R$ is Noetherian iff $\converse{R}$ is well-founded.
\end{definition}

\begin{definition}\label{noetherian_on}
    Let $R$ be a binary relation on $X$.
    $R$ is Noetherian on $X$ iff
        $\converse{R}$ is well-founded on $X$.
\end{definition}

\begin{proposition}\label{noetherian_on_iff_noetherian}
    Let $R$ be a binary relation on $X$.
    Then $R$ is Noetherian on $X$ iff $R$ is Noetherian.
\end{proposition}
\begin{proof}
    $R$ is a relation by notion.
    $\converse{R}$ is a relation by \obv{converse_is_relation}.
    $\fld{\converse{R}}=\fld{R}$ by \obv{fld_converse}.
    $\fld{R}\subseteq X$ by \obv{binary_relation_field_subseteq}.
    Hence $\fld{\converse{R}}\subseteq X$.
    Follows by
        \auto{noetherian,noetherian_on,
            well_founded_iff_well_founded_on_ambient}.
\end{proof}

\begin{proposition}\label{noetherian_iff_noetherian_on_field}
    Let $R$ be a relation.
    Then $R$ is Noetherian iff $R$ is Noetherian on $\fld{R}$.
\end{proposition}
\begin{proof}
    $R$ is a binary relation on $\fld{R}$ by notion.
    Follows by \obv{noetherian_on_iff_noetherian}.
\end{proof}

\begin{proposition}\label{noetherian_on_sink_characterization}
    Let $R$ be a binary relation on $X$.
    $R$ is Noetherian on $X$ iff
        for every inhabited subset $Y$ of $X$ there exists $m\in Y$
            such that for every $y\in Y$ we have
                $m\not\mathrel{R}y$.
\end{proposition}
\begin{proof}
    $\converse{R}$ is a binary relation on $X$ by notion.
    Hence $\converse{R}$ is a relation by notion.
    Follows by \auto{noetherian_on,well_founded_on,converse_iff}.
\end{proof}

\begin{theorem}\label{noetherian_induction_set}
    Let $R$ be a binary relation on $X$.
    Suppose $R$ is Noetherian on $X$.
    Suppose for every $x\in X$ we have
        if for every $y\in X$ such that $x\mathrel{R}y$ we have $y\in B$,
        then $x\in B$.
    Then $X\subseteq B$.
\end{theorem}
\begin{proof}
    $\converse{R}$ is well-founded on $X$ by definition.
    $B$ is progressive for $\converse{R}$ on $X$ by
        \obv{progressive_on,converse_iff}.
    Follows by \obv{well_founded_induction_set}.
\end{proof}

\begin{theorem}\label{noetherian_induction}
    Let $R$ be a binary relation on $X$.
    Suppose $R$ is Noetherian on $X$.
    Then $\forall P:\Set\to\Prop.
        (\forall x:\Set.\ x\in X\implies
            ((\forall y:\Set.\ (y\in X\land x\mathrel{R}y)
                \implies\Apply{P}{y})
            \implies\Apply{P}{x}))\implies
        (\forall x:\Set.\ x\in X\implies\Apply{P}{x})$.
\end{theorem}
\begin{proof}
    $\converse{R}$ is well-founded on $X$ by definition.
    Follows by \obv{well_founded_induction,converse_iff}.
\end{proof}

\begin{proposition}\label{noetherian_on_subrelation}
    Let $R$ be a binary relation on $X$.
    Let $S$ be a binary relation on $X$.
    Suppose $R$ is Noetherian on $X$ and $S\subseteq R$.
    Then $S$ is Noetherian on $X$.
\end{proposition}
\begin{proof}
    $S$ is a relation by notion.
    $\converse{R}$ is well-founded on $X$ by definition.
    $\converse{S}\subseteq\converse{R}$
        by \obv{converse_subseteq_intro}.
    Hence $\converse{S}$ is well-founded on $X$
        by \obv{well_founded_on_subset_relation}.
    Follows by definition.
\end{proof}

\begin{proposition}\label{noetherian_subrelation}
    Let $R$ be a relation.
    Let $S$ be a relation.
    Suppose $R$ is Noetherian and $S\subseteq R$.
    Then $S$ is Noetherian.
\end{proposition}
\begin{proof}
    $R$ is a binary relation on $\fld{R}$ by notion.
    $R$ is Noetherian on $\fld{R}$
        by \obv{noetherian_iff_noetherian_on_field}.
    $R\subseteq\fld{R}\times\fld{R}$
        by \obv{relation_subseteq_times_fld}.
    Hence $S\subseteq\fld{R}\times\fld{R}$
        by \obv{subseteq_transitive}.
    Thus $S$ is a binary relation on $\fld{R}$ by definition.
    Hence $S$ is Noetherian on $\fld{R}$
        by \obv{noetherian_on_subrelation}.
    Follows by \obv{noetherian_on_iff_noetherian}.
\end{proof}

\begin{proposition}\label{relation_trace_noetherian_on}
    Let $R$ be a binary relation on $X$.
    Suppose $R$ is Noetherian on $X$.
    Let $Y\subseteq X$.
    Then $\relationTrace{R}{Y}$ is Noetherian on $Y$.
\end{proposition}
\begin{proof}
    $\converse{R}$ is well-founded on $X$ by definition.
    Hence $\converse{R}$ is well-founded on $Y$
        by \obv{well_founded_on_subset_carrier}.
    $\relationTrace{R}{Y}\subseteq R$
        by \obv{relation_trace_subseteq}.
    $\relationTrace{R}{Y}$ is a binary relation on $Y$ by notion.
    Hence $\relationTrace{R}{Y}$ is a relation by notion.
    Hence $\converse{\relationTrace{R}{Y}}\subseteq\converse{R}$
        by \obv{converse_subseteq_intro}.
    Thus $\converse{\relationTrace{R}{Y}}$ is well-founded on $Y$
        by \obv{well_founded_on_subset_relation}.
    Follows by definition.
\end{proof}

\begin{proposition}\label{noetherian_is_irreflexive}
    Let $R$ be a relation.
    Suppose $R$ is Noetherian.
    Then $R$ is irreflexive.
\end{proposition}
\begin{proof}
    $\converse{R}$ is well-founded by definition.
    Hence $\converse{R}$ is well-founded on $\fld{\converse{R}}$
        by \obv{well_founded}.
    $\fld{\converse{R}}=\fld{R}$ by \obv{fld_converse}.
    Show that for every $x$ we have $(x,x)\notin R$.
    \begin{subproof}
        Fix $x$.
        Suppose $(x,x)\in R$.
        Then $x\in\fld{R}$ by \obv{fld_intro_left}.
        Hence $x\in\fld{\converse{R}}$.
        Thus $x\not\mathrel{\converse{R}}x$
            by \obv{well_founded_on_irreflexive}.
        $x\mathrel{\converse{R}}x$ by \obv{converse_iff}.
        Contradiction.
    \end{subproof}
    Follows by \obv{irreflexive}.
\end{proof}
