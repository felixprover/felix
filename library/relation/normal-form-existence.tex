\section{Existence of normal forms}

\begin{theorem}\label{accessible_implies_normal_form}
    Let $R$ be a binary relation on $X$.
    Let $x\in X$.
    Suppose $x\in\accessiblePart{X}{\converse{R}}$.
    Then there exists $y\in X$ such that
        $y$ is a normal form of $x$ in $X$ with respect to $R$.
\end{theorem}
\begin{proof}
    Let $B=\{u\in X\mid\text{there exists $v\in X$ such that
        $v$ is a normal form of $u$ in $X$ with respect to $R$}\}$.
    Show that $B$ is progressive for $\converse{R}$ on $X$.
    \begin{subproof}
        Show that for every $u\in X$ we have if for every $v\in X$ such that
            $v\mathrel{\converse{R}}u$ we have $v\in B$, then $u\in B$.
        \begin{subproof}
            Fix $u\in X$.
            Suppose for every $v\in X$ such that
                $v\mathrel{\converse{R}}u$ we have $v\in B$.

            Show that if $u$ is irreducible in $X$ with respect to $R$,
                then $u\in B$.
            \begin{subproof}
                Suppose $u$ is irreducible in $X$ with respect to $R$.
                $u\mathrel{\reflexiveTransitiveClosure{X}{R}}u$
                    by \auto{reflexive_transitive_closure_is_reflexive,
                        reflexive_on}.
                Hence $u$ is a normal form of $u$ in $X$ with respect to $R$
                    by \obv{normal_form_with_respect_to_on}.
            \end{subproof}

            Show that if $u$ is not irreducible in $X$ with respect to $R$,
                then $u\in B$.
            \begin{subproof}
                Suppose $u$ is not irreducible in $X$ with respect to $R$.
                Take $v\in X$ such that $u\mathrel{R}v$
                    by \auto{irreducible_with_respect_to_on}.
                $v\mathrel{\converse{R}}u$ by \obv{converse_iff}.
                Hence $v\in B$.
                Take $n\in X$ such that
                    $n$ is a normal form of $v$ in $X$ with respect to $R$.
                $v\mathrel{\reflexiveTransitiveClosure{X}{R}}n$ and
                    $n$ is irreducible in $X$ with respect to $R$
                    by \obv{normal_form_with_respect_to_on}.
                $u\mathrel{\reflexiveTransitiveClosure{X}{R}}v$
                    by \auto{reflexive_transitive_closure_contains,subseteq}.
                $\reflexiveTransitiveClosure{X}{R}$ is transitive
                    by \obv{reflexive_transitive_closure_is_transitive}.
                Hence $u\mathrel{\reflexiveTransitiveClosure{X}{R}}n$
                    by \obv{transitive}.
                Thus $n$ is a normal form of $u$ in $X$ with respect to $R$
                    by \obv{normal_form_with_respect_to_on}.
                Follows by \auto.
            \end{subproof}
        \end{subproof}
        Follows by \obv{progressive_on}.
    \end{subproof}
    Hence $x\in B$ by \obv{accessible_induction_set}.
\end{proof}

\begin{corollary}\label{noetherian_on_implies_normalizing_on}
    Let $R$ be a binary relation on $X$.
    Suppose $R$ is Noetherian on $X$.
    Then $R$ is normalizing on $X$.
\end{corollary}
\begin{proof}
    $\converse{R}$ is well-founded on $X$ by definition.
    Hence $X\subseteq\accessiblePart{X}{\converse{R}}$
        by \obv{well_founded_on_implies_accessible}.
    For every $x\in X$ there exists $y\in X$ such that
        $y$ is a normal form of $x$ in $X$ with respect to $R$
        by \obv{subseteq,accessible_implies_normal_form}.
    Follows by \obv{normalizing_on}.
\end{proof}

\begin{corollary}\label{noetherian_implies_normalizing_on_field}
    Let $R$ be a relation.
    Suppose $R$ is Noetherian.
    Then $R$ is normalizing on $\fld{R}$.
\end{corollary}
\begin{proof}
    $R$ is a binary relation on $\fld{R}$ by notion.
    $R$ is Noetherian on $\fld{R}$
        by \obv{noetherian_iff_noetherian_on_field}.
    Follows by \obv{noetherian_on_implies_normalizing_on}.
\end{proof}
