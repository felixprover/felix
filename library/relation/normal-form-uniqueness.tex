\section{Uniqueness of normal forms}

\begin{theorem}\label{confluent_at_normal_forms_unique}
    Let $R$ be a binary relation on $X$.
    Let $x,y,z\in X$.
    Suppose $R$ is confluent at $x$ on $X$.
    Suppose $y$ and $z$ are normal forms of $x$ in $X$ with respect to $R$.
    Then $y=z$.
\end{theorem}
\begin{proof}
    $x\mathrel{\reflexiveTransitiveClosure{X}{R}}y$ and
        $y$ is irreducible in $X$ with respect to $R$
        by \obv{normal_form_with_respect_to_on}.
    $x\mathrel{\reflexiveTransitiveClosure{X}{R}}z$ and
        $z$ is irreducible in $X$ with respect to $R$
        by \obv{normal_form_with_respect_to_on}.
    Hence $y$ is joinable with $z$ in $X$ with respect to $R$
        by \obv{confluent_at_on}.
    Take $d\in X$ such that
        $y\mathrel{\reflexiveTransitiveClosure{X}{R}}d$ and
        $z\mathrel{\reflexiveTransitiveClosure{X}{R}}d$
        by \obv{joinability_relation_iff}.
    $y=d$ by \obv{irreducible_reflexive_transitive_elim}.
    $z=d$ by \obv{irreducible_reflexive_transitive_elim}.
\end{proof}

\begin{corollary}\label{confluent_on_normal_forms_unique}
    Let $R$ be a binary relation on $X$.
    Suppose $R$ is confluent on $X$.
    Let $x,y,z\in X$.
    Suppose $y$ and $z$ are normal forms of $x$ in $X$ with respect to $R$.
    Then $y=z$.
\end{corollary}
\begin{proof}
    $R$ is confluent at $x$ on $X$ by \obv{confluent_on}.
    Follows by \obv{confluent_at_normal_forms_unique}.
\end{proof}
