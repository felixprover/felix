\section{Normal forms in abstract reduction systems}

\begin{definition}\label{irreducible_in_reduction_system}
    Let $A$ be an abstract reduction system.
    Let $y\in\carrier[A]$.
    $y$ is irreducible in $A$ iff
        $y$ is irreducible in $\carrier[A]$ with respect to $\reduction[A]$.
\end{definition}

\begin{definition}\label{normal_form_in_reduction_system}
    Let $A$ be an abstract reduction system.
    Let $x,y\in\carrier[A]$.
    $y$ is a normal form of $x$ in $A$ iff
        $y$ is a normal form of $x$ in $\carrier[A]$
            with respect to $\reduction[A]$.
\end{definition}

\begin{definition}\label{normalizing_reduction_system}
    Let $A$ be an abstract reduction system.
    $A$ is normalizing iff
        $\reduction[A]$ is normalizing on $\carrier[A]$.
\end{definition}

\begin{proposition}\label{irreducible_reduction_star_elim}
    Let $A$ be an abstract reduction system.
    Let $y,z\in\carrier[A]$.
    Suppose $y$ is irreducible in $A$.
    Suppose $y\mathrel{\reductionStar{A}}z$.
    Then $y=z$.
\end{proposition}
\begin{proof}
    $\reduction[A]$ is a binary relation on $\carrier[A]$ by notion.
    $y$ is irreducible in $\carrier[A]$ with respect to $\reduction[A]$
        by \obv{irreducible_in_reduction_system}.
    $y\mathrel{\reflexiveTransitiveClosure{\carrier[A]}{\reduction[A]}}z$
        by \obv{reduction_star}.
    Follows by \obv{irreducible_reflexive_transitive_elim}.
\end{proof}
