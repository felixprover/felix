\section{Newman's lemma for abstract reduction systems}

\begin{theorem}[Newman's lemma]\label{newman}
    Let $A$ be an abstract reduction system.
    Suppose $A$ is locally confluent and terminating.
    Then $A$ is confluent.
\end{theorem}
\begin{proof}
    $\reduction[A]$ is Noetherian on $\carrier[A]$ by definition.
    $\reduction[A]$ is locally confluent on $\carrier[A]$ by definition.
    Hence $\reduction[A]$ is confluent on $\carrier[A]$
        by \obv{newman_relation}.
    Follows by definition.
\end{proof}

\begin{corollary}\label{newman_convergent}
    Let $A$ be an abstract reduction system.
    Suppose $A$ is locally confluent and terminating.
    Then $A$ is convergent.
\end{corollary}
\begin{proof}
    $A$ is confluent by \obv{newman}.
    Follows by \obv{convergent_reduction_system_iff}.
\end{proof}
