\section{Properties of abstract reduction systems}

\begin{definition}\label{joinable_in_reduction_system}
    Let $A$ be an abstract reduction system.
    Let $b,c\in\carrier[A]$.
    $b$ is joinable with $c$ in $A$ iff
        $b$ is joinable with $c$ in $\carrier[A]$
            with respect to $\reduction[A]$.
\end{definition}

\begin{definition}\label{confluent_at_reduction_system}
    Let $A$ be an abstract reduction system.
    Let $a\in\carrier[A]$.
    $A$ is confluent at $a$ iff
        $\reduction[A]$ is confluent at $a$ on $\carrier[A]$.
\end{definition}

\begin{definition}\label{locally_confluent_at_reduction_system}
    Let $A$ be an abstract reduction system.
    Let $a\in\carrier[A]$.
    $A$ is locally confluent at $a$ iff
        $\reduction[A]$ is locally confluent at $a$ on $\carrier[A]$.
\end{definition}

\begin{definition}\label{terminating_reduction_system}
    Let $A$ be an abstract reduction system.
    $A$ is terminating iff
        $\reduction[A]$ is Noetherian on $\carrier[A]$.
\end{definition}

\begin{definition}\label{confluent_reduction_system}
    Let $A$ be an abstract reduction system.
    $A$ is confluent iff
        $\reduction[A]$ is confluent on $\carrier[A]$.
\end{definition}

\begin{definition}\label{locally_confluent_reduction_system}
    Let $A$ be an abstract reduction system.
    $A$ is locally confluent iff
        $\reduction[A]$ is locally confluent on $\carrier[A]$.
\end{definition}

\begin{definition}\label{convergent_reduction_system}
    Let $A$ be an abstract reduction system.
    $A$ is convergent iff
        $\reduction[A]$ is convergent on $\carrier[A]$.
\end{definition}

\begin{proposition}\label{convergent_reduction_system_iff}
    Let $A$ be an abstract reduction system.
    Then $A$ is convergent iff
        $A$ is terminating and $A$ is confluent.
\end{proposition}
\begin{proof}
    $\reduction[A]$ is a binary relation on $\carrier[A]$ by notion.
    Follows by \auto{convergent_reduction_system,convergent_on,
        terminating_reduction_system,confluent_reduction_system}.
\end{proof}
