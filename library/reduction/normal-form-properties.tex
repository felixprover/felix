\section{Normal-form consequences for abstract reduction systems}

\begin{corollary}\label{terminating_reduction_system_is_normalizing}
    Let $A$ be a terminating abstract reduction system.
    Then $A$ is normalizing.
\end{corollary}
\begin{proof}
    $\reduction[A]$ is a binary relation on $\carrier[A]$ by notion.
    $\reduction[A]$ is Noetherian on $\carrier[A]$ by definition.
    Hence $\reduction[A]$ is normalizing on $\carrier[A]$
        by \obv{noetherian_on_implies_normalizing_on}.
    Follows by definition.
\end{proof}

\begin{theorem}\label{confluent_at_reduction_system_normal_forms_unique}
    Let $A$ be an abstract reduction system.
    Let $x,y,z\in\carrier[A]$.
    Suppose $A$ is confluent at $x$.
    Suppose $y$ and $z$ are normal forms of $x$ in $A$.
    Then $y=z$.
\end{theorem}
\begin{proof}
    $\reduction[A]$ is a binary relation on $\carrier[A]$ by notion.
    $\reduction[A]$ is confluent at $x$ on $\carrier[A]$ by definition.
    $y$ and $z$ are normal forms of $x$ in $\carrier[A]$
        with respect to $\reduction[A]$ by definition.
    Follows by \obv{confluent_at_normal_forms_unique}.
\end{proof}

\begin{corollary}\label{confluent_reduction_system_normal_forms_unique}
    Let $A$ be a confluent abstract reduction system.
    Let $x,y,z\in\carrier[A]$.
    Suppose $y$ and $z$ are normal forms of $x$ in $A$.
    Then $y=z$.
\end{corollary}
\begin{proof}
    $\reduction[A]$ is a binary relation on $\carrier[A]$ by notion.
    $\reduction[A]$ is confluent on $\carrier[A]$ by definition.
    $y$ and $z$ are normal forms of $x$ in $\carrier[A]$
        with respect to $\reduction[A]$ by definition.
    Follows by \obv{confluent_on_normal_forms_unique}.
\end{proof}

\begin{theorem}\label{convergent_unique_normal_forms}
    Let $A$ be a convergent abstract reduction system.
    Then for every $x\in\carrier[A]$ there exists $y\in\carrier[A]$ such that
        $y$ is a normal form of $x$ in $A$ and
        for every $z\in\carrier[A]$ we have if
            $z$ is a normal form of $x$ in $A$, then $z=y$.
\end{theorem}
\begin{proof}
    $\reduction[A]$ is a binary relation on $\carrier[A]$ by notion.
    $\reduction[A]$ is convergent on $\carrier[A]$ by definition.
    Fix $x\in\carrier[A]$.
    Take $y\in\carrier[A]$ such that
        $y$ is a normal form of $x$ in $\carrier[A]$ with respect to
        $\reduction[A]$ and for every $z\in\carrier[A]$ we have if
        $z$ is a normal form of $x$ in $\carrier[A]$ with respect to
        $\reduction[A]$, then $z=y$
        by \obv{convergent_on_unique_normal_form}.
    $y$ is a normal form of $x$ in $A$ by definition.
    Show that for every $z\in\carrier[A]$ we have if
        $z$ is a normal form of $x$ in $A$, then $z=y$.
    \begin{subproof}
        Fix $z\in\carrier[A]$.
        Suppose $z$ is a normal form of $x$ in $A$.
        Then $z$ is a normal form of $x$ in $\carrier[A]$ with respect to
            $\reduction[A]$ by definition.
    \end{subproof}
\end{proof}
