\section{Codomain-free well-founded recursion}

This chapter constructs recursive set functions without prescribing a common
set of possible values.  The proof uses recursive partials on arbitrary
initial segments.  Any two such partials agree where both are defined; the
points covered by a partial therefore have unique values, which relational
Replacement collects into the global graph.  No set of all partials, Zorn
argument, or set-level choice function enters the construction.

\subsection{Recursive partials for a logical step}

\begin{definition}\label{recursive_set_function}
    $\recursiveSetFunction{f}$ iff $f$ is a function.
\end{definition}

\begin{definition}\label{recursive_initial_segment}
    $\recursiveInitialSegment{f}{A}{R}$ iff
        $\dom{f}$ is an initial segment of $A$ with respect to $R$.
\end{definition}

\begin{definition}\label{recursive_set_function_op}
    $\recursiveSetFunctionOp:\Set\to\Prop
        =\fun{f:\Set}{\recursiveSetFunction{f}}$.
\end{definition}

\begin{definition}\label{recursive_initial_segment_op}
    $\recursiveInitialSegmentOp:\Set\to\Set\to\Set\to\Prop
        =\fun{f,A,R:\Set}{\recursiveInitialSegment{f}{A}{R}}$.
\end{definition}

\begin{definition}\label{unbounded_recursive_history_op}
    $\unboundedRecursiveHistoryOp:
        \Set\to\Set\to\Set\to\Set\to\Prop
        =\fun{A,R,x,p:\Set}{
                        \Apply{\recursiveSetFunctionOp}{p}\land
                        \dom{p}=\downward{R}{x}}$.
\end{definition}

\begin{proposition}\label{unbounded_recursive_history_iff}
    $\forall A,R,x,p:\Set.
        \Apply{\unboundedRecursiveHistoryOp}
            {A,R,x,p}
        \iff
        (\Apply{\recursiveSetFunctionOp}{p}\land
         \dom{p}=\downward{R}{x})$.
\end{proposition}
\begin{proof}
    Follows by \obv{unbounded_recursive_history_op}.
\end{proof}

\begin{proposition}\label{restriction_is_unbounded_recursive_history}
    Suppose $R$ is a binary relation on $A$.
    Suppose $f$ is a function and $\dom{f}=A$.
    Let $x\in A$.
    Then $\Apply{\unboundedRecursiveHistoryOp}
        {A,R,x,\restrl{f}{\downward{R}{x}}}$.
\end{proposition}
\begin{proof}
    $\downward{R}{x}\subseteq A$ by
        \obv{binary_relation_downward_subseteq}.
    Hence $\downward{R}{x}\subseteq\dom{f}$.
    Hence $\recursiveSetFunction{\restrl{f}{\downward{R}{x}}}$ by
        \obv{recursive_set_function}.
    $\dom{\restrl{f}{\downward{R}{x}}}
        =\dom{f}\inter\downward{R}{x}$ by
        \obv{dom_of_restrl_eq_inter}.
    Hence $\dom{\restrl{f}{\downward{R}{x}}}
        =\downward{R}{x}$ by
        \obv{inter_absorb_supseteq_left}.
    Follows by
        \obv{unbounded_recursive_history_op,recursive_set_function_op}.
\end{proof}

\begin{definition}\label{unbounded_recursive_partial_op}
    $\unboundedRecursivePartialOp:
        \Set\to\Set\to
        (\Set\to\Set\to\Set)\to\Set\to\Prop
        =\fun{A,R:\Set}{
                \fun{H:\Set\to\Set\to\Set}{
                    \fun{f:\Set}{
                        \Apply{\recursiveSetFunctionOp}{f}\land
                        \Apply{\recursiveInitialSegmentOp}
                            {f,A,R}\land
                        (\forall x:\Set.\ x\in\dom{f}\implies
                            \apply{f}{x}=\Apply{H}
                                {x,\restrl{f}{\downward{R}{x}}})}}}$.
\end{definition}

\begin{definition}\label{unbounded_recursive_solution_op}
    $\unboundedRecursiveSolutionOp:
        \Set\to\Set\to
        (\Set\to\Set\to\Set)\to\Set\to\Prop
        =\fun{A,R:\Set}{
                \fun{H:\Set\to\Set\to\Set}{
                    \fun{f:\Set}{
                        \Apply{\unboundedRecursivePartialOp}
                            {A,R,H,f}\land
                        \dom{f}=A}}}$.
\end{definition}

\begin{proposition}\label{unbounded_recursive_partial_iff}
    $\forall A,R:\Set.\ \forall H:\Set\to\Set\to\Set.
     \forall f:\Set.
        \Apply{\unboundedRecursivePartialOp}
            {A,R,H,f}
        \iff
        (\Apply{\recursiveSetFunctionOp}{f}\land
         \Apply{\recursiveInitialSegmentOp}{f,A,R}\land
         (\forall x:\Set.\ x\in\dom{f}\implies
            \apply{f}{x}=\Apply{H}
                {x,\restrl{f}{\downward{R}{x}}}))$.
\end{proposition}
\begin{proof}
    Follows by \obv{unbounded_recursive_partial_op}.
\end{proof}

\begin{proposition}\label{unbounded_recursive_solution_iff}
    $\forall A,R:\Set.\ \forall H:\Set\to\Set\to\Set.
     \forall f:\Set.
        \Apply{\unboundedRecursiveSolutionOp}
            {A,R,H,f}
        \iff
        (\Apply{\unboundedRecursivePartialOp}
            {A,R,H,f}\land
         \dom{f}=A)$.
\end{proposition}
\begin{proof}
    Follows by \obv{unbounded_recursive_solution_op}.
\end{proof}

\subsection{Agreement of partials}

\begin{proposition}\label{unbounded_recursive_partial_function}
    $\forall A,R:\Set.\ \forall H:\Set\to\Set\to\Set.
     \forall f:\Set.
        \Apply{\unboundedRecursivePartialOp}
            {A,R,H,f}
        \implies \recursiveSetFunction{f}$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{unbounded_recursive_partial_iff,recursive_set_function_op}.
\end{proof}

\begin{proposition}\label{unbounded_recursive_partial_initial}
    $\forall A,R:\Set.\ \forall H:\Set\to\Set\to\Set.
     \forall f:\Set.
        \Apply{\unboundedRecursivePartialOp}
            {A,R,H,f}
        \implies
        \recursiveInitialSegment{f}{A}{R}$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{unbounded_recursive_partial_iff,recursive_initial_segment_op}.
\end{proof}

\begin{proposition}\label{unbounded_recursive_partial_equation}
    $\forall A,R:\Set.\ \forall H:\Set\to\Set\to\Set.
     \forall f,x:\Set.
        (\Apply{\unboundedRecursivePartialOp}
            {A,R,H,f}\land x\in\dom{f})
        \implies
        \apply{f}{x}=\Apply{H}
            {x,\restrl{f}{\downward{R}{x}}}$.
\end{proposition}
\begin{proof}
    Follows by \obv{unbounded_recursive_partial_iff}.
\end{proof}

\begin{proposition}\label{unbounded_recursive_partial_domain_subseteq}
    $\forall A,R:\Set.\ \forall H:\Set\to\Set\to\Set.
     \forall f:\Set.
        \Apply{\unboundedRecursivePartialOp}
            {A,R,H,f}
        \implies \dom{f}\subseteq A$.
\end{proposition}
\begin{proof}
    Fix $A,R,H,f$.
    Assume
        $\Apply{\unboundedRecursivePartialOp}
            {A,R,H,f}$.
    $\recursiveInitialSegment{f}{A}{R}$ by
        \obv{unbounded_recursive_partial_initial}.
    Follows by
        \auto{recursive_initial_segment,initial_segment}.
\end{proof}

\begin{proposition}\label{recursive_history_predecessor_common}
    Suppose $\recursiveInitialSegment{f}{A}{R}$ and
        $\recursiveInitialSegment{g}{A}{R}$.
    Let $x\in\dom{f}\inter\dom{g}$.
    Let $y\in A$.
    Suppose $y\mathrel{R}x$.
    Then $y\in\dom{f}\inter\dom{g}$.
\end{proposition}
\begin{proof}
    $x\in\dom{f}$ and $x\in\dom{g}$ by \obv{inter}.
    Hence $y\in\dom{f}$ and $y\in\dom{g}$ by
        \auto{recursive_initial_segment,initial_segment}.
    Follows by \obv{inter_intro}.
\end{proof}

\begin{proposition}\label{recursive_history_agreement}
    Suppose $R$ is a binary relation on $A$.
    Suppose $\recursiveSetFunction{f}$ and $\recursiveSetFunction{g}$.
    Suppose $\recursiveInitialSegment{f}{A}{R}$ and
        $\recursiveInitialSegment{g}{A}{R}$.
    Let $x\in\dom{f}\inter\dom{g}$.
    Suppose for every $y\in A$ such that $y\mathrel{R}x$ we have
        $f(y)=g(y)$.
    Then
        $\restrl{f}{\downward{R}{x}}=
            \restrl{g}{\downward{R}{x}}$.
\end{proposition}
\begin{proof}
    Show that for every $y\in\downward{R}{x}$ such that
        $y\in\dom{f}\union\dom{g}$ we have
        $y\in\dom{f}\inter\dom{g}$ and $f(y)=g(y)$.
    \begin{subproof}
        Fix $y$.
        Assume $y\in\downward{R}{x}$.
        Assume $y\in\dom{f}\union\dom{g}$.
        $y\mathrel{R}x$ by \obv{downward_closure_iff}.
        $y\in A$ by \obv{subseteq,times_tuple_elim}.
        $y\in\dom{f}\inter\dom{g}$ by
            \obv{recursive_history_predecessor_common}.
        $f(y)=g(y)$.
    \end{subproof}
    Follows by \obv{restrictions_agree}.
\end{proof}

\begin{theorem}\label{unbounded_recursive_partials_agree_on_carrier}
    Suppose $R$ is a binary relation on $A$ and $R$ is well-founded on $A$.
    Then $\forall f,g:\Set.\ \forall H:\Set\to\Set\to\Set.
        (\Apply{\unboundedRecursivePartialOp}
            {A,R,H,f}\land
         \Apply{\unboundedRecursivePartialOp}
            {A,R,H,g})
        \implies
        (\forall x:\Set.\ x\in A\implies
            (x\in\dom{f}\inter\dom{g}\implies
                \apply{f}{x}=\apply{g}{x}))$.
\end{theorem}
\begin{proof}
    Fix $f,g,H$.
    Assume
        $\Apply{\unboundedRecursivePartialOp}
            {A,R,H,f}\land
         \Apply{\unboundedRecursivePartialOp}
            {A,R,H,g}$.
    Show that $\forall x:\Set.\ x\in A\implies
        (x\in\dom{f}\inter\dom{g}\implies
            \apply{f}{x}=\apply{g}{x})$.
    \begin{subproof}
        $\recursiveSetFunction{f}$ by
            \obv{unbounded_recursive_partial_function}.
        $\recursiveSetFunction{g}$ by
            \obv{unbounded_recursive_partial_function}.
        $\recursiveInitialSegment{f}{A}{R}$ by
            \obv{unbounded_recursive_partial_initial}.
        $\recursiveInitialSegment{g}{A}{R}$ by
            \obv{unbounded_recursive_partial_initial}.
        Let $P:\Set\to\Prop=\fun{x:\Set}{
            x\in\dom{f}\inter\dom{g}\implies
                \apply{f}{x}=\apply{g}{x}}$.
        Show that $\forall x:\Set.\ x\in A\implies
            ((\forall y:\Set.\ (y\in A\land y\mathrel{R}x)
                \implies\Apply{P}{y})
            \implies\Apply{P}{x})$.
        \begin{subproof}
            Fix $x$.
            Assume $x\in A$.
            Assume $\forall y:\Set.\ (y\in A\land y\mathrel{R}x)
                \implies\Apply{P}{y}$.
            Show that $x\in\dom{f}\inter\dom{g}\implies
                \apply{f}{x}=\apply{g}{x}$.
            \begin{subproof}
                Assume $x\in\dom{f}\inter\dom{g}$.
                Show that
                    $\restrl{f}{\downward{R}{x}}=
                        \restrl{g}{\downward{R}{x}}$.
                \begin{subproof}
                    Show that $\forall y:\Set.
                        (y\in\downward{R}{x}\land
                         y\in\dom{f}\union\dom{g})
                        \implies
                        (y\in\dom{f}\inter\dom{g}\land
                         \apply{f}{y}=\apply{g}{y})$.
                    \begin{subproof}
                        Fix $y$.
                        Assume
                            $y\in\downward{R}{x}\land
                             y\in\dom{f}\union\dom{g}$.
                        $y\mathrel{R}x$ by \obv{downward_closure_iff}.
                        $y\in A$ by \obv{subseteq,times_tuple_elim}.
                        $y\in\dom{f}\inter\dom{g}$ by
                            \obv{recursive_history_predecessor_common}.
                        $\Apply{P}{y}$.
                    \end{subproof}
                    Follows by
                        \obv{restrictions_agree,recursive_set_function}.
                \end{subproof}
                $f(x)=\Apply{H}
                    {x,\restrl{f}{\downward{R}{x}}}$ by
                    \obv{unbounded_recursive_partial_equation,inter}.
                $g(x)=\Apply{H}
                    {x,\restrl{g}{\downward{R}{x}}}$ by
                    \obv{unbounded_recursive_partial_equation,inter}.
                \begin{align*}
                    f(x)
                        &= \Apply{H}
                            {x,\restrl{f}{\downward{R}{x}}}
                        \\
                        &= \Apply{H}
                            {x,\restrl{g}{\downward{R}{x}}}
                        \\
                        &= g(x)
                \end{align*}
            \end{subproof}
        \end{subproof}
        Follows by \auto{well_founded_induction}.
    \end{subproof}
\end{proof}

\begin{corollary}\label{unbounded_recursive_partials_agree}
    Suppose $R$ is a binary relation on $A$ and $R$ is well-founded on $A$.
    Then $\forall f,g:\Set.\ \forall H:\Set\to\Set\to\Set.
        (\Apply{\unboundedRecursivePartialOp}
            {A,R,H,f}\land
         \Apply{\unboundedRecursivePartialOp}
            {A,R,H,g})
        \implies
        (\forall x:\Set.\ x\in\dom{f}\inter\dom{g}\implies
            \apply{f}{x}=\apply{g}{x})$.
\end{corollary}
\begin{proof}
    Fix $f,g,H$.
    Assume
        $\Apply{\unboundedRecursivePartialOp}
            {A,R,H,f}\land
         \Apply{\unboundedRecursivePartialOp}
            {A,R,H,g}$.
    Fix $x$.
    Assume $x\in\dom{f}\inter\dom{g}$.
    $x\in A$ by
        \auto{inter,unbounded_recursive_partial_domain_subseteq,subseteq}.
    Follows by \obv{unbounded_recursive_partials_agree_on_carrier}.
\end{proof}

\subsection{The canonical covered-point graph}

\begin{definition}\label{unbounded_recursive_covered_op}
    $\unboundedRecursiveCoveredOp:
        \Set\to\Set\to(\Set\to\Set\to\Set)\to\Set
        =\fun{A,R:\Set}{
                \fun{H:\Set\to\Set\to\Set}{
                    \{x\in A\mid
                        \exists f:\Set.
                            \Apply{\unboundedRecursivePartialOp}
                                {A,R,H,f}\land
                            x\in\dom{f}\}}}$.
\end{definition}

\begin{proposition}\label{unbounded_recursive_covered_iff}
    $\forall A,R:\Set.\ \forall H:\Set\to\Set\to\Set.
     \forall x:\Set.
        x\in\Apply{\unboundedRecursiveCoveredOp}
            {A,R,H}
        \iff
        (x\in A\land
         (\exists f:\Set.
            \Apply{\unboundedRecursivePartialOp}
                {A,R,H,f}\land
            x\in\dom{f}))$.
\end{proposition}
\begin{proof}
    Follows by \auto{unbounded_recursive_covered_op}.
\end{proof}

\begin{definition}\label{unbounded_recursive_value_candidate_op}
    $\unboundedRecursiveValueCandidateOp:
        \Set\to\Set\to(\Set\to\Set\to\Set)\to
        \Set\to\Prop
        =\fun{A,R:\Set}{
                \fun{H:\Set\to\Set\to\Set}{
                    \fun{p:\Set}{
                        \exists f:\Set.
                            \Apply{\unboundedRecursivePartialOp}
                                {A,R,H,f}\land
                            \fst{p}\in\dom{f}\land
                            \snd{p}=\apply{f}{\fst{p}}}}}$.
\end{definition}

\begin{definition}\label{unbounded_recursive_value_op}
    $\unboundedRecursiveValueOp:
        \Set\to\Set\to(\Set\to\Set\to\Set)\to\Set\to\Set
        =\fun{A,R:\Set}{
                \fun{H:\Set\to\Set\to\Set}{
                    \fun{x:\Set}{
                        \Select[\Set]{
                            \fun{y:\Set}{
                                \Apply{\unboundedRecursiveValueCandidateOp}
                                    {A,R,H,(x,y)}}}}}}$.
\end{definition}

\begin{proposition}\label{unbounded_recursive_covered_has_value_candidate}
    $\forall A,R:\Set.\ \forall H:\Set\to\Set\to\Set.
     \forall x:\Set.
        x\in\Apply{\unboundedRecursiveCoveredOp}
            {A,R,H}
        \implies
        (\exists y:\Set.
            \Apply{\unboundedRecursiveValueCandidateOp}
                {A,R,H,(x,y)})$.
\end{proposition}
\begin{proof}
    Fix $A,R$, $H$, and $x$.
    Assume
        $x\in\Apply{\unboundedRecursiveCoveredOp}
            {A,R,H}$.
    Take $f$ such that
        $\Apply{\unboundedRecursivePartialOp}
            {A,R,H,f}\land x\in\dom{f}$ by
        \auto{unbounded_recursive_covered_op}.
    Let $y=\apply{f}{x}$.
    $\Apply{\unboundedRecursiveValueCandidateOp}
        {A,R,H,(x,y)}$ by
        \auto{unbounded_recursive_value_candidate_op,fst_eq,snd_eq}.
\end{proof}

\begin{proposition}\label{unbounded_recursive_selected_value_candidate}
    $\forall A,R:\Set.\ \forall H:\Set\to\Set\to\Set.
     \forall x:\Set.
        (\exists y:\Set.
            \Apply{\unboundedRecursiveValueCandidateOp}
                {A,R,H,(x,y)})
        \implies
        \Apply{\unboundedRecursiveValueCandidateOp}
            {A,R,H,(x,
                \Apply{\unboundedRecursiveValueOp}
                    {A,R,H,x})}$.
\end{proposition}
\begin{proof}
    Fix $A,R,H,x$.
    Assume $\exists y:\Set.
        \Apply{\unboundedRecursiveValueCandidateOp}
            {A,R,H,(x,y)}$.
    Follows by \auto{unbounded_recursive_value_op}.
\end{proof}

\begin{theorem}\label{unbounded_recursive_value_candidate_unique}
    Suppose $R$ is a binary relation on $A$ and $R$ is well-founded on $A$.
    Then $\forall H:\Set\to\Set\to\Set.\ \forall x,y,z:\Set.
        (\Apply{\unboundedRecursiveValueCandidateOp}
            {A,R,H,(x,y)}\land
         \Apply{\unboundedRecursiveValueCandidateOp}
            {A,R,H,(x,z)})
        \implies y=z$.
\end{theorem}
\begin{proof}
    Fix $H$ and $x,y,z$.
    Assume
        $\Apply{\unboundedRecursiveValueCandidateOp}
            {A,R,H,(x,y)}\land
         \Apply{\unboundedRecursiveValueCandidateOp}
            {A,R,H,(x,z)}$.
    Take $f$ such that
        $\Apply{\unboundedRecursivePartialOp}
            {A,R,H,f}\land x\in\dom{f}\land y=\apply{f}{x}$ by
        \auto{unbounded_recursive_value_candidate_op,fst_eq,snd_eq}.
    Take $g$ such that
        $\Apply{\unboundedRecursivePartialOp}
            {A,R,H,g}\land x\in\dom{g}\land z=\apply{g}{x}$ by
        \auto{unbounded_recursive_value_candidate_op,fst_eq,snd_eq}.
    $\Apply{\unboundedRecursivePartialOp}
        {A,R,H,f}$.
    $\Apply{\unboundedRecursivePartialOp}
        {A,R,H,g}$.
    $x\in\dom{f}$.
    $x\in\dom{g}$.
    $y=\apply{f}{x}$.
    $z=\apply{g}{x}$.
    $x\in A$ by
        \auto{unbounded_recursive_partial_domain_subseteq,subseteq}.
    $x\in\dom{f}\inter\dom{g}$ by \obv{inter_intro}.
    Hence $\apply{f}{x}=\apply{g}{x}$ by
        \obv{unbounded_recursive_partials_agree_on_carrier}.
\end{proof}

\begin{corollary}\label{unbounded_recursive_selected_value_unique}
    Suppose $R$ is a binary relation on $A$ and $R$ is well-founded on $A$.
    Then $\forall H:\Set\to\Set\to\Set.\ \forall x,y:\Set.
        (x\in\Apply{\unboundedRecursiveCoveredOp}
            {A,R,H}\land
         \Apply{\unboundedRecursiveValueCandidateOp}
            {A,R,H,(x,y)})
        \implies
        \Apply{\unboundedRecursiveValueOp}
            {A,R,H,x}=y$.
\end{corollary}
\begin{proof}
    Fix $H$ and $x,y$.
    Assume
        $x\in\Apply{\unboundedRecursiveCoveredOp}
            {A,R,H}\land
         \Apply{\unboundedRecursiveValueCandidateOp}
            {A,R,H,(x,y)}$.
    $\exists z:\Set.
        \Apply{\unboundedRecursiveValueCandidateOp}
            {A,R,H,(x,z)}$ by
        \obv{unbounded_recursive_covered_has_value_candidate}.
    $\Apply{\unboundedRecursiveValueCandidateOp}
        {A,R,H,(x,
            \Apply{\unboundedRecursiveValueOp}
                {A,R,H,x})}$ by
        \obv{unbounded_recursive_selected_value_candidate}.
    Follows by \obv{unbounded_recursive_value_candidate_unique}.
\end{proof}

\begin{definition}\label{unbounded_recursive_glue_op}
    $\unboundedRecursiveGlueOp:
        \Set\to\Set\to(\Set\to\Set\to\Set)\to\Set
        =\fun{A,R:\Set}{
                \fun{H:\Set\to\Set\to\Set}{
                    \Apply{\graphOnDomainOp}
                        {A,\fun{x:\Set}{

                                {A,R,H,x}}}}}$.
\end{definition}

\begin{proposition}\label{unbounded_recursive_glue_domain}
    $\forall A,R:\Set.\ \forall H:\Set\to\Set\to\Set.
        \dom{\Apply{\unboundedRecursiveGlueOp}{A,R,H}}
        =A$.
\end{proposition}
\begin{proof}
    Follows by
        \auto{unbounded_recursive_glue_op,logical_graph_on_domain_dom}.
\end{proof}

\begin{definition}\label{unbounded_recursive_covered_graph_op}
    $\unboundedRecursiveCoveredGraphOp:
        \Set\to\Set\to(\Set\to\Set\to\Set)\to\Set
        =\fun{A,R:\Set}{
                \fun{H:\Set\to\Set\to\Set}{
                    \restrl{
                        \Apply{\unboundedRecursiveGlueOp}
                            {A,R,H}}
                        {\Apply{\unboundedRecursiveCoveredOp}
                            {A,R,H}}}}$.
\end{definition}

\subsection{Extension without a codomain bound}

\begin{proposition}\label{extend_set_function_at_new_point}
    Let $f$ be a function.
    Suppose $x\notin\dom{f}$.
    Then $\cons{(x,b)}{f}$ is a function and
        $\dom{\cons{(x,b)}{f}}=\cons{x}{\dom{f}}$.
\end{proposition}
\begin{proof}
    Let $B=\cons{b}{\ran{f}}$.
    Show that $f$ is a function to $B$.
    \begin{subproof}
        For every $u\in\dom{f}$ we have $f(u)\in\ran{f}$
            by \auto{fun_ran_iff}.
        Hence for every $u\in\dom{f}$ we have $f(u)\in B$
            by \auto{cons_iff}.
    \end{subproof}
    Hence $f\in\funs{\dom{f}}{B}$ by \obv{funs_intro}.
    $b\in B$ by \auto{cons_iff}.
    Hence $\cons{(x,b)}{f}\in\funs{\cons{x}{\dom{f}}}{B}$
        by \obv{extend_function_at_new_point}.
    Follows by \obv{funs_is_function,funs}.
\end{proof}

\begin{proposition}\label{extend_set_function_at_new_point_new_value}
    Let $f$ be a function.
    Suppose $x\notin\dom{f}$.
    Then $\apply{\cons{(x,b)}{f}}{x}=b$.
\end{proposition}
\begin{proof}
    $\cons{(x,b)}{f}$ is a function by
        \auto{extend_set_function_at_new_point}.
    $(x,b)\in\cons{(x,b)}{f}$ by \auto{cons_iff}.
    Follows by \obv{function_apply_intro}.
\end{proof}

\begin{proposition}\label{extend_set_function_at_new_point_old_value}
    Let $f$ be a function.
    Suppose $x\notin\dom{f}$.
    Let $y\in\dom{f}$.
    Then $\apply{\cons{(x,b)}{f}}{y}=f(y)$.
\end{proposition}
\begin{proof}
    $\cons{(x,b)}{f}$ is a function by
        \auto{extend_set_function_at_new_point}.
    $(y,f(y))\in f$ by \obv{function_apply_elim}.
    Hence $(y,f(y))\in\cons{(x,b)}{f}$ by \auto{cons_iff}.

        (\Apply{\unboundedRecursiveSolutionOp}
            {A,R,H,f}\land
         \Apply{\unboundedRecursiveSolutionOp}
            {A,R,H,g})
        \implies f=g$.
\end{theorem}
\begin{proof}
    Fix $H$ and $f,g$.
    Assume
        $\Apply{\unboundedRecursiveSolutionOp}
            {A,R,H,f}\land
         \Apply{\unboundedRecursiveSolutionOp}
            {A,R,H,g}$.
    $\Apply{\unboundedRecursivePartialOp}
        {A,R,H,f}\land
     \Apply{\unboundedRecursivePartialOp}
        {A,R,H,g}$ by \obv{unbounded_recursive_solution_iff}.
    $\dom{f}=A$ and $\dom{g}=A$ by
        \obv{unbounded_recursive_solution_iff}.
    $\dom{f}=\dom{g}$.
    $\dom{f}\subseteq\dom{g}$ and $\dom{g}\subseteq\dom{f}$ by
        \obv{subseteq}.
    $f$ and $g$ are functions by
        \auto{unbounded_recursive_partial_function,recursive_set_function}.
    Show that $f\subseteq g$.
    \begin{subproof}
        Show that for every $x\in\dom{f}$ we have $f(x)=g(x)$.
        \begin{subproof}
            Fix $x\in\dom{f}$.
            $x\in\dom{g}$.
            Hence $x\in\dom{f}\inter\dom{g}$ by \obv{inter_intro}.
            Follows by \obv{unbounded_recursive_partials_agree}.
        \end{subproof}
        Follows by \obv{fun_subseteq}.
    \end{subproof}
    Show that $g\subseteq f$.
    \begin{subproof}
        Show that for every $x\in\dom{g}$ we have $g(x)=f(x)$.
        \begin{subproof}
            Fix $x\in\dom{g}$.
            $x\in\dom{f}$.
            Hence $x\in\dom{f}\inter\dom{g}$ by \obv{inter_intro}.
            $f(x)=g(x)$ by \obv{unbounded_recursive_partials_agree}.
        \end{subproof}
        Follows by \obv{fun_subseteq}.
    \end{subproof}
    Follows by \obv{subseteq_antisymmetric}.
\end{proof}

\subsection{The named recursor}

\begin{definition}\label{unbounded_well_founded_recursion_op}
    $\unboundedWellFoundedRecursionOp:
        \Set\to\Set\to(\Set\to\Set\to\Set)\to\Set
        =\fun{A,R:\Set}{
                \fun{H:\Set\to\Set\to\Set}{
                    \Select[\Set]{\fun{f:\Set}{
                        \Apply{\unboundedRecursiveSolutionOp}
                            {A,R,H,f}}}}}$.
\end{definition}

The term above is total in the logical framework.  Its specification is
available under the well-foundedness hypotheses below; no off-guard value is
claimed.  Naming the unique solution uses typed logical selection only after
existence and uniqueness have been proved.

\begin{theorem}\label{unbounded_well_founded_recursion_solution}
    Suppose $R$ is a binary relation on $A$ and $R$ is well-founded on $A$.
    Then $\forall H:\Set\to\Set\to\Set.
        \Apply{\unboundedRecursiveSolutionOp}
            {A,R,H,\Apply{\unboundedWellFoundedRecursionOp}
                {A,R,H}}$.
\end{theorem}
\begin{proof}
    Fix $H$.
    $\exists f:\Set.
        \Apply{\unboundedRecursiveSolutionOp}
            {A,R,H,f}$ by \obv{unbounded_well_founded_recursion_exists}.
    Follows by \auto{unbounded_well_founded_recursion_op}.
\end{proof}

\begin{theorem}\label{unbounded_well_founded_recursion_function}
    Suppose $R$ is a binary relation on $A$ and $R$ is well-founded on $A$.
    Then $\forall H:\Set\to\Set\to\Set.
        \Apply{\recursiveSetFunctionOp}{
            \Apply{\unboundedWellFoundedRecursionOp}
            {A,R,H}}$.
\end{theorem}
\begin{proof}
    Fix $H$.
    Let $f=\Apply{\unboundedWellFoundedRecursionOp}
        {A,R,H}$.
    $\Apply{\unboundedRecursiveSolutionOp}
        {A,R,H,f}$ by \obv{unbounded_well_founded_recursion_solution}.
    Hence $\Apply{\unboundedRecursivePartialOp}
        {A,R,H,f}$ by \obv{unbounded_recursive_solution_iff}.
    Hence $\recursiveSetFunction{f}$ by
        \obv{unbounded_recursive_partial_function}.
    Follows by \obv{recursive_set_function_op}.
\end{proof}

\begin{theorem}\label{unbounded_well_founded_recursion_domain}
    Suppose $R$ is a binary relation on $A$ and $R$ is well-founded on $A$.
    Then $\forall H:\Set\to\Set\to\Set.
        \dom{\Apply{\unboundedWellFoundedRecursionOp}
            {A,R,H}}=A$.
\end{theorem}
\begin{proof}
    Fix $H$.
    Follows by \auto{unbounded_well_founded_recursion_solution,unbounded_recursive_solution_iff}.
\end{proof}

\begin{theorem}\label{unbounded_well_founded_recursion_equation}
    Suppose $R$ is a binary relation on $A$ and $R$ is well-founded on $A$.
    Then $\forall H:\Set\to\Set\to\Set.\ \forall x:\Set.\ x\in A
        \implies
        \apply{
            \Apply{\unboundedWellFoundedRecursionOp}
                {A,R,H}}
            {x}
        =\Apply{H}{x,
            \restrl{
                \Apply{\unboundedWellFoundedRecursionOp}
                    {A,R,H}}
                {\downward{R}{x}}}$.
\end{theorem}
\begin{proof}
    Fix $H$ and $x\in A$.
    Let $f=\Apply{\unboundedWellFoundedRecursionOp}
        {A,R,H}$.
    $\Apply{\unboundedRecursiveSolutionOp}
        {A,R,H,f}$ by \obv{unbounded_well_founded_recursion_solution}.
    Hence $\Apply{\unboundedRecursivePartialOp}
        {A,R,H,f}$ by \obv{unbounded_recursive_solution_iff}.
    $\dom{f}=A$ by \obv{unbounded_recursive_solution_iff}.
    Hence $x\in\dom{f}$.
    Follows by \auto{unbounded_recursive_partial_equation}.
\end{proof}

\begin{theorem}\label{unbounded_well_founded_recursion_characterization}
    Suppose $R$ is a binary relation on $A$ and $R$ is well-founded on $A$.
    Then $\forall H:\Set\to\Set\to\Set.\ \forall f:\Set.
        (\Apply{\recursiveSetFunctionOp}{f}\land\dom{f}=A\land
         (\forall x:\Set.\ x\in A\implies
            \apply{f}{x}=\Apply{H}{x,
                \restrl{f}{\downward{R}{x}}}))
        \implies
        f=\Apply{\unboundedWellFoundedRecursionOp}
            {A,R,H}$.
\end{theorem}
\begin{proof}
    Fix $H$ and $f$.
    Assume $\Apply{\recursiveSetFunctionOp}{f}\land\dom{f}=A\land
        (\forall x:\Set.\ x\in A\implies
            \apply{f}{x}=\Apply{H}{x,
                \restrl{f}{\downward{R}{x}}})$.
    $\Apply{\recursiveSetFunctionOp}{f}$.
    Hence $\recursiveSetFunction{f}$ by \obv{recursive_set_function_op}.
    $\dom{f}=A$.
    $\forall x:\Set.\ x\in A\implies
        \apply{f}{x}=\Apply{H}{x,
            \restrl{f}{\downward{R}{x}}}$.
    $\dom{f}$ is an initial segment of $A$ with respect to $R$ by
        \obv{carrier_initial_segment}.
    Hence $\recursiveInitialSegment{f}{A}{R}$ by
        \obv{recursive_initial_segment}.
    Hence $\Apply{\unboundedRecursivePartialOp}
        {A,R,H,f}$ by \obv{unbounded_recursive_partial_op,recursive_set_function_op,recursive_initial_segment_op}.
    Hence $\Apply{\unboundedRecursiveSolutionOp}
        {A,R,H,f}$ by \obv{unbounded_recursive_solution_op}.
    Let $g=\Apply{\unboundedWellFoundedRecursionOp}
        {A,R,H}$.
    $\Apply{\unboundedRecursiveSolutionOp}
        {A,R,H,g}$ by \obv{unbounded_well_founded_recursion_solution}.
    Follows by \auto{unbounded_well_founded_recursion_unique}.
\end{proof}
